import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def improved_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=20, n_cand_end=100, max_no_improve=500,
                n_active_shifts=5, reheat_interval=100, reheat_scale=0.05):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0
    best_c5_at_start = best_c5

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            if sweeps_since_improve > reheat_interval * 3:
                break
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(8, n // 30)
            top_k_idx = np.argsort(corr)[-n_active_shifts:]
            worst_shifts = top_k_idx - mid
            pert_idx = np.random.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 0.001:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                consecutive_no_improve = 0
                continue
            else:
                h = best_h.copy()
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                consecutive_no_improve = 0
                continue

        c_max = float(np.max(corr))
        threshold = c_max - 0.02 * max(c_max, 0.01)
        active_shifts = np.where(corr >= threshold)[0]
        if len(active_shifts) > n_active_shifts:
            active_shifts = np.argsort(corr)[-n_active_shifts:]
        n_act = len(active_shifts)

        corr_top_sum = np.sum(corr[active_shifts])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_act) / n_act
        else:
            weights = corr[active_shifts] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(active_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k])
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_max_abs = np.max(np.abs(g))
                if g_max_abs < 1e-15:
                    continue
                if g[k_star_idx] > 0.4 * g_max_abs:
                    continue

                a_d = max(delta_lo, -1e-3)
                b_d = min(delta_hi, 1e-3)
                if b_d <= a_d:
                    a_d = delta_lo
                    b_d = delta_hi

                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    if abs(mid_d) < 1e-14:
                        break
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            # No improving single-pair move found; try a small random move
            if sweeps_since_improve % 20 == 19:
                rng = np.random.RandomState(sweep * 31 + 11)
                i_rand = int(rng.randint(len(high_set)))
                j_rand = int(rng.randint(len(low_set)))
                i = int(high_set[i_rand])
                j = int(low_set[j_rand])
                if i != j:
                    delta_small = 0.01 * min(h[i], 1.0 - h[j])
                    if delta_small > 1e-8:
                        h[i] -= delta_small
                        h[j] += delta_small
                        h = project_feasible(h, target)
                        update_change_row(change, i, h, n, shifts)
                        update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=3, max_sweeps=20000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n-1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.conf = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = int(k_star_idx) - mid

        # Find the worst pair for the bottleneck shift
        if k_star >= 0:
            valid_i = np.arange(n - k_star)
        else:
            valid_i = np.arange(-k_star, n)

        if len(valid_i) == 0:
            continue

        products = h[valid_i] * (1.0 - h[valid_i + k_star]) if k_star >= 0 else h[valid_i] * (1.0 - h[valid_i + k_star])
        # Find the index with maximum product (worst contributor)
        if len(products) == 0:
            continue
        worst_local = int(np.argmax(products))
        i_worst = int(valid_i[worst_local])
        j_worst = int(i_worst + k_star)

        # Try to decrease h[i_worst] or h[j_worst]
        # and increase some other point
        grad_approx = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad_approx[v1] += (1.0 - h[idxs[v1] - k_star])
        grad_approx[v2] -= h[idxs[v2] + k_star])
        grad_approx -= np.mean(grad_approx)

        n_cand = max(20, n // 20)
        sorted_idx = np.argsort(grad_approx)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_c5_trial = c5_val
        best_move = None

        # Try moving from i_worst to best low candidate
        for jj in range(min(10, len(low_set))):
            j = int(low_set[jj])
            if j == i_worst:
                continue
            d_max = min(h[i_worst], 1.0 - h[j])
            if d_max < 1e-10:
                continue
            for frac in [0.5, 1.0, 0.25, 0.75]:
                d = frac * d_max
                h_trial = h.copy()
                h_trial[i_worst] -= d
                h_trial[j] += d
                corr_t = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                ct = float(np.max(corr_t)) * dx
                if ct < best_c5_trial:
                    best_c5_trial = ct
                    best_move = (i_worst, j, d)

        if best_move is not None:
            i_m, j_m, d_m = best_move
            h[i_m] -= d_m
            h[j_m] += d_m
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_reconstruct(h_src, n_dst, target_dst, n_modes=None):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    if n_modes is None:
        n_modes = n_src // 2
    # Compute DFT of h_src
    H = np.fft.rfft(h_src)
    # Reconstruct at n_dst using only first n_modes
    H_full = np.zeros(n_dst // 2 + 1, dtype=complex)
    n_copy = min(n_modes, len(H), len(H_full))
    H_full[:n_copy] = H[:n_copy] * (n_dst / n_src)
    h_out = np.fft.irfft(H_full, n=n_dst)
    return project_feasible(h_out, target_dst)


def targeted_reheat(h, n, target, corr, scale=0.04):
    mid = n - 1
    top_k = np.argsort(corr)[-8:]
    worst_shifts = top_k - mid
    # Find points involved in worst overlaps
    involved = set()
    for s in worst_shifts:
        k = int(s)
        if k >= 0:
            for i in range(n - k):
                if h[i] > 0.3 and h[i + k] < 0.7:
                    involved.add(i)
                    involved.add(i + k)
                elif h[i] < 0.7 and h[i + k] > 0.3:
                    involved.add(i)
                    involved.add(i + k)
        else:
            for i in range(-k, n):
                if h[i] > 0.3 and h[i + k] < 0.7:
                    involved.add(i)
                    involved.add(i + k)
                elif h[i] < 0.7 and h[i + k] > 0.3:
                    involved.add(i)
                    involved.add(i + k)
    if len(involved) < 5:
        involved = set(np.random.choice(n, size=10, replace=False).tolist())
    involved_list = list(involved)[:50]
    if len(involved_list) == 0:
        return h
    h_new = h.copy()
    h_new[involved_list] += np.random.randn(len(involved_list)) * scale
    return project_feasible(h_new, target)

        if tleft() > 60:

            h_pol1, sc_pol1 = slsqp_round(best_h, current_n, current_target,
                                          [5000, 20000, 100000],
                                          1200, 1e-15, deadline=deadline)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    # Phase 5: More CD at current resolution
    if tleft() > 200:
        h_cd2, c5_cd2 = improved_cd(best_h, current_n, current_target,
                                    time_limit=min(120, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=30, n_cand_end=120,
                                    max_no_improve=600,
                                    n_active_shifts=8,
                                    reheat_interval=80, reheat_scale=0.03)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Try higher resolution n=1600
    if tleft() > 500:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = fourier_reconstruct(best_h, n2, target2, n_modes=current_n // 2)
        sc2 = compute_c5(h2)
        if sc2 > best_score + 0.005:
            h2 = interpolate_to(best_h, n2, target2)
            sc2 = compute_c5(h2)

        if tleft() > 100:
            deadline = time.time() + min(60, tleft() * 0.3)
            h2_pol, sc2_pol = slsqp_round(h2, n2, target2,
                                          [1000, 5000, 20000],
                                          800, 1e-15, deadline=deadline)
            if sc2_pol < sc2:
                h2 = h2_pol
                sc2 = sc2_pol

        if tleft() > 200:
            h_cd3, c5_cd3 = improved_cd(h2, n2, target2,
                                        time_limit=min(180, tleft() * 0.55),
                                        max_sweeps=15000,
                                        n_cand_start=30, n_cand_end=100,
                                        max_no_improve=400,
                                        n_active_shifts=8,
                                        reheat_interval=100, reheat_scale=0.03)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                current_n = n2
                current_target = target2
        if tleft() > 50:
            deadline = time.time() + min(40, tleft() * 0.4)
            h_pol2, sc_pol2 = slsqp_round(best_h, current_n, current_target,
                                          [10000, 50000, 200000],
                                          1000, 1e-15, deadline=deadline)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

    # Phase 7: Multi-start with targeted perturbations
    if tleft() > 120:
        for attempt in range(3):
            if tleft() < 50:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            noise_scales = [0.03, 0.06, 0.1]
            noise = noise_scales[attempt % 3]
            h_pert = targeted_reheat(best_h, current_n, current_target, corr, scale=noise)
            deadline = time.time() + min(30, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 2000, 10000, 50000],
                                        600, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            # Follow up with CD if improved
            if sc_ref < best_score + 0.001 and tleft() > 30:
                h_cd_f, c5_f = improved_cd(h_ref, current_n, current_target,
                                           time_limit=min(20, tleft() * 0.4),
                                           max_sweeps=5000,
                                           n_cand_start=30, n_cand_end=100,
                                           max_no_improve=200,
                                           n_active_shifts=6,
                                           reheat_interval=60, reheat_scale=0.02)
                if c5_f < best_score:
                    best_score = c5_f
                    best_h = h_cd_f

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        deadline = time.time() + min(20, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [10000, 50000, 200000, 500000],
                                        1500, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final feasibility check
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
