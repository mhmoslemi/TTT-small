```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_grad(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    
    grad = np.zeros(n)
    mid = n - 1
    active = weights > 1e-10
    if np.any(active):
        active_idx = np.where(active)[0]
        for m in active_idx:
            k = int(m) - mid
            w = float(weights[m])
            lo1 = max(0, k)
            hi1 = min(n, k + n)
            if hi1 > lo1:
                grad[lo1:hi1] += w * (1.0 - x[lo1-k:hi1-k])
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if hi2 > lo2:
                grad[lo2:hi2] -= w * x[lo2+k:hi2+k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def run_lse(h_start, target, alphas, maxiter=200, penalty_start=50.0, time_limit=60.0):
    start = time.time()
    n = len(h_start)
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for i, alpha in enumerate(alphas):
        if time.time() - start > time_limit:
            break
        pen = penalty_start * (10 ** i)
        try:
            res = minimize(lse_obj, h_cur, args=(alpha, target, pen),
                          jac=lse_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def pair_cd(h_init, target, time_limit, n_cand=40, max_sweeps=100000, n_shifts=2):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Gradient from top shifts
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            lo1 = max(0, k_star)
            hi1 = min(n, k_star + n)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1-k_star:hi1-k_star])
            lo2 = max(0, -k_star)
            hi2 = min(n, n - k_star)
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2+k_star:hi2+k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 30.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                lo1 = max(0, k)
                hi1 = min(n, k + n)
                if hi1 > lo1:
                    grad[lo1:hi1] += float(wi) * (1.0 - h[lo1-k:hi1-k])
                lo2 = max(0, -k)
                hi2 = min(n, n - k)
                if hi2 > lo2:
                    grad[lo2:hi2] -= float(wi) * h[lo2+k:hi2+k]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        cand_pos = {int(all_cands[p]): p for p in range(nc)}
        
        # Profiles
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n-1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

    n = len(x)
    dx = 2.0 / n

                        vals = corr + mid_d * g
                        f_mid = float(np.max(vals))
                        active = vals >= (f_mid - 1e-13)
                        if not np.any(active):
                            break
                        f_prime = float(np.max(g[active]))
                        if f_prime > 1e-13:
                            hi_d = mid_d
                        else:
                            lo = mid_d
                    delta_star = 0.5 * (lo + hi_d)
                    if delta_star < 1e-10:
                        continue
                    new_max = float(np.max(corr + delta_star * g))
                    improve = cur_max - new_max
                    if improve > best_improve:
                        best_improve = improve
                        best_i = int(i)
                        best_j = int(j)
                        best_delta = delta_star

            if best_i >= 0 and best_improve > 1e-14:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_max = float(np.max(corr))
                found_in_batch = True
            else:
                break

        if not found_in_batch:
            no_improve += 5
            if no_improve > 500:
                break

    return best_h, best_c5

def interp_to(src, n_dst, target_dst):
    n_src = len(src)
    if n_src == n_dst:
        return project_feasible(src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, src)
    return project_feasible(h_out, target_dst)

def structured_random_init(n, target, rng):
    r = rng.uniform()
    if r < 0.3:
        h = np.zeros(n)
        ones = rng.choice(n, target // 2, replace=False)
        h[ones] = 1.0
        h += rng.randn(n) * 0.03
    elif r < 0.6:
        n_blocks = rng.randint(5, 20)
        block_size = max(2, n // n_blocks)
        h = np.zeros(n)
        for start in range(0, n, block_size * 2):
            end = min(start + block_size, n)
            h[start:end] = 1.0
        h = np.convolve(h, np.ones(3) / 3.0, mode='same')
    else:
        freq = rng.randint(2, 8)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = project_feasible(h0.copy(), target0)
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Quick LSE at original resolution
    if tleft() > 60:
        hp, sc = run_lse(best_h, target0, alphas=[100, 500, 2000],
                         maxiter=200, penalty_start=50.0,
                         time_limit=min(45, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Low-resolution global search
    for n_low in [120, 200]:
        if tleft() < 150:
            break
        target_low = n_low / 2.0
        h_low_init = interp_to(h0, n_low, target_low)
        best_low = h_low_init.copy()
        best_low_score = compute_c5(h_low_init)

        n_attempts = 5
        for attempt in range(n_attempts):
            if tleft() < 80:
                break
            if attempt == 0:
                h_try = h_low_init.copy()
            else:
                h_try = structured_random_init(n_low, target_low, rng)
            hp_low, sc_low = run_lse(h_try, target_low,
                                     alphas=[50, 500, 5000],
                                     maxiter=150, penalty_start=20.0,
                                     time_limit=min(25, tleft() * 0.15))
            if sc_low < best_low_score:
                best_low_score = sc_low
                best_low = hp_low

        h_full = interp_to(best_low, n0, target0)
        hp_full, sc_full = run_lse(h_full, target0, alphas=[500, 5000],
                                   maxiter=150, penalty_start=100.0,
                                   time_limit=min(30, tleft() * 0.2))
        if sc_full < best_score:
            best_score = sc_full
            best_h = hp_full

    # Phase 3: Pair CD multi-pair at full resolution
    if tleft() > 120:
        cd_time = min(250, tleft() * 0.35)
        hcd, c5cd = pair_cd(best_h, target0, time_limit=cd_time,
                            n_cand=40, n_pairs=3, n_shifts=2)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 4: LSE high alpha
    if tleft() > 60:
        hp, sc = run_lse(best_h, target0, alphas=[20000, 100000, 500000],
                         maxiter=200, penalty_start=500.0,
                         time_limit=min(50, tleft() * 0.35))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: More pair CD with single shift
    if tleft() > 120:
        hcd, c5cd = pair_cd(best_h, target0,
                            time_limit=min(150, tleft() * 0.4),
                            n_cand=50, n_pairs=3, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: Higher resolution attempt
    if tleft() > 250:
        n_hires = 1200
        target_hires = n_hires / 2.0
        h_hires = interp_to(best_h, n_hires, target_hires)
        hp, sc = run_lse(h_hires, target_hires,
                         alphas=[50, 500, 5000, 50000],
                         maxiter=200, penalty_start=100.0,
                         time_limit=min(80, tleft() * 0.3))
        if sc < best_score:
            best_score = sc
            best_h = hp
        if tleft() > 100:
            hcd, c5cd = pair_cd(best_h, target_hires,
                                time_limit=min(80, tleft() * 0.35),
                                n_cand=40, n_pairs=2, n_shifts=2)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 7: Final LSE polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = run_lse(best_h, target_use,
                         alphas=[500000, 2000000],
                         maxiter=150, penalty_start=5000.0,
                         time_limit=min(35, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, target_use,
                            time_limit=min(25, tleft() * 0.7),
                            n_cand=30, n_pairs=3, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
```
