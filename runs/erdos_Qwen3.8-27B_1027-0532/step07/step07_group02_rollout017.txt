def binary_pair_cd(h_init, n, target_int, time_limit, seed=42,

    """Binary pair coordinate descent: only full swaps allowed."""
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)
    n_corr = 2 * n - 1
    rng = np.random.RandomState(seed)
    
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)
    
    ones_mask = (h == 1.0)
    zeros_mask = (h == 0.0)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        cur_max = float(np.max(corr))
        
        # Find bottleneck shift
        k_star_idx = int(np.argmax(corr))
        
        # Compute gradient to select candidates
        grad = np.zeros(n)
        k_star = k_star_idx - (n - 1)
        idxs = np.arange(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star])
        
        # Candidates: ones with high grad (decreasing helps), zeros with low grad (increasing helps)
        ones_indices = np.where(ones_mask)[0]
        zeros_indices = np.where(zeros_mask)[0]
        
        # Sort ones by grad descending (highest grad = most beneficial to decrease)
        ones_sorted = ones_indices[np.argsort(grad[ones_indices])[::-1]]
        zeros_sorted = zeros_indices[np.argsort(grad[zeros_indices])]
        
        # Take top candidates
        hi_set = ones_sorted[:min(n_cand, len(ones_sorted))]
        lo_set = zeros_sorted[:min(n_cand, len(zeros_sorted))]
        
        best_trial = cur_max
        best_i = -1
        best_j = -1
        
        for i in hi_set:
            ci = change[i]
            for j in lo_set:
                g = change[j] - ci
                new_max = np.max(corr + g)
                if new_max < best_trial:
                    best_trial = new_max
                    best_i = int(i)
                    best_j = int(j)
        
        if best_i >= 0:
            # Execute swap
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones_mask[best_i] = False
            ones_mask[best_j] = True
            zeros_mask[best_i] = True
            zeros_mask[best_j] = False
            corr += change[best_j] - change[best_i]
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            
            new_c5 = best_trial * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

            # Try random swaps to escape

                j = int(rng.choice(zeros_indices))
                h[i] = 0.0
                h[j] = 1.0
                ones_mask[i] = False
                ones_mask[j] = True
                zeros_mask[i] = True
                zeros_mask[j] = False
                corr += change[j] - change[i]
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
        
        if no_improve > max_no_improve:
            break
    
    return best_h, best_c5

                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_prime_check(g, vals, f_mid) > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def f_prime_check(g, vals, f_mid):
    active = vals >= f_mid - 1e-14
    if active.any():
        return float(np.max(g[active]))
    return 0.0

def make_structured_binary(n, target_int, pattern='block'):
    h = np.zeros(n)
    if pattern == 'block':
        h[:target_int] = 1.0
    elif pattern == 'spread':
        positions = np.linspace(0, n - 1, target_int).astype(int)
        h[positions] = 1.0
    elif pattern == 'interleaved':
        step = n // target_int
        for i in range(target_int):
            pos = min(i * step, n - 1)
            h[pos] = 1.0
    elif pattern == 'random':
        rng = np.random.RandomState(0)
        positions = rng.choice(n, size=target_int, replace=False)
        h[positions] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    target_int = int(n // 2)
    dx = 2.0 / n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    h_bin = to_binary(h_init, n, target_int)
    bin_score = compute_c5(h_bin)
    if bin_score < best_score:
        best_score = bin_score
        best_h = h_bin.copy()

    best_bin_h = h_bin.copy()
    best_bin_score = bin_score

    # Phase 1: Binary pair CD from thresholded parent
    if tleft() > 250:
        alloc = min(250, tleft() * 0.3)
        h_b1, sc_b1 = binary_pair_cd(h_bin, n, target_int, time_limit=alloc,
                                      seed=seed, max_sweeps=100000,
                                      n_cand=100, max_no_improve=300)
        if sc_b1 < best_bin_score:
            best_bin_score = sc_b1
            best_bin_h = h_b1
        if sc_b1 < best_score:
            best_score = sc_b1
            best_h = h_b1

    # Phase 2: Binary SA from best binary
    if tleft() > 200:
        alloc = min(180, tleft() * 0.2)
        h_b2, sc_b2 = binary_sa(best_bin_h, n, target_int, time_limit=alloc,
                                 seed=seed + 100, T_init=3.0, cooling=0.9998)
        if sc_b2 < best_bin_score:
            best_bin_score = sc_b2
            best_bin_h = h_b2
        if sc_b2 < best_score:
            best_score = sc_b2
            best_h = h_b2

    # Phase 3: Binary pair CD again with different params
    if tleft() > 200:
        alloc = min(180, tleft() * 0.2)
        h_b3, sc_b3 = binary_pair_cd(best_bin_h, n, target_int, time_limit=alloc,
                                      seed=seed + 200, max_sweeps=80000,
                                      n_cand=120, max_no_improve=400)
        if sc_b3 < best_bin_score:
            best_bin_score = sc_b3
            best_bin_h = h_b3
        if sc_b3 < best_score:
            best_score = sc_b3
            best_h = h_b3

    # Phase 4: Try structured initializations
    if tleft() > 200:
        patterns = ['spread', 'interleaved', 'block']
        for pat in patterns:
            if tleft() < 80:
                break
            h_struct = make_structured_binary(n, target_int, pattern=pat)
            alloc = min(60, tleft() * 0.3)
            h_s, sc_s = binary_pair_cd(h_struct, n, target_int, time_limit=alloc,
                                        seed=seed + 300, max_sweeps=50000,
                                        n_cand=80, max_no_improve=200)
            if sc_s < best_bin_score:
                best_bin_score = sc_s
                best_bin_h = h_s
            if sc_s < best_score:
                best_score = sc_s
                best_h = h_s

    # Phase 5: More binary SA from best
    if tleft() > 150:
        alloc = min(120, tleft() * 0.2)
        h_b4, sc_b4 = binary_sa(best_bin_h, n, target_int, time_limit=alloc,
                                 seed=seed + 400, T_init=5.0, cooling=0.9997)
        if sc_b4 < best_bin_score:
            best_bin_score = sc_b4
            best_bin_h = h_b4
        if sc_b4 < best_score:
            best_score = sc_b4
            best_h = h_b4

    # Phase 6: Final binary pair CD
    if tleft() > 150:
        alloc = min(150, tleft() * 0.3)
        h_b5, sc_b5 = binary_pair_cd(best_bin_h, n, target_int, time_limit=alloc,
                                      seed=seed + 500, max_sweeps=100000,
                                      n_cand=150, max_no_improve=500)
        if sc_b5 < best_bin_score:
            best_bin_score = sc_b5
            best_bin_h = h_b5
        if sc_b5 < best_score:
            best_score = sc_b5
            best_h = h_b5

    # Phase 7: Continuous SLSQP refinement from best binary
    if tleft() > 120:
        h_c1, sc_c1 = slsqp_optimize(best_bin_h, n, target,
                                      alphas=[100, 500, 2000], maxiter_per_alpha=1500)
        if sc_c1 < best_score:
            best_score = sc_c1
            best_h = h_c1

    if tleft() > 120:
        alloc = min(120, tleft() * 0.5)
        h_c2, sc_c2 = pair_cd_continuous(best_h, n, target, time_limit=alloc,
                                          max_sweeps=30000, n_cand_start=50,
                                          n_cand_end=130, max_no_improve=250,
                                          n_binsearch=18, stochastic_frac=0.05,
                                          seed=seed + 600)
        if sc_c2 < best_score:
            best_score = sc_c2
            best_h = h_c2

    # Phase 9: Another SLSQP pass
    if tleft() > 80:
        h_c3, sc_c3 = slsqp_optimize(best_h, n, target,
                                      alphas=[5000, 20000], maxiter_per_alpha=2000)
        if sc_c3 < best_score:
            best_score = sc_c3
            best_h = h_c3

    # Phase 10: Final continuous pair CD polish
    if tleft() > 60:
        alloc = min(60, tleft() * 0.6)
        h_c4, sc_c4 = pair_cd_continuous(best_h, n, target, time_limit=alloc,
                                          max_sweeps=20000, n_cand_start=60,
                                          n_cand_end=150, max_no_improve=200,
                                          n_binsearch=20, stochastic_frac=0.03,
                                          seed=seed + 700)
        if sc_c4 < best_score:
            best_score = sc_c4
            best_h = h_c4

    # Phase 11: Final SLSQP
    if tleft() > 40:
        h_c5, sc_c5 = slsqp_optimize(best_h, n, target,

            best_score = sc_c5
            best_h = h_c5

    final_h = project_feasible(best_h, target)
    final_score = compute_c5(final_h)
    return final_h, final_score, n
