def vec_pair_cd(h_init, n, target, time_limit, start_time, 
                max_sweeps=50000, n_cand_start=20, n_cand_end=60,
                bisection_iters=18, max_no_improve=400):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    mid = n - 1
    idxs = np.arange(n)
    
    # Precompute index arrays for change matrix
    p_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    pos = p_idx + s_idx
    neg = p_idx - s_idx
    pos_valid = (pos >= 0) & (pos < n)
    neg_valid = (neg >= 0) & (neg < n)
    pos_clipped = np.clip(pos, 0, n-1)
    neg_clipped = np.clip(neg, 0, n-1)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Compute change matrix
        change = np.where(pos_valid, 1.0 - h[pos_clipped], 0.0) \
               - np.where(neg_valid, h[neg_clipped], 0.0)
        
        # Gradient for active constraint
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Adaptive candidate set
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 5)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Build pair matrix
        n_h = len(high_set)
        n_l = len(low_set)
        # delta_corr[pair_idx, k_idx] = change[j, k] - change[i, k]
        # where pair_idx = ii * n_l + jj, i = high_set[ii], j = low_set[jj]
        
        # Reshape for vectorized computation
        # high_set shape: (n_h,), low_set shape: (n_l,)
        # We want delta_corr shape: (n_h, n_l, n_corr)
        # = change[low_set[None, :, :], None] - change[high_set[:, None, :], None]
        
        ch_high = change[high_set, :, None]  # (n_h, n_corr, 1)
        ch_low = change[low_set, None, :, :]  # Wait, this doesn't work directly
        
        # Better: 
        # ch_high: (n_h, 1, n_corr) = change[high_set][:, None, :]
        # ch_low: (1, n_l, n_corr) = change[low_set][None, :, :]
        # delta_corr: (n_h, n_l, n_corr) = ch_low - ch_high
        
        ch_high = change[high_set][:, None, :]  # (n_h, 1, n_corr)
        ch_low = change[low_set][None, :, :]     # (1, n_l, n_corr)
        delta_corr = ch_low - ch_high             # (n_h, n_l, n_corr)
        
        # Bounds for δ
        h_high = h[high_set]   # (n_h,)
        h_low = h[low_set]     # (n_l,)
        delta_max = np.minimum(h_high[:, None], 1.0 - h_low[None, :])  # (n_h, n_l)
        
        # Reshape to 2D for vectorized search
        delta_corr_2d = delta_corr.reshape(-1, n_corr)  # (n_pairs, n_corr)
        delta_max_2d = delta_max.reshape(-1)            # (n_pairs,)
        
        # Filter out invalid pairs
        valid = delta_max_2d > 1e-12
        if not valid.any():
            continue
        
        delta_corr_v = delta_corr_2d[valid]
        delta_max_v = delta_max_2d[valid]
        n_v = len(delta_max_v)
        
        # Ternary search over all valid pairs
        d_lo = np.zeros(n_v)
        d_hi = delta_max_v.copy()
        
        for _ in range(bisection_iters):
            d1 = d_lo + (d_hi - d_lo) / 3.0
            d2 = d_lo + 2.0 * (d_hi - d_lo) / 3.0
            f1 = np.max(corr[None, :] + d1[:, None] * delta_corr_v, axis=1)
            f2 = np.max(corr[None, :] + d2[:, None] * delta_corr_v, axis=1)
            mask = f1 < f2
            d_hi[mask] = d2[mask]
            d_lo[~mask] = d1[~mask]
        
        d_star = (d_lo + d_hi) / 2.0
        f_star = np.max(corr[None, :] + d_star[:, None] * delta_corr_v, axis=1)
        
        # Find best pair
        best_pair_idx = int(np.argmin(f_star))
        best_delta = d_star[best_pair_idx]
        best_f = f_star[best_pair_idx]
        
        if best_f >= c5_val / dx - 1e-14 or best_delta < 1e-12:
            continue
        
        # Map back to original pair indices

        orig_idx = int(valid_indices[best_pair_idx])
        ii = orig_idx // n_l
        jj = orig_idx % n_l

        j = int(low_set[jj])
        
        h[i] -= best_delta
        h[j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_lse, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            f2 = np.max(corr[None, :] + d2[:, None] * delta_corr_v, axis=1)
            mask = f1 < f2
            d_hi[mask] = d2[mask]
            d_lo[~mask] = d1[~mask]
        
        d_star = (d_lo + d_hi) / 2.0
        f_star = np.max(corr[None, :] + d_star[:, None] * delta_corr_v, axis=1)
        
        best_pair_reld = int(np.argmin(f_star))
        best_delta = d_star[best_pair_reld]
        best_f = f_star[best_pair_reld]
        
        if best_f >= c5_val / dx - 1e-14 or best_delta < 1e-12:
            continue
        
        # Map back
        valid_indices = np.where(valid)[0]
        orig_idx = int(valid_indices[best_pair_reld])
        ii = orig_idx // n_l
        jj = orig_idx % n_l
        
        i = int(high_set[ii])
        j = int(low_set[jj])
        
        h[i] -= best_delta
        h[j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_lse, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def vec_pair_cd(h_init, n, target, time_limit, start_time,
                max_sweeps=80000, n_cand_start=20, n_cand_end=60,
                bisection_iters=18, max_no_improve=500, top_k=1):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    p_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    pos = p_idx + s_idx
    neg = p_idx - s_idx
    pos_valid = (pos >= 0) & (pos < n)
    neg_valid = (neg >= 0) & (neg < n)
    pos_clipped = np.clip(pos, 0, n - 1)
    neg_clipped = np.clip(neg, 0, n - 1)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        change = np.where(pos_valid, 1.0 - h[pos_clipped], 0.0) \
               - np.where(neg_valid, h[neg_clipped], 0.0)

        if top_k <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idxs = np.argsort(corr)[-top_k:]
            weights = corr[top_idxs].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(top_k) / top_k
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for wi, s_idx2 in zip(weights, top_idxs):
                k2 = int(s_idx2) - mid
                v1 = (idxs - k2 >= 0) & (idxs - k2 < n)
                v2 = (idxs + k2 >= 0) & (idxs + k2 < n)
                grad[v1] += wi * (1.0 - h[idxs[v1] - k2])
                grad[v2] -= wi * h[idxs[v2] + k2]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        n_l = len(low_set)
        n_h = len(high_set)

        ch_high = change[high_set][:, None, :]
        ch_low = change[low_set][None, :, :]
        delta_corr = ch_low - ch_high

        h_high = h[high_set]
        h_low = h[low_set]
        delta_max = np.minimum(h_high[:, None], 1.0 - h_low[None, :])

        delta_corr_2d = delta_corr.reshape(-1, n_corr)
        delta_max_2d = delta_max.reshape(-1)

        valid = delta_max_2d > 1e-12
        if not valid.any():
            continue

        delta_corr_v = delta_corr_2d[valid]
        delta_max_v = delta_max_2d[valid]
        n_v = len(delta_max_v)

        if n_v > 50000:
            perm = np.random.RandomState(sweep % 1000).permutation(n_v)
            delta_corr_v = delta_corr_v[perm[:50000]]
            delta_max_v = delta_max_v[perm[:50000]]
            n_v = 50000

        d_lo = np.zeros(n_v)
        d_hi = delta_max_v.copy()

        for _ in range(bisection_iters):
            d1 = d_lo + (d_hi - d_lo) / 3.0
            d2 = d_lo + 2.0 * (d_hi - d_lo) / 3.0
            f1 = np.max(corr[None, :] + d1[:, None] * delta_corr_v, axis=1)
            f2 = np.max(corr[None, :] + d2[:, None] * delta_corr_v, axis=1)
            mask = f1 < f2
            d_hi[mask] = d2[mask]
            d_lo[~mask] = d1[~mask]

        d_star = (d_lo + d_hi) / 2.0
        f_star = np.max(corr[None, :] + d_star[:, None] * delta_corr_v, axis=1)

        best_pair_reld = int(np.argmin(f_star))
        best_delta = d_star[best_pair_reld]
        best_f = f_star[best_pair_reld]

        if best_f >= c5_val / dx - 1e-14 or best_delta < 1e-13:
            continue

        valid_indices = np.where(valid)[0]
        orig_idx = int(valid_indices[best_pair_reld])
        ii = orig_idx // n_l
        jj = orig_idx % n_l

        i = int(high_set[ii])
        j = int(low_set[jj])

        h[i] -= best_delta
        h[j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_vec_pair_cd(h_init, n, target, time_limit, start_time,
                            max_sweeps=50000, n_cand=40, n_top=3,
                            bisection_iters=16, max_no_improve=400):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    p_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    pos = p_idx + s_idx
    neg = p_idx - s_idx
    pos_valid = (pos >= 0) & (pos < n)
    neg_valid = (neg >= 0) & (neg < n)
    pos_clipped = np.clip(pos, 0, n - 1)
    neg_clipped = np.clip(neg, 0, n - 1)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        change = np.where(pos_valid, 1.0 - h[pos_clipped], 0.0) \
               - np.where(neg_valid, h[neg_clipped], 0.0)

        top_idxs = np.argsort(corr)[-n_top:]
        weights = corr[top_idxs].astype(float)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idxs):
            k2 = int(si) - mid
            v1 = (idxs - k2 >= 0) & (idxs - k2 < n)
            v2 = (idxs + k2 >= 0) & (idxs + k2 < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k2])
            grad[v2] -= wi * h[idxs[v2] + k2]
        grad -= np.mean(grad)

        n_cand_adj = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand_adj]
        high_set = sorted_idx[-n_cand_adj:]
        n_l = len(low_set)
        n_h = len(high_set)

        ch_high = change[high_set][:, None, :]
        ch_low = change[low_set][None, :, :]
        delta_corr = ch_low - ch_high

        h_high = h[high_set]
        h_low = h[low_set]
        delta_max = np.minimum(h_high[:, None], 1.0 - h_low[None, :])

        delta_corr_2d = delta_corr.reshape(-1, n_corr)
        delta_max_2d = delta_max.reshape(-1)

        valid = delta_max_2d > 1e-12
        if not valid.any():
            continue

        delta_corr_v = delta_corr_2d[valid]
        delta_max_v = delta_max_2d[valid]
        n_v = len(delta_max_v)

        d_lo = np.zeros(n_v)
        d_hi = delta_max_v.copy()

        for _ in range(bisection_iters):
            d1 = d_lo + (d_hi - d_lo) / 3.0
            d2 = d_lo + 2.0 * (d_hi - d_lo) / 3.0
            f1 = np.max(corr[None, :] + d1[:, None] * delta_corr_v, axis=1)
            f2 = np.max(corr[None, :] + d2[:, None] * delta_corr_v, axis=1)
            mask = f1 < f2
            d_hi[mask] = d2[mask]
            d_lo[~mask] = d1[~mask]

        d_star = (d_lo + d_hi) / 2.0
        f_star = np.max(corr[None, :] + d_star[:, None] * delta_corr_v, axis=1)

        best_pair_reld = int(np.argmin(f_star))
        best_delta = d_star[best_pair_reld]
        best_f = f_star[best_pair_reld]

        if best_f >= c5_val / dx - 1e-14 or best_delta < 1e-13:
            continue

        valid_indices = np.where(valid)[0]
        orig_idx = int(valid_indices[best_pair_reld])
        ii = orig_idx // n_l
        jj = orig_idx % n_l

        i = int(high_set[ii])
        j = int(low_set[jj])

        h[i] -= best_delta
        h[j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    return best_h, best_c5


def stochastic_refine(h_init, n, target, time_limit, start_time,
                      max_steps=15000, base_scale=0.03, seed=777):
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 2000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 60:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:
        h1, sc1 = slsqp_refine(best_h, n0, target0,
                                [200, 1000, 3000], 600,
                                time.time() + min(25, remaining - 5), start_time)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        t_end = time.time() + min(150, remaining - 15)
        h2, sc2 = vec_pair_cd(best_h, n0, target0, t_end, start_time,
                               max_sweeps=120000, n_cand_start=20, n_cand_end=60,
                               bisection_iters=18, max_no_improve=500, top_k=1)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:
        h3, sc3 = slsqp_refine(best_h, n0, target0,
                                [500, 2000], 800,
                                time.time() + min(25, remaining - 5), start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 150:
        n1 = 800
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        t_end = time.time() + min(250, remaining - 20)
        h4, sc4 = vec_pair_cd(h_int, n1, target1, t_end, start_time,
                               max_sweeps=120000, n_cand_start=25, n_cand_end=70,
                               bisection_iters=18, max_no_improve=500, top_k=1)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
            n0 = n1
            target0 = target1

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 50:
        h5, sc5 = slsqp_refine(best_h, n0, target0,
                                [2000, 8000], 1000,
                                time.time() + min(40, remaining - 10), start_time)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 120:
        n2 = 1200
        target2 = n2 / 2.0
        h_int2 = interpolate_to(best_h, n2, target2)
        t_end = time.time() + min(150, remaining - 20)
        h6, sc6 = vec_pair_cd(h_int2, n2, target2, t_end, start_time,
                               max_sweeps=100000, n_cand_start=20, n_cand_end=50,
                               bisection_iters=16, max_no_improve=400, top_k=1)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6
            n0 = n2
            target0 = target2

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        multi_n = n0
        t_end = time.time() + min(multi_n <= 800 and 80 or 50, remaining - 10)
        h7, sc7 = multi_shift_vec_pair_cd(best_h, multi_n, target0, t_end, start_time,
                                           max_sweeps=60000, n_cand=40, n_top=3,
                                           bisection_iters=16, max_no_improve=400)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 40:
        h8, sc8 = slsqp_refine(best_h, n0, target0,
                                [10000, 30000], 800,
                                time.time() + min(35, remaining - 5), start_time)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    remaining = budget_s - 40.0 - (time.time() - start_time)

                               max_sweeps=80000, n_cand_start=30, n_cand_end=80,

        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
