def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_smooth_max(h, alpha):
    """Compute gradient of smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = alpha * (corr - np.max(corr))
    weights = np.exp(np.clip(corr_shifted, -500, 0))
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(n_corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad * dx

                break
            
            # Nesterov extrapolation
            beta_t = (1 + np.sqrt(1 + 4 * t * t)) / 2
            h_y = h_t + (t / (t + 1 + beta_t)) * (h_t - h_prev_t)
            h_y = np.clip(h_y, 0.0, 1.0)
            
            # Compute gradient at y

            # Projected gradient step
            h_new = h_y - step * grad
            h_new = project_feasible(h_new, target)
            
            # Backtracking if needed

                obj_curr = compute_c5(h_t)

def block_cd(h_init, n, target, time_limit, max_sweeps=20000,
             block_sizes=None, seed=456):
    """Block coordinate descent - move mass between contiguous blocks."""
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    if block_sizes is None:
        block_sizes = [5, 10, 20, 40]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 200:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        
        # Find block with highest average h and block with lowest average h
        bs = block_sizes[rng.randint(len(block_sizes))]
        
        if bs >= n:
            bs = n // 4
        
        # Compute correlation contribution per position
        # For the active shift, the "pressure" at position i is:
        # contributes to corr[k_star] via h(i)*(1-h(i+k_star))
        # So positions with high h(i) and low h(i+k_star) are "bad"
        
        # Compute a "pressure" gradient
        idxs = np.arange(n)
        pressure = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        pressure[v1] += (1.0 - h[idxs[v1] - k_star])
        pressure[v2] -= h[idxs[v2] + k_star]
        
        # Divide into blocks and find high/low pressure blocks
        n_blocks = n // bs
        block_mean = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_mean[b] = np.mean(pressure[b*bs:(b+1)*bs])
        
        high_b = int(np.argmax(block_mean))
        low_b = int(np.argmin(block_mean))
        
        if high_b == low_b:
            continue
        
        # Move mass from high block to low block
        h_block = h[high_b*bs:(high_b+1)*bs].copy()
        l_block = h[low_b*bs:(low_b+1)*bs].copy()
        
        # Try to move mass: reduce h in high block, increase in low block
        # Find the max we can move
        max_move = min(np.sum(h_block), np.sum(1.0 - l_block))
        max_move = max_move * 0.5  # be conservative
        
        if max_move < 1e-10:
            continue
        
        # Binary search for optimal amount to move
        a = 0.0
        b = max_move
        best_local = c5_val
        best_frac = 0.0
        
        for _ in range(10):
            mid = 0.5 * (a + b)
            # Scale: reduce high block proportionally, increase low block proportionally
            h_trial = h.copy()
            h_trial[high_b*bs:(high_b+1)*bs] = h_block - mid / bs
            h_trial[low_b*bs:(low_b+1)*bs] = l_block + mid / bs
            h_trial = np.clip(h_trial, 0.0, 1.0)
            
            # Check if we maintained sum
            if abs(np.sum(h_trial) - target) > 1e-6:
                continue
            
            trial_c5 = float(np.max(np.correlate(h_trial, 1.0 - h_trial, mode='full'))) * dx
            if trial_c5 < best_local:
                best_local = trial_c5
                best_frac = mid
                a = mid
            else:
                b = mid
        
        if best_frac > 1e-10:
            h[high_b*bs:(high_b+1)*bs] -= best_frac / bs
            h[low_b*bs:(low_b+1)*bs] += best_frac / bs
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    idxs = np.arange(n)
    grads = np.zeros((n_corr, n))
    for s in range(n_corr):
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grads[s, v1] += (1.0 - h[idxs[v1] - k])
        grads[s, v2] -= h[idxs[v2] + k]
    return corr, grads

def sla_optimize(h0, n, target, time_limit, max_iter=200, damping=0.5):
    """Successive Linear Approximation."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr, grads = compute_corr_and_grads(h, n)
        c5_val = float(np.max(corr)) * dx
        
        # Build LP: min t s.t. corr_k(h) + grads[k,:] @ (h_var - h) <= t, bounds, sum constraint
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        # Linear constraints for each k
        # corr_k(h) + grads[k,:] @ (h_var - h) <= t_var
        # = corr_k(h) + grads[k,:] @ h_var - grads[k,:] @ h <= t_var
        lhs_coeff = grads  # shape (n_corr, n)
        rhs_const = corr - dx * grads  # wait, need to be careful with units
        
        # Actually: corr[s] + grads[s,:] @ (h_var - h) <= t
        # The corr values are already in "raw" units (before *dx)
        # We want: (corr[s] + grads[s,:] @ (h_var - h)) * dx <= t
        # => corr[s]*dx + dx*grads[s,:] @ h_var - dx*grads[s,:] @ h <= t
        
        # Let me define: a_s = dx * grads[s,:] (row vector)
        # b_s = corr[s]*dx - dx*grads[s,:] @ h
        # constraint: a_s @ h_var + b_s <= t
        
        a = dx * grads  # (n_corr, n)
        b = corr * dx - dx * (grads @ h)  # (n_corr,)
        
        # But wait, corr[s] = h^T A_s h type thing... let me just verify:
        # corr_k(h) = sum_i h(i)(1-h(i+k))
        # d/dh(j) corr_k = (1-h(j+k)) - h(j-k)
        # Linear approx: corr_k(h0) + sum_j [(1-h0(j+k)) - h0(j-k)] (h(j) - h0(j))
        
        # So the constraint is:
        # corr_k(h0) + grads[k,:] @ (h_var - h0) <= t/dx
        # (working in unscaled units)
        # => corr_k(h0) + grads[k,:] @ h_var - grads[k,:] @ h0 <= t/dx
        
        # In scaled units (multiply by dx):
        # corr_k(h0)*dx + dx*grads[k,:] @ h_var - dx*grads[k,:] @ h0 <= t_var
        
        for s in range(len(corr)):
            constraints.append(a[s] @ h_var + (corr[s] * dx - dx * grads[s] @ h) <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-7)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_sol = np.array(h_var.value, dtype=float).flatten()
        h_sol = project_feasible(h_sol, target)
        c5_sol = compute_c5(h_sol)
        
        # Damped update
        h_new = (1 - damping) * h + damping * h_sol
        
        if c5_sol < best_c5 - 1e-12:
            best_c5 = c5_sol
            best_h = h_sol.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve > 30:
            break
    
    return best_h, best_c5
