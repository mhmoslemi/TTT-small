```python
def frank_wolfe_step(h, n, alpha, eta):
    """One Frank-Wolfe step with log-sum-exp smoothing."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    
    # Compute gradient of smoothed objective
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    # Frank-Wolfe: find v = argmin grad·v s.t. v ∈ P
    # v[i] = 1 if grad[i] is small, 0 if large
    # Need sum(v) = n/2
    sorted_g = np.sort(grad)
    # v[i] = 1 for the smallest n/2 values of grad
    threshold_idx = n // 2
    threshold = sorted_g[threshold_idx]
    v = (grad < threshold).astype(float)
    # Handle ties
    tie_mask = np.isclose(grad, threshold, atol=1e-14)
    n_ties = tie_mask.sum()
    deficit = n // 2 - v.sum()
    if deficit > 0 and n_ties > 0:
        v[tie_mask] = deficit / n_ties
    elif deficit < 0:
        # Shouldn't happen but handle
        v[~tie_mask] = 0
        v[tie_mask] = 1.0
    
    # Update
    h_new = (1 - eta) * h + eta * v
    return h_new, grad
```

def fw_step(h, n, target, alpha):
    grad = smooth_obj_grad(h, alpha)
    sorted_g = np.sort(grad)
    threshold = sorted_g[n // 2]
    v = np.zeros(n)
    below = grad < threshold - 1e-14
    v[below] = 1.0
    deficit = n / 2.0 - v.sum()
    if abs(deficit) > 1e-10:
        tie = np.isclose(grad, threshold, atol=1e-10)
        n_ties = max(1, tie.sum())
        v[tie] = deficit / n_ties
    return project_feasible(v, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    h_cur = h_start.copy()
    cur = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cm = np.max(corr)
            sh = np.clip(alpha * (corr - cm), -500.0, 0.0)
            return float(cm + np.log(np.sum(np.exp(sh))) / alpha)
        def objg(x):
            return smooth_obj_grad(x, alpha)
        try:
            res = minimize(obj, h_cur, jac=objg, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                cur = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        hcf, ccf = pair_cd(best_h, cur_n, cur_target, min(25, tl * 0.7),
                           n_cand_start=60, n_cand_end=150, top_k=100,
                           max_no_improve=300, n_shifts=5,
                           use_weighted_score=True)
        if ccf < best_score:
            best_score = ccf
            best_h = hcf

    tl = _time_left(start_time, budget_s)
    if tl > 20:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        hfin = slsqp_optimize(best_h, cur_n, cur_target, [10000, 50000], 400, 1e-15)
        sc = compute_c5(hfin)
        if sc < best_score:
            best_score = sc
            best_h = hfin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
