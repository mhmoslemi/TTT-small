import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha

def _slsqp_con_fun(x, target):
    return np.sum(x) - target

def _slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h_start, target, alphas, maxiter_per_alpha):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _slsqp_con_fun(x, target),
             'jac': lambda x: _slsqp_con_jac(x)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x, a=alpha: _slsqp_obj(x, a, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_out = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_out, target_dst)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        for ell in range(-(n - 1), n):
            k = mid + ell
            val = 0.0
            if 0 <= p + ell < n:
                val += 1.0 - h[p + ell]
            if 0 <= p - ell < n:
                val -= h[p - ell]
            change[p, k] = val
    return change

def update_change_transfer(change, n, i, j, delta):
    mid = n - 1
    ps = np.arange(i + 1)
    ks = mid + i - ps
    change[ps, ks] += delta
    ps = np.arange(i, n)
    ks = mid + ps - i
    change[ps, ks] += delta
    ps = np.arange(j + 1)
    ks = mid + j - ps
    change[ps, ks] -= delta
    ps = np.arange(j, n)
    ks = mid + ps - j
    change[ps, ks] -= delta

def fast_cd(h_init, n, target, time_limit, max_sweeps=200000,
            n_active=3, n_cand=30, max_no_improve=400,
            n_bisect=8, max_pairs=80, recompute_every=500):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    change = build_change_matrix(h, n)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    sweep_count = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        sweep_count += 1

        if sweep_count % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            change = build_change_matrix(h, n)

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        w_sum = float(corr[top_idx].sum())
        if w_sum < 1e-15:
            w_sum = 1.0

        grad = np.zeros(n)

        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        total_pairs = n_h * n_l

        if total_pairs <= max_pairs:

            pairs_j = []

                    pairs_i.append(iv)
                    pairs_j.append(jv)

        for idx in range(len(pairs_i)):
            i = pairs_i[idx]
            j = pairs_j[idx]
            delta_max = min(h[j], 1.0 - h[i])
            if delta_max < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_max
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = float(np.max(g[active]))
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue

            approx_max = float(np.max(corr + delta_star * g)) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            corr += best_delta * (change[best_j] - change[best_i])
            update_change_transfer(change, n, best_i, best_j, best_delta)

    return best_h, best_c5
