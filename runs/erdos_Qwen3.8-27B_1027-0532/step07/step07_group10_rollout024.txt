import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        v_pos = (idxs - k >= 0) & (idxs - k < n)
        v_neg = (idxs + k >= 0) & (idxs + k < n)
        grad[v_pos] += weights[k_idx] * (1.0 - x[idxs[v_pos] - k])
        grad[v_neg] -= weights[k_idx] * x[idxs[v_neg] + k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p - shifts[pos_mask]])
        change[p, neg_mask] -= h[p + shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p - shifts[pos_mask]])
    change[p, neg_mask] -= h[p + shifts[neg_mask]]
    return change

            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 150:
            h1_cd, c5_1cd = fast_pair_cd(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.6),
                                         max_sweeps=25000,
                                         n_cand_start=25, n_cand_end=80,
                                         max_no_improve=400, n_active=3)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 60:
            h1_gd, c5_1gd = gradient_descent_active(h1, n1, target1,
                                                    time_limit=min(50, tleft() * 0.4),
                                                    max_steps=30000, base_lr=0.02,
                                                    n_active=8)
            if c5_1gd < best_score:
                best_score = c5_1gd
                best_h = h1_gd
                best_n = n1
                best_target = target1

        if tleft() > 50:
            h1_pol2, sc1p2 = slsqp_round(best_h, best_n, best_target,
                                         [5000, 20000, 100000], 1000, 1e-15)
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2

    n2 = 1000
    if tleft() > 300:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 250:
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [100, 500, 2000, 10000], 600, 1e-15)
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 180:
            h2_cd, c5_2cd = fast_pair_cd(h2, n2, target2,
                                         time_limit=min(200, tleft() * 0.55),
                                         max_sweeps=20000,
                                         n_cand_start=25, n_cand_end=70,
                                         max_no_improve=400, n_active=3)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 60:
            h2_gd, c5_2gd = gradient_descent_active(h2, n2, target2,
                                                    time_limit=min(50, tleft() * 0.4),
                                                    max_steps=20000, base_lr=0.015,
                                                    n_active=10)
            if c5_2gd < best_score:
                best_score = c5_2gd
                best_h = h2_gd
                best_n = n2
                best_target = target2

        if tleft() > 40:
            h2_pol2, sc2p2 = slsqp_round(best_h, best_n, best_target,
                                         [10000, 50000], 800, 1e-15)
            if sc2p2 < best_score:
                best_score = sc2p2
                best_h = h2_pol2

    if tleft() > 80:
        h_b3, c5_b3 = block3_cd(best_h, best_n, best_target,
                                time_limit=min(80, tleft() * 0.6),
                                max_sweeps=10000, n_cand=25, max_no_improve=200)
        if c5_b3 < best_score:
            best_score = c5_b3
            best_h = h_b3

    if tleft() > 80:
        h_cd_final, c5_final = fast_pair_cd(best_h, best_n, best_target,
                                            time_limit=min(100, tleft() * 0.6),
                                            max_sweeps=15000,
                                            n_cand_start=40, n_cand_end=100,
                                            max_no_improve=300, n_active=5)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 50:
        h_pol3, sc3 = slsqp_round(best_h, best_n, best_target,
                                  [50000, 200000], 1000, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = slsqp_round(h_pert, best_n, best_target,
                                    [500, 5000, 50000], 400, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                best_n = len(h_ref)
                best_target = best_n / 2.0

    if tleft() > 30:
        h_cd_last, c5_last = fast_pair_cd(best_h, best_n, best_target,
                                          time_limit=min(25, tleft() * 0.5),
                                          max_sweeps=5000,
                                          n_cand_start=50, n_cand_end=100,
                                          max_no_improve=100, n_active=1)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    if tleft() > 20:
        h_pol4, sc4 = slsqp_round(best_h, best_n, best_target,
                                  [100000, 500000], 500, 1e-15)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_pol4

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
