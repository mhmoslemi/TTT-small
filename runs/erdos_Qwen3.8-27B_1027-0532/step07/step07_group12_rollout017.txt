        elif no_improve > 200:

    return best_h, best_c5

def active_set_lbfgs(h_start, n, target, alphas, n_active_list, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for na in n_active_list:
            if time.time() - start > time_limit:
                break
            for pen in [10.0, 100.0, 1000.0]:
                if time.time() - start > time_limit:
                    break
                try:
                    res = minimize(
                        lse_active_objective, h_cur,
                        args=(alpha, target, pen, na),
                        jac=lambda x, a=t, p=pen, n=na: lse_active_gradient(x, alpha, t, p, n),
                        method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
                    h_new = project_feasible(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h_cur = h_new
                    if abs(np.sum(h_cur) - target) < 1e-9:
                        break
                except Exception:
                    pass

    return best_h, best_c5

def full_lse_lbfgs(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_pair_cd(h_init, n, target, time_limit, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    ks_all = np.arange(-(n - 1), n)

    while time.time() - start < time_limit:
        no_improve += 1
        if no_improve > 600:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))

        grad = np.zeros(n)
        n_top = max(5, int(0.01 * n_corr))
        top_shifts = np.argsort(corr)[-n_top:]
        for si in top_shifts:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]

        sorted_idx = np.argsort(-grad)
        high_set = sorted_idx[:n_cand]
        low_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        valid_plus = (all_cands[:, None] + ks_all[None, :] >= 0) & (all_cands[:, None] + ks_all[None, :] < n)
        valid_minus = (all_cands[:, None] - ks_all[None, :] >= 0) & (all_cands[:, None] - ks_all[None, :] < n)
        plus_idx = np.clip(all_cands[:, None] + ks_all[None, :], 0, n - 1)
        minus_idx = np.clip(all_cands[:, None] - ks_all[None, :], 0, n - 1)
        prof_plus = np.where(valid_plus, 1.0 - h[plus_idx], 0.0)
        prof_minus = np.where(valid_minus, h[minus_idx], 0.0)
        profiles = prof_plus - prof_minus

        updated = 0
        n_high = min(len(high_set), 30)
        n_low = min(len(low_set), 30)
        hi_cand = high_set[:n_high]
        lo_cand = low_set[:n_low]

        hi_prof = np.array([list(all_cands).index(int(x)) for x in hi_cand])
        lo_prof = np.array([list(all_cands).index(int(x)) for x in lo_cand])

        pair_hi, pair_lo = np.meshgrid(hi_prof, lo_prof, indexing='ij')
        pair_hi = pair_hi.ravel()
        pair_lo = pair_lo.ravel()

        n_pairs = len(pair_hi)
        batch_size = 400

        for bstart in range(0, n_pairs, batch_size):
            if time.time() - start > time_limit:
                break
            bend = min(bstart + batch_size, n_pairs)
            bh = pair_hi[bstart:bend]
            bl = pair_lo[bstart:bend]

            g_batch = profiles[bl] - profiles[bh]
            hi_orig = np.tile(hi_cand, n_low)[bstart:bend]
            lo_orig = np.repeat(lo_cand, n_high)[bstart:bend]
            max_delta = np.minimum(h[hi_orig], 1.0 - h[lo_orig])
            max_delta = np.maximum(max_delta, 0.0)

            lo_b = np.zeros(bend - bstart)
            hi_b = max_delta.copy()

            for _ in range(12):
                mid_d = 0.5 * (lo_b + hi_b)
                new_corr = corr[None, :] + mid_d[:, None] * g_batch
                new_max = np.max(new_corr, axis=1)
                active_k = np.argmax(new_corr, axis=1)
                f_prime = g_batch[np.arange(len(mid_d)), active_k]
                hi_b[f_prime > 0] = mid_d[f_prime > 0]
                lo_b[f_prime <= 0] = mid_d[f_prime <= 0]

            opt_delta = 0.5 * (lo_b + hi_b)
            new_max = np.max(corr[None, :] + opt_delta[:, None] * g_batch, axis=1)
            improve = cur_max - new_max

            for lb in np.where(improve > 1e-8)[0]:
                i_idx = int(hi_orig[lb])
                j_idx = int(lo_orig[lb])
                d = float(opt_delta[lb])
                h[i_idx] -= d
                h[j_idx] += d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                updated += 1

        if updated > 0:
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
            no_improve = 0
        else:
            if no_improve > 100:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast pair CD (main workhorse)
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.35)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=50, top_frac=0.02)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 2: Active-set LSE-LBFGS
    if tleft() > 100:
        hp, sc = active_set_lbfgs(best_h, n, target,
                                  alphas=[200, 1000, 5000],
                                  n_active_list=[10, 20, 50],
                                  maxiter=300, time_limit=min(120, tleft() * 0.25),
                                  penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 3: Multi-pair CD (update multiple pairs per sweep)
    if tleft() > 150:
        hcd, c5cd = multi_pair_cd(best_h, n, target, time_limit=min(150, tleft() * 0.25), n_cand=40)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 4: Fast pair CD with different parameters
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.25)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=60, top_frac=0.05)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Full LSE-LBFGS with very high alpha
    if tleft() > 80:
        hp, sc = full_lse_lbfgs(best_h, n, target,
                                alphas=[5000, 20000, 100000],
                                maxiter=300, time_limit=min(100, tleft() * 0.25),
                                penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 6: Another round of fast pair CD
    if tleft() > 100:
        cd_time = min(120, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=70, top_frac=0.01)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 7: Try higher resolution
    if tleft() > 250:
        n_new = 1000
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = full_lse_lbfgs(h_interp, n_new, target_new,
                                alphas=[100, 1000, 5000],
                                maxiter=200, time_limit=min(100, tleft() * 0.25),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        if tleft() > 100:
            hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.3), n_cand=40, top_frac=0.02)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 8: Active set polish
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = active_set_lbfgs(best_h, n_use, target_use,
                                  alphas=[5000, 20000],
                                  n_active_list=[5, 10, 20],
                                  maxiter=200, time_limit=min(80, tleft() * 0.3),
                                  penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 9: Final pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.6), n_cand=50, top_frac=0.02)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
