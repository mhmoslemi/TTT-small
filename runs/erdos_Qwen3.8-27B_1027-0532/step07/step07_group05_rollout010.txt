def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_vec(h, p, n):
    """Compute row p of change matrix: d/dh[p] corr."""
    k_range = np.arange(-(n-1), n)
    pk = p - k_range
    ppk = p + k_range
    valid_pk = (pk >= 0) & (pk < n)
    valid_ppk = (ppk >= 0) & (ppk < n)
    term1 = np.where(valid_pk, 1.0 - h[pk], 0.0)
    term2 = np.where(valid_ppk, -h[ppk], 0.0)
    return term1 + term2

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_row(h, p, n):
    k_range = np.arange(-(n - 1), n)
    pk = p - k_range
    ppk = p + k_range
    valid_pk = (pk >= 0) & (pk < n)
    valid_ppk = (ppk >= 0) & (ppk < n)
    term1 = np.where(valid_pk, 1.0 - h[pk], 0.0)
    term2 = np.where(valid_ppk, -h[ppk], 0.0)
    return term1 + term2


def fast_cd(h_init, n, target, time_limit,
            n_top=5, n_cand_low=30, n_cand_high=30,
            bisection_iters=10, max_no_improve=300,
            max_pairs=250, n_restarts=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    restarts = 0
    rng = np.random.RandomState(12345)

    corr = np.correlate(h, 1.0 - h, mode='full')
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p] = compute_change_row(h, p, n)

    while True:
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                if restarts >= n_restarts:
                    break
                restarts += 1
                h = h + rng.randn(n) * (0.03 + 0.02 * restarts)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                for p in range(n):
                    change[p] = compute_change_row(h, p, n)
                no_improve = 0
                continue

        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += change[:, idx]

        sorted_g = np.argsort(grad)
        low_set = sorted_g[:n_cand_low]
        high_set = sorted_g[-n_cand_high:]

        best_gain = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_l = len(low_set)
        n_h = len(high_set)
        total_pairs = n_l * n_h

        if total_pairs <= max_pairs:
            pair_list = [(int(high_set[i]), int(low_set[j]))
                         for i in range(n_h) for j in range(n_l)

        else:

            for fh in np.linspace(0, 1, 20):

                    jv = int(low_set[int(fl * (n_l - 1))])

            n_extra = max(0, max_pairs - len(pair_list))

                    pair_list.append((iv, jv))
            pair_list = pair_list[:max_pairs]

        for i, j in pair_list:
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max < 1e-12:
                continue
            g_vec = change[j] - change[i]
            a_d = 0.0
            b_d = delta_max
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_vec
                f_mid = np.max(vals)

                    f_prime = np.max(g_vec[active])

                    f_prime = 0.0
                if f_prime > 1e-10:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:

            new_c5 = np.max(corr + delta_star * g_vec) * dx
            gain = c5_val - new_c5
            if gain > best_gain:

                best_i = i
                best_j = j

        if best_i >= 0 and best_gain > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-10:

                if mask.any():
                    h[mask] += diff / mask.sum()
                    h = np.clip(h, 0.0, 1.0)
                else:
                    h = project_feasible(h, target)
            change[best_i] = compute_change_row(h, best_i, n)
            change[best_j] = compute_change_row(h, best_j, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
        elif no_improve % 100 == 0:
            idx1 = rng.randint(n)
            idx2 = rng.randint(n)
            if idx1 != idx2:
                d = 0.002
                h[idx1] -= d
                h[idx2] += d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                change[idx1] = compute_change_row(h, idx1, n)
                change[idx2] = compute_change_row(h, idx2, n)
                corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5
