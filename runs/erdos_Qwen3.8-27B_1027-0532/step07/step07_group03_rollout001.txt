    """Find delta in [delta_lo, delta_hi] that minimizes max(corr + delta * g_vec)."""
    # Check endpoints
    f_lo = np.max(corr + delta_lo * g_vec)
    f_hi = np.max(corr + delta_hi * g_vec)
    
    # If f is non-decreasing on [lo, hi], return lo
    # If f is non-increasing, return hi
    # Otherwise, find the minimum

        f_mid = np.max(vals)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_corr_gradient(h, weights, n):
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def softmax_weights(corr, alpha, top_k=20):
    n_corr = len(corr)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    if top_k < n_corr:
        idx = np.argsort(w)[::-1][:top_k]
        w_masked = np.zeros_like(w)
        w_masked[idx] = w[idx]
        w = w_masked
    s = np.sum(w)
    if s < 1e-15:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w = w / s
    return w

def grad_cd(h_init, n, target, time_limit, alpha_init=50, alpha_end=200,
            top_k_init=5, top_k_end=30, max_iters=5000, n_cand=30):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break

        progress = min(float(it) / max_iters, 1.0)
        alpha = alpha_init + (alpha_end - alpha_init) * progress
        top_k = int(top_k_init + (top_k_end - top_k_init) * progress)
        top_k = min(top_k, n_corr)

        weights = softmax_weights(corr, alpha, top_k)

        grad = weighted_corr_gradient(h, weights, n)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        change = np.zeros((n, n_corr))
        for p in range(n):
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
            change[p, neg_mask] -= h[p - shifts[neg_mask]]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = min(n_cand, len(high_set))
        n_l = min(n_cand, len(low_set))
        for ii in range(n_h):
            i = int(high_set[n_cand - 1 - ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_vec = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        g_max = np.max(g_vec[active])
                    else:
                        g_max = 0.0
                    if g_max > 1e-10:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g_vec) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_score

def slsqp_polish(h_init, n, target, alphas, maxiter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    def obj_func(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)

    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    for alpha in alphas:
        try:
            res = minimize(obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def cutting_plane_opt(h_init, n, target, time_limit, max_iters=5000, tol=1e-10, max_cuts=100):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_b = []
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        active_mask = corr >= corr[k_star] - 1e-6
        active_lags = np.where(active_mask)[0]
        n_act = len(active_lags)
        w = np.ones(n_act) / n_act

        idxs = np.arange(n)
        g = np.zeros(n)
        for ks, wi in zip(active_lags, w):
            k = int(ks) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += wi * (1.0 - h[idxs[v1] + k])
            g[v2] -= wi * h[idxs[v2] - k]

        cuts_g.append(g)
        cuts_b.append(float(g @ h) - f_val)

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_b = [cuts_b[0]] + cuts_b[-keep:]

        nc = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((nc, n + 1))
        b_ub = np.zeros(nc)
        for i in range(nc):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_b[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
            if gap < tol:
                break
        except Exception:
            break

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_restart(h, n, target, rng, mode=0):
    if mode == 0:
        h_new = h + rng.randn(n) * 0.03
    elif mode == 1:
        h_new = h + rng.randn(n) * 0.08
    elif mode == 2:
        h_bin = (h > 0.5).astype(float)
        idx_ones = np.where(h_bin == 1.0)[0]
        idx_zeros = np.where(h_bin == 0.0)[0]
        n_swap = max(1, int(0.05 * n))
        for _ in range(n_swap):
            if len(idx_ones) == 0 or len(idx_zeros) == 0:
                break
            i = int(rng.choice(idx_ones))
            j = int(rng.choice(idx_zeros))
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            idx_ones = np.where(h_bin == 1.0)[0]

        h_new[i:i+block] = 1.0 - h_new[i:i+block]
    else:
        h_new = h + rng.randn(n) * 0.05
    return project_feasible(h_new, target)

        h600 = interpolate_to(best_h, n2, target2)
        sc600 = compute_c5(h600)
        if tleft() > 250:
            h600_cd, sc600_cd = grad_cd(h600, n2, target2, time_limit=min(180, tleft() * 0.4),
                                         alpha_init=50, alpha_end=250, top_k_init=5, top_k_end=30,
                                         max_iters=5000, n_cand=25)
            if sc600_cd < best_score:
                best_score = sc600_cd
                best_h = h600_cd
        if tleft() > 100:
            h600_sls, sc600_sls = slsqp_polish(best_h, len(best_h), len(best_h)/2.0,
                                                [2000, 8000], 400, 1e-14)
            if sc600_sls < best_score:
                best_score = sc600_sls
                best_h = h600_sls

    if tleft() > 200:
        n3 = 800
        target3 = n3 / 2.0
        h800 = interpolate_to(best_h, n3, target3)
        sc800 = compute_c5(h800)
        if tleft() > 150:
            h800_cd, sc800_cd = grad_cd(h800, n3, target3, time_limit=min(120, tleft() * 0.45),
                                         alpha_init=50, alpha_end=200, top_k_init=5, top_k_end=25,
                                         max_iters=4000, n_cand=20)
            if sc800_cd < best_score:
                best_score = sc800_cd
                best_h = h800_cd
        if tleft() > 60:
            h800_sls, sc800_sls = slsqp_polish(best_h, len(best_h), len(best_h)/2.0,
                                                [5000], 300, 1e-14)
            if sc800_sls < best_score:
                best_score = sc800_sls
                best_h = h800_sls

    if tleft() > 60:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_fin, sc_fin = grad_cd(best_h, n_cur, tgt_cur, time_limit=min(45, tleft() * 0.6),
                                alpha_init=200, alpha_end=500, top_k_init=10, top_k_end=30,
                                max_iters=3000, n_cand=40)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    if tleft() > 40:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_sls, sc_sls = slsqp_polish(best_h, n_cur, tgt_cur, [10000, 30000], 400, 1e-14)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
