def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):

    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        if tleft() > 80:
            h_a3, sc3 = adam_optimize(best_h, n0, target0,
                                      time_limit=min(60, tleft() * 0.6),
                                      alpha=200.0, lr=0.3, top_k_shifts=3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_a3

    # Phase 6: Multi-shift CD
    if tleft() > 120:
        h_ms, c5_ms = fast_cd(best_h, n0, target0,
                              time_limit=min(100, tleft() * 0.7),
                              max_sweeps=25000,
                              n_cand_start=40, n_cand_end=90,
                              max_no_improve=200,
                              n_shifts=3, alpha_smooth=5.0)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 7: SA for escape
    if tleft() > 100:
        h_sa, c5_sa = sa_optimize(best_h, n0, target0,
                                  time_limit=min(80, tleft() * 0.7),
                                  max_steps=25000,
                                  t_start=0.02, t_end=0.0001,
                                  n_cand=50, alpha_smooth=3.0, n_shifts=2)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 8: Go to n=800
    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        h_cd2, c5_cd2 = fast_cd(h2, n2, target2,
                                time_limit=min(150, tleft() * 0.6),
                                max_sweeps=30000,
                                n_cand_start=30, n_cand_end=70,
                                max_no_improve=250)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n0 = n2
            target0 = target2

        if tleft() > 60:
            h_a4, sc4 = adam_optimize(best_h, n0, target0,
                                      time_limit=min(50, tleft() * 0.6),
                                      alpha=500.0, lr=0.15, top_k_shifts=2)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_a4

    # Phase 9: Final Adam with very high alpha
    if tleft() > 60:
        h_a5, sc5 = adam_optimize(best_h, n0, target0,
                                  time_limit=min(50, tleft() * 0.7),
                                  alpha=5000.0, lr=0.1, top_k_shifts=1)
        if sc5 < best_score:
            best_score = sc5
            best_h = h_a5

    # Phase 10: Final CD polish
    if tleft() > 30:
        h_cd_f, c5_cf = fast_cd(best_h, n0, target0,
                                time_limit=min(25, tleft() * 0.7),
                                max_sweeps=10000,
                                n_cand_start=50, n_cand_end=100,
                                max_no_improve=100)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cd_f

    # Final verification
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    if final_score > best_score:
        final_h = best_h
        final_score = best_score

    return final_h, final_score, len(final_h)
