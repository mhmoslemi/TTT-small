```python
import numpy as np
from scipy.optimize import minimize
import time

_n_global = 0
_target_global = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def _cons_sum(x):
    return np.sum(x) - _target_global

def _cons_sum_jac(x):
    return np.ones(_n_global)

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-12, deadline=None):
    global _n_global, _target_global
    _n_global = n
    _target_global = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_sum, 'jac': _cons_sum_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

Let me try a different approach: **use the problem structure more directly**.

The key insight is that we want to equalize the correlation values. If we could make all c_k equal, that would be optimal (by a minimax argument). The "least squares" version of this would be to minimize ∑(c_k - c̄)² where c̄ = mean(c_k). This is a convex problem (since the variance is a convex function of the quadratic forms... wait, no, it's not convex in h).

But we can use it as a **descent direction**: at each step, compute the variance of the c_k values, find the direction that reduces it most, and take a step in that direction.

The variance of c_k is:
Var = (1/|K|) ∑(c_k - c̄)²

The gradient of Var with respect to h is:
∂Var/∂h[i] = (2/|K|) ∑ (c_k - c̄) * ∂c_k/∂h[i]

This is a weighted sum of the individual gradients, weighted by (c_k - c̄). Shifts with high c_k get positive weight (decrease them), shifts with low c_k get negative weight (increase them).

This is essentially a **water-filling** approach: reduce the peaks and raise the valleys.

Let me implement this as an alternative to the CD:

```python
def waterfilling(h_init, n, target, time_limit, alpha0=0.1, alpha_min=0.0001):

            best_c5 = c5_val
            best_h = h.copy()

        else:
            alpha *= 0.97
        
        # Compute waterfilling direction
        c_mean = np.mean(corr)
        weights = corr - c_mean  # positive for peaks, negative for valleys
        # Normalize
        w_norm = np.sqrt(np.sum(weights**2))
        if w_norm < 1e-12:
            break
        weights /= w_norm
        
        # Compute gradient of weighted sum
        mid = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        for s in range(len(corr)):
            if abs(weights[s]) < 1e-8:

            grad[v2] -= weights[s] * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
```

```python
def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _gt


def _cons_jac(x):
    return np.ones(_gn)


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_var(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    c_mean = np.mean(corr)
    var = np.mean((corr - c_mean) ** 2)
    return float(lse + 0.1 * var)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None, use_var=False):
    global _gn, _gt
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    obj_func = smooth_obj_var if use_var else smooth_obj
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        else:
            stale += 1
            alpha = max(alpha * 0.97, alpha_min)
            if stale > 800:
                break

        c_mean = np.mean(corr)
        weights = corr - c_mean
        w_norm = np.sqrt(np.sum(weights ** 2))
        if w_norm < 1e-12:
            break
        weights /= w_norm

        n_active = max(5, int(len(corr) * n_active_frac))
        top_idx = np.argsort(np.abs(weights))[-n_active:]

        grad = np.zeros(n)
        for s in top_idx:
            k = int(s) - mid
            ws = weights[s]
            if abs(ws) < 1e-10:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += ws * (1.0 - h[idxs[v1] - k])
            grad[v2] -= ws * h[idxs[v2] + k]

        step_size = alpha / (np.sqrt(np.sum(grad ** 2)) + 1e-8)
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
