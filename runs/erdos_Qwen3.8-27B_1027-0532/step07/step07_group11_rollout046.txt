def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:

    return h_cur, cur_score

def lp_linearize(h, n, target, tol_frac=0.05):
    """Successive linearization step using LP."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    active = np.where(corr >= c_max * (1 - tol_frac))[0]
    
    if len(active) == 0:
        return h, compute_c5(h)
    
    # Build LP
    n_corr = 2 * n - 1
    mid = n - 1
    j = np.arange(n)
    
    m = len(active)
    n_vars = n + 1
    A = np.zeros((m + 2 * n, n_vars))
    b = np.zeros(m + 2 * n)
    
    for idx, s_idx in enumerate(active):
        k = int(s_idx) - mid
        c = np.zeros(n)
        m1 = (j + k >= 0) & (j + k < n)
        m2 = (j - k >= 0) & (j - k < n)
        c[m1] += (1.0 - h[j[m1] + k])
        c[m2] -= h[j[m2] - k]
        d = float(np.sum(h[m1] * h[m1 + k]))
        A[idx, :n] = c
        A[idx, n] = -1.0
        b[idx] = -d
    
    A[m:m+n, :n] = -np.eye(n)
    b[m:m+n] = 0
    A[m+n:m+2*n, :n] = np.eye(n)
    b[m+n:m+2*n] = 1.0
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    bounds_lp = [(0, 1)] * n + [(-0.001 * c_max, 0.001 * c_max)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_lp, method='highs')
        if res.success:
            h_new = project_feasible(res.x[:n], target)
            return h_new
    except Exception:
        pass
    return h

def successive_lp(h_init, n, target, time_limit, max_iter=200, tol_frac=0.05):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        h_new = lp_linearize(h, n, target, tol_frac)
        sc = compute_c5(h_new)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_new.copy()
        h = h_new
        # If LP solution is same as current, try smaller tolerance
        if np.allclose(h, h_init if it == 0 else best_h, atol=1e-12):
            # Stuck, try different tolerance
            tol_frac = max(0.01, tol_frac * 0.5)
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000, n_cand=50, n_shifts=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        top = np.argsort(corr)[-n_shifts:]
        sw = corr[top]
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        sw /= sw_sum
        
        grad = np.zeros(n)
        for si in range(n_shifts):
            k = int(top[si]) - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sec = n * (2*n-1)  # size of change matrix
        # Recompute change for top shifts only
        # delta corr when we decrease h[i] by d and increase h[j] by d:
        # For shift k: change = d * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
        #             = d * [h[j+k] - h[i+k] + h[j-k] - h[i-k]]  (for valid indices)
        
        # Actually let me use the full change matrix approach
        # time() is n * (2n-1) which for n=400 is 400*799 = 319600, manageable
        change = np.zeros((n, n_corr))
        for p in range(n):
            pos = (p + shifts >= 0) & (p + shifts < n)
            neg = (p - shifts >= 0) & (p - shifts < n)
            change[p, pos] += (1.0 - h[p + shifts[pos]])
            change[p, neg] -= h[p - shifts[neg]]
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5
        best_move = None
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a, b2 = d_lo, d_hi
                for _ in range(14):
                    mid_d = 0.5 * (a + b2)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mk = vals >= f_mid - 1e-14
                    if active_mk.any():
                        f_prime = np.max(g[active_mk])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b2 = mid_d
                    else:
                        a = mid_d
                d_star = 0.5 * (a + b2)
                if abs(d_star) < 1e-12:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial - 1e-14:
                    best_trial = approx
                    best_move = (i, j, d_star)
        
        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            # Try single point moves
            for i in range(0, n, 3):
                # Try to decrease h[i] and spread to random others
                # Skip for now
                pass
            no_improve += 10
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):

def lp_single_step(h, n, target, tol_frac):

    c_max = float(np.max(corr))
    active = np.where(corr >= c_max * (1.0 - tol_frac))[0]
    if len(active) == 0 or len(active) > 2 * n:
        return h

    j = np.arange(n)

        c[m1] += (1.0 - h[j[m1] + k])
        c[m2] -= h[j[m2] - k]
        d = float(np.sum(h[m1] * h[m1 + k]))
        A[idx, :n] = c
        A[idx, n] = -1.0
        b[idx] = -d
    A[m:m+n, :n] = -np.eye(n)
    b[m:m+n] = 0.0
    A[m+n:m+2*n, :n] = np.eye(n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
