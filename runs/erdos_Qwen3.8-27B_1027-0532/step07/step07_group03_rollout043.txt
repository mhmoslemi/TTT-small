- corr_k changes by -δ * h[i-k] for the k where i-k is valid

The `change[i, :]` matrix stores exactly this: the change in the full correlation vector when h[i] increases by 1.

So for a swap (h[i] -= δ, h[j] += δ), the correlation changes by:
δ * (change[j] - change[i])

We want this to decrease the max correlation. The gradient of the max at the current point tells us the direction of increase, so we want (change[j] - change[i]) to be in the opposite direction.

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 5:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def cutting_plane_method(h_init, n, target, time_limit, max_iters=50000, tol=1e-10, max_cuts=300, act_tol=1e-6):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    no_improve = 0
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            if len(cuts_g) > 10:
                keep = max(10, len(cuts_g) // 3)
                cuts_g = [cuts_g[0]] + cuts_g[-keep:]
                cuts_f = [cuts_f[0]] + cuts_f[-keep:]
                cuts_h = [cuts_h[0]] + cuts_h[-keep:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_mask = corr >= corr[k_star] - act_tol
        n_active = int(active_mask.sum())

        if n_active <= 3:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            g = np.zeros(n)
            for k_star_i, wi in zip(active_lags, w):
                k = int(k_star_i) - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                g[v1] += wi * (1.0 - h[idxs[v1] + k])
                g[v2] -= wi * h[idxs[v2] - k]
        else:
            k = k_star - mid
            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]

        cuts_g.append(g.copy())
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                if hasattr(res, 'status') and res.status == 2:
                    break
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interceptive_cd(h_init, n, target, time_limit, max_sweeps=20000,
                    n_cand_start=15, n_cand_end=40, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        active_mask = corr >= corr.max() - 1e-7
        if int(active_mask.sum()) <= 5:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            grad = np.zeros(n)
            for k_star_i, wi in zip(active_lags, w):
                k = int(k_star_i) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                grad[v2] -= wi * h[idxs[v2] + k]
        else:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_subgrad(h_init, n, target, time_limit, max_iters=5000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h_curr = h_init.copy().astype(float)
    h_mom = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    mu = 1e-4
    step_size = 0.05

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * it * it))
        h_mom = h_curr + (1.0 - t_k) * (h_curr - h_mom)
        h_mom = np.clip(h_mom, 0.0, 1.0)

        corr = compute_corr(h_mom)
        c_max = np.max(corr)
        alpha = 200.0
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-15:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = p / p_sum

        grad = np.zeros(n)
        for m in range(len(corr)):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - h_mom[idxs[v1] + k])
            grad[v2] -= p[m] * h_mom[idxs[v2] - k]

        h_new = h_mom - step_size * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc <= compute_c5(h_curr) + 1e-12:
            h_curr = h_new
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_curr.copy()
        else:
            h_new2 = h_curr - 0.5 * step_size * grad
            h_new2 = project_feasible(h_new2, target)
            h_curr = h_new2
            sc2 = compute_c5(h_curr)
            if sc2 < best_score - 1e-14:
                best_score = sc2
                best_h = h_curr.copy()

        step_size *= 0.9995

    return best_h, best_score


def anneal_refine(h_init, n, target, time_limit, T_start=0.005, T_end=0.0005, n_cand=25):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(12345)

    n_steps = 5000
    for it in range(n_steps):
        if time.time() - start > time_limit:
            break

        progress = float(it) / n_steps
        T = T_start * (T_end / T_start) ** progress

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        hi = h[i] - delta
        hj = h[j] + delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue

        h[i] = hi
        h[j] = hj
        h = project_feasible(h, target)
        sc_new = compute_c5(h)
        sc_old = compute_c5(best_h)

        if sc_new < best_score - 1e-14 or rng.rand() < np.exp(-(sc_new - best_score) / T):
            if sc_new < best_score - 1e-14:
                best_score = sc_new
                best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
            h = project_feasible(h, target)

    return best_h, best_score

            best_score, best_h = sc, h_final_cd

    if tleft() > 60:
        h_final_cp, sc = cutting_plane_method(best_h, n_cur, target_cur,
                                              time_limit=min(40, tleft() * 0.5),
                                              max_iters=10000, max_cuts=200, act_tol=5e-7)
        if sc < best_score:
            best_score, best_h = sc, h_final_cp

    if tleft() > 30:
        h_final_sls, sc = slsqp_round(best_h, n_cur, target_cur,
                                      [20000], 300, 1e-14,
                                      time_limit=min(20, tleft() * 0.6),
                                      start_time=start_time)
        if sc < best_score:
            best_score, best_h = sc, h_final_sls

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    if final_score > best_score:
        final_score = best_score

    return final_h, final_score, len(final_h)
