import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def lbfgs_smooth_obj(x, alpha, target, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    penalty = lam * (np.sum(x) - target) ** 2
    return obj + penalty


def lbfgs_smooth_grad(x, alpha, target, lam):
    n = len(x)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(n_corr) / n_corr
    else:
        p /= p_sum
    grad = np.zeros(n)
    for m in range(n_corr):
        w = p[m]
        if w < 1e-15:
            continue
        k = m - (n - 1)
        q1s = max(0, k)
        q1e = min(n, k + n)
        if q1s < q1e:
            grad[q1s:q1e] += w * x[q1s - k:q1e - k]
        q2s = max(0, -k)
        q2e = min(n, n - k)
        if q2s < q2e:
            grad[q2s:q2e] += w * x[q2s + k:q2e + k]
    grad *= -dx
    grad += 2.0 * lam * (np.sum(x) - target) * dx
    return grad


def lbfgs_optimize(h_init, n, target, alpha, lam, maxiter, time_limit):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(lbfgs_smooth_obj, h, fprime=lbfgs_smooth_grad,
                            args=(alpha, target, lam),
                            pgtol=1e-12, maxiter=maxiter,
                            iprint=0, factr=1e8, bounds=bounds)
    except Exception:
        return best_h, best_score
    if time.time() - start > time_limit:
        return best_h, best_score
    h_new = project_feasible(res[0], target)
    sc = compute_c5(h_new)
    if sc < best_score:
        best_score = sc
        best_h = h_new
    return best_h, best_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.93
        if step % 2000 == 0 and step > 0:
            eta = max(step_scale * max(0.15, 1.0 - step / n_steps), 1e-6)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 180:
        h_cd, c5_cd = pair_cd_advanced(best_h, cur_n, cur_target,
                                       time_limit=min(160, tleft() * 0.7),
                                       max_sweeps=50000,
                                       n_cand_start=50, n_cand_end=220,
                                       n_shifts=4, max_no_improve=800,
                                       anneal_temp=0.015, anneal_decay=0.9998)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        for alpha in [200, 800, 3000]:
            if tleft() < 30:
                break
            h_lb, sc = lbfgs_optimize(best_h, cur_n, cur_target,
                                      alpha=alpha, lam=500.0,
                                      maxiter=200, time_limit=min(25, tleft() * 0.4))
            if sc < best_score:
                best_score = sc
                best_h = h_lb

    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            cur_n = n1
            cur_target = target1
        if tleft() > 300:
            h_cd1, c5_cd1 = pair_cd_advanced(h1, n1, target1,
                                             time_limit=min(300, tleft() * 0.75),
                                             max_sweeps=40000,
                                             n_cand_start=40, n_cand_end=180,
                                             n_shifts=3, max_no_improve=600,
                                             anneal_temp=0.008, anneal_decay=0.9998)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1

    if tleft() > 80:
        for alpha in [500, 2000]:
            if tleft() < 30:
                break
            h_lb, sc = lbfgs_optimize(best_h, cur_n, cur_target,
                                      alpha=alpha, lam=300.0,
                                      maxiter=150, time_limit=min(30, tleft() * 0.4))
            if sc < best_score:
                best_score = sc
                best_h = h_lb

    if tleft() > 100:
        h_sd, c5_sd = subgradient_multi(best_h, cur_n, cur_target,
                                        time_limit=min(70, tleft() * 0.6),
                                        n_steps=20000, step_scale=0.003)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [3000, 8000], 1000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd_advanced(best_h, cur_n, cur_target,
                                         time_limit=min(120, tleft() * 0.7),
                                         max_sweeps=30000,
                                         n_cand_start=80, n_cand_end=250,
                                         n_shifts=5, max_no_improve=500,
                                         anneal_temp=0.005, anneal_decay=0.9995)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 80:
        rng = np.random.RandomState(seed + 1)
        for noise_scale in [0.03, 0.06, 0.1]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc = pair_cd_advanced(h_pert, cur_n, cur_target,
                                         time_limit=min(20, tleft() * 0.3),
                                         max_sweeps=10000,
                                         n_cand_start=50, n_cand_end=150,
                                         n_shifts=3, max_no_improve=300,
                                         anneal_temp=0.0, anneal_decay=1.0)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 40:
        h_cd3, c5_cd3 = pair_cd_advanced(best_h, cur_n, cur_target,
                                         time_limit=min(30, tleft() * 0.7),
                                         max_sweeps=15000,
                                         n_cand_start=100, n_cand_end=300,
                                         n_shifts=5, max_no_improve=300,
                                         anneal_temp=0.0, anneal_decay=1.0)
        if c5_cd3 < best_score:

            best_h = h_cd3

    if tleft() > 30:
        h_final, sc = slsqp_round(best_h, cur_n, cur_target,
                                  [5000, 15000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
