import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                                 n_cand_start=15, n_cand_end=60,
                                                 max_no_improve=350,
                                                 n_active_shifts=4,
                                                 reheat_interval=80, reheat_scale=0.03,
                                                 n_swings=2)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3
                    current_n = n3
                    current_target = target3
            if tleft() > 80:
                deadline = time.time() + min(60, tleft() * 0.5)
                h_pol4, sc4 = slsqp_round(best_h, current_n, current_target,
                                          [3000, 15000, 60000],
                                          800, 1e-14, deadline=deadline)
                if sc4 < best_score:
                    best_score = sc4
                    best_h = h_pol4

    if tleft() > 200:
        for attempt in range(5):
            if tleft() < 50:
                break
            noise_scales = [0.02, 0.04, 0.07, 0.1, 0.15]
            noise_scale = noise_scales[attempt % 5]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            sc_pert = compute_c5(h_pert)
            if sc_pert > best_score + 0.01:
                continue
            deadline = time.time() + min(30, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 2000, 10000, 40000],
                                        400, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 150:
        h_cd_f, c5_f = pair_cd_improved(best_h, current_n, current_target,
                                         time_limit=min(90, tleft() * 0.5),
                                         max_sweeps=15000,
                                         n_cand_start=30, n_cand_end=100,
                                         max_no_improve=300,
                                         n_active_shifts=6,
                                         reheat_interval=70, reheat_scale=0.02,
                                         n_swings=3)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    if tleft() > 100:
        h_sg, c5_sg = subgradient_cd(best_h, current_n, current_target,
                                      time_limit=min(50, tleft() * 0.5),
                                      alpha0=0.15, alpha_min=0.003)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.6)
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [5000, 20000, 80000, 200000],
                                            1000, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
