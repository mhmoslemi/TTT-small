    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1

        peak_idx = int(np.argmax(corr))
        peak_k = peak_idx - mid
        idxs = np.arange(n)

        grad = np.zeros(n)
        v1 = (idxs - peak_k >= 0) & (idxs - peak_k < n)
        v2 = (idxs + peak_k >= 0) & (idxs + peak_k < n)
        grad[v1] += (1.0 - h_bin[idxs[v1] - peak_k])
        grad[v2] -= h_bin[idxs[v2] + peak_k]
        grad -= np.mean(grad)

        one_idx = np.where(h_bin > 0.5)[0]
        zero_idx = np.where(h_bin < 0.5)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        one_grads = grad[one_idx]
        zero_grads = grad[zero_idx]

        top_ones = one_idx[np.argsort(one_grads)[-n_cand:]]
        top_zeros = zero_idx[np.argsort(zero_grads)[:n_cand]]

        best_trial = c5_val
        best_i = -1
        best_j = -1

        for i in top_ones:
            for j in top_zeros:
                g = change[j] - change[i]
                approx_max = np.max(corr + g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            update_change_row(change, best_i, h_bin, n, shifts)
            update_change_row(change, best_j, h_bin, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):

                                           max_sweeps=20000,
                                           n_cand_start=25, n_cand_end=70,
                                           max_no_improve=400)
        if c5_cd2 < best_score:

                                           n_cand_start=20, n_cand_end=60,

        if c5_cd3 < best_score:

            curr_target = target2

    if tleft() > 80:
        h_pol2 = slsqp_round(best_h, curr_n, curr_target,

    return final_h, final_score, len(final_h)
