    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-10
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

                        max_other = diff_val
                scores[li] = -g_at_max - 0.3 * max_other

            top_low = np.argsort(-scores)[:n_best_low]
            for li in top_low:
                j = int(low_set[li])
                if int(i) == j:
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[j]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def triple_cd(h_init, n, target, time_limit, n_cand=15, n_perturb=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        cand_pos = {}
        for ci in range(nc):
            m = int(all_cands[ci])
            cand_pos[m] = ci
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        rng = np.random.default_rng(int(time.time() * 1000) % (2**31))

        best_improve = 0.0
        best_move = None

        for _ in range(n_perturb):
            i1 = int(rng.choice(high_set, size=1)[0])
            i2 = int(rng.choice(high_set, size=1)[0])
            j1 = int(rng.choice(low_set, size=1)[0])
            j2 = int(rng.choice(low_set, size=1)[0] if len(low_set) > 1 else low_set[0])
            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2 or i1 == i2:
                continue

            d1_max = min(float(h[i1]), 1.0 - float(h[j1]))
            d2_max = min(float(h[i2]), 1.0 - float(h[j2]))
            if d1_max < 1e-12 or d2_max < 1e-12:
                continue

            ci1 = cand_pos[i1]
            ci2 = cand_pos[i2]
            cj1 = cand_pos[j1]
            cj2 = cand_pos[j2]
            g1 = profiles[ci1] - profiles[cj1]
            g2 = profiles[ci2] - profiles[cj2]

            for frac in [0.3, 0.5, 0.7, 1.0]:
                d1 = frac * d1_max
                d2 = frac * d2_max * 0.5
                combined = d1 * g1 + d2 * g2
                new_max = float(np.max(corr + combined))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_move = (i1, j1, d1, i2, j2, d2)

        if best_move is not None and best_improve > 1e-13:
            i1, j1, d1, i2, j2, d2 = best_move
            h[i1] -= d1
            h[j1] += d1
            h[i2] -= d2
            h[j2] += d2
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 1500:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_random_init(n, rng):
    target = n / 2.0
    pattern = rng.choice([0, 1], size=n, p=[0.5, 0.5])
    for _ in range(5):
        pos = rng.randint(5, n - 5)
        width = rng.randint(3, min(20, n // 10))
        if rng.random() < 0.5:
            pattern[max(0, pos - width):pos + width + 1] = np.clip(
                pattern[max(0, pos - width):pos + width + 1] + 1, 0, 1)
        else:
            pattern[max(0, pos - width):pos + width + 1] = np.clip(
                pattern[max(0, pos - width):pos + width + 1] - 1, 0, 1)
    h = pattern + rng.randn(n) * 0.1
    return project_feasible(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B to polish initial
    if tleft() > 100:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[100, 500, 2000],
                            maxiter=300, time_limit=min(50, tleft() * 0.15),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Main pair CD with single-shift gradient
    if tleft() > 300:
        cd_time = min(400, tleft() * 0.3)
        hcd, c5cd = pair_cd_v2(best_h, n, target, time_limit=cd_time,
                               n_cand=20, n_best_low=12, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: L-BFGS-B intermediate
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[2000, 10000, 50000],
                            maxiter=300, time_limit=min(60, tleft() * 0.2),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Pair CD with multi-shift gradient
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.25)
        hcd, c5cd = pair_cd_v2(best_h, n, target, time_limit=cd_time,
                               n_cand=25, n_best_low=10, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Triple CD
    if tleft() > 100:
        tc_time = min(80, tleft() * 0.2)
        ht, c5t = triple_cd(best_h, n, target, time_limit=tc_time,
                            n_cand=12, n_perturb=8)
        if c5t < best_score:
            best_score = c5t
            best_h = ht

    # Phase 6: Subgradient
    if tleft() > 60:
        hsg, c5sg = subgradient_phase(best_h, n, target,
                                       time_limit=min(50, tleft() * 0.25),
                                       max_steps=50000, n_shifts=7, base_lr=0.06)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 7: L-BFGS-B high alpha
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[50000, 200000, 1000000],
                            maxiter=250, time_limit=min(70, tleft() * 0.3),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Pair CD final pass
    if tleft() > 150:
        cd_time = min(130, tleft() * 0.3)
        hcd, c5cd = pair_cd_v2(best_h, n, target, time_limit=cd_time,
                               n_cand=20, n_best_low=10, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Random restarts
    if tleft() > 200:
        rng = np.random.default_rng(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for restart in range(3):
            if tleft() < 80:
                break
            noise_scales = [0.03, 0.05, 0.08]
            h_pert = best_h + rng.randn(n_use) * noise_scales[restart]
            h_pert = project_feasible(h_pert, target_use)
            hp, sc = lbfgs_phase(h_pert, n_use, target_use,
                                alphas=[2000, 20000, 100000],
                                maxiter=200, time_limit=min(40, tleft() * 0.25),
                                penalty=500.0)
            if sc < best_score and tleft() > 60:
                hcd, c5cd = pair_cd_v2(hp, n_use, target_use, time_limit=min(40, tleft() * 0.3),
                                       n_cand=20, n_best_low=8, n_shifts=1)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd
            elif sc < best_score:
                best_score = sc
                best_h = hp

    # Phase 10: Higher resolution attempt
    if tleft() > 250:
        n_new = 1000
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                            alphas=[100, 1000, 10000, 50000],
                            maxiter=250, time_limit=min(100, tleft() * 0.3),
                            penalty=100.0)
        if tleft() > 80:
            hcd, c5cd = pair_cd_v2(hp, n_new, target_new, time_limit=min(80, tleft() * 0.35),
                                   n_cand=15, n_best_low=8, n_shifts=1)
            if c5cd < sc:
                sc = c5cd
                hp = hcd
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final L-BFGS-B
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[200000, 1000000, 5000000],
                            maxiter=150, time_limit=min(30, tleft() * 0.5),
                            penalty=10000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 12: Final pair CD
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_v2(best_h, n_use, target_use, time_limit=min(18, tleft() * 0.6),
                               n_cand=15, n_best_low=8, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
