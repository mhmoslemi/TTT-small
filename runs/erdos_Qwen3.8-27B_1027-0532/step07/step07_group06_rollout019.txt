        func = _pnorm_obj
    else:
        func = _smooth_lse

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    idxs = np.arange(n)
    grads = np.zeros((n_corr, n))
    for m in range(n_corr):
        k = m - mid
        if k >= 0:
            if n - k > 0:
                grads[m, k:n] += dx
                grads[m, :n-k] -= dx * h[k:]
        else:
            kk = -k
            if n - kk > 0:
                grads[m, :n-kk] += dx
                grads[m, kk:] -= dx * h[:n-kk]
    return corr, grads, dx, mid


def cutting_plane_lp(h_init, n, target, max_iters=100, time_limit=120, start_time=0,
                     active_frac=0.15, use_all_first=True):
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr, grads, dx_v, mid = compute_corr_and_grads(h)
        c_max = float(np.max(corr)) * dx_v

        if c_max < best_score - 1e-13:
            best_score = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            break

        threshold = c_max - active_frac * (c_max * 0.05 + 1e-8) / dx_v
        active_m = np.where(corr >= threshold)[0]

        if use_all_first and it == 0:
            active_m = np.arange(len(corr))

        n_active = len(active_m)
        if n_active == 0:
            active_m = np.array([np.argmax(corr)])
            n_active = 1

        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0

        A_rows = []
        b_rows = []

        for m in active_m:
            row = np.zeros(n + 1)
            row[:n] = grads[m]
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(float(np.dot(grads[m], h) - corr[m]))

        I_mat = np.eye(n)
        z1 = np.zeros(1)
        A_rows.append(np.hstack([I_mat, z1]))
        b_rows.append(np.ones(n))
        A_rows.append(np.hstack([-I_mat, z1]))
        b_rows.append(np.zeros(n))

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'presolve': True})
            if res.status == 0:
                h_lp = res.x[:n]
                step_vec = h_lp - h
                step_norm = np.linalg.norm(step_vec)
                if step_norm < 1e-12:
                    break
                accepted = False
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                    h_trial = h + alpha * step_vec
                    h_trial = project_feasible(h_trial, target)
                    sc = compute_c5(h_trial)
                    if sc <= c_max + 1e-12:
                        h = h_trial
                        accepted = True
                        break
                if not accepted:
                    active_frac = max(0.05, active_frac * 0.7)
            else:
                break
        except Exception:
            break

    return best_h, best_score


def subgradient_momentum(h_init, n, target, max_iters=500, time_limit=60, start_time=0,
                         mom=0.9, base_step=0.01):
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr, grads, dx_v, mid = compute_corr_and_grads(h)
        c_max = float(np.max(corr)) * dx_v

        if c_max < best_score - 1e-13:
            best_score = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            break

        active_thresh = c_max - 0.02 / dx_v
        active_m = np.where(corr >= active_thresh)[0]
        if len(active_m) == 0:
            active_m = np.array([np.argmax(corr)])

        weights = corr[active_m].astype(float)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(len(active_m))
            w_sum = len(active_m)
        weights = weights / w_sum

        g = np.zeros(n)
        for idx, w in enumerate(weights):
            g += w * grads[active_m[idx]]
        g -= np.mean(g)

        step_size = base_step / (1.0 + 0.01 * it)
        v = mom * v + (1.0 - mom) * g
        h_new = h - step_size * v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < c_max + 1e-12:
            h = h_new
        else:
            v *= 0.5

    return best_h, best_score


def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=30000, n_cand_start=15, n_cand_end=60,
            max_no_improve=400, n_top_shifts=1, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = weights.sum()
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = np.zeros(n_corr)
                for t_idx in range(n_corr):
                    k_t = t_idx - mid
                    val = 0.0
                    if 0 <= i - k_t < n and 0 <= (i - k_t) + k_t < n:
                        pass
                    v1 = (i - k_t >= 0) and (i - k_t < n)
                    v2 = (i + k_t >= 0) and (i + k_t < n)
                    if v1:
                        val += (1.0 - h[i - k_t])
                    if v2:
                        val -= h[i + k_t]
                    v1j = (j - k_t >= 0) and (j - k_t < n)
                    v2j = (j + k_t >= 0) and (j + k_t < n)
                    if v1j:
                        val -= (1.0 - h[j - k_t])
                    if v2j:
                        val += h[j + k_t]
                    g_vec[t_idx] = val * dx

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=20000, base_scale=0.03):
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 3000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 80:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

    return final_h, final_score, len(final_h)
