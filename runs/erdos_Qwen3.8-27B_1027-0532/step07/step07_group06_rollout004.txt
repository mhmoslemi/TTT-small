            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lse_objective, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def gradient_descent(h_init, n, target, time_limit, start_time,
                     max_iters=20000, n_top=5, lr_start=0.002, lr_end=0.0001):

        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break

        progress = it / max_iters
        lr = lr_start * (lr_end / lr_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_raw = float(np.max(corr))

        top_idx = np.argsort(corr)[-n_top:]
        top_corr = corr[top_idx]

        if wsum < 1e-15:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < best_c5 - 1e-14:
            best_c5 = sc

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:

                no_improve = 0

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:
        h5, sc5 = slsqp_optimize(best_h, n0, target0,
                                  [5000, 20000], 2000,
                                  start_time + min(30, remaining - 10), start_time)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        h6, sc6 = fast_pair_cd(best_h, n0, target0,
                               start_time + min(80, remaining - 10), start_time,
                               max_sweeps=40000, n_cand=35, max_no_improve=350,
                               n_top=5, bisection_iters=10)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        h7, sc7 = gradient_descent(best_h, n0, target0,
                                    start_time + min(60, remaining - 10), start_time,
                                    max_iters=30000, n_top=5,
                                    lr_start=0.001, lr_end=0.00005)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 40:
        h8, sc8 = stochastic_refine(best_h, n0, target0,
                                     start_time + min(40, remaining - 10), start_time,
                                     max_steps=20000, base_scale=0.02, seed=seed)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        rng = np.random.RandomState(seed + 1)
        for noise_scale in [0.02, 0.05, 0.1]:
            remaining = budget_s - 40.0 - (time.time() - start_time)
            if remaining < 30:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_r, sc_r = fast_pair_cd(h_pert, n0, target0,
                                     start_time + min(25, remaining - 10), start_time,
                                     max_sweeps=20000, n_cand=30, max_no_improve=300,
                                     n_top=3, bisection_iters=8)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:
        h_fin, sc_fin = slsqp_optimize(best_h, n0, target0,
                                        [10000, 50000], 2000,
                                        start_time + min(30, remaining - 10), start_time)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:
        h_fin2, sc_fin2 = fast_pair_cd(best_h, n0, target0,
                                       start_time + min(30, remaining - 10), start_time,
                                       max_sweeps=20000, n_cand=40, max_no_improve=200,
                                       n_top=5, bisection_iters=12)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
