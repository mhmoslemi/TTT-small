def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(x, target):
    """Project x onto {0<=h<=1, sum(h)=target} using bisection on Lagrange multiplier."""
    n = len(x)
    # Find t such that sum(clip(x+t, 0, 1)) = target
    lo, hi = -max(np.abs(x)) - 1, max(np.abs(x)) + 1
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5*(lo+hi), 0.0, 1.0)

def grad_Lk(h, k, n):
    """Gradient of L_k w.r.t. h, where k is the shift (can be negative)."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    v_pos = (idxs - k >= 0) & (idxs - k < n)
    v_neg = (idxs + k >= 0) & (idxs + k < n)
    grad[v_pos] += (1.0 - h[idxs[v_pos] - k])
    grad[v_neg] -= h[idxs[v_neg] + k]
    return grad

def compute_combined_gradient(h, n, corr, n_active=5, smoothness=0.1):
    """Compute weighted gradient of active L_k's."""
    c_max = np.max(corr)
    # Soft active set: weight = exp(-smoothness * (c_max - corr) / max(c_max, eps))
    eps = max(c_max * 1e-6, 1e-12)
    weights = np.exp(-smoothness * (c_max - corr) / eps)
    # Only keep top n_active
    top_idx = np.argsort(corr)[-n_active:]
    weights[:len(weights)] = 0
    weights[top_idx] = np.exp(-smoothness * (c_max - corr[top_idx]) / eps)
    if np.sum(weights) < 1e-15:
        weights[top_idx[-1]] = 1.0
    else:
        weights /= np.sum(weights)

    grad = np.zeros(n)
    idxs = np.arange(n)
    for s_idx in top_idx:

        w = weights[s_idx]
        v_pos = (idxs - k >= 0) & (idxs - k < n)
        v_neg = (idxs + k >= 0) & (idxs + k < n)
        grad[v_pos] += w * (1.0 - h[idxs[v_pos] - k])
        grad[v_neg] -= w * h[idxs[v_neg] + k]
    
    return grad

                n_active=5, smoothness=0.1, max_steps=50000):

    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * (2.0 / n)
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 2000:
                break
        
        # Compute gradient
        grad = compute_combined_gradient(h, n, corr, n_active, smoothness)
        grad -= np.mean(grad)  # preserve sum
        
        # Momentum update
        v = momentum * v - alpha0 * grad
        
        # Trial step with backtracking

        improved = False
        for _ in range(10):

            h_trial = project_box_sum(h_trial, target)

            v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def project_bisect(x, target):
    n = len(x)
    lo = -np.max(np.abs(x)) - 2.0
    hi = np.max(np.abs(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_opt(h_start, n, target, alphas, maxiter, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def cd_optimize(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=10, n_cand_end=60, max_no_improve=300,
                n_active=4, reheat_interval=100, reheat_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    sweeps_since = 0

    corr = compute_corr(h)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            sweeps_since = 0
        else:
            no_improve += 1
            sweeps_since += 1

        if no_improve > max_no_improve:
            break

        if sweeps_since > reheat_interval and sweeps_since % 30 == 0:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(3, n // 50)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                corr = compute_corr(h)
                c5_val = float(np.max(corr)) * dx
                sweeps_since = 0
                continue

        top_k = np.argsort(corr)[-n_active:]
        c_max = float(np.max(corr))

        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(top_k)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        k_star_idx = int(np.argmax(corr))
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(2 * n - 1)
                for s_idx in top_k:
                    k = int(s_idx) - mid
                    g[s_idx] = (1.0 - h[i + k]) * (0 <= i + k < n) + \
                               h[j - k] * (0 <= j - k < n) - \
                               (1.0 - h[j + k]) * (0 <= j + k < n) - \
                               h[i - k] * (0 <= i - k < n)
                # delta_i = -delta, delta_j = +delta
                # change in corr[s] = delta * (dL_s/dh[j] - dL_s/dh[i])
                # dL_s/dh[x] where s corresponds to shift k: (1-h[x-k])*[valid] - h[x+k]*[valid]

                if g[k_star_idx] > 0.2 * np.max(np.abs(g)):
                    continue

                a_d, b_d = delta_lo, delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] = np.clip(h[best_i] - best_delta, 0.0, 1.0)
            h[best_j] = np.clip(h[best_j] + best_delta, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = compute_corr(h)

    return best_h, best_c5


def momentum_pg(h_init, n, target, time_limit, alpha0=0.3, momentum_init=0.7,
                n_active=5, smooth_param=0.05, max_steps=100000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    momentum = momentum_init
    alpha = alpha0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if alpha < 1e-6:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            momentum = min(momentum * 1.05, 0.9)
            alpha = min(alpha * 1.02, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            momentum *= 0.99
            if stale > 3000:
                break

        top_k = np.argsort(corr)[-n_active:]
        c_max = float(np.max(corr))
        eps = max(c_max * 1e-9, 1e-12)
        active_weights = np.exp(-smooth_param * (c_max - corr[top_k]) / eps)
        if np.sum(active_weights) < 1e-15:
            active_weights = np.ones(n_active) / n_active
        else:
            active_weights /= np.sum(active_weights)

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, active_weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = momentum * v - alpha * grad
        h_trial = project_bisect(h + v, target)
        c5_trial = compute_c5(h_trial)

        if c5_trial < c5_val - 1e-15:
            h = h_trial
        else:
            v = np.zeros(n)
            h_trial = project_bisect(h - 0.1 * alpha * grad, target)
            h = h_trial

    return best_h, best_c5
