def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(corr, alpha):
    """Compute smooth max approximation and its gradient w.r.t. corr."""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    smooth_val = c_max + np.log(np.sum(exp_shifted)) / alpha
    return smooth_val, weights

def grad_h_from_weights(h, weights, n):
    """Compute gradient of smooth max corr w.r.t. h given weights on shifts."""
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-10:
            continue
        k = s - mid

def fista_smooth_max(h_init, n, target, alpha_seq, steps_per_alpha, time_limit):
    """FISTA optimization with increasing smooth max parameter."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy()
    h_prev = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alpha_seq:
        t = 1.0
        h_prev = h.copy()
        for step in range(steps_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5
            
            # Compute y (FISTA extrapolation)
            y = h + ((t - 1) / (t + 1)) * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)  # Keep in bounds for gradient eval
            
            # Compute smooth max and gradient
            corr = np.correlate(y, 1.0 - y, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_vals = np.exp(shifted)
            wsum = np.sum(exp_vals)
            if wsum < 1e-30:
                weights = np.zeros(len(corr))
                weights[int(np.argmax(corr))] = 1.0
            else:
                weights = exp_vals / wsum
            
            # Gradient w.r.t. h
            grad = np.zeros(n)
            for s in range(len(corr)):
                w = weights[s]
                if w < 1e-10:
                    continue
                k = s - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - y[idxs[v1] - k])
                grad[v2] -= w * y[idxs[v2] + k]
            
            # Step size (conservative)
            step_size = 1.0 / (2.0 * alpha * n + 4.0)
            
            # Gradient step
            z = y - step_size * grad
            
            # Project onto feasible set
            h_new = project_feasible(z, target)
            
            # Check improvement (use actual C5, not smooth)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h_prev = h
            h = h_new
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights = exp_vals / wsum
    return weights

def compute_grad(h, weights):
    n = len(h)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for s in active:
        k = int(s) - mid
        w = weights[s]
        if k >= 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            if n - kk > 0:
                grad[:n-kk] += w * (1.0 - h[kk:])
                grad[kk:] -= w * h[:n-kk]
    return grad

def fista(h_init, n, target, alpha_list, iters_per_alpha, time_start, time_limit):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alpha_list:
        t = 1.0
        h_prev = h.copy()
        for _ in range(iters_per_alpha):
            if time.time() - time_start > time_limit:
                return best_h, best_c5
            # FISTA extrapolation
            mom = (t - 1.0) / (t + 1.0)
            y = h + mom * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            
            # Gradient
            corr = np.correlate(y, 1.0 - y, mode='full')
            weights = get_weights(corr, alpha)
            grad = compute_grad(y, weights)
            
            # Step size: conservative estimate of 1/L
            # L ≈ 2*alpha (from the smooth max) + 2 (from the bilinear part)
            # But we also have the constraint that h is bounded, so effective L is smaller
            step = 1.0 / (alpha * 4.0 + 4.0)
            
            z = y - step * grad
            h_new = project_feasible(z, target)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h_prev = h
            h = h_new
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
    
    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    rng = np.random.RandomState(seed)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute the change matrix
    # change[p, s] = effect on corr[s] when h[p] increases by 1
    # = (1-h[p-k_s])*1{p-k_s valid} - h[p+k_s]*1{p+k_s valid}
    # where k_s = s - mid
    shifts = np.arange(-(n-1), n)
    
    def build_change_row(p):
        row = np.zeros(n_corr)
        for s in range(n_corr):
            k = s - mid
            val = 0.0
            if 0 <= p - k < n:
                val += (1.0 - h[p - k])
            if 0 <= p + k < n:
                val -= h[p + k]
            row[s] = val
        return row
    
    # Build full change matrix

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_single(h, k):
    """Gradient of corr at shift k w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if n > k:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        kk = -k
        if n > kk:
            grad[:n-kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n-kk]
    return grad

def fista_optimize(h_init, n, target, alpha_list, max_iters, time_start, time_limit):
    """FISTA with log-sum-exp smooth max."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    
    for alpha in alpha_list:
        t = 1.0
        h_prev = h.copy()
        
        for it in range(max_iters):
            if time.time() - time_start > time_limit:
                return best_h, best_c5
            
            # Nesterov extrapolation
            mom = (t - 1.0) / (t + 1.0)
            y = h + mom * (h - h_prev)
            y_clipped = np.clip(y, 0.0, 1.0)
            
            # Compute correlation and gradient
            corr = np.correlate(y_clipped, 1.0 - y_clipped, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_vals = np.exp(shifted)
            wsum = np.sum(exp_vals)
            
            if wsum < 1e-30:
                # Effectively the max
                grad = compute_grad_single(y_clipped, int(np.argmax(corr)) - mid)
            else:
                weights = exp_vals / wsum
                # Compute gradient (weighted sum)
                grad = np.zeros(n)
                active = np.where(weights > 1e-10)[0]
                for s in active:
                    k = int(s) - mid
                    w = weights[s]
                    if k >= 0:
                        if n > k:
                            grad[k:] += w * (1.0 - y_clipped[:n-k])
                            grad[:n-k] -= w * y_clipped[k:]
                    else:
                        kk = -k
                        if n > kk:
                            grad[:n-kk] += w * (1.0 - y_clipped[kk:])
                            grad[kk:] -= w * y_clipped[:n-kk]
            
            # Step size
            step = 1.0 / (2.0 * alpha + 4.0)
            
            # Gradient step + projection
            z = y_clipped - step * grad
            h_new = project_feasible(z, target)
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h_prev = h
            h = h_new
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
    
    return best_h, best_c5
