```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

```python
def lp_descent_direction(h, corr, n, verbose=False):
    """Find optimal descent direction via LP."""
    from scipy.optimize import linprog
    n_corr = 2 * n - 1
    # Build the change matrix
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n_corr, n))  # change[k, i] = d(corr_k)/d(h[i])
    for k in range(n_corr):
        s = k - (n - 1)
        idxs = np.arange(n)
        v1 = (idxs + s >= 0) & (idxs + s < n)
        v2 = (idxs - s >= 0) & (idxs - s < n)
        change[k, v1] += (1.0 - h[idxs[v1] + s])
        change[k, v2] -= h[idxs[v2] - s]
    
    # LP: min t
    # s.t. change[k] @ d - t ≤ -corr[k] for all k
    #      sum(d) = 0
    #      -1 ≤ d ≤ 1
    #      t free (but since we're minimizing, it'll be bounded)
    
    # Variables: x = [d_0, ..., d_{n-1}, t]
    n_vars = n + 1
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    # Inequality constraints: change[k] @ d - t ≤ -corr[k]
    A_ub = np.zeros((n_corr, n_vars))
    A_ub[:, :n] = change  # change[k, i] * d[i]
    A_ub[:, -1] = -1.0    # -1 * t
    b_ub = -corr
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 ≤ d_i ≤ 1, t free
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        d = result.x[:n]
        t_opt = result.x[-1]
        return d, t_opt
    else:
        return None, None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        change[k, v1] += (1.0 - h[idxs[v1] + s])
        change[k, v2] -= h[idxs[v2] - s]

            break

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(30, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        improvement = np.zeros(n_corr)
        n_updated = 0
        for _ in range(n_pairs):
            best_improv = float(np.max(corr + improvement)) * dx
            best_move = None
            n_h = len(high_set)
            n_l = len(low_set)
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue
                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(14):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + improvement + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = float(np.max(corr + improvement + delta_star * g)) * dx
                    if approx_max < best_improv:
                        best_improv = approx_max
                        best_move = (i, j, delta_star)
            if best_move is not None:
                i, j, delta = best_move
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                improvement += delta * (change[j] - change[i])
                change[i, :] = 0.0
                change[j, :] = 0.0
                pos_i = (i + shifts >= 0) & (i + shifts < n)
                neg_i = (i - shifts >= 0) & (i - shifts < n)
                change[i, pos_i] += (1.0 - h[i + shifts[pos_i]])
                change[i, neg_i] -= h[i - shifts[neg_i]]
                pos_j = (j + shifts >= 0) & (j + shifts < n)
                neg_j = (j - shifts >= 0) & (j - shifts < n)
                change[j, pos_j] += (1.0 - h[j + shifts[pos_j]])
                change[j, neg_j] -= h[j - shifts[neg_j]]
                n_updated += 1

        if n_updated == 0:
            consecutive_no_improve += 5

    return best_h, best_c5

def nesterov_smooth(h_init, n, target, time_limit, alpha=5000, momentum=0.9,
                    step_init=0.01, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum_val = momentum
    step = step_init
    y = h.copy()
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr_y)
        shifted = alpha * (corr_y - c_max)
        shifted = np.clip(shifted, -50.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-30:
            p = np.ones(n_corr) / n_corr
        else:
            p = p / p_sum

        grad = np.zeros(n)
        top_k = min(50, n_corr)
        top_probs = np.argsort(p)[-top_k:]
        for s_idx in top_probs:
            wi = p[s_idx]
            if wi < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = y - step * grad
        h_new = project_feasible(h_new, target)

        momentum_new = 0.5 * (1.0 + momentum_val)
        y = h_new + momentum_new * (h_new - h)
        y = np.clip(y, 0.0, 1.0)

        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
            step = min(step * 1.02, 0.1)
        else:
            no_improve += 1
            if no_improve > 200:
                step *= 0.9
                momentum_val = max(0.5, momentum_val * 0.99)

        h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_lp, sc_lp = lp_steepest_descent(best_h, None, n0,
                                          max_iters=300, time_limit=min(150, tleft() * 0.5),
                                          active_fraction=0.3)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    if tleft() > 200:
        h_cd, c5_cd = weighted_pair_cd(best_h, n0, target0,
                                       time_limit=min(150, tleft() * 0.5),
                                       max_sweeps=30000,
                                       n_cand_start=20, n_cand_end=70,
                                       max_no_improve=400,
                                       weight_beta=5.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        h_mp, c5_mp = multi_pair_cd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.5),
                                    max_sweeps=5000,
                                    n_cand=40, n_pairs=5,
                                    weight_beta=6.0, max_no_improve=200)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    if tleft() > 100:
        h_lp2, sc_lp2 = lp_steepest_descent(best_h, None, n0,
                                            max_iters=300, time_limit=min(80, tleft() * 0.5),
                                            active_fraction=0.2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_h = h_lp2

    if tleft() > 80:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [5000, 20000], 500, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 200:
        n1 = 1000
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 150:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [100, 500, 2000], 400, 1e-14)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
            else:
                best_h = h_pol1

        if tleft() > 100:
            h_cd1, c5_cd1 = weighted_pair_cd(best_h, n1, target1,
                                             time_limit=min(80, tleft() * 0.5),
                                             max_sweeps=20000,
                                             n_cand_start=20, n_cand_end=60,
                                             max_no_improve=300,
                                             weight_beta=6.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 80:
            h_lp1, sc_lp1 = lp_steepest_descent(best_h, None, n1,
                                                max_iters=200, time_limit=min(60, tleft() * 0.5),
                                                active_fraction=0.3)
            if sc_lp1 < best_score:
                best_score = sc_lp1
                best_h = h_lp1

        if tleft() > 50:
            h_pol1b, sc1b = slsqp_round(best_h, n1, target1,
                                        [10000, 50000], 400, 1e-14)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        n_hav = np.linspace(0, 2 * np.pi, 5)
        for freq in n_hav:
            if tleft() < 60:
                break
            t = np.linspace(0, 2, n_cur)
            pert = np.sin(freq * t * np.pi / 2.0) * 0.05
            h_pert = best_h + pert
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = slsqp_round(h_pert, n_cur, target_cur,
                                        [200, 2000, 10000], 300, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 40:
                h_cd_ref, c5_cd_ref = weighted_pair_cd(best_h, n_cur, target_cur,
                                                       time_limit=min(30, tleft() * 0.4),
                                                       max_sweeps=8000,
                                                       n_cand_start=20, n_cand_end=50,
                                                       max_no_improve=150,
                                                       weight_beta=5.0)
                if c5_cd_ref < best_score:

                    best_h = h_cd_ref

    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_nt, c5_nt = nesterov_smooth(best_h, n_cur, target_cur,
                                      time_limit=min(60, tleft() * 0.6),
                                      alpha=5000, momentum=0.9,
                                      step_init=0.005, max_iter=10000)
        if c5_nt < best_score:
            best_score = c5_nt
            best_h = h_nt

    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd_fin, c5_fin = weighted_pair_cd(best_h, n_cur, target_cur,
                                            time_limit=min(45, tleft() * 0.6),
                                            max_sweeps=15000,
                                            n_cand_start=30, n_cand_end=70,
                                            max_no_improve=200,
                                            weight_beta=8.0)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc_final = slsqp_round(best_h, n_cur, target_cur,
                                        [50000, 100000], 400, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
