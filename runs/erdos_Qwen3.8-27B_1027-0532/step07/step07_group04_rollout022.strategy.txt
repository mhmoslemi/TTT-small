   ```python
                             jac=grad_objective, options={'maxiter': 1000, 'ftol': 1e-8})
           if result.success:
               h_med = result.x
       except:
           pass

       best_score = min(best_score, objective(h_med))
       best_h = h_med  # Update with refined solution

       # Final SA on n=500 (local refinement)
       t_initial = 0.2
       t_final = 0.0003
       steps = 8000
       t_current = t_initial
       h_final = h_med.copy()

       for i in range(steps):
           if time.time() - start_time > budget_s:  # Total budget check
               break
           # Small random perturbation around current point
           perturb = np.random.randn(len(h_final)) * t_current * 0.3
           h_new = np.clip(h_final + perturb, 0, 1)
           # Repair sum
           diff_sum = target500 - np.sum(h_new)
           h_new += diff_sum / len(h_new)
           score_new = objective(h_new)

           # Metropolis acceptance
           diff = score_new - best_score
           if diff < 0 or np.log(np.random.rand()) < diff / t_current:
               h_final = h_new
           if score_new < best_score:
               best_score = score_new
           t_current *= 0.998

       # Project to parent dimensions if needed
       final_h = h_final

       return final_h, best_score, n
   ```
