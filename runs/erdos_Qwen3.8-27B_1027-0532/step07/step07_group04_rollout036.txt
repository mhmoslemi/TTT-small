    shifted = np.zeros(len(corr))
    shifted[top_idx] = alpha * (corr[top_idx] - c_max)
    # For non-top, they're very negative, so exp is ~0
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    total = weights.sum()
    weights /= total
    
    obj = c_max + np.log(total) / alpha
    
    grad = np.zeros(n)
    idx = np.arange(n)
    mid = n - 1
    for k_idx in top_idx:
        w = weights[k_idx]
        if abs(w) < 1e-14:
            continue
        k = k_idx - mid
        pos = idx + k
        valid_pos = (pos >= 0) & (pos < n)
        neg = idx - k
        valid_neg = (neg >= 0) & (neg < n)
        grad[valid_pos] += w * (1.0 - h[pos[valid_pos]])
        grad[valid_neg] -= w * h[neg[valid_neg]]
    
    return obj, grad, corr

**Key mathematical insight**: The problem is equivalent to finding a function h that minimizes the maximum "auto-cross-correlation" with its complement. This is related to the theory of "optimal binary sequences" and "Golay complementary sequences."

For a binary sequence (h ∈ {0,1}), the problem becomes: find a binary sequence of length n with exactly n/2 ones such that the maximum cross-correlation between the sequence and its complement (over all shifts) is minimized.

This is a combinatorial optimization problem. For continuous h, we get a relaxation.

One approach from signal processing: use the "periodic autocorrelation" spectrum. The sum of squared autocorrelation values is related to the "energy" of the sequence (Parseval's theorem). If we could make the autocorrelation as flat as possible (except at zero lag), we'd minimize the maximum.

r(k) = sum_i h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]

The sum over all k of r(k)^2 can be computed via FFT. Minimizing this sum (rather than the max) might be a good relaxation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_grad_trunc(h, alpha, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_idx = np.argsort(corr)[-k_top:]
    
    shifted = np.zeros(len(corr))
    shifted[top_idx] = alpha * (corr[top_idx] - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    obj = c_max + np.log(total) / alpha
    
    grad = np.zeros(n)
    idx = np.arange(n)
    mid = n - 1
    for k_idx in top_idx:
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - mid
        pos = idx + k
        valid_pos = (pos >= 0) & (pos < n)
        neg = idx - k
        valid_neg = (neg >= 0) & (neg < n)
        grad[valid_pos] += w * (1.0 - h[pos[valid_pos]])
        grad[valid_neg] -= w * h[neg[valid_neg]]
    
    return obj, grad

def adam_run(h_init, n, target, alpha_schedule, steps, lr_init, lr_end, 
             k_top, time_limit, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    total_steps = sum(s for _, s in alpha_schedule)
    step_count = 0
    
    for alpha, n_steps in alpha_schedule:
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        for step in range(n_steps):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            t += 1
            step_count += 1
            
            progress = step_count / total_steps
            lr = lr_init * (lr_end / lr_init) ** progress
            
            obj, grad = compute_c5_grad_trunc(h, alpha, k_top)
            
            beta1, beta2, eps = 0.85, 0.999, 1e-8
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            h = h_new
            
            if step % 200 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    sc = compute_c5(h)
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def cd_refine(h_init, n, target, time_limit, max_sweeps=50000,

                    best_delta = delta_star

            h[best_i] -= best_delta

            h = np.clip(h, 0.0, 1.0)

    return best_h, best_c5

    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    
    def obj_fun(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(lambda x, a=alpha: obj_fun(x, a), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _slsqp_cons_fun(x):
    return np.sum(x) - _g_target

def _slsqp_cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                   time_limit=None, start_time=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_grad_trunc(h, alpha, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_idx = np.argsort(corr)[-k_top:]
    shifted = np.zeros(len(corr))
    shifted[top_idx] = alpha * (corr[top_idx] - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    mid = n - 1
    for k_idx in top_idx:
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - mid
        pos = idx + k
        vp = (pos >= 0) & (pos < n)
        neg = idx - k
        vn = (neg >= 0) & (neg < n)
        grad[vp] += w * (1.0 - h[pos[vp]])
        grad[vn] -= w * h[neg[vn]]
    obj = c_max + np.log(total) / alpha
    return obj, grad

def adam_optimize(h_init, n, target, alpha_schedule, lr_init, lr_end,
                  k_top, time_limit, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    total_steps = sum(s for _, s in alpha_schedule)
    step_count = 0
    for alpha, n_steps in alpha_schedule:
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        for step in range(n_steps):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            t += 1
            step_count += 1
            progress = step_count / max(total_steps, 1)
            lr = lr_init * (lr_end / lr_init) ** progress
            obj, grad = compute_grad_trunc(h, alpha, k_top)
            beta1, beta2, eps = 0.85, 0.999, 1e-8
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            h = h_new
            if step % 300 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    sc = compute_c5(h)
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=float)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask].astype(int)])
        change[p, neg_mask] -= h[p - shifts[neg_mask].astype(int)]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask].astype(int)])
    change[p, neg_mask] -= h[p - shifts[neg_mask].astype(int)]
    return change

def cd_optimize(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=30, n_cand_end=90, max_no_improve=300):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=float)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def nesterov_optimize(h_init, n, target, alpha_schedule, lr_init, lr_end,
                      k_top, time_limit, start_time):

    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    total_steps = sum(s for _, s in alpha_schedule)
    step_count = 0
    beta = 0.9
    for alpha, n_steps in alpha_schedule:
        for step in range(n_steps):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            step_count += 1
            progress = step_count / max(total_steps, 1)
            lr = lr_init * (lr_end / lr_init) ** progress
            y = h + beta * v
            obj, grad = compute_grad_trunc(y, alpha, k_top)
            v_new = -lr * grad
            h_new = y + ((np.sqrt(beta) - 1.0) / (np.sqrt(1.0 + beta) - 1.0)) * (v_new - v)
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            v = h_new - y
            h = h_new
            if step % 200 == 0:

                if sc < best_score:

                    best_h = h.copy()
    sc = compute_c5(h)
    if sc < best_score:

        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
