           grad[wm] -= w * x[idxs[wm] - k]
       if _g_penalty > 0:
           grad += 2.0 * _g_penalty * (np.sum(x) - _g_target)
       return grad

   def lbfhsb_optimize(h_start, n, target, alphas, maxiter=300, penalty=1e6):
       global _g_n, _g_target, _g_alpha, _g_penalty
       _g_n = n
       _g_target = target
       _g_penalty = penalty
       bounds = [(0.0, 1.0)] * n
       h_cur = h_start.copy()
       cur_score = compute_c5(h_cur)
       for alpha in alphas:
           _g_alpha = alpha
           try:
               res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxiter, 'gtol': 1e-14,
                                       'ftol': 1e-18, 'disp': False})
               h_new = project_feasible(res.x, target)
               sc = compute_c5(h_new)
               if sc < cur_score - 1e-15:
                   cur_score = sc
                   h_cur = h_new
           except Exception:
               pass
       return h_cur, cur_score

   def compute_change_matrix(h, n):
       n_corr = 2 * n - 1
       shifts = np.arange(-(n - 1), n)
       change = np.zeros((n, n_corr))
       for p in range(n):
           pos_mask = (p + shifts >= 0) & (p + shifts < n)
           neg_mask = (p - shifts >= 0) & (p - shifts < n)
           change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
           change[p, neg_mask] -= h[p - shifts[neg_mask]]
       return change

   def update_change_row(change, p, h, n, shifts):
       change[p, :] = 0.0
       pos_mask = (p + shifts >= 0) & (p + shifts < n)
       neg_mask = (p - shifts >= 0) & (p - shifts < n)
       change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
       change[p, neg_mask] -= h[p - shifts[neg_mask]]
       return change

   def coordinate_descent(h_init, n, target, time_limit, max_sweeps=50000,

       start = time.time()

       mid = n - 1

       best_c5 = compute_c5(h)
       best_h = h.copy()
       consecutive_no_improve = 0

           if time.time() - start > time_limit:
               break

           c5_val = float(np.max(corr)) * dx

           if c5_val < best_c5 - 1e-14:
               best_c5 = c5_val
               best_h = h.copy()
               consecutive_no_improve = 0
           else:
               consecutive_no_improve += 1

           if consecutive_no_improve > max_no_improve:
               break

           k_star = k_star_idx - mid

           grad[v2] -= h[idxs[v2] + k_star]
