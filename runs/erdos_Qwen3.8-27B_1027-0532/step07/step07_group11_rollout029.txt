```python
def build_change_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        # change[p, k] = (1-h[p+k]) * [0<=p+k<n] - h[p-k] * [0<=p-k<n]
        pk = p + shifts  # array of n_corr values
        pmk = p - shifts
        valid_pk = (pk >= 0) & (pk < n)
        valid_pmk = (pmk >= 0) & (pmk < n)
        change[p, valid_pk] += (1.0 - h[pk[valid_pk]])
        change[p, valid_pmk] -= h[pmk[valid_pmk]]
    return change
```

The key insight: we want to find δ that minimizes the maximum. Since f(δ) = max_k (corr[k] + δ*g[k]) is convex in δ, we can use a simple approach:

2. If g[k*] < 0, then increasing δ helps (decreases the active constraint)
3. The optimal δ is either where the active constraint is "released" (another constraint becomes active at the same level) or at the boundary

A simple heuristic: 
- If g[k*] < 0, try δ = min(h[i], 1-h[j]) (full move) or a fraction thereof

```python
def find_optimal_delta(corr, g, delta_max, n_iter=10):
    a, b = 0.0, delta_max
    # f(delta) = max(corr + delta * g)
    # Since f is convex, binary search for where f'(delta) = 0
    # f'(delta) = max{g[k] : corr[k] + delta*g[k] = f(delta)}
    # At the minimum, f'(delta) changes from negative to positive
    
    if b <= 0:
        return 0.0
    
    f_a = np.max(corr)  # f(0)
    f_b = np.max(corr + b * g)  # f(b)
    
    if f_b >= f_a:
        # f is not decreasing over [0, b], minimum is at 0 or in between
        # Try a smaller step
        b = b * 0.5
        if b <= 1e-15:
            return 0.0
    
    for _ in range(n_iter):
        mid = 0.5 * (a + b)
        f_mid = np.max(corr + mid * g)
        # Check if f is decreasing at mid
        active = (corr + mid * g) >= f_mid - 1e-14
        if active.any():
            f_prime = np.max(g[active])
        else:
            f_prime = 0.0
        
        if f_prime < -1e-12:
            a = mid  # f is still decreasing, go right
        elif f_prime > 1e-12:
            b = mid  # f is increasing, go left
        else:
            return mid  # at the minimum
    
    return 0.5 * (a + b)
```

Alternatively, I can use a **gradient projection** method instead of SLSQP:
1. Compute gradient of smooth objective
2. Take a step in negative gradient direction
3. Project onto feasible set (capped simplex)

The projection onto {h : 0≤h≤1, sum(h)=target} is:
1. Start with x = h - step * grad
2. Find τ such that sum(clip(x - τ, 0, 1)) = target
3. Return clip(x - τ, 0, 1)

This can be done in O(n log n) time by sorting.

Actually, the "gradient projection" method (also known as the "projected gradient method" or "Fischer's method") converges slowly for non-smooth objectives. The log-sum-exp smoothing helps, but the convergence rate depends on the smoothness parameter α.

A better approach: **accelerated projected gradient** (Nesterov's method). This gives O(1/k²) convergence for smooth convex problems. Our problem is non-convex, but locally it might work well.

Let me implement:
1. Nesterov accelerated gradient with log-sum-exp smoothing
2. Step size: L = Lipschitz constant of the gradient (or adaptive)
3. Projection at each step

For the log-sum-exp objective:
f(h) = max_k corr_k + (1/α) log(sum_k exp(α * corr_k - α * max_k corr_k))

The gradient:
∂f/∂h_i = (1/α) * sum_k exp(α * corr_k - α * max_k) * (∂corr_k/∂h_i)

And ∂corr_k/∂h_i is the (i, k-(n-1)) entry of the "correlation gradient" which I can precompute.

Actually, corr_k = sum_j h[j] * (1-h[j+k-something])... this is getting complicated. Let me just use the change matrix.

∂corr_k/∂h_i = change[i, k] (from the change matrix definition)

So the gradient of the smooth objective is:
∇f = (1/α) * sum_k w_k * change[:, k]

where w_k = exp(α * corr_k - α * max_k) / sum_j exp(α * corr_j - α * max_j)

This is a weighted sum of columns of the change matrix, which I can compute in O(n * n_corr).

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_change = None


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-13
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_change_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pk = p + shifts
        pmk = p - shifts
        valid_pk = (pk >= 0) & (pk < n)
        valid_pmk = (pmk >= 0) & (pmk < n)
        change[p, valid_pk] += (1.0 - h[pk[valid_pk]])
        change[p, valid_pmk] -= h[pmk[valid_pmk]]
    return change


def update_change_row(change, p, h, n, shifts):
    pk = p + shifts
    pmk = p - shifts
    change[p, :] = 0.0
    valid_pk = (pk >= 0) & (pk < n)
    valid_pmk = (pmk >= 0) & (pmk < n)
    change[p, valid_pk] += (1.0 - h[pk[valid_pk]])
    change[p, valid_pmk] -= h[pmk[valid_pmk]]


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        global _g_alpha
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def fast_vectorized_cd(h_init, n, target, time_limit, max_sweeps=50000,
                       k_start=20, k_end=80, n_shifts=5, max_no_improve=800):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h)
    changes_since_rebuild = 0

    fracs = np.array([0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0])

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        sw = corr[top_shifts]
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw = np.ones(n_shifts) / n_shifts
        else:
            sw = sw / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        cur_k = int(k_start + (k_end - k_start) * progress)
        cur_k = min(cur_k, n // 3)
        cur_k = max(cur_k, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_k]
        high_set = sorted_idx[-cur_k:]

        h_low = h[low_set]
        h_high = h[high_set]
        delta_max = np.minimum(h_high[:, np.newaxis], 1.0 - h_low[np.newaxis, :])

        neg_mask = delta_max < 1e-12
        delta_max[neg_mask] = 0.0

        G = change[high_set][:, np.newaxis, :] - change[low_set][np.newaxis, :, :]

        best_score = c5_val
        best_ii = -1
        best_jj = -1
        best_frac_idx = 0

        for fi in range(len(fracs)):
            f = fracs[fi]
            trial = corr[np.newaxis, np.newaxis, :] + f * delta_max * G
            scores = np.max(trial, axis=2) * dx
            scores[neg_mask] = best_score + 1.0

            idx = np.argmin(scores)
            s_val = scores.flat[idx]
            if s_val < best_score - 1e-14:
                best_score = s_val
                best_ii = idx // cur_k
                best_jj = idx % cur_k
                best_frac_idx = fi

        if best_ii >= 0:
            i = int(high_set[best_ii])
            j = int(low_set[best_jj])
            delta = fracs[best_frac_idx] * min(h[i], 1.0 - h[j])
            if delta > 1e-12:
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                changes_since_rebuild += 2
                if changes_since_rebuild > n // 3:
                    change = build_change_matrix(h)
                    changes_since_rebuild = 0

    return best_h, best_c5

def gradient_descent_smooth(h_init, n, target, time_limit, alpha=300,
                            n_steps=30000, step_init=0.01):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    h_yn = h.copy()
    t_val = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = step_init

    change = build_change_matrix(h)
    rebuild_counter = 0

    for s in range(n_steps):
        if time.time() - start > time_limit:
            break

        if rebuild_counter >= 5 or s == 0:
            change = build_change_matrix(h)
            rebuild_counter = 0
        rebuild_counter += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)

        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()

        grad = change.T @ weights
        grad -= np.mean(grad)

        h_trial = h - step * grad
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)

        if sc < compute_c5(h) - 1e-15:
            h_yn = h
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            h_inc = h_trial + ((t_val - 1.0) / t_new) * (h_trial - h_yn)
            h = project_feasible(h_inc, target)
            h = project_feasible(h, target)
            t_val = t_new
            step = min(step * 1.01, 0.1)
        else:
            step *= 0.8

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def random_perturb_reoptimize(h_base, n, target, rng, scales, time_per=30):
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    for scale in scales:
        if time_per <= 0:
            break
        h_pert = h_base + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()
    return best_h, best_c5

    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: SLSQP quick polish at n=400
    if tleft() > 60:
        alloc = min(55, tleft() * 0.4)
        h1, sc1 = slsqp_optimize(best_h, cur_n, cur_target, [100, 400],
                                  int(alloc * 3))
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Fast vectorized CD at n=400
    if tleft() > 180:
        alloc = min(180, tleft() * 0.5)
        h2, sc2 = multi_shift_cd(best_h, cur_n, cur_target,
                                  time_limit=alloc, max_sweeps=30000,
                                  k_start=30, k_end=100, n_shifts=8,
                                  max_no_improve=1000, n_fracs=12)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: SLSQP polish
    if tleft() > 50:
        alloc = min(45, tleft() * 0.4)
        h3, sc3 = slsqp_optimize(best_h, cur_n, cur_target, [200, 800],
                                  int(alloc * 4))
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Interpolate to n=800 and optimize
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h_inter = interpolate_to(best_h, n1, target1)
        sc_inter = compute_c5(h_inter)

        if tleft() > 250:
            alloc_cd = min(220, tleft() * 0.5)
            h4, sc4 = multi_shift_cd(h_inter, n1, target1,
                                      time_limit=alloc_cd, max_sweeps=30000,
                                      k_start=25, k_end=80, n_shifts=6,
                                      max_no_improve=800, n_fracs=10)
            if sc4 < best_score:
                best_score = sc4
                best_h = h4
                cur_n = n1
                cur_target = target1
            elif sc_inter < best_score:
                best_score = sc_inter
                best_h = h_inter
                cur_n = n1
                cur_target = target1

    # Phase 5: Gradient descent at current resolution
    if tleft() > 120:
        alloc = min(100, tleft() * 0.5)
        h5, sc5 = gradient_descent_smooth(best_h, cur_n, cur_target,
                                           time_limit=alloc, alpha=300,
                                           n_steps=20000, step_init=0.008)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: More CD (different parameters)
    if tleft() > 150:
        alloc = min(130, tleft() * 0.5)
        h6, sc6 = fast_vectorized_cd(best_h, cur_n, cur_target,
                                      time_limit=alloc, max_sweeps=40000,
                                      k_start=20, k_end=60, n_shifts=4,
                                      max_no_improve=600)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: SLSQP polish
    if tleft() > 50:
        alloc = min(45, tleft() * 0.4)
        h7, sc7 = slsqp_optimize(best_h, cur_n, cur_target, [500, 1500],
                                  int(alloc * 3))
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Try higher resolution if time allows
    if tleft() > 300 and cur_n < 1200:
        n2 = 1200
        target2 = n2 / 2.0
        h_inter2 = interpolate_to(best_h, n2, target2)
        sc_inter2 = compute_c5(h_inter2)

        if tleft() > 200:
            alloc_cd2 = min(180, tleft() * 0.5)
            h8, sc8 = fast_vectorized_cd(h_inter2, n2, target2,
                                          time_limit=alloc_cd2, max_sweeps=30000,
                                          k_start=20, k_end=60, n_shifts=4,
                                          max_no_improve=500)
            if sc8 < best_score:
                best_score = sc8
                best_h = h8
                cur_n = n2
                cur_target = target2

    # Phase 9: Random perturbation + reoptimize
    if tleft() > 80:
        for ns in [0.03, 0.06, 0.1]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(cur_n) * ns
            h_pert = project_feasible(h_pert, cur_target)
            sc_p = compute_c5(h_pert)
            if sc_p < best_score:
                best_score = sc_p
                best_h = h_pert
            if tleft() > 40:
                h_ref, sc_ref = slsqp_optimize(h_pert, cur_n, cur_target,
                                                 [300, 1000], 200)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 10: Final CD
    if tleft() > 60:
        alloc_final = min(50, tleft() * 0.7)
        h9, sc9 = multi_shift_cd(best_h, cur_n, cur_target,
                                  time_limit=alloc_final, max_sweeps=20000,
                                  k_start=40, k_end=120, n_shifts=6,
                                  max_no_improve=300, n_fracs=15)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 11: Final SLSQP
    if tleft() > 35:
        h_final, sc_final = slsqp_optimize(best_h, cur_n, cur_target,
                                            [2000], int(min(30, tleft() * 0.8) * 3))
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
