import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

        return np.zeros(n)

    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,

            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new
        except Exception:
            pass

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def paired_cd(h_init, n, target, time_limit, max_sweeps=60000,
              n_cand_start=30, n_cand_end=90, top_k_eval=60,
              n_top_shifts=3, max_no_improve=1000, n_bisect=15):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shift_ids = np.argsort(corr)[-n_top_shifts:]
        k_star_idx = int(top_shift_ids[-1])

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        grad = np.zeros(n)
        for s_idx in top_shift_ids:
            w = corr[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def jump_cd(h_init, n, target, time_limit, max_sweeps=40000,
            n_cand=60, top_k_eval=50, n_top_shifts=3,
            max_no_improve=600, jump_interval=300, jump_size=0.04,
            seed=99):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if no_improve > 0 and no_improve % jump_interval == 0:
            n_pert = rng.randint(3, 10)
            n_pert = min(n_pert, n)
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            jumps = rng.randn(n_pert) * jump_size
            h[idxs_pert] += jumps
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in idxs_pert:
                update_change_row(change, h, int(p), shifts, n)
            no_improve = 0

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shift_ids = np.argsort(corr)[-n_top_shifts:]
        k_star_idx = int(top_shift_ids[-1])

        grad = np.zeros(n)
        for s_idx in top_shift_ids:
            w = corr[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_init(n, K, rng, target):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.ones(n) * 0.5
    for k in range(1, K + 1):
        a = rng.uniform(-0.8, 0.8)
        b = rng.uniform(-0.8, 0.8)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = project_feasible(h, target)
    return h


def subgradient_nesterov(h_init, n, target, time_limit, max_iter=5000,
                         momentum=0.9, step_init=0.01):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    step = step_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step *= 0.9

        top_k = 5
        top_shift_ids = np.argsort(corr)[-top_k:]
        weights = np.zeros(n_corr)
        for s_idx in top_shift_ids:
            weights[s_idx] = corr[s_idx]
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights /= w_sum

        grad = np.zeros(n)
        for s_idx in top_shift_ids:
            w = weights[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new < c5_val * 1.001:
            h = h_new
        else:
            v *= 0.5
            h_new = h + v
            h_new = project_feasible(h_new, target)
            h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at n=400
    if time_left() > 50:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [200, 1000, 5000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Go to n=800
    if time_left() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        # CD round 1 at n=800
        tl = time_left()
        if tl > 150:
            h_cd1, c5_cd1 = paired_cd(h1, n1, target1,
                                      time_limit=min(130, tl * 0.7),
                                      max_sweeps=100000,
                                      n_cand_start=35, n_cand_end=90,
                                      top_k_eval=70, n_top_shifts=3,
                                      max_no_improve=1200, n_bisect=15)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        # SLSQP at n=800
        tl = time_left()
        if tl > 60:
            h_pol2, sc2 = slsqp_optimize(best_h, n1, target1,
                                         [2000, 10000, 30000], 500, 1e-14)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

        # CD round 2
        tl = time_left()
        if tl > 120:
            h_cd2, c5_cd2 = paired_cd(best_h, n1, target1,
                                      time_limit=min(100, tl * 0.7),
                                      max_sweeps=80000,
                                      n_cand_start=45, n_cand_end=100,
                                      top_k_eval=80, n_top_shifts=4,
                                      max_no_improve=1000, n_bisect=15)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        # Subgradient at n=800
        tl = time_left()
        if tl > 80:
            h_sub, c5_sub = subgradient_nesterov(best_h, n1, target1,
                                                 time_limit=min(60, tl * 0.6),
                                                 max_iter=3000,
                                                 momentum=0.85, step_init=0.005)
            if c5_sub < best_score:
                best_score = c5_sub
                best_h = h_sub

        # SLSQP again
        tl = time_left()
        if tl > 50:
            h_pol3, sc3 = slsqp_optimize(best_h, n1, target1,
                                         [10000, 50000], 400, 1e-14)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 3: Jump CD at n=800
    if time_left() > 100:
        tl = time_left()
        n1 = len(best_h)
        target1 = n1 / 2.0
        h_jump, c5_jump = jump_cd(best_h, n1, target1,
                                  time_limit=min(80, tl * 0.7),
                                  max_sweeps=50000, n_cand=70,
                                  top_k_eval=60, n_top_shifts=3,
                                  max_no_improve=800, jump_interval=200,
                                  jump_size=0.03, seed=42)
        if c5_jump < best_score:
            best_score = c5_jump
            best_h = h_jump

        tl = time_left()
        if tl > 40:
            h_pol4, sc4 = slsqp_optimize(best_h, n1, target1,
                                         [20000, 80000], 400, 1e-14)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_pol4

    # Phase 4: Try n=1200
    if time_left() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = time_left()
        if tl > 180:
            h_cd4, c5_cd4 = paired_cd(h2, n2, target2,
                                      time_limit=min(150, tl * 0.7),
                                      max_sweeps=60000,
                                      n_cand_start=35, n_cand_end=80,
                                      top_k_eval=60, n_top_shifts=3,
                                      max_no_improve=800, n_bisect=15)
            if c5_cd4 < best_score:
                best_score = c5_cd4
                best_h = h_cd4

        tl = time_left()
        if tl > 60:
            h_pol5, sc5 = slsqp_optimize(best_h, n2, target2,
                                         [5000, 20000], 400, 1e-14)
            if sc5 < best_score:
                best_score = sc5
                best_h = h_pol5

    # Phase 5: Fourier multi-start
    if time_left() > 180:
        n_try = 800
        target_try = n_try / 2.0
        for trial in range(4):
            tl = time_left()
            if tl < 60:
                break
            h_four = fourier_init(n_try, K=12, rng=rng, target=target_try)
            h_fcd, c5_fcd = paired_cd(h_four, n_try, target_try,
                                      time_limit=min(50, tl * 0.4),
                                      max_sweeps=25000,
                                      n_cand_start=40, n_cand_end=80,
                                      top_k_eval=50, n_top_shifts=3,
                                      max_no_improve=500, n_bisect=14)
            tl = time_left()
            if tl > 30 and c5_fcd < best_score + 0.001:
                h_fpol, sc_fpol = slsqp_optimize(h_fcd, n_try, target_try,
                                                 [5000, 20000], 300, 1e-14)
                if sc_fpol < best_score:
                    best_score = sc_fpol
                    best_h = h_fpol

    # Phase 6: Final intensive CD
    if time_left() > 60:
        n_final = len(best_h)
        target_final = n_final / 2.0
        tl = time_left()
        h_cd_f, c5_cd_f = paired_cd(best_h, n_final, target_final,
                                    time_limit=min(50, tl * 0.7),
                                    max_sweeps=50000,
                                    n_cand_start=50, n_cand_end=100,
                                    top_k_eval=80, n_top_shifts=4,
                                    max_no_improve=500, n_bisect=16)
        if c5_cd_f < best_score:
            best_score = c5_cd_f
            best_h = h_cd_f

    # Final SLSQP polish
    if time_left() > 20:
        n_last = len(best_h)
        target_last = n_last / 2.0
        h_last, sc_last = slsqp_optimize(best_h, n_last, target_last,
                                         [20000, 100000], 300, 1e-14)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
