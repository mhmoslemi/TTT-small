    g[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    g[neg_mask] -= h[p - shifts[neg_mask]]
    return g

                sweeps_since_improve = 0
                continue

        grad = compute_grad_active(h, corr, n, idxs, mid, n_active_shifts, 5.0)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        k_star_idx = int(np.argmax(corr))

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.5 * np.max(np.abs(g)) + 1e-12:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i] = _change_at(best_i, h, n, shifts, mid, idxs)
            change[best_j] = _change_at(best_j, h, n, shifts, mid, idxs)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
        else:
            consecutive_no_improve += 2

    return best_h, best_c5


def projected_gradient(h_init, n, target, time_limit, alpha0=0.05, n_active=5):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0

    for step in range(100000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            alpha *= 0.97
            if stale > 200 or alpha < 1e-6:
                break

        grad = compute_grad_active(h, corr, n, idxs, mid, n_active, 3.0)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        h_new = h + alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Quick SLSQP at n=400
    if tleft() > 60:
        deadline = time.time() + min(40, tleft() * 0.35)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000],
                                500, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at n=400 to get a better starting point
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_fast(best_h, n0, target0,
                                   time_limit=min(60, tleft() * 0.3),
                                   max_sweeps=20000,
                                   n_cand_start=10, n_cand_end=40,
                                   max_no_improve=200,
                                   n_active_shifts=4,
                                   reheat_interval=80, reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 200:
            h_cd1, c5_cd1 = pair_cd_fast(h1, n1, target1,
                                         time_limit=min(180, tleft() * 0.5),
                                         max_sweeps=40000,
                                         n_cand_start=15, n_cand_end=60,
                                         max_no_improve=300,
                                         n_active_shifts=4,
                                         reheat_interval=100, reheat_scale=0.04)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if tleft() > 80:
            deadline = time.time() + min(50, tleft() * 0.4)
            h_pol1, sc1 = slsqp_round(best_h, current_n, current_target,
                                      [500, 2000, 10000, 50000],
                                      800, 1e-14, deadline=deadline)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    # Phase 4: Interpolate to n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2

        if tleft() > 150:
            h_cd2, c5_cd2 = pair_cd_fast(h2, n2, target2,
                                         time_limit=min(130, tleft() * 0.55),
                                         max_sweeps=25000,
                                         n_cand_start=12, n_cand_end=50,
                                         max_no_improve=250,
                                         n_active_shifts=5,
                                         reheat_interval=80, reheat_scale=0.035)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                current_n = n2
                current_target = target2

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.4)
            h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                      [1000, 5000, 20000],
                                      600, 1e-14, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 5: Try n=1600 if time permits
    if tleft() > 250:
        n3 = 1600
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h3)
        if sc3 < best_score + 0.001:
            if tleft() > 200:
                h_cd3, c5_cd3 = pair_cd_fast(h3, n3, target3,
                                             time_limit=min(150, tleft() * 0.55),
                                             max_sweeps=15000,
                                             n_cand_start=10, n_cand_end=40,
                                             max_no_improve=200,
                                             n_active_shifts=5,
                                             reheat_interval=70, reheat_scale=0.03)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3
                    current_n = n3
                    current_target = target3

                if tleft() > 50:
                    deadline = time.time() + min(35, tleft() * 0.4)
                    h_pol3, sc3 = slsqp_round(best_h, current_n, current_target,
                                              [2000, 10000],
                                              500, 1e-14, deadline=deadline)
                    if sc3 < best_score:
                        best_score = sc3
                        best_h = h_pol3

    # Phase 6: Multi-start with structured perturbations
    if tleft() > 150:
        for attempt in range(5):
            if tleft() < 50:
                break
            noise_scale = [0.02, 0.05, 0.08, 0.12, 0.18][attempt % 5]
            if attempt % 2 == 0:
                h_pert = fourier_perturb(best_h, rng, n_modes=4, scale=noise_scale)
            else:
                h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            if tleft() < 30:
                break
            deadline = time.time() + min(25, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 2000, 10000],
                                        400, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 7: Final CD polishing
    if tleft() > 80:
        h_cd_f, c5_f = pair_cd_fast(best_h, current_n, current_target,
                                    time_limit=min(50, tleft() * 0.5),
                                    max_sweeps=10000,
                                    n_cand_start=20, n_cand_end=70,
                                    max_no_improve=200,
                                    n_active_shifts=5,
                                    reheat_interval=60, reheat_scale=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 8: Final SLSQP
    if tleft() > 40:
        deadline = time.time() + min(25, tleft() * 0.5)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [5000, 20000, 80000],
                                        600, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
