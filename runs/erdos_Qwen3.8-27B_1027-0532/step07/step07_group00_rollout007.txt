import numpy as np

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha, top_m=None):
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_m is not None and top_m < len(corr):
        # Only consider top m values
        idx = np.argpartition(corr, -top_m)[-top_m:]
        corr_top = corr[idx]
        c_max = np.max(corr_top)
        shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:

                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

        n_act = min(top_m, n_corr)

        grad = np.zeros(n)

            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k]])
            grad[v2] -= h[idxs[v2] + k]]
        grad /= len(active_idx)
        grad -= np.mean(grad)
        
        # Adaptive step size
        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.7
        
        # Periodic step reset
        if it % 500 == 0 and step < 0.001:
            step = 0.005
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def active_set_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_start=30, n_cand_end=100, max_no_improve=500,
                       weight_beta=8.0, use_anneal=True, T0=0.001,
                       top_m_shifts=50, max_pairs_to_try=1500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(777)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweep_count = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        sweep_count += 1
        progress = min(float(sweep_count) / max(max_no_improve * 2, 1), 1.0)

        if use_anneal:
            T = T0 * (0.4 ** progress)
        else:
            T = 0.0

        n_active = min(top_m_shifts, n_corr)
        active_idx = np.argpartition(corr, -n_active)[-n_active:]

        c_max = float(np.max(corr))
        weights = np.zeros(n_corr)
        weights[active_idx] = np.exp(weight_beta * (corr[active_idx] - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        for s_idx in active_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k]])
            grad[v2] -= w * h[idxs[v2] + k]]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        pair_scores = np.zeros((n_h, n_l))
        for ii in range(n_h):
            pair_scores[ii, :] = grad[high_set[ii]] - grad[low_set]

        flat_scores = pair_scores.flatten()
        n_try = min(max_pairs_to_try, len(flat_scores))
        if n_try < len(flat_scores):
            top_flat = np.argpartition(flat_scores, -n_try)[-n_try:]
            top_flat = top_flat[np.argsort(flat_scores[top_flat])[::-1]]
        else:
            top_flat = np.argsort(flat_scores)[::-1]

        for pf in top_flat:
            ii = int(pf // n_l)
            jj = int(pf % n_l)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(15):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx

            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
                break
            elif T > 1e-12 and approx_max - c5_val < T * 5:
                if rng.random() < np.exp(-(approx_max - c5_val) / T):
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_cd(h_init, n, target, time_limit, top_m=30,
                   max_iter=20000, step_init=0.008):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_act = min(top_m, n_corr)
        active_idx = np.argpartition(corr, -n_act)[-n_act:]

        grad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k]])
            grad[v2] -= h[idxs[v2] + k]]
        grad /= len(active_idx)
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.08, 0.08)
        else:
            step *= 0.75

        if it % 500 == 0 and step < 0.0005:
            step = 0.004

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_restart_optimize(h_best, n, target, time_limit, n_restarts=4, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)

    scales = np.linspace(0.02, 0.12, n_restarts)
    per_restart_time = (time_limit - 20.0) / n_restarts

    for scale in scales:
        if time.time() - start > time_limit - 15:
            break

        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        h_res, c5_res = active_set_pair_cd(h_pert, n, target,
                                            time_limit=per_restart_time,
                                            max_sweeps=10000,
                                            n_cand_start=30, n_cand_end=80,
                                            max_no_improve=250,
                                            weight_beta=6.0,
                                            use_anneal=True, T0=0.003,
                                            top_m_shifts=40)

        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    return best_h, best_c5

        h_cd, c5_cd = active_set_pair_cd(best_h, n0, target0,
                                          time_limit=t_lim,
                                          max_sweeps=80000,
                                          n_cand_start=30, n_cand_end=110,
                                          max_no_improve=600,
                                          weight_beta=8.0,
                                          use_anneal=True, T0=0.001,
                                          top_m_shifts=50,
                                          max_pairs_to_try=2000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with higher alpha
    if tleft() > 100:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                   [5000, 20000],
                                   400, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 4: Pair CD with different parameters (wider exploration)
    if tleft() > 200:
        t_lim = min(tleft() * 0.3, 280)
        h_cd2, c5_cd2 = active_set_pair_cd(best_h, n0, target0,
                                            time_limit=t_lim,
                                            max_sweeps=80000,
                                            n_cand_start=50, n_cand_end=130,
                                            max_no_improve=400,
                                            weight_beta=5.0,
                                            use_anneal=True, T0=0.002,
                                            top_m_shifts=30,
                                            max_pairs_to_try=1500)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Subgradient method
    if tleft() > 100:
        h_grad, c5_grad = subgradient_cd(best_h, n0, target0,
                                          time_limit=min(100, tleft() * 0.25),
                                          top_m=30, max_iter=15000)
        if c5_grad < best_score:
            best_score = c5_grad
            best_h = h_grad

    # Phase 6: Multi-restart exploration
    if tleft() > 200:
        h_mr, c5_mr = multi_restart_optimize(best_h, n0, target0,
                                              time_limit=min(220, tleft() * 0.45),
                                              n_restarts=4, seed=seed)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 7: Try finer grid
    if tleft() > 300:
        n1 = min(1000, n0 + 200)
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 250:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                       [100, 500],
                                       250, 1e-14)

            if tleft() > 200:
                h_cd1, c5_cd1 = active_set_pair_cd(h_pol1, n1, target1,
                                                    time_limit=min(180, tleft() * 0.5),
                                                    max_sweeps=40000,
                                                    n_cand_start=30, n_cand_end=90,
                                                    max_no_improve=350,
                                                    weight_beta=7.0,
                                                    use_anneal=True, T0=0.001,
                                                    top_m_shifts=40,
                                                    max_pairs_to_try=1500)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1

    # Phase 8: Final SLSQP polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                     [10000, 50000],
                                     350, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 9: Final aggressive pair CD
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fc, sc_fc = active_set_pair_cd(best_h, n_cur, target_cur,
                                          time_limit=min(40, tleft() * 0.6),
                                          max_sweeps=20000,
                                          n_cand_start=60, n_cand_end=120,
                                          max_no_improve=150,
                                          weight_beta=10.0,
                                          use_anneal=False,
                                          top_m_shifts=60,
                                          max_pairs_to_try=2000)
        if sc_fc < best_score:
            best_score = sc_fc
            best_h = h_fc

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
