    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
            n_cand_start=20, n_cand_end=60, max_no_improve=300,
            n_top_lags=5, weight_beta=5.0, use_anneal=False, T0=0.0003):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(42)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0
        
        c_max = float(np.max(corr))
        
        # Use weighted gradient from top lags
        weights = np.zeros(n_corr)
        top_idx = np.argsort(corr)[-n_top_lags:]
        for s_idx in top_idx:
            weights[s_idx] = np.exp(weight_beta * (corr[s_idx] - c_max) / max(c_max, 1e-10))
        
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 5:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def block_move_cd(h_init, n, target, time_limit, max_sweeps=10000,
                  block_sizes=(3, 5, 7, 11), max_no_improve=200):
    """Try moving contiguous blocks of h values to different positions."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(99)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[-5:]
        
        # Compute gradient from top lags
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Try block moves guided by gradient
        best_trial_c5 = c5_val
        best_move = None
        
        for bs in block_sizes:
            # Find blocks to move (high gradient sum = wants to decrease)
            block_grads = np.array([np.sum(grad[i:i+bs]) for i in range(n - bs + 1)])
            # Find blocks to receive (low gradient sum = wants to increase)
            # Actually, we want to move mass from high-grad to low-grad locations
            
            # Pick top source and top destination
            top_sources = np.argsort(block_grads)[-10:]
            top_dests = np.argsort(block_grads)[:10]
            
            for si in top_sources:
                for di in top_dests:
                    # Check overlap
                    si, di = int(si), int(di)
                    if si <= di < si + bs or di <= si < di + bs:
                        continue  # blocks overlap
                    
                    # Compute the change: move block [si, si+bs) to [di, di+bs)
                    # This changes h at positions si..si+bs-1 and di..di+bs-1
                    # The new h would have h_new[si:si+bs] = h[di:di+bs] and h_new[di:di+bs] = h[si:si+bs]
                    # But we need to maintain the constraint... swapping preserves the sum.
                    
                    # Actually, for a swap, the sum is preserved.
                    # Let's just compute the new correlation directly (expensive but accurate)
                    h_try = h.copy()
                    h_try[si:si+bs], h_try[di:di+bs] = h_try[di:di+bs].copy(), h_try[si:si+bs].copy()
                    # h_try is already feasible (swap preserves bounds and sum)
                    
                    sc_try = compute_c5(h_try)
                    if sc_try < best_trial_c5 - 1e-14:
                        best_trial_c5 = sc_try
                        best_move = (si, di, bs)
            
            if best_move is not None:
                break
        
        if best_move is not None:
            si, di, bs = best_move
            h[si:si+bs], h[di:di+bs] = h[di:di+bs].copy(), h[si:si+bs].copy()
    
    return best_h, best_c5

Wait, the block_move_cd as written is too expensive because it computes compute_c5 for each candidate move, which is O(n^2) each time. Let me optimize this.

Actually, for n=800, compute_c5 is O(n^2) ≈ 640,000 operations. With 100 candidate moves per sweep, that's 64M operations per sweep. At 100 sweeps, that's 6.4B operations, which might be too slow.

Let me use the change matrix approach for block moves too, or limit the number of candidates.

Actually, let me reconsider the whole approach. The block move is O(n) to compute the new correlation if I use the change matrix properly. But it's complicated because a block move affects multiple rows of the change matrix.

Let me simplify: instead of block swaps, let me try "block transfers" - move a small amount of mass from one region to another. This is like a multi-coordinate version of the pair CD.

Actually, I think the most promising approach is:

1. Start from initial_h_values
2. Do a quick optimization at n=800 (SLSQP + pair CD) to get a good starting point
3. Then use a "multi-start" approach: create several perturbed versions and optimize each
4. The key difference from the previous code: use a **different perturbation strategy** - instead of Gaussian noise, use structured perturbations (e.g., add a sinusoidal component, or shift parts of the function)

Let me also consider: maybe the issue is that at n=800, we're already near the optimum for that resolution, and we need n=1000 or n=1200 to get below 0.3808.
