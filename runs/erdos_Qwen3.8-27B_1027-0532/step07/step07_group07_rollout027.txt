def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def smooth_max_and_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    obj = float(c_max + np.log(s) / alpha)
    if s < 1e-300:
        return obj, np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return obj, grad

def slsqp_solve(h_start, n, target, alphas, maxiter_per_alpha=500):
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_max_and_grad(x, _g_alpha[0])[0]
    
    def _grad(x):
        return smooth_max_and_grad(x, _g_alpha[0])[1]
    
    def _cons(x):
        return np.sum(x) - target
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        _g_alpha[0] = alpha
        try:
            res = minimize(_obj, h_cur, jac=_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, top_k_eval=60,
            max_no_improve=400, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    stall_count = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        improved = False
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Try a random restart
                if time.time() - start > time_limit - 30:
                    break
                # Perturb and reset
                h = best_h.copy()
                h += np.random.randn(n) * 0.01
                h = project_feasible(h, target)
                # Rebuild change matrix
                change = build_change_matrix(h, n)
                no_improve = 0
                stall_count += 1
                if stall_count > 5:
                    break
                continue

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        # Multi-shift gradient
        n_gs = min(n_shifts_grad, n_corr)
        top_shifts = np.argsort(corr)[-n_gs:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(n_gs)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Score by effect on top shift
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        
        # Get top candidates
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            # Bisection
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(n_gs)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        if use_multishift_score:
            score_mat = np.zeros((len(high_set), len(low_set)))
            for s_idx in top_shifts:
                ch_h = change[high_set, s_idx]
                ch_l = change[low_set, s_idx]
                score_mat += (ch_l[None, :] - ch_h[:, None])
        else:
            ch_h_star = change[high_set, k_star_idx]
            ch_l_star = change[low_set, k_star_idx]
            score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(13):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def get_time_left(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    tl = get_time_left(start_time, budget_s)
    if tl > 80:
        h_pol, sc = slsqp_solve(best_h, n0, target0, [200, 1000, 5000], 500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = get_time_left(start_time, budget_s)
    if tl > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        tl = get_time_left(start_time, budget_s)
        if tl > 250:
            h_cd, c5_cd = pair_cd(h1, n1, target1, time_limit=min(220, tl * 0.75),
                                   max_sweeps=50000, n_cand_start=30, n_cand_end=100,
                                   top_k_eval=70, max_no_improve=500, n_shifts_grad=4,
                                   use_multishift_score=True)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        tl = get_time_left(start_time, budget_s)
        if tl > 80:
            h_pol, sc = slsqp_solve(best_h, n1, target1, [2000, 10000], 600)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    tl = get_time_left(start_time, budget_s)
    if tl > 350:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = get_time_left(start_time, budget_s)
        if tl > 250:
            h_cd, c5_cd = pair_cd(h2, n2, target2, time_limit=min(200, tl * 0.75),
                                   max_sweeps=30000, n_cand_start=30, n_cand_end=90,
                                   top_k_eval=60, max_no_improve=400, n_shifts_grad=3,
                                   use_multishift_score=False)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        tl = get_time_left(start_time, budget_s)
        if tl > 60:
            h_pol, sc = slsqp_solve(best_h, n2, target2, [5000, 20000], 500)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    n_final = len(best_h)
    target_final = n_final / 2.0

    tl = get_time_left(start_time, budget_s)
    if tl > 120:
        h_cd, c5_cd = pair_cd(best_h, n_final, target_final, time_limit=min(100, tl * 0.75),
                               max_sweeps=25000, n_cand_start=40, n_cand_end=110,
                               top_k_eval=80, max_no_improve=350, n_shifts_grad=4,
                               use_multishift_score=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    tl = get_time_left(start_time, budget_s)
    if tl > 80:
        h_pol, sc = slsqp_solve(best_h, n_final, target_final, [10000, 50000], 400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = get_time_left(start_time, budget_s)
    if tl > 80:
        rng = np.random.RandomState(seed + 1)
        for noise_scale in [0.015, 0.04, 0.08]:
            tl = get_time_left(start_time, budget_s)
            if tl < 50:
                break
            h_pert = best_h + rng.randn(n_final) * noise_scale
            h_pert = project_feasible(h_pert, target_final)
            h_ref, sc_ref = slsqp_solve(h_pert, n_final, target_final,
                                         [500, 3000, 15000], 400)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    tl = get_time_left(start_time, budget_s)
    if tl > 50:
        h_cd_final, c5_final = pair_cd(best_h, n_final, target_final,
                                        time_limit=min(40, tl * 0.8),
                                        max_sweeps=15000, n_cand_start=50,
                                        n_cand_end=120, top_k_eval=90,
                                        max_no_improve=250, n_shifts_grad=4,
                                        use_multishift_score=True)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = get_time_left(start_time, budget_s)
    if tl > 30:
        h_final_pol, sc_final = slsqp_solve(best_h, n_final, target_final,
                                             [20000, 100000], 300)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
