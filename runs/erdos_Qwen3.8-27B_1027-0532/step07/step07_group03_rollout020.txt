_g_n = 0
_g_target = 0.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

def block_cd(h_init, n, target, time_limit, block_size=4, max_sweeps=10000):
    """Block coordinate descent: move mass in contiguous blocks."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(123)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            break
        
        # Pick a random block to move
        bs = block_size
        i_start = rng.randint(0, n - bs)
        j_start = rng.randint(0, n - bs)
        
        # Skip if blocks overlap
        if abs(i_start - j_start) < bs:
            continue
        
        # Compute the change in correlation if we move mass from block i to block j
        # Simplified: try moving a small amount
        delta = 0.01
        h_trial = h.copy()
        total_i = h_trial[i_start:i_start+bs].sum()
        total_j = h_trial[j_start:j_start+bs].sum()
        
        max_move = min(total_i - 1e-10, 1.0 * bs - total_j - 1e-10)
        if max_move < 1e-10:
            continue
        
        move = min(delta, max_move)
        h_trial[i_start:i_start+bs] -= move / bs
        h_trial[j_start:j_start+bs] += move / bs

        sc = compute_c5(h_trial)
        if sc < c5_val - 1e-14:
            h = h_trial
    
    return best_h, best_c5

This is exactly what `compute_single_grad` computes. So the cutting plane is using the correct linearization.

The issue is that the linearization is only valid locally. For a tighter relaxation, we could use **second-order** information, but that would make it a QP instead of an LP.

Alternatively, we can use **valid inequalities** for the bilinear terms. For h[i] ∈ [0,1] and h[i+k] ∈ [0,1]:
h[i]*(1-h[i+k]) ≤ h[i] (since 1-h[i+k] ≤ 1)
h[i]*(1-h[i+k]) ≤ 1-h[i+k] (since h[i] ≤ 1)

These are the McCormick-type bounds. Adding these as additional constraints to the LP would tighten the relaxation.

But adding O(n²) constraints might be too expensive. Let me think about which ones to add.

Actually, the most useful McCormick inequalities for our problem are:
For the term h[i]*(1-h[i+k]):
- Upper bound: min(h[i], 1-h[i+k])
- Lower bound: max(0, h[i] + h[i+k] - 1)

So: max(0, h[i] + h[i+k] - 1) ≤ h[i]*(1-h[i+k]) ≤ min(h[i], 1-h[i+k])

In the cutting plane, we're using the upper bound (linearization). Adding the lower bound as a constraint would help:
h[i]*(1-h[i+k]) ≥ h[i] + h[i+k] - 1 (for the case where h[i] + h[i+k] > 1)

But this is a constraint on the bilinear term, not on h directly. To use it in the LP, we'd need to introduce auxiliary variables for each bilinear term.

This is getting complex. Let me stick with the simpler improvements and focus on making the CD and cutting plane work better with the time available.
