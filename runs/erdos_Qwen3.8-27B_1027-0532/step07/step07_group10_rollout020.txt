def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change, shifts


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]
    return change


def find_optimal_delta(corr, g, d_lo, d_hi, n_iter=22):

        else:
            f_prime = 0.0
        if f_prime > 1e-14:
            b = mid
        else:
            a = mid

            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_new = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
