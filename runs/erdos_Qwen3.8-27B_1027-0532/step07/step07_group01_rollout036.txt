import numpy as np
from scipy.optimize import minimize
import time

_start_time = 0.0
_budget = 1300.0

def _remaining():
    return _budget - 20.0 - (time.time() - _start_time)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha, n, target, penalty=1000.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    shifted = np.clip(shifted, -700.0, 0.0)
    s = np.sum(np.exp(shifted))
    f = mx + np.log(s) / alpha
    f += penalty * (np.sum(x) - target) ** 2
    return f

def _smooth_grad(x, alpha, n, target, penalty=1000.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(corr)
    shifted = alpha * (corr - mx)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    sw = np.sum(w)
    w /= sw
    
    grad = np.zeros(n)
    mid = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        offset = k - mid
        if offset >= 0:
            if offset == 0:
                grad += wk * (1.0 - 2.0 * x)
            else:
                # c_k = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
                # d/dh[j] = (1-h[j-k]) for k<=j<n, minus h[j+k] for 0<=j<=n-1-k
                if k < n:
                    grad[k:] += wk * (1.0 - x[:n-k])
                if n - 1 - k >= 0:
                    grad[:n-k] -= wk * x[k:]
        else:
            m = -offset
            # c_{-m} = sum_{n=m}^{n-1} h[n]*(1-h[n-m])
            # d/dh[j] = (1-h[j-m]) for m<=j<n, minus h[j+m] for 0<=j<=n-1-m
            if m < n:
                grad[m:] += wk * (1.0 - x[:n-m])
            if n - 1 - m >= 0:
                grad[:n-m] -= wk * x[m:]
    
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_init, n, target, alphas, max_iter=500, time_limit=100):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()

                jac=_smooth_grad, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-8}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=20, n_cand_end=80, max_no_improve=500,
            restart_every=200, restart_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consec = 0
    rng = np.random.RandomState(12345)
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    def _update_change(p):
        change[p, :] = 0.0
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit or _remaining() < 30:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()
            consec = 0
        else:
            consec += 1

            break

            h = best_h + rng.randn(n) * restart_scale

            for p in range(n):
                _update_change(p)
            consec = 0
        
        # Compute subgradient (uniform over top shifts)

        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_delta = 0.0
        best_ij = (-1, -1)

                        b = mid_d
                    else:
                        a = mid_d
                delta = 0.5 * (a + b)

                    best_delta = delta

            h[j] += best_delta

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    # Midpoint interpolation
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0/n_src
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0/n_dst
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                trial = np.max(corr + delta * g) * dx
                if trial < best_trial:
                    best_trial = trial
                    best_i = i
                    best_j = j
                    best_delta = delta
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                pos = (p + shifts >= 0) & (p + shifts < n)
                neg = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0.0
                change[p, pos] += (1.0 - h[p + shifts[pos]])
                change[p, neg] -= h[p - shifts[neg]]
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
