For each coordinate i, the optimal value can be found by noting that the smooth objective as a function of h[i] (with other coordinates fixed) is a 1D function that's a sum of exponentials of linear functions. We can minimize it with golden section search or by finding the critical point.

Actually, for the smooth objective f(h) = (1/alpha)*log(∑_k exp(alpha * corr_k * dx)), with all coordinates fixed except h[i]:

where a_k is the part not involving h[i] and b_k = (1-h[i+k]) - h[i-k] (the coefficient of h[i] in corr_k).

where w_k = exp(alpha * corr_k * dx) / ∑_j exp(alpha * corr_j * dx) (the softmax weights).

This is a weighted average of the terms (1-h[i+k]) - h[i-k], weighted by the softmax of the correlations.     return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def generate_binary_patterns(n, rng, num):
    patterns = []
    for _ in range(num):
        ones = rng.choice(n, size=n // 2, replace=False)
        h_b = np.zeros(n)
        h_b[ones] = 1.0
        patterns.append(h_b)
    return patterns

def generate_periodic_patterns(n, rng, num):
    patterns = []
    for period in [3, 5, 7, 11, 13]:
        h_per = np.zeros(n)
        phase = rng.uniform(0, period)
        duty = 0.3 + 0.4 * rng.random()
        for i in range(n):
            if ((i + phase) % period) < period * duty:
                h_per[i] = 1.0
        patterns.append(h_per.copy())
        if len(patterns) >= num:
            break
    while len(patterns) < num:
        period = int(rng.randint(3, 15))
        h_per = np.zeros(n)
        phase = rng.uniform(0, period)
        duty = 0.3 + 0.4 * rng.random()
        for i in range(n):
            if ((i + phase) % period) < period * duty:
                h_per[i] = 1.0
        patterns.append(h_per)
    return patterns[:num]

def generate_block_swaps(h_base, n, rng, num, num_swaps=5):
    patterns = []
    for _ in range(num):
        h_s = h_base.copy()
        for _ in range(num_swaps):
            bs = int(rng.randint(5, max(6, n // 20)))
            pos1 = int(rng.randint(0, n - bs))
            pos2 = int(rng.randint(0, n - bs))
            if pos1 == pos2:
                continue
            tmp = h_s[pos1:pos1 + bs].copy()
            h_s[pos1:pos1 + bs] = h_s[pos2:pos2 + bs]
            h_s[pos2:pos2 + bs] = tmp
        patterns.append(h_s)
    return patterns

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(seed)

    if tleft() > 50:
        h_pol, sc_pol = slsqp_optimize(best_h, n, target, [50, 200, 1000], 600, 1e-15)
        if sc_pol < best_c5:
            best_c5 = sc_pol
            best_h = h_pol

    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                               time_limit=min(200, tleft() * 0.7),
                               max_sweeps=50000,
                               n_cand_start=40, n_cand_end=120,
                               max_no_improve=500,
                               n_binsearch=20,
                               stochastic_frac=0.06, seed=42)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if tleft() > 350:
        n_starts = 12
        per_start_time = min(28, int((tleft() - 30) / n_starts))
        starts = []
        bin_pats = generate_binary_patterns(n, rng, 4)
        starts.extend(bin_pats)
        peri_pats = generate_periodic_patterns(n, rng, 3)
        starts.extend(peri_pats)
        for scale in [0.05, 0.1, 0.2, 0.35]:
            h_p = best_h + rng.randn(n) * scale
            h_p = project_feasible(h_p, target)
            starts.append(h_p)
        block_pats = generate_block_swaps(best_h, n, rng, 3)
        starts.extend(block_pats)
        for s in starts:
            remaining = tleft()
            if remaining < 35:
                break
            h_c, c5_c = pair_cd(s, n, target, time_limit=min(per_start_time, remaining - 20),
                                  max_sweeps=30000, n_cand_start=30, n_cand_end=90,
                                  max_no_improve=200, n_binsearch=18,
                                  stochastic_frac=0.1, seed=int(rng.randint(99999)))
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c

    if tleft() > 150:
        h_g, c5_g = gradient_cd(best_h, n, target,
                                 time_limit=min(120, tleft() * 0.6),
                                 max_sweeps=20000, n_cand=70,
                                 n_binsearch=18, seed=789, reheat_interval=250)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g

    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                               time_limit=min(200, tleft() * 0.7),
                               max_sweeps=50000,
                               n_cand_start=50, n_cand_end=150,
                               max_no_improve=500,
                               n_binsearch=22,
                               stochastic_frac=0.04, seed=137)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n, target, [5000, 20000, 80000], 2000, 1e-15)
        if sc_pol < best_c5:
            best_c5 = sc_pol
            best_h = h_pol

    if tleft() > 100:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                               time_limit=min(80, tleft() * 0.7),
                               max_sweeps=30000,
                               n_cand_start=60, n_cand_end=160,
                               max_no_improve=400,
                               n_binsearch=22,
                               stochastic_frac=0.05, seed=501)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc_pol = slsqp_optimize(best_h, n, target, [20000, 100000, 400000], 3000, 1e-15)
        if sc_pol < best_c5:
            best_c5 = sc_pol
            best_h = h_pol

    if tleft() > 30:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                               time_limit=min(tleft() - 10, 60),
                               max_sweeps=20000,
                               n_cand_start=80, n_cand_end=200,
                               max_no_improve=300,
                               n_binsearch=24,
                               stochastic_frac=0.03, seed=777)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    final_h = project_feasible(best_h, target)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n
