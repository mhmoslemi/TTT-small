```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur
```

```python
def fw_subgradient(h, active_lags, n):
    """Compute subgradient of max_k c_k at h, averaged over active lags."""
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros(n)
    for k_idx in active_lags:
        k = k_idx - mid
        mask1 = (idxs + k >= 0) & (idxs + k < n)
        g[mask1] += (1.0 - h[idxs[mask1] + k])
        mask2 = (idxs - k >= 0) & (idxs - k < n)
        g[mask2] -= h[idxs[mask2] - k]
    g /= len(active_lags)
    return g

def fw_linear_min(g, target):
    """Solve min g^T h' s.t. 0≤h'≤1, sum(h')=target."""
    n = len(g)
    order = np.argsort(g)
    h_prime = np.zeros(n)
    n_full = int(target)
    h_prime[order[:n_full]] = 1.0
    frac = target - n_full
    if 0 < frac < 1:
        h_prime[order[n_full]] = frac
    return h_prime

def frank_wolfe(h_init, n, target, time_limit, n_active_lags=5, max_iters=5000):
    """Frank-Wolfe method with line search."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0/n)
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Get active lags
        top_idx = np.argsort(corr)[-n_active_lags:]
        
        # Compute subgradient
        g = fw_subgradient(h, top_idx, n)
        
        # Linear minimization
        h_prime = fw_linear_min(g, target)
        
        # Line search
        best_t = 0.0
        best_f = c5_val
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            h_trial = (1 - t) * h + t * h_prime
            h_trial = project_feasible(h_trial, target)
            f_trial = compute_c5(h_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_t = t
        
        if best_t == 0.0:
            break
        
        h = (1 - best_t) * h + best_t * h_prime
        h = project_feasible(h, target)
    
    return best_h, best_score
```

```python
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=3, n_cand_start=15, n_cand_end=50,
                max_no_improve=400, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        max_pairs = 300
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            pair_list = []
            # Use stratified sampling
            n_strata = min(max_pairs, n_h * n_l)
            fracs_h = np.linspace(0, 1, n_h)
            fracs_l = np.linspace(0, 1, n_l)
            for ii in range(n_h):
                for jj in range(n_l):
                    if len(pair_list) >= max_pairs:
                        break
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
                if len(pair_list) >= max_pairs:
                    break
        
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            
            g = change[j] - change[i]
            
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5

def fw_method(h_init, n, target, time_limit, n_active=5, max_iters=3000,
              cool_rate=0.999):
    """Frank-Wolfe with line search for global moves."""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    temp = 0.1  # exploration parameter
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Get active lags (top n_active)
        top_idx = np.argsort(corr)[-n_active:]
        mid = n - 1
        idxs = np.arange(n)
        
        # Compute subgradient
        g = np.zeros(n)
        for k_idx in top_idx:
            k = int(k_idx) - mid
            mask1 = (idxs + k >= 0) & (idxs + k < n)
            g[mask1] += (1.0 - h[idxs[mask1] + k])
            mask2 = (idxs - k >= 0) & (idxs - k < n)
            g[mask2] -= h[idxs[mask2] - k]
        g /= len(top_idx)
        
        # Linear minimization: min g^T h' s.t. 0≤h'≤1, sum(h')=target
        order = np.argsort(g)
        h_prime = np.zeros(n)
        n_full = int(target)
        h_prime[order[:n_full]] = 1.0
        frac = target - n_full
        if 0 < frac < 1:
            h_prime[order[n_full]] = frac
        
        # Line search over t
        best_t = 0.0
        best_f = c5_val
        for t in [1.0, 0.8, 0.5, 0.3, 0.15, 0.05]:
            h_trial = (1.0 - t) * h + t * h_prime
            h_trial = project_feasible(h_trial, target)
            f_trial = compute_c5(h_trial)
            if f_trial < best_f - 1e-15:
                best_f = f_trial
                best_t = t
        
        if best_t > 0:
            h = (1.0 - best_t) * h + best_t * h_prime
            h = project_feasible(h, target)
        
        temp *= cool_rate
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

**Strategy: Alternating minimization over "blocks"**

Instead of changing one pair at a time (as in CD), change an entire block:
- Select a block of m consecutive indices
- Optimize the values in that block while keeping the rest fixed
- This is a small subproblem (m variables) that can be solved with a local QP or line search

For m=2, this is the same as pair CD. For m=3 or m=4, we can explore a larger neighborhood.

For a block of size m, the objective is:
max_k [c_k + ∑_{i in block} δ_i * change[i][k]]

where δ_i = h_new[i] - h_old[i].

This is a linear function of δ (for each k), so the max is a piecewise-linear function of δ. We want to minimize this subject to:
- 0 ≤ h_old[i] + δ_i ≤ 1
- ∑ δ_i = 0 (to maintain the sum constraint)

This is a **linear programming** problem in the δ variables (with one additional variable t for the max):
min t
s.t. c_k + ∑ δ_i * change[i][k] ≤ t for all k
     -h_old[i] ≤ δ_i ≤ 1-h_old[i] for i in block
     ∑ δ_i = 0

This is an LP with n_corr + m + 1 variables and n_corr + m + 1 constraints. For m=4 and n_corr=799, this is ~800 variables and ~803 constraints. Very solvable!

We can use scipy.optimize.linprog or cvxpy for this.

```python
def block_lp_update(h, block_indices, n, target, time_per_call=1.0):
    """Solve LP to optimize a block of indices."""
    from scipy.optimize import linprog
    start = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    m = len(block_indices)
    
    # Build change vectors for each index in the block
    # change[i][k] = effect on c_k of increasing h[i] by 1
    block_changes = np.zeros((m, n_corr))
    mid = n - 1
    for ii, p in enumerate(block_indices):
        lags = np.arange(0, p + 1)
        block_changes[ii, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        block_changes[ii, mid + lags2] -= h[p + lags2]
    
    # Variables: [delta_1, ..., delta_m, t]
    # Objective: minimize t
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0  # coefficient of t
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each lag k: sum_i delta_i * change[i][k] - t <= -c_k
    A_ineq = np.zeros((n_corr, m + 1))
    A_ineq[:, :m] = block_changes.T  # shape (n_corr, m)
    A_ineq[:, m] = -1.0
    b_ineq = -corr
    
    # Equality constraint: sum delta_i = 0
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] <= delta_i <= 1-h[i] for each i in block, t >= 0
    bounds = []
    for ii, p in enumerate(block_indices):
        bounds.append((-h[p], 1.0 - h[p]))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs')
        if res.status == 0:
            deltas = res.x[:m]
            h_new = h.copy()
            for ii, p in enumerate(block_indices):
```

```python
    dx = 2.0 / n
```

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
