The vectorization in step 5: compute all g's at once (n_cand² × n_corr array), then for each, do the binary search in parallel.

For the binary search in parallel:
- Initialize δ_lo and δ_hi for each pair (n_cand² vector)
- For each iteration of binary search:
  - Compute mid = (δ_lo + δ_hi) / 2 (n_cand² vector)
  - Compute vals = corr[None,:] + mid[:,None] * G (n_cand² × n_corr)
  - Compute f_mid = vals.max(axis=1) (n_cand²)
  - Find active: vals >= f_mid[:,None] - eps (n_cand² × n_corr boolean)
  - Compute f_prime = G[active].max() for each pair (this is the tricky part)
  
The "f_prime" computation is not trivially vectorizable because the active set varies per pair. One approach:
- For each pair, f_prime = max of G[pair, s] over s where corr[s] + mid[pair]*G[pair,s] >= f_mid[pair] - eps
- This is equivalent to: max of G[pair, s] over s where G[pair,s] >= (f_mid[pair] - corr[s]) / mid[pair] (for mid > 0)

Hmm, this is getting complicated. Let me just use the binary search approach but vectorize the inner product:

For each pair (i,j):
- g = change[j] - change[i]  (this is precomputed, shape n_corr)
- Binary search for optimal δ:
  - mid = 0.5*(a+b)
  - vals = corr + mid * g  (shape n_corr)
  - f_mid = vals.max()
  - active = vals >= f_mid - 1e-14
  - f_prime = g[active].max() if active.any() else 0
  - if f_prime > 0: b = mid else: a = mid

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=100000,
                       n_cand_start=40, n_cand_end=120, n_binsearch=16,
                       max_no_improve=500, sa_temp=0.001, sa_prob=0.02,
                       seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_nc = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_nc = min(current_nc, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_nc]
        high_set = sorted_idx[-current_nc:]
        
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        
        if n_pairs == 0:
            continue
        
        hi_idx = high_set.astype(np.intp)
        lo_idx = low_set.astype(np.intp)
        
        hi_2d = hi_idx[:, None]
        lo_2d = lo_idx[None, :]
        
        same_mask = hi_2d == lo_2d
        hi_2d = hi_2d.ravel()
        lo_2d = lo_2d.ravel()
        same_mask = same_mask.ravel()
        
        valid_pairs = ~same_mask
        hi_flat = hi_2d[valid_pairs]
        lo_flat = lo_2d[valid_pairs]
        
        if len(hi_flat) == 0:
            continue
        
        G = change[lo_flat] - change[hi_flat]
        
        delta_lo = np.maximum(h[hi_flat] - 1.0, -h[lo_flat])
        delta_hi = np.minimum(h[hi_flat], 1.0 - h[lo_flat])
        
        feasible = delta_hi - delta_lo > 1e-12
        if not feasible.any():
            continue
        
        G = G[feasible]
        hi_flat = hi_flat[feasible]
        lo_flat = lo_flat[feasible]
        delta_lo = delta_lo[feasible]
        delta_hi = delta_hi[feasible]
        
        n_f = len(hi_flat)
        if n_f == 0:
            continue
        
        a_d = delta_lo.copy()
        b_d = delta_hi.copy()
        
        corr_row = corr[None, :]
        for _ in range(n_binsearch):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr_row + mid_d[:, None] * G
            f_mid = vals.max(axis=1)
            active = vals >= (f_mid[:, None] - 1e-14)
            G_masked = np.where(active, G, -1e30)
            f_prime = G_masked.max(axis=1)
            inc = f_prime > 1e-12
            b_d[inc] = mid_d[inc]
            a_d[~inc] = mid_d[~inc]
        
        delta_star = 0.5 * (a_d + b_d)
        
        valid_delta = np.abs(delta_star) > 1e-12
        if not valid_delta.any():
            continue
        
        approx_vals = corr_row + delta_star[valid_delta][:, None] * G[valid_delta]
        approx_max = approx_vals.max(axis=1) * dx
        approx_max = np.concatenate([np.full(n_f - valid_delta.sum(), c5_val + 1.0), approx_max])
        
        best_idx = int(np.argmin(approx_max))
        best_trial_c5 = float(approx_max[best_idx])
        
        do_sa_move = (rng.random() < sa_prob) and best_trial_c5 >= c5_val
        if do_sa_move and (rng.random() < sa_prob):
            best_idx = int(rng.randint(n_f))
        
        if best_trial_c5 < c5_val or do_sa_move:
            i = int(hi_flat[best_idx])
            j = int(lo_flat[best_idx])
            d = float(delta_star[best_idx])
            if abs(d) > 1e-12:
                h[i] -= d
                h[j] += d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_cd(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()

        if remaining < 30:
            break

            best_h = h_cd
        if time.time() - start > time_limit * 0.85:

    return best_h, best_c5
