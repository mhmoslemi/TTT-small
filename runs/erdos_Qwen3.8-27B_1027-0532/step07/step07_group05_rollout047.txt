import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_and_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exp_vals / total
    soft_max = c_max + np.log(total) / alpha
    return soft_max, weights

def grad_smooth_max(h, weights, alpha):
    """Compute gradient of soft max of correlations w.r.t. h."""
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idx = np.arange(n)
    grad = np.zeros(n)
    
    # Only compute for significant weights
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    
    for k_idx in active:
        w = weights[k_idx]
        m = k_idx - mid
        im = idx + m
        valid1 = (im >= 0) & (im < n)
        grad[valid1] += w * (1.0 - h[im[valid1]])
        im2 = idx - m
        valid2 = (im2 >= 0) & (im2 < n)
        grad[valid2] -= w * h[im2[valid2]]
    
    return grad

def l_bfgs_smoothing(h_init, n, target_sum, alphas, maxiter, penalty_factor=1e6):
    """Optimize using L-BFGS-B with penalty for sum constraint."""
    h = h_init.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            sm, _ = soft_max_and_weights(corr, alpha)
            pen = penalty_factor * (np.sum(x) - target_sum) ** 2
            return sm + pen
        
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            _, weights = soft_max_and_weights(corr, alpha)
            g = grad_smooth_max(x, weights, alpha)
            g += 2.0 * penalty_factor * (np.sum(x) - target_sum)
            return g
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h = h_new
        except Exception:
            pass
    
    return h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def soft_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    s = np.sum(e)
    return e / s

def grad_soft_max(h, weights):
    n = len(h)
    mid = n - 1
    idx = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k_idx in active:
        w = weights[k_idx]
        m = k_idx - mid
        im = idx - m
        v1 = (im >= 0) & (im < n)
        grad[v1] += w * (1.0 - h[im[v1]])
        im2 = idx + m
        v2 = (im2 >= 0) & (im2 < n)
        grad[v2] -= w * h[im2[v2]]
    return grad

def lbfgs_optimize(h_init, n, target_sum, alphas, maxiter_per_alpha, penalty):
    h = h_init.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return soft_max(corr, alpha) + penalty * (np.sum(x) - target_sum) ** 2
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            w = soft_max_weights(corr, alpha)
            g = grad_soft_max(x, w)
            g += 2.0 * penalty * (np.sum(x) - target_sum)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h = h_new
        except:
            pass
    return h, best_score

        grad[v1] += w * (1.0 - h[im[v1]])
        im2 = idx + m
        v2 = (im2 >= 0) & (im2 < n)
        grad[v2] -= w * h[im2[v2]]
    return grad

def lbfgs_round(h_start, n, target_sum, alphas, maxiter_per_alpha, penalty=1e5):
    h = h_start.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return soft_max_val(corr, alpha) + penalty * (np.sum(x) - target_sum) ** 2
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            w = soft_max_weights(corr, alpha)
            g = grad_soft_max(x, w)
            g += 2.0 * penalty * (np.sum(x) - target_sum)
            return g
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h = h_new
        except Exception:
            pass
    return h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    n_corr = 2 * n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def cd_refine(h_init, n, target_sum, time_limit, max_sweeps=20000,
              n_active_start=3, n_active_end=7,
              n_cand_start=20, n_cand_end=60,
              max_no_improve=300, bisection_iters=12, max_pairs=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            pair_list = []
            n_frac = min(30, max_pairs // 5)
            for fi in range(n_frac):
                frac = fi / max(n_frac - 1, 1)
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            rng_local = np.random.RandomState(sweep)
            while len(pair_list) < max_pairs:
                ii = rng_local.randint(n_h)
                jj = rng_local.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B at original resolution
    if tleft() > 120:
        tl = min(150, tleft() * 0.2)
        h_new, sc = lbfgs_round(best_h, n0, target0,
                                 [50, 200, 1000], 400, penalty=1e5)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: CD refinement at original resolution
    if tleft() > 150:
        tl = min(200, tleft() * 0.25)
        h_new, sc = cd_refine(best_h, n0, target0, tl,
                               max_sweeps=15000,
                               n_active_start=3, n_active_end=6,
                               n_cand_start=20, n_cand_end=50,
                               max_no_improve=250, bisection_iters=12, max_pairs=350)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Increase resolution to 800
    cur_n = n0
    cur_target = target0
    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl = min(200, tleft() * 0.3)
        h_new, sc = lbfgs_round(h1, n1, target1,
                                 [200, 1000, 5000], 300, penalty=1e5)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            cur_n = n1
            cur_target = target1

    # Phase 4: CD at higher resolution
    if tleft() > 250:
        tl = min(300, tleft() * 0.35)
        h_new, sc = cd_refine(best_h, cur_n, cur_target, tl,
                               max_sweeps=20000,
                               n_active_start=3, n_active_end=7,
                               n_cand_start=25, n_cand_end=60,
                               max_no_improve=300, bisection_iters=12, max_pairs=400)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: L-BFGS-B polishing
    if tleft() > 150:
        tl = min(150, tleft() * 0.3)
        h_new, sc = lbfgs_round(best_h, cur_n, cur_target,
                                 [5000, 20000, 50000], 400, penalty=1e6)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: CD with more aggressive settings
    if tleft() > 150:
        tl = min(150, tleft() * 0.4)
        h_new, sc = cd_refine(best_h, cur_n, cur_target, tl,
                               max_sweeps=15000,
                               n_active_start=5, n_active_end=8,
                               n_cand_start=30, n_cand_end=70,
                               max_no_improve=250, bisection_iters=14, max_pairs=450)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Perturbation + re-optimization
    if tleft() > 200:
        for scale in [0.02, 0.05, 0.1]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(cur_n) * scale
            h_pert = project_feasible(h_pert, cur_target)
            tl = min(60, tleft() * 0.3)
            h_new, sc = cd_refine(h_pert, cur_n, cur_target, tl,
                                   max_sweeps=8000,
                                   n_active_start=3, n_active_end=6,
                                   n_cand_start=25, n_cand_end=55,
                                   max_no_improve=200, bisection_iters=10, max_pairs=300)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Final L-BFGS-B
    if tleft() > 80:
        h_new, sc = lbfgs_round(best_h, cur_n, cur_target,
                                 [20000, 100000], 300, penalty=1e6)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Final CD
    if tleft() > 50:
        tl = min(40, tleft() * 0.5)
        h_new, sc = cd_refine(best_h, cur_n, cur_target, tl,
                               max_sweeps=8000,
                               n_active_start=5, n_active_end=8,
                               n_cand_start=35, n_cand_end=70,
                               max_no_improve=150, bisection_iters=12, max_pairs=400)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    final_h = project_feasible(best_h, cur_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
