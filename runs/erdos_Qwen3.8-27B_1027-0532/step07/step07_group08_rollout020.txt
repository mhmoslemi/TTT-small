import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_jac(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-16, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_jac, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=500,
            n_active_shifts=4, reheat_interval=120, reheat_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(5, n // 40)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                consecutive_no_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.3 * np.max(np.abs(g)):
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)
        momentum = beta * momentum + (1.0 - beta) * grad
        h_new = h - alpha * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP with analytic gradient at n=400
    if tleft() > 100:
        deadline = time.time() + min(80, tleft() * 0.55)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [300, 1000, 5000, 20000, 80000, 300000],
                                   1500, 1e-16, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD refinement
    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(100, tleft() * 0.5),
                              max_sweeps=30000,
                              n_cand_start=25, n_cand_end=80,
                              max_no_improve=400,
                              n_active_shifts=4,
                              reheat_interval=130, reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP again after CD
    if tleft() > 80:
        deadline = time.time() + min(60, tleft() * 0.5)
        h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                   [5000, 20000, 100000, 500000],
                                   2000, 1e-16, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Subgradient with momentum
    if tleft() > 100:
        h_sg, c5_sg = subgradient_momentum(best_h, current_n, current_target,
                                           time_limit=min(60, tleft() * 0.5),
                                           alpha0=0.25, alpha_min=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Try n=500

        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 150:
            deadline = time.time() + min(70, tleft() * 0.45)
            h_pol1, sc1 = slsqp_optimize(h1, n1, target1,
                                         [1000, 5000, 30000, 100000, 400000],
                                         1500, 1e-16, deadline=deadline)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
                current_n = n1
                current_target = target1

        if tleft() > 80:
            h_cd1, c5_cd1 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(50, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=25, n_cand_end=75,
                                    max_no_improve=350,
                                    n_active_shifts=4,
                                    reheat_interval=100, reheat_scale=0.03)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 40:
            deadline = time.time() + min(30, tleft() * 0.6)
            h_pol1b, sc1b = slsqp_optimize(best_h, current_n, current_target,
                                           [20000, 100000, 500000],
                                           1500, 1e-16, deadline=deadline)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 6: Try n=600
    if tleft() > 250:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2

        if tleft() > 150:
            deadline = time.time() + min(70, tleft() * 0.45)
            h_pol2, sc2 = slsqp_optimize(h2, n2, target2,
                                         [1000, 5000, 30000, 100000, 400000],
                                         1500, 1e-16, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2
                current_n = n2

                                    max_sweeps=15000,
                                    n_cand_start=20, n_cand_end=70,
                                    max_no_improve=350,

    if tleft() > 150:
        scales = [0.01, 0.02, 0.04, 0.07, 0.12, 0.2]
        for s in scales:

                break
            h_pert = best_h + rng.randn(current_n) * s
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(22, tleft() * 0.35)

    # Phase 9: Final SLSQP polish
    if tleft() > 30:
        deadline = time.time() + min(25, tleft() * 0.7)
        h_final, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                           [50000, 200000, 800000, 2000000],

            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
