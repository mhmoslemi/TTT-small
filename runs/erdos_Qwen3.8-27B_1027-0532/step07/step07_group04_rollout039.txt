The minimum of a convex piecewise-linear function is where the subgradient crosses zero, which happens at a "kink" - i.e., where two components of (corr + delta*g) are equal and both are the maximum.

So the optimal delta is the one where the maximum is achieved by (at least) two lags simultaneously. We can find this by:
1. Computing all "critical" delta values where two components become equal
2. Evaluating max at each critical point and at the boundary
3. Taking the best

But this is O(M^2) where M = 2n-1, which is too expensive.

A faster approach: the convex piecewise-linear function f(delta) = max_k (corr[k] + delta * g[k]) has its minimum at the point where the slope changes from negative to positive. The slope at any point is g[k*] where k* is an argmax.

So we can do:
1. Start at delta = 0 (or the lower bound)
2. Find the argmax k*
3. If g[k*] < 0, we can increase delta (the max is decreasing)
4. The next "kink" is where corr[k*] + delta*g[k*] = corr[k**] for some k** that's the second max
5. delta_next = (corr[k**] - corr[k*]) / (g[k*] - g[k**])
6. Jump to delta_next and repeat

This is the "active set" method for LP, and it takes at most M iterations in the worst case but typically much fewer.

Actually, even simpler: since f(delta) is convex piecewise-linear, we can use the fact that the minimum is where the subgradient contains 0. The subgradient at delta is [min_{k in argmax} g[k], max_{k in argmax} g[k]]. If this interval contains 0, we're at the minimum.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def pnorm_objective(h, p=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.mean(corr ** p) ** (1.0 / p)

def pnorm_gradient(h, p=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = len(corr)
    mean_p = np.mean(corr ** p)
    f_val = dx * mean_p ** (1.0 / p)
    prefactor = dx * (1.0 / M) * p * mean_p ** (1.0 / p - 1.0)
    weights = prefactor * corr ** (p - 1)
    shifts = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in range(n):
        valid1 = (m + shifts >= 0) & (m + shifts < n)
        t1 = m + shifts[valid1]
        grad[m] += np.sum(weights[valid1] * one_minus_h[t1])
        valid2 = (m - shifts >= 0) & (m - shifts < n)
        t2 = m - shifts[valid2]
        grad[m] -= np.sum(weights[valid2] * h[t2])
    return f_val, grad

_pnorm_state = {}

def _smooth_fn(x):
    return pnorm_objective(x, _pnorm_state['p'])

def _smooth_jac(x):
    _, g = pnorm_gradient(x, _pnorm_state['p'])
    return g

def _cons_fun(x):
    return np.sum(x) - _pnorm_state['target']

def _cons_jac(x):
    return np.ones(_pnorm_state['n'])

def slsqp_pnorm(h_start, n, target, p_values, maxiter_per_p, time_limit, start_time):
    global _pnorm_state
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    for p in p_values:
        if time.time() - start_time > time_limit:
            break
        _pnorm_state = {'p': p, 'target': target, 'n': n}
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(_smooth_fn, best_h, jac=_smooth_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_p, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def weighted_gradient(corr, h, n, beta, n_shifts):
    M = len(corr)
    mid = n - 1
    top_idx = np.argsort(corr)[-n_shifts:]
    c_max = corr[int(np.argmax(corr))]
    weights = np.exp(-beta * (c_max - corr[top_idx]))
    weights = weights / np.sum(weights)
    idxs = np.arange(n)
    grad = np.zeros(n)
    for wi, s_idx in enumerate(top_idx):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[wi] * h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad

def active_set_cd(h_init, n, target, time_limit, max_sweeps=50000,
                  n_cand_start=20, n_cand_end=50, beta=3.0, n_shifts=3,
                  max_no_improve=400, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        grad = weighted_gradient(corr, h, n, beta, n_shifts)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = find_best_delta(corr, g, delta_lo, delta_hi)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

            if consecutive_no_improve == 0 and beta_idx < len(betas) - 1:
                beta_idx += 1
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            if beta_idx < len(betas) - 1:
                beta_idx += 1
                consecutive_no_improve = 0
                h += rng.randn(n) * 0.01
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
            else:
                break

        current_ncand = min(n_cand, n // 2)
        grad = weighted_gradient(corr, h, n, betas[beta_idx], 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = find_best_delta(corr, g, delta_lo, delta_hi)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _pnorm_state, start_ref
    start_time = time.time()
    rng = np.random.RandomState(seed)
    start_ref = start_time + budget_s

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = project_feasible(h0, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with p-norm at n=400
    if tleft() > 120:
        h_pol, sc = slsqp_pnorm(best_h, n0, target0,
                                [10, 20, 50, 100],
                                600,
                                time_limit=tleft() * 0.7,
                                start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active-set CD at n=400 (main workhorse)
    if tleft() > 200:
        h_cd, c5_cd = active_set_cd(best_h, n0, target0,
                                    time_limit=min(300, tleft() * 0.7),
                                    max_sweeps=80000,
                                    n_cand_start=20, n_cand_end=50,
                                    beta=3.0, n_shifts=3,
                                    max_no_improve=400, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Multi-beta CD for escape
    if tleft() > 150:
        h_mb, c5_mb = multi_beta_cd(best_h, n0, target0,
                                    time_limit=min(150, tleft() * 0.7),
                                    max_sweeps=30000,
                                    n_cand=40, max_no_improve=250, rng=rng)
        if c5_mb < best_score:
            best_score = c5_mb
            best_h = h_mb

    # Phase 4: SLSQP refinement with higher p
    if tleft() > 100:
        h_pol, sc = slsqp_pnorm(best_h, n0, target0,
                                [50, 100, 200],
                                800,
                                time_limit=tleft() * 0.7,
                                start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Another CD round
    if tleft() > 150:
        h_cd2, c5_cd2 = active_set_cd(best_h, n0, target0,
                                      time_limit=min(150, tleft() * 0.7),
                                      max_sweeps=50000,
                                      n_cand_start=25, n_cand_end=55,
                                      beta=5.0, n_shifts=4,
                                      max_no_improve=300, rng=rng)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Try higher resolution n=600
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd6, c5_cd6 = active_set_cd(h1, n1, target1,
                                      time_limit=min(200, tleft() * 0.6),
                                      max_sweeps=40000,
                                      n_cand_start=20, n_cand_end=45,
                                      beta=3.0, n_shifts=3,
                                      max_no_improve=350, rng=rng)
        if c5_cd6 < best_score:
            best_score = c5_cd6
            best_h = h_cd6
            n0 = n1
            target0 = target1

        if tleft() > 80:
            h_pol6, sc6 = slsqp_pnorm(best_h, n0, target0,
                                      [50, 100],
                                      600,
                                      time_limit=tleft() * 0.6,
                                      start_time=start_time)
            if sc6 < best_score:
                best_score = sc6
                best_h = h_pol6

    # Phase 7: Final CD polish
    if tleft() > 100:
        h_cd7, c5_cd7 = active_set_cd(best_h, n0, target0,
                                      time_limit=min(80, tleft() * 0.7),
                                      max_sweeps=30000,
                                      n_cand_start=30, n_cand_end=60,
                                      beta=8.0, n_shifts=4,
                                      max_no_improve=200, rng=rng)
        if c5_cd7 < best_score:
            best_score = c5_cd7
            best_h = h_cd7

    # Phase 8: Final SLSQP
    if tleft() > 60:
        h_fp, sc_fp = slsqp_pnorm(best_h, n0, target0,
                                  [100, 200, 500],
                                  500,
                                  time_limit=tleft() * 0.7,
                                  start_time=start_time)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    # Phase 9: Final rapid CD
    if tleft() > 30:
        h_cf, c5_cf = active_set_cd(best_h, n0, target0,
                                    time_limit=min(20, tleft() * 0.7),
                                    max_sweeps=10000,
                                    n_cand_start=40, n_cand_end=60,
                                    beta=10.0, n_shifts=4,
                                    max_no_improve=100, rng=rng)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
