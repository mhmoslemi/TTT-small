    h = np.clip(h.astype(float), 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)
    return h

    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

    total = np.sum(exp_vals)
    weights = exp_vals / total
    n_corr = len(corr)
    mid = n - 1

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]

                h_cur = h_new

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=50, n_top=5, n_bisect=12,
            max_no_improve=300, restart_every=150, restart_scale=0.015):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(999)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if no_improve > restart_every:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            no_improve = 0

        # Top shifts for gradient
        top_idx = np.argsort(corr)[-n_top:]

        # Weighted gradient
        top_vals = corr[top_idx]
        top_sum = np.sum(np.abs(top_vals))
        if top_sum < 1e-15:
            w = np.ones(n_top) / n_top
        else:
            w = top_vals / top_sum
        grad = np.zeros(n)
        for s, wi in zip(top_idx, w):
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate size
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        top_corr = corr[top_idx]
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ci = change[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                g = change[j] - ci
                top_g = g[top_idx]
                a, b = d_lo, d_hi
                for _ in range(n_bisect):
                    m = 0.5 * (a + b)
                    vals = top_corr + m * top_g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = np.max(top_g[active])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m
                d = 0.5 * (a + b)
                if abs(d) < 1e-13:
                    continue
                approx = np.max(corr + d * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

Actually, one more idea: **use the projected gradient method as a fast initial optimization before SLSQP**. The projected gradient is:
1. Compute subgradient g of f at h
2. h_new = h - α * g
3. Project h_new onto {x: 0≤x≤1, sum(x)=target}
4. If f(h_new) < f(h), accept; else reduce α

The projection onto {x: 0≤x≤1, sum(x)=target} can be done in O(n log n) time (sort and find the threshold). This is much faster than SLSQP per iteration, and might find a good starting point for SLSQP.

But actually, the subgradient of f(h) = max_k corr_k is just the gradient of the active corr (or a convex combination of active ones). Computing this is O(n * n_top) which is fast.

def projected_subgrad(h_init, n, target, time_limit, max_iter=10000):
    start = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.1  # initial step size
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0/n)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Subgradient: gradient of the active correlation
        k_star = int(np.argmax(corr)) - (n - 1)
        idxs = np.arange(n)
        g = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        g[v1] += (1.0 - h[idxs[v1] - k_star])
        g[v2] -= h[idxs[v2] + k_star]
        g -= np.mean(g)  # project out the constraint direction
        
        # Take a step
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_val:
            h = h_new
            alpha = min(alpha * 1.1, 0.5)  # increase step
        else:
            alpha *= 0.7  # decrease step
        
        if alpha < 1e-8:
            break
