def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    n = len(x)

        progress = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        if it > 0:
            t_curr = (1 + np.sqrt(1 + 4 * momentum_param**2)) / 2

            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        momentum_param = t_curr if it > 0 else 0.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        i1 = max(0, -k)
        i2 = min(n, n - k)
        if i2 <= i1:
            continue
        grad[i1:i2] += w * (1.0 - x[i1 + k:i2 + k])
        grad[i1 + k:i2 + k] -= w * x[i1:i2]
    return grad, float(c_max + np.log(np.sum(exp_shifted)) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def nesterov_pg(h_init, n, target, time_limit, alpha_start=100, alpha_end=8000,
                max_iter=80000, lr_base=0.008, momentum_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_prev = h.copy()
    t_nest = 1.0
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break
        progress = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            beta = (t_nest - 1.0) / t_new
            y = h + beta * (h - h_prev)
            t_nest = t_new
        else:
            y = h.copy()

        grad, _ = smooth_obj_grad(y, alpha)
        grad -= np.mean(grad)

        lr = lr_base * (1.0 + 0.5 * progress) / (1.0 + 0.3 * alpha / alpha_start)
        h_new = project_feasible(y - lr * grad, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_prev = h
        h = h_new
        t_nest *= momentum_decay
        t_nest = max(t_nest, 1.0)

    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=30, n_cand_end=60, max_no_improve=500, top_k=6):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(top_k, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            h_i = h[i]
            if h_i < 1e-12:
                continue
            hr = hi_rows[ii]
            for jj in range(n_l):
                j = int(low_set[jj])
                h_j = h[j]
                if h_j > 1.0 - 1e-12:
                    continue
                lr = lo_rows[jj]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                a_d = delta_min
                b_d = delta_max
                for _ in range(8):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid > c5_val + 1e-14:
                        b_d = mid_d
                        continue
                    active_mask = vals >= f_mid - 1e-12
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, max_steps=20000, block_sizes=(2, 3, 4, 5, 8)):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, rng, mode):
    h_pert = h.copy()
    if mode == 0:
        block_size = rng.randint(5, 20)
        pos = rng.randint(0, n - block_size)
        src = h_pert[pos:pos + block_size].copy()
        dst_pos = rng.randint(0, n - block_size)
        if dst_pos != pos:
            h_pert[pos:pos + block_size] = h_pert[dst_pos:dst_pos + block_size]
            h_pert[dst_pos:dst_pos + block_size] = src
    elif mode == 1:
        scale = 1.0 + rng.uniform(-0.1, 0.1)
        h_pert = h * scale
    elif mode == 2:
        center = rng.randint(0, n)
        width = rng.randint(20, 60)
        lo = max(0, center - width)
        hi = min(n, center + width)
        h_pert[lo:hi] += rng.uniform(-0.05, 0.05)
    elif mode == 3:
        idx = rng.choice(n, size=rng.randint(5, 30), replace=False)
        h_pert[idx] += rng.uniform(-0.08, 0.08, size=len(idx))
    elif mode == 4:
        period = rng.randint(37, 73)
        h_pert += 0.03 * np.sin(2.0 * np.pi * np.arange(n) / period)
    return project_feasible(h_pert, target)

                              max_no_improve=600, top_k=6)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish at n=800
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n1, target1, [500, 2000, 10000, 40000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Block CD at n=800
    if tleft() > 150:
        t_limit = min(120, tleft() * 0.5)
        h_bc, c5_bc = block_cd(best_h, n1, target1, time_limit=t_limit)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 7: More pair CD
    if tleft() > 150:
        t_limit = min(120, tleft() * 0.5)
        h_cd2, c5_cd2 = pair_cd(best_h, n1, target1, time_limit=t_limit,
                                max_sweeps=30000, n_cand_start=35, n_cand_end=70,
                                max_no_improve=500, top_k=8)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Nesterov PG again
    if tleft() > 100:
        t_limit = min(80, tleft() * 0.5)
        h_pg2, c5_pg2 = nesterov_pg(best_h, n1, target1, time_limit=t_limit,
                                    alpha_start=500, alpha_end=20000,
                                    max_iter=40000, lr_base=0.004)
        if c5_pg2 < best_score:
            best_score = c5_pg2
            best_h = h_pg2

    # Phase 9: SLSQP polish
    if tleft() > 80:
        h_pol2 = slsqp_round(best_h, n1, target1, [2000, 10000, 40000],
                             1000, 1e-14)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 10: Try higher resolution n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 200:
            t_limit = min(120, tleft() * 0.5)
            h_cd3, c5_cd3 = pair_cd(h2, n2, target2, time_limit=t_limit,
                                    max_sweeps=20000, n_cand_start=20, n_cand_end=50,
                                    max_no_improve=400, top_k=5)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd2
                best_h = h_cd3

        if tleft() > 100:
            t_limit = min(60, tleft() * 0.5)
            h_pg3, c5_pg3 = nesterov_pg(best_h, n2, target2, time_limit=t_limit,
                                        alpha_start=200, alpha_end=8000,
                                        max_iter=30000, lr_base=0.005)
            if c5_pg3 < best_score:
                best_score = c5_pg3
                best_h = h_pg3

        if tleft() > 60:
            h_pol3 = slsqp_round(best_h, n2, target2, [2000, 8000, 30000],
                                 600, 1e-14)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 11: Perturbation and restart
    if tleft() > 120:
        for mode in range(5):
            if tleft() < 50:
                break
            h_pert = structured_perturb(best_h, len(best_h), len(best_h) / 2.0,
                                        rng, mode)
            t_limit = min(40, tleft() * 0.4)
            h_ref = slsqp_round(h_pert, len(best_h), len(best_h) / 2.0,
                                [500, 3000, 15000], 500, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if tleft() > 30:
                t_limit2 = min(20, tleft() * 0.3)
                h_cd_p, c5_cd_p = pair_cd(h_ref, len(best_h), len(best_h) / 2.0,
                                          time_limit=t_limit2, max_sweeps=5000,
                                          n_cand_start=25, n_cand_end=50,
                                          max_no_improve=200, top_k=5)
                if c5_cd_p < best_score:
                    best_score = c5_cd_p
                    best_h = h_cd_p

    # Phase 12: Final pair CD
    if tleft() > 60:
        n_cur = len(best_h)
        t_limit = min(40, tleft() * 0.6)
        h_final_cd, c5_final_cd = pair_cd(best_h, n_cur, n_cur / 2.0,
                                          time_limit=t_limit, max_sweeps=10000,
                                          n_cand_start=40, n_cand_end=70,
                                          max_no_improve=300, top_k=7)
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    # Phase 13: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        h_fin = slsqp_round(best_h, n_cur, n_cur / 2.0,
                            [5000, 20000, 80000], 600, 1e-14)
        sc_fin = compute_c5(h_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
