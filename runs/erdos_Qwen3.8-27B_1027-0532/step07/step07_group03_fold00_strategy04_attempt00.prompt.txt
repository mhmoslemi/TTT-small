def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_pg(h_init, n, target, time_limit, alpha_init=500.0, alpha_end=50000.0,
                max_iters=20000, momentum=0.9):

    dx = 2.0 / n
    mid = n - 1
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        alpha = alpha_init * (alpha_end / alpha_init) ** (it / max_iters)
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p = p / np.sum(p)

        idxs = np.arange(n)
        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-15:
                continue

            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        step_backup = 0.01
        h_new = y - step_backup * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < compute_c5(h_curr) + 1e-10:
            h_prev = h_curr.copy()
            h_curr = h_new
