            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def interp_cubic(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    # Use cubic interpolation
    cs = CubicSpline(x_old, h_src, bc_type='not-a-knot')
    h_out = cs(x_new)
    h_out = np.clip(h_out, 0.0, 1.0)
    return project_feasible(h_out, target_dst)

def gradient_flow(h_init, n, target, time_limit, lr_init=0.01, lr_decay=0.999,
                  n_active=5, max_iter=50000):
    """Simple projected gradient method with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient using top active lags
        top_idxs = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for s_idx in top_idxs:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_idxs))

        # Project gradient step
        h_trial = h - lr * grad
        h_trial = project_feasible(h_trial, target)

        # Check if it improved
        trial_c5 = compute_c5(h_trial)
        if trial_c5 < c5_val:
            h = h_trial
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.9
            # Try a smaller step
            h_trial2 = h - lr * grad
            h_trial2 = project_feasible(h_trial2, target)
            trial_c5_2 = compute_c5(h_trial2)
            if trial_c5_2 < c5_val:
                h = h_trial2
            else:
                lr *= 0.5

    return best_h, best_c5

def frank_wolfe_step(h, grad, target, n):
    """Solve the linear oracle: min g·h' s.t. 0≤h'≤1, sum(h')=target."""
    sorted_g = np.sort(grad)
    # Find the threshold: set h'=1 for smallest g values, 0 for largest
    # Need sum = target, so target positions get h'=1, one gets fractional
    if target >= n:
        return np.ones(n)
    if target <= 0:
        return np.zeros(n)
    
    # Binary search for threshold
    # h'[i] = 1 if g[i] < thr, 0 if g[i] > thr, fractional if g[i] == thr
    lo, hi = float(np.min(grad)), float(np.max(grad))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_trial = np.where(grad < mid, 1.0, 0.0)
        s = np.sum(h_trial)
        if s < target:
            lo = mid
        else:
            hi = mid
    
    thr = 0.5 * (lo + hi)
    h_fw = np.where(grad < thr, 1.0, 0.0)
    # Fix the fractional part
    diff = target - np.sum(h_fw)
    if diff > 1e-12:
        # Need to set some entries to fractional
        frac_mask = grad >= thr
        n_frac = int(np.sum(frac_mask))
        if n_frac > 0:
            h_fw[frac_mask] = diff / n_frac
            h_fw = np.clip(h_fw, 0.0, 1.0)
    elif diff < -1e-12:
        frac_mask = grad < thr
        n_frac = int(np.sum(frac_mask))
        if n_frac > 0:
            # Take away from the largest among those set to 1
            h_fw[frac_mask] = 1.0 + diff / n_frac
            h_fw = np.clip(h_fw, 0.0, 1.0)
    
    return project_feasible(h_fw, target)

def fw_heuristic(h_init, n, target, time_limit, n_active=5, max_iter=20000):
    """Frank-Wolfe inspired heuristic for the non-convex problem."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    t = 0.0  # FW step size parameter

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        t = 2.0 / (2.0 + it + 1)  # NW step size

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        top_idxs = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for s_idx in top_idxs:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_idxs))

        # FW step: find vertex, then move toward it
        h_vertex = frank_wolfe_step(h, grad, target, n)
        h_new = (1.0 - t) * h + t * h_vertex
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < c5_val:
            h = h_new

    return best_h, best_c5

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = weights > 1e-10
    for k_idx in np.where(active)[0]:
        w = weights[k_idx]
        k = int(k_idx) - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if v1.any():
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
        if v2.any():
            grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd_multilag(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_start=30, n_cand_end=80, top_k_eval=50,
                     max_no_improve=600, n_active_lags=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_lag_idxs = np.argsort(corr)[-n_active_lags:]
        k_star_idx = int(top_lag_idxs[-1])

        grad = np.zeros(n)
        for s_idx in top_lag_idxs:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_lag_idxs))
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        for s_idx in top_lag_idxs[:-1]:
            ch_h = change[high_set, s_idx]
            ch_l = change[low_set, s_idx]
            score_mat += 0.4 * (ch_l[None, :] - ch_h[:, None])

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interp_pchip(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    pchip = PchipInterpolator(x_old, h_src)
    h_out = pchip(x_new)
    h_out = np.clip(h_out, 0.0, 1.0)
    return project_feasible(h_out, target_dst)


def gradient_descent_bb(h_init, n, target, time_limit, n_active=5,
                        max_iter=30000, lr_init=0.005):
    """Projected gradient with Barzilai-Borwein step size."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    prev_h = h.copy()
    prev_grad = None
    lr = lr_init

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idxs = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for s_idx in top_idxs:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_idxs))

        if prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            s_norm = np.dot(s, s)
            y_norm = np.dot(y, y)
            if s_norm > 1e-16 and y_norm > 1e-16:
                sy = np.dot(s, y)
                if abs(sy) > 1e-16:
                    lr_bb = sy / y_norm
                    lr = 0.5 * max(1e-6, min(lr_bb, 0.05))
                else:
                    lr *= 0.7
            else:
                lr *= 0.8

        h_trial = project_feasible(h - lr * grad, target)
        trial_c5 = compute_c5(h_trial)

        if trial_c5 <= c5_val:
            prev_h = h.copy()
            prev_grad = grad.copy()
            h = h_trial
        else:
            lr *= 0.5
            h_trial = project_feasible(h - lr * grad, target)
            trial_c5 = compute_c5(h_trial)
            if trial_c5 <= c5_val:
                prev_h = h.copy()
                prev_grad = grad.copy()
                h = h_trial
            else:
                prev_h = h.copy()
                prev_grad = grad.copy()

    return best_h, best_c5


def equalize_top_lags(h_init, n, target, time_limit, n_lags=4,
                      max_iter=20000, lr=0.002):
    """Try to equalize the top n_lags correlation values."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        top_idxs = np.argsort(corr)[-n_lags:]
        top_vals = corr[top_idxs]
        spread = top_vals.max() - top_vals.min()

        if spread < 1e-10:
            break

        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] - top_vals.min()
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_trial = project_feasible(h - lr * grad, target)
        trial_c5 = compute_c5(h_trial)
        if trial_c5 < c5_val:
            h = h_trial
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.8

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target, _g_alpha
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP at initial resolution
    tl = _time_left(start_time, budget_s)
    if tl > 50:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [200, 800, 3000],
                               300, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: BB gradient descent at initial resolution
    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_gd, c5_gd = gradient_descent_bb(best_h, n0, target0,
                                          time_limit=min(45, tl * 0.7),
                                          n_active=5, lr_init=0.004)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd

    # Phase 3: Upsample to n=700
    tl = _time_left(start_time, budget_s)
    if tl > 350:
        n1 = 700
        target1 = n1 / 2.0
        h1 = interp_pchip(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        # SLSQP at n=700
        tl = _time_left(start_time, budget_s)
        if tl > 180:
            h_pol1 = slsqp_optimize(best_h, n1, target1,
                                    [300, 1000, 4000],
                                    500, 1e-14)
            sc = compute_c5(h_pol1)
            if sc < best_score:
                best_score = sc
                best_h = h_pol1

        # Pair CD at n=700
        tl = _time_left(start_time, budget_s)
        if tl > 130:
            h_cd, c5_cd = pair_cd_multilag(best_h, n1, target1,
                                           time_limit=min(110, tl * 0.8),
                                           max_sweeps=40000,
                                           n_cand_start=30, n_cand_end=75,
                                           top_k_eval=55, max_no_improve=500,
                                           n_active_lags=5)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        # Equalize top lags
        tl = _time_left(start_time, budget_s)
        if tl > 40:
            h_eq, c5_eq = equalize_top_lags(best_h, n1, target1,
                                            time_limit=min(30, tl * 0.7),
                                            n_lags=4, lr=0.002)
            if c5_eq < best_score:
                best_score = c5_eq
                best_h = h_eq

        # More SLSQP
        tl = _time_left(start_time, budget_s)
        if tl > 50:
            h_pol1b = slsqp_optimize(best_h, n1, target1,
                                     [5000, 15000],
                                     400, 1e-14)
            sc = compute_c5(h_pol1b)
            if sc < best_score:
                best_score = sc
                best_h = h_pol1b

    # Phase 4: Upsample to n=1000
    tl = _time_left(start_time, budget_s)
    if tl > 350:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interp_pchip(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0

        # SLSQP at n=1000
        tl = _time_left(start_time, budget_s)
        if tl > 130:
            h_pol2 = slsqp_optimize(best_h, n_cur, tgt_cur,
                                    [500, 2000, 6000],
                                    400, 1e-14)
            sc = compute_c5(h_pol2)
            if sc < best_score:
                best_score = sc
                best_h = h_pol2

        # Pair CD at n=1000
        tl = _time_left(start_time, budget_s)
        if tl > 100:
            h_cd2, c5_cd2 = pair_cd_multilag(best_h, n_cur, tgt_cur,
                                             time_limit=min(85, tl * 0.8),
                                             max_sweeps=25000,
                                             n_cand_start=35, n_cand_end=80,
                                             top_k_eval=55, max_no_improve=400,
                                             n_active_lags=5)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        # BB gradient
        tl = _time_left(start_time, budget_s)
        if tl > 40:
            h_gd2, c5_gd2 = gradient_descent_bb(best_h, n_cur, tgt_cur,
                                                time_limit=min(30, tl * 0.7),
                                                n_active=6, lr_init=0.003)
            if c5_gd2 < best_score:
                best_score = c5_gd2
                best_h = h_gd2

        # SLSQP polish
        tl = _time_left(start_time, budget_s)
        if tl > 40:
            h_pol2b = slsqp_optimize(best_h, n_cur, tgt_cur,
                                     [10000, 30000],
                                     350, 1e-14)
            sc = compute_c5(h_pol2b)
            if sc < best_score:
                best_score = sc
                best_h = h_pol2b

    # Phase 5: Restart from perturbation
    tl = _time_left(start_time, budget_s)
    if tl > 70:
        rng = np.random.RandomState(seed)
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        for noise_scale in [0.02, 0.05]:
            tl = _time_left(start_time, budget_s)
            if tl < 35:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, tgt_cur)
            h_ref = slsqp_optimize(h_pert, n_cur, tgt_cur,
                                   [1000, 5000, 15000],
                                   350, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 6: Final CD
    tl = _time_left(start_time, budget_s)
    if tl > 50:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cd_final, c5_final = pair_cd_multilag(best_h, n_cur, tgt_cur,
                                                time_limit=min(40, tl * 0.75),
                                                max_sweeps=20000,
                                                n_cand_start=40, n_cand_end=90,
                                                top_k_eval=65, max_no_improve=300,
                                                n_active_lags=6)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 7: Final SLSQP with very high alpha
    tl = _time_left(start_time, budget_s)
    if tl > 30:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_final = slsqp_optimize(best_h, n_cur, tgt_cur,
                                 [20000, 60000, 100000],
                                 300, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
