               break

           progress = it / total_iters
           alpha = alpha_start * (alpha_end / alpha_start) ** progress

           h_look = np.clip(h_look, 0.0, 1.0)

    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    cv[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    cv[neg_mask] -= h[p - shifts[neg_mask]]
    return cv


def greedy_pair_cd(h_init, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            idxs_pert = rng.choice(n, size=max(5, n // 20), replace=False)
            h[idxs_pert] += rng.randn(len(idxs_pert)) * 0.04
            h = project_feasible(h, target)
            no_improve = 0
            continue
        top_k = 8
        if no_improve > 1000:
            top_k = min(20, 2 * n - 1)
        top_idxs = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idxs]
        top_max = np.max(top_vals)
        temp = 5.0 + 20.0 * (no_improve / 2000.0)
        weights = np.exp((top_vals - top_max) * temp)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(len(top_idxs))
            weights[0] = 1.0
        else:
            weights /= wsum
        grad = np.zeros(n)
        for si in range(len(top_idxs)):
            w = weights[si]
            if w < 1e-6:
                continue
            s = int(top_idxs[si])
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        if rng.random() < 0.1:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
        else:
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                sorted_idx = np.argsort(grad)
                i = int(sorted_idx[-1])
                j = int(sorted_idx[0])
                if i == j:
                    continue
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        g = change_vector(h, j, n) - change_vector(h, i, n)
        a_d = delta_lo
        b_d = delta_hi
        for _ in range(20):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-12
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-10:
                b_d = mid_d
            else:
                a_d = mid_d
        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue
        h[i] -= delta_star
        h[j] += delta_star
        h = project_feasible(h, target)
    return best_h, best_c5


def slsqp_polish(h_init, n, target, time_limit, seed=42):
    start = time.time()
    _gn = n
    _gt = target

    def _obj(x):
        obj, _ = smooth_obj_grad(x, 10000.0)
        return obj

    def _cons_fun(x):
        return np.sum(x) - _gt

    def _cons_jac(x):
        return np.ones(_gn)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    alphas = [1000, 5000, 20000, 80000]
    for alpha in alphas:
        if time.time() - start > time_limit - 5:
            break

        def _smooth_obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient-based optimization at original resolution
    if tleft() > 200:
        p1_time = min(600, tleft() * 0.55)
        h_p1, c5_p1 = gradient_optimize(best_h, n0, target0, p1_time, seed=seed)
        if c5_p1 < best_c5:
            best_c5 = c5_p1
            best_h = h_p1

    # Phase 2: Greedy pair CD at original resolution
    if tleft() > 150:
        p2_time = min(300, tleft() * 0.4)
        h_p2, c5_p2 = greedy_pair_cd(best_h, n0, target0, p2_time, seed=seed + 100)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2

    # Phase 3: SLSQP polish
    if tleft() > 80:
        p3_time = min(100, tleft() * 0.4)
        h_p3, c5_p3 = slsqp_polish(best_h, n0, target0, p3_time, seed=seed + 200)
        if c5_p3 < best_c5:
            best_c5 = c5_p3
            best_h = h_p3

    # Phase 4: Interpolate to higher resolution and optimize
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        c5_1 = compute_c5(h1)
        if tleft() > 200:
            p4_time = min(200, tleft() * 0.5)
            h_p4, c5_p4 = gradient_optimize(h1, n1, target1, p4_time, seed=seed + 300)
            if c5_p4 < c5_1:
                c5_1 = c5_p4
                h1 = h_p4
        if tleft() > 100:
            p4b_time = min(80, tleft() * 0.4)
            h_p4b, c5_p4b = greedy_pair_cd(h1, n1, target1, p4b_time, seed=seed + 400)
            if c5_p4b < c5_1:
                c5_1 = c5_p4b
                h1 = h_p4b
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 5: Try even higher resolution if time permits
    if tleft() > 200:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        c5_2 = compute_c5(h2)
        if tleft() > 150:
            p5_time = min(120, tleft() * 0.5)
            h_p5, c5_p5 = gradient_optimize(h2, n2, target2, p5_time, seed=seed + 500)
            if c5_p5 < c5_2:
                c5_2 = c5_p5
                h2 = h_p5
        if tleft() > 60:
            p5b_time = min(50, tleft() * 0.4)
            h_p5b, c5_p5b = greedy_pair_cd(h2, n2, target2, p5b_time, seed=seed + 600)
            if c5_p5b < c5_2:
                c5_2 = c5_p5b
                h2 = h_p5b
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 6: Final refinement
    if tleft() > 60:
        nf = len(best_h)
        tf = nf / 2.0
        p6_time = min(50, tleft() * 0.6)
        h_p6, c5_p6 = greedy_pair_cd(best_h, nf, tf, p6_time, seed=seed + 700)
        if c5_p6 < best_c5:
            best_c5 = c5_p6
            best_h = h_p6

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
