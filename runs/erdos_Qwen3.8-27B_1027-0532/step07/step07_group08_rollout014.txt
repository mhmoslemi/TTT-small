               n_cand=40, max_no_improve=500, n_top=5,
               reheat_every=200, reheat_scale=0.05):
    """CD that focuses on reducing the top-n_top correlations simultaneously."""
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Reheat
        if no_improve > reheat_every and no_improve % 50 == 0:
            rng = np.random.RandomState(sweep * 31 + 11)
            # Try multiple perturbations and keep the best
            best_pert = h
            best_pert_sc = c5_val
            for trial in range(5):
                n_pert = max(8, n // 25)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale * (trial + 1) * 0.3
                h_pert = project_feasible(h_pert, target)
                sc = compute_c5(h_pert)
                if sc < best_pert_sc + 0.001:
                    best_pert = h_pert.copy()
                    best_pert_sc = sc
            if best_pert_sc < c5_val + 1e-6:
                h = best_pert
                change = compute_change_matrix(h, n)
                no_improve = 0
                continue

        # Focus on top n_top correlations
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        
        # Weight: emphasize the max but include others
        w_max = np.max(top_vals)
        weights = (top_vals / max(w_max, 1e-20)) ** 2

        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        
        # Limit search for efficiency
        n_h = min(len(high_set), 30)
        n_l = min(len(low_set), 30)

            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.2 * np.max(np.abs(g) + 1e-30):
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(8):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:

        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, target, alphas, maxiters, deadline=None):
    global _g_n, _g_target
    n = len(h_start)
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiters, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def annealed_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=20, n_cand_end=55, max_no_improve=600,
                n_top=4, reheat_every=150, reheat_scale=0.05,
                seed_val=42):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed_val)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = reheat_scale * 0.5

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        temp = max(reheat_scale * 0.01, temp * 0.9995)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        if no_improve > reheat_every and no_improve % 40 == 0:
            best_pert = None
            best_pert_sc = c5_val + 1e-6
            for trial in range(8):
                scale = reheat_scale * (0.3 + 0.7 * rng.random())
                n_pert = max(8, n // 20)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.randn(n_pert) * scale
                h_pert = project_feasible(h_pert, target)
                sc = compute_c5(h_pert)
                if sc < best_pert_sc:
                    best_pert = h_pert.copy()
                    best_pert_sc = sc
            if best_pert is not None and best_pert_sc < c5_val + 1e-5:
                h = best_pert
                change = compute_change_matrix(h, n)
                no_improve = 0
                continue

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        w_max = max(np.max(top_vals), 1e-20)
        weights = (top_vals / w_max) ** 3

        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = min(len(high_set), 35)
        n_l = min(len(low_set), 35)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_max_abs = np.max(np.abs(g))
                if g_max_abs > 1e-30 and g[k_star_idx] > 0.25 * g_max_abs:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(9):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        elif temp > 1e-7 and rng.random() < 0.2:
            for _attempt in range(10):
                ci = rng.randint(n)
                cj = rng.randint(n)
                if ci != cj and abs(h[ci] - h[cj]) > 0.05:
                    if h[ci] > h[cj]:
                        d = (h[ci] - h[cj]) * 0.05
                        h[ci] -= d
                        h[cj] += d
                    else:
                        d = (h[cj] - h[ci]) * 0.05
                        h[cj] -= d
                        h[ci] += d
                    h = project_feasible(h, target)
                    new_sc = compute_c5(h)
                    delta_sc = new_sc - c5_val
                    if delta_sc <= 0 or rng.random() < np.exp(-delta_sc / max(temp, 1e-10)):
                        update_change_row(change, ci, h, n, shifts)
                        update_change_row(change, cj, h, n, shifts)
                        break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Aggressive SLSQP at native resolution
    if tleft() > 250:
        deadline = time.time() + min(150, tleft() * 0.4)
        h_pol, sc = slsqp_optimize(best_h, target0,
                                   [500, 2000, 10000, 40000, 100000, 200000],
                                   1500, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Annealed CD at native resolution
    if tleft() > 250:
        h_cd, c5_cd = annealed_cd(best_h, current_n, current_target,
                                  time_limit=min(180, tleft() * 0.5),
                                  max_sweeps=60000,
                                  n_cand_start=20, n_cand_end=55,
                                  max_no_improve=600, n_top=4,
                                  reheat_every=150, reheat_scale=0.05,
                                  seed_val=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Second SLSQP round
    if tleft() > 120:
        deadline = time.time() + min(80, tleft() * 0.4)
        h_pol2, sc2 = slsqp_optimize(best_h, current_target,
                                     [20000, 80000, 200000],
                                     2000, deadline=deadline)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 4: More CD with different parameters
    if tleft() > 150:
        h_cd2, c5_cd2 = annealed_cd(best_h, current_n, current_target,
                                    time_limit=min(100, tleft() * 0.4),
                                    max_sweeps=40000,
                                    n_cand_start=25, n_cand_end=65,
                                    max_no_improve=500, n_top=5,
                                    reheat_every=120, reheat_scale=0.04,
                                    seed_val=seed + 100)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Interpolate to n=600
    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 250:
            deadline = time.time() + min(80, tleft() * 0.3)
            h_pol1, sc_pol1 = slsqp_optimize(h1, target1,
                                             [1000, 5000, 20000, 80000],
                                             1000, deadline=deadline)
            best_h1 = h_pol1 if sc_pol1 < sc1 else h1
            best_sc1 = min(sc_pol1, sc1)

            h_cd1, c5_cd1 = annealed_cd(best_h1, n1, target1,
                                        time_limit=min(120, tleft() * 0.4),
                                        max_sweeps=35000,
                                        n_cand_start=20, n_cand_end=50,
                                        max_no_improve=500, n_top=4,
                                        reheat_every=130, reheat_scale=0.04,
                                        seed_val=seed + 200)
            if c5_cd1 < best_sc1:
                best_h1 = h_cd1
                best_sc1 = c5_cd1

            if tleft() > 60:
                deadline = time.time() + min(40, tleft() * 0.3)
                h_p1, sc_p1 = slsqp_optimize(best_h1, target1,
                                             [40000, 100000],
                                             1000, deadline=deadline)
                if sc_p1 < best_sc1:
                    best_h1 = h_p1
                    best_sc1 = sc_p1

            if best_sc1 < best_score:
                best_score = best_sc1
                best_h = best_h1
                current_n = n1
                current_target = target1

    # Phase 6: Interpolate to n=800
    if tleft() > 300:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2_base = compute_c5(h2)
        if tleft() > 200:
            deadline = time.time() + min(70, tleft() * 0.3)
            h_pol2, sc_pol2 = slsqp_optimize(h2, target2,
                                             [2000, 10000, 40000],
                                             1000, deadline=deadline)
            best_h2 = h_pol2 if sc_pol2 < sc2_base else h2
            best_sc2 = min(sc_pol2, sc2_base)

            h_cd2b, c5_cd2b = annealed_cd(best_h2, n2, target2,
                                          time_limit=min(100, tleft() * 0.35),
                                          max_sweeps=25000,
                                          n_cand_start=18, n_cand_end=45,
                                          max_no_improve=400, n_top=4,
                                          reheat_every=100, reheat_scale=0.04,
                                          seed_val=seed + 300)
            if c5_cd2b < best_sc2:
                best_h2 = h_cd2b
                best_sc2 = c5_cd2b

            if best_sc2 < best_score:
                best_score = best_sc2
                best_h = best_h2
                current_n = n2
                current_target = target2

    # Phase 7: Multi-start perturbation
    if tleft() > 120:
        for attempt in range(5):
            if tleft() < 30:
                break
            noise_scales = [0.02, 0.04, 0.07, 0.1, 0.15]
            h_pert = best_h + rng.randn(current_n) * noise_scales[attempt]
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(20, tleft() * 0.3)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_target,
                                           [2000, 10000, 40000],
                                           600, deadline=deadline)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 8: Final CD
    if tleft() > 60:
        h_cd_f, c5_f = annealed_cd(best_h, current_n, current_target,
                                   time_limit=min(40, tleft() * 0.5),
                                   max_sweeps=15000,
                                   n_cand_start=30, n_cand_end=70,
                                   max_no_improve=300, n_top=6,
                                   reheat_every=80, reheat_scale=0.03,
                                   seed_val=seed + 500)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 9: Final SLSQP polish
    if tleft() > 25:
        deadline = time.time() + min(15, tleft() * 0.5)
        h_final, sc_final = slsqp_optimize(best_h, current_target,
                                           [50000, 200000],
                                           1000, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
