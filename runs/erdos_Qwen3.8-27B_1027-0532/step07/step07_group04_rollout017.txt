    # Each constraint: grad_k · h' ≤ t - (corr_k(h) - grad_k · h)
    # i.e., grad_k · h' - t ≤ corr_k(h) - grad_k · h = corr_k(h) - corr_k(h) = 0
    # Wait, let me redo this.

    # We want: corr[k](h0) + grad_k · (h - h0) ≤ t for all active k
    # i.e.: grad_k · h ≤ t - corr[k](h0) + grad_k · h0
    # i.e.: grad_k · h - t ≤ corr[k](h0) - grad_k · h0
    # Note: corr[k](h0) - grad_k · h0 = constant (depends only on h0)
    
    A_rows = []
    b_vals = []
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] = (1.0 - h_cur[idxs[v1] + k_star])
        grad[v2] -= h_cur[idxs[v2] - k_star]

        rhs = corr[k_star_idx] - np.dot(grad, h_cur)
        A_rows.append(np.concatenate([grad, [-1.0]]))
        b_vals.append(rhs)
        
        # Also add all previously active constraints (or just the top few)
        # For efficiency, keep only top m constraints

        # Solve the LP
        # Variables: [h_0, ..., h_{n-1}, t]
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # A_eq: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [target]

        bounds = [(0, 1)] * n + [(None, None)]

            if res.success:

                if actual_max <= c_max + 1e-10:

            else:
                break

    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    v = np.zeros(n)  # momentum
    lr = 1.0
    momentum = 0.9
    mid = n - 1
    idxs = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        k_star = int(np.argmax(corr)) - mid

        grad = np.zeros(n)

        grad[v2] -= h[idxs[v2] - k_star]

        v = momentum * v - lr * grad

            if lr < 1e-15:
                break
            h_new = project_feasible(h + v, target)
            new_c5 = compute_c5(h_new)
            if new_c5 < c5_val:
                h = h_new

def lbfgs_optimize(h_init, n, target, time_limit, alpha=100, top_k=10, maxiter=300):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-k:]
        c_max = corr[top_idx].max()
        shifted = alpha * (corr[top_idx] - c_max)
        logsum = c_max + np.log(np.exp(np.clip(shifted, -500, 0)).sum()) / alpha
        penalty = 100.0 * (np.sum(x) - target) ** 2
        return logsum * dx + penalty
    
    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-k:]
        c_max = corr[top_idx].max()
        shifted = alpha * (corr[top_idx] - c_max)
        weights = np.exp(np.clip(shifted, -500, 0))
        weights /= weights.sum()

def fast_cd(h_init, n, target, time_limit, n_cand=50, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    while time.time() - start < time_limit:
        if consecutive_no_improve > max_no_improve:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        k_star = int(np.argmax(corr)) - mid
        
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] = (1.0 - h[idxs[v1] + k_star])
        grad[v2] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    n = _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(_g_top_k, len(corr))
    top_idx = np.argsort(corr)[-k:]
    c_max = corr[top_idx].max()
    shifted = _g_alpha * (corr[top_idx] - c_max)
    logsum = c_max + np.log(np.exp(np.clip(shifted, -500, 0)).sum()) / _g_alpha
    penalty = _g_penalty * (np.sum(x) - _g_target) ** 2
    return logsum * _g_dx + penalty

def _grad(x):
    n = _g_n
    idxs = _g_ids
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(_g_top_k, len(corr))
    top_idx = np.argsort(corr)[-k:]
    c_max = corr[top_idx].max()
    shifted = _g_alpha * (corr[top_idx] - c_max)
    weights = np.exp(np.clip(shifted, -500, 0))
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.ones(k) / k
    else:
        weights /= ws
    grad = np.zeros(n)
    for wi in range(k):
        ki = int(top_idx[wi]) - _g_mid
        w = weights[wi]
        v1 = (idxs + ki >= 0) & (idxs + ki < n)
        v2 = (idxs - ki >= 0) & (idxs - ki < n)
        grad[v1] += w * (1.0 - x[idxs[v1] + ki])
        grad[v2] -= w * x[idxs[v2] - ki]
    grad *= _g_dx
    grad += 2.0 * _g_penalty * (np.sum(x) - _g_target)
    return grad

def lbfgs_optimize(h_init, n, target, time_limit, alphas=None, top_k=10,
                   maxiter_per_alpha=200, penalty=500.0):
    global _g_n, _g_target, _g_alpha, _g_top_k, _g_ids, _g_mid, _g_dx, _g_penalty
    start = time.time()
    _g_n = n
    _g_target = target
    _g_top_k = top_k
    _g_ids = np.arange(n)
    _g_mid = n - 1
    _g_dx = 2.0 / n
    _g_penalty = penalty
    if alphas is None:
        alphas = [50, 200, 1000, 5000]
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _g_alpha = alpha
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': maxiter_per_alpha * 20})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_score

def lbfgs_optimize_multi_topk(h_init, n, target, time_limit, alphas=None,
                              top_k_list=None, maxiter_per_alpha=150, penalty=500.0):
    global _g_n, _g_target, _g_alpha, _g_top_k, _g_ids, _g_mid, _g_dx, _g_penalty
    start = time.time()
    _g_n = n
    _g_target = target
    _g_ids = np.arange(n)
    _g_mid = n - 1
    _g_dx = 2.0 / n
    _g_penalty = penalty
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000]
    if top_k_list is None:
        top_k_list = [5, 10, 20, 40]
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for tk in top_k_list:
        if time.time() - start > time_limit:
            break
        _g_top_k = tk
        for alpha in alphas:
            if time.time() - start > time_limit:
                break
            _g_alpha = alpha
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16,
                                       'gtol': 1e-10, 'maxfun': maxiter_per_alpha * 20})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass
    return best_h, best_score

def fast_cd(h_init, n, target, time_limit, n_cand=50, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    while time.time() - start < time_limit:
        if consecutive_no_improve > max_no_improve:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] = (1.0 - h[idxs[v1] + k_star])
        grad[v2] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in range(n):
                change[best_i, :] = 0.0
                pm = (best_i + shifts >= 0) & (best_i + shifts < n)
                nm = (best_i - shifts >= 0) & (best_i - shifts < n)
                change[best_i, pm] = (1.0 - h[best_i + shifts[pm]])
                change[best_i, nm] -= h[best_i - shifts[nm]]
                change[best_j, :] = 0.0
                pm = (best_j + shifts >= 0) & (best_j + shifts < n)
                nm = (best_j - shifts >= 0) & (best_j - shifts < n)
                change[best_j, pm] = (1.0 - h[best_j + shifts[pm]])
                change[best_j, nm] -= h[best_j - shifts[nm]]
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_lbfgs(h_best, n, target, time_limit, n_restarts=5, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    start = time.time()
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    alphas = [200, 1000, 5000, 20000]
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        scale = 0.03 * (r + 1)
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_opt, sc = lbfgs_optimize(h_pert, n, target,
                                   time_limit=min(30, time_limit - (time.time() - start)),
                                   alphas=alphas, top_k=15, maxiter_per_alpha=150)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
    return best_h, best_score

def projected_subgrad(h_init, n, target, time_limit, max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = 0.01
    momentum = 0.85
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            lr *= 0.95
        if no_improve > 2000:
            break
        if lr < 1e-7:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] = (1.0 - h[idxs[v1] + k_star])
        grad[v2] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val:
            h = h_new
        else:
            h_new2 = project_feasible(h + 0.5 * v, target)
            if compute_c5(h_new2) < c5_val:
                h = h_new2
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_top_k, _g_ids, _g_mid, _g_dx, _g_penalty
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS at n=400, multiple top-k and alpha
    if tleft() > 60:
        h_opt, sc = lbfgs_optimize_multi_topk(
            best_h, n0, target0, time_limit=min(50, tleft() * 0.85),
            alphas=[50, 200, 1000, 5000, 20000],
            top_k_list=[5, 10, 20, 40], maxiter_per_alpha=200, penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 2: Interpolate to n=800 and optimize
    if tleft() > 150:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_opt, sc = lbfgs_optimize_multi_topk(
            h1, n1, target1, time_limit=min(120, tleft() * 0.8),
            alphas=[200, 1000, 5000, 20000],
            top_k_list=[5, 10, 20, 40], maxiter_per_alpha=150, penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
            n0 = n1
            target0 = target1

    # Phase 3: CD polishing at current resolution
    if tleft() > 80:
        h_cd, sc = fast_cd(best_h, n0, target0,
                          time_limit=min(60, tleft() * 0.8),
                          n_cand=60, max_no_improve=200)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 4: Interpolate to n=1600 and optimize
    if tleft() > 200:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_opt, sc = lbfgs_optimize_multi_topk(
            h2, n2, target2, time_limit=min(150, tleft() * 0.8),
            alphas=[500, 2000, 10000, 50000],
            top_k_list=[5, 10, 20], maxiter_per_alpha=120, penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
            n0 = n2
            target0 = target2

    # Phase 5: CD at higher resolution
    if tleft() > 80:
        h_cd, sc = fast_cd(best_h, n0, target0,
                          time_limit=min(60, tleft() * 0.8),
                          n_cand=80, max_no_improve=200)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 6: Multi-restart to escape local optima
    if tleft() > 120:
        h_mr, sc = multi_restart_lbfgs(
            best_h, n0, target0, time_limit=min(100, tleft() * 0.8),
            n_restarts=4, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_mr

    # Phase 7: Projected subgradient
    if tleft() > 60:
        h_pg, sc = projected_subgrad(best_h, n0, target0,
                                     time_limit=min(45, tleft() * 0.8),
                                     max_iters=30000)
        if sc < best_score:
            best_score = sc
            best_h = h_pg

    # Phase 8: Try n=3200 if time allows
    if tleft() > 250:
        n3 = 3200
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        h_opt, sc = lbfgs_optimize_multi_topk(
            h3, n3, target3, time_limit=min(180, tleft() * 0.8),
            alphas=[1000, 5000, 20000],
            top_k_list=[5, 10, 20], maxiter_per_alpha=100, penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
            n0 = n3
            target0 = target3

    # Phase 9: Final L-BFGS with very high alpha
    if tleft() > 60:
        h_opt, sc = lbfgs_optimize(
            best_h, n0, target0, time_limit=min(45, tleft() * 0.8),
            alphas=[20000, 100000, 500000],
            top_k=5, maxiter_per_alpha=200, penalty=1000.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 10: Final CD polish
    if tleft() > 30:
        h_cd, sc = fast_cd(best_h, n0, target0,
                          time_limit=min(20, tleft() * 0.8),
                          n_cand=80, max_no_improve=100)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 11: One more L-BFGS pass
    if tleft() > 25:
        h_opt, sc = lbfgs_optimize(
            best_h, n0, target0, time_limit=min(15, tleft() * 0.8),
            alphas=[50000, 200000],
            top_k=5, maxiter_per_alpha=100, penalty=1000.0)
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
