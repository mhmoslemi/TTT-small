        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = int(stochastic_frac * n_pairs)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            if consecutive_no_improve > 100 and consecutive_no_improve % 100 == 0:
                idxs_pert = rng.choice(n, size=max(10, n // 15), replace=False)
                h[idxs_pert] += rng.randn(len(idxs_pert)) * 0.03
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, max_sweeps=5000,
             block_sizes=(3, 5, 8, 13), n_binsearch=15, seed=777):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 800:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        block_size = block_sizes[min(sweep % len(block_sizes), len(block_sizes) - 1)]
        best_trial_c5 = c5_val
        best_start = -1
        best_delta = 0.0
        n_starts = max(1, n - block_size + 1)
        n_try = min(n_try_starts := max(20, n // 5), n_starts)
        try_starts = rng.choice(n_starts, size=n_try, replace=False)
        for s_start in try_starts:
            s_start = int(s_start)
            src = np.arange(s_start, s_start + block_size)
            dst = src + k_star
            dst_valid = (dst >= 0) & (dst < n)
            src_valid = True
            if not dst_valid.all():
                continue
            overlap = np.intersect1d(src, dst)
            if len(overlap) > 0:
                continue
            g = np.zeros(n_corr)
            for idx in src:
                g -= change[int(idx)]
            for idx in dst:
                g += change[int(idx)]
            delta_lo = max(float(np.max(h[src])) - 1.0, -float(np.min(h[dst])))
            delta_hi = min(float(np.min(h[src])), 1.0 - float(np.max(h[dst])))
            if delta_hi - delta_lo < 1e-12:
                continue
            a_d, b_d = delta_lo, delta_hi
            for _ in range(n_binsearch):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_start = s_start
                best_delta = delta_star
        if best_start >= 0:
            src = np.arange(best_start, best_start + block_size)
            dst = src + k_star
            h[src] -= best_delta
            h[dst] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for idx in set(list(src) + list(dst)):
                update_change_row(change, int(idx), h, n, shifts)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_optimize(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]
    attempts = 0
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 40:
            break
        scale = scales[attempts % len(scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        alloc = min(remaining * 0.3, 60)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=alloc,
                               max_sweeps=3000, n_cand_start=30, n_cand_end=80,
                               max_no_improve=200, n_binsearch=16,
                               stochastic_frac=0.1, seed=int(rng.randint(99999)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        attempts += 1
    return best_h, best_c5
