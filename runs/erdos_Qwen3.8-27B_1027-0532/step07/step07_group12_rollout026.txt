def block_cd(h_init, n, target, time_limit, block_size=5, max_sweeps=50000):
    """Block coordinate descent: update a block of consecutive coordinates using line search."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Find the dominant shift
        k_star = int(np.argmax(corr)) - mid
        
        # Compute gradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Select block: centered around the most negative gradient coordinate
        worst_idx = int(np.argmin(grad))
        block_start = max(0, worst_idx - block_size // 2)
        block_end = min(n, block_start + block_size)
        block_start = max(0, block_end - block_size)
        block = np.arange(block_start, block_end)
        
        # Try moving the block in the -grad direction (projected)
        block_grad = grad[block]
        if np.max(-block_grad) < 1e-14:
            # No improvement possible in this direction for this block
            continue
        
        # Line search: h_block - lr * block_grad, projected to [0,1]
        best_lr = 0
        best_new_max = cur_max
        for lr in [0.01, 0.005, 0.001, 0.05, 0.0005]:
            h_try = h.copy()
            h_try[block] -= lr * block_grad
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            new_corr = np.correlate(h_try, 1.0 - h_try, mode='full')
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_lr = lr
        
        if best_lr > 0 and best_new_max < cur_max - 1e-15:
            h = h.copy()
            h[block] -= best_lr * block_grad
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]

                      jac=lse_gradient, method='L-BFGS-B',

        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)
        
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        
        if not improved and step > 1000:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B with low-to-medium alpha
    if tleft() > 80:
        for alpha in [100, 500, 2000]:
            if tleft() < 50:
                break
            hp, sc = lbfgs_optimize(best_h, n, target, alpha, maxiter=400, time_limit=min(45, tleft()*0.3), penalty=100.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
    
    # Phase 2: Pair CD single shift, many candidates
    if tleft() > 180:
        cd_time = min(220, tleft() * 0.28)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=45, n_shifts=1, max_sweeps=100000, stochastic_rate=0.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    
    # Phase 3: L-BFGS-B medium alpha
    if tleft() > 70:
        for alpha in [5000, 20000, 100000]:
            if tleft() < 40:
                break
            hp, sc = lbfgs_optimize(best_h, n, target, alpha, maxiter=250, time_limit=min(35, tleft()*0.3), penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
    
    # Phase 4: Pair CD with stochastic restarts for escape
    if tleft() > 100:
        cd_time = min(100, tleft() * 0.2)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, n_shifts=1, max_sweeps=60000, stochastic_rate=0.1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    
    # Phase 5: Block CD
    if tleft() > 80:
        for bs in [5, 8]:
            if tleft() < 40:
                break
            hbc, c5bc = block_cd(best_h, n, target, time_limit=min(50, tleft()*0.4), block_size=bs, max_sweeps=25000)
            if c5bc < best_score:
                best_score = c5bc
                best_h = hbc
    
    # Phase 6: Pair CD multi-shift
    if tleft() > 120:
        cd_time = min(120, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=50, n_shifts=3, max_sweeps=80000, stochastic_rate=0.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    
    # Phase 7: Subgradient with many shifts
    if tleft() > 60:
        hsg, c5sg = subgradient_optimize(best_h, n, target, time_limit=min(50, tleft()*0.3), n_shifts=7, base_lr=0.06, max_steps=100000)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg
    
    # Phase 8: L-BFGS-B very high alpha
    if tleft() > 80:
        for alpha in [500000, 2000000, 10000000]:
            if tleft() < 40:
                break
            hp, sc = lbfgs_optimize(best_h, n, target, alpha, maxiter=150, time_limit=min(35, tleft()*0.3), penalty=5000.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
    
    # Phase 9: Pair CD final polish at current resolution
    if tleft() > 60:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(50, tleft()*0.6), n_cand=35, n_shifts=1, max_sweeps=50000, stochastic_rate=0.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    
    # Phase 10: Higher resolution
    if tleft() > 280:
        n_new = 1000
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        n_use = n_new
        target_use = target_new
        for alpha in [100, 1000, 5000, 50000]:
            if tleft() < 70:
                break
            hp, sc = lbfgs_optimize(h_interp, n_use, target_use, alpha, maxiter=200, time_limit=min(50, tleft()*0.2), penalty=100.0)
            if sc < best_score:
                best_score = sc
                h_interp = hp
                best_h = hp
        if tleft() > 100:
            hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(100, tleft()*0.4), n_cand=30, n_shifts=1, max_sweeps=50000, stochastic_rate=0.0)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n_use = len(best_h)
                target_use = n_use / 2.0
        if tleft() > 50:
            hp, sc = lbfgs_optimize(best_h, n_use, target_use, 100000, maxiter=150, time_limit=min(40, tleft()*0.5), penalty=5000.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
    
    # Phase 11: Structured perturbation
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for freq in [3, 5, 7, 11]:
            if tleft() < 40:
                break
            x_axis = np.arange(n_use) / n_use
            perturbation = 0.008 * np.sin(2.0 * np.pi * freq * x_axis)
            h_pert = best_h + perturbation
            h_pert = project_feasible(h_pert, target_use)
            hp, sc = lbfgs_optimize(h_pert, n_use, target_use, 10000, maxiter=150, time_limit=min(20, tleft()*0.25), penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
    
    # Phase 12: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(25, tleft()*0.7), n_cand=25, n_shifts=1, max_sweeps=30000, stochastic_rate=0.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    
    # Final projection
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, n_use
