def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_capped_simplex(v, target):
    """Efficient projection onto {x : 0 <= x_i <= 1, sum(x) = target} using bisection."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5*(lo + hi), 0.0, 1.0)
    # Fix any numerical issues
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x > 1e-15) & (x < 1.0 - 1e-15)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)

def compute_smoothed_gradient(h, alpha):
    """Compute gradient of smoothed max of correlation."""
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]

def nesterov_gd(h_init, n, target, time_limit, alpha_start=50.0, alpha_end=5000.0,
                 lr_start=0.05, beta=0.9, max_steps=100000, seed=42):

    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    no_improve = 0
    max_no_improve = 5000
    
    for step in range(max_steps):

        if step % 500 == 0:

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Adaptive alpha (sharpening)
        progress = min(float(step) / max_steps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Adaptive learning rate (decreasing)
        lr = lr_start * (0.999 ** step)
        
        # Nesterov lookahead
        h_look = h + beta * momentum
        grad, _ = compute_smoothed_gradient(h_look, alpha)
        
        # Update momentum
        momentum = beta * momentum - lr * grad
        
        # Step
        h_new = h + momentum
        h_new = project_capped_simplex(h_new, target)
        
        # Occasional check
        if step % 100 == 0:
            cur_c5 = compute_c5(h_new)
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h_new.copy()
        h = h_new
    
    cur_c5 = compute_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5

_g_n = 0
_g_target = 0.0

    return float(np.max(corr)) * dx

def project_capped_simplex(v, target):

    if target <= 0:
        return np.zeros(n)

        return np.ones(n)
    lo = -2.0

def nesterov_gd(h_init, n, target, time_limit, alpha_start=30.0, alpha_end=10000.0,
                 lr_start=0.02, beta=0.85, max_steps=200000, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    no_improve = 0
    max_no_improve = 8000
    check_interval = 200
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step % check_interval == 0:
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_interval
            if no_improve > max_no_improve:
                break
        progress = min(float(step) / max_steps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (0.9995 ** step)
        h_look = h + beta * momentum
        grad, _, _ = compute_grad_sm(h_look, alpha, top_k=50)
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = project_capped_simplex(h_new, target)
        h = h_new
    cur_c5 = compute_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_capped_simplex(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=120, max_no_improve=500,
            n_binsearch=20, stochastic_frac=0.05, alpha_sm=100.0, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        shifted_w = np.clip(alpha_sm * (corr - np.max(corr)), -500.0, 0.0)
        weights = np.exp(shifted_w)

        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = int(stochastic_frac * n_pairs)
        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                lj_val = h[j]
                delta_lo = max(-lj_val, hi_val - 1.0)
                delta_hi = min(hi_val, 1.0 - lj_val)
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_i = np.zeros(n_corr)
                m_i = i - mid
                for s in active:
                    w = weights[s]
                    m = s - mid
                    if m >= 0:
                        if i >= m and i < n:
                            g_i[s] += w * (1.0 - h[i - m])
                        if i + m < n:
                            g_i[s] -= w * h[i + m]
                    else:
                        q = -m
                        if i < n - q:
                            g_i[s] += w * (1.0 - h[i + q])
                        if i >= q:
                            g_i[s] -= w * h[i - q]
                g_j = np.zeros(n_corr)
                for s in active:
                    w = weights[s]
                    m = s - mid
                    if m >= 0:
                        if j >= m:
                            g_j[s] += w * (1.0 - h[j - m])
                        if j + m < n:
                            g_j[s] -= w * h[j + m]
                    else:
                        q = -m
                        if j < n - q:
                            g_j[s] += w * (1.0 - h[j + q])
                        if j >= q:
                            g_j[s] -= w * h[j - q]
                g = g_j - g_i
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                hi_val = h[i]
                lj_val = h[j]
                delta_lo = max(-lj_val, hi_val - 1.0)
                delta_hi = min(hi_val, 1.0 - lj_val)
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                for idx in [i, j]:
                    sign = -1.0 if idx == i else 1.0
                    for s in active:
                        w = weights[s]
                        m = s - mid
                        if m >= 0:
                            if idx >= m:
                                g[s] += sign * w * (1.0 - h[idx - m])
                            if idx + m < n:
                                g[s] -= sign * w * h[idx + m]
                        else:
                            q = -m
                            if idx < n - q:
                                g[s] += sign * w * (1.0 - h[idx + q])
                            if idx >= q:
                                g[s] -= sign * w * h[idx - q]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_capped_simplex(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            if consecutive_no_improve % 200 == 0 and consecutive_no_improve > 100:
                n_pert = max(n // 20, 10)
                idxs_pert = rng.choice(n, size=n_pert, replace=False)
                h[idxs_pert] += rng.randn(n_pert) * 0.03
                h = project_capped_simplex(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=40, n_cand_end=150, max_no_improve=600,
                 n_binsearch=20, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    alpha_sm = 200.0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        shifted_w = np.clip(alpha_sm * (corr - np.max(corr)), -500.0, 0.0)
        weights = np.exp(shifted_w)
        wsum = np.sum(weights)
        if wsum > 1e-30:
            weights /= wsum
        else:
            weights = np.zeros(n_corr)
            weights[int(np.argmax(corr))] = 1.0
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for s in active:
            w = weights[s]
            m = s - mid
            if m >= 0:
                grad[m:] += w * (1.0 - h[:n - m])
                grad[:n - m] -= w * h[m:]
            else:
                q = -m
                grad[:n - q] += w * (1.0 - h[q:])
                grad[q:] -= w * h[:n - q]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                lj_val = h[j]
                delta_lo = max(-lj_val, hi_val - 1.0)
                delta_hi = min(hi_val, 1.0 - lj_val)
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                for s in active:
                    w = weights[s]
                    m = s - mid
                    if m >= 0:
                        if i >= m:
                            g[s] += w * (1.0 - h[i - m])
                        if i + m < n:
                            g[s] -= w * h[i + m]
                        if j >= m:
                            g[s] -= w * (1.0 - h[j - m])
                        if j + m < n:
                            g[s] += w * h[j + m]
                    else:
                        q = -m
                        if i < n - q:
                            g[s] += w * (1.0 - h[i + q])
                        if i >= q:
                            g[s] -= w * h[i - q]
                        if j < n - q:
                            g[s] -= w * (1.0 - h[j + q])
                        if j >= q:
                            g[s] += w * h[j - q]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_capped_simplex(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            if consecutive_no_improve % 300 == 0 and consecutive_no_improve > 150:
                n_pert = max(n // 15, 10)
                idxs_pert = rng.choice(n, size=n_pert, replace=False)
                h[idxs_pert] += rng.randn(n_pert) * 0.02
                h = project_capped_simplex(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_capped_simplex(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_capped_simplex(h_out, target_dst)

def multi_restart(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.01, 0.02, 0.03, 0.05, 0.08]
    attempt = 0
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 60:
            break
        scale = scales[attempt % len(scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_capped_simplex(h_pert, target)
        alloc = min(remaining * 0.4, 120)
        h_cd, c5_cd = fast_pair_cd(h_pert, n, target, time_limit=alloc,
                                     max_sweeps=20000,
                                     n_cand_start=30, n_cand_end=100,
                                     max_no_improve=300,
                                     n_binsearch=18,
                                     seed=int(rng.randint(10000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        attempt += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_score = compute_c5(best_h)

    if tleft() > 180:
        h_gd, sc_gd = nesterov_gd(best_h, n0, target0,
                                   time_limit=min(150, tleft() * 0.4),
                                   alpha_start=30.0, alpha_end=5000.0,

                                   max_steps=200000, seed=seed)
        if sc_gd < best_score:
            best_score = sc_gd
            best_h = h_gd

        h_cd, c5_cd = fast_pair_cd(best_h, n0, target0,
                                    time_limit=min(200, tleft() * 0.5),
                                    max_sweeps=50000,
                                    n_cand_start=40, n_cand_end=150,
                                    max_no_improve=500,
                                    n_binsearch=20,seed=42)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:

        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 150:

    if tleft() > 300:

            if sc_gd1 < best_score:

        if tleft() > 150:
