        grad[v1] += 1.0 - h[idxs[v1] - k_star]
        grad[v2] -= h[idxs[v2] + k_star]

        # For zeros: low grad means adding helps most
        ones_cand = ones_idx[np.argsort(-grad[ones_idx])[:n_c]]

            else:
                no_improve += 1
        else:
            no_improve += 1

        if n_c < 2:
            break

        ones_cand = ones_idx[np.argsort(-grad[ones_idx])[:n_c]]
        zeros_cand = zeros_idx[np.argsort(grad[zeros_idx])[:n_c]]

        best_new = cur_c5max
        best_i = -1
        best_j = -1

        for ii in range(n_c):
            i = int(ones_cand[ii])
            deltas = change[zeros_cand] - change[i]
            new_maxs = np.max(corr[np.newaxis, :] + deltas, axis=1)
            j_local = int(np.argmin(new_maxs))
            if new_maxs[j_local] < best_new:
                best_new = new_maxs[j_local]
                best_i = i
                best_j = int(zeros_cand[j_local])

        for _ in range(n_random):
            i = int(ones_idx[rng.randint(len(ones_idx))])
            j = int(zeros_idx[rng.randint(len(zeros_idx))])
            new_val = float(np.max(corr + change[j] - change[i]))
            if new_val < best_new:
                best_new = new_val
                best_i = i
                best_j = j

        if best_i < 0:
            no_improve += 1
            if no_improve > 3000:
                break
            continue

        delta = best_new - cur_c5max
        if delta <= 0 or (T > 0 and rng.rand() < np.exp(-delta / T)):
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = corr + change[best_j] - change[best_i]
            update_row(change, best_i, h, n, shifts)
            update_row(change, best_j, h, n, shifts)
            cur_c5max = float(np.max(corr))
            cur_c5 = cur_c5max * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def multi_seed_binary_sa(h_init, n, time_limit, n_seeds=3, n_cand=50):
    start_time = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    per_seed = time_limit / n_seeds
    for s in range(n_seeds):
        remaining = time_limit - (time.time() - start_time)
        if remaining < 10:
            break
        h_s, c5_s = binary_sa(h_init, n, min(per_seed, remaining),
                              seed=42 + s * 100, T_start=0.02,
                              T_end=0.0003, max_iters=200000, n_cand=n_cand)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init_raw = np.array(initial_h_values, dtype=float)
    n0 = len(h_init_raw)
    target0 = n0 // 2

    h_bin = to_binary(h_init_raw, target0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    if tleft() > 150:
        h1, c5_1 = binary_sa(h_bin, n0, min(140, tleft()), seed=seed,
                             T_start=0.02, T_end=0.0003,
                             max_iters=300000, n_cand=50, n_random=3)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    if tleft() > 100:
        h2, c5_2 = binary_sa(best_h, n0, min(90, tleft()), seed=seed + 500,
                             T_start=0.03, T_end=0.001,
                             max_iters=200000, n_cand=60, n_random=5,
                             greedy_frac=0.1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    if tleft() > 200:
        n1 = 600
        h_interp = interpolate_binary(best_h, n1)
        h3, c5_3 = binary_sa(h_interp, n1, min(180, tleft()), seed=seed + 1000,
                             T_start=0.015, T_end=0.0002,
                             max_iters=200000, n_cand=45, n_random=3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    if tleft() > 100:
        h4, c5_4 = binary_sa(best_h, len(best_h), min(90, tleft()),
                             seed=seed + 2000, T_start=0.01,
                             T_end=0.0001, max_iters=150000,
                             n_cand=50, n_random=3, greedy_frac=0.5)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    if tleft() > 200:
        n2 = 800
        h_interp2 = interpolate_binary(best_h, n2)
        h5, c5_5 = binary_sa(h_interp2, n2, min(180, tleft()), seed=seed + 3000,
                             T_start=0.012, T_end=0.0002,
                             max_iters=150000, n_cand=40, n_random=2)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    if tleft() > 120:
        n_cur = len(best_h)
        h6, c5_6 = multi_seed_binary_sa(best_h, n_cur, min(110, tleft()),
                                        n_seeds=2, n_cand=55)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    if tleft() > 60:
        h7, c5_7 = binary_sa(best_h, len(best_h), min(55, tleft()),
                             seed=seed + 9999, T_start=0.005,
                             T_end=0.00005, max_iters=100000,
                             n_cand=60, n_random=2, greedy_frac=0.7)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    final_h = to_binary(best_h, len(best_h) // 2)
    final_c5 = compute_c5(final_h)

    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, len(final_h)
