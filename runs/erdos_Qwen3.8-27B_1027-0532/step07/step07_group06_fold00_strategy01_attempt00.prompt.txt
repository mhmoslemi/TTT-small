def block_cd(h_init, n, target, time_limit, start_time, max_sweeps=5000, block_size=3):

        else:
            no_improve += 1

        if no_improve > max_sweeps // 3:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        # Pick the block starting near the most negative gradient
        window = block_size
        best_block_start = -1
        best_block_score = np.inf
        for start in range(0, n - window + 1, max(1, (n - window) // 500)):
            block_grad = grad[start:start + window]
            score = np.sum(block_grad ** 2)
            if score < best_block_score:
                best_block_score = score
                best_block_start = start

        if best_block_start < 0:
            best_block_start = 0

        block_idx = np.arange(best_block_start, best_block_start + window)
        h_fixed = h.copy()
        h_block = h[block_idx].copy()
        target_block = target - np.sum(h_fixed) + np.sum(h_block)

            h_trial = h_fixed.copy()

            return float(cm + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / 500.0)

                           bounds=bounds_block,

        except Exception:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=20000, base_scale=0.03):

        scale = base_scale * max(0.05, 1.0 - step / max_steps)

def smooth_surrogate(x, alpha):  
    g = np.correlate(x, 1-x, mode='full')  
    c_max = np.max(g)  
    if c_max < 1e-12:  
        return 0.0  
    shift = alpha*(g - c_max)  
    np.clip(shift, -500.0, 0.0, out=shift)  
    return float(c_max + np.log(1 + np.sum(np.exp(shift - np.max(shift)))/alpha)  
```

def project_feasible(x, target):

    dx = 2.0/len(x)

        if abs(steps) < 5e-10:  
            break  
        x_new = np.clip(x + steps, 0, 1)  
        x = x_new  
        s_new = np.sum(x) - target  
        if abs(s_new) < 1e-8:  
            return x  
        s = s_new  
    return x  
```
