def fast_pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Gradient for active shifts
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        # Two-stage search
        # Stage 1: Quick evaluation of all pairs
        n_h = len(high_set)
        n_l = len(low_set)
        
        # Precompute change vectors for candidate sets
        high_changes = change[high_set]  # (n_h, n_corr)
        low_changes = change[low_set]    # (n_l, n_corr)
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # For each pair, compute g = change[j] - change[i]
        # Quick estimate: try delta = midpoint
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                g = low_changes[jj] - high_changes[ii]
                
                # Quick midpoint evaluation
                mid_delta = 0.5 * (delta_lo + delta_hi)
                approx_max_quick = np.max(corr + mid_delta * g)
                
                # Only do full binary search if promising
                if approx_max_quick < best_trial_c5 / dx:
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(14):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        if f_mid < best_trial_c5 / dx:
                            active_mask = vals >= f_mid - 1e-14
                            if active_mask.any():
                                f_prime = np.max(g[active_mask])
                            else:
                                f_prime = 0.0
                            if f_prime > 1e-12:
                                b_d = mid_d
                            else:
                                a_d = mid_d
                        else:
                            b_d = mid_d
                    
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-13:
                        continue
                    
                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial_c5 / dx:
                        best_trial_c5 = float(approx_max) * dx
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5
