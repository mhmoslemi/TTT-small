import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0
_g_beta = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _obj_balanced(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)
    if _g_beta > 0:
        sorted_c = np.sort(corr)[::-1]
        top_k = min(30, len(sorted_c))
        val += _g_beta * float(np.std(sorted_c[:top_k]))
    return val

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_opt(h_start, n, target, alphas, betas, maxiter, use_balance=False):
    global _g_n, _g_target, _g_alpha, _g_beta
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    obj_fn = _obj_balanced if use_balance else _obj_smooth
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        for beta in betas:
            _g_beta = beta
            try:
                res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                              constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturbation(h, n, target, rng):
    """Apply a structured perturbation (segment swap or flip)."""
    h_new = h.copy()
    choice = rng.randint(0, 3)
    if choice == 0:
        # Swap two segments
        len_seg = rng.randint(n // 10, n // 4)
        start1 = rng.randint(0, n - len_seg)
        start2 = rng.randint(0, n - len_seg)
        if start1 != start2 and abs(start1 - start2) >= len_seg:
            h_new[start1:start1+len_seg], h_new[start2:start2+len_seg] = \
                h_new[start2:start2+len_seg].copy(), h_new[start1:start1+len_seg].copy()
    elif choice == 1:
        # Invert a segment
        len_seg = rng.randint(n // 8, n // 3)
        start = rng.randint(0, n - len_seg)
        h_new[start:start+len_seg] = 1.0 - h_new[start:start+len_seg]
    else:
        # Local noise
        len_seg = rng.randint(n // 5, n // 2)
        start = rng.randint(0, n - len_seg)
        h_new[start:start+len_seg] += rng.randn(len_seg) * 0.1
    return project_feasible(h_new, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _obj_balanced(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)
    if _g_beta > 0:
        sorted_c = np.sort(corr)[::-1]
        top_k = min(30, len(sorted_c))
        val += _g_beta * float(np.std(sorted_c[:top_k]))
    return val

    return np.sum(x) - _g_target

def slsqp_opt(h_start, n, target, alphas, betas, maxiter, use_balance=False):
    global _g_n, _g_target, _g_alpha, _g_beta
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    obj_fn = _obj_balanced if use_balance else _obj_smooth
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        for beta in betas:
            _g_beta = beta
            try:
                res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if _tl() > 100:
        h_pol, sc = slsqp_opt(best_h, current_n, current_target,
                              [100, 500, 2000], [0.0], 500, use_balance=False)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 350:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(380, _tl() * 0.55),
                              max_sweeps=60000,
                              n_cand_start=25, n_cand_end=130,
                              max_no_improve=800,
                              bisection_iters=16,
                              restart_every=200, restart_scale=0.02,
                              seed=999)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tl() > 120:
        h_pol, sc = slsqp_opt(best_h, current_n, current_target,
                              [2000, 10000, 50000], [0.02, 0.05, 0.1], 800,
                              use_balance=True)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 200:
        h_ms, c5_ms = multi_shift_cd(best_h, current_n, current_target,
                                     time_limit=min(220, _tl() * 0.55),
                                     max_sweeps=25000, n_cand=60, n_shifts=5,
                                     seed=42)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if _tl() > 150:
        h_bc, c5_bc = block_cd(best_h, current_n, current_target,
                               time_limit=min(130, _tl() * 0.5),
                               max_sweeps=15000, block_size=3, n_cand=50,
                               bisection_iters=14, seed=777)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    if _tl() > 200:
        rng = np.random.RandomState(seed)
        for attempt in range(6):
            if _tl() < 80:
                break
            h_pert = structured_perturbation(best_h, current_n, current_target, rng)
            h_ref, sc_ref = slsqp_opt(h_pert, current_n, current_target,
                                       [100, 500, 2000], [0.0], 400, use_balance=False)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if _tl() > 100:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(50, _tl() * 0.4),
                                       max_sweeps=10000,
                                       n_cand_start=30, n_cand_end=90,
                                       max_no_improve=400,
                                       bisection_iters=14,
                                       seed=1000 + attempt)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if _tl() > 300:
        n1 = 512
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if _tl() > 200:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                    time_limit=min(200, _tl() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=20, n_cand_end=80,
                                    max_no_improve=500,
                                    bisection_iters=14,
                                    seed=555)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1
            if current_n == n1 and _tl() > 60:
                h_pol1, sc1 = slsqp_opt(best_h, n1, target1,
                                        [5000, 20000], [0.01, 0.05], 600,
                                        use_balance=True)
                if sc1 < best_score:
                    best_score = sc1
                    best_h = h_pol1

    if _tl() > 120:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(100, _tl() * 0.6),
                                       max_sweeps=20000,
                                       n_cand_start=40, n_cand_end=140,
                                       max_no_improve=500,
                                       bisection_iters=16,
                                       seed=333)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if _tl() > 50:
        h_final_pol, sc_final = slsqp_opt(best_h, current_n, current_target,
                                           [50000, 200000], [0.0, 0.01], 800,
                                           use_balance=False)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
