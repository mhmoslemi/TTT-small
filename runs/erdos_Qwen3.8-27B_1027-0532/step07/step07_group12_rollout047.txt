0 ≤ h[i] ≤ 1
sum(h) = n/2

This is a linear program (LP) since the linearized constraints are linear in h and t!

```python
def linearization_step(h, n, target, n_constraints=200):
    """
    Solve the linearized problem to find a descent direction.
    """
    from scipy.optimize import linprog
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    # Select most important constraints (top n_constraints)
    top_k = np.argsort(-corr)[:n_constraints]
    
    # Build the LP: minimize t
    # Variables: [h[0], ..., h[n-1], t]
    # Objective: minimize t → c = [0, ..., 0, 1]
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    # Constraints: for each selected shift k:
    # corr[k] + Σ_i grad_corr_k[i] * (h[i] - h[i]) <= t
    # → Σ_i grad_corr_k[i] * h[i] - t <= -corr[k]
    # → -Σ_i grad_corr_k[i] * h[i] + t >= corr[k]
    # For linprog (minimize c^T x subject to A_ub x <= b_ub):
    # Σ_i grad_corr_k[i] * h[i] - t <= -corr[k]
    
    A_ub = np.zeros((n_constraints, n_vars))
    b_ub = np.zeros(n_constraints)
    
    for ii, kk in enumerate(top_k):
        k_shift = int(kk) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
        v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
        if np.any(v1):
            grad[v1] = (1.0 - h[idxs[v1] - k_shift])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_shift]
        
        A_ub[ii, :n] = grad * dx
        A_ub[ii, -1] = -1.0
        b_ub[ii] = -corr[kk] * dx
    
    # Sum constraint: sum(h) = n/2
    # For linprog: sum(h) <= n/2 AND -sum(h) <= -n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h[i] <= 1, t is free (but we can bound it)
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]  # The proposed h values
    except:
        pass
    return None
```

```python
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

            continue

        for ci in range(nc):
            m = int(all_cands[ci])
            idx1 = m + k_all
            idx2 = m - k_all
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                profiles[ci, valid2] -= h[idx2[valid2]]
        
        cand_map = {int(all_cands[ci]): ci for ci in range(nc)}

        best_i = -1
        best_j = -1
        best_delta = 0.0

            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_map[i]

                    continue

                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue

                g = profiles[ci] - profiles[cj]
                
                # Binary search (10 iterations)
                lo = 0.0
                hi = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (lo + hi)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-13)
                    if np.any(active):
                        if float(np.max(g[active])) > 1e-12:
                            hi = mid_d
                        else:
                            lo = mid_d
                    else:
                        break
                
                delta_star = 0.5 * (lo + hi)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            sum_diff = target - np.sum(h)
            if abs(sum_diff) > 1e-10:
                h = project_feasible(h, target)
```

Let me also think about the LP approach more carefully.

For the LP:
- Variables: h[0], ..., h[n-1], t (total n+1 variables)
- Objective: minimize t
- Constraints:
  - For top K shifts: Σ_i grad_k[i]*dx*h[i] - t ≤ -b_k (K constraints)
  - sum(h) = n/2 (1 equality)
  - 0 ≤ h[i] ≤ 1 (n bound pairs)

With n=600, K=100: 601 variables, ~101 inequality constraints + 1 equality. This is a small LP that should solve in milliseconds.

The direction from the LP: d = h_LP - h_current
Line search: minimize C₅(h_current + α*d) for α ∈ [0, 1]

This is a 1D minimization that we can do with golden section search.

Wait, I'm confusing myself with the RHS. Let me redo:

Constraint: corr[k](h) * dx ≤ t

Linearization: corr[k](h) ≈ corr[k](h₀) + g·(h - h₀) where g = ∇h corr[k]

So: (corr[k](h₀) + g·h - g·h₀) * dx ≤ t
→ g*dx·h - t ≤ t_max - (corr[k](h₀) - g·h₀)*dx

→ -t ≤ -corr[k](h₀)*dx
→ t ≥ corr[k](h₀)*dx ✓

```python
def lp_step(h, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    n_corr = 2*n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((n_corr, n_vars))
    b_ub = np.zeros(n_corr)
    
    for kk in range(n_corr):
        k_shift = kk - mid
        grad = np.zeros(n)
        v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
        v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
        if np.any(v1):
            grad[v1] = (1.0 - h[idxs[v1] - k_shift])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_shift]
        
        A_ub[kk, :n] = grad * dx
        A_ub[kk, n] = -1.0
        b_ub[kk] = (np.sum(grad * h) - corr[kk]) * dx
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    except:
        pass
    return None
```
