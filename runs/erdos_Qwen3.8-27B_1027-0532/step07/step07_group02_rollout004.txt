```python
def subgradient_optimize(h_init, n, target, time_limit, 
                         n_steps=50000, base_lr=0.01, seed=42):
    """Projected subgradient descent with Nesterov momentum."""
    start = time.time()
    rng = np.random.RandomState(seed)
    h_cur = h_init.copy().astype(float)
    h_prev = h_cur.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    
    for t in range(n_steps):
        if time.time() - start > time_limit:
            break
        # Nesterov momentum
        momentum = (t - 1) / (t + 2) if t > 0 else 0.0
        y = h_cur + momentum * (h_cur - h_prev)
        
        # Compute subgradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        active = corr >= c_max - 1e-10
        n_active = active.sum()
        
        # Average gradient over active shifts
        mid = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        for s in np.where(active)[0]:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y[idxs[v1] - k]) / n_active
            grad[v2] -= y[idxs[v2] + k] / n_active
        
        # Adaptive step size
        lr = base_lr / (1.0 + t * 0.001)
        
        # Projected update
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Update
        h_prev = h_cur
        h_cur = h_new
        
        # Track best
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    
    return best_h, best_c5
```

```python
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        
        # Best swap: move 1 mass from high-grad one to low-grad zero
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score each potential swap
        best_delta_c5 = c5_val
        best_i = -1
        best_j = -1
        
        # Try top candidates
        ones_sorted = ones[np.argsort(grad[ones])][-30:]  # highest grad among ones
        zeros_sorted = zeros[np.argsort(grad[zeros])][:30]  # lowest grad among zeros
        
        for i in ones_sorted:
            for j in zeros_sorted:
                h_trial = h.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                c5_trial = compute_c5(h_trial)

                    best_delta_c5 = c5_trial
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
        else:
            consecutive_no_improve += 1
    
    return best_h, best_c5
```

Actually, this binary approach is O(n^2) per evaluation (since compute_c5 is O(n^2) for the correlation), and we're doing 30*30 = 900 evaluations per sweep. That's too slow.

Let me instead use the change matrix approach for the binary case:
- If we flip h[i] from 1 to 0 and h[j] from 0 to 1, the change is:
  new_corr[s] = corr[s] + (change[j, s] - change[i, s])
  where change is defined with h[i] going from 1→0 (so we need to be careful)

Actually, for binary h, if h[i]=1 and h[j]=0:
- Changing h[i] to 0: corr changes by -change[i] (removing the contribution of h[i]=1)
  Wait, change[p] is defined as the change in corr when h[p] increases by 1.
  So decreasing h[i] by 1 changes corr by -change[i].
  Increasing h[j] by 1 changes corr by +change[j].
  Total change: change[j] - change[i].

So the new max is max_s (corr[s] + change[j][s] - change[i][s]).

This is O(n_corr) per evaluation, and we have 900 pairs, so 900 * 800 = 720K per sweep. That's reasonable.

```python
def iterated_local_search(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    
    # Adaptive perturbation schedule
    base_scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
    
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        
        # Choose perturbation scale (biased towards smaller when we're close to time limit)
        scale_idx = rng.randint(len(base_scales))
        scale = base_scales[scale_idx]
        
        # Create perturbed solution
        # Option 1: Gaussian noise
        if rng.random() < 0.5:
            h_pert = best_h + rng.randn(n) * scale
        else:
            # Option 2: Block perturbation (perturb a random contiguous block)
            block_size = rng.randint(n // 20, n // 4)
            block_start = rng.randint(0, n - block_size)
            h_pert = best_h.copy()
            h_pert[block_start:block_start + block_size] += rng.randn(block_size) * scale
            # Also try a "swap" perturbation
            n_swaps = rng.randint(3, 15)
            for _ in range(n_swaps):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    tmp = h_pert[i]
                    h_pert[i] = h_pert[j]
                    h_pert[j] = tmp
        
        h_pert = project_feasible(h_pert, target)
        
        # Run short CD on perturbed solution
        cd_time = min(remaining * 0.3, 60)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=cd_time,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=400,
            n_binsearch=20, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])

        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1

                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:

                approx_max = np.max(corr + delta_star * g) * dx

            h = np.clip(h, 0.0, 1.0)

    return best_h, best_c5

            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            if consecutive_no_improve % 40 == 0 and consecutive_no_improve > 60:
                n_pert = max(n // 15, 5)
                idxs_pert = rng.choice(n, size=n_pert, replace=False)
                h[idxs_pert] += rng.randn(n_pert) * 0.03
                h = project_feasible(h, target)
    return best_h, best_c5

def iterated_local_search(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 35:
            break
        scale = scales[rng.randint(len(scales))]
        pert_type = rng.random()
        if pert_type < 0.4:
            h_pert = best_h + rng.randn(n) * scale
        elif pert_type < 0.7:
            h_pert = best_h.copy()
            block_size = rng.randint(n // 15, n // 3)
            block_start = rng.randint(0, n - block_size)
            h_pert[block_start:block_start + block_size] += rng.randn(block_size) * scale
        else:
            h_pert = best_h.copy()
            n_swaps = rng.randint(5, 25)
            for _ in range(n_swaps):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
            h_pert += rng.randn(n) * scale * 0.3
        h_pert = project_feasible(h_pert, target)
        cd_time = min(remaining * 0.25, 50)
        if cd_time < 10:
            break
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=cd_time,
                               max_sweeps=5000, n_cand_start=30, n_cand_end=90,
                               max_no_improve=150, n_binsearch=16,
                               stochastic_frac=0.15, seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5

def downsample_upsample(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    x_fine = np.linspace(0.0, 1.0, n)
    for n_coarse in [80, 120, 160, 200]:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 50:
            break
        x_coarse = np.linspace(0.0, 1.0, n_coarse)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        target_coarse = n_coarse / 2.0
        h_coarse = project_feasible(h_coarse, target_coarse)
        cd_time = min(remaining * 0.35, 35)
        h_c_opt, _ = pair_cd(h_coarse, n_coarse, target_coarse,
                              time_limit=cd_time, max_sweeps=10000,
                              n_cand_start=10, n_cand_end=25,
                              max_no_improve=200, n_binsearch=14,
                              stochastic_frac=0.2, seed=int(rng.randint(100000)))
        h_up = np.interp(x_fine, x_coarse, h_c_opt)
        h_up = project_feasible(h_up, target)
        cd_time2 = min(remaining * 0.2, 25)
        h_f_opt, c5_f = pair_cd(h_up, n, target,
                                 time_limit=cd_time2, max_sweeps=4000,
                                 n_cand_start=30, n_cand_end=80,
                                 max_no_improve=150, n_binsearch=16,
                                 stochastic_frac=0.1, seed=int(rng.randint(100000)))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f_opt
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        h_tmp, sc_tmp = slsqp_optimize(best_h, n, target, [50, 200, 800], 800, 1e-15)
        if sc_tmp < best_score:

                                 stochastic_frac=0.03, seed=42)

        if sc_tmp < best_score:

    elapsed = time.time() - start_time

        if sc_tmp < best_score:

                                 stochastic_frac=0.05, seed=501)

    final_h = project_feasible(best_h, target)

    return final_h, final_score, n
```
