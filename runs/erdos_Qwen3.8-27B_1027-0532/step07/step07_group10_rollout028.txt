        weights_k /= np.sum(weights_k)

        grad = np.zeros(n)
        for wi, si in zip(weights_k, top_k_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        # Quick screening: estimate improvement using only the max correlation shift
        k_max = int(np.argmax(corr)) - mid
        scores = np.zeros(n_h * n_l)
        count = 0
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    scores[count] = -1e30
                else:
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-13:
                        scores[count] = -1e30
                    else:
                        g_val = (change[j, k_max] - change[i, k_max])
                        if g_val > 0:
                            scores[count] = -1e30
                        else:
                            est_improve = -g_val * delta_hi
                            scores[count] = est_improve
                count += 1

        # Get top pairs from screening
        top_pairs = np.argsort(scores)[-min(30, n_h * n_l):]
        top_pairs = top_pairs[scores[top_pairs] > -1e29]

        for pi in range(len(top_pairs)):
            pidx = int(top_pairs[pi])
            ii = pidx // n_l
            jj = pidx % n_l
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(16):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-16:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_polish(h_init, n, target, time_limit, max_steps=80000,
                    base_lr=0.01, n_active=8, lr_decay_patience=400):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > lr_decay_patience:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[-1]]
        alpha_w = 40.0
        w = np.exp(alpha_w * (corr[top_shifts] - c_max))
        w /= np.sum(w)

        grad = np.zeros(n)
        for wi, si in zip(w, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-16:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, max_sweeps=10000,
             block_size=4, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 500:
            break

        # Pick random block
        b_start = rng.randint(0, max(1, n - block_size))
        block = np.arange(b_start, b_start + block_size)

        # Find best delta to add to this block (and remove from its complement)
        # We add delta to the block and subtract from a random other block
        b2_start = rng.randint(0, max(1, n - block_size))
        while b2_start + block_size <= b_start and b2_start >= b_start - block_size:
            b2_start = rng.randint(0, max(1, n - block_size))
        block2 = np.arange(b2_start, b2_start + block_size)

        # Combined change: add delta to block, subtract from block2
        g = np.zeros(n_corr)
        for p in block:
            g += change[p]
        for p in block2:
            g -= change[p]

        # Binary search for best delta
        delta_lo = max(-(min(h[p2] for p2 in block2)), -1.0)
        # More carefully: we need h[block] + delta <= 1 and h[block2] - delta >= 0
        delta_lo = max(-min(h[block2].min()), 0.0 - h[block2].min() * 0)
        # Actually: h[p] + delta <= 1 for p in block => delta <= 1 - max(h[block])
        #            h[p] - delta >= 0 for p in block2 => delta <= min(h[block2])
        #            h[p] + delta >= 0 for p in block => delta >= -min(h[block])
        #            h[p] - delta <= 1 for p in block2 => delta >= min(h[block2]) - 1
        d_lo = max(-min(h[block].min()), min(h[block2].min()) - 1.0)
        d_hi = min(1.0 - max(h[block].max()), min(h[block2].min()))

        if d_hi - d_lo < 1e-13:
            continue

        a_d = d_lo
        b_d = d_hi
        for _ in range(14):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-13:
            continue

        # Try the move
        h_try = h.copy()
        h_try[block] += delta_star
        h_try[block2] -= delta_star
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)

        sc_tried = compute_c5(h_try)
        if sc_tried < c5_val - 1e-15:
            h = h_try
            for p in block:
                update_change_row(change, p, h, n, shifts)
            for p in block2:
                update_change_row(change, p, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Polish at n=400
    if tleft() > 50:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                    [200, 1000, 5000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 50:
        h_cd, c5_cd = fast_cd_vectorized(best_h, n0, target0,
                                          time_limit=min(40, tleft() * 0.5),
                                          max_sweeps=20000,
                                          n_cand_start=15, n_cand_end=50,
                                          max_no_improve=300, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: n=600
    n1 = 600
    if tleft() > 250:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h1_pol, sc1p = slsqp_optimize(h1, n1, target1,
                                           [300, 1500, 8000, 30000], 1200, 1e-15)
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 140:
            h1_cd, c5_1cd = fast_cd_vectorized(h1, n1, target1,
                                                time_limit=min(100, tleft() * 0.5),
                                                max_sweeps=25000,
                                                n_cand_start=20, n_cand_end=70,
                                                max_no_improve=400, top_k=6)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 60:
            h1_gd, c5_1gd = gradient_polish(best_h, best_n, best_target,
                                             time_limit=min(40, tleft() * 0.5),
                                             max_steps=40000, base_lr=0.02,
                                             n_active=8)
            if c5_1gd < best_score:
                best_score = c5_1gd
                best_h = h1_gd

    # Phase 3: n=800
    n2 = 800
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 280:
            h2_pol, sc2p = slsqp_optimize(h2, n2, target2,
                                           [500, 3000, 15000, 50000], 1500, 1e-15)
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 200:
            h2_cd, c5_2cd = fast_cd_vectorized(h2, n2, target2,
                                                time_limit=min(140, tleft() * 0.5),
                                                max_sweeps=30000,
                                                n_cand_start=25, n_cand_end=80,
                                                max_no_improve=500, top_k=8)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_gd, c5_2gd = gradient_polish(best_h, best_n, best_target,
                                             time_limit=min(50, tleft() * 0.4),
                                             max_steps=50000, base_lr=0.012,
                                             n_active=10)
            if c5_2gd < best_score:
                best_score = c5_2gd
                best_h = h2_gd

        if tleft() > 60:
            h2_bc, c5_2bc = block_cd(best_h, best_n, best_target,
                                      time_limit=min(40, tleft() * 0.4),
                                      max_sweeps=8000, block_size=3, n_cand=40)
            if c5_2bc < best_score:
                best_score = c5_2bc
                best_h = h2_bc

    # Phase 4: n=1000 (if time allows)
    n3 = 1000
    if tleft() > 350:
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)

        if tleft() > 300:
            h3_pol, sc3p = slsqp_optimize(h3, n3, target3,
                                           [1000, 5000, 25000], 1500, 1e-15)
            if sc3p < best_score:
                best_score = sc3p
                best_h = h3_pol
                best_n = n3
                best_target = target3

        if tleft() > 200:
            h3_cd, c5_3cd = fast_cd_vectorized(best_h, best_n, best_target,
                                                time_limit=min(140, tleft() * 0.5),
                                                max_sweeps=25000,
                                                n_cand_start=30, n_cand_end=90,
                                                max_no_improve=500, top_k=10)
            if c5_3cd < best_score:
                best_score = c5_3cd
                best_h = h3_cd
                best_n = len(h3_cd)
                best_target = best_n / 2.0

    # Phase 5: Randomized restarts at best resolution
    if tleft() > 150:
        for noise_scale in [0.005, 0.01, 0.02, 0.04]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = slsqp_optimize(h_pert, best_n, best_target,
                                        [1000, 10000, 50000], 1000, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if tleft() > 30:
                h_cd_r, c5_r = fast_cd_vectorized(h_ref, best_n, best_target,
                                                   time_limit=min(25, tleft() * 0.3),
                                                   max_sweeps=8000,
                                                   n_cand_start=30, n_cand_end=80,
                                                   max_no_improve=150, top_k=8)
                if c5_r < best_score:
                    best_score = c5_r
                    best_h = h_cd_r

    # Phase 6: Final intensive polish
    if tleft() > 80:
        h_pol2, sc2 = slsqp_optimize(best_h, best_n, best_target,
                                      [20000, 100000, 300000], 2000, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 50:
        h_cd_final, c5_final = fast_cd_vectorized(best_h, best_n, best_target,
                                                   time_limit=min(35, tleft() * 0.6),
                                                   max_sweeps=20000,
                                                   n_cand_start=40, n_cand_end=100,
                                                   max_no_improve=300, top_k=10)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 40:
        h_gd_final, c5_gf = gradient_polish(best_h, best_n, best_target,
                                             time_limit=min(25, tleft() * 0.5),
                                             max_steps=50000, base_lr=0.005,
                                             n_active=15)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gd_final

    if tleft() > 30:
        h_pol3, sc3 = slsqp_optimize(best_h, best_n, best_target,
                                      [500000], 3000, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 20:
        h_cd_last, c5_last = fast_cd_vectorized(best_h, best_n, best_target,
                                                 time_limit=min(12, tleft() * 0.5),
                                                 max_sweeps=5000,
                                                 n_cand_start=50, n_cand_end=100,
                                                 max_no_improve=100, top_k=12)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
