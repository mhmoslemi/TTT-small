        if w < 1e-10:
            continue
        k = shifts[idx]
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:n-kp])
            grad[:n-kp] -= w * h[kp:]
    grad -= np.mean(grad)
    return grad

def projected_gradient(h_init, target, time_limit,
                       alpha_start=100.0, alpha_end=10000.0,
                       lr_start=0.3, momentum_start=0.8, momentum_end=0.95,
                       max_iters=300000):
    n = len(h_init)
    start = time.time()
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = lr_start
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        progress = min(it / max_iters, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        momentum = momentum_start + (momentum_end - momentum_start) * progress
        grad = lse_gradient(h, alpha)
        grad_norm = np.max(np.abs(grad))
        if grad_norm < 1e-14:
            break
        v = momentum * v - (lr / grad_norm) * grad
        v = np.clip(v, -0.05, 0.05)
        h_new = h + v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 3000:
            lr *= 0.5
            v *= 0.3
            no_improve = 0
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
            n_cand_start=30, n_cand_end=80, max_no_improve=400,
            bisection_iters=12):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=10000,
                   n_cand=50, n_shifts=4):
    dx = 2.0 / n
    start = time.time()
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 300:
            break
        top_shifts = np.argsort(corr)[-n_shifts:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-5.0 * (c_max - corr[top_shifts]) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)
        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternate_pg_cd(h_init, n, target, time_limit, n_alternations=6,
                    pg_frac=0.4, alpha_base=500.0):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_per = time_limit / max(n_alternations, 1)
    for alt in range(n_alternations):
        if time.time() - start > time_limit:
            break
        pg_time = t_per * pg_frac
        cd_time = t_per * (1.0 - pg_frac)
        remaining = time_limit - (time.time() - start)
        pg_time = min(pg_time, remaining * 0.5)
        cd_time = min(cd_time, remaining * 0.5)
        if pg_time > 5:
            alpha_s = alpha_base * (1.0 + 2.0 * alt)
            alpha_e = alpha_s * 20.0
            h_pg, sc_pg = projected_gradient(h, target, pg_time,
                                             alpha_start=alpha_s, alpha_end=alpha_e,
                                             lr_start=0.15, momentum_start=0.85,
                                             momentum_end=0.95)
            if sc_pg < best_c5:
                best_c5 = sc_pg
                best_h = h_pg.copy()
            h = h_pg
        if cd_time > 5:
            h_cd, sc_cd = pair_cd(h, n, target, cd_time,
                                  max_sweeps=5000,
                                  n_cand_start=40, n_cand_end=70,
                                  max_no_improve=200, bisection_iters=10)
            if sc_cd < best_c5:
                best_c5 = sc_cd
                best_h = h_cd.copy()
            h = h_cd
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 35.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: SLSQP on initial grid
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, best_n, best_target,
                                    [100, 500, 2000],
                                    400, 1e-14,
                                    time_limit=tleft() * 0.4, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: PG on initial grid
    if tleft() > 60:
        h_pg, sc_pg = projected_gradient(best_h, best_target,
                                         time_limit=min(40, tleft() * 0.5),
                                         alpha_start=200.0, alpha_end=5000.0,
                                         lr_start=0.2, momentum_start=0.8,
                                         momentum_end=0.93)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg

    # Phase 3: CD on initial grid
    if tleft() > 80:
        h_cd, sc_cd = pair_cd(best_h, best_n, best_target,
                               time_limit=min(60, tleft() * 0.5),
                               max_sweeps=15000,
                               n_cand_start=30, n_cand_end=70,
                               max_no_improve=300, bisection_iters=12)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Move to n=600

                                            n_alternations=4, alpha_base=300.0)
            if sc_alt < best_score:
                best_score = sc_alt
                best_h = h_alt
                best_n = n1
                best_target = target1
            else:
                best_n = n1
                best_target = target1
                best_h = h1
        if tleft() > 80:
            h_cd1, sc_cd1 = pair_cd(best_h, best_n, best_target,
                                     time_limit=min(60, tleft() * 0.4),
                                     max_sweeps=10000,
                                     n_cand_start=35, n_cand_end=70,
                                     max_no_improve=250, bisection_iters=11)
            if sc_cd1 < best_score:
                best_score = sc_cd1
                best_h = h_cd1

    # Phase 5: Move to n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_n = n2
        best_target = target2
        if tleft() > 200:

                                                alpha_start=500.0, alpha_end=20000.0,
                                                lr_start=0.1, momentum_start=0.85,
                                                momentum_end=0.95)

                best_score = sc_pg2
                best_h = h_pg2

                                     max_sweeps=8000,
                                     n_cand_start=35, n_cand_end=75,
                                     max_no_improve=300, bisection_iters=11)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2
        if tleft() > 60:

                                          max_sweeps=5000, n_cand=40, n_shifts=4)
            if sc_ms < best_score:
                best_score = sc_ms
                best_h = h_ms

    # Phase 6: Alternate PG+CD at current resolution
    if tleft() > 150:
        h_alt2, sc_alt2 = alternate_pg_cd(best_h, best_n, best_target,
                                          time_limit=min(100, tleft() * 0.5),
                                          n_alternations=5, alpha_base=1000.0)

    if tleft() > 250:

        target3 = n3 / 2.0

                                            momentum_end=0.97)
        if sc_pg4 < best_score:

    # Phase 10: Final CD
    if tleft() > 30:
        h_cd4, sc_cd4 = pair_cd(best_h, best_n, best_target,
                                 time_limit=min(20, tleft() * 0.7),

                                 max_no_improve=100, bisection_iters=10)
        if sc_cd4 < best_score:

            best_h = h_cd4

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
