def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12

            break

        h = np.clip(h, 0.0, 1.0)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

    cur_score = compute_c5(h_cur)

        mask[top_w] = True

        grad = np.zeros(n)
        for k_idx in np.where(mask)[0]:
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad = grad / alpha - np.mean(grad) / alpha

        lr = lr_init / (1.0 + 0.005 * it)
        h_prev = h.copy()
        h = y - lr * grad
        h = project_feasible(h, target)

        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_val = t_new

    return best_h, best_c5


def active_set_gradient(h_init, n, target, time_limit, max_iter=30000, n_active=5):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = 0.004

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.9

        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx] - corr[top_idx[0]]
        w = np.exp(np.clip(w, -50, 0))
        sw = np.sum(w)
        if sw > 0:
            w = w / sw
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            k = int(ki) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new
        else:
            h_new = project_feasible(h - 0.5 * lr * grad, target)
            if compute_c5(h_new) < c5_val - 1e-15:
                h = h_new
                lr *= 0.7

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 45.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    if tleft() > 50:
        h_pol = slsqp_round(best_h, best_n, best_n / 2.0, [100, 500, 2000], 600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 180:
        h_cd, c5_cd = vectorized_pair_cd(best_h, best_n, best_n / 2.0,
                                         time_limit=min(120, tleft() * 0.5),
                                         max_sweeps=30000,
                                         n_cand_start=15, n_cand_end=50,
                                         max_no_improve=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, best_n, best_n / 2.0, [5000, 20000, 80000], 1000, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    resolutions = [600, 800, 1000]
    for n_res in resolutions:
        if tleft() < 220:
            break
        target_res = n_res / 2.0
        h_res = interpolate_to(best_h, n_res, target_res)
        cur_h = h_res
        cur_n = n_res
        cur_t = target_res
        cur_score = compute_c5(cur_h)

        if tleft() > 180:
            h_cd2, c5_cd2 = vectorized_pair_cd(cur_h, cur_n, cur_t,
                                               time_limit=min(100, tleft() * 0.4),
                                               max_sweeps=18000,
                                               n_cand_start=12, n_cand_end=40,
                                               max_no_improve=250)
            if c5_cd2 < cur_score:
                cur_score = c5_cd2
                cur_h = h_cd2

        if tleft() > 80:
            h_pol2 = slsqp_round(cur_h, cur_n, cur_t, [5000, 20000, 60000], 800, 1e-14)
            sc2 = compute_c5(h_pol2)
            if sc2 < cur_score:
                cur_score = sc2
                cur_h = h_pol2

        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h
            best_n = cur_n

    if tleft() > 100:
        h_nes, c5_nes = nesterov_smooth(best_h, best_n, best_n / 2.0,
                                        time_limit=min(70, tleft() * 0.5),
                                        alpha=5000.0, lr_init=0.004, max_iter=40000)
        if c5_nes < best_score:
            best_score = c5_nes
            best_h = h_nes

    if tleft() > 120:
        h_cd3, c5_cd3 = vectorized_pair_cd(best_h, best_n, best_n / 2.0,
                                           time_limit=min(90, tleft() * 0.55),
                                           max_sweeps=15000,
                                           n_cand_start=20, n_cand_end=55,
                                           max_no_improve=250)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 70:
        h_asg, c5_asg = active_set_gradient(best_h, best_n, best_n / 2.0,
                                            time_limit=min(50, tleft() * 0.6),
                                            max_iter=25000, n_active=5)
        if c5_asg < best_score:
            best_score = c5_asg
            best_h = h_asg

    if tleft() > 70:
        h_pol_final = slsqp_round(best_h, best_n, best_n / 2.0,
                                  [10000, 40000, 150000], 1200, 1e-14)
        sc_final = compute_c5(h_pol_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_pol_final

    if tleft() > 40:
        h_cd_final, c5_final = vectorized_pair_cd(best_h, best_n, best_n / 2.0,
                                                  time_limit=min(30, tleft() * 0.6),
                                                  max_sweeps=8000,
                                                  n_cand_start=25, n_cand_end=60,
                                                  max_no_improve=150)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_last = slsqp_round(best_h, best_n, best_n / 2.0,
                             [50000, 200000], 800, 1e-14)
        sc_last = compute_c5(h_last)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
