```python
def dual_ascend(h, n, mu, beta, lr, momentum, max_iter, time_limit, start_time):
    """
    Alternating: fix mu -> minimize smooth objective, fix h -> update mu.
    """
    h = h.copy()
    v = np.zeros(n)  # momentum
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * (2.0/n)
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        # Update mu: softmax over shifts
        # Use top-k shifts with softmax
        sorted_idx = np.argsort(corr)[::-1]
        top_k = sorted_idx[:min(50, len(corr))]
        w = np.exp(beta * (corr[top_k] - corr[top_k[0]]))
        mu_new = np.zeros(len(corr))
        mu_new[top_k] = w / w.sum()
        
        # Compute smooth gradient
        g = np.zeros(n)
        for k_idx in top_k:
            k = int(k_idx) - (n-1)  # actual shift
            w_k = mu_new[k_idx]
            if w_k < 1e-10:
                continue
            # ∂c_k/∂h[j] = (1-h[j+k])*1[j+k<n] - h[j-k]*1[j-k>=0]
            for j in range(n):
                if j+k < n and j+k >= 0:  # j+k valid
                    g[j] += w_k * (1.0 - h[j+k])
                if j-k >= 0 and j-k < n:  # j-k valid
                    g[j] -= w_k * h[j-k]
        
        # Remove mean to maintain sum constraint
        g -= np.mean(g)
        
        # Momentum update
        v = 0.9 * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n/2.0)
        
        # Check if improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = np.max(corr_new) * (2.0/n)
        
        if c5_new < c5:
            h = h_new
        else:
            # Reject, reduce step
            v *= 0.5
        
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def multi_shift_cd(h_init, n, target, time_limit, start_time,
```

```python
                   max_sweeps=50000, max_no_improve=800,

    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_raw = float(np.max(corr))

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        c5_val = c5_raw * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            h = h + rng.randn(n) * reheat_sigma
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_raw = float(np.max(corr))
            no_improve = 0
            continue

        top_idx = np.argsort(corr)[-n_shifts:]
        c_max_val = float(corr[top_idx[0]])
        w = np.exp(-8.0 * (c_max_val - corr[top_idx]) / max(c5_raw, 1e-10))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)

        high_set = sorted_idx[-ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

                    continue

                delta_hi = min(h[i], 1.0 - h[j])

                g = change[j] - change[i]
                a_d = 0.0
                b_d = delta_hi

                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-12
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-10:
                    continue

                trial = np.max(corr + delta_star * g)
                if trial < best_trial - 1e-12:
                    best_trial = trial
                    best_i = i
                    best_j = j
                    best_delta = delta_star
            if count > max_pairs:
                break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_raw = float(np.max(corr))

    return best_h, best_c5

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        h_pol2, sc = slsqp_polish(best_h, n0, target0,
                                  [50000, 200000], 400,
                                  min(25, tleft() * 0.7), start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
