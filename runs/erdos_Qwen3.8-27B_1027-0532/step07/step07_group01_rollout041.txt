    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_adaptive(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_min=15, n_cand_max=80, n_shifts=5,
                     max_no_improve=500, restart_every=200, restart_scale=0.015):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)
    sweep = 0

    while sweep < max_sweeps:
        if time.time() - start > time_limit:
            break
        sweep += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 40 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        progress = float(sweep) / max_sweeps
        if consecutive_no_improve > 200:
            current_ncand = n_cand_max
        elif consecutive_no_improve > 100:
            current_ncand = (n_cand_min + n_cand_max) // 2
        else:
            current_ncand = int(n_cand_min + (n_cand_max - n_cand_min) * min(progress * 2.0, 1.0))
        current_ncand = min(current_ncand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = (corr[s_idx] / top_sum) ** 2
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h_init, n, target, time_limit, max_sweeps=20000, block_size=3):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(999)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_cand = min(30, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_move = None

        for ii in range(min(n_cand, 15)):
            i = int(high_set[ii])
            for jj in range(min(n_cand, 15)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j] * block_size)
                delta_hi = min(h[i] / block_size, 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = (change[j] * block_size - change[i])
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d * block_size
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [30, 100, 400, 1500, 5000, 20000, 80000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd, c5_cd = pair_cd_adaptive(best_h, current_n, current_target,
                                       time_limit=min(220, tleft() * 0.6),
                                       max_sweeps=60000,
                                       n_cand_min=15, n_cand_max=75,
                                       n_shifts=5, max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [10000, 50000, 200000],
                                800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 600
    if tleft() > 350:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd_adaptive(h1, n1, target1,
                                              time_limit=min(200, tleft() * 0.55),
                                              max_sweeps=40000,
                                              n_cand_min=20, n_cand_max=80,
                                              n_shifts=5, max_no_improve=400)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1
        if current_n == n1 and tleft() > 80:
            h_pol1, sc1b = slsqp_round(best_h, n1, target1,
                                       [10000, 50000, 200000],
                                       800, 1e-15)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1

    n2 = 800
    if tleft() > 400:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 300:
            h_cd2, c5_cd2 = pair_cd_adaptive(h2, n2, target2,
                                              time_limit=min(250, tleft() * 0.6),
                                              max_sweeps=50000,
                                              n_cand_min=20, n_cand_max=70,
                                              n_shifts=4, max_no_improve=400)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                current_n = n2
                current_target = target2
        if current_n == n2 and tleft() > 80:
            h_pol2, sc2b = slsqp_round(best_h, n2, target2,
                                       [10000, 50000, 200000],
                                       800, 1e-15)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_pol2

    if tleft() > 150:
        h_blk, c5_blk = block_cd(best_h, current_n, current_target,
                                 time_limit=min(120, tleft() * 0.6),
                                 max_sweeps=15000, block_size=2)
        if c5_blk < best_score:
            best_score = c5_blk
            best_h = h_blk

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10, 0.20]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [100, 500, 2000, 8000, 30000],
                                        400, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 80:
                h_cd_p, c5_p = pair_cd_adaptive(h_ref, current_n, current_target,
                                                 time_limit=min(50, tleft() * 0.3),
                                                 max_sweeps=10000,
                                                 n_cand_min=20, n_cand_max=60,
                                                 n_shifts=5, max_no_improve=200)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 60:
        h_final_cd, c5_final = pair_cd_adaptive(best_h, current_n, current_target,
                                                time_limit=min(50, tleft() * 0.7),
                                                max_sweeps=20000,
                                                n_cand_min=25, n_cand_max=85,
                                                n_shifts=6, max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 30:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [30000, 100000, 300000],
                                            600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
