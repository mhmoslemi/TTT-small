    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > start_time + time_limit:
                break
        try:
            def cons_fun(x):
                return np.sum(x) - target
            def cons_jac(x):
                return np.ones(n)
            res = minimize(_smooth_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_descent(h_init, n, target, time_limit, start_time,
                        topk=5, alpha_ls=100.0, step0=0.05,
                        momentum=0.85, max_iters=10000, patience=300):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros(n)
    step_size = step0
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for it in range(max_iters):
        if time.time() - start_time > start_time + time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        top_idx = np.argsort(corr)[-topk:]
        c_max = np.max(corr)
        weights = np.exp(alpha_ls * (corr[top_idx] - c_max))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for idx in range(topk):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)
        velocity = momentum * velocity + (1.0 - momentum) * grad

        h_new = h - step_size * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            step_size = min(step_size * 1.02, 0.3)
        else:
            h = h_new
            no_improve += 1
            step_size = max(step_size * 0.97, 0.001)
            if no_improve > patience:
                break

    return best_h, best_score

def enhanced_cd(h_init, n, target, time_limit, start_time,
                max_sweeps=20000, n_active=5, n_cand_start=15, n_cand_end=50,
                max_no_improve=300, bisection_iters=10):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(123)

    for sweep in range(max_sweeps):
        if time.time() - start_time > start_time + time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        max_pairs = 300
        pair_list = []
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            step_h = max(1, n_h // 8)
            step_l = max(1, n_l // 8)
            for ii in range(0, n_h, step_h):
                for jj in range(0, n_l, step_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
            while len(pair_list) < max_pairs:
                ii = int(rng.randint(n_h))
                jj = int(rng.randint(n_l))
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

def structured_perturb_best(h, n, target, scales, freqs, start_time, time_limit):
    best_h = h.copy()
    best_score = compute_c5(h)
    for scale in scales:
        for freq in freqs:
            if time.time() - start_time > start_time + time_limit:
                break
            perturbation = scale * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
            h_pert = h + perturbation
            h_pert = project_feasible(h_pert, target)
            h_re, sc_re = enhanced_cd(h_pert, n, target,
                                       time_limit=min(20, time_limit * 0.15),
                                       start_time=start_time,
                                       max_sweeps=3000,
                                       n_active=3,
                                       n_cand_start=15, n_cand_end=35,
                                       max_no_improve=100, bisection_iters=8)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [50, 200, 1000, 3000], 400, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 2: Subgradient descent to escape local optima
    if tleft() > 120:
        tl_sg = min(90, tleft() * 0.25)
        h_sg, sc_sg = subgradient_descent(best_h, n0, target0,
                                           time_limit=tl_sg,
                                           start_time=start_time,
                                           topk=5, alpha_ls=80.0,
                                           step0=0.03, momentum=0.8,
                                           max_iters=8000, patience=200)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 3: CD at initial resolution
    if tleft() > 120:
        tl_cd = min(100, tleft() * 0.3)
        h_cd, sc_cd = enhanced_cd(best_h, n0, target0,
                                   time_limit=tl_cd, start_time=start_time,
                                   max_sweeps=10000, n_active=5,
                                   n_cand_start=10, n_cand_end=30,
                                   max_no_improve=250, bisection_iters=10)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [3000, 10000, 30000], 500, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 5: Interpolate to n=800 and optimize
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        h_pol1, sc_pol1 = slsqp_optimize(h1, n1, target1,
                                          [100, 500, 2000], 350, 1e-14)
        if sc_pol1 < best_score:
            best_score = sc_pol1
            best_h = h_pol1

        if tleft() > 150:
            tl_sg1 = min(80, tleft() * 0.2)
            h_sg1, sc_sg1 = subgradient_descent(best_h, n1, target1,
                                                 time_limit=tl_sg1,
                                                 start_time=start_time,
                                                 topk=7, alpha_ls=100.0,
                                                 step0=0.02, momentum=0.8,
                                                 max_iters=6000, patience=200)
            if sc_sg1 < best_score:
                best_score = sc_sg1
                best_h = h_sg1

    # Phase 6: CD at n=800 (main optimization phase)
    if tleft() > 200:
        tl_cd1 = min(250, tleft() * 0.45)
        h_cd1, sc_cd1 = enhanced_cd(best_h, n1, target1,
                                     time_limit=tl_cd1, start_time=start_time,
                                     max_sweeps=25000, n_active=5,
                                     n_cand_start=20, n_cand_end=45,
                                     max_no_improve=400, bisection_iters=10)
        if sc_cd1 < best_score:
            best_score = sc_cd1
            best_h = h_cd1

    # Phase 7: SLSQP at n=800
    if tleft() > 100:
        h_pol1b, sc_pol1b = slsqp_optimize(best_h, n1, target1,
                                            [5000, 20000, 50000], 600, 1e-15)
        if sc_pol1b < best_score:
            best_score = sc_pol1b
            best_h = h_pol1b

    # Phase 8: Try n=1200
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        h_pol2, sc_pol2 = slsqp_optimize(h2, n2, target2,
                                          [200, 1000, 5000], 300, 1e-14)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_h = h_pol2

        if tleft() > 150:
            tl_cd2 = min(150, tleft() * 0.4)
            h_cd2, sc_cd2 = enhanced_cd(best_h, n2, target2,
                                         time_limit=tl_cd2, start_time=start_time,
                                         max_sweeps=15000, n_active=5,
                                         n_cand_start=25, n_cand_end=55,
                                         max_no_improve=300, bisection_iters=10)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

        if tleft() > 80:
            h_pol2b, sc_pol2b = slsqp_optimize(best_h, n2, target2,
                                                [10000, 30000, 80000], 500, 1e-15)
            if sc_pol2b < best_score:
                best_score = sc_pol2b
                best_h = h_pol2b

    # Phase 9: Structured perturbations at best resolution
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sp, sc_sp = structured_perturb_best(best_h, n_cur, target_cur,
                                               scales=[0.02, 0.05, 0.08],
                                               freqs=[1, 2, 3, 5],
                                               start_time=start_time,
                                               time_limit=min(100, tleft() * 0.4))
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    # Phase 10: Final CD
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl_final_cd = min(50, tleft() * 0.4)
        h_fd, sc_fd = enhanced_cd(best_h, n_cur, target_cur,
                                   time_limit=tl_final_cd, start_time=start_time,
                                   max_sweeps=8000, n_active=5,
                                   n_cand_start=30, n_cand_end=60,
                                   max_no_improve=200, bisection_iters=8)
        if sc_fd < best_score:
            best_score = sc_fd
            best_h = h_fd

    # Phase 11: Final SLSQP
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc_final = slsqp_optimize(best_h, n_cur, target_cur,
                                            [20000, 80000, 200000], 400, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
