import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def subgradient_opt(h_init, n, target, time_limit, n_active=30,
                    step_init=0.005, alpha_scale=8.0,
                    max_no_improve=300, bb_blend=0.4):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(7)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    s_prev = None
    g_prev = None
    no_improve = 0

    for it in range(1000000):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            scale = min(0.005 * (1.0 + no_improve / max_no_improve), 0.04)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            step = step_init * 0.5
            s_prev = None
            g_prev = None
            no_improve = 0
            continue

        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[-n_active:]
        c_sub = corr[top_idx]
        alpha_ws = alpha_scale * n / max(c_max, 1e-10)
        weights = np.exp(alpha_ws * (c_sub - c_sub.max()))
        weights /= weights.sum()

        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            w = weights[idx]
            if w < 1e-10:
                continue
            k = int(top_idx[idx]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        if s_prev is not None and g_prev is not None:
            ys = np.dot(g_prev, s_prev)
            ss = np.dot(s_prev, s_prev)
            if ys > 1e-12 and ss > 1e-12:
                step_bb = ss / ys
                step = (1.0 - bb_blend) * step + bb_blend * np.clip(step_bb, 1e-6, 0.08)

        h_old = h.copy()
        h_try = h - step * grad
        h_try = project_feasible(h_try, target)

        s_prev = h_try - h_old
        g_prev = grad
        h = h_try

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand=60, max_no_improve=400, weight_beta=6.0,
            n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k = min(40, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        found = False

        for ii in range(n_h):
            if found:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found = True
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

    def cons_jac(x):
        return np.ones(len(x))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    alphas = [200, 1000, 5000, 20000, 100000]
    per_alpha_time = time_limit / len(alphas)

    for alpha in alphas:
        if time.time() - start > time_limit - 5:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15,
                                    'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score

def multi_start_refine(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)

    perturbations = []
    for s in range(1, 6):
        perturbations.append(('shift', s))
    perturbations.append(('reflect', None))
    for ns in [3, 7, 15, 30]:
        perturbations.append(('freq', ns))
    for amp in [0.01, 0.03, 0.06]:
        perturbations.append(('noise', amp))

    per_pert_time = max(time_limit / len(perturbations), 5.0)

    for ptype, pval in perturbations:
        if time.time() - start > time_limit - 10:
            break

        h_pert = best_h.copy()
        if ptype == 'shift':
            s = pval % n
            h_pert = np.roll(h_pert, s)
        elif ptype == 'reflect':
            h_pert = h_pert[::-1]
        elif ptype == 'freq':
            h_diff = h_pert - 0.5
            freq = np.fft.rfft(h_diff)
            hfreq = len(freq)
            for k in range(1, min(pval, hfreq - 1)):
                magnitude = np.abs(freq[k])
                phase = np.angle(freq[k])
                new_mag = magnitude * (1.0 + 0.3 * rng.randn())
                new_phase = phase + 0.2 * rng.randn()
                freq[k] = new_mag * np.exp(1j * new_phase)
                if hfreq - k > 0 and hfreq - k < hfreq:
                    freq[hfreq - k] = np.conj(freq[k])
            h_pert = np.fft.irfft(freq, n) + 0.5
        elif ptype == 'noise':
            h_pert = h_pert + rng.randn(n) * pval

        h_pert = project_feasible(h_pert, target)
        t_left = time_limit - (time.time() - start)
        if t_left < 10:
            break

        t_alloc = min(per_pert_time, t_left * 0.6)
        h_sub, c5_sub = subgradient_opt(h_pert, n, target, time_limit=t_alloc,
                                        n_active=25, step_init=0.003,
                                        max_no_improve=150)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

        t_left2 = time_limit - (time.time() - start)
        if t_left2 > 20 and c5_sub < best_c5 + 0.001:
            t_alloc2 = min(per_pert_time * 0.5, t_left2 * 0.5)
            h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=t_alloc2,
                                   max_sweeps=5000, n_cand=50,
                                   max_no_improve=150)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick subgradient at initial resolution
    if tleft() > 50:
        h_sub, c5_sub = subgradient_opt(best_h, n0, target0,
                                        time_limit=min(40, tleft() * 0.3),
                                        n_active=30, step_init=0.005)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 2: Pair CD at initial resolution
    if tleft() > 100:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(80, tleft() * 0.4),
                              max_sweeps=20000, n_cand=60,
                              max_no_improve=300, weight_beta=6.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Go to n=1000
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 350:
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 300:
            h_sub1, c5_sub1 = subgradient_opt(h1, n1, target1,
                                              time_limit=min(80, tleft() * 0.3),
                                              n_active=30, step_init=0.004,
                                              max_no_improve=250)
            best_h = h_sub1
            if c5_sub1 < best_score:
                best_score = c5_sub1

        if tleft() > 200:
            h_cd1, c5_cd1 = pair_cd(best_h, n1, target1,
                                    time_limit=min(150, tleft() * 0.5),
                                    max_sweeps=25000, n_cand=70,
                                    max_no_improve=350, weight_beta=6.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    # Phase 4: Go to n=1200
    n2 = 1200
    target2 = n2 / 2.0
    if tleft() > 350:
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 300:
            h_sub2, c5_sub2 = subgradient_opt(h2, n2, target2,
                                              time_limit=min(80, tleft() * 0.3),
                                              n_active=35, step_init=0.003,
                                              max_no_improve=200)
            best_h = h_sub2
            if c5_sub2 < best_score:
                best_score = c5_sub2

        if tleft() > 200:
            h_cd2, c5_cd2 = pair_cd(best_h, n2, target2,
                                    time_limit=min(150, tleft() * 0.5),
                                    max_sweeps=20000, n_cand=70,
                                    max_no_improve=300, weight_beta=7.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 5: Annealed pair CD at current best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 120:
        h_an, c5_an = annealed_pair_cd(best_h, n_cur, target_cur,
                                       time_limit=min(100, tleft() * 0.5),
                                       max_sweeps=15000, n_cand=55,
                                       max_no_improve=400, weight_beta=4.0,
                                       T0=0.001)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 6: Multi-start at current resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 150:
        h_ms, c5_ms = multi_start_refine(best_h, n_cur, target_cur,
                                         time_limit=min(130, tleft() * 0.6),
                                         seed=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 7: SLSQP refine
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 60:
        h_sl, c5_sl = slsqq_refine(best_h, n_cur, target_cur,
                                   time_limit=min(50, tleft() * 0.6))
        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl

    # Phase 8: Final pair CD polish
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 40:
        h_fin, c5_fin = pair_cd(best_h, n_cur, target_cur,
                                time_limit=min(30, tleft() * 0.7),
                                max_sweeps=10000, n_cand=80,
                                max_no_improve=200, weight_beta=8.0,
                                n_bisect=16)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Phase 9: Final subgradient with tiny steps
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 20:
        h_tiny, c5_tiny = subgradient_opt(best_h, n_cur, target_cur,
                                          time_limit=min(15, tleft() * 0.7),
                                          n_active=50, step_init=0.001,
                                          alpha_scale=15.0,
                                          max_no_improve=100)
        if c5_tiny < best_score:
            best_score = c5_tiny
            best_h = h_tiny

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
