def p_norm_obj(x, p, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    return np.sum(vals ** p) ** (1.0 / p)

def p_norm_grad(x, p, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    base = np.sum(vals ** p)
    if base < 1e-30:
        return np.zeros(len(x))
    coeff = (base ** (-1.0 / p)) * vals ** (p - 1)
    n = len(x)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s_idx in range(n_corr):
        c = coeff[s_idx]
        if abs(c) < 1e-20:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += c * (1.0 - x[idxs[v1] - k])
        grad[v2] -= c * x[idxs[v2] + k]
    return grad

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fast_vectorized_pair_cd(h_init, n, target, time_limit,
                            max_sweeps=30000, n_cand_start=20, n_cand_end=60,
                            batch_size=400, max_no_improve=250,
                            weight_beta=5.0, use_anneal=False, T0=0.0005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(777)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    shifts = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        c_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            if use_anneal and consecutive_no_improve < max_no_improve + 50:
                h = h + rng.randn(n) * 0.005
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
                continue
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        nc = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        nc = min(nc, n // 3)

        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(n_corr) / n_corr
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for s in range(n_corr):
            w = weights[s]
            if w < 1e-12:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]

        n_h = len(high_set)
        n_l = len(low_set)
        total_pairs = n_h * n_l

        best_trial = float('inf')
        best_ii = -1
        best_jj = -1
        best_delta = 0.0

        for batch_start in range(0, total_pairs, batch_size):
            batch_end = min(batch_start + batch_size, total_pairs)
            n_batch = batch_end - batch_start

            ii_idx = np.arange(batch_start, batch_end) // n_l
            jj_idx = np.arange(batch_start, batch_end) % n_l

            i_vals = high_set[ii_idx]
            j_vals = low_set[jj_idx]

            valid = i_vals != j_vals
            if not valid.any():
                continue
            ii_idx = ii_idx[valid]
            jj_idx = jj_idx[valid]
            i_vals = i_vals[valid]
            j_vals = j_vals[valid]
            n_batch = len(ii_idx)
            if n_batch == 0:
                continue

            g_batch = change[j_vals] - change[i_vals]

            delta_lo = np.maximum(h[i_vals] - 1.0, -h[j_vals])
            delta_hi = np.minimum(h[i_vals], 1.0 - h[j_vals])

            valid_range = (delta_hi - delta_lo) > 1e-12
            if not valid_range.any():
                continue
            g_batch = g_batch[valid_range]
            delta_lo = delta_lo[valid_range]
            delta_hi = delta_hi[valid_range]
            ii_idx = ii_idx[valid_range]
            jj_idx = jj_idx[valid_range]
            i_vals = i_vals[valid_range]
            j_vals = j_vals[valid_range]
            n_batch = len(ii_idx)
            if n_batch == 0:
                continue

            a_d = delta_lo.copy()
            b_d = delta_hi.copy()

            for _ in range(20):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr[None, :] + mid_d[:, None] * g_batch
                f_mid = np.max(vals, axis=1)
                active = vals >= (f_mid[:, None] - 1e-14)
                g_masked = np.where(active, g_batch, -np.inf)
                slope = np.max(g_masked, axis=1)
                go_left = slope > 1e-12
                b_d[go_left] = mid_d[go_left]
                a_d[~go_left] = mid_d[~go_left]

            delta_star = 0.5 * (a_d + b_d)
            vals_final = corr[None, :] + delta_star[:, None] * g_batch
            f_final = np.max(vals_final, axis=1)

            best_in_batch = int(np.argmin(f_final))
            if f_final[best_in_batch] < best_trial:
                best_trial = float(f_final[best_in_batch])
                best_ii = int(ii_idx[best_in_batch])
                best_jj = int(jj_idx[best_in_batch])
                best_delta = float(delta_star[best_in_batch])

        T = T0 * 0.01 ** progress if use_anneal else 0.0

        if best_ii >= 0:
            if best_trial < c_max - 1e-12:
                i = int(high_set[best_ii])
                j = int(low_set[best_jj])
                h[i] -= best_delta
                h[j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
            elif T > 1e-12 and (best_trial - c_max) < T * dx * 3:
                if rng.random() < np.exp(-(best_trial - c_max) / (T * dx)):
                    i = int(high_set[best_ii])
                    j = int(low_set[best_jj])
                    h[i] -= best_delta
                    h[j] += best_delta
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def p_norm_smooth_opt(h_init, n, target, time_limit, p, n_iter, step_init):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        vals = corr * dx
        base = np.sum(vals ** p)
        if base < 1e-30:
            break
        coeff = (base ** (-1.0 / p)) * vals ** (p - 1)

        grad = np.zeros(n)
        for s in range(n_corr):
            c = coeff[s]
            if abs(c) < 1e-20:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += c * (1.0 - h[idxs[v1] - k])
            grad[v2] -= c * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.88

    return best_h, best_c5


def multi_start_refine(h_base, n, target, time_limit, seed):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_base.copy()
    best_score = compute_c5(h_base)

    noise_scales = [0.005, 0.01, 0.02, 0.05]
    for ns in noise_scales:
        if time.time() - start > time_limit:
            break
        h_pert = best_h + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        h_tried, sc_tried = fast_vectorized_pair_cd(
            h_pert, n, target,
            time_limit=min(60, time_limit - (time.time() - start)),
            max_sweeps=5000, n_cand_start=20, n_cand_end=50,
            batch_size=300, max_no_improve=150, weight_beta=5.0)
        if sc_tried < best_score:
            best_score = sc_tried
            best_h = h_tried

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_polish(best_h, n0, target0,
                                 [50, 200, 1000], 300)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Fast vectorized pair CD at initial resolution
    if tleft() > 200:
        h_cd, c5_cd = fast_vectorized_pair_cd(
            best_h, n0, target0,
            time_limit=min(120, tleft() * 0.4),
            max_sweeps=15000, n_cand_start=20, n_cand_end=55,
            batch_size=400, max_no_improve=200, weight_beta=5.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: p-norm gradient for smoothing
    if tleft() > 150:
        h_pp, c5_pp = p_norm_smooth_opt(
            best_h, n0, target0,
            time_limit=min(60, tleft() * 0.3), p=12, n_iter=4000, step_init=0.004)
        if c5_pp < best_score:
            best_score = c5_pp
            best_h = h_pp

    # Phase 4: Interpolate to n=1000
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 500:
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 400:
            h_pol1, sc1 = slsqp_polish(h1, n1, target1,
                                       [50, 200, 500], 300)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
            else:
                best_h = h_pol1

        if tleft() > 350:
            h_cd1, c5_cd1 = fast_vectorized_pair_cd(
                best_h, n1, target1,
                time_limit=min(300, tleft() * 0.5),
                max_sweeps=20000, n_cand_start=20, n_cand_end=60,
                batch_size=400, max_no_improve=250, weight_beta=5.5)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 80:
            h_pol1b, sc1b = slsqp_polish(best_h, n1, target1,
                                         [1000, 5000], 400)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 5: Multi-start at n=1000
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_ms, c5_ms = multi_start_refine(best_h, n_cur, target_cur,
                                         time_limit=min(150, tleft() * 0.5),
                                         seed=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 6: Interpolate to n=1200
    n2 = 1200
    target2 = n2 / 2.0
    if tleft() > 350 and len(best_h) < n2:
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 300:
            h_pol2, sc2 = slsqp_polish(h2, n2, target2,
                                       [50, 200], 250)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2
            else:
                best_h = h_pol2

        if tleft() > 200:
            h_cd2, c5_cd2 = fast_vectorized_pair_cd(
                best_h, n2, target2,
                time_limit=min(180, tleft() * 0.5),
                max_sweeps=15000, n_cand_start=25, n_cand_end=65,
                batch_size=350, max_no_improve=250, weight_beta=6.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        if tleft() > 60:
            h_pol2b, sc2b = slsqp_polish(best_h, n2, target2,
                                         [2000, 5000], 350)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_pol2b

    # Phase 7: Final pair CD polish with higher beta
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin_cd = fast_vectorized_pair_cd(
            best_h, n_cur, target_cur,
            time_limit=min(80, tleft() * 0.6),
            max_sweeps=10000, n_cand_start=30, n_cand_end=70,
            batch_size=400, max_no_improve=200, weight_beta=8.0)
        if c5_fin_cd < best_score:
            best_score = c5_fin_cd
            best_h = h_fin_cd

    # Phase 8: Final p-norm polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_pp, c5_fin_pp = p_norm_smooth_opt(
            best_h, n_cur, target_cur,
            time_limit=min(40, tleft() * 0.6), p=16, n_iter=3000, step_init=0.002)
        if c5_fin_pp < best_score:
            best_score = c5_fin_pp
            best_h = h_fin_pp

    # Phase 9: Final SLSQP
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_polish(best_h, n_cur, target_cur,
                                     [10000, 50000], 300)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
