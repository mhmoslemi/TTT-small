                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
            penalty *= 0.4
        except Exception:
            break
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_v3(h_init, n, target, time_limit, max_sweeps=40000,
               n_cand_start=20, n_cand_end=120, max_no_improve=800,
               n_active_shifts=5, reheat_interval=80, reheat_scale=0.05,
               lookahead_k=5, filter_thresh=0.25):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)
    corr = compute_corr(h)
    c5_val = float(np.max(corr)) * dx

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if sweep % 5 == 0:
            corr = compute_corr(h)
            c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval:
            if sweeps_since_improve % 15 == 0:
                rng = np.random.RandomState(sweep * 17 + 11)
                n_pert = max(8, n // 25)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
                h_pert = project_feasible(h_pert, target)
                sc_pert = compute_c5(h_pert)
                if sc_pert < c5_val + 3e-6:
                    h = h_pert
                    change = compute_change_matrix(h, n)
                    corr = compute_corr(h)
                    c5_val = float(np.max(corr)) * dx
                    sweeps_since_improve = 0
                    continue

        top_k_idx = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k_idx])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = (corr[top_k_idx] - corr[top_k_idx].min() + 0.01)
            weights = weights / weights.sum()

        grad = np.zeros(n)
        for s_idx, w in zip(top_k_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        top_lookahead_idx = np.argsort(corr)[-lookahead_k:]
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_top = g[top_lookahead_idx]
                g_top_abs_max = np.max(np.abs(g_top))
                if g_top_abs_max > 1e-14 and np.max(g_top) > filter_thresh * g_top_abs_max:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def proximal_logsumexp(h_init, n, target, time_limit, alpha0=0.02, alpha_decay=0.98,
                       smoothing=200.0, max_iters=50000):
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha0
    stale = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if alpha < 1e-5:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx if (dx := 2.0 / n) else 0
        c5_val = float(np.max(corr)) * (2.0 / n)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= alpha_decay
            if stale > 800:
                break

        c_max = np.max(corr)
        weights = np.exp(smoothing * (corr - c_max))
        total_w = np.sum(weights)
        if total_w < 1e-30:
            break
        weights = weights / total_w

        grad = np.zeros(n)
        for s in range(len(corr)):
            w = weights[s]
            if w < 1e-10:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP + L-BFGS-B polish at initial resolution
    if tleft() > 100:
        deadline = time.time() + min(45, tleft() * 0.35)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [100, 500, 2000, 8000, 30000],
                                   800, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 80:
        deadline = time.time() + min(40, tleft() * 0.3)
        h_lb, sc_lb = lbfgsb_round(best_h, current_n, current_target,
                                   alpha=5000.0, maxiter=500, deadline=deadline)
        if sc_lb < best_score:
            best_score = sc_lb
            best_h = h_lb

    # Phase 2: Pair CD at initial resolution
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_v3(best_h, current_n, current_target,
                                 time_limit=min(80, tleft() * 0.4),
                                 max_sweeps=20000,
                                 n_cand_start=25, n_cand_end=100,
                                 max_no_improve=600,
                                 n_active_shifts=5,
                                 reheat_interval=80, reheat_scale=0.04,
                                 lookahead_k=5, filter_thresh=0.25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Proximal log-sum-exp at initial resolution
    if tleft() > 150:
        h_pl, c5_pl = proximal_logsumexp(best_h, current_n, current_target,
                                         time_limit=min(60, tleft() * 0.35),
                                         alpha0=0.015, alpha_decay=0.985,
                                         smoothing=300.0, max_iters=30000)
        if c5_pl < best_score:
            best_score = c5_pl
            best_h = h_pl

    # Phase 4: SLSQP polish again
    if tleft() > 80:
        deadline = time.time() + min(40, tleft() * 0.35)
        h_pol2, sc2 = slsqp_optimize(best_h, current_n, current_target,
                                     [500, 2000, 10000, 50000],
                                     600, 1e-15, deadline=deadline)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 5: Interpolate to n=800
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd_v3(h1, n1, target1,
                                       time_limit=min(150, tleft() * 0.4),
                                       max_sweeps=25000,
                                       n_cand_start=20, n_cand_end=100,
                                       max_no_improve=700,
                                       n_active_shifts=5,
                                       reheat_interval=100, reheat_scale=0.04,
                                       lookahead_k=5, filter_thresh=0.3)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if tleft() > 80:
            h_pl1, c5_pl1 = proximal_logsumexp(best_h, current_n, current_target,
                                               time_limit=min(50, tleft() * 0.3),
                                               alpha0=0.01, alpha_decay=0.985,
                                               smoothing=400.0, max_iters=20000)
            if c5_pl1 < best_score:
                best_score = c5_pl1
                best_h = h_pl1

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.35)
            h_pol3, sc3 = slsqp_optimize(best_h, current_n, current_target,
                                         [2000, 8000, 30000, 100000],
                                         800, 1e-15, deadline=deadline)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 6: More CD at current resolution
    if tleft() > 200:
        h_cd2, c5_cd2 = pair_cd_v3(best_h, current_n, current_target,
                                   time_limit=min(120, tleft() * 0.45),
                                   max_sweeps=20000,
                                   n_cand_start=30, n_cand_end=130,
                                   max_no_improve=800,
                                   n_active_shifts=7,
                                   reheat_interval=100, reheat_scale=0.03,
                                   lookahead_k=7, filter_thresh=0.2)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: Interpolate to n=1200 if time allows
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 200:
            h_cd3, c5_cd3 = pair_cd_v3(h2, n2, target2,
                                       time_limit=min(120, tleft() * 0.4),
                                       max_sweeps=15000,
                                       n_cand_start=15, n_cand_end=80,
                                       max_no_improve=600,
                                       n_active_shifts=5,
                                       reheat_interval=100, reheat_scale=0.04,
                                       lookahead_k=5, filter_thresh=0.3)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                current_n = n2
                current_target = target2

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.35)
            h_pol4, sc4 = slsqp_optimize(best_h, current_n, current_target,
                                         [3000, 10000, 50000],
                                         600, 1e-15, deadline=deadline)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_pol4

    # Phase 8: Multi-start perturbation with SLSQP
    if tleft() > 150:
        for attempt in range(5):
            if tleft() < 50:
                break
            noise_scales = [0.02, 0.04, 0.07, 0.12, 0.18]
            noise_scale = noise_scales[attempt % 5]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(30, tleft() * 0.25)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [500, 2000, 10000, 50000],
                                           500, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final proximal + CD
    if tleft() > 120:
        h_plf, c5_plf = proximal_logsumexp(best_h, current_n, current_target,
                                           time_limit=min(50, tleft() * 0.35),
                                           alpha0=0.008, alpha_decay=0.99,
                                           smoothing=500.0, max_iters=15000)
        if c5_plf < best_score:
            best_score = c5_plf
            best_h = h_plf

    if tleft() > 80:
        h_cdf, c5_cdf = pair_cd_v3(best_h, current_n, current_target,
                                   time_limit=min(50, tleft() * 0.4),
                                   max_sweeps=10000,
                                   n_cand_start=40, n_cand_end=140,
                                   max_no_improve=400,
                                   n_active_shifts=7,
                                   reheat_interval=80, reheat_scale=0.02,
                                   lookahead_k=7, filter_thresh=0.2)
        if c5_cdf < best_score:
            best_score = c5_cdf
            best_h = h_cdf

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.5)
        h_final, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                           [5000, 20000, 80000, 200000],
                                           800, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final projection and verification
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
