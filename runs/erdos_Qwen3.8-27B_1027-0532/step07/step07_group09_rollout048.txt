                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def fista_cd(h_init, n, target, time_limit, 
             lr_start=None, max_iter=200000):
    """FISTA-style optimization with subgradient."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    if lr_start is None:
        lr_start = 2.0 / (n * n)
    
    h = h_init.copy().astype(float)
    y = h.copy()
    t_mom = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr_y)
        tol = max(1e-8, 5e-6 * max_corr)
        active = np.where(corr_y >= max_corr - tol)[0]
        
        # Weighted subgradient
        w = np.exp(50.0 * (corr_y[active] - max_corr))
        w_sum = w.sum()
        w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k_idx in zip(w, active):
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Gradient step
        h_new = project_feasible(y - lr_start * grad, target)
        
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        y_new = np.clip(y_new, 0.0, 1.0)
        y = y_new
        h = h_new
        t_mom = t_new
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        for wi, k_idx in zip(w, active):
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = project_feasible(y - lr_start * grad, target)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        y_new = np.clip(y_new, 0.0, 1.0)
        y = y_new
        h = h_new
        t_mom = t_new

        new_c5 = compute_c5(h)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_cd(h_init, n, target, time_limit, n_restarts=3, rng_seed=99):
    start = time.time()
    rng = np.random.RandomState(rng_seed)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    slice_per = time_limit / n_restarts
    for r in range(n_restarts):
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        if r == 0:
            h_cur = h_init.copy()
        else:
            noise = rng.randn(n) * 0.03
            h_cur = project_feasible(h_init + noise, target)

        h_out, c5_out = pair_cd(h_cur, n, target,
                                 time_limit=min(slice_per * 0.8, remaining - 10),
                                 max_sweeps=20000,
                                 n_active_start=5, n_active_end=12,
                                 n_cand_start=30, n_cand_end=100,
                                 max_no_improve=400,
                                 explore_every=300, explore_n=35,
                                 rng_seed=rng_seed + r)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    return best_h, best_c5
