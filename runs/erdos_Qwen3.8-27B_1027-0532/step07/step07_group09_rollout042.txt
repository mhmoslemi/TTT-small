from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, n, tol=1e-6):
    """Compute subgradient of max_k corr[k] at h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    max_corr = np.max(corr)
    
    # Active set with tolerance
    active = np.where(corr >= max_corr - tol * max_corr)[0]
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    
    grad /= len(active)
    grad -= np.mean(grad)  # Remove mean to respect sum constraint
    return grad, corr, max_corr

def momentum_gradient_descent(h_init, n, target, time_limit, 
                                 lr_start=0.005, max_iter=50000,
                                 momentum=0.9, tol=1e-6):
    """Nesterov momentum gradient descent with backtracking line search"""
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = lr_start
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        # Nesterov look-ahead
        h_look = h - momentum * v
        grad, corr, max_corr = compute_gradient(h_look, n, tol=tol)
        
        # Gradient step
        v = momentum * v + lr * grad
        h_new = project_feasible(h - v, target)
        
        # Check improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            v *= 0.5
            no_improve = 0
        elif it > 0 and it % 500 == 0 and no_improve < 100:
            lr = min(lr * 1.1, lr_start * 2)
    
    return best_h, best_c5

def smoothed_max_objective(h, alpha):
    """Smoothed max of correlations using log-sum-exp"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    """SLSQP optimization with smoothed max objective"""
    _g_n = n
    _g_target = target
    
    def cons_fun(x):
        return np.sum(x) - _g_target
    
    def cons_jac(x):
        return np.ones(_g_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(lambda x: smoothed_max_objective(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    
    # Get top M indices
    top_idx = np.argsort(corr)[-top_m:]
    
    # Softmax weights
    c_max = corr[top_idx].max()
    weights = np.exp(200 * (corr[top_idx] - c_max))  # alpha=200
    weights /= weights.sum()
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    
    for wi, k_idx in zip(weights, top_idx):
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    
    grad -= np.mean(grad)
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_topm_gradient(h, n, top_m, alpha=300.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    max_corr = np.max(corr)
    top_idx = np.argsort(corr)[-top_m:]
    c_top_max = corr[top_idx].max()
    raw_w = np.exp(alpha * (corr[top_idx] - c_top_max))
    w_sum = raw_w.sum()
    if w_sum < 1e-30:
        weights = np.ones(top_m) / top_m
    else:
        weights = raw_w / w_sum
    grad = np.zeros(n)
    idxs = np.arange(n)
    for wi, k_idx in zip(weights, top_idx):
        k = int(k_idx) - mid
        if k == 0:

        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def frequency_denoise(h, n, target, cutoff_frac=0.3):

    mask[0] = True
    H[~mask] *= 0.1
    h_f = np.fft.irfft(H, n=n)
    return project_feasible(h_f, target)
