def nesterov_projected_grad(h_init, n, target, alphas, time_limit, start_time, 
                              max_iters=5000, step_sizes=None):
    """Nesterov-accelerated projected gradient on the smooth surrogate."""
    h = h_init.copy().astype(float)
    y = h.copy()  # Nesterov momentum variable
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if step_sizes is None:
        step_sizes = []
        for alpha in alphas:
            # Adaptive step size based on Lipschitz constant estimate
            dx = 2.0 / n
            # The gradient of corr w.r.t. h has entries bounded by 1 in magnitude
            # So L ≈ n_corr (number of terms in the sum) ~ 2n
            L_est = 2 * n
            step_sizes.append(1.0 / (alpha * L_est))
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        alpha_idx = min(it // 1000, len(alphas) - 1)
        alpha = alphas[alpha_idx]
        step = step_sizes[alpha_idx]
        
        # Nesterov step
        momentum = 2.0 / (3.0 + it + 1.0)
        y_new = h + momentum * (h - y)
        y = y_new
        
        # Compute gradient at y
        h_y = project_feasible(y, target)
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Gradient of smooth surrogate w.r.t. h
        n_corr = len(corr)
        mid = n - 1
        g = np.zeros(n)
        idxs = np.arange(n)
        
        # Vectorized gradient computation
        for m_idx in range(n_corr):
            wm = weights[m_idx]
            if wm < 1e-14:
                continue
            k = m_idx - mid
            if k >= 0:
                g[:n-k] += wm * (1.0 - h_y[k:])
                g[k:] += wm * (-h_y[:n-k])
            else:
                kk = -k
                g[kk:] += wm * (1.0 - h_y[:-kk])
                g[:-kk] += wm * (-h_y[kk:])
        
        # Projected gradient step
        h_trial = h_y - step * g
        h_trial = project_feasible(h_trial, target)
        
        # Backtracking line search
        f_y = smooth_max_surrogate(corr, alpha)
        f_trial = smooth_max_surrogate(np.correlate(h_trial, 1.0 - h_trial, mode='full'), alpha)
        
        ls = step
        for _ in range(20):
            if f_trial < f_y - 0.5 * ls * np.dot(g, h_trial - h_y):
                break
            ls *= 0.5
            h_trial = project_feasible(h_y - ls * g, target)
            f_trial = smooth_max_surrogate(np.correlate(h_trial, 1.0 - h_trial, mode='full'), alpha)
        
        h = h_trial
        h_c5 = compute_c5(h)
        if h_c5 < best_c5 - 1e-13:
            best_c5 = h_c5
            best_h = h.copy()
    
    return best_h, best_c5

def multi_lag_cd(h_init, n, target, time_limit, start_time, 
                 max_sweeps=20000, n_top=3, n_cand=30, bisection_iters=16):
    """Coordinate descent considering multiple lags simultaneously."""
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Get top lags
        top_indices = np.argsort(corr)[-n_top:]
        top_weights = corr[top_indices].astype(float)
        wsum = np.sum(top_weights)
        if wsum < 1e-15:
            top_weights = np.ones(n_top) / n_top
        else:
            top_weights = top_weights / wsum
        
        # Compute gradient using weighted top lags
        grad = np.zeros(n)
        for s_idx, w in zip(top_indices, top_weights):
            k = int(s_idx) - mid
            if k >= 0:
                grad[:n-k] += w * (1.0 - h[k:])
                grad[k:] += w * (-h[:n-k])
            else:
                kk = -k
                grad[kk:] += w * (1.0 - h[:-kk])
                grad[:-kk] += w * (-h[kk:])
        grad -= np.mean(grad)
        
        # Pick candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_idx[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Change in correlation for pair (i, j) with delta
                # h[i] -> h[i] - delta, h[j] -> h[j] + delta
                # For each lag k, the change in corr_k is:
                # delta * [(1-h[i+k]) - h[i] ... ] (complex, depends on which indices are in range)
                
                # Use the change matrix approach
                # Δcorr[m] = delta * (change[i, m] - change[j, m])
                # where change[p, m] = d(corr[m])/d(h[p])
                
                # Compute change vectors for i and j
                ci = np.zeros(2*n - 1)
                cj = np.zeros(2*n - 1)
                for m_idx in range(2*n - 1):
                    k = m_idx - mid
                    if k >= 0:
                        if 0 <= i < n - k:
                            ci[m_idx] = 1.0 - h[i + k]
                        if k <= i < n:
                            ci[m_idx] -= h[i - k]
                        if 0 <= j < n - k:
                            cj[m_idx] = 1.0 - h[j + k]
                        if k <= j < n:
                            cj[m_idx] -= h[j - k]
                    else:
                        kk = -k
                        if kk <= i < n:

            h[best_j] += best_delta

Another idea: **Duality gap approach**. The problem min_h max_k f_k(h) has a dual max_{λ≥0, Σλ=1} min_h Σλ_k f_k(h). The inner problem is minimizing a convex combination of bilinear forms, which is a QP! We can solve this QP approximately and use the dual variables to guide the primal.

Actually, f_k(h) = sum_i h[i](1-h[i+k]) = sum_i h[i] - sum_i h[i]h[i+k]. The first term is linear (and mostly constant for interior lags), the second is a quadratic form. So the dual problem is:
max_λ min_h [linear terms - h^T (Σ λ_k A_k) h] subject to 0≤h≤1, sum(h)=n/2

where A_k is the matrix representing the quadratic form for lag k. This is a concave QP (maximizing a concave function), which can be solved!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_grad(h, alpha):
    """Compute gradient of smooth LSE surrogate w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    g = np.zeros(n)
    mid = n - 1
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - mid
        if k >= 0:
            g[:n-k] += w * (1.0 - h[k:])
            g[k:] += w * (-h[:n-k])
        else:
            kk = -k
            g[kk:] += w * (1.0 - h[:-kk])
            g[:-kk] += w * (-h[kk:])
    return g

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global_n = n
    global_target = target
    
    def cons_fun(x):
        return np.sum(x) - global_target
    
    def cons_jac(x):
        return np.ones(global_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        def func(x, p=param):
            return smooth_lse(x, p)
        try:
            res = minimize(func, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def nesterov_cd(h_init, n, target, alphas, time_limit, start_time, max_iters=10000):
    """Nesterov-accelerated projected gradient descent."""
    h = h_init.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 2000:
            break
        
        # Cycle through alphas
        alpha_idx = min(it // 2000, len(alphas) - 1)
        alpha = alphas[alpha_idx]
        
        # Nesterov momentum
        t_k = it + 1
        momentum = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2  # standard Nesterov
        # Actually, use the simpler form: y = h + ((t-1)/(t+1)) * (h - y_prev)
        # Let me use the standard form: beta_k = (1 - 1/(k+1)) / (1 + 1/(k+1))
        beta = (t_k - 1) / (t_k + 1) if t_k > 1 else 0.0
        y = h + beta * (h - y)
        
        # Compute gradient at y
        h_y = project_feasible(y, target)
        g = compute_grad(h_y, alpha)
        
        # Step size with backtracking
        f_y = smooth_lse(h_y, alpha)
        step = 1.0 / (alpha * (2 * n + 1))  # initial estimate
        h_trial = project_feasible(h_y - step * g, target)
        f_trial = smooth_lse(h_trial, alpha)
        
        for _ in range(30):
            if f_trial <= f_y - 0.5 * step * np.dot(g, h_y - h_trial):
                break
            step *= 0.5
            h_trial = project_feasible(h_y - step * g, target)
            f_trial = smooth_lse(h_trial, alpha)
        
        h = h_trial
        h_c5 = compute_c5(h)
        if h_c5 < best_c5 - 1e-13:
            best_c5 = h_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def fourier_refine(h_init, n, target, K, alphas, time_limit, start_time, max_iters=500):
    """Optimize in Fourier coefficient space."""
    x = np.arange(n) * (2.0 / n)
    basis = np.zeros((n, 2 * K + 1))
    basis[:, 0] = 1.0 / np.sqrt(n)
    for k in range(1, K + 1):
        basis[:, 2 * k - 1] = np.sqrt(2.0 / n) * np.cos(k * np.pi * x)
        basis[:, 2 * k] = np.sqrt(2.0 / n) * np.sin(k * np.pi * x)
    
    coefs = basis.T @ h_init
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def fobj(c):
            h = basis @ c
            h = project_feasible(h, target)
            return smooth_lse(h, alpha)
        
        def fgrad(c):
            h = basis @ c
            h = project_feasible(h, target)
            g_h = compute_grad(h, alpha)
            return basis.T @ g_h
        
        bounds = [(-5, 5)] * len(coefs)
        try:
            res = minimize(fobj, coefs, method='L-BFGS-B', jac=fgrad,
                          bounds=bounds, options={'maxiter': max_iters, 'ftol': 1e-14})
            h_new = project_feasible(basis @ res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                coefs = res.x
        except Exception:
            pass
    
    return best_h, best_c5

def active_pair_cd(h_init, n, target, time_limit, start_time,
                    max_sweeps=30000, n_top=3, n_cand_start=20, n_cand_end=60,
                    max_no_improve=300, bisection_iters=16):
    """Paired coordinate descent with multi-lag bisection."""
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Weighted top lags
        top_idx = np.argsort(corr)[-n_top:]
        top_w = corr[top_idx].astype(float)
        ws = np.sum(top_w)
        if ws < 1e-15:
            top_w = np.ones(n_top) / n_top
        else:
            top_w = top_w / ws
        
        # Gradient
        grad = np.zeros(n)
        for si, w in zip(top_idx, top_w):
            k = int(si) - mid
            if k >= 0:
                grad[:n-k] += w * (1.0 - h[k:])
                grad[k:] += w * (-h[:n-k])
            else:
                kk = -k
                grad[kk:] += w * (1.0 - h[:-kk])
                grad[:-kk] += w * (-h[kk:])
        grad -= np.mean(grad)
        
        # Candidates
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 5)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Compute change in corr
                g_change = np.zeros(n_corr)
                for m in range(n_corr):
                    k = m - mid
                    if k >= 0:
                        if 0 <= i < n - k:
                            g_change[m] += 1.0 - h[i + k]
                        if k <= i < n:
                            g_change[m] -= h[i - k]
                        if 0 <= j < n - k:
                            g_change[m] -= (1.0 - h[j + k])
                        if k <= j < n:
                            g_change[m] += h[j - k]
                    else:
                        kk = -k
                        if kk <= i < n:
                            g_change[m] += 1.0 - h[i - kk]
                        if 0 <= i < n - kk:
                            g_change[m] -= h[i + kk]
                        if kk <= j < n:
                            g_change[m] -= (1.0 - h[j - kk])
                        if 0 <= j < n - kk:
                            g_change[m] += h[j + kk]
                
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_change
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g_change[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_change) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def lse_grad_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_v = np.exp(shifted)
    total = np.sum(exp_v)
    return exp_v / total


def compute_grad_fast(h, corr, weights, n_top=20):
    n = len(h)
    mid = n - 1
    g = np.zeros(n)
    top_idx = np.argsort(weights)[-n_top:]
    for m in top_idx:
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - mid
        if k >= 0:
            g[:n - k] += w * (1.0 - h[k:])
            g[k:] += w * (-h[:n - k])
        else:
            kk = -k
            g[kk:] += w * (1.0 - h[:-kk])
            g[:-kk] += w * (-h[kk:])
    return g


def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target

    def cons_fun(x):
        return np.sum(x) - _g_target

    def cons_jac(x):
        return np.ones(_g_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        def func(x, p=param):
            return smooth_lse(x, p)
        try:
            res = minimize(func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def nesterov_optimize(h_init, n, target, alphas, time_limit, start_time, max_iters=15000):
    h = h_init.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 3000:
            break

        alpha_idx = min(it // 3000, len(alphas) - 1)
        alpha = alphas[alpha_idx]

        t_k = float(it + 1)
        if it > 0:
            beta = (1.0 - 1.0 / t_k) / (1.0 + 1.0 / t_k)
        else:
            beta = 0.0
        y = h + beta * (h - y)
        h_y = project_feasible(y, target)

        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        weights = lse_grad_weights(corr, alpha)
        g = compute_grad_fast(h_y, corr, weights, n_top=25)

        f_y = smooth_lse(h_y, alpha)
        step = 0.8 / (alpha * (2.0 * n + 1.0))
        h_trial = project_feasible(h_y - step * g, target)
        f_trial = smooth_lse(h_trial, alpha)

        for _ in range(25):
            if f_trial <= f_y - 0.4 * step * np.dot(g, h_y - h_trial):
                break
            step *= 0.5
            h_trial = project_feasible(h_y - step * g, target)
            f_trial = smooth_lse(h_trial, alpha)

        h = h_trial
        h_c5 = compute_c5(h)
        if h_c5 < best_c5 - 1e-13:
            best_c5 = h_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5


def fourier_refine(h_init, n, target, K, alphas, time_limit, start_time, max_iters=400):
    x = np.arange(n) * (2.0 / n)
    basis = np.zeros((n, 2 * K + 1))
    basis[:, 0] = 1.0 / np.sqrt(n)
    for k in range(1, K + 1):
        basis[:, 2 * k - 1] = np.sqrt(2.0 / n) * np.cos(k * np.pi * x)
        basis[:, 2 * k] = np.sqrt(2.0 / n) * np.sin(k * np.pi * x)

    coefs = basis.T @ h_init
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break

        def fobj(c):
            h = basis @ c
            h = project_feasible(h, target)
            return smooth_lse(h, alpha)

        def fgrad(c):
            h = basis @ c
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            weights = lse_grad_weights(corr, alpha)
            g_h = compute_grad_fast(h, corr, weights, n_top=20)
            return basis.T @ g_h

        bounds_c = [(-5.0, 5.0)] * len(coefs)
        try:
            res = minimize(fobj, coefs, method='L-BFGS-B', jac=fgrad,
                           bounds=bounds_c, options={'maxiter': max_iters, 'ftol': 1e-14})
            h_new = project_feasible(basis @ res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                coefs = res.x
        except Exception:
            pass

    return best_h, best_c5


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        for m in range(n_corr):
            k = m - mid
            if k >= 0:
                if 0 <= p < n - k:
                    change[p, m] += 1.0 - h[p + k]
                if k <= p < n:
                    change[p, m] -= h[p - k]
            else:
                kk = -k
                if kk <= p < n:
                    change[p, m] += 1.0 - h[p - kk]
                if 0 <= p < n - kk:
                    change[p, m] -= h[p + kk]
    return change


def update_change_row(change, p, h, n, mid):
    n_corr = 2 * n - 1
    change[p, :] = 0.0
    for m in range(n_corr):
        k = m - mid
        if k >= 0:
            if 0 <= p < n - k:
                change[p, m] += 1.0 - h[p + k]
            if k <= p < n:
                change[p, m] -= h[p - k]
        else:
            kk = -k
            if kk <= p < n:
                change[p, m] += 1.0 - h[p - kk]
            if 0 <= p < n - kk:
                change[p, m] -= h[p + kk]


def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=25000, n_top=3, n_cand_start=15, n_cand_end=50,
            max_no_improve=350, bisection_iters=16):
    dx = 2.0 / n
    mid = n - 1
    n_corr = 2 * n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_top:]
        top_w = corr[top_idx].astype(float)
        ws = np.sum(top_w)
        if ws < 1e-15:
            top_w = np.ones(n_top) / n_top
        else:
            top_w = top_w / ws

        grad = np.zeros(n)
        for si, w in zip(top_idx, top_w):
            k = int(si) - mid
            if k >= 0:
                grad[:n - k] += w * (1.0 - h[k:])
                grad[k:] += w * (-h[:n - k])
            else:
                kk = -k
                grad[kk:] += w * (1.0 - h[:-kk])
                grad[:-kk] += w * (-h[kk:])
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, mid)
            update_change_row(change, best_j, h, n, mid)

    return best_h, best_c5

    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
