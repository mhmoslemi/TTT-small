import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h onto {0≤x≤1, sum(x)=target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

# Main pair CD with incremental correlation update
def pair_cd(h_init, n, target, time_limit, max_sweeps=100000,
            n_cand_start=50, n_cand_end=120, max_no_improve=500):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()

        if no_improve > max_no_improve:

        c5_val = float(np.max(corr))

            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:

- h[i]*h[i-k] → (h[i]-δ)*h[i-k] = h[i]*h[i-k] - δ*h[i-k] (if i ∈ S_k)
- h[i+k]*h[i] → h[i+k]*(h[i]-δ) = h[i+k]*h[i] - δ*h[i+k] (if i+k ∈ S_k)
- Similarly for j.

So the exact change is:
Δc_k = [1_{i∈S_k} - 1_{j∈S_k}] * δ 
       - δ * [h[i-k]*1_{i∈S_k} + h[i+k]*1_{i+k∈S_k} - h[j-k]*1_{j∈S_k} - h[j+k]*1_{j+k∈S_k}]

Which is exactly:
Δc_k = δ * (-change[i,k] + change[j,k])

where change[p,k] is computed at the OLD h. Wait, that's not right either, because change[p,k] includes the (1-h[p-k]) term which changes when h changes.

Let me just be explicit. The change matrix entry change[p,k] = ∂c_k/∂h[p] = (1-h[p-k])*1_{p∈S_k} - h[p+k]*1_{p+k∈S_k}

After changing h[i] by -δ and h[j] by +δ:
- change[i,k] changes because h[i-k] and h[i+k] might change (if i-k=j or i+k=j)
- change[j,k] changes similarly
- change[p,k] for other p changes if p-k=i or p-k=j or p+k=i or p+k=j

So the incremental update is NOT exact. The first-order update is:
corr_new ≈ corr_old + δ * (change[j,:] - change[i,:])

This is exact to first order in δ. For δ < 1, the error is O(δ^2) which is small.

To be safe, I'll recompute the correlation every 3-5 iterations. The cost is O(n * n_corr) = O(n^2) per recompute, which for n=800 is 640K operations, about 0.5ms. Over 10000 iterations, that's 5 extra seconds, which is fine.

f_smooth(h) = (max(corr) + log(∑ exp(α*(corr - max(corr))/α)) / α) * dx

Wait, the standard log-sum-exp is:
log_sum_exp(x) = max(x) + log(∑ exp(x - max(x)))

So the smooth approximation is:
f_smooth(h) = log_sum_exp(corr * dx) ≈ max(corr * dx) = C₅

When α → ∞, log_sum_exp(α * corr) / α → max(corr).
Actually, log_sum_exp(x) / α where x = α * corr:
= (max(α*corr) + log(∑ exp(α*(corr - max(corr)))))/α
= max(corr) + (1/α) * log(∑ exp(α*(corr - max(corr))))
→ max(corr) as α → ∞

So the smooth approximation f_smooth(h) = log_sum_exp(α * corr) / α → max(corr) as α → ∞.

The gradient of f_smooth is:
∂f_smooth/∂h[p] = (1/α) * ∑_k α * exp(α*(c_k - max_c)) / ∑_j exp(α*(c_j - max_c)) * ∂c_k/∂h[p]
= ∑_k softmax_k * ∂c_k/∂h[p]

where softmax_k = exp(α*(c_k - max_c)) / ∑_j exp(α*(c_j - max_c))

This is a weighted average of the individual gradients, with weights concentrated on the active shifts. It's a smooth approximation of the subgradient of the max function.

For SLSQP to work well, we need α large enough that the smooth approximation is tight, but not so large that the optimization is ill-conditioned.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:

                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
