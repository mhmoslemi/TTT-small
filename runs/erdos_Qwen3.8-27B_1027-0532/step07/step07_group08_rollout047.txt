_g_target = 0.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_target = target

    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_eq_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas=None, maxiter=1000, deadline=None):
    global _g_n, _g_target
    if alphas is None:
        alphas = [100, 500, 2000, 10000]
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=60, max_no_improve=300,
            n_active_shifts=4, reheat_interval=80, reheat_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0
    
    # Build change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        sm = shifts
        pos_mask = (p + sm >= 0) & (p + sm < n)
        neg_mask = (p - sm >= 0) & (p - sm < n)
        change[p, pos_mask] += (1.0 - h[p + sm[pos_mask]])
        change[p, neg_mask] -= h[p - sm[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Reheat
        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 20 == 0:
            rng = np.random.RandomState(sweep * 17 + 11)
            n_pert = max(5, n // 50)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            if compute_c5(h_pert) < c5_val + 1e-5:
                h = h_pert
                for p in range(n):
                    sm = shifts
                    pos_mask = (p + sm >= 0) & (p + sm < n)
                    neg_mask = (p - sm >= 0) & (p - sm < n)
                    change[p, pos_mask] = (1.0 - h[p + sm[pos_mask]])  # Reset
                    change[p, neg_mask] = 0.0
                    change[p, pos_mask] += (1.0 - h[p + sm[pos_mask]])
                    change[p, neg_mask] -= h[p - sm[neg_mask]]
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue
        
        k_star_idx = int(np.argmax(corr))
        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        weights = corr[top_k] / max(corr_top_sum, 1e-10)
        
        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                # Quick check: if the change at the critical shift is wrong, skip
                if g[k_star_idx] > 0.5 * np.max(np.abs(g)):
                    continue
                
                # Binary search
                a_d, b_d = delta_lo, delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial_c5:
                    best_trial_c5 = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def nesterov_smooth(h_init, n, target, time_limit, alpha=500, lr=0.1, momentum=0.9):
    """Nesterov accelerated gradient descent on the smoothed objective."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    y = h.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Nesterov step
        beta = momentum
        y_new = h + beta * (h - y + v)  # Eh, standard Nesterov
        # Actually, let's use the standard form:
        # y = h + (beta - 1)/beta * (h - h_prev)  -- nah, let me use the simpler form
        
        # Simpler: standard momentum GD
        # Compute gradient of smoothed objective at y
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        # To avoid O(n^2) gradient computation, only use top shifts
        top_shifts = np.argsort(corr_y)[-20:]
        for s in top_shifts:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = weights[s]
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        
        # Momentum update
        v = beta * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Nesterov lookahead
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        
        h = h_new
        
        # Evaluate
        sc = compute_c5(h)
        if sc < best_c5 - 1e-10:
            best_c5 = sc
            best_h = h.copy()
    
    return best_h, best_c5

        b_const[ii] = corr[s] - A[ii] @ h  # Should be 0 (linearization error)

    # The linearization is: f_k(h + d) ≈ f_k(h) + A[ii] @ d
    # We want: f_k(h) + A[ii] @ d ≤ t
    # i.e., A[ii] @ d - t ≤ -f_k(h)
    
    # Variables: d[0], ..., d[n-1], t
    # Objective: minimize t  (c = [0,...,0, 1])
    # Constraints:
    #   A @ d - t ≤ -f_k(h)  for each active k  (n_active rows)
    #   sum(d) = 0  (1 row, equality)
    #   -eps ≤ d[i] ≤ eps  for each i  (2n rows)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ vars ≤ b_ub
    # A @ d - 1*t ≤ -corr[top_k[i]]
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -corr[top_k].copy()
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -eps ≤ d[i] ≤ min(eps, 1-h[i]), and t ≥ 0
    bounds = []
    for i in range(n):
        lo = max(-eps, -h[i])
        hi = min(eps, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t ≥ 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    d = res.x[:n]
    t_val = res.x[n]
    return d, t_val
