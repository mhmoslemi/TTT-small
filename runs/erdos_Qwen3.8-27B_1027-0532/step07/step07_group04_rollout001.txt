from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, k_star, n):
    """Compute subgradient of corr[k_star] w.r.t. h."""
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    g[v1] += (1.0 - h[idxs[v1] - k_star])
    g[v2] -= h[idxs[v2] + k_star]
    return g

def subgradient_descent(h_init, n, target, time_limit, steps=1000000, 
                        alpha_init=0.01, use_bb=True):
    """Subgradient descent with Barzilai-Borwein step size."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    h_prev = h.copy()
    g_prev = None
    alpha = alpha_init
    
    for step in range(steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        
        g = compute_subgradient(h, k_star, n)
        g -= np.mean(g)  # preserve sum constraint
        
        if use_bb and g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            sy = np.dot(s, y)
            nn = np.dot(s, s)
            yy = np.dot(y, y)
            if abs(sy) > 1e-12 and nn > 1e-12:
                # BB step size (use the one that gives smaller step if negative)
                if sy > 0:
                    alpha = nn / sy
                else:
                    alpha = nn / sy  # will be negative, so we use abs
                    alpha = abs(alpha)
                alpha = min(max(alpha, 1e-6), 0.5)
        
        h_prev = h.copy()
        g_prev = g.copy()
        
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target)
        
        # Check if we made progress, if not reduce alpha
        new_c5 = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * dx
        if new_c5 > c5_val * 1.01:
            alpha *= 0.5
            h_new = h - alpha * g
            h_new = project_feasible(h_new, target)
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj_topk(x, alpha, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-k:]
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr[idx] - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_full(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_init, n, target, time_limit, start_time, alphas, k_smooth, maxiter):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    if k_smooth > 0:
        obj_func = lambda x, a: _smooth_obj_topk(x, a, k_smooth)
    else:
        obj_func = lambda x, a: _smooth_obj_full(x, a)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        n_l = len(low_set)
        n_h = len(high_set)

        delta_lo_arr = np.maximum(h[high_set][np.newaxis, :] - 1.0, -h[low_set][:, np.newaxis])
        delta_mid = 0.5 * (delta_lo_arr + delta_hi_arr)

        new_corr = corr[np.newaxis, np.newaxis, :] + delta_mid[:, :, np.newaxis] * g_all
        new_max = np.max(new_corr, axis=2)

        diag_mask = (low_set[:, np.newaxis] == high_set[np.newaxis, :])
        new_max[diag_mask] = np.inf

        bi, bj = np.unravel_index(int(np.argmin(new_max)), new_max.shape)
        i = int(high_set[bj])
        j = int(low_set[bi])
        delta_init = float(delta_mid[bi, bj])

        if abs(delta_init) < 1e-12:
            continue

        g = change[j] - change[i]

        d_lo = max(h[i] - 1.0, -h[j])
        d_hi = min(h[i], 1.0 - h[j])
        if d_hi - d_lo < 1e-12:

            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:

            if f_prime > 1e-12:

            else:
                a_d = mid_d

        h[j] += delta_star
        h = np.clip(h, 0.0, 1.0)

def subgrad_momentum(h_init, n, target, time_limit, steps=5000000,
                     alpha_init=0.02, momentum=0.9):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    h_prev = h.copy()
    g_prev = None
    alpha = alpha_init
    no_improve = 0

    for step in range(steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 200000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        idxs = np.arange(n)
        g = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        g[v1] += (1.0 - h[idxs[v1] - k_star])
        g[v2] -= h[idxs[v2] + k_star]
        g -= np.mean(g)

        if g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            sy = np.dot(s, y)
            nn = np.dot(s, s)
            if abs(sy) > 1e-10 and nn > 1e-10:
                alpha_bb = max(min(nn / abs(sy), 0.1), 1e-5)
                alpha = 0.5 * alpha + 0.5 * alpha_bb

        h_prev = h.copy()
        g_prev = g.copy()

        velocity = momentum * velocity - alpha * g
        h_new = h + velocity
        h_new = project_feasible(h_new, target)

        new_c5 = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))) * dx
        if new_c5 > c5_val * 1.05 + 1e-6:
            alpha *= 0.5
            velocity = momentum * velocity - alpha * g
            h_new = h + velocity
            h_new = project_feasible(h_new, target)

        h = h_new

    return best_h, best_c5

def compute_c5_val_fast(h, n):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    cur_n = n0
    cur_target = target0

    # Phase 1: SLSQP quick start
    if tleft() > 60:
        tl = min(50, tleft() * 0.5)
        h_pol, sc = slsqp_refine(best_h, cur_n, cur_target, tl, start_time,
                                  [200, 800], 5, 500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Vectorized CD at n=400
    if tleft() > 150:
        tl = min(120, tleft() * 0.6)
        h_cd, c5_cd = vec_cd(best_h, cur_n, cur_target, tl,
                              max_sweeps=100000, n_cand_start=25, n_cand_end=70,
                              max_no_improve=600, n_multi_shifts=1)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Subgradient with momentum
    if tleft() > 80:
        tl = min(60, tleft() * 0.5)
        h_sg, c5_sg = subgrad_momentum(best_h, cur_n, cur_target, tl,
                                        steps=2000000, alpha_init=0.015, momentum=0.9)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: Multi-shift CD
    if tleft() > 100:
        tl = min(80, tleft() * 0.5)
        h_ms, c5_ms = vec_cd(best_h, cur_n, cur_target, tl,
                              max_sweeps=60000, n_cand_start=30, n_cand_end=70,
                              max_no_improve=500, n_multi_shifts=3)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 5: SA for escape
    if tleft() > 80:
        tl = min(60, tleft() * 0.5)
        h_sa, c5_sa = sa_block(best_h, cur_n, cur_target, tl,
                                max_steps=150000, t_start=0.025, t_end=0.0003,
                                n_cand=50, block_size=4)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 6: SLSQP refinement
    if tleft() > 60:
        tl = min(50, tleft() * 0.5)
        h_pol2, sc2 = slsqp_refine(best_h, cur_n, cur_target, tl, start_time,
                                    [1000, 5000], 3, 800)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 7: Go to n=800
    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl = min(150, tleft() * 0.55)
        h_cd1, c5_cd1 = vec_cd(h1, n1, target1, tl,
                                max_sweeps=80000, n_cand_start=30, n_cand_end=80,
                                max_no_improve=500, n_multi_shifts=2)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            cur_n = n1
            cur_target = target1

    # Phase 8: SLSQP at n=800
    if tleft() > 80:
        tl = min(60, tleft() * 0.5)
        h_pol3, sc3 = slsqp_refine(best_h, cur_n, cur_target, tl, start_time,
                                    [3000, 10000], 3, 1000)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    # Phase 9: More CD at current resolution
    if tleft() > 120:
        tl = min(100, tleft() * 0.5)
        h_cd2, c5_cd2 = vec_cd(best_h, cur_n, cur_target, tl,
                                max_sweeps=60000, n_cand_start=40, n_cand_end=90,
                                max_no_improve=400, n_multi_shifts=1)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 10: Try alternative starts + SA
    if tleft() > 100:
        for attempt in range(3):
            if tleft() < 40:
                break
            h_alt = generate_alternative_start(cur_n, cur_target, rng)
            tl_sa = min(30, tleft() * 0.25)
            h_sa_alt, c5_sa_alt = sa_block(h_alt, cur_n, cur_target, tl_sa,
                                            max_steps=50000, t_start=0.04, t_end=0.0005,
                                            n_cand=50, block_size=6)
            if c5_sa_alt < best_score:
                best_score = c5_sa_alt
                best_h = h_sa_alt

    # Phase 11: Go to n=1200
    if tleft() > 200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl = min(150, tleft() * 0.55)
        h_cd3, c5_cd3 = vec_cd(h2, n2, target2, tl,
                                max_sweeps=50000, n_cand_start=25, n_cand_end=70,
                                max_no_improve=500, n_multi_shifts=1)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            cur_n = n2
            cur_target = target2

    # Phase 12: SLSQP at final resolution
    if tleft() > 60:
        tl = min(50, tleft() * 0.6)
        h_pol4, sc4 = slsqp_refine(best_h, cur_n, cur_target, tl, start_time,
                                    [5000, 20000], 3, 600)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_pol4

    # Phase 13: Final CD polish
    if tleft() > 40:
        tl = min(25, tleft() * 0.7)
        h_cd4, c5_cd4 = vec_cd(best_h, cur_n, cur_target, tl,
                                max_sweeps=30000, n_cand_start=50, n_cand_end=100,
                                max_no_improve=200, n_multi_shifts=2)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Final SLSQP touch
    if tleft() > 30:
        tl = min(20, tleft() * 0.8)
        h_pf, sc_pf = slsqp_refine(best_h, cur_n, cur_target, tl, start_time,
                                    [10000], 2, 300)
        if sc_pf < best_score:
            best_score = sc_pf
            best_h = h_pf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
