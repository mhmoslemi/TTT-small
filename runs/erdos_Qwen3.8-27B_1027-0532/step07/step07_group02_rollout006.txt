```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    s = np.sum(weights)
    weights /= s
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s_idx in range(len(weights)):
        w = weights[s_idx]
        if w < 1e-12:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

One more optimization: in the vectorized CD, instead of iterating over t values and computing max(corr + delta*g) for all pairs, I can be smarter:

For a given pair (i,j) with g = change[j] - change[i], the function f(delta) = max_s(corr[s] + delta * g[s]) is convex piecewise linear. The minimum is at the point where the "slope" changes from negative to positive. The slope at delta is max(g[s] for s in argmax(corr[s] + delta*g[s])).

So for a given delta, the "active set" is argmax(corr + delta*g), and the slope is max(g[active_set]).

This means for each pair, the optimal delta is where max(g[active_set]) changes sign. We can binary search for this, but doing it for all pairs simultaneously is the challenge.

- delta ≥ 0 (we're moving from high to low)
- The function is convex, so we can use ternary search or the fact that the minimum is where the subgradient contains 0.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    s = np.sum(weights)
    weights = weights / s
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s_idx in range(len(weights)):
        w = weights[s_idx]
        if w < 1e-12:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def vec_pair_cd(h_init, n, target, time_limit, max_sweeps=80000,
                n_cand_start=40, n_cand_end=100, max_no_improve=500,
                n_deltas=25, n_refine=15, n_binsearch=22,
                stochastic_frac=0.03, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    t_vals = np.linspace(0.02, 0.98, n_deltas)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        top_k = np.argsort(corr)[-10:]
        k_weights = np.zeros(n_corr)
        top_vals = corr[top_k]
        k_weights[top_k] = np.exp((top_vals - top_vals.max()) / 2.0)
        k_weights /= k_weights.sum()

        grad = np.zeros(n)
        for s_idx in range(n_corr):
            w = k_weights[s_idx]
            if w < 1e-4:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        n_h = len(high_set)
        n_l = len(low_set)

        g_mat = change[low_set][None, :, :] - change[high_set][:, None, :]
        h_high = h[high_set]
        h_low = h[low_set]
        delta_lo = np.maximum(h_high[:, None] - 1.0, -h_low[None, :])
        delta_hi = np.minimum(h_high[:, None], 1.0 - h_low[None, :])
        valid = (delta_hi - delta_lo) > 1e-12

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        if valid.any():
            best_overall = np.full((n_h, n_l), np.inf)
            corr_exp = corr[None, None, :]
            for t in t_vals:
                deltas = delta_lo + t * (delta_hi - delta_lo)
                new_vals = corr_exp + deltas[:, :, None] * g_mat
                cur_max = np.max(new_vals, axis=2)
                cur_max[~valid] = np.inf
                np.minimum(best_overall, cur_max, out=best_overall)
                del new_vals

            flat_best = best_overall[valid]
            if len(flat_best) > 0:
                min_val = np.min(flat_best)
                if min_val * dx < best_trial_c5:
                    valid_flat = np.where(valid)[0]
                    best_local = valid_flat[np.argmin(flat_best)]
                    best_ii, best_jj = np.unravel_index(best_local, (n_h, n_l))
                    candidate_pairs = [(best_ii, best_jj)]
                    n_valid = len(flat_best)
                    if n_valid > n_refine:
                        sorted_valid = np.argsort(flat_best)
                        for fi in range(1, min(n_refine, n_valid)):
                            vpos = valid_flat[sorted_valid[fi]]
                            candidate_pairs.append(np.unravel_index(vpos, (n_h, n_l)))

                    for (cii, cjj) in candidate_pairs:
                        i = int(high_set[cii])
                        j = int(low_set[cjj])
                        g = g_mat[cii, cjj]
                        a_d = float(delta_lo[cii, cjj])
                        b_d = float(delta_hi[cii, cjj])
                        if b_d - a_d < 1e-12:
                            continue
                        for _ in range(n_binsearch):
                            mid_d = 0.5 * (a_d + b_d)
                            vals = corr + mid_d * g
                            f_mid = np.max(vals)
                            active_mask = vals >= f_mid - 1e-14
                            if active_mask.any():
                                f_prime = np.max(g[active_mask])
                            else:
                                f_prime = 0.0
                            if f_prime > 1e-12:
                                b_d = mid_d
                            else:
                                a_d = mid_d
                        d_star = 0.5 * (a_d + b_d)
                        if abs(d_star) < 1e-12:
                            continue
                        approx = np.max(corr + d_star * g) * dx
                        if approx < best_trial_c5:
                            best_trial_c5 = approx
                            best_i = i
                            best_j = j
                            best_delta = d_star

        if stochastic_frac > 0:
            n_rand = max(1, int(stochastic_frac * n_h * n_l / 10))
            for _ in range(n_rand):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if h[i] <= h[j]:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) > 1e-12:
                    approx = np.max(corr + d_star * g) * dx
                    if approx < best_trial_c5:
                        best_trial_c5 = approx
                        best_i = i
                        best_j = j
                        best_delta = d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def subgradient_polish(h_init, n, target, time_limit, n_steps=50000, decay=0.999, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    step = 0.01
    no_improve = 0

    for step_i in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve > 200:
                step *= decay
            no_improve = min(no_improve, 1000)

        if no_improve % 100 == 50:
            top_m = np.argsort(corr)[-20:]
            w = np.exp((corr[top_m] - corr[top_m].max()) / 3.0)
            w = w / w.sum()
        else:
            top_m = np.argsort(corr)[-5:]
            w = np.ones(len(top_m)) / len(top_m)

        grad = np.zeros(n)
        for mi in range(len(top_m)):
            s = int(top_m[mi])
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[mi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[mi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        h_new_c5 = compute_c5(h_new)
        if h_new_c5 < c5_val + 1e-12:
            h = h_new
        else:
            h = h_new * 0.5 + h * 0.5
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def anneal_from_binary(h_binary, n, target, time_limit, n_steps=20000, seed=99):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_binary.copy().astype(float)
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    temp = 5.0
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    for step_i in range(n_steps):
        if time.time() - start > time_limit:
            break
        progress = float(step_i) / n_steps
        temp = 5.0 * (1.0 - progress) + 0.5 * progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr_shifted = (corr - np.max(corr)) / temp
        weights = np.exp(corr_shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum

        grad = np.zeros(n)
        for s_idx in range(len(weights)):
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        lr = 0.02 * (1.0 - progress * 0.5)
        i_move = int(np.argmax(grad))
        j_move = int(np.argmin(grad))
        if i_move == j_move:
            continue
        d_lo = max(h[i_move] - 1.0, -h[j_move])
        d_hi = min(h[i_move], 1.0 - h[j_move])
        if d_hi - d_lo < 1e-12:
            continue
        delta = lr
        delta = max(d_lo, min(delta, d_hi))
        if abs(delta) < 1e-12:
            continue
        h[i_move] -= delta
        h[j_move] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    return best_h, best_c5

            h_cd1, c5_cd1 = vec_pair_cd(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.5),
                                         max_sweeps=50000,
                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=500,
                                         n_deltas=20, n_refine=8,
                                         n_binsearch=20,
                                         stochastic_frac=0.03, seed=251)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                n0 = n1
                target0 = target1
        if tleft() > 60:
            h_pol1, sc_pol1 = slsqp_optimize(best_h, n0, target0,
                                              [5000, 20000, 100000],
                                              1500, 1e-15)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    if tleft() > 100:
        h_cd3, c5_cd3 = vec_pair_cd(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.6),
                                     max_sweeps=50000,

                                     stochastic_frac=0.04, seed=501)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 50:
        h_final, sc_final = slsqp_optimize(best_h, n0, target0,
                                            [50000, 200000, 500000],
                                            2000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
