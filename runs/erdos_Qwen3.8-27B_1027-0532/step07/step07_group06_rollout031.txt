import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            if use_pnorm:
                res = minimize(lambda x: pnorm_obj(x, param), h_cur,
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            else:
                res = minimize(lambda x: smooth_lse(x, param), h_cur,
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def vectorized_pair_cd(h_init, n, target, time_limit, start_time,
                       max_sweeps=50000, n_cand=30, n_top_shifts=3,
                       bisection_iters=18, batch_size=1000):

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr)
        
        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Compute gradient using top shifts
        top_idx = np.argsort(corr)[-n_top_shifts:]
        weights = corr[top_idx].astype(float)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top_shifts) / n_top_shifts
        else:
            weights /= wsum
        
        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidates

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0
_g_p = 4.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _obj_smooth(x):
    return _smooth_lse(x, _g_alpha)


def _obj_pnorm(x):
    return _pnorm_obj(x, _g_p)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target, _g_alpha, _g_p
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        if use_pnorm:
            _g_p = param
            func = _obj_pnorm
        else:
            _g_alpha = param
            func = _obj_smooth
        try:

                          constraints=cons,

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            change[p1, q] += (1.0 - h[p1 - kk])
            p2 = np.arange(n - kk)
            change[p2, q] -= h[p2 + kk]

import numpy as np

import time

_g_n = 0

_g_alpha = 1.0
_g_p = 4.0
_g_h = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _obj_smooth(x):
    return _smooth_lse(x, _g_alpha)


def _obj_pnorm(x):
    return _pnorm_obj(x, _g_p)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target, _g_alpha, _g_p
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        if use_pnorm:
            _g_p = int(param)
            func = _obj_pnorm
        else:
            _g_alpha = param
            func = _obj_smooth
        try:
            res = minimize(func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
            else:
                h_cur = project_feasible(res.x, target)
                sc2 = compute_c5(h_cur)
                if sc2 < cur_score:
                    cur_score = sc2
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for q in range(n_corr):
        k = q - mid
        if k >= 0:
            n_terms = n - k
            if n_terms > 0:
                p1 = np.arange(n_terms)
                change[p1, q] += (1.0 - h[p1 + k])
            if k > 0:
                p2 = np.arange(k, n)
                change[p2, q] -= h[p2 - k]
        else:
            kk = -k
            p1 = np.arange(kk, n)
            change[p1, q] += (1.0 - h[p1 - kk])
            p2 = np.arange(n - kk)
            change[p2, q] -= h[p2 + kk]
    return change


def update_change_row(change, p, h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change[p, :] = 0.0
    for q in range(n_corr):
        k = q - mid
        if k >= 0:
            if p <= n - 1 - k:
                change[p, q] += (1.0 - h[p + k])
            if k > 0 and p >= k:
                change[p, q] -= h[p - k]
        else:
            kk = -k
            if p >= kk:
                change[p, q] += (1.0 - h[p - kk])
            if p <= n - 1 - kk:
                change[p, q] -= h[p + kk]

                h = project_feasible(h, target)
                val = compute_c5(h)
                if val < best_val - 1e-15:
                    best_val = val
                    best_hi = h[i]

            lo_t = 0.0
            hi_t = 1.0
            h[i] = best_hi
            h = project_feasible(h, target)
            f_mid = compute_c5(h)

            for _ in range(25):
                if time.time() - start_time > time_limit:
                    break
                mid_t = 0.5 * (lo_t + hi_t)
                h[i] = mid_t
                h = project_feasible(h, target)
                f_lo = compute_c5(h)
                h[i] = lo_t
                h = project_feasible(h, target)
                f_at_lo = compute_c5(h)
                h[i] = hi_t
                h = project_feasible(h, target)
                f_at_hi = compute_c5(h)

                if f_lo <= f_at_lo and f_lo <= f_at_hi:
                    lo_t = mid_t
                else:
                    hi_t = mid_t

            h[i] = 0.5 * (lo_t + hi_t)
            h = project_feasible(h, target)
            final_val = compute_c5(h)
            if final_val < best_val - 1e-15:
                best_val = final_val
                improved = True
            h[i] = best_hi if best_hi != h_save else 0.5 * (lo_t + hi_t)
            h = project_feasible(h, target)
            if compute_c5(h) < best_c5 - 1e-14:
                best_c5 = compute_c5(h)
                best_h = h.copy()

    return best_h, best_c5


def stochastic_hill_climb(h_init, n, target, time_limit, start_time,
                          max_steps=30000, base_scale=0.04):
    rng = np.random.RandomState(42)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break

        scale = base_scale * max(0.05, 1.0 - float(step) / max_steps)
        n_pert = int(rng.choice([1, 2, 3]))
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_p
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at original resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        h1, sc1 = slsqp_round(best_h, n0, target0,
                              [100, 500, 2000],
                              600, start_time + 80, start_time, use_pnorm=False)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: p-norm SLSQP at original resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        h2, sc2 = slsqp_round(best_h, n0, target0,
                              [100, 1000, 10000],
                              500, start_time + 60, start_time, use_pnorm=True)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Interpolate to n=800 and run vectorized pair CD
    n1 = 800
    target1 = n1 / 2.0
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 250:
        h_int = interpolate_to(best_h, n1, target1)
        h3, sc3 = vectorized_pair_cd(h_int, n1, target1,
                                     start_time + min(350, remaining - 40), start_time,
                                     max_sweeps=80000, n_cand_start=15, n_cand_end=60,
                                     n_top_shifts=3, bisection_iters=20,
                                     batch_size=2000, max_no_improve=500)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            n0 = n1
            target0 = target1

    # Phase 4: SLSQP polish at new resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 100:
        h4, sc4 = slsqp_round(best_h, n0, target0,
                              [500, 2000, 10000],
                              800, start_time + min(100, remaining - 20), start_time,
                              use_pnorm=False)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: More pair CD with different parameters
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        h5, sc5 = vectorized_pair_cd(best_h, n0, target0,
                                     start_time + min(250, remaining - 40), start_time,
                                     max_sweeps=60000, n_cand_start=25, n_cand_end=80,
                                     n_top_shifts=5, bisection_iters=20,
                                     batch_size=2000, max_no_improve=600)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Block SLSQP
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 120:
        h6, sc6 = block_slsqp_round(best_h, n0, target0,
                                    start_time + min(120, remaining - 20), start_time,
                                    block_size=8, n_rounds=300)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Try higher resolution n=1000
    n2 = 1000
    target2 = n2 / 2.0
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        h_int2 = interpolate_to(best_h, n2, target2)
        h7, sc7 = vectorized_pair_cd(h_int2, n2, target2,
                                     start_time + min(200, remaining - 40), start_time,
                                     max_sweeps=40000, n_cand_start=15, n_cand_end=50,
                                     n_top_shifts=3, bisection_iters=18,
                                     batch_size=1500, max_no_improve=400)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
            n0 = n2
            target0 = target2

    # Phase 8: Stochastic refinement
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        h8, sc8 = stochastic_hill_climb(best_h, n0, target0,
                                        start_time + min(80, remaining - 20), start_time,
                                        max_steps=30000, base_scale=0.03)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP polish
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        h9, sc9 = slsqp_round(best_h, n0, target0,
                              [5000, 20000],
                              600, start_time + min(60, remaining - 10), start_time,
                              use_pnorm=False)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final pair CD
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h10, sc10 = vectorized_pair_cd(best_h, n0, target0,
                                       start_time + min(50, remaining - 10), start_time,
                                       max_sweeps=20000, n_cand_start=30, n_cand_end=80,
                                       n_top_shifts=1, bisection_iters=22,
                                       batch_size=2000, max_no_improve=200)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    # Random restart attempts
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_r, sc_r = vectorized_pair_cd(h_pert, n0, target0,
                                           start_time + min(50, remaining - 20), start_time,
                                           max_sweeps=15000, n_cand_start=20, n_cand_end=50,
                                           n_top_shifts=3, bisection_iters=18,
                                           batch_size=2000, max_no_improve=200)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    # Final feasibility check
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
