```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def enforce_palindromic(h):
    n = len(h)
    h = h.copy()
    for i in range(n // 2):
        avg = (h[i] + h[n - 1 - i]) / 2.0
        h[i] = avg
        h[n - 1 - i] = avg
    return h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    h = enforce_palindromic(h)
    n = len(h)

    # We need sum = target_sum
    half = h[:n//2].copy()

        if abs(diff) < 1e-14:
```

```python
def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def enforce_palindromic(h):
    n = len(h)
    h = h.copy()
    for i in range(n // 2):
        avg = (h[i] + h[n - 1 - i]) * 0.5
        h[i] = avg
        h[n - 1 - i] = avg
    return h

def project_feasible_pal(h, target_sum):
    h = enforce_palindromic(h.copy().astype(float))
    n = len(h)
    half_len = n // 2
    half = h[:half_len].copy()
    target_half = target_sum / 2.0
    for _ in range(100):
        diff = target_half - np.sum(half)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = half < 1.0 - 1e-15
        else:
            mask = half > 1e-15
        if not mask.any():
            half = np.clip(half + diff / half_len, 0.0, 1.0)
            break
        half[mask] += diff / mask.sum()
        half = np.clip(half, 0.0, 1.0)
    h[:half_len] = half
    h[half_len:] = half[::-1]
    return h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _g_n_global = 0
def _g_target_global = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target_global

def _cons_jac(x):
    return np.ones(_g_n_global)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n_global, _g_target_global
    _g_n_global = n
    _g_target_global = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible_pal(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

        h_pol2 = slsqp_round(best_h, n0, target0, [2000, 8000, 30000], 800, 1e-14)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 7: Try upscaling to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd3, c5_cd3 = vectorized_pair_cd(h1, n1, target1,
                                           time_limit=min(180, tleft() * 0.5),
                                           max_sweeps=15000,
                                           n_cand_start=12, n_cand_end=30,
                                           max_no_improve=400,
                                           reheat_threshold=150)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            n0 = n1
            target0 = target1
        if tleft() > 60:
            h_pol3 = slsqp_round(best_h, n0, target0, [1000, 4000, 15000], 500, 1e-14)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 8: Random restarts
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for attempt in range(4):
            if tleft() < 50:
                break
            noise_scale = 0.015 + 0.01 * attempt
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible_pal(h_pert, target0)
            h_ref = vectorized_pair_cd(h_pert, n0, target0,
                                       time_limit=min(30, tleft() * 0.3),
                                       max_sweeps=5000,
                                       n_cand_start=20, n_cand_end=45,
                                       max_no_improve=200,
                                       reheat_threshold=100)
            sc = h_ref[1]
            if sc < best_score:
                best_score = sc
                best_h = h_ref[0]
            if tleft() > 20:
                h_ref2 = slsqp_round(h_pert, n0, target0, [500, 2000], 300, 1e-14)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 9: Final CD
    if tleft() > 60:
        h_cd_final, c5_final = vectorized_pair_cd(best_h, n0, target0,
                                                  time_limit=min(40, tleft() * 0.6),
                                                  max_sweeps=10000,
                                                  n_cand_start=25, n_cand_end=60,
                                                  max_no_improve=200,
                                                  reheat_threshold=100)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 10: Final SLSQP polish
    if tleft() > 25:
        h_final = slsqp_round(best_h, n0, target0, [5000, 20000], 400, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible_pal(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
