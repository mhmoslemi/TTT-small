import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change(h, n):
    """Build the change matrix: change[p, k] = d corr[k] / d h[p]"""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(_g_alpha * (corr - m), -500.0, 0.0)
    return float(m + np.log(np.sum(np.exp(s))) / _g_alpha)

def _slsqp_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(_g_alpha * (corr - m), -500.0, 0.0)
    exp_s = np.exp(s)
    total = np.sum(exp_s)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_s / total
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _slsqp_cons(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

             block_sizes=(1, 2, 3), n_cand_start=30, n_cand_end=80,

    Block coordinate descent for the Erdős overlap problem.
    Moves blocks of mass from high-gradient to low-gradient regions.
    """

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = build_change(h, n)
    corr = get_corr(h)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Compute gradient from top shifts
        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])
        
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            if w < 0.1 * c5_val / dx:  # only significant shifts
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Score for block CD
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_move = None  # (i, j, block_size, delta)
        
        n_tl = len(low_set)
        n_th = len(high_set)
        
        # Score pairs using the top shift
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            
            # Try block sizes
            for bs in block_sizes:
                # Donor block: [i, i+1, ..., i+bs-1]
                # Receiver block: [j, j+1, ..., j+bs-1]
                i_end = min(i + bs, n)
                j_end = min(j + bs, n)
                actual_bs = min(bs, n - i, n - j)
                if actual_bs < 1:
                    continue
                i_end = i + actual_bs
                j_end = j + actual_bs
                
                # Check bounds
                i_block = np.arange(i, i_end)
                j_block = np.arange(j, j_end)
                
                h_i_max = np.min(h[i_block])
                h_j_min = np.min(1.0 - h[j_block])
                delta_hi = min(h_i_max, h_j_min)
                delta_lo = max(0.0, h_i_max - (1.0 - h[j_block]).max())
                # Actually: for the move to be feasible, we need:
                # h[i_m] - delta/actual_bs >= 0 for all m -> delta <= actual_bs * min(h[i_block])
                # h[j_m] + delta/actual_bs <= 1 for all m -> delta <= actual_bs * min(1-h[j_block])
                delta_lo = 0.0
                delta_hi = actual_bs * min(np.min(h[i_block]), np.min(1.0 - h[j_block]))
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                # Compute g vector for this block move
                # corr_new[k] = corr[k] + (delta/actual_bs) * sum_{m=0}^{actual_bs-1} (change[j+m, k] - change[i+m, k])
                g = np.sum(change[j_block], axis=0) - np.sum(change[i_block], axis=0)
                g_scale = 1.0 / actual_bs
                
                # Find optimal delta via bisection on the max
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_scale * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-12
                    f_prime = np.max(g_scale * g[active_mask])
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = np.max(corr + delta_star * g_scale * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (i, j, actual_bs, delta_star)
        
        if best_move is not None:
            i, j, bs, delta = best_move
            per = delta / bs
            h[i:i+bs] -= per
            h[j:j+bs] += per
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in range(i, i+bs):
                update_change_row(change, h, p, shifts, n)
            for p in range(j, j+bs):
                update_change_row(change, h, p, shifts, n)
            corr = get_corr(h)
    
    return best_h, best_c5

def subgrad_descent(h_init, n, target, time_limit, lr=0.01, momentum=0.9, 
                    max_steps=100000, patience=2000):
    """
    Projected subgradient descent with momentum (Nesterov acceleration).
    """
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > patience:
            break
        
        corr = get_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient (average over active shifts)
        threshold = np.max(corr) - 0.05 * np.max(corr)
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)  # project onto tangent of sum constraint
        
        # Nesterov step
        lr_decay = lr / (1.0 + 0.01 * step)
        v = momentum * v + lr_decay * grad
        h_new = h - v
        h = project_feasible(h_new, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_search(h_init, n, target, time_limit, max_steps=20000, seed=777):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        
        scale = 0.03 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(_g_alpha * (corr - m), -500.0, 0.0)
    return float(m + np.log(np.sum(np.exp(s))) / _g_alpha)


def _slsqp_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(_g_alpha * (corr - m), -500.0, 0.0)
    exp_s = np.exp(s)
    total = np.sum(exp_s)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_s / total
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def block_cd(h_init, n, target, time_limit, max_sweeps=50000,
             block_sizes=(1, 2, 3), n_cand_start=30, n_cand_end=90,
             top_k_eval=60, max_no_improve=800, n_shifts=3,
             use_multi_score=True):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change(h, n)
    corr = get_corr(h)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])
        max_corr = corr[k_star_idx]

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            if w < 0.05 * max_corr:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_tl = len(low_set)
        n_th = len(high_set)
        if n_tl == 0 or n_th == 0:
            continue

        if use_multi_score:
            score vectors for multi-shift
            score_mat = np.zeros((n_th, n_tl))
            for s_idx in top_shifts[:min(3, n_shifts)]:
                wi = corr[s_idx] / max_corr
                score_mat += wi * (change[low_set, s_idx][None, :] -
                                   change[high_set, s_idx][:, None])
        else:
            score_mat = change[low_set, k_star_idx][None, :] - \
                        change[high_set, k_star_idx][:, None]

        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_move = None

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            for bs in block_sizes:
                i_end = min(i + bs, n)
                j_end = min(j + bs, n)
                actual_bs = min(bs, n - i, n - j)
                if actual_bs < 1:
                    continue
                i_end = i + actual_bs
                j_end = j + actual_bs

                i_block = np.arange(i, i_end)
                j_block = np.arange(j, j_end)

                min_h_i = np.min(h[i_block])
                min_1_mh_j = np.min(1.0 - h[j_block])
                delta_hi = actual_bs * min(min_h_i, min_1_mh_j)
                delta_lo = 0.0
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = np.sum(change[j_block], axis=0) - np.sum(change[i_block], axis=0)
                g_scale = 1.0 / actual_bs

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_scale * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-12
                    f_prime = np.max(g_scale * g[active_mask])
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g_scale * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (i, j, actual_bs, delta_star)

        if best_move is not None:
            i, j, bs, delta = best_move
            per = delta / bs
            h[i:i + bs] -= per
            h[j:j + bs] += per
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in range(i, i + bs):
                update_change_row(change, h, p, shifts, n)
            for p in range(j, j + bs):
                update_change_row(change, h, p, shifts, n)
            corr = get_corr(h)

    return best_h, best_c5


def subgrad_descent(h_init, n, target, time_limit, lr=0.03, momentum=0.9,
                    max_steps=80000, patience=3000, threshold_frac=0.1):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > patience:
            break

        corr = get_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_c = np.max(corr)
        if max_c < 1e-15:
            break
        threshold = max_c * (1.0 - threshold_frac)
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        lr_decay = lr / (1.0 + 0.005 * step)
        v = momentum * v + lr_decay * grad
        h_new = h - v
        h = project_feasible(h_new, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_search(h_init, n, target, time_limit, max_steps=20000, seed=777):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        scale = 0.03 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 400:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()

    def _tl():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if _tl() > 120:
        h_pol = slsqp_optimize(best_h, n0, target0, [300, 1500, 6000], 500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 350:
        h_cd, c5_cd = block_cd(best_h, n0, target0,
                               time_limit=min(280, _tl() * 0.65),
                               max_sweeps=50000,
                               block_sizes=(1, 2),
                               n_cand_start=30, n_cand_end=80,
                               top_k_eval=60, max_no_improve=800,
                               n_shifts=3, use_multi_score=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tl() > 150:
        h_sg, c5_sg = subgrad_descent(best_h, n0, target0,
                                      time_limit=min(100, _tl() * 0.5),
                                      lr=0.04, momentum=0.9,
                                      max_steps=60000, patience=4000,
                                      threshold_frac=0.15)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if _tl() > 100:
        h_pol = slsqp_optimize(best_h, n0, target0, [2000, 10000, 40000], 600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            n0 = n1
            target0 = target1

        if _tl() > 300:
            h_cd, c5_cd = block_cd(h1, n1, target1,
                                   time_limit=min(220, _tl() * 0.6),
                                   max_sweeps=40000,
                                   block_sizes=(1, 2, 3),
                                   n_cand_start=40, n_cand_end=100,
                                   top_k_eval=70, max_no_improve=1000,
                                   n_shifts=4, use_multi_score=True)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        if _tl() > 80:
            h_sg, c5_sg = subgrad_descent(best_h, n0, target0,
                                          time_limit=min(50, _tl() * 0.5),
                                          lr=0.02, momentum=0.92,
                                          max_steps=40000, patience=3000,
                                          threshold_frac=0.1)
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg

        if _tl() > 60:
            h_pol = slsqp_optimize(best_h, n0, target0, [3000, 15000], 500, 1e-14)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    if _tl() > 200:
        h_cd, c5_cd = block_cd(best_h, n0, target0,
                               time_limit=min(150, _tl() * 0.7),
                               max_sweeps=35000,
                               block_sizes=(1, 2, 3),
                               n_cand_start=50, n_cand_end=110,
                               top_k_eval=80, max_no_improve=600,
                               n_shifts=4, use_multi_score=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tl() > 100:
        h_ss, c5_ss = stochastic_search(best_h, n0, target0,
                                        time_limit=min(50, _tl() * 0.5),
                                        max_steps=15000, seed=12345)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    if _tl() > 60:
        h_pol = slsqp_optimize(best_h, n0, target0, [5000, 20000, 80000], 400, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04, 0.08]:
            if _tl() < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_optimize(h_pert, n0, target0,
                                   [1000, 5000, 20000], 400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

            if _tl() > 40:
                h_cd, c5_cd = block_cd(h_ref, n0, target0,
                                       time_limit=min(25, _tl() * 0.4),
                                       max_sweeps=10000,

                                       n_cand_start=40, n_cand_end=70,
                                       top_k_eval=40, max_no_improve=300,
                                       n_shifts=3, use_multi_score=True)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    if _tl() > 50:
        h_cd, c5_cd = block_cd(best_h, n0, target0,
                               time_limit=min(35, _tl() * 0.7),
                               max_sweeps=15000,
                               block_sizes=(1, 2, 3),
                               n_cand_start=60, n_cand_end=120,
                               top_k_eval=100, max_no_improve=200,
                               n_shifts=5, use_multi_score=True)

            best_h = h_cd

    if _tl() > 30:
        h_pol = slsqp_optimize(best_h, n0, target0, [20000, 80000], 300, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tl() > 15:
        h_sg = subgrad_descent(best_h, n0, target0,
                               time_limit=min(10, _tl() * 0.7),
                               lr=0.01, momentum=0.95,
                               max_steps=20000, patience=1000,
                               threshold_frac=0.05)
        sc = compute_c5(h_sg[0])
        if sc < best_score:
            best_score = sc
            best_h = h_sg[0]

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
