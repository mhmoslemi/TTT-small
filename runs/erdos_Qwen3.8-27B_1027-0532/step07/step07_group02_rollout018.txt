import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_max_check(corr, delta, g, top_k=50):
    """Check max(corr + delta*g) using only top-k candidates"""
    # The new max can only come from positions that are already high
    # or positions where delta*g significantly boosts
    scores = corr + delta * g
    return float(np.max(scores))

# Actually the fast version is to note that if corr is sorted, 
# the new max is at most max(top_k of corr) + |delta| * max(|g|)
# So we only need to check top_k + a few extra positions

def fast_bin_search(corr, g, delta_lo, delta_hi, n_iters=20, top_k=100):
    """Binary search for optimal delta with fast max check"""
    # Pre-compute: which positions are "active" (likely to be max)
    # A position s is active if corr[s] + delta * g[s] could be the max
    # Upper bound on max: max(corr) + |delta| * max(|g|)
    # A position s can be max only if corr[s] + delta_hi * g[s] >= some_threshold
    
    for _ in range(n_iters):
        mid_d = 0.5 * (delta_lo + delta_hi)
        vals = corr + mid_d * g
        f_mid = float(np.max(vals))
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = float(np.max(g[active_mask]))
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            delta_hi = mid_d
        else:
            delta_lo = mid_d
    return 0.5 * (delta_lo + delta_hi)

def smooth_logsumexp(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def _smooth_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        return np.zeros(len(x))
    weights /= wsum
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

_g_state = {}

def _cons_fun(x):
    return np.sum(x) - _g_state['target']

def _cons_jac(x):
    return np.ones(_g_state['n'])

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_state
    _g_state = {'n': n, 'target': target}
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: _smooth_obj(x, alpha, n, target), h_cur,
                           jac=_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_state
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP warmup
    if tleft() > 120:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [50, 200, 1000], 1500)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 2: Main pair CD with spread gradient
    if tleft() > 200:
        h_cd, c5_cd = core_pair_cd(best_h, n0, target0,
                                    time_limit=min(280, tleft() * 0.75),
                                    max_sweeps=100000, n_cand=120,
                                    max_no_improve=400, n_binsearch=22,
                                    seed=42, use_spread_grad=True,
                                    spread_weight=0.3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with higher alpha
    if tleft() > 100:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [3000, 10000], 2000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 4: Pair CD with beam search for efficiency and larger candidates
    if tleft() > 200:
        h_cd, c5_cd = core_pair_cd(best_h, n0, target0,
                                    time_limit=min(250, tleft() * 0.7),
                                    max_sweeps=100000, n_cand=180,
                                    max_no_improve=350, n_binsearch=22,
                                    seed=137, use_spread_grad=True,
                                    spread_weight=0.4, n_beam=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Multi-restart for exploration
    if tleft() > 150:
        h_ms, c5_ms = multi_restart(best_h, n0, target0,
                                     time_limit=min(120, tleft() * 0.5),
                                     seed=seed + 1)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [5000, 20000, 50000], 2000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 7: Try n=500
    if tleft() > 250:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 200:
            h_cd1, c5_cd1 = core_pair_cd(h1, n1, target1,
                                          time_limit=min(180, tleft() * 0.6),
                                          max_sweeps=40000, n_cand=80,
                                          max_no_improve=300, n_binsearch=20,
                                          seed=251, use_spread_grad=True,
                                          spread_weight=0.3)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                n0 = n1
                target0 = target1
        if tleft() > 60:
            h_pol1, sc_pol1 = slsqp_optimize(best_h, n0, target0,
                                               [5000, 20000], 1500)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    # Phase 8: Final pair CD
    if tleft() > 100:
        h_cd_final, c5_final = core_pair_cd(best_h, n0, target0,
                                             time_limit=min(100, tleft() * 0.7),
                                             max_sweeps=60000, n_cand=150,
                                             max_no_improve=300, n_binsearch=24,
                                             seed=777, use_spread_grad=True,
                                             spread_weight=0.35)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 9: Final SLSQP
    if tleft() > 50:
        h_final, sc_final = slsqp_optimize(best_h, n0, target0,
                                            [50000, 200000], 2000)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
