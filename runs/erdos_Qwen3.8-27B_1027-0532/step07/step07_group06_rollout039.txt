                max_sweeps=100000, n_cand_start=20, n_cand_end=60,
                max_no_improve=500, n_top_shifts=3, bisection_iters=14,
                anneal_temp=0.001):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = anneal_temp
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        temp *= 0.999
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            no_improve = 0
            if sweep % (max_no_improve * 3) == 0:
                # Larger perturbation
                rng = np.random.RandomState(sweep)
                h_pert = h + rng.randn(n) * 0.01
                h_pert = project_feasible(h_pert, target)
                if compute_c5(h_pert) < best_c5 + 0.001:
                    h = h_pert
        
        # Compute gradient from top-k lags
        top_k = min(n_top_shifts, n_corr)
        top_idxs = np.argsort(corr)[-top_k:]
        weights = corr[top_idxs].astype(float)
        wsum = weights.sum()
        if wsum > 1e-15:
            # Use softmax for smoother weighting
            exp_w = np.exp(10.0 * (weights - weights.max()))
            weights = exp_w / exp_w.sum()

                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 8)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        # Try all pairs
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Also track second-best for diversity
        second_trial = c5_val
        second_i = -1
        second_j = -1
        second_delta = 0.0
        
        nh = len(high_set)
        nl = len(low_set)
        
        for ii in range(nh):
            i = int(high_set[ii])
            for jj in range(nl):
                j = int(low_set[jj])
                if i == j:
                    continue
                
                # Compute change in corr for this pair

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_phase(h_start, n, target, alphas, maxiter, deadline, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    func = pnorm_obj if use_pnorm else smooth_lse
    for param in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst, noise=0.0, seed=0):
    n_src = len(h_src)
    if n_src == n_dst:
        h = h_src.copy()
        if noise > 0:
            rng = np.random.RandomState(seed)
            h = h + rng.randn(n_dst) * noise
        return project_feasible(h, target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    if noise > 0:
        rng = np.random.RandomState(seed)
        h_out = h_out + rng.randn(n_dst) * noise
    return project_feasible(h_out, target_dst)


def change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def compute_grad_topk(corr, h, n, n_top, softmax_temp=5.0):
    idxs = np.arange(n)
    mid = n - 1
    n_corr = len(corr)
    k = min(n_top, n_corr)
    top_idxs = np.argsort(corr)[-k:]
    weights = corr[top_idxs].astype(float)
    wmin = weights.min()
    exp_w = np.exp(softmax_temp * (weights - wmin))
    wsum = exp_w.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = exp_w / wsum
    grad = np.zeros(n)
    for s_idx, w in zip(top_idxs, weights):
        kk = int(s_idx) - mid
        if kk == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            v1 = (idxs - kk >= 0) & (idxs - kk < n)
            v2 = (idxs + kk >= 0) & (idxs + kk < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - kk])
            grad[v2] -= w * h[idxs[v2] + kk]
    grad -= np.mean(grad)
    return grad, top_idxs


def enhanced_cd(h_init, n, target, deadline,
                max_sweeps=100000, n_cand_start=15, n_cand_end=65,
                max_no_improve=600, n_top_shifts=3, bisection_iters=14,
                anneal_budget=0.002, restart_every=800):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = change_matrix(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if no_improve % restart_every == 0:
                h = best_h.copy()
                nn = int(rng.randint(3, 15))
                nn = min(nn, n // 4)
                perm = rng.permutation(n)
                idxs_p = perm[:nn]
                h_pert = h.copy()
                h_pert[idxs_p] += rng.randn(nn) * anneal_budget
                h_pert = project_feasible(h_pert, target)
                sc_pert = compute_c5(h_pert)
                if sc_pert < best_c5 + anneal_budget:
                    h = h_pert
                    change = change_matrix(h, n)
                else:
                    h = best_h.copy()
                    change = change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
            no_improve = 0

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 8)

        grad, top_idxs = compute_grad_topk(corr, h, n, n_top_shifts)

        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1

        best_delta = 0.0

        nl = len(low_set)

                j = int(low_set[jj])
                if i == j:

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

    return best_h, best_c5

        h_ms, sc_ms = multi_start_refine(best_h, n0, target0, ms_deadline, n_starts=3, perturb=0.015)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 80:
        dl_fin = start_time + min(80, remaining * 0.7)
        h_fin, sc_fin = enhanced_cd(best_h, n0, target0, dl_fin,
                                    max_sweeps=50000, n_cand_start=25, n_cand_end=70,
                                    max_no_improve=300, n_top_shifts=1, bisection_iters=16,
                                    anneal_budget=0.001)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 50:
        dl_s = start_time + min(50, remaining * 0.8)
        h_s, sc_s = slsqp_phase(best_h, n0, target0, [5000, 20000], 800, dl_s, use_pnorm=False)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 40:
        dl_cd = start_time + min(40, remaining * 0.8)
        h_cd, sc_cd = enhanced_cd(best_h, n0, target0, dl_cd,
                                  max_sweeps=20000, n_cand_start=30, n_cand_end=80,
                                  max_no_improve=200, n_top_shifts=2, bisection_iters=14,
                                  anneal_budget=0.0005)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
