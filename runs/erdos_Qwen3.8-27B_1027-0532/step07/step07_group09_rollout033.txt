The binding constraint for pair (i,j) at the current point is:
δ_max_feasible = min(h_i, 1 - h_j)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _smooth_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[k_idx] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= weights[k_idx] * x[idxs[v2] + k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def fast_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=25, n_cand_end=60, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = 1e-10 * max(1.0, c5_val)
        active = np.where(corr >= c5_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        weights = corr[active]
        w_norm = np.sum(weights)
        if w_norm > 0:
            weights = weights / w_norm
        else:
            weights = np.ones(len(active)) / len(active)

        grad = np.zeros(n)
        for wi, si in zip(weights, active):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=np.int32)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)], dtype=np.int32)
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)], dtype=np.int32)

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            if h_i < 1e-12:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]
                if h_j > 1.0 - 1e-12:
                    continue

                delta_max = min(h_i, 1.0 - h_j)
                if delta_max < 1e-12:
                    continue

                g = G[lr] - G[hr]

                a_d = 0.0
                b_d = delta_max
                for _ in range(7):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid >= c5_val - 1e-14:
                        idx_active = np.where(vals >= f_mid - 1e-14)[0]
                        if len(idx_active) > 0:
                            f_prime = np.max(g[idx_active])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-14:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def projected_subgradient(h_init, n, target, time_limit, max_iter=50000, lr_start=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-10, 1e-7 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        progress = min(float(it) / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.95 * progress) ** 1.5

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new
        else:
            h = project_feasible(h - 0.5 * lr * grad, target)
            new_c5 = compute_c5(h)
            if new_c5 < c5_val - 1e-15:
                h = h_new if compute_c5(h_new) < new_c5 else h

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def triple_cd(h_init, n, target, time_limit, max_sweeps=10000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = 1e-10 * max(1.0, c5_val)
        active = np.where(corr >= c5_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for si in active:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        try_count = 0
        for tries in range(30):
            if try_count >= 30:
                break
            i = int(high_set[np.random.randint(n_cand)])
            j = int(low_set[np.random.randint(n_cand)])
            l = int(low_set[np.random.randint(n_cand)])
            if j == l:
                continue
            try_count += 1

            h_i = h[i]
            h_j = h[j]
            h_l = h[l]
            if h_i < 1e-12 or h_j > 1.0 - 1e-12 or h_l > 1.0 - 1e-12:
                continue

            delta_max = min(h_i, 0.5 * (1.0 - h_j), 0.5 * (1.0 - h_l))
            if delta_max < 1e-12:
                continue

            g = (change[j, :] - change[i, :]) + (change[l, :] - change[i, :]) * 0.5
            g = (change[j, :] + change[l, :]) / 2.0 - change[i, :]

            a_d = 0.0
            b_d = delta_max
            for _ in range(6):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                if f_mid >= c5_val - 1e-14:
                    idx_active = np.where(vals >= f_mid - 1e-14)[0]
                    f_prime = np.max(g[idx_active]) if len(idx_active) > 0 else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g)
            if approx_max < best_trial - 1e-14:
                best_trial = approx_max
                best_move = (i, j, l, delta_star)

        if best_move is not None:
            i, j, l, delta_star = best_move
            h[i] -= delta_star
            h[j] += 0.5 * delta_star
            h[l] += 0.5 * delta_star
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, target)
            for p in [i, j, l]:
                update_change_row(change, p, h, n, shifts)

    return best_h, best_c5
