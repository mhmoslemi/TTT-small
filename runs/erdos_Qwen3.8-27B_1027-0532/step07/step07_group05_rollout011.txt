    """Compute d(corr[k])/d(h[p]) for all k."""
    mid = n - 1
    n_corr = 2 * n - 1
    row = np.zeros(n_corr)

    # d(corr[k])/d(h[p]) = (1-h[p+lag])*[0≤p+lag<n] - h[p-lag]*[0≤p-lag<n]

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        lag = int(k) - mid
        w = weights[k]
        v1 = (idxs - lag >= 0) & (idxs - lag < n)
        v2 = (idxs + lag >= 0) & (idxs + lag < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - lag])
        grad[v2] -= w * x[idxs[v2] + lag]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           jac=lambda x, a=alpha: smooth_grad(x, a),
                           method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_row(p, h, n):
    mid = n - 1
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - mid
    row = np.zeros(n_corr)
    mask1 = (p + lags >= 0) & (p + lags < n)
    mask2 = (p - lags >= 0) & (p - lags < n)
    row[mask1] += (1.0 - h[p + lags[mask1]])
    row[mask2] -= h[p - lags[mask2]]
    return row


def ecdd(h_init, n, target, time_limit, max_sweeps=50000,
         n_active_start=2, n_active_end=5,
         n_cand_start=15, n_cand_end=55,
         max_no_improve=350, bisection_iters=10):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    consecutive_no_improve = 0
    sweep = 0

    while sweep < max_sweeps:
        if time.time() - start > time_limit:
            break
        sweep += 1

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = max(n_active_start, int(n_active_start + (n_active_end - n_active_start) * progress))

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - (n - 1)
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, max(10, n // 4))

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        cand_unique = np.unique(np.concatenate([low_set, high_set]))
        change_cache = {}
        for p in cand_unique:
            change_cache[int(p)] = compute_change_row(int(p), h, n)

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 120 + int(150 * progress)

        if n_h * n_l <= max_pairs:
            pairs = [(int(high_set[ii]), int(low_set[jj]))
                     for ii in range(n_h) for jj in range(n_l)
                     if high_set[ii] != low_set[jj]]
        else:
            pairs = []
            n_sys = min(max_pairs, 80)
            for t in range(n_sys):
                frac = t / max(n_sys - 1, 1)
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pairs.append((iv, jv))
            n_rand = max_pairs - len(pairs)
            for _ in range(max(0, n_rand)):
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pairs.append((iv, jv))

        for i, j in pairs:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change_cache[j] - change_cache[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at original resolution
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0, [100, 500, 2000], 400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: ECD at original resolution
    if tleft() > 100:
        h_cd, c5_cd = ecdd(best_h, n0, target0,
                           time_limit=min(60, tleft() * 0.2),
                           max_sweeps=8000,
                           n_active_start=2, n_active_end=4,
                           n_cand_start=12, n_cand_end=35,
                           max_no_improve=200, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Upgrade to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h1, sc1 = slsqp_optimize(h1, n1, target1, [200, 1000], 300)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 4: ECD at n=800
    if tleft() > 250:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl = min(180, tleft() * 0.3)
        h_cd, c5_cd = ecdd(best_h, n_cur, target_cur,
                           time_limit=tl,
                           max_sweeps=20000,
                           n_active_start=2, n_active_end=5,
                           n_cand_start=15, n_cand_end=50,
                           max_no_improve=350, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pol, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                   [2000, 10000, 30000], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Upgrade to n=1200
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h2, sc2 = slsqp_optimize(h2, n2, target2, [500, 2000], 300)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 7: ECD at n=1200 (main optimization)
    if tleft() > 250:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl = min(250, tleft() * 0.4)
        h_cd, c5_cd = ecdd(best_h, n_cur, target_cur,
                           time_limit=tl,
                           max_sweeps=30000,
                           n_active_start=2, n_active_end=5,
                           n_cand_start=20, n_cand_end=60,
                           max_no_improve=400, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Multi-active ECD for refinement
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl = min(120, tleft() * 0.3)
        h_cd, c5_cd = ecdd_multi_active(best_h, n_cur, target_cur,
                                         time_limit=tl,
                                         max_sweeps=20000,
                                         n_active=5, n_cand=45,
                                         max_no_improve=250, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: SLSQP high-alpha polish
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pol, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                   [10000, 50000, 100000], 500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Structured perturbations and restarts
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed + 1)
        for scale in [0.02, 0.05, 0.1, 0.15]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, target_cur)
            h_re, sc_re = ecdd(h_pert, n_cur, target_cur,
                               time_limit=min(50, tleft() * 0.2),
                               max_sweeps=10000,
                               n_active_start=3, n_active_end=6,
                               n_cand_start=20, n_cand_end=55,
                               max_no_improve=200, bisection_iters=10)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    # Phase 11: Final ECD pass
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl = min(60, tleft() * 0.4)
        h_cd, c5_cd = ecdd(best_h, n_cur, target_cur,
                           time_limit=tl,
                           max_sweeps=15000,
                           n_active_start=3, n_active_end=6,
                           n_cand_start=30, n_cand_end=70,
                           max_no_improve=200, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_pol, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                         [50000, 200000], 400)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Final cleanup
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    final_h = project_feasible(best_h, target_cur)
    final_score = compute_c5(final_h)
    return final_h, final_score, n_cur
