def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_shift_gradient(h, k, n):
    """Compute gradient of corr[k] with respect to h."""
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

    """Compute weighted average gradient of top shifts."""
    corr = compute_corr(h)
    c_max = np.max(corr)
    # Find shifts within epsilon of max
    active = np.where(corr >= c_max - max(epsilon * c_max, 1e-8))[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    # Weight by proximity to max
    weights = np.exp(-(corr[active] - c_max + max(epsilon * c_max, 1e-8)) / (epsilon * c_max + 1e-8))
    weights /= weights.sum()
    mid = n - 1
    grad = np.zeros(n)
    for ak, w in zip(active, weights):
        k = int(ak) - mid
        grad += w * compute_shift_gradient(h, k, n)
    return grad, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_shift_gradient(h, k, n):
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def multi_shift_gradient(h, corr, n):
    mid = n - 1
    c_max = float(np.max(corr))
    thresh = c_max - max(5e-7 * c_max, 1e-10)
    active = np.where(corr >= thresh)[0]
    idxs = np.arange(n)
    grad = np.zeros(n)
    if len(active) == 1:
        k = int(active[0]) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    else:
        w = np.ones(len(active)) / len(active)
        for ak, wi in zip(active, w):
            k = int(ak) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
    return grad


def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change, shifts


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def bisect_delta(corr, g_vec, delta_lo, delta_hi, n_iter=14):
    a_d = delta_lo
    b_d = delta_hi
    for _ in range(n_iter):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g_vec
        f_mid = np.max(vals)
        active_m = vals >= f_mid - 1e-14
        if active_m.any():
            f_prime = np.max(g_vec[active_m])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)

def slsqp_polish(h_init, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    def _cons_fun(x):
        return np.sum(x) - _g_target

    def _cons_jac(x):
        return np.ones(_g_n)

    def _smooth_obj_func(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

            g[v2] -= h[idxs[v2] - k]
        else:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            g = np.zeros(n)
            for ak, wi in zip(active_lags, w):
                kk = int(ak) - mid
                v1 = (idxs + kk >= 0) & (idxs + kk < n)
                v2 = (idxs - kk >= 0) & (idxs - kk < n)
                g[v1] += wi * (1.0 - h[idxs[v1] + kk])
                g[v2] -= wi * h[idxs[v2] - kk]

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 200:
        h_cd, sc_cd = vectorized_cd(best_h, n0, target0,
                                     time_limit=min(180, tleft() * 0.85),
                                     max_sweeps=30000,
                                     n_cand_start=25, n_cand_end=90,
                                     max_no_improve=500)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 100:
        h_cp, sc_cp = cutting_plane(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.7),
                                     max_iters=5000, tol=1e-10)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if tleft() > 80:
        h_cd2, sc_cd2 = vectorized_cd(best_h, n0, target0,
                                       time_limit=min(60, tleft() * 0.7),
                                       max_sweeps=20000,
                                       n_cand_start=40, n_cand_end=110,
                                       max_no_improve=400)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 120:
        n1 = 600
        target1 = n1 / 2.0
        h_hi = interpolate_to(best_h, n1, target1)
        sc_hi = compute_c5(h_hi)
        if sc_hi < best_score:

            best_h = h_hi
        if tleft() > 80:

                                               max_sweeps=15000,
                                               n_cand_start=20, n_cand_end=60,
                                               max_no_improve=400)
            if sc_hi_cd < best_score:
                best_score = sc_hi_cd
                best_h = h_hi_cd
        if tleft() > 50:
            h_hi_cp, sc_hi_cp = cutting_plane(best_h, len(best_h), len(best_h) / 2.0,
                                               time_limit=min(40, tleft() * 0.7),
                                               max_iters=3000, tol=1e-10)
            if sc_hi_cp < best_score:
                best_score = sc_hi_cp
                best_h = h_hi_cp

    if tleft() > 100:
        n_final = 400
        target_final = n_final / 2.0
        h_back = interpolate_to(best_h, n_final, target_final)
        sc_back = compute_c5(h_back)
        if tleft() > 60:
            h_back_cd, sc_back_cd = vectorized_cd(h_back, n_final, target_final,
                                                   time_limit=min(50, tleft() * 0.7),
                                                   max_sweeps=20000,
                                                   n_cand_start=30, n_cand_end=100,
                                                   max_no_improve=400)
            if sc_back_cd < best_score:
                best_score = sc_back_cd
                best_h = h_back_cd
            elif sc_back < best_score:
                best_score = sc_back
                best_h = h_back
        else:
            if sc_back < best_score:
                best_score = sc_back
                best_h = h_back

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for perturb_size in [0.03, 0.06, 0.1]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(len(best_h)) * perturb_size
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_ref, sc_ref = vectorized_cd(h_pert, len(best_h), len(best_h) / 2.0,
                                           time_limit=min(30, tleft() * 0.4),
                                           max_sweeps=10000,
                                           n_cand_start=25, n_cand_end=80,
                                           max_no_improve=300)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 60:
        h_sls, sc_sls = slsqp_polish(best_h, len(best_h), len(best_h) / 2.0,
                                      [2000, 10000, 30000], 400, 1e-14)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    if tleft() > 40:
        h_cd3, sc_cd3 = vectorized_cd(best_h, len(best_h), len(best_h) / 2.0,
                                       time_limit=min(30, tleft() * 0.7),
                                       max_sweeps=15000,
                                       n_cand_start=50, n_cand_end=120,
                                       max_no_improve=300)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

        h_cp2, sc_cp2 = cutting_plane(best_h, len(best_h), len(best_h) / 2.0,

            best_score = sc_sls2
            best_h = h_sls2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
