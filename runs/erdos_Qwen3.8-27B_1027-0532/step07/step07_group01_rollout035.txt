def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

    return change

def find_optimal_delta_subgrad(corr, g, delta_lo, delta_hi, n_iter=40):
    """Find delta minimizing max_k(corr[k] + delta*g[k]) using subgradient bisection."""
    if delta_hi - delta_lo < 1e-15:
        return 0.5 * (delta_lo + delta_hi)
    
    lo = delta_lo
    hi = delta_hi
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * g
        max_val = np.max(vals)
        active = vals >= max_val - 1e-14
        g_min = np.min(g[active])
        g_max = np.max(g[active])
        if g_min <= 0.0 and g_max >= 0.0:
            # 0 is in the subgradient, this is optimal
            return mid
        elif g_max < 0.0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

Wait, I realize there might be an issue with the gradient computation in the log-sum-exp part. Let me double-check.

The log-sum-exp smoothed objective is:
LSE(x) = (1/alpha) * log(sum_k exp(alpha * corr_k(x)))

where corr_k(x) = sum_i x[i] * (1 - x[i+k]) (for valid indices).

The gradient is:
dLSE/dx[p] = (1/alpha) * sum_k [alpha * exp(alpha*corr_k) / sum_j exp(alpha*corr_j)] * d(corr_k)/dx[p]
= sum_k [exp(alpha*corr_k) / sum_j exp(alpha*corr_j)] * d(corr_k)/dx[p]

Let p_k = exp(alpha*corr_k) / sum_j exp(alpha*corr_j) = softmax(alpha * corr)_k

Then:
dLSE/dx[p] = sum_k p_k * d(corr_k)/dx[p]

And d(corr_k)/dx[p]:
corr_k = sum_i x[i] * (1-x[i+k]) for valid i, i+k
= sum_i x[i] - sum_i x[i]*x[i+k]

d/dx[p] of sum_i x[i] = 1 (if p is in range, which it always is)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha=500, ftol=1e-16):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
        remaining = budget_tracker['total'] - 10.0 - (time.time() - budget_tracker['start'])
        if remaining < 30:
            break
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < len(h))
    neg_mask = (p - shifts >= 0) & (p - shifts < len(h))
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def find_optimal_delta(corr, g, delta_lo, delta_hi, n_iter=35):
    if delta_hi - delta_lo < 1e-15:
        return 0.5 * (delta_lo + delta_hi)
    lo = delta_lo
    hi = delta_hi
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * g
        max_val = np.max(vals)
        active = vals >= max_val - 1e-14
        g_min = np.min(g[active])
        g_max = np.max(g[active])
        if g_min <= 0.0 and g_max >= 0.0:
            return mid
        elif g_max < 0.0:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def pair_cd_exact(h_init, n, target, time_limit, max_sweeps=50000,
                  n_cand_start=30, n_cand_end=130, max_no_improve=800,
                  restart_every=200, restart_scale=0.01, alpha_ls=100.0):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(123)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 25 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            continue

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        corr_norm = corr - np.max(corr)
        corr_clip = np.clip(alpha_ls * corr_norm, -500.0, 0.0)
        weights_ls = np.exp(corr_clip)
        w_sum = np.sum(weights_ls)
        probs_ls = weights_ls / w_sum

        grad_ls = np.zeros(n)
        significant = probs_ls > 1e-8
        for k_idx in np.where(significant)[0]:
            k = int(k_idx) - mid
            p_k = probs_ls[k_idx]
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            if np.any(v1):
                grad_ls[idxs[v1]] += p_k * (1.0 - h[idxs[v1] + k])
            if np.any(v2):
                grad_ls[idxs[v2]] -= p_k * h[idxs[v2] - k]
        grad_ls -= np.mean(grad_ls)

        combined_grad = 0.5 * grad + 0.5 * grad_ls

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(combined_grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        current_max_raw = np.max(corr)

        for ii in range(n_h):
            i = int(high_set[ii])
            ch_i = change[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue
                g = change[j] - ch_i
                val_lo = np.max(corr + delta_lo * g)
                val_hi = np.max(corr + delta_hi * g)
                if val_lo >= current_max_raw and val_hi >= current_max_raw:
                    continue
                delta_star = find_optimal_delta(corr, g, delta_lo, delta_hi)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = float(np.max(corr + delta_star * g))
                if approx_max * dx < best_trial_c5:
                    best_trial_c5 = approx_max * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, shifts)
            update_change_row(change, best_j, h, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

        alphas_p3 = [50000, 200000, 800000]
        h3, sc3 = slsqp_round(best_h, n0, target0, alphas_p3, maxiter_per_alpha=800)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Multi-pair CD on initial grid
    if tleft() > 150:
        h4, sc4 = multi_pair_cd(best_h, n0, target0,
                                time_limit=min(140, tleft() * 0.6),
                                max_sweeps=20000, n_pairs=3, n_cand=80)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Interpolate to finer grid and optimize
    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h_fine = interpolate_to(best_h, n1, target1)
        sc_fine = compute_c5(h_fine)

        if tleft() > 250:
            h5, sc5 = pair_cd_exact(h_fine, n1, target1,
                                    time_limit=min(250, tleft() * 0.55),
                                    max_sweeps=35000,
                                    n_cand_start=40, n_cand_end=140,
                                    max_no_improve=900, restart_scale=0.008)
            if sc5 < best_score:
                best_score = sc5
                best_h = h5
                best_n = n1

        if best_n == n1 and tleft() > 100:
            h5b, sc5b = multi_pair_cd(best_h, n1, target1,
                                      time_limit=min(90, tleft() * 0.5),
                                      max_sweeps=15000, n_pairs=2, n_cand=100)
            if sc5b < best_score:
                best_score = sc5b
                best_h = h5b

    # Phase 6: Even finer grid if time and improvement warrant
    if best_n > n0 and tleft() > 300:
        n2 = 800
        target2 = n2 / 2.0
        h_finest = interpolate_to(best_h, n2, target2)
        sc_finest = compute_c5(h_finest)

        if tleft() > 200:
            h6, sc6 = pair_cd_exact(h_finest, n2, target2,
                                    time_limit=min(200, tleft() * 0.55),
                                    max_sweeps=25000,
                                    n_cand_start=50, n_cand_end=160,
                                    max_no_improve=800, restart_scale=0.006)
            if sc6 < best_score:
                best_score = sc6
                best_h = h6
                best_n = n2

    # Phase 7: Diversity - try perturbed restarts
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n / 2.0)
            h_ref, sc_ref = slsqp_round(h_pert, best_n, best_n / 2.0,
                                        [200, 2000, 20000, 100000],
                                        maxiter_per_alpha=400)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 80:
                h_cd_p, sc_p = pair_cd_exact(h_ref, best_n, best_n / 2.0,
                                             time_limit=min(60, tleft() * 0.35),
                                             max_sweeps=12000,
                                             n_cand_start=35, n_cand_end=120,
                                             max_no_improve=400)
                if sc_p < best_score:
                    best_score = sc_p
                    best_h = h_cd_p

    # Phase 8: Final CD polish
    if tleft() > 60:
        h8, sc8 = pair_cd_exact(best_h, best_n, best_n / 2.0,
                                time_limit=min(50, tleft() * 0.6),
                                max_sweeps=15000,
                                n_cand_start=50, n_cand_end=150,
                                max_no_improve=400, restart_scale=0.005)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP on original grid size for precision
    if tleft() > 40:
        nf = n0
        target_f = nf / 2.0
        if best_n > nf:
            h_polish = interpolate_to(best_h, nf, target_f)
        else:
            h_polish = best_h.copy()
        alphas_final = [100000, 500000, 2000000]
        h9, sc9 = slsqp_round(h_polish, nf, target_f, alphas_final, maxiter_per_alpha=600)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9
            best_n = nf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
