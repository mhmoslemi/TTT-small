import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _smooth_max_obj(x, alpha, target_sum, coeff_cache):
    n = len(x)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:

def _cons_fun_eq(x, target):
    return np.sum(x) - target

def slsqp_polish(h_start, n, target, alphas, time_limit, start_time, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_max_obj, h_cur, args=(param, target, None),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - 20.0 - elapsed()

    # Phase 1: Anti-periodic optimization (reduces dimension, may find better structure)
    if n0 % 2 == 0 and remaining() > 200:
        alloc = min(200, remaining() * 0.25)
        h_ap, score_ap = anti_periodic_cd(best_h, n0,
                                          start_time + alloc, start_time,
                                          max_sweeps=100000,
                                          n_cand_range=(8, 25),
                                          bisection_iters=45,
                                          max_no_improve=400)
        h_full_ap = np.concatenate([h_ap, 1.0 - h_ap])
        h_full_ap = project_feasible(h_full_ap, target0)
        score_ap = compute_c5(h_full_ap)
        if score_ap < best_score:
            best_score = score_ap
            best_h = h_full_ap

    # Phase 2: Aggressive pair CD at original resolution
    if remaining() > 250:
        alloc = min(350, remaining() * 0.35)
        h2, score2 = aggressive_pair_cd(best_h, n0, target0,
                                        start_time + alloc, start_time,
                                        max_sweeps=300000,
                                        n_cand_range=(10, 45),
                                        n_top_shifts=3,
                                        bisection_iters=55,
                                        max_no_improve=600,
                                        pert_interval=400,
                                        pert_scale_init=0.04,
                                        pert_scale_min=0.003)
        if score2 < best_score:
            best_score = score2
            best_h = h2

    # Phase 3: SLSQP polish
    if remaining() > 80:
        alloc = min(80, remaining() * 0.2)
        h3, score3 = slsqp_polish(best_h, n0, target0,
                                  [100, 500, 2000],
                                  start_time + alloc, start_time, maxiter=1500)
        if score3 < best_score:
            best_score = score3
            best_h = h3

    # Phase 4: Higher resolution interpolation and pair CD
    if remaining() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        alloc = min(300, remaining() * 0.35)
        h4, score4 = aggressive_pair_cd(h_int, n1, target1,
                                        start_time + alloc, start_time,
                                        max_sweeps=200000,
                                        n_cand_range=(8, 35),
                                        n_top_shifts=3,
                                        bisection_iters=50,
                                        max_no_improve=500,
                                        pert_interval=300,
                                        pert_scale_init=0.03,
                                        pert_scale_min=0.003)
        if score4 < best_score:
            best_score = score4
            best_h = h4
            n0 = n1
            target0 = target1

    # Phase 5: Multi-start exploration
    if remaining() > 200:
        alloc = min(200, remaining() * 0.3)
        h5, score5 = multi_start_explore(best_h, n0, target0,
                                         start_time + alloc, start_time, n_starts=4)
        if score5 < best_score:
            best_score = score5
            best_h = h5

    # Phase 6: Higher resolution again
    if remaining() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h_int2 = interpolate_to(best_h, n2, target2)
        alloc = min(280, remaining() * 0.35)
        h6, score6 = aggressive_pair_cd(h_int2, n2, target2,
                                        start_time + alloc, start_time,
                                        max_sweeps=150000,
                                        n_cand_range=(8, 30),
                                        n_top_shifts=2,
                                        bisection_iters=45,
                                        max_no_improve=400,
                                        pert_interval=300,
                                        pert_scale_init=0.025,
                                        pert_scale_min=0.002)
        if score6 < best_score:
            best_score = score6
            best_h = h6
            n0 = n2
            target0 = target2

    # Phase 7: Final SLSQP polish
    if remaining() > 60:
        alloc = min(60, remaining() * 0.4)
        h7, score7 = slsqp_polish(best_h, n0, target0,
                                  [500, 2000, 10000],
                                  start_time + alloc, start_time, maxiter=1000)
        if score7 < best_score:
            best_score = score7
            best_h = h7

    # Phase 8: Final aggressive pair CD
    if remaining() > 80:
        alloc = min(80, remaining() * 0.5)
        h8, score8 = aggressive_pair_cd(best_h, n0, target0,
                                        start_time + alloc, start_time,
                                        max_sweeps=100000,
                                        n_cand_range=(15, 50),
                                        n_top_shifts=1,
                                        bisection_iters=60,
                                        max_no_improve=300,
                                        pert_interval=500,
                                        pert_scale_init=0.02,
                                        pert_scale_min=0.001)
        if score8 < best_score:
            best_score = score8
            best_h = h8

    # Final validation
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    if final_score > best_score:
        final_h = project_feasible(best_h, len(best_h) / 2.0)
        final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
