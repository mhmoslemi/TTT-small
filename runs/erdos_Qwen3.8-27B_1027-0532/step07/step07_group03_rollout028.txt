import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def block_pattern(n, n_blocks, block_values):
    """Create a pattern from block values."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i, v in enumerate(block_values):
        start = i * block_size
        end = min(start + block_size, n)
        h[start:end] = v
    return h

def make_block_init(n, n_blocks, seed):
    """Create random block initialization."""
    rng = np.random.RandomState(seed)
    # Random values in [0,1]
    values = rng.random(n_blocks)
    h = block_pattern(n, n_blocks, values)
    return project_feasible(h, n / 2.0)

def smooth_max(corr, alpha):
    """Smooth maximum of correlation values."""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(smooth_max(corr, alpha))

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:

    return h_cur, cur_score

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _power_mean_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    vals = np.maximum(corr, 0.0)
    return float((np.sum(vals ** p) / n_corr) ** (1.0 / p))


def _power_obj_wrapper(x, p):
    return _power_mean_obj(x, p)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def power_mean_slsqp(h_start, n, target, p_values, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for p in p_values:
        try:
            res = minimize(_power_obj_wrapper, h_cur, args=(p,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad


def cutting_plane_multi(h_init, n, target, time_limit, max_iters=5000,
                        tol=1e-9, max_cuts=80, top_k=5):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_b = []
    cuts_h = []

    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            if len(cuts_g) > 20:
                cuts_g = cuts_g[-10:]
                cuts_b = cuts_b[-10:]
                cuts_h = cuts_h[-10:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        threshold = corr[k_star] * 0.95
        active_mask = corr >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([k_star])
        elif len(active_lags) > top_k:
            top_indices = np.argsort(corr[active_lags])[::-1][:top_k]
            active_lags = active_lags[top_indices]

        n_active = len(active_lags)
        w = np.ones(n_active) / n_active
        g = compute_corr_gradient_multi(h, active_lags, w)

        cuts_g.append(g)
        cuts_b.append(float(g @ h) - f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            drops = len(cuts_g) - keep
            cuts_g = [cuts_g[0]] + cuts_g[1 + drops:]
            cuts_b = [cuts_b[0]] + cuts_b[1 + drops:]
            cuts_h = [cuts_h[0]] + cuts_h[1 + drops:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_b[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            continue

    return best_h, best_score


def fast_interceptive_cd(h_init, n, target, time_limit, max_sweeps=50000,
                         n_cand_start=10, n_cand_end=40, max_no_improve=300,
                         use_linear=False):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            hi = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj = h[j]
                delta_lo = max(-hi, -hj)
                delta_hi = min(hi, 1.0 - hj)
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]

                if use_linear:
                    f_prime = np.max(g_vec)
                    if f_prime > 1e-12:
                        delta_star = 0.0
                    else:
                        delta_star = delta_hi
                    approx_max = np.max(corr + delta_star * g_vec) * dx
                else:
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(10):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g_vec
                        f_mid = np.max(vals)
                        active_m = vals >= f_mid - 1e-14
                        if active_m.any():
                            f_prime = np.max(g_vec[active_m])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g_vec) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def trilinear_cd(h_init, n, target, time_limit, max_sweeps=20000,
                 n_cand=15, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    def recompute_change(p):
        change[p, :] = 0.0
        pos_m = (p + shifts >= 0) & (p + shifts < n)
        neg_m = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
        change[p, neg_m] -= h[p - shifts[neg_m]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_move = None

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(high_set)):
                j = int(high_set[jj])
                if i == j:
                    continue
                for kk in range(len(low_set)):
                    l = int(low_set[kk])
                    if l == i or l == j:
                        continue

                    delta_lo = max(-h[i], -h[j], 0.0)
                    delta_hi = min(h[i], h[j], 1.0 - h[l])
                    if delta_hi - delta_lo < 1e-12:
                        continue

                    g_vec = 2.0 * change[l] - change[i] - change[j]

                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(8):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g_vec
                        f_mid = np.max(vals)
                        active_m = vals >= f_mid - 1e-14
                        f_prime = np.max(g_vec[active_m]) if active_m.any() else 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g_vec) * dx

                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_move = (i, j, l, delta_star)

        if best_move is not None:
            i, j, l, delta_star = best_move
            h[i] -= delta_star
            h[j] -= delta_star
            h[l] += 2.0 * delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            recompute_change(i)
            recompute_change(j)
            recompute_change(l)

    return best_h, best_c5


def generate_structured_init(n, method, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0

    if method == 'blocks':
        n_blocks = rng.choice([8, 12, 16, 20, 25, 30])
        vals = rng.random(n_blocks)
        h = np.zeros(n)
        bs = n // n_blocks
        for i in range(n_blocks):
            s = i * bs
            e = min(s + bs, n)
            h[s:e] = vals[i]
        return project_feasible(h, target)
    elif method == 'sine':
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        return project_feasible(h, target)
    elif method == 'alt_blocks':
        n_blocks = rng.choice([4, 6, 8, 10])
        h = np.zeros(n)
        bs = n // n_blocks
        for i in range(n_blocks):
            s = i * bs
            e = min(s + bs, n)
            if i % 2 == 0:
                h[s:e] = rng.uniform(0.4, 0.9)
            else:
                h[s:e] = rng.uniform(0.1, 0.5)
        return project_feasible(h, target)
    elif method == 'bimodal':
        h = 0.5 + np.zeros(n)
        bs = n // 2
        h[:bs] = 0.5 + rng.uniform(0.1, 0.3)
        h[bs:] = 0.5 - rng.uniform(0.1, 0.3)
        return project_feasible(h, target)
    else:
        h = rng.random(n)
        return project_feasible(h, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    n0 = 400
    target0 = n0 / 2.0

    h_init = np.array(initial_h_values, dtype=float)
    h_init = project_feasible(h_init, target0)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    candidate_pool = [(h_init.copy(), best_score)]

    rng = np.random.RandomState(seed)
    init_methods = ['blocks', 'sine', 'alt_blocks', 'bimodal', 'random']
    for m in init_methods:
        h_try = generate_structured_init(n0, m, seed + hash(m) % 1000)
        sc_try = compute_c5(h_try)
        candidate_pool.append((h_try, sc_try))

    candidate_pool.sort(key=lambda x: x[1])

    for h_cand, sc_cand in candidate_pool[:4]:
        if tleft() < 60:
            break
        if sc_cand < best_score:
            best_score = sc_cand
            best_h = h_cand.copy()

        t_alloc = min(40, tleft() * 0.25)
        h_cd, sc_cd = fast_interceptive_cd(h_cand, n0, target0,
                                           time_limit=t_alloc,
                                           max_sweeps=8000,
                                           n_cand_start=8, n_cand_end=25,
                                           max_no_improve=150)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    if tleft() > 100:
        h_cp, sc_cp = cutting_plane_multi(best_h, n0, target0,
                                          time_limit=min(120, tleft() * 0.4),
                                          max_iters=4000, tol=1e-10,
                                          max_cuts=60, top_k=5)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp.copy()

    if tleft() > 80:
        h_cd, sc_cd = fast_interceptive_cd(best_h, n0, target0,
                                           time_limit=min(100, tleft() * 0.5),
                                           max_sweeps=20000,
                                           n_cand_start=10, n_cand_end=35,
                                           max_no_improve=250)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    if tleft() > 80:
        h_sls, sc_sls = slsqp_optimize(best_h, n0, target0,
                                       [200, 1000, 5000], 400, 1e-14)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls.copy()

    if tleft() > 60:
        h_pm, sc_pm = power_mean_slsqp(best_h, n0, target0,
                                       [50, 100, 200], 400, 1e-14)
        if sc_pm < best_score:
            best_score = sc_pm
            best_h = h_pm.copy()

    if tleft() > 60:
        h_tl, sc_tl = trilinear_cd(best_h, n0, target0,
                                   time_limit=min(60, tleft() * 0.4),
                                   max_sweeps=10000, n_cand=8,
                                   max_no_improve=150)
        if sc_tl < best_score:
            best_score = sc_tl
            best_h = h_tl.copy()

    if tleft() > 80:
        h_cp2, sc_cp2 = cutting_plane_multi(best_h, n0, target0,
                                            time_limit=min(90, tleft() * 0.4),
                                            max_iters=5000, tol=1e-10,
                                            max_cuts=80, top_k=8)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2.copy()

    if tleft() > 60:
        h_cd2, sc_cd2 = fast_interceptive_cd(best_h, n0, target0,
                                             time_limit=min(70, tleft() * 0.5),
                                             max_sweeps=15000,
                                             n_cand_start=15, n_cand_end=50,
                                             max_no_improve=200)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2.copy()

    if tleft() > 50:
        h_sls2, sc_sls2 = slsqp_optimize(best_h, n0, target0,
                                         [2000, 8000, 30000], 500, 1e-14)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2.copy()

    if tleft() > 50:
        h_cd3, sc_cd3 = fast_interceptive_cd(best_h, n0, target0,
                                             time_limit=min(50, tleft() * 0.6),
                                             max_sweeps=15000,
                                             n_cand_start=20, n_cand_end=60,
                                             max_no_improve=150)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3.copy()

    if tleft() > 40:
        perturbed_found = False
        for ns in [0.03, 0.08, 0.15]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert.copy()
                perturbed_found = True
                break
            h_ref, sc_ref = fast_interceptive_cd(h_pert, n0, target0,
                                                 time_limit=min(25, tleft() * 0.3),
                                                 max_sweeps=5000,
                                                 n_cand_start=10, n_cand_end=30,
                                                 max_no_improve=100)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref.copy()

    if tleft() > 30:
        h_final_cd, sc_final = fast_interceptive_cd(best_h, n0, target0,
                                                    time_limit=min(25, tleft() * 0.6),
                                                    max_sweeps=10000,
                                                    n_cand_start=25, n_cand_end=70,
                                                    max_no_improve=100)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cd.copy()

    if tleft() > 20:
        h_final_sls, sc_final2 = slsqp_optimize(best_h, n0, target0,
                                                [5000, 20000], 300, 1e-15)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_h = h_final_sls.copy()

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
