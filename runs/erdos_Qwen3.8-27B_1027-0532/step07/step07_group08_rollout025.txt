def benders_master(cuts, n, target, solver='GLPK'):
    import cvxpy as cp
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [0 <= h, h <= 1, cp.sum(h) == target]
    for a, b in cuts:
        constraints.append(a @ h + b <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
        if h.value is not None:
            return h.value.copy(), t.value
    except:
        pass
    return None, None

7. Add cuts to the set (maybe limit the number)
8. Repeat

This is a standard Benders decomposition. The key is that the master LP is solved exactly (by an LP solver), giving us a valid lower bound at each step.

    best_h = h.copy()
    best_score = compute_c5(h)

    cuts = []  # list of (gradient, constant)

        if time.time() - start > time_limit:
            break

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target]
        for a, b in cuts:
            constraints.append(a @ h_var + b <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except:
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except:
                continue
        
        if h_var.value is None:
            # Try with fewer cuts
            if len(cuts) > 10:
                cuts = cuts[-10:]
                continue
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Evaluate true objective
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        score = float(np.max(corr)) * dx
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_new.copy()
        
        # Check convergence: LP objective vs true objective
        lp_gap = score - float(t_var.value)
        if lp_gap < 1e-5:
            break

        c_max = float(np.max(corr))
        threshold = c_max - 0.02 * max(c_max, 0.01)

        # The objective is max(corr) * dx, so active shifts are those with corr close to max(corr)
        active = np.where(corr >= c_max - 0.02 * max(c_max, 1.0))[0]
        
        # Use top n_active shifts
        sorted_active = active[np.argsort(corr[active])[::-1]][:n_active]

        for s in sorted_active:

            # (simplify: keep last 200)
            cuts = cuts[-200:]

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    top_k = np.argsort(corr)[-n_pert * 3:]

    idxs = np.arange(n)

    for s in top_k:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        c_max = float(np.max(corr))
        active_frac = 0.1
        threshold = c_max - active_frac * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]

        alpha = alpha_base / (t_new + 10.0)
        h_new = y - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_new.copy()

        h = h_new
        t_k = t_new

    return best_h, best_c5


def multi_start_cd(h_base, n, target, time_limit, n_starts=6):
    start = time.time()
    rng = np.random.RandomState(12345)
    best_h = h_base.copy()
    best_score = compute_c5(h_base)
    time_per_start = time_limit / n_starts

    for s in range(n_starts):
        if time.time() - start > time_limit - 10:
            break
        noise_scale = 0.02 + 0.02 * s
        h_pert = h_base + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)

        remaining = time_limit - (time.time() - start)
        if remaining < 10:
            break
        h_cd, c5_cd = pair_cd(h_pert, n, target,
                              time_limit=remaining * 0.6,
                              max_sweeps=5000,
                              n_cand_start=10, n_cand_end=50,
                              max_no_improve=200,
                              n_active_start=2, n_active_end=4,
                              reheat_interval=80, reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Quick SLSQP at n=400
    if tleft() > 80:
        deadline = time.time() + min(45, tleft() * 0.4)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000, 50000],
                                500, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Accelerated gradient at n=400
    if tleft() > 200:
        h_ag, sc_ag = accelerated_grad_descent(best_h, current_n, current_target,
                                               time_limit=min(60, tleft() * 0.25))
        if sc_ag < best_score:
            best_score = sc_ag
            best_h = h_ag

    # Phase 3: Interpolate to n=800
    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        # Quick SLSQP at n=800
        if tleft() > 350:
            deadline = time.time() + min(40, tleft() * 0.2)
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [200, 1000, 5000],
                                      400, 1e-14, deadline=deadline)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
                current_n = n1
                current_target = target1
            else:
                current_n = n1
                current_target = target1
                best_h = h1 if sc1 < best_score else best_h

        # Main CD at n=800
        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(350, tleft() * 0.55),
                                    max_sweeps=40000,
                                    n_cand_start=12, n_cand_end=70,
                                    max_no_improve=500,
                                    n_active_start=2, n_active_end=5,
                                    reheat_interval=100, reheat_scale=0.035,
                                    use_fourier=True)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

    # Phase 4: SLSQP polish at n=800
    if tleft() > 200:
        deadline = time.time() + min(80, tleft() * 0.35)
        h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                  [2000, 10000, 50000, 200000],
                                  1000, 1e-15, deadline=deadline)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 5: More CD with different parameters
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, current_n, current_target,
                                time_limit=min(120, tleft() * 0.55),
                                max_sweeps=25000,
                                n_cand_start=20, n_cand_end=90,
                                max_no_improve=400,
                                n_active_start=3, n_active_end=7,
                                reheat_interval=80, reheat_scale=0.025,
                                use_fourier=True)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Multi-start at current resolution
    if tleft() > 150:
        h_ms, sc_ms = multi_start_cd(best_h, current_n, current_target,
                                     time_limit=min(100, tleft() * 0.5),
                                     n_starts=4)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    # Phase 7: Try n=1200 if time allows
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2_init = compute_c5(h2)

        if tleft() > 250:
            deadline = time.time() + min(50, tleft() * 0.2)
            h_pol3, sc3 = slsqp_round(h2, n2, target2,
                                      [500, 2000, 8000],
                                      500, 1e-14, deadline=deadline)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3
                current_n = n2
                current_target = target2

        if tleft() > 150:
            h_cd3, c5_cd3 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(150, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=15, n_cand_end=60,
                                    max_no_improve=350,
                                    n_active_start=2, n_active_end=5,
                                    reheat_interval=90, reheat_scale=0.03,
                                    use_fourier=True)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.4)
            h_pol4, sc4 = slsqp_round(best_h, current_n, current_target,
                                      [5000, 20000, 80000],
                                      800, 1e-15, deadline=deadline)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_pol4

    # Phase 8: Subgradient at current best
    if tleft() > 60:
        h_sg, c5_sg = subgradient_cd(best_h, current_n, current_target,
                                     time_limit=min(40, tleft() * 0.5),
                                     alpha0=0.15, alpha_min=0.003)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 9: Final CD
    if tleft() > 60:
        h_cd_f, c5_f = pair_cd(best_h, current_n, current_target,
                               time_limit=min(50, tleft() * 0.55),
                               max_sweeps=15000,
                               n_cand_start=30, n_cand_end=100,
                               max_no_improve=250,
                               n_active_start=4, n_active_end=8,
                               reheat_interval=60, reheat_scale=0.02,
                               use_fourier=True)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 10: Final SLSQP
    if tleft() > 30:
        deadline = time.time() + min(25, tleft() * 0.6)
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [10000, 50000, 200000],
                                            1000, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    # Return at n=400 if that's better (the harness evaluates at whatever n we return)
    # Actually, let's also try returning at n=400 by down-sampling
    # But higher n should give better bounds, so return the best
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
