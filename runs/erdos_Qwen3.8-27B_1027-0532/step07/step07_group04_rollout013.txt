import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0
_g_mu = 0.0

def _bimodal_obj(x, alpha, alpha_mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smooth_max = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    bimodal_penalty = alpha_mu * np.sum(x * (1.0 - x))
    return smooth_max + bimodal_penalty

def _bimodal_obj_grad(x, alpha, alpha_mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-15:
            continue
        k = k_idx - mid
        idxs = np.arange(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        w = weights[k_idx]
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    
    grad += 2.0 * alpha_mu * (0.5 - x)
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_bimodal(h_start, n, target, alphas, mu_alphas, maxiter_per_alpha,
                  time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for i in range(len(alphas)):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        alpha = alphas[i]
        alpha_mu = mu_alphas[i] if i < len(mu_alphas) else 0.0
        try:
            res = minimize(_bimodal_obj, h_cur, args=(alpha, alpha_mu),
                           jac=_bimodal_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def focused_cd(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=40, n_cand_end=120, max_no_improve=400,
               active_tol=0.5):

    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 5e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = np.max(corr)
        active = np.where(corr >= c_max - active_tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(0.0, h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < best_trial_c5 / dx - 1e-15:
                        pass
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-16:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

        for ii in range(min(n_h, 20)):
            i = int(high_set[ii])
            for jj in range(min(n_l, 20)):
                j = int(low_set[jj])
                if i == j:
                    continue
                for kk in range(min(n_l, 10)):
                    m = int(low_set[kk])
                    if m == i or m == j:
                        continue
                    pair_lo = max(0.0, h[i] - 1.0, -h[j], -h[m])
                    pair_hi = min(h[i], 1.0 - h[j], 1.0 - h[m])
                    if pair_hi - pair_lo < 1e-14:
                        continue
                    g = change[j] + change[m] - 2.0 * change[i]
                    a_d = pair_lo
                    b_d = pair_hi
                    for _ in range(10):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-14:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial_c5 - 1e-16:
                        best_trial_c5 = approx_max
                        best_move = (i, j, m, delta_star)
                        if best_trial_c5 < c5_val * (1.0 - 0.001):
                            break
                if best_trial_c5 < c5_val * (1.0 - 0.002):
                    break

        if best_move is not None:
            i, j, m, delta_star = best_move
            h[i] -= delta_star
            h[j] += 0.5 * delta_star
            h[m] += 0.5 * delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            update_change_row(change, m, h, n, shifts)

    return best_h, best_c5

def sa_focused_cd(h_init, n, target, time_limit, max_steps=30000,
                  t_start=0.03, t_end=0.0001, n_cand=80, eta=5.0):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break

        progress = min(step / max(max_steps, 1), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        c_max = np.max(corr)
        active = np.where(corr >= c_max - 0.5)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(0.0, h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-14:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-14:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-16:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-15:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_local(h_best, n, target, rng, n_restarts, time_limit, start_time):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        scale = 0.02 + 0.03 * (r % 5)
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_cd, c5_cd = focused_cd(h_pert, n, target,
                                  time_limit=min(30, (time_limit - (time.time() - start_time)) * 0.3),
                                  max_sweeps=5000, n_cand_start=40, n_cand_end=70,
                                  max_no_improve=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Bimodal SLSQP at initial resolution
    if tleft() > 120:
        h_pol, sc = slsqp_bimodal(best_h, n0, target0,
                                  alphas=[200, 800, 3000],
                                  mu_alphas=[0.01, 0.05, 0.1],
                                  maxiter_per_alpha=500,
                                  time_limit=min(90, tleft() * 0.6),
                                  start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Focused CD at initial resolution
    if tleft() > 150:
        h_cd, c5_cd = focused_cd(best_h, n0, target0,
                                  time_limit=min(120, tleft() * 0.6),
                                  max_sweeps=40000,
                                  n_cand_start=40, n_cand_end=100,
                                  max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Triple moves
    if tleft() > 100:
        h_tr, c5_tr = triple_cd(best_h, n0, target0,
                                 time_limit=min(70, tleft() * 0.5),
                                 max_sweeps=10000, n_cand=50)
        if c5_tr < best_score:
            best_score = c5_tr
            best_h = h_tr

    # Phase 4: SA to escape local optima
    if tleft() > 100:
        h_sa, c5_sa = sa_focused_cd(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.5),
                                     max_steps=25000,
                                     t_start=0.025, t_end=0.0002,
                                     n_cand=80)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 5: Go to higher resolution n=600
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd1, c5_cd1 = focused_cd(h1, n1, target1,
                                    time_limit=min(180, tleft() * 0.6),
                                    max_sweeps=40000,
                                    n_cand_start=50, n_cand_end=110,
                                    max_no_improve=350)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1

        if tleft() > 80:
            h_pol1, sc1 = slsqp_bimodal(best_h, n0, target0,
                                         alphas=[3000, 10000],
                                         mu_alphas=[0.05, 0.15],
                                         maxiter_per_alpha=800,
                                         time_limit=min(60, tleft() * 0.5),
                                         start_time=start_time)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    # Phase 6: More CD at current resolution
    if tleft() > 100:
        h_cd2, c5_cd2 = focused_cd(best_h, n0, target0,
                                    time_limit=min(80, tleft() * 0.6),
                                    max_sweeps=30000,
                                    n_cand_start=60, n_cand_end=120,
                                    max_no_improve=250)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: Multi-restart
    if tleft() > 120:
        h_mr, c5_mr = multi_restart_local(best_h, n0, target0, rng,
                                           n_restarts=6,
                                           time_limit=min(100, tleft() * 0.6),
                                           start_time=start_time)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 8: Go to n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd3, c5_cd3 = focused_cd(h2, n2, target2,
                                    time_limit=min(180, tleft() * 0.6),
                                    max_sweeps=35000,
                                    n_cand_start=50, n_cand_end=100,
                                    max_no_improve=300)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            n0 = n2
            target0 = target2

        if tleft() > 80:
            h_pol2, sc2 = slsqp_bimodal(best_h, n0, target0,
                                         alphas=[5000, 20000],
                                         mu_alphas=[0.1, 0.3],
                                         maxiter_per_alpha=600,
                                         time_limit=min(60, tleft() * 0.5),
                                         start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 9: SA at higher resolution
    if tleft() > 80:
        h_sa2, c5_sa2 = sa_focused_cd(best_h, n0, target0,
                                       time_limit=min(60, tleft() * 0.5),
                                       max_steps=20000,
                                       t_start=0.02, t_end=0.0001,
                                       n_cand=70)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2

    # Phase 10: Final CD polish
    if tleft() > 60:
        h_cd4, c5_cd4 = focused_cd(best_h, n0, target0,
                                    time_limit=min(45, tleft() * 0.6),
                                    max_sweeps=20000,
                                    n_cand_start=80, n_cand_end=130,
                                    max_no_improve=150)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Phase 11: Final SLSQP
    if tleft() > 40:
        h_fp, sc_fp = slsqp_bimodal(best_h, n0, target0,
                                     alphas=[10000, 50000],
                                     mu_alphas=[0.2, 0.5],
                                     maxiter_per_alpha=400,
                                     time_limit=min(30, tleft() * 0.7),
                                     start_time=start_time)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    # Final rapid CD
    if tleft() > 20:
        h_cf, c5_cf = focused_cd(best_h, n0, target0,
                                  time_limit=min(15, tleft() * 0.7),
                                  max_sweeps=8000,
                                  n_cand_start=80, n_cand_end=120,
                                  max_no_improve=100)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
