import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-10
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, n_cand=35, max_sweeps=200000, n_bisect=12):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                g_max = float(np.max(np.abs(g)))
                if g_max < 1e-14:
                    continue

                lo = 0.0
                hi_d = delta_max
                for _ in range(n_bisect):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def momentum_subgrad(h_init, n, target, time_limit, max_steps=50000, n_shifts=5, base_lr=0.03, momentum=0.9):
    start = time.time()

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    velocity = np.zeros(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 15.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (1.0 / n) * np.sum(grad)

        decay = 1.0 / (1.0 + 0.002 * step)
        lr = base_lr * decay
        velocity = momentum * velocity - lr * grad
        h_try = h + velocity
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_now:
            h = h_try
        else:
            velocity *= 0.5
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now:
                h = h_try
                velocity = -lr * grad

    return best_h, best_c5

                            maxiter=300, time_limit=min(60, tleft() * 0.3),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Pair CD multi-shift
    if tleft() > 180:
        hcd, c5cd = pair_cd_multi(best_h, n, target,
                                   time_limit=min(200, tleft() * 0.22),
                                   n_cand=40, max_sweeps=200000, n_shifts=3, n_bisect=12)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Momentum subgradient
    if tleft() > 80:
        hsg, c5sg = momentum_subgrad(best_h, n, target,
                                      time_limit=min(60, tleft() * 0.25),
                                      max_steps=30000, n_shifts=5, base_lr=0.04, momentum=0.85)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 6: Pair CD (single shift, more candidates)
    if tleft() > 150:
        hcd, c5cd = pair_cd_fast(best_h, n, target,
                                  time_limit=min(180, tleft() * 0.25),
                                  n_cand=50, max_sweeps=200000, n_bisect=12)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 7: Perturb and re-optimize (restart 1)
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * 0.015
        h_pert = project_feasible(h_pert, target_use)
        hp, sc = lbfgs_phase(h_pert, n_use, target_use,
                            alphas=[2000, 10000, 50000],
                            maxiter=200, time_limit=min(40, tleft() * 0.15),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            if tleft() > 100:
                hcd, c5cd = pair_cd_fast(best_h, n_use, target_use,
                                          time_limit=min(100, tleft() * 0.3),
                                          n_cand=35, max_sweeps=100000, n_bisect=12)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd

    # Phase 8: L-BFGS-B very high alpha
    if tleft() > 70:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[100000, 500000, 2000000],
                            maxiter=200, time_limit=min(50, tleft() * 0.4),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 9: Final pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_fast(best_h, n_use, target_use,
                                  time_limit=min(50, tleft() * 0.6),
                                  n_cand=30, max_sweeps=100000, n_bisect=12)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: Perturb (smaller) and final refinement
    if tleft() > 120:
        rng2 = np.random.RandomState(seed + 1)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert2 = best_h + rng2.randn(n_use) * 0.008
        h_pert2 = project_feasible(h_pert2, target_use)
        hp, sc = lbfgs_phase(h_pert2, n_use, target_use,
                            alphas=[5000, 50000],
                            maxiter=150, time_limit=min(30, tleft() * 0.2),
                            penalty=1000.0)
        if sc < best_score and tleft() > 50:
            best_score = sc
            best_h = hp
            hcd, c5cd = pair_cd_fast(best_h, n_use, target_use,
                                      time_limit=min(40, tleft() * 0.5),
                                      n_cand=30, max_sweeps=80000, n_bisect=12)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
