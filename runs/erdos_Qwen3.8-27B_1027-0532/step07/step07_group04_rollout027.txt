        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]

                if use_kink:
                    k_star = int(np.argmax(corr))
                    g_active = g[k_star]
                    if g_active >= -1e-14:
                        continue
                    if delta_hi <= 0:
                        continue
                    c_active = corr[k_star]
                    others = corr.copy()
                    others[k_star] = -1e30
                    denom = g - g_active
                    mask_pos = denom > 1e-14
                    if mask_pos.any():
                        candidates = (others - c_active)[mask_pos] / denom[mask_pos]
                        candidates = candidates[candidates > 1e-12]
                        if len(candidates) > 0:
                            delta_kink = min(float(np.min(candidates)), delta_hi)
                        else:
                            delta_kink = delta_hi
                    else:
                        delta_kink = delta_hi
                    delta_star = delta_kink
                else:
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(12):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)

                if abs(delta_star) < 1e-12:
                    continue
                delta_star = max(delta_star, delta_lo)
                delta_star = min(delta_star, delta_hi)

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def fista_optimize(h_init, n, target, alpha, time_limit, max_iter=2000):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy()
    h_prev = h.copy()
    t_k = 1.0
    L = alpha * 3.0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        max_c = np.max(c_vals)
        shifted = alpha * (c_vals - max_c)
        weights = np.exp(np.clip(shifted, -100.0, 0.0))
        w_sum = np.sum(weights)
        if w_sum < 1e-100:
            break
        weights /= w_sum

        grad = np.zeros(n)
        mid = n - 1
        for k_idx in range(2 * n - 1):
            w = weights[k_idx]
            if w < 1e-10:
                continue
            k = k_idx - mid
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += w * (1.0 - h[k:])
                if k > 0:
                    grad[k:] -= w * h[:n-k]
            else:
                kn = -k
                if n - kn > 0:
                    grad[:n-kn] += w * (1.0 - h[kn:])
                if kn > 0:
                    grad[kn:] -= w * h[:n-kn]
        grad *= dx

        step = 1.0 / L
        h_new = np.clip(h - step * grad, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        h_y = h_new + ((t_k - 1.0) / t_new) * (h_new - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_y = project_feasible(h_y, target)

        h_prev = h.copy()
        h = h_y
        t_k = t_new

    return h, compute_c5(h)

def slsqp_optimize(h_init, n, target, alphas, maxiter, time_limit, start_time):
    _n = n
    _target = target
    bounds = [(0.0, 1.0)] * n

    def obj_fun(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_vals = corr
        c_max = np.max(c_vals)
        shifted = np.clip(alpha * (c_vals - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def obj_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_vals = corr
        c_max = np.max(c_vals)
        shifted = alpha * (c_vals - c_max)
        weights = np.exp(np.clip(shifted, -100.0, 0.0))
        w_sum = np.sum(weights)
        if w_sum < 1e-100:
            return np.zeros(n)
        weights /= w_sum
        dx = 2.0 / n
        grad = np.zeros(n)
        mid = n - 1
        for k_idx in range(2 * n - 1):
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = k_idx - mid
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += w * (1.0 - x[k:])
                if k > 0:
                    grad[k:] -= w * x[:n-k]
            else:
                kn = -k
                if n - kn > 0:
                    grad[:n-kn] += w * (1.0 - x[kn:])
                if kn > 0:
                    grad[kn:] -= w * x[:n-kn]
        grad *= dx
        return grad

    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_fun, h_cur, args=(alpha,), method='SLSQP',
                          jac=lambda x, a=alpha: obj_grad(x, a),
                          bounds=bounds,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sa_cd(h_init, n, target, time_limit, max_steps=20000,
          t_start=0.03, t_end=0.0002, n_cand=80):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break

        progress = min(step / max(1, max_steps), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-15:
            accept = True
        elif temperature > 1e-12:
            log_accept = -max(delta_c, 0.0) / max(temperature * dx, 1e-15)
            if log_accept > -50:
                if rng.rand() < np.exp(min(log_accept, 0.0)):
                    accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    n_cur = n0
    target_cur = target0
    h_cur = best_h.copy()

    # Phase 1: Quick FISTA to polish initial
    if tleft() > 120:
        for alpha in [50, 200, 1000]:
            if tleft() < 30:
                break
            h_f, sc_f = fista_optimize(h_cur, n_cur, target_cur, alpha,
                                       time_limit=min(25, tleft() * 0.3), max_iter=500)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f
                h_cur = h_f

    # Phase 2: CD at current resolution (n=400)
    if tleft() > 200:
        h_cd, c5_cd = cd_optimize(h_cur, n_cur, target_cur,
                                  time_limit=min(250, tleft() * 0.55),
                                  max_sweeps=40000,
                                  n_cand_start=50, n_cand_end=130,
                                  max_no_improve=300,
                                  n_shifts_active=1, use_kink=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            h_cur = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 80:
        h_p, sc_p = slsqp_optimize(h_cur, n_cur, target_cur,
                                   [500, 2000], 500,
                                   time_limit=min(60, tleft() * 0.6), start_time=start_time)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
            h_cur = h_p

    # Phase 4: CD again after SLSQP
    if tleft() > 100:
        h_cd2, c5_cd2 = cd_optimize(h_cur, n_cur, target_cur,
                                    time_limit=min(100, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=60, n_cand_end=140,
                                    max_no_improve=200,
                                    n_shifts_active=1, use_kink=True)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            h_cur = h_cd2

    # Phase 5: SA for exploration
    if tleft() > 80:
        h_sa, c5_sa = sa_cd(h_cur, n_cur, target_cur,
                            time_limit=min(60, tleft() * 0.5),
                            max_steps=15000, t_start=0.025, t_end=0.0003, n_cand=90)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
            h_cur = h_sa

    # Phase 6: Interpolate to n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(h_cur, n2, target2)
        n_cur = n2
        target_cur = target2
        h_cur = h2
        sc2 = compute_c5(h_cur)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_cur

    # Phase 7: CD at n=800
    if tleft() > 200:
        h_cd3, c5_cd3 = cd_optimize(h_cur, n_cur, target_cur,
                                    time_limit=min(280, tleft() * 0.55),
                                    max_sweeps=25000,
                                    n_cand_start=40, n_cand_end=120,
                                    max_no_improve=250,
                                    n_shifts_active=1, use_kink=True)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            h_cur = h_cd3

    # Phase 8: FISTA at n=800
    if tleft() > 80:
        for alpha in [100, 500]:
            if tleft() < 30:
                break
            h_f2, sc_f2 = fista_optimize(h_cur, n_cur, target_cur, alpha,
                                         time_limit=min(30, tleft() * 0.4), max_iter=400)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_h = h_f2
                h_cur = h_f2

    # Phase 9: SLSQP at n=800
    if tleft() > 60:
        h_p2, sc_p2 = slsqp_optimize(h_cur, n_cur, target_cur,
                                     [1000, 5000], 400,
                                     time_limit=min(50, tleft() * 0.7), start_time=start_time)
        if sc_p2 < best_score:
            best_score = sc_p2
            best_h = h_p2
            h_cur = h_p2

    # Phase 10: Final CD
    if tleft() > 60:
        h_cd4, c5_cd4 = cd_optimize(h_cur, n_cur, target_cur,
                                    time_limit=min(60, tleft() * 0.6),
                                    max_sweeps=15000,
                                    n_cand_start=70, n_cand_end=150,
                                    max_no_improve=150,
                                    n_shifts_active=1, use_kink=True)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4
            h_cur = h_cd4

    # Phase 11: Try n=1200 if time allows
    if tleft() > 200:
        n3 = 1200
        target3 = n3 / 2.0
        h3 = interpolate_to(h_cur, n3, target3)
        n_cur = n3
        target_cur = target3
        h_cur = h3
        sc3 = compute_c5(h_cur)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_cur
        if tleft() > 150:
            h_cd5, c5_cd5 = cd_optimize(h_cur, n_cur, target_cur,
                                        time_limit=min(150, tleft() * 0.6),
                                        max_sweeps=10000,
                                        n_cand_start=40, n_cand_end=100,
                                        max_no_improve=200,
                                        n_shifts_active=1, use_kink=True)
            if c5_cd5 < best_score:
                best_score = c5_cd5
                best_h = h_cd5
                h_cur = h_cd5

    # Final rapid CD
    if tleft() > 30:
        h_cf, c5_cf = cd_optimize(h_cur, n_cur, target_cur,
                                  time_limit=min(20, tleft() * 0.7),
                                  max_sweeps=8000,
                                  n_cand_start=80, n_cand_end=150,
                                  max_no_improve=100,
                                  n_shifts_active=1, use_kink=True)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
