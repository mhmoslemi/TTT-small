import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def sharpen_h(h, target, strength=0.5):
    """Push h towards binary {0,1} while maintaining sum constraint."""
    n = len(h)
    # Apply a transformation that amplifies deviations from 0.5
    center = 0.5
    h_shifted = (h - center) * (1.0 + strength)
    h_new = h_shifted + center
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, target)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False, p_param=8):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    if use_pnorm:
        def func(x, param):
            return pnorm_obj(x, p_param)
    else:
        def func(x, param):
            return smooth_lse_obj(x, param)

            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_subgradient(h, n, top_k=1):
    """Compute subgradient of C5 at h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idxs = np.arange(n)
    mid = n - 1
    
    # Get top shifts
    if top_k == 1:
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        return grad * dx, corr
    else:
        top_shifts = np.argsort(corr)[-top_k:]
        weights = corr[top_shifts].astype(float)
        wsum = np.sum(weights)
        weights = weights / wsum
        grad = np.zeros(n)
        for s_idx, w in zip(top_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= w * h[idxs[v2] + k_star]
        return grad * dx, corr


def subgradient_method(h_init, n, target, time_limit, start_time, 
                       max_iters=50000, base_lr=0.01, decay=0.999,
                       momentum=0.9, sharpen_every=500, sharpen_strength=0.3):
    """Nesterov-accelerated subgradient method with periodic binarization."""
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')

            best_c5 = c5
            best_h = h.copy()
            no_improve = 0

            no_improve += 1
            if no_improve > 5000:
                break

        weights = corr[top_shifts].astype(float)
        wsum = np.sum(weights)

        else:
            weights = np.ones(3) / 3.0

            grad[v2] -= w * h[idxs[v2] + k]
        
        dx = 2.0 / n
        grad *= dx
        # Remove mean to maintain sum constraint during gradient step
        grad -= np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)

            h = h_new
            lr *= 1.0  # keep learning rate

            h_sharp = project_feasible(h_sharp, target)

def block_optimize(h_init, n, target, time_limit, start_time, 
                   block_size=5, max_iters=5000):
    """Optimize blocks of coordinates using Nelder-Mead."""
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_iters // 4:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Pick the block with the most negative gradient sum
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        idxs = np.arange(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        
        # Find block with most negative sum of gradients
        best_block_start = 0
        best_block_sum = np.inf
        for start in range(0, n - block_size + 1, max(1, (n - block_size) // 200)):
            bs = np.sum(grad[start:start + block_size])
            if bs < best_block_sum:
                best_block_sum = bs
                best_block_start = start
        
        block_idx = np.arange(best_block_start, best_block_start + block_size)
        
        # Optimize this block
        h_fixed = h.copy()
        h_block = h[block_idx].copy()
        
        def block_obj(xb):
            h_trial = h_fixed.copy()
            h_trial[block_idx] = np.clip(xb, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            cm = np.max(c)
            if cm <= 0:
                return 0.0
            return float(cm + np.log(np.sum(np.exp(np.clip(200.0 * (c - cm), -500, 0)))) / 200.0)
        
        bounds_block = [(0.0, 1.0)] * block_size
        try:
            res = minimize(block_obj, h_block, method='Nelder-Mead',
                           bounds=bounds_block,
                           options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_trial = h.copy()
            h_trial[block_idx] = h_new_block
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < c5_val - 1e-14:
                h = h_trial
        except Exception:

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def sharpen_h(h, target, strength=0.5):
    n = len(h)
    h_shifted = (h - 0.5) * (1.0 + strength)
    h_new = h_shifted + 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, target)

    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    if use_pnorm:
        def func(x, param):
            return pnorm_obj(x, p_param)
    else:
        def func(x, param):
            return smooth_lse_obj(x, param)

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

                change[p, neg_mask] -= h[p - shifts[neg_mask]]

        if use_sharpen and sharpen_interval > 0 and sweep > 0 and sweep % sharpen_interval == 0:
            h_sharp = sharpen_h(h, target, sharpen_strength)
            h_sharp = project_feasible(h_sharp, target)
            if compute_c5(h_sharp) < c5_val:
                h = h_sharp
                for p in range(n):
                    change[p, :] = 0.0
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def subgradient_momentum(h_init, n, target, time_limit, start_time,
                         max_iters=40000, base_lr=0.008, decay=0.9997,
                         momentum=0.92, sharpen_every=400, sharpen_strength=0.25):
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    idxs = np.arange(n)
    mid_shift = n - 1

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 6000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * (2.0 / n)

        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-3:]
        weights = corr[top_shifts].astype(float)
        wsum = np.sum(weights)
        if wsum > 1e-15:
            weights = weights / wsum
        else:
            weights = np.ones(3) / 3.0

        grad = np.zeros(n)
        for s_idx, w in zip(top_shifts, weights):
            k = int(s_idx) - mid_shift
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        dx = 2.0 / n
        grad *= dx
        grad -= np.mean(grad)

        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new < c5:
            h = h_new
        else:
            h = h_new
            lr *= decay

        if sharpen_every > 0 and it > 0 and it % sharpen_every == 0:
            h_sharp = sharpen_h(h, target, sharpen_strength * max(0.1, 1.0 - float(it) / max_iters))
            h_sharp = project_feasible(h_sharp, target)
            if compute_c5(h_sharp) < c5:
                h = h_sharp

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_refine(h_init, n, target, time_limit, start_time,
                       n_starts=5, noise_scales=None, max_sweeps_per_start=8000):
    if noise_scales is None:
        noise_scales = [0.02, 0.05, 0.08, 0.12, 0.18]
    best_h = h_init.copy().astype(float)
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(99999)

    for i in range(n_starts):
        if time.time() - start_time > time_limit - 20:
            break
        scale = noise_scales[i % len(noise_scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        tlimit = min(time.time() + 80, time_limit - 20)
        h_r, sc_r = pair_cd_improved(h_pert, n, target, tlimit, time.time(),
                                     max_sweeps=max_sweeps_per_start,
                                     n_cand_start=10, n_cand_end=40,
                                     max_no_improve=300, n_top_shifts=3,
                                     bisection_iters=16, use_sharpen=True,
                                     sharpen_interval=2000, sharpen_strength=0.25)
        if sc_r < best_c5:
            best_c5 = sc_r
            best_h = h_r

    return best_h, best_c5


def block_optimize(h_init, n, target, time_limit, start_time,
                   block_size=4, max_iters=4000):
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_iters // 3:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        best_block_start = 0
        best_block_sum = np.inf
        step_size = max(1, (n - block_size) // 300)
        for start in range(0, n - block_size + 1, step_size):
            bs = np.sum(grad[start:start + block_size])
            if bs < best_block_sum:
                best_block_sum = bs
                best_block_start = start

        block_idx = np.arange(best_block_start, best_block_start + block_size)
        h_fixed = h.copy()
        h_block = h[block_idx].copy()

        def block_obj(xb):
            h_trial = h_fixed.copy()
            h_trial[block_idx] = np.clip(xb, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            cm = np.max(c)
            if cm <= 0:
                return 0.0
            return float(cm + np.log(np.sum(np.exp(np.clip(300.0 * (c - cm), -500, 0)))) / 300.0)

        bounds_block = [(0.0, 1.0)] * block_size
        try:
            res = minimize(block_obj, h_block, method='Nelder-Mead',
                           bounds=bounds_block,
                           options={'maxiter': 800, 'xatol': 1e-11, 'fatol': 1e-13})
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_trial = h.copy()
            h_trial[block_idx] = h_new_block
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < c5_val - 1e-14:
                h = h_trial
        except Exception:
            pass

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - 30.0 - elapsed()

    # Phase 1: SLSQP with smooth LSE
    if remaining() > 60:
        h1, sc1 = slsqp_optimize(best_h, n0, target0,
                                  [30, 100, 500, 2000],
                                  600, start_time + min(60, remaining() - 30), start_time, use_pnorm=False)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: SLSQP p-norm
    if remaining() > 45:
        h2, sc2 = slsqp_optimize(best_h, n0, target0,
                                  [4, 8, 16, 32],
                                  500, start_time + min(45, remaining() - 30), start_time, use_pnorm=True, p_param=12)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Main pair CD with sharpening
    if remaining() > 140:
        h3, sc3 = pair_cd_improved(best_h, n0, target0,
                                    start_time + min(140, remaining() - 30), start_time,
                                    max_sweeps=60000, n_cand_start=10, n_cand_end=55,
                                    max_no_improve=400, n_top_shifts=3,
                                    bisection_iters=18, use_sharpen=True,
                                    sharpen_interval=3000, sharpen_strength=0.3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Subgradient with momentum
    if remaining() > 70:
        h4, sc4 = subgradient_momentum(best_h, n0, target0,
                                        start_time + min(70, remaining() - 30), start_time,
                                        max_iters=30000, base_lr=0.006, decay=0.9997,
                                        momentum=0.92, sharpen_every=300, sharpen_strength=0.25)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Upsample to n=600 and CD
    if remaining() > 140:
        n1 = 600
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        h5, sc5 = pair_cd_improved(h_int, n1, target1,
                                    start_time + min(140, remaining() - 30), start_time,
                                    max_sweeps=40000, n_cand_start=10, n_cand_end=45,
                                    max_no_improve=350, n_top_shifts=3,
                                    bisection_iters=18, use_sharpen=True,
                                    sharpen_interval=2500, sharpen_strength=0.25)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5
            n0 = n1
            target0 = target1

    # Phase 6: SLSQP polish
    if remaining() > 55:
        h6, sc6 = slsqp_optimize(best_h, n0, target0,
                                  [5000, 15000],
                                  1000, start_time + min(55, remaining() - 25), start_time, use_pnorm=False)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: More pair CD with wider candidate set
    if remaining() > 110:
        h7, sc7 = pair_cd_improved(best_h, n0, target0,
                                    start_time + min(110, remaining() - 30), start_time,
                                    max_sweeps=45000, n_cand_start=15, n_cand_end=65,
                                    max_no_improve=350, n_top_shifts=5,
                                    bisection_iters=18, use_sharpen=True,
                                    sharpen_interval=2000, sharpen_strength=0.2)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Multi-start
    if remaining() > 130:
        h8, sc8 = multi_start_refine(best_h, n0, target0,
                                      start_time + min(130, remaining() - 30), start_time,
                                      n_starts=4, noise_scales=[0.03, 0.06, 0.1, 0.15],
                                      max_sweeps_per_start=7000)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Block optimization
    if remaining() > 60:
        h9, sc9 = block_optimize(best_h, n0, target0,
                                  start_time + min(60, remaining() - 25), start_time,
                                  block_size=4, max_iters=3000)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final SLSQP
    if remaining() > 45:
        h10, sc10 = slsqp_optimize(best_h, n0, target0,
                                    [20000, 60000],
                                    800, start_time + min(45, remaining() - 20), start_time, use_pnorm=False)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    # Phase 11: Final pair CD (single shift, precise)
    if remaining() > 35:
        h11, sc11 = pair_cd_improved(best_h, n0, target0,
                                      start_time + min(35, remaining() - 15), start_time,
                                      max_sweeps=15000, n_cand_start=25, n_cand_end=80,
                                      max_no_improve=200, n_top_shifts=1,
                                      bisection_iters=22, use_sharpen=False)
        if sc11 < best_score:
            best_score = sc11
            best_h = h11

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
