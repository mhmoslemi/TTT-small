    h = np.clip(h.astype(float), 0.0, 1.0)

            break

        h = np.clip(h, 0.0, 1.0)
    return h

                h_cur = h_new

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _obj_grad(x):
    # Gradient of logsumexp approximation
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = _g_alpha * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    probs = exp_shifted / np.sum(exp_shifted)
    # d/dx of sum_i x_i (1-x_i+k) for each shift k
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k_idx, p in enumerate(probs):
        if p < 1e-15:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += p * (1.0 - x[idxs[v1] - k])
        grad[v2] -= p * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        h_cur = project_feasible(h_cur, target)
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=_obj_grad,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
        # Verify and re-project
        h_cur = project_feasible(h_cur, target)
    return h_cur, compute_c5(h_cur)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _obj_grad(x):
    n = _g_n
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - cmax), -500.0, 0.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    sig = np.where(p > 1e-10)[0]
    for k_idx in sig:
        k = int(k_idx) - mid
        wi = p[k_idx]
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wi * (1.0 - x[idx - k])
        if k >= 0:
            j_lo = 0
            j_hi = n - 1 - k
        else:
            j_lo = -k
            j_hi = n - 1
        jdx = np.arange(j_lo, j_hi + 1)
        grad[jdx] += wi * (-x[jdx + k])
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_list):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for i, alpha in enumerate(alphas):
        _g_alpha = alpha
        h_cur = project_feasible(h_cur, target)
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=_obj_grad,
                           options={'maxiter': maxiter_list[i], 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
        h_cur = project_feasible(h_cur, target)
    return h_cur, compute_c5(h_cur)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_balanced(h_init, n, target, time_limit, max_sweeps=100000,
                     n_cand_start=40, n_cand_end=100, n_shifts=4,

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        top10_idx = np.argsort(corr)[-10:]
        top10_spread = float(np.std(corr[top10_idx]))

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_spread = top10_spread

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-11:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                new_corr = corr + delta_star * g
                approx_max = np.max(new_corr) * dx
                new_spread = float(np.std(new_corr[np.argsort(new_corr)[-10:]]))
                score = approx_max + balance_weight * (new_spread - top10_spread)
                if score < best_trial_c5 + balance_weight * (best_spread - top10_spread) - 1e-14:
                    best_trial_c5 = approx_max
                    best_spread = new_spread
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
        else:
            pass

    return best_h, best_c5

def stochastic_restarts(h_best, n, target, time_limit, n_restarts=8):

        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)

    start = time.time()
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    rng = np.random.RandomState(999)

    for t in range(n_trials):
        if time.time() - start > time_limit:
            break
        h_try = h_best.copy()
        ptype = t % 4
        if ptype == 0:
            lo = rng.randint(0, n - 50)
            width = rng.randint(10, 50)
            h_try[lo:lo + width] += 0.1
        elif ptype == 1:
            lo = rng.randint(0, n - 30)
            width = rng.randint(5, 30)
            h_try[lo:lo + width] -= 0.1
        elif ptype == 2:
            freq = rng.randint(2, 8)
            h_try += 0.05 * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
        else:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i != j:
                swap = h_try[i]
                h_try[i] = h_try[j]
                h_try[j] = swap

        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5:
            h_pol, sc2 = slsqp_round(h_try, n, target, [1000, 5000], [300, 300])
            if sc2 < best_c5:
                best_c5 = sc2
                best_h = h_pol.copy()

    return best_h, best_c5
