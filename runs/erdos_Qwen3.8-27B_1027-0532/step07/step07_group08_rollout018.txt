1. Compute the "change vector" a_k for a given pair (i, j): a_k = (h[i+k] + h[i-k] - h[j+k] - h[j-k]) with 0 for out-of-range.

3. Find the optimal δ by bisection on this 1D problem.

For step 3, the optimal δ that minimizes max_k (corr[k] + δ * a_k) can be found as follows:
- If all a_k ≤ 0: increase δ as much as possible (δ_hi)
- If all a_k ≥ 0: δ* = 0 (no improvement)
- Otherwise: the optimal δ* is where the maximum of the increasing lags equals the maximum of the decreasing lags (or one of the lags hits the boundary)

- N = {k : a_k < 0} (decreasing lags)

The function max_{k in P} (corr[k] + δ*a_k) is increasing in δ.

        sc1 = compute_c5(h1)
        current_n = n1
        current_target = target1
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        else:
            best_h = h1

        # Phase 3: Momentum descent at n=800
        if tleft() > 250:
            h_md, c5_md = momentum_descend(best_h, current_n, current_target,
                                           time_limit=min(100, tleft() * 0.3),
                                           alpha=5000, lr0=0.004, momentum=0.82)
            if c5_md < best_score:
                best_score = c5_md
                best_h = h_md

        # Phase 4: SLSQP at n=800
        if tleft() > 150:
            deadline = time.time() + min(100, tleft() * 0.4)
            h_pol2, sc2 = slsqp_optimize(best_h, current_n, current_target,
                                         [3000, 10000, 30000, 80000],
                                         1500, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

        # Phase 5: Pair CD at n=800
        if tleft() > 200:
            h_cd, c5_cd = pair_cd_fast(best_h, current_n, current_target,
                                       time_limit=min(180, tleft() * 0.5),
                                       n_active=10, max_no_improve=600, reheat_scale=0.05)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        # Phase 6: SLSQP after CD
        if tleft() > 80:
            deadline = time.time() + min(60, tleft() * 0.4)
            h_pol3, sc3 = slsqp_optimize(best_h, current_n, current_target,
                                         [10000, 30000, 80000, 200000],
                                         1200, deadline=deadline)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 7: Multi-restart with perturbations
    if tleft() > 250:
        for attempt in range(8):
            if tleft() < 50:
                break
            noise = [0.015, 0.025, 0.04, 0.06, 0.08, 0.12, 0.18, 0.25][attempt % 8]
            h_pert = best_h + rng.randn(current_n) * noise
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(25, tleft() * 0.2)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [1000, 5000, 15000, 40000],
                                           800, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 8: Momentum descent
    if tleft() > 120:
        h_md2, c5_md2 = momentum_descend(best_h, current_n, current_target,
                                         time_limit=min(80, tleft() * 0.5),
                                         alpha=8000, lr0=0.003, momentum=0.80)
        if c5_md2 < best_score:
            best_score = c5_md2
            best_h = h_md2

    # Phase 9: Pair CD
    if tleft() > 100:
        h_cd2, c5_cd2 = pair_cd_fast(best_h, current_n, current_target,
                                     time_limit=min(80, tleft() * 0.5),
                                     n_active=12, max_no_improve=500, reheat_scale=0.02)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 10: Try n=1200 if time allows
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_md3, c5_md3 = momentum_descend(h2, n2, target2,
                                             time_limit=min(80, tleft() * 0.3),
                                             alpha=5000, lr0=0.003, momentum=0.82)
            if c5_md3 < best_score:
                best_score = c5_md3
                best_h = h_md3
                current_n = n2
                current_target = target2
        if tleft() > 100:
            deadline = time.time() + min(80, tleft() * 0.4)
            h_pol4, sc4 = slsqp_optimize(best_h, current_n, current_target,
                                         [5000, 15000, 40000, 100000],
                                         1000, deadline=deadline)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_pol4
        if tleft() > 80:
            h_cd3, c5_cd3 = pair_cd_fast(best_h, current_n, current_target,
                                         time_limit=min(60, tleft() * 0.4),
                                         n_active=10, max_no_improve=400, reheat_scale=0.03)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.5)
        h_final_pol, sc_final = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                               [20000, 60000, 150000, 400000],
                                               1500, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
