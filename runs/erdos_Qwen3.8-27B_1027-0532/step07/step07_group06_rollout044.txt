def project_feasible(h, target_sum):

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = weights.sum()
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue

                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = d_lo
                b_d = d_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5

def stochastic_refine(h_init, n, target, time_limit, start_time,
                      max_steps=30000, base_scale=0.03, seed=999):
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 4000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max(max_steps, 1))
        n_pert = int(rng.choice([1, 2, 3, 4, 6]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: DE Global Search
    if remaining() > 350:
        phase1_limit = start_time + min(330, remaining() - 20)
        h_de, sc_de = de_search(best_h, n, target, phase1_limit, start_time,
                                pop_size=25, max_gens=400, seed=seed)
        if sc_de < best_score:
            best_score = sc_de
            best_h = h_de

    # Phase 2: Gradient-based local search
    if remaining() > 200:
        phase2_limit = start_time + min(180, remaining() - 15)
        h_grad, sc_grad = gradient_local_search(best_h, n, target, phase2_limit, start_time,
                                                max_iters=40000, top_m_start=3, top_m_end=8)
        if sc_grad < best_score:
            best_score = sc_grad
            best_h = h_grad

    # Phase 3: Pair CD fine-tuning
    if remaining() > 200:
        phase3_limit = start_time + min(180, remaining() - 15)
        h_pc, sc_pc = pair_cd_fine(best_h, n, target, phase3_limit, start_time,
                                   max_sweeps=30000, n_cand_start=15, n_cand_end=50,
                                   max_no_improve=300, n_top_shifts=1, bisection_iters=18)
        if sc_pc < best_score:
            best_score = sc_pc
            best_h = h_pc

    # Phase 4: SLSQP refinement
    if remaining() > 80:
        phase4_limit = start_time + min(75, remaining() - 10)
        h_sl, sc_sl = slsqp_refine(best_h, n, target, [100, 500, 2000],
                                   800, phase4_limit, start_time)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 5: Second DE round with smaller population from improved start
    if remaining() > 150:
        phase5_limit = start_time + min(140, remaining() - 15)
        h_de2, sc_de2 = de_search(best_h, n, target, phase5_limit, start_time,
                                  pop_size=20, max_gens=200, seed=seed + 1)
        if sc_de2 < best_score:
            best_score = sc_de2
            best_h = h_de2

    # Phase 6: Gradient + Pair CD cycle
    if remaining() > 150:
        phase6_limit = start_time + min(130, remaining() - 15)
        h_g2, sc_g2 = gradient_local_search(best_h, n, target, phase6_limit, start_time,
                                            max_iters=30000, top_m_start=2, top_m_end=6)
        if sc_g2 < best_score:
            best_score = sc_g2
            best_h = h_g2

    if remaining() > 100:
        phase6b_limit = start_time + min(90, remaining() - 10)
        h_pc2, sc_pc2 = pair_cd_fine(best_h, n, target, phase6b_limit, start_time,
                                     max_sweeps=25000, n_cand_start=20, n_cand_end=55,
                                     max_no_improve=250, n_top_shifts=3, bisection_iters=18)
        if sc_pc2 < best_score:
            best_score = sc_pc2
            best_h = h_pc2

    # Phase 7: Multi-start with structured perturbations
    if remaining() > 200:
        rng = np.random.RandomState(seed + 77)
        for pert_type in range(5):
            if remaining() < 40:
                break
            if pert_type == 0:
                h_pert = project_feasible(best_h + rng.randn(n) * 0.02, target)
            elif pert_type == 1:
                block = n // 5
                s = rng.randint(0, block)
                t = rng.randint(block, n - block)
                h_pert = pop_init_block(best_h, s, t, block)
                h_pert = project_feasible(h_pert, target)
            elif pert_type == 2:
                shift = rng.randint(n // 20, n // 10)
                h_pert = project_feasible(np.roll(best_h, shift), target)
            elif pert_type == 3:
                h_pert = project_feasible(best_h + rng.randn(n) * 0.05, target)
            else:
                mask = rng.rand(n) < 0.3
                h_pert = best_h.copy()
                h_pert[mask] += rng.randn(int(mask.sum())) * 0.04
                h_pert = project_feasible(h_pert, target)

            phase7_limit = start_time + min(35, remaining() - 10)
            h_g3, sc_g3 = gradient_local_search(h_pert, n, target, phase7_limit, start_time,
                                                max_iters=15000, top_m_start=2, top_m_end=5)
            if remaining() < 20:
                continue
            phase7b_limit = start_time + min(20, remaining() - 10)
            h_pc3, sc_pc3 = pair_cd_fine(h_g3, n, target, phase7b_limit, start_time,
                                         max_sweeps=10000, n_cand_start=15, n_cand_end=40,
                                         max_no_improve=150, n_top_shifts=1, bisection_iters=16)
            if sc_pc3 < best_score:
                best_score = sc_pc3
                best_h = h_pc3

    # Phase 8: Stochastic refinement
    if remaining() > 80:
        phase8_limit = start_time + min(70, remaining() - 10)
        h_st, sc_st = stochastic_refine(best_h, n, target, phase8_limit, start_time,
                                        max_steps=20000, base_scale=0.02, seed=seed + 313)
        if sc_st < best_score:
            best_score = sc_st
            best_h = h_st

    # Phase 9: Final SLSQP polish
    if remaining() > 60:
        phase9_limit = start_time + min(50, remaining() - 10)
        h_fin, sc_fin = slsqp_refine(best_h, n, target, [5000, 20000, 60000],
                                     1000, phase9_limit, start_time)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 10: Final pair CD
    if remaining() > 40:
        phase10_limit = start_time + min(30, remaining() - 8)
        h_fin2, sc_fin2 = pair_cd_fine(best_h, n, target, phase10_limit, start_time,
                                       max_sweeps=10000, n_cand_start=30, n_cand_end=70,
                                       max_no_improve=100, n_top_shifts=1, bisection_iters=20)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
