The key new idea is the "correlation equalization" step. Here's how it works:

Given the current correlation vector c = (c_1, ..., c_m), let c_max = max(c).
The active set A = {k : c_k ≥ c_max - ε} for small ε.
We want to find a direction d (change to h) that:
- Decreases c_k for k in A
- Doesn't increase c_k for k not in A
- Preserves the sum constraint

The gradient of c_k with respect to h is:
∂c_k/∂h[p] = (1-h[p-k]) - h[p+k] (with valid range constraints)

So the gradient of the max (subgradient) is the average of gradients over the active set:
g = (1/|A|) Σ_{k∈A} ∇c_k

We want to move in -g direction (to decrease the max), subject to the constraint that we don't increase any other c_k too much.

1. Computing the exact subgradient (not just top-k approximations)
2. Using a line search along the projected subgradient
3. Checking feasibility of all correlations, not just the max

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def _top_m_obj(x, alpha, m):
    """Smooth approximation of the sum of top-m correlations."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    # Soft top-m using log-sum-exp tricks
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    # Approximate top-m by thresholding
    threshold = c_max - m / alpha * np.log(n_corr / m)
    top_mask = corr >= threshold
    if not top_mask.any():
        top_mask = corr >= c_max - 1e-10
    top_sum = np.sum(corr[top_mask])
    return float(top_sum / max(np.sum(top_mask), 1))

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:

    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def _topk_obj(x, alpha, k_top):

    sorted_c = np.sort(corr)[::-1]
    top_sum = np.sum(sorted_c[:k_top])
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

        else:
            consecutive_no_improve += 1

    return best_h, best_c5


def momentum_subgradient(h_init, n, target, time_limit, n_iter=15000,
                         momentum=0.85, lr_init=0.008, top_k=25,
                         weight_beta=5.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.04)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.92

        top_idx = np.argsort(corr)[-top_k:]
        c_max = float(np.max(corr))
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = np.exp(weight_beta * (corr[s_idx] - c_max) / max(c_max, 1e-10))
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm
        grad -= np.mean(grad)

        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        sc = compute_c5(h_new)
        if sc <= c5_val:
            h = h_new
        else:
            h_new2 = project_feasible(h - 0.3 * lr * grad, target)
            sc2 = compute_c5(h_new2)
            if sc2 < c5_val:
                h = h_new2
            v *= 0.5

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    _g_n = n0
    _g_target = target0

    # Phase 1: Quick SLSQP polish
    if tleft() > 40:
        h_pol, sc = slsqp_round(best_h, n0, target0, [500, 5000], 250, 1e-14)
        best_h = h_pol
        best_score = min(best_score, sc)

    # Phase 2: Main enhanced pair CD
    if tleft() > 250:
        h_cd, c5_cd = enhanced_pair_cd(best_h, n0, target0,
                                       time_limit=min(230, tleft() * 0.45),
                                       max_sweeps=50000,
                                       n_cand_start=30, n_cand_end=80,
                                       max_no_improve=400,
                                       weight_beta=6.0,
                                       top_k_shifts=40,
                                       bisection_iters=22,
                                       max_pairs_per_sweep=200)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Top-k SLSQP (penalize top-3 correlations)
    if tleft() > 100:
        _g_n = n0
        _g_target = target0
        h_tk, sc_tk = topk_slsqp_round(best_h, n0, target0,
                                       k_top=3,
                                       alphas=[500, 5000],
                                       maxiter_per_alpha=300)
        if sc_tk < best_score:
            best_score = sc_tk
            best_h = h_tk

    # Phase 4: Second pair CD with different params
    if tleft() > 180:
        h_cd2, c5_cd2 = enhanced_pair_cd(best_h, n0, target0,
                                         time_limit=min(160, tleft() * 0.4),
                                         max_sweeps=40000,
                                         n_cand_start=40, n_cand_end=100,
                                         max_no_improve=350,
                                         weight_beta=8.0,
                                         top_k_shifts=50,
                                         bisection_iters=24,
                                         max_pairs_per_sweep=150)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Momentum subgradient
    if tleft() > 100:
        h_mom, c5_mom = momentum_subgradient(best_h, n0, target0,
                                             time_limit=min(80, tleft() * 0.3),
                                             n_iter=12000, momentum=0.85,
                                             lr_init=0.006, top_k=30)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 6: SLSQP with high alpha
    if tleft() > 80:
        _g_n = n0
        _g_target = target0
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [20000, 100000], 350, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 7: Interpolate to n=1000
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 300:
        h1 = interpolate_to(best_h, n1, target1)
        _g_n = n1
        _g_target = target1

        if tleft() > 250:
            h_pol1, sc1 = slsqp_round(h1, n1, target1, [1000, 10000], 300, 1e-14)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
            else:
                best_h = h_pol1

        if tleft() > 180:
            h_cd1, c5_cd1 = enhanced_pair_cd(best_h, n1, target1,
                                             time_limit=min(160, tleft() * 0.4),
                                             max_sweeps=35000,
                                             n_cand_start=35, n_cand_end=90,
                                             max_no_improve=350,
                                             weight_beta=6.0,
                                             top_k_shifts=45,
                                             bisection_iters=22,
                                             max_pairs_per_sweep=200)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 60:
            _g_n = n1
            _g_target = target1
            h_pol1b, sc1b = slsqp_round(best_h, n1, target1,
                                        [50000, 200000], 400, 1e-15)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 8: Try n=1200
    n2 = 1200
    target2 = n2 / 2.0
    if tleft() > 280:
        h2 = interpolate_to(best_h, n2, target2)
        _g_n = n2
        _g_target = target2

        if tleft() > 230:
            h_pol2, sc2b = slsqp_round(h2, n2, target2, [2000, 20000], 300, 1e-14)
            if tleft() > 170:
                h_cd2h, c5_cd2h = enhanced_pair_cd(h_pol2, n2, target2,
                                                   time_limit=min(150, tleft() * 0.45),
                                                   max_sweeps=25000,
                                                   n_cand_start=40, n_cand_end=100,
                                                   max_no_improve=300,
                                                   weight_beta=7.0,
                                                   top_k_shifts=50,
                                                   bisection_iters=22,
                                                   max_pairs_per_sweep=180)
                if c5_cd2h < best_score:
                    best_score = c5_cd2h
                    best_h = h_cd2h
                else:
                    best_h = h_pol2

                if tleft() > 60:
                    _g_n = n2
                    _g_target = target2
                    h_pol3, sc3 = slsqp_round(best_h, n2, target2,
                                              [100000], 400, 1e-15)
                    if sc3 < best_score:
                        best_score = sc3
                        best_h = h_pol3

    # Phase 9: Final pair CD
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin_cd = enhanced_pair_cd(best_h, n_cur, target_cur,
                                               time_limit=min(40, tleft() * 0.55),
                                               max_sweeps=15000,
                                               n_cand_start=50, n_cand_end=100,
                                               max_no_improve=200,
                                               weight_beta=10.0,
                                               top_k_shifts=50,
                                               bisection_iters=24,
                                               max_pairs_per_sweep=250)
        if c5_fin_cd < best_score:
            best_score = c5_fin_cd
            best_h = h_fin_cd

    # Phase 10: Final SLSQP
    if tleft() > 25:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        _g_n = n_cur
        _g_target = target_cur
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [200000], 300, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
