```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=np.int32)
    change = np.zeros((n, n_corr), dtype=np.float64)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        if pos_mask.any():
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        if neg_mask.any():
            change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    if pos_mask.any():
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    if neg_mask.any():
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

def slsqp_smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_smooth_grad(x, alpha, top_k=80):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    n_corr = 2 * n - 1
    if top_k < n_corr:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(n_corr)
    shifted = np.clip(alpha * (corr[top_idx] - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_s / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for ki in range(len(top_idx)):
        w = weights[ki]
        if w < 1e-9:
            continue
        s_idx = int(top_idx[ki])
        k = s_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _slsqp_obj(x):
    return slsqp_smooth_obj(x, _g_alpha)

def _slsqp_grad(x):
    return slsqp_smooth_grad(x, _g_alpha, top_k=100)

def _slsqp_cons(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14, time_limit=None):
    global _g_alpha, _g_target
    start = time.time()
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit and time.time() - start > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=30, n_cand_end=80, top_k_eval=60,
               max_no_improve=1000, n_shifts_grad=3, block_size=1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=np.int32)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        # Multi-shift gradient
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Score matrix based on the max shift
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            # Bisection to find optimal delta
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(13):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5
```

At delta = 0, corr doesn't change. As delta increases, we're redistributing mass. The function f(delta) = max(corr + delta*g) where g = change[j] - change[i] tells us how the correlation vector changes.

The bisection finds the delta that minimizes f, which is exactly what we want.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=np.int32)
    change = np.zeros((n, n_corr), dtype=np.float64)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        if pos_mask.any():
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        if neg_mask.any():
            change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    if pos_mask.any():
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    if neg_mask.any():
        change[p, neg_mask] -= h[p - shifts[neg_mask]]


def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_grad(x, alpha, top_k=100):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    n_corr = 2 * n - 1
    if top_k < n_corr:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(n_corr)
    shifted = np.clip(alpha * (corr[top_idx] - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_s / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for ki in range(len(top_idx)):
        w = weights[ki]
        if w < 1e-9:
            continue
        s_idx = int(top_idx[ki])
        k = s_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _smo(x):
    return smooth_max_obj(x, _g_alpha)


def _smg(x):
    return smooth_max_grad(x, _g_alpha, top_k=120)


def _scon(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14, time_limit=None):
    global _g_alpha, _g_target
    start = time.time()
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_smo, h_cur, jac=_smg,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit=None):
    global _g_alpha, _g_target
    start = time.time()
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    def obj_pen(x):
        return smooth_max_obj(x, _g_alpha) + 1e8 * (np.sum(x) - target) ** 2

    def grad_pen(x):
        return smooth_max_grad(x, _g_alpha, top_k=120) + 2e8 * (np.sum(x) - target) * np.ones(n)

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(obj_pen, h_cur, jac=grad_pen,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-6})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=80, top_k_eval=60,
            max_no_improve=800, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n, dtype=np.int32)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def stochastic_refine(h_init, n, target, time_limit, max_steps=50000):
    start = time.time()
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        progress = float(step) / max_steps
        scale = 0.03 * max(0.05, 1.0 - progress)
        n_pert = int(rng.choice([2, 3, 5, 7, 10]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 500:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

    if _tl() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, n1, target1, time_limit=min(130, _tl() * 0.85),
                                max_sweeps=40000, n_cand_start=40, n_cand_end=90,
                                top_k_eval=80, n_shifts_grad=4, max_no_improve=600)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if _tl() > 80:
        h_ms, c5_ms = pair_cd_multitarget(best_h, n1, target1,
                                          time_limit=min(70, _tl() * 0.85),
                                          max_sweeps=25000, n_cand=60,
                                          top_k_eval=60, n_shifts=5, max_no_improve=400)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    n2 = 1600
    target2 = n2 / 2.0
    if _tl() > 350:
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        h_cd3, c5_cd3 = pair_cd(h2, n2, target2, time_limit=min(280, _tl() * 0.85),
                                max_sweeps=35000, n_cand_start=30, n_cand_end=70,
                                top_k_eval=50, n_shifts_grad=3, max_no_improve=600)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

        if _tl() > 50:
            best_h = slsqp_optimize(best_h, n2, target2,
                                    [5000, 20000], 600, 1e-15, time_limit=_tl() - 5)
            sc = compute_c5(best_h)
            if sc < best_score:
                best_score = sc

    nf = len(best_h)
    tf = nf / 2.0

    if _tl() > 50:
        h_ss, c5_ss = stochastic_refine(best_h, nf, tf,
                                        time_limit=min(40, _tl() * 0.8), max_steps=30000)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    if _tl() > 40:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04]:
            if _tl() < 30:
                break
            h_pert = best_h + rng.randn(nf) * noise_scale
            h_pert = project_feasible(h_pert, tf)
            h_ref = slsqp_optimize(h_pert, nf, tf,
                                   [1000, 5000, 20000], 400, 1e-15, time_limit=_tl() - 5)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if _tl() > 60:
        h_final_cd, c5_final = pair_cd(best_h, nf, tf, time_limit=min(50, _tl() * 0.8),
                                       max_sweeps=25000, n_cand_start=40, n_cand_end=90,
                                       top_k_eval=70, n_shifts_grad=3, max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if _tl() > 25:
        best_h = slsqp_optimize(best_h, nf, tf, [10000, 50000], 500, 1e-15, time_limit=_tl() - 3)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
