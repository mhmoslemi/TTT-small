import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen in [penalty, penalty * 10, penalty * 100]:
            if time.time() - start > time_limit:
                break
            try:
                res = minimize(lse_objective, h_cur, 
                              args=(alpha, target, pen, dx, n),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-10:
                    break
            except Exception:
                pass
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000, n_shifts=1, alpha_ctrl=50.0):

        for ci in range(nc):

            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):

        cand_pos = {int(all_cands[p]): p for p in range(nc)}
        
        best_improve = 0.0
        best_i = -1

                continue

                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[j]

                # Bisection to find optimal delta
                lo = 0.0
                hi_d = delta_max

                    vals = corr + mid_d * g

                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:

                    else:
                        lo = mid_d

                    best_j = j

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, alpha, target, penalty, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_grad(x, alpha, target, penalty, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS at original resolution
    if tleft() > 200:
        hp, sc = lbfgs_optimize(best_h, n, target,
                              alphas=[100, 500, 2000, 10000],
                              maxiter=300, time_limit=min(50, tleft() * 0.15),
                              penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Pair CD at original resolution
    if tleft() > 250:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(80, tleft() * 0.2),
                           n_cand=50, max_sweeps=100000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: L-BFGS high alpha at original resolution
    if tleft() > 200:
        hp, sc = lbfgs_optimize(best_h, n, target,
                              alphas=[10000, 50000, 200000, 1000000],
                              maxiter=250, time_limit=min(50, tleft() * 0.15),
                              penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Flatten active constraints at original resolution
    if tleft() > 200:
        hf, scf = flatten_active(best_h, n, target, time_limit=min(40, tleft() * 0.15),
                                  n_active=5, lr=0.01)
        if scf < best_score:
            best_score = scf
            best_h = hf

    # Phase 5: Upsample to n=1200 and optimize
    if tleft() > 500:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        c5_interp = compute_c5(h_interp)

        # Quick L-BFGS to settle
        hp, sc = lbfgs_optimize(h_interp, n_new, target_new,
                              alphas=[200, 2000, 20000],
                              maxiter=200, time_limit=min(40, tleft() * 0.1),
                              penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
        else:
            best_h = h_interp
            best_score = c5_interp

        # Pair CD at higher resolution - main workhorse
        if tleft() > 250:
            hcd, c5cd = pair_cd(best_h, n_new, target_new, time_limit=min(150, tleft() * 0.3),
                               n_cand=40, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

        # L-BFGS high alpha at higher resolution
        if tleft() > 150:
            hp, sc = lbfgs_optimize(best_h, n_new, target_new,
                                  alphas=[50000, 200000, 1000000],
                                  maxiter=200, time_limit=min(60, tleft() * 0.15),
                                  penalty=1000.0)
            if sc < best_score:
                best_score = sc
                best_h = hp

        # Pair CD multi-shift at higher resolution
        if tleft() > 150:
            hcd, c5cd = pair_cd(best_h, n_new, target_new, time_limit=min(80, tleft() * 0.25),
                               n_cand=50, max_sweeps=50000, n_shifts=3, alpha_ctrl=40.0)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

        # Flatten at higher resolution
        if tleft() > 100:
            hf, scf = flatten_active(best_h, n_new, target_new, time_limit=min(40, tleft() * 0.2),
                                      n_active=7, lr=0.005)
            if scf < best_score:
                best_score = scf
                best_h = hf

        # More pair CD at higher resolution
        if tleft() > 100:
            hcd, c5cd = pair_cd(best_h, n_new, target_new, time_limit=min(60, tleft() * 0.3),
                               n_cand=50, max_sweeps=50000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

        n = n_new
        target = target_new

    # Phase 6: Multi-start perturbation
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        scales = [0.02, 0.04, 0.06]
        for trial in range(3):
            if tleft() < 60:
                break
            noise_scale = scales[trial]
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            hp, sc = lbfgs_optimize(h_pert, n_use, target_use,
                                    alphas=[5000, 50000, 500000],
                                    maxiter=150, time_limit=min(25, tleft() * 0.2),
                                    penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
            if tleft() > 40:
                hcd, c5cd = pair_cd(hp, n_use, target_use, time_limit=min(20, tleft() * 0.3),
                                    n_cand=35, max_sweeps=30000, n_shifts=1)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd

    # Phase 7: Subgradient with momentum at current resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hsg, c5sg = subgradient_momentum(best_h, n_use, target_use,
                                         time_limit=min(40, tleft() * 0.4),
                                         max_steps=50000, n_shifts=7, base_lr=0.03)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 8: Final L-BFGS very high alpha
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                              alphas=[500000, 2000000, 10000000],
                              maxiter=200, time_limit=min(40, tleft() * 0.5),
                              penalty=10000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 9: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(25, tleft() * 0.7),
                           n_cand=35, max_sweeps=30000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: Final flatten
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hf, scf = flatten_active(best_h, n_use, target_use, time_limit=min(15, tleft() * 0.7),
                                  n_active=5, lr=0.005)
        if scf < best_score:
            best_score = scf
            best_h = hf

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
