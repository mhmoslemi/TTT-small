def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

    return np.sum(x) - target

def _cons_jac(x):
    return np.ones_like(x)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, use_pnorm=False, p_val=4):

    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (target,), 'jac': _cons_jac}]

    for param in alphas:
        if time.time() > time_limit:
            break
        try:
            if use_pnorm:
                res = minimize(_pnorm_obj, h_cur, args=(p_val,), method='SLSQP',
                           bounds=bounds, constraints=cons,

                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

                                n_trials=20, scale_start=0.05, scale_end=0.015,
                                pair_cd_sweeps=3000, mix_ratio=0.5)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    remaining = budget_s - 70.0 - (time.time() - start_time)
    if remaining > 120:
        h4, sc4 = pair_cd(best_h, n0, target0,
                          start_time + min(180, remaining - 30),
                          max_sweeps=50000, n_cand_start=18, n_cand_end=55,

    remaining = budget_s - 60.0 - (time.time() - start_time)

        h5, sc5 = slsqp_round(best_h, n0, target0, [5000, 20000],
                              1200, start_time + min(60, remaining - 20), start_time, use_pnorm=False)

    remaining = budget_s - 70.0 - (time.time() - start_time)

        h6, sc6 = basin_hopping(best_h, n0, target0,

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 30:

    remaining = budget_s - 30.0 - (time.time() - start_time)

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
