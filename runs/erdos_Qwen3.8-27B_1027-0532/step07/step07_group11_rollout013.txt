    top = corr[idx]
    c_max = np.max(top)
    shifted = np.clip(200.0 * (top - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / 200.0)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-13, use_topm=False, m=20):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            if use_topm:
                obj_func = _topm_obj
                args = (m,)
            else:
                obj_func = _smooth_obj
                args = (alpha,)
            res = minimize(obj_func, h_cur, args=args,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=150, n_shifts=3,
            max_no_improve=400, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 500.0
                    if np.random.random() < np.exp(-max(delta_obj, -100) / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.93, 1e-10)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=4, n_shifts=3, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps if 'max_sweeps' in dir() else 100000):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = shift_weights.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            s = b * block_size
            block_grads[b] = np.mean(grad[s:s + block_size])

        n_sel = min(8, n_blocks)
        sorted_blocks = np.argsort(block_grads)
        low_blocks = sorted_blocks[:n_sel]
        high_blocks = sorted_blocks[-n_sel:]

        best_trial = c5_val
        best_move = None

        for hb in high_blocks:
            for lb in low_blocks:
                if hb == lb:
                    continue
                hi_s = int(hb) * block_size
                hi_e = hi_s + block_size
                lo_s = int(lb) * block_size
                lo_e = lo_s + block_size

                max_move = min(float(np.sum(h[hi_s:hi_e])),
                               float(np.sum(1.0 - h[lo_s:lo_e])))
                if max_move < 1e-10:
                    continue

                total_delta = max_move / block_size
                g = np.zeros(len(corr))
                for p in range(block_size):
                    if h[hi_s + p] > 0 and h[lo_s + p] < 1:
                        g += -compute_single_change(h, n, hi_s + p) + compute_single_change(h, n, lo_s + p)
                if np.max(np.abs(g)) < 1e-15:
                    continue

                a_d = 0.0
                b_d = total_delta
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = float(np.max(g[active]))
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial - 1e-14:
                    h_try = h.copy()
                    h_try[hi_s:hi_e] -= delta_star
                    h_try[lo_s:lo_e] += delta_star
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial:
                        best_trial = sc
                        best_move = h_try.copy()

        if best_move is not None:
            h = best_move
        else:
            no_improve += 1

    return best_h, best_c5


def compute_single_change(h, n, p):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[neg_mask] -= h[p - shifts[neg_mask]]
    return change


def polyak_subgradient(h_init, n, target, time_limit, n_steps=50000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_avg = h.copy()
    no_improve = 0

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 20000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        g_norm_sq = float(np.sum(grad * grad))
        if g_norm_sq < 1e-20:
            break

        eta = (c5_val / dx) / g_norm_sq
        eta = min(eta, 0.05)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)

        h_avg = (step * h_avg + h_try) / (step + 1)
        h = h_try

    sc_avg = compute_c5(h_avg)
    if sc_avg < best_c5:
        best_c5 = sc_avg
        best_h = h_avg

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coarse_fine_cycle(h_in, n_fine, target_fine, time_limit):
    start = time.time()
    n_coarse = n_fine // 4
    target_coarse = n_coarse / 2.0

    h_coarse = np.array([np.mean(h_in[i * (n_fine // n_coarse):(i + 1) * (n_fine // n_coarse)])
                         for i in range(n_coarse)])
    h_coarse = project_feasible(h_coarse, target_coarse)

    if time_limit - (time.time() - start) > 60:
        h_coarse, _ = slsqp_round(h_coarse, n_coarse, target_coarse,
                                  [100, 500, 2000], 300, 1e-13)
    if time_limit - (time.time() - start) > 40:
        h_coarse, _ = pair_cd(h_coarse, n_coarse, target_coarse,
                              time_limit=min(30, time_limit - (time.time() - start)),
                              n_cand_start=10, n_cand_end=30, n_shifts=2,
                              max_no_improve=100)

    h_up = interpolate_to(h_coarse, n_fine, target_fine)
    return h_up


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: SLSQP warmup
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [50, 200, 1000], 800, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD
    if tleft() > 250:
        h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                              time_limit=min(200, tleft() * 0.6),
                              n_cand_start=30, n_cand_end=130, n_shifts=3,
                              max_no_improve=500, anneal_temp=0.015,
                              anneal_decay=0.9995)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement with top-m
    if tleft() > 150:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [2000, 8000], 1200, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [15, 25, 40], 800, 1e-13, use_topm=True, m=20)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Coarse-fine cycle
    if tleft() > 180:
        h_cf = coarse_fine_cycle(best_h, cur_n, cur_target, tleft())
        sc_cf = compute_c5(h_cf)
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cf
        if tleft() > 100:
            h_cf_pol, sc_cf_pol = slsqp_round(h_cf, cur_n, cur_target,
                                               [500, 2000, 8000], 600, 1e-13)
            if sc_cf_pol < best_score:
                best_score = sc_cf_pol
                best_h = h_cf_pol

    # Phase 5: More pair CD
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, cur_n, cur_target,
                                time_limit=min(120, tleft() * 0.6),
                                n_cand_start=50, n_cand_end=150, n_shifts=4,
                                max_no_improve=300, anneal_temp=0.005,
                                anneal_decay=0.999)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Polyak subgradient with averaging
    if tleft() > 100:
        h_pg, c5_pg = polyak_subgradient(best_h, cur_n, cur_target,
                                         time_limit=min(80, tleft() * 0.6))
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 7: High-alpha SLSQP
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [20000, 50000], 800, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Random restarts
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert.copy()
            if tleft() > 50:
                h_mr, sc_mr = slsqp_round(h_pert, cur_n, cur_target,
                                          [500, 3000, 15000], 500, 1e-13)
                if sc_mr < best_score:
                    best_score = sc_mr
                    best_h = h_mr

    # Phase 9: Final pair CD polish
    if tleft() > 50:
        h_final_cd, c5_final = pair_cd(best_h, cur_n, cur_target,
                                       time_limit=min(35, tleft() * 0.7),
                                       n_cand_start=80, n_cand_end=200, n_shifts=5,
                                       max_no_improve=150, anneal_temp=0.0,
                                       anneal_decay=1.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        h_final_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                      [50000, 100000], 500, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
