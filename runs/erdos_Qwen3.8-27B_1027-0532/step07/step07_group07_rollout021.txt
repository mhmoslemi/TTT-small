```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _slsqp_grad(x):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):

def pair_cd_improved(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_start=30, n_cand_end=100, top_k_eval=60,
                     n_shifts_weight=5, max_no_improve=800,
                     anneal_temp=0.0, anneal_decay=0.999):
    """Improved pair CD with multi-shift weighting and optional annealing."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = anneal_temp

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        temp *= anneal_decay

        # Multi-shift weighted gradient
        corr_sorted = np.sort(corr)[::-1]
        threshold = corr_sorted[min(n_shifts_weight, n_corr) - 1]
        active = np.where(corr >= threshold - 1e-12)[0]
        
        # Weight by proximity to max
        max_corr = corr[active]
        w = max_corr - max_corr.min() + 1e-10
        w = w / w.sum()

        grad = np.zeros(n)
        for idx in active:
            wi = w[np.where(active == idx)[0][0]]
            k = int(idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
        elif temp > 1e-6:
            # Annealing: try a random move
            if np.random.random() < 0.3:
                i = int(np.random.choice(high_set))
                j = int(np.random.choice(low_set))
                if i != j:
                    delta = np.random.random() * min(h[i], 1.0 - h[j]) * 0.5
                    if delta > 1e-6:
                        h[best_i if best_i >= 0 else i] -= delta
                        h[best_j if best_j >= 0 else j] += delta
                        h = np.clip(h, 0.0, 1.0)
                        h = project_feasible(h, target)
                        update_change_row(change, h, i, shifts, n)
                        update_change_row(change, h, j, shifts, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_init(h_base, n_fourier, n_target, target):
    """Create initial guess using low-order Fourier fit of h_base, upscaled."""
    n_src = len(h_base)
    x_src = np.linspace(0, 2, n_src)
    # Fit low-order Fourier series
    a0 = np.mean(h_base)
    coeffs = []
    for k in range(1, n_fourier + 1):
        ak = np.sum(h_base * np.cos(np.pi * k * x_src)) / (n_src / 2)
        bk = np.sum(h_base * np.sin(np.pi * k * x_src)) / (n_src / 2)
        coeffs.append((ak, bk))
    x_new = np.linspace(0, 2, n_target)
    h_new = np.full(n_target, a0)
    for k in range(1, n_fourier + 1):
        ak, bk = coeffs[k-1]
        h_new += ak * np.cos(np.pi * k * x_new) + bk * np.sin(np.pi * k * x_new)
    return project_feasible(h_new, target)

def directed_local_search(h_init, n, target, time_limit, max_sweeps=20000):
    """Directional local search using active-set subgradient."""

            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        # Find best pair (i,j) with i in top-grad, j in bottom-grad

            d_hi = min(h[i], 1.0-h[j])

                continue

            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
```

```python
def cutting_plane_solve(h_init, n, target, max_iter=100, tol=1e-6):
    """Solve using cutting plane / Benders decomposition."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    
    h = h_init.copy()
    cuts = []  # List of (g, c_k_star) tuples
    
    for it in range(max_iter):
        # Compute current c_k and add top cuts
        corr = np.correlate(h, 1-h, mode='full')
        c_max = float(np.max(corr))
        
        # Add top K shifts as cuts
        top_k = np.argsort(corr)[-min(50, n_corr):]
        for ki in top_k:
            k = int(ki) - (n-1)
            idxs = np.arange(n)
            g = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] = (1.0 - h[idxs[v1] - k]) * dx
            g[v2] = -h[idxs[v2] + k] * dx
            c_star = float(corr[ki])
            cuts.append((g, c_star, h.copy()))
        
        # Solve LP
        # Variables: h[0..n-1], t
        # Minimize t
        # Subject to: g·(h - h*) ≤ t - c_star for each cut (g, c_star, h_star)
        #             0 ≤ h ≤ 1
        #             sum(h) * dx = 1
        
        n_vars = n + 1  # h[0..n-1] + t
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0  # minimize t
        
        A_ub = []
        b_ub = []
        for g, c_star, h_star in cuts[-200:]:  # limit cuts
            row = np.zeros(n_vars)
            row[:n] = -g  # -g·h ≤ -(t - c_star + g·h_star)
            row[-1] = 1.0  # + t
            rhs = -(c_star + np.dot(g, h_star))
            A_ub.append(row)
            b_ub.append(rhs)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = dx  # sum(h) * dx = 1
        b_eq = [1.0]
        
        bounds_lin = [(0, 1)] * n + [(None, None)]
        
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds_lin, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        t_new = res.x[-1]
        
        # Check actual c_k at h_new
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, target)
        corr_new = np.correlate(h_new, 1-h_new, mode='full')
        actual_max = float(np.max(corr_new)) * dx
        
        if actual_max <= t_new + tol:
            # LP solution is feasible for original problem
            h = h_new
            break
        
        # Update h for next iteration
        h = h_new
        if actual_max < c_max - 1e-8:
            c_max = actual_max
        else:
            # No improvement, stop
            break
    
    return h, compute_c5(h)
```

Let me also add one more idea: **projected subgradient descent with adaptive step size**. This is simpler than SLSQP and might be more robust:

At each iteration:
1. Compute c_k for all k
2. Find the active set A (shifts with c_k > c_max - ε)
3. Compute the subgradient: g = (1/|A|) Σ_{k∈A} grad(c_k)
4. Project -g onto the tangent space of the constraint set (i.e., remove the mean to satisfy Σh = const, and project onto [0,1])
5. Take a step: h ← h - α * projected_g
6. Project h onto [0,1] and adjust to satisfy Σh = target

```python
def subgradient_search(h_init, n, target, time_limit, max_iter=100000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        c_max = float(np.max(corr)) * dx
        corr_raw = np.max(corr)
        
        if c_max < best_c5 - 1e-15:
            best_c5 = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Active set: shifts within 5% of max
        threshold = corr_raw * 0.95
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for ki in active:
            k = int(ki) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        
        # Project onto tangent space (remove mean for equality constraint)
        grad -= np.mean(grad)
        
        # Adaptive step size
        alpha = 1.0 / (1.0 + 0.01 * it)
        alpha = min(alpha, 0.1)
        
        # Try multiple step sizes
        best_trial = c_max
        best_alpha = 0
        for a_mult in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a = alpha * a_mult
            h_try = h - a * grad
            h_try = np.clip(h_try, 0, 1)
            h_try = project_feasible(h_try, target)
            trial = compute_c5(h_try)
            if trial < best_trial:
                best_trial = trial
                best_alpha = a
        
        if best_alpha > 0:
            h = h - best_alpha * grad
            h = np.clip(h, 0, 1)
            h = project_feasible(h, target)
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _slsqp_grad(x):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _cons_eq(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=30, n_cand_end=120, top_k_eval=80,
               n_active=5, max_no_improve=1000, use_score_aware=True):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        if use_score_aware:
            corr_sorted = np.sort(corr)[::-1]
            n_act = min(n_active, n_corr)
            threshold = corr_sorted[n_act - 1]
            active_idx = np.where(corr >= threshold - 1e-12)[0]
            w = (corr[active_idx] - corr[active_idx].min() + 1e-8)
            w = w / w.sum()
        else:
            active_idx = np.argsort(corr)[-n_active:]
            w = np.ones(len(active_idx)) / len(active_idx)

        grad = np.zeros(n)
        for ai, wv in zip(active_idx, w):
            k = int(ai) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wv * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wv * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5
