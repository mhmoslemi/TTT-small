def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    # Use smooth max approximation with SLSQP
    cons_fun = lambda x: np.sum(x) - target
    cons_jac = lambda x: np.ones(n)
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: _smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=4, n_cand_start=20, n_cand_end=60,
                max_no_improve=300, bisection_iters=12,
                n_top_pairs=200, seed_cd=42):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(seed_cd)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        high_set = sorted_idx[-n_cand:]

                    jv = int(low_set[jj])

        else:

            per_h = max(1, n_top_pairs // n_h)

                    if iv != jv:

                    if len(pair_list) >= n_top_pairs:
                        break
                if len(pair_list) >= n_top_pairs:
                    break
            while len(pair_list) < n_top_pairs:
                ii = rng.randint(n_h)
                jj = rng.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
                if len(pair_list) >= n_top_pairs * 2:
                    break

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
        else:
            if consecutive_no_improve > 80 and sweep % 40 == 0:
                pert_scale = 0.015
                mask = (h > 0.05) & (h < 0.95)

                    chosen = rng.choice(pert_idx, size=n_pert, replace=False)
                    h[chosen] += rng.randn(n_pert) * pert_scale
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    change = compute_change_matrix(h, n)

    return best_h, best_c5
