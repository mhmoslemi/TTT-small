```python
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _obj_wrapper(x):
    return _smooth_lse(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        _g_alpha = param
        try:
            res = minimize(_obj_wrapper, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
```

```python
def lp_step(h, n, target, max_step=0.1, n_active=None):
    """Take one LP-based step to reduce the max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    
    # Compute change matrix (gradient of corr wrt h)
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    # LP: minimize t
    # subject to: change.T @ d - t <= -corr  (for each shift)
    #             sum(d) = 0
    #             -h[i] <= d[i] <= 1-h[i]
    
    # Variables: d[0..n-1], t (total n+1 variables)
    # c = [0, ..., 0, 1] (minimize t)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ [d, t] <= b
    # For each shift k: sum_i change[i,k] * d[i] - t <= -corr[k]
    # => [change[:,k], -1] @ [d, t] <= -corr[k]
    
    A = np.hstack([change.T, -np.ones((n_corr, 1))])
    b = -corr
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -h[i] <= d[i] <= 1-h[i], t >= 0 (or some lower bound)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            t_val = res.x[n]
            # Scale d if it's too large
            d_norm = np.max(np.abs(d))
            if d_norm > max_step:
                d = d * (max_step / d_norm)
            
            h_new = h + d
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            cur_sc = compute_c5(h)
            if sc < cur_sc:
                return h_new, sc, d
            else:
                # Try with smaller step
                d = d * 0.5
                h_new = h + d
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_sc:
                    return h_new, sc, d
        return h, compute_c5(h), None
    except Exception:
        return h, compute_c5(h), None
```

- This means: for all k with b_k > 0: d <= (t - a_k) / b_k

  For all k with b_k = 0: need a_k <= t

  AND max_k { a_k : b_k = 0 } <= t

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _obj_wrapper(x):
    return _smooth_lse(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        _g_alpha = param
        try:
            res = minimize(_obj_wrapper, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def lp_single_step(h, n, target, max_step):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([change.T, -np.ones((n_corr, 1))])
    b_ub = -corr.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lb = -h.copy()
    ub = 1.0 - h.copy()
    bounds_list = []
    for i in range(n):
        bounds_list.append((lb[i], ub[i]))
    bounds_list.append((0.0, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            d_norm = np.max(np.abs(d))
            if d_norm > max_step and d_norm > 1e-15:
                d = d * (max_step / d_norm)
            if np.max(np.abs(d)) < 1e-15:
                return h, compute_c5(h), None
            h_new = h + d
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            cur_sc = compute_c5(h)
            if sc < cur_sc - 1e-15:
                return h_new, sc, d
    except Exception:
        pass
    return h, compute_c5(h), None

def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=30000, n_cand_start=15, n_cand_end=60,
            max_no_improve=400, n_top_shifts=1, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def triple_cd(h_init, n, target, time_limit, start_time,

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

        grad[v2] -= h[idxs[v2] + k_star]

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        best_improve = 1e-15
        best_trip = None

        for ii in range(min(n_h, 8)):
            i = int(high_set[ii])
            for jj in range(min(n_l, 8)):
                j = int(low_set[jj])
                for kk in range(min(n_l, 4)):
                    k = int(low_set[kk])
                    if i == j or i == k or j == k:
                        continue
                    d1_lo = max(-h[i], -(1.0 - h[k]) * 0.5)
                    d1_hi = min(h[i], 1.0 - h[j])

                        d2 = 0.5 * d2

                    g_vec = -d1 * change[i] - d2 * change[j] + (d1 + d2) * change[k]

            h[k] += (d1 + d2)
            h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_sg, sc_sg = subgradient_descent(best_h, n0, target0,
                                           start_time + min(50, remaining - 5), start_time,
                                           max_iters=3000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 40:
        h_lp4, sc_lp4 = lp_local_search(best_h, n0, target0,
                                         start_time + min(40, remaining - 5), start_time,
                                         max_iters=100, init_step=0.02)
        if sc_lp4 < best_score:
            best_score = sc_lp4
            best_h = h_lp4

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 50:
        h_fin, sc_fin = pair_cd(best_h, n0, target0,
                                 start_time + min(50, remaining - 5), start_time,
                                 max_sweeps=20000, n_cand_start=30, n_cand_end=80,
                                 max_no_improve=150, n_top_shifts=1, bisection_iters=20)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 30:
        h_sg2, sc_sg2 = subgradient_descent(best_h, n0, target0,
                                             start_time + min(30, remaining - 5), start_time,
                                             max_iters=2000)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 30:
        h_fin2, sc_fin2 = slsqp_optimize(best_h, n0, target0,
                                          [20000, 60000],
                                          600, start_time + min(30, remaining - 5), start_time)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
