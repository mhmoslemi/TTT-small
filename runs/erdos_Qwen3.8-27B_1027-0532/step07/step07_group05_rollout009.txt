def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    # Smooth approximation to max(corr) using log-sum-exp
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

- If i = j+k: the term (1-h[i]) in the h[j] term... wait, this is the same as i = j+k, i.e., k = i-j = -(j-i)
  The term h[j]*(1-h[i]) becomes h[j]*(1-(h[i]-δ)) = h[j]*(1-h[i]) + δ*h[j]
  Extra: δ*h[j]
- If j = i+k: the term h[i]*(1-h[j]) becomes (h[i]-δ)*(1-(h[j]+δ)) as before

So the second-order correction is non-zero only when k = ±(j-i), and it's O(δ * max(h[i], h[j])). For small δ, this is negligible. The bisection in the ECD should handle this correctly because it evaluates the actual max (using the approximate corr + delta * g) and the approximation is good for small δ.

For larger δ, the approximation might be off. But the bisection finds the point where the (approximate) max is minimized, and since the approximation is close for the relevant range of δ, it should still give a good result.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        k_pos = idxs - p
        valid = (k_pos >= 0) & (k_pos < n)
        change[p, mid + k_pos[valid]] += (1.0 - h[idxs[valid]])
        k_neg = idxs + p
        valid2 = (k_neg >= 0) & (k_neg < n)
        change[p, mid - p + np.arange(n)[valid2]] -= h[k_neg[valid2]]
    return change

def compute_change_matrix_v2(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def enhanced_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_active=3, n_cand_start=15, n_cand_end=60,
                max_no_improve=500, bisection_iters=12, max_pairs=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix_v2(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        n_act = min(n_active, n_corr)
        top_idx = np.argsort(corr)[-n_act:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pairs = [(int(high_set[ii]), int(low_set[jj]))
                     for ii in range(n_h) for jj in range(n_l)
                     if int(high_set[ii]) != int(low_set[jj])]
        else:
            pairs = []
            for frac in np.linspace(0.0, 1.0, min(max_pairs, 20)):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                pairs.append((int(high_set[hi]), int(low_set[lo])))
            while len(pairs) < max_pairs:
                ii = int(rng.randint(n_h))
                jj = int(rng.randint(n_l))
                pairs.append((int(high_set[ii]), int(low_set[jj])))

        for (i, j) in pairs:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-14:
                continue

            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-14:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha=300):
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_s = np.exp(shifted)
        S = np.sum(exp_s)
        val = c_max + np.log(S) / alpha
        weights = exp_s / S / alpha
        n_corr = 2 * n - 1
        mid_local = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        for k in range(n_corr):
            wx = weights[k]
            if wx < 1e-16:
                continue
            lag = k - mid_local
            v1 = (idxs - lag >= 0) & (idxs - lag < n)
            v2 = (idxs + lag >= 0) & (idxs + lag < n)
            if v1.any():
                grad[v1] += wx * (1.0 - x[idxs[v1] - lag])
            if v2.any():
                grad[v2] -= wx * x[idxs[v2] + lag]
        return val, grad

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        try:
            res = minimize(obj_and_grad, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur
