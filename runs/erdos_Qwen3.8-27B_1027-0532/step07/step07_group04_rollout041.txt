import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]
    return change

def topm_corr_grad(corr, h, n, top_m):
    """Compute weighted gradient from top m correlation values."""
    mid = n - 1
    idxs = np.arange(n)
    
    # Get top m indices
    top_idx = np.argsort(corr)[-top_m:]
    c_max = np.max(corr)
    
    # Softmax weights over top m (higher temp = more uniform)
    beta = 10.0
    w = np.exp(-beta * (c_max - corr[top_idx]))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(top_m) / top_m
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in enumerate(top_idx):
        lag = int(k) - mid
        if lag >= 0:
            ns, ne = 0, n - lag
        else:
            ns, ne = -lag, n
        vn = np.arange(ns, ne)
        grad[vn] += w[wi] * (1.0 - h[vn + lag])
        jr = vn + lag
        vm = (jr >= 0) & (jr < n)
        grad[jr[vm]] -= w[wi] * h[vn[vm]]
    
    grad -= np.mean(grad)
    return grad, top_idx, w

def slsqp_round(h_start, n, target, alphas, maxiter, ftol=1e-14,
                time_limit=None, start_time=None):
    dx = 2.0 / n
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                ns, ne = -lag, n
            vn = np.arange(ns, ne)
            grad[vn] += w[k] * (1.0 - y[vn + lag])
            jr = vn + lag
            vm = (jr >= 0) & (jr < n)
            grad[jr[vm]] -= w[k] * y[vn[vm]]
        
        grad *= dx / alpha
        grad -= np.mean(grad)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * it)
        
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        
        h_prev = h
        h = h_new
        t_curr = t_next
        
        sc = compute_c5(h)
        if sc < best_c5:
            best_c5 = sc
            best_h = h.copy()
    
    return best_h, best_c5
