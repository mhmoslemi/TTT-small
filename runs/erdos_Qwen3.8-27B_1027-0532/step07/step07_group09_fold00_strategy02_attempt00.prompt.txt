$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

def project_feasible(h, target_sum):

def _smooth_obj(x, alpha):

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_change_matrix(h, n):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    def project_feasible(x: np.ndarray, n: int) -> np.ndarray:
        """Maintains integral and bounds"""
        x = np.clip(x, 0, 1)
        proj_sum = np.sum(x)
        desired_sum = n / 2
        imbalance = (desired_sum - proj_sum) / n
        x = np.minimum(x, 1 - imbalance).clip(0)
        # Additional distribution via diffusion matrix
        return x + (desired_sum/proj_sum - 1) * np.random.rand(x.shape[0]) * 0.01

    def localized_cd(f: Callable, grad: Callable, x0, n_trials=10):
        """Adaptive low-frequency coordinate descent"""
        x_candidate = np.array(x0)
        x_temp = np.array(x0)
        for _ in range(n_trials):
            if random() < 0.2:  # Local perturbation
                var_idx = np.argmax(np.abs(grad(x_temp))) if grad else random.randint(0, len(x0)-1)
                x_temp[var_idx] += sign_flip(random())
            corr = f(x_temp)
            if corr < f(x_candidate):
                x_candidate = x_temp.copy()
            # Simulated anneal
            x_temp -= random.normal(size=len(x_temp)) * 0.001
            x_temp = project_feasible(x_temp, n)
        return x_candidate
