    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

        if abs(diff) < 1e-14:

        if diff > 0:

        else:

        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)

        weights /= wsum

    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return obj, grad

def slsqp_with_grad(h_start, n, target, dx, alphas, maxiter_per_alpha):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        def obj(x):
            return smooth_obj_grad(x, alpha, dx)[0]

            return smooth_obj_grad(x, alpha, dx)[1]

            return np.sum(x) - target

            return np.ones(n)

Actually, a better approach for this specific problem: **Frank-Wolfe (conditional gradient) method**.

The feasible set is the polytope P = {h : 0 ≤ h[i] ≤ 1, sum(h) = n/2}.
The Frank-Wolfe method:
1. Compute gradient g of the smooth approximation
2. Find s = argmin_{v ∈ P} g·v (the "linear oracle")
3. Update: h ← h + step * (s - h)

The linear oracle for our constraint set:
minimize g·v subject to 0 ≤ v[i] ≤ 1, sum(v) = n/2

This is: set v[i] = 0 if g[i] > threshold, v[i] = 1 if g[i] < threshold, and one index at threshold to make the sum work out. The threshold is the n/2-th smallest value of g.

This is much more efficient than SLSQP for each iteration!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    shifted = np.clip(alpha * (corr - corr_max), -500.0, 0.0)
    wsum = np.sum(np.exp(shifted))
    return float((corr_max + np.log(wsum) / alpha) * dx)

def smooth_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    shifted = np.clip(alpha * (corr - corr_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    n_corr = 2 * n - 1
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return grad

def slsqp_with_grad(h_start, n, target, dx, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            return smooth_obj(x, alpha, dx)
        
        def jac(x):
            return smooth_grad(x, alpha, dx)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, h_cur, jac=jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return float((cmax + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return grad

def slsqp_opt(h_start, n, target, dx, alphas, maxiter, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x):
            return smooth_obj(x, alpha, dx)
        def jac(x):
            return smooth_grad(x, alpha, dx)
        def cf(x):
            return np.sum(x) - target
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(obj, h_cur, jac=jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=400,
            n_binsearch=20, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

        grad[v2] -= h[idxs[v2] + k_star]

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_j = -1

                b_d = delta_hi

                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random := int(stochastic_frac * len(high_set) * len(low_set)):
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)

                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            pos_mask = (best_i + shifts >= 0) & (best_i + shifts < n)
            neg_mask = (best_i - shifts >= 0) & (best_i - shifts < n)
            change[best_i, pos_mask] += (1.0 - h[best_i + shifts[pos_mask]])
            change[best_i, neg_mask] -= h[best_i - shifts[neg_mask]]
            change[best_j, :] = 0.0
            pos_mask = (best_j + shifts >= 0) & (best_j + shifts < n)
            neg_mask = (best_j - shifts >= 0) & (best_j - shifts < n)
            change[best_j, pos_mask] += (1.0 - h[best_j + shifts[pos_mask]])
            change[best_j, neg_mask] -= h[best_j - shifts[neg_mask]]
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def soften_block_cd(h_init, n, target, time_limit, block_size=10, seed=42):
    """Optimize block by block using SLSQP on each block."""
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    n_blocks = (n + block_size - 1) // block_size
    
    for epoch in range(10):
        if time.time() - start > time_limit:
            break
        block_order = list(range(n_blocks))
        rng.shuffle(block_order)
        for bi in block_order:
            if time.time() - start > time_limit:
                break
            start_idx = bi * block_size
            end_idx = min(start_idx + block_size, n)
            bs = end_idx - start_idx
            if bs < 2:
                continue
            block_sum = np.sum(h[start_idx:end_idx])
            # Optimize the block: minimize smooth max of the whole corr
            # but only vary the block
            alpha = 500.0
            def block_obj(x):
                h_trial = h.copy()
                h_trial[start_idx:end_idx] = x
                h_trial = project_feasible(h_trial, target)
                return smooth_obj(h_trial, alpha, dx)
            
            def block_jac(x):
                h_trial = h.copy()
                h_trial[start_idx:end_idx] = x
                h_trial = project_feasible(h_trial, target)
                return smooth_grad(h_trial, alpha, dx)[start_idx:end_idx]
            
            bounds = [(0.0, 1.0)] * bs
            try:
                res = minimize(block_obj, h[start_idx:end_idx], jac=block_jac,
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 50, 'ftol': 1e-14})
                h_new = project_feasible(res.x, block_sum)  # Keep same block sum? No, we need total sum
                # Actually we need to maintain total sum = target
                # Just project the full h
                h_trial = h.copy()
                h_trial[start_idx:end_idx] = res.x
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_score:
                    best_score = sc
                    best_h = h_trial.copy()
                    h = h_trial
            except Exception:
                pass
    return best_h, best_score
