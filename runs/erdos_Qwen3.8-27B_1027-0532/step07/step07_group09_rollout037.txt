            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        n_rand = max(3, ncand // 6)
        rand_cands = rng.choice(n, size=n_rand, replace=False)

        all_cands = np.unique(np.concatenate([low_set, high_set, rand_cands]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            G[r, :] = change[int(all_cands[r]), :]

        n_h = len(high_set)
        n_l = len(low_set)

        best_trial = c5_raw
        best_i = -1
        best_j = -1
        best_delta = 0.0

        if n_h * n_l > 4000:
            perm_i = rng.permutation(n_h)
            perm_j = rng.permutation(n_l)
            check_count = 0
            for ii in perm_i:
                for jj in perm_j:
                    if check_count >= 4000:
                        break
                    i = int(high_set[ii])
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    hr = cand_map[i]
                    lr = cand_map[j]
                    h_i = h[i]
                    h_j = h[j]
                    delta_max = min(h_i, 1.0 - h_j)
                    delta_min = max(h_i - 1.0, -h_j)
                    if delta_max - delta_min < 1e-12:
                        continue
                    g = G[lr] - G[hr]
                    a_d = delta_min + 1e-12
                    b_d = delta_max - 1e-12
                    for _ in range(8):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                    check_count += 1
        else:
            for ii in range(n_h):
                i = int(high_set[ii])
                hr = cand_map[i]
                h_i = h[i]
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    lr = cand_map[j]
                    h_j = h[j]
                    delta_max = min(h_i, 1.0 - h_j)
                    delta_min = max(h_i - 1.0, -h_j)
                    if delta_max - delta_min < 1e-12:
                        continue
                    g = G[lr] - G[hr]
                    a_d = delta_min + 1e-12
                    b_d = delta_max - 1e-12
                    for _ in range(8):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def spectral_perturb(h, n, scale, rng):
    ft = np.fft.rfft(h)
    n_freq = len(ft)
    n_smooth = max(3, int(n // 20))
    mask = np.zeros(n_freq)
    mask[:n_smooth] = 1.0
    noise = (rng.randn(n_freq) + 1j * rng.randn(n_freq)) * scale * mask
    ft_pert = ft + noise
    h_pert = np.fft.irfft(ft_pert, n)
    h_pert = project_feasible(h_pert, n / 2.0)
    return h_pert


def block_cd(h_init, n, target, time_limit, max_steps=15000,
             block_sizes=(2, 3, 4, 5, 6, 8), rng_seed=456):
    start = time.time()
    rng = np.random.RandomState(rng_seed)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = compute_change_matrix(h, n)
    n_pos_max = n - max(block_sizes) + 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_raw = float(np.max(corr))

        if c5_raw * dx < best_c5 - 1e-15:
            best_c5 = c5_raw * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        bs = block_sizes[step % len(block_sizes)]
        n_pos = n - bs + 1
        if n_pos <= 1:
            continue

        best_block_move = (c5_raw, -1, -1, 0.0)

        n_trials = min(30, n_pos)
        trial_starts = rng.choice(n_pos, size=n_trials, replace=False)

        for src_start in trial_starts:
            block_grad = grad[src_start:src_start + bs].sum()

            best_target = -1
            best_target_grad = -np.inf
            for dst_start in trial_starts:
                if dst_start == src_start:
                    continue
                if dst_start + bs > n:
                    continue
                if src_start + bs > dst_start or src_start >= dst_start + bs:
                    overlap = min(src_start + bs, dst_start + bs) - max(src_start, dst_start)
                    if overlap > 0:
                        continue
                bg = grad[dst_start:dst_start + bs].sum()
                if bg > best_target_grad:
                    best_target_grad = bg
                    best_target = dst_start

            if best_target < 0:
                continue

            if block_grad > best_target_grad:
                s_start, d_start = src_start, best_target
            else:
                s_start, d_start = best_target, src_start

            src_block = h[s_start:s_start + bs].copy()
            dst_block = h[d_start:d_start + bs].copy()

            delta_max = min(src_block.max(), 1.0 - dst_block.min())
            delta_min = max(-src_block.min(), dst_block.min() - 1.0)
            if delta_max - delta_min < 1e-12:
                continue

            g = np.zeros(n_corr)
            for t in range(bs):
                g += change[d_start + t, :] - change[s_start + t, :]

            a_d = delta_min + 1e-12
            b_d = delta_max - 1e-12
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g)
            if approx_max < best_block_move[0]:
                best_block_move = (approx_max, s_start, d_start, delta_star)

        if best_block_move[1] >= 0:
            s_start, d_start, delta_star = best_block_move[1], best_block_move[2], best_block_move[3]
            h[s_start:s_start + bs] -= delta_star
            h[d_start:d_start + bs] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(bs):
                update_change_row(change, s_start + t, h, n, shifts)
                update_change_row(change, d_start + t, h, n, shifts)

    return best_h, best_c5


def subgradient_cd(h_init, n, target, time_limit, max_iter=20000, lr_start=0.003):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-10, 1e-7 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(1, len(active))
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.95 * progress) ** 2

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val * dx - 1e-15:
            h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0,
                            [200, 1000, 5000],
                            500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800

        best_h = h1
        n0 = n1
        target0 = target1

    if tleft() > 400:
        h_cd, c5_cd = enhanced_pair_cd(h1, n1, target1,
                                       time_limit=min(300, tleft() * 0.5),
                                       max_sweeps=30000,
                                       n_cand_start=25, n_cand_end=65,
                                       max_no_improve=400,
                                       temp_scale=0.02,
                                       rng_seed=123)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

            target0 = target1

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0,
                            [500, 2000, 10000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_bm, c5_bm = block_cd(best_h, n0, target0,

                               max_steps=12000,
                               block_sizes=(2, 3, 4, 5, 6, 8),
                               rng_seed=456)
        if c5_bm < best_score:
            best_score = c5_bm
            best_h = h_bm

    if tleft() > 100:
        h_cd2, c5_cd2 = enhanced_pair_cd(best_h, n0, target0,
                                         time_limit=min(80, tleft() * 0.5),
                                         max_sweeps=20000,
                                         n_cand_start=30, n_cand_end=70,
                                         max_no_improve=300,
                                         temp_scale=0.015,
                                         rng_seed=789)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_cd3, c5_cd3 = enhanced_pair_cd(h2, n2, target2,
                                             time_limit=min(150, tleft() * 0.55),
                                             max_sweeps=20000,
                                             n_cand_start=20, n_cand_end=55,
                                             max_no_improve=350,
                                             temp_scale=0.02,

            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                n0 = n2
                target0 = target2
        if tleft() > 80:
            h_pol3 = slsqp_round(best_h, n0, target0,

            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 120:

            scale = 0.03 + 0.02 * attempt

    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
