        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - tolerance:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 30 == 0:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(8, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                lj_val = h[j]
                delta_lo = max(hi_val - 1.0, -lj_val)
                delta_hi = min(hi_val, 1.0 - lj_val)
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_max_abs = np.max(np.abs(g))
                if g_max_abs < 1e-15:
                    continue
                if g[k_star_idx] > 0.2 * g_max_abs:
                    continue

                a_d = delta_lo + 1e-10
                b_d = delta_hi - 1e-10
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def active_set_slsqp(h_init, n, target, time_limit, top_k=3):
    start = time.time()
    global _g_n, _g_target
    _g_n = n
    _g_target = target

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1

    alphas = [2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.argsort(corr)[-top_k:]

        def focused_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            active_vals = c[active]
            c_max = np.max(c)
            exp_vals = np.exp((active_vals - c_max) * alpha)
            return float(c_max + np.log(np.sum(exp_vals)) / alpha)

        def focused_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(c)
            exp_vals = np.exp((c[active] - c_max) * alpha)
            probs = exp_vals / np.sum(exp_vals)
            g = np.zeros(n)
            idxs = np.arange(n)
            for idx, p in zip(active, probs):
                k = int(idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += p * (1.0 - x[idxs[v1] - k])
                g[v2] -= p * x[idxs[v2] + k]
            return g / alpha

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

        maxiter = min(2000, max(200, int((time_limit - (time.time() - start)) * 3)))
        try:
            res = minimize(focused_obj, h, jac=focused_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h = h_new
        except Exception:
            pass

    return best_h, best_score


def gurobi_qp_polish(h_init, n, target, time_limit, top_k=4):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)

    try:
        import gurobipy as grb
    except ImportError:
        return best_h, best_score

    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(corr)[-top_k:]

    Q = np.zeros((n, n))
    for a in active:
        k = int(a) - mid
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                Q[i, j] -= 0.5 / top_k
                Q[j, i] -= 0.5 / top_k

    model = grb.Model("erdos_qp")
    model.setParam(grb.GRB.Param.TimeLimit, min(time_limit - (time.time() - start), 120))
    model.setParam(grb.GRB.Param.NonConvex, 2)
    model.setParam(grb.GRB.Param.OutputFlag, 0)

    x = model.addVars(n, lb=0.0, ub=1.0, name="h")
    model.addConstr(grb.quicksum(x[i] for i in range(n)) == target)
    model.setObjective(grb.quicksum(Q[i, j] * x[i] * x[j] for i in range(n) for j in range(n) if Q[i, j] != 0),
                       grb.GRB.MINIMIZE)
    model.optimize()

    if model.status == grb.GRB.OPTIMAL or model.status == grb.GRB.OPTIMAL_INFEASIBLE:
        h_gurobi = np.array([x[i].X for i in range(n)])
        h_gurobi = project_feasible(h_gurobi, target)
        sc = compute_c5(h_gurobi)
        if sc < best_score:
            best_score = sc
            best_h = h_gurobi

    return best_h, best_score


def momentum_subgradient(h_init, n, target, time_limit, alpha0=0.1,
                         momentum0=0.5, decay=0.999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    mu = momentum0
    stale = 0

    for step in range(100000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 300:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.1 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        if len(active) > 10:
            active = np.argsort(corr)[-10:]

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        v = mu * v - alpha * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        if compute_c5(h_new) < c5_val + alpha * 0.1:
            h = h_new
            alpha *= 0.99
            mu = max(0.3, mu * decay)
        else:
            alpha *= 0.7
            mu = max(0.3, mu * 0.9)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        deadline = time.time() + min(40, tleft() * 0.3)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000, 5000, 20000],
                                800, 1e-14, deadline=deadline, top_k=5)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = pair_cd_improved(best_h, n0, target0,
                                       time_limit=min(150, tleft() * 0.45),
                                       max_sweeps=40000,
                                       n_cand_start=20, n_cand_end=100,
                                       max_no_improve=600,
                                       n_active_shifts=5,
                                       reheat_interval=100, reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_act, sc_act = active_set_slsqp(best_h, n0, target0,
                                         time_limit=min(60, tleft() * 0.4), top_k=4)
        if sc_act < best_score:
            best_score = sc_act
            best_h = h_act

    if tleft() > 100:
        h_gurobi, sc_gurobi = gurobi_qp_polish(best_h, n0, target0,
                                               time_limit=min(80, tleft() * 0.5), top_k=4)
        if sc_gurobi < best_score:
            best_score = sc_gurobi
            best_h = h_gurobi

    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 150:
            h_cd1, c5_cd1 = pair_cd_improved(h1, n1, target1,
                                             time_limit=min(100, tleft() * 0.45),
                                             max_sweeps=25000,
                                             n_cand_start=15, n_cand_end=70,
                                             max_no_improve=400,
                                             n_active_shifts=4,
                                             reheat_interval=80, reheat_scale=0.03)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if current_n == n1 and tleft() > 80:
            deadline = time.time() + min(60, tleft() * 0.4)
            h_pol1, sc1 = slsqp_round(best_h, n1, target1,
                                      [2000, 10000, 50000],
                                      1000, 1e-14, deadline=deadline, top_k=5)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    if tleft() > 100:
        h_cd2, c5_cd2 = pair_cd_improved(best_h, current_n, current_target,
                                         time_limit=min(80, tleft() * 0.5),
                                         max_sweeps=20000,
                                         n_cand_start=25, n_cand_end=110,
                                         max_no_improve=400,
                                         n_active_shifts=6,
                                         reheat_interval=60, reheat_scale=0.02)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 60:
        h_act2, sc_act2 = active_set_slsqp(best_h, current_n, current_target,
                                           time_limit=min(45, tleft() * 0.5), top_k=3)
        if sc_act2 < best_score:
            best_score = sc_act2
            best_h = h_act2

    if tleft() > 80:
        h_mom, c5_mom = momentum_subgradient(best_h, current_n, current_target,
                                             time_limit=min(50, tleft() * 0.5),
                                             alpha0=0.05, momentum0=0.6, decay=0.9995)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [10000, 50000, 200000],
                                        1500, 1e-15, deadline=deadline, top_k=4)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
