import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _golden_search_1d(f, a, b, tol=1e-6, max_iter=50):
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(max_iter):
        if b - a < tol:
            break
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    return (a + b) / 2

    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',

                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _smooth_obj(h, alpha):
            return res.x
    except Exception:
        pass
    return h

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum_inplace(h, target_sum):
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return
    if diff > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
    else:
        order = np.argsort(h)
        for idx in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub

def _fix_sum(h, target_sum):
    h_new = h.copy()
    _fix_sum_inplace(h_new, target_sum)
    return h_new

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _golden_search_1d(f, a, b, tol=1e-8, max_iter=30):
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

    h = h.copy()

            if time_limit is not None and time.time() - t0 > time_limit:
                break
            base = h.copy()
            old_val = base[i]

                ht = base.copy()
                ht[i] = t
                _fix_sum_inplace(ht, target_sum)
                return _c5_of(ht)

            f_0 = f(0.0)
            f_1 = f(1.0)
            f_cur = f(old_val)

            best_t = old_val
            best_f = f_cur
            if f_0 < best_f:
                best_t, best_f = 0.0, f_0
            if f_1 < best_f:
                best_t, best_f = 1.0, f_1

            if 0.0 < best_t < 1.0:
                t_opt = _golden_search_1d(f, 0.0, 1.0, tol=1e-8, max_iter=22)
                f_opt = f(t_opt)
                if f_opt < best_f - 1e-15:
                    best_t = t_opt
            elif best_t <= 0.0:

def _slsqp_refine(h, n, target_sum, alpha, maxiter=200, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    h_start = _fix_sum(h, target_sum)
    c5_start = _c5_of(h_start)
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        x_new = _fix_sum(res.x, target_sum)
        if _c5_of(x_new) < c5_start:
            return x_new
    except Exception:
        pass
    return h_start
