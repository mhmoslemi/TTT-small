import numpy as np
from scipy.optimize import minimize
import time

def project_to_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    values = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(values)
    x = (values - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    values = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(values)
    x = (values - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * omh[:n - s]
                grad[:n - s] -= wk * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * omh[r:]
                grad[r:] -= wk * h[:n - r]
    return grad * dx

def power_mean_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    R = np.maximum(R, 1e-12)
    return dx * np.sum(R ** p) ** (1.0 / p)

def power_mean_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    R_safe = np.maximum(R, 1e-12)
    weights = R_safe ** (p - 1)
    sum_Rp = np.sum(R_safe ** p)
    factor = dx * sum_Rp ** (1.0 / p - 1.0)
    omh = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * omh[:n - s]
                grad[:n - s] -= wk * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * omh[r:]
                grad[r:] -= wk * h[:n - r]
    return factor * grad

def constraint_fun(h):
    return np.sum(h) - len(h) / 2.0

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_smooth(h_init, eps, maxiter, target_sum):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    h0 = project_to_constraint(h_init, target_sum)
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint_fun,
                                       'jac': constraint_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_to_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_power(h_init, p, maxiter, target_sum):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    h0 = project_to_constraint(h_init, target_sum)
    try:
        result = minimize(power_mean_obj, h0, args=(p,),
                          jac=power_mean_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint_fun,
                                       'jac': constraint_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_to_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def compute_coeff_A(h, i):
    n = len(h)
    A = np.zeros(2 * n - 1)
    for s in range(-n + 1, n):
        k = n - 1 + s
        coef = 0.0
        if s >= 0:
            if s <= i <= n - 1:
                coef += (1.0 - h[i - s])
            if 0 <= i <= n - 1 - s:
                coef -= h[i + s]
        else:
            r = -s
            if 0 <= i <= n - 1 - r:
                coef += (1.0 - h[i + r])
            if r <= i <= n - 1:
                coef -= h[i - r]
        A[k] = coef
    return A

def coordinate_descent_sweep(h, target_sum, n_grid=31):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    ts = np.linspace(0.0, 1.0, n_grid)
    for i in range(n):
        A = compute_coeff_A(h, i)
        old_val = h[i]
        B = corr - old_val * A
        best_t = old_val
        best_obj = float(np.max(B + old_val * A))
        for t in ts:
            val = float(np.max(B + t * A))
            if val < best_obj:
                best_obj = val
                best_t = t
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        ts_fine = np.linspace(lo, hi, 21)
        for t in ts_fine:
            val = float(np.max(B + t * A))
            if val < best_obj:
                best_obj = val
                best_t = t
        delta = best_t - old_val
        if abs(delta) > 1e-10:
            h[i] = best_t
            corr += delta * A
    h = project_to_constraint(h, target_sum)
    return h

def coordinate_descent_multi(h_init, n_sweeps, target_sum):
    h = project_to_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_sweeps):

        c5 = compute_c5(h)

            best_c5 = c5
    return best_h, best_c5

def grad_descent_power(h_init, p, n_steps, target_sum, base_lr=0.02):
    h = project_to_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h.copy()

    for step in range(n_steps):
        grad = power_mean_grad(h, p)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        lr = base_lr * (0.97 ** (step // 100)) / gnorm
        h_new = project_to_constraint(h - lr * grad, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5
