    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def project_to_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_func(h, alpha):
    return smooth_max_full(h, alpha)

def eq_constraint(h, n):
    return np.sum(h) - n / 2.0

def eq_jacobian(h, n):
    return np.ones(n)

def compute_d_vec(h, i, j):
    n = len(h)
    m_range = np.arange(2 * n - 1) - (n - 1)
    
    idx1 = i + m_range
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.zeros(2 * n - 1)
    if np.any(mask1):
        term1[mask1] = 1.0 - h[idx1[mask1]]
    
    idx2 = i - m_range
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.zeros(2 * n - 1)
    if np.any(mask2):
        term2[mask2] = h[idx2[mask2]]
    
    idx3 = j + m_range
    mask3 = (idx3 >= 0) & (idx3 < n)
    term3 = np.zeros(2 * n - 1)
    if np.any(mask3):
        term3[mask3] = 1.0 - h[idx3[mask3]]
    
    idx4 = j - m_range
    mask4 = (idx4 >= 0) & (idx4 < n)
    term4 = np.zeros(2 * n - 1)
    if np.any(mask4):
        term4[mask4] = h[idx4[mask4]]
    
    return term1 - term2 - term3 + term4

def ternary_search_delta(corr, d_vec, lo, hi, iters=15):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr + m1 * d_vec)
        f2 = np.max(corr + m2 * d_vec)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coordinate_descent(h, n_swaps, rng, time_limit):
    n = len(h)
    corr = compute_correlations(h)
    current_max = np.max(corr)
    n_above = np.sum(h > 0.5)
    n_below = n - n_above
    
    for idx in range(n_swaps):
        if time.time() > time_limit:
            break
        
        # Sample a pair: prefer one above and one below 0.5
        if rng.random() < 0.7 and n_above > 0 and n_below > 0:
            if rng.random() < 0.5:
                i = rng.choice(np.where(h > 0.5)[0])
                j = rng.choice(np.where(h <= 0.5)[0])
                direction = -1  # decrease i, increase j
            else:
                i = rng.choice(np.where(h <= 0.5)[0])
                j = rng.choice(np.where(h > 0.5)[0])
                direction = +1  # increase i, decrease j
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            direction = rng.choice([-1, 1])
        
        if i == j:
            continue
        
        d_vec = compute_d_vec(h, i, j)
        
        # Determine valid delta range: h[i] + delta in [0,1], h[j] - delta in [0,1]
        delta_min = max(-h[i], h[j] - 1.0)
        delta_max = min(1.0 - h[i], h[j])
        
        if delta_min >= delta_max - 1e-15:
            continue
        
        # Check if moving helps
        current_max_val = np.max(corr)
        
        # Try the direction we prefer
        if direction > 0:
            test_d = min(0.05, delta_max)
        else:
            test_d = max(-0.05, delta_min)
        
        f_test = np.max(corr + test_d * d_vec)
        if f_test >= current_max_val - 1e-12:
            # Try the other direction
            if direction > 0:
                test_d = max(-0.05, delta_min)
            else:
                test_d = min(0.05, delta_max)
            f_test = np.max(corr + test_d * d_vec)
            if f_test >= current_max_val - 1e-12:
                continue
        
        # Full ternary search
        opt_delta = ternary_search_delta(corr, d_vec, delta_min, delta_max, 20)
        f_opt = np.max(corr + opt_delta * d_vec)
        
        if f_opt < current_max_val - 1e-12:
            h[i] += opt_delta
            h[j] -= opt_delta
            corr = corr + opt_delta * d_vec
            current_max = f_opt
    
    return h, current_max * (2.0 / n)
