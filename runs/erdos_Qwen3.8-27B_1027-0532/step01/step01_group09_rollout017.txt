import numpy as np
import time

def project_feasible(x, target):

    x = np.clip(x.astype(float), 0.0, 1.0)

    for _ in range(60):

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def soft_max_grad(h, t, dx):
    """Compute gradient of soft-max of overlap values."""
    n = len(h)
    full = np.correlate(h, 1.0 - h, 'full') * dx
    max_val = full.max()
    exps = np.exp((full - max_val) / t)
    p = exps / exps.sum()
    
    B = np.correlate(h, p, 'full')[n-1 : 2*n-1]
    C = np.convolve(p, h, 'full')[n-1 : 2*n-1]
    
    grad = dx * (1.0 - B - C)
    val = max_val + t * np.log(exps.sum())
    return val, grad

def adam_opt(h0, budget, t_init=0.1, t_min=0.005, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.time()
    it = 0
    t = t_init
    
    while time.time() - t0 < budget:
        it += 1
        val, grad = soft_max_grad(h, t, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, tgt)

            if c5 < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-10:
        return x
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def soft_max_val_grad(h, t, dx):
    n = len(h)
    full = np.correlate(h, 1.0 - h, 'full')
    vals = full * dx
    max_val = vals.max()
    exps = np.exp((vals - max_val) / t)
    p = exps / exps.sum()
    val = max_val + t * np.log(exps.sum())
    
    B = np.correlate(h, p, 'full')[n-1 : 2*n-1]
    C = np.convolve(p, h, 'full')[n-1 : 2*n-1]
    grad = dx * (1.0 - B - C)
    return val, grad, p, max_val

def adam_soft_max(h0, time_limit, t_init=0.15, t_min=0.01, lr=0.05, anneal_rate=0.995):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.time()
    it = 0
    t = t_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    while time.time() - t0 < time_limit:
        it += 1
        val, grad, p, max_val = soft_max_val_grad(h, t, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, tgt)
        
        # Occasionally check actual C5

            c5 = eval_c5(h)
            if c5 < best_c5:

                best = h.copy()
            t = max(t_min, t * anneal_rate ** 100)
    
    c5 = eval_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best = h.copy()
    return best

def resample(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def slsqp_polish(h0, time_limit):
    """Use SLSQP for final polish at smaller n."""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    t0 = time.time()
    
    def objective(x):
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()
    
    try:

                         bounds=[(0.0, 1.0)] * n,

                         options={'maxiter': 5000, 'ftol': 1e-12})
        if result.success or eval_c5(project_feasible(result.x, tgt)) < eval_c5(h0):

            h[idx] = 1
    
    best = h.copy()
    best_c5 = eval_c5(best)
    t0 = time.time()

        improved = False

        # Based on gradient information

        np.random.shuffle(zeros_arr)

        tried = 0
        while tried < n_tries and time.time() - t0 < time_limit:

            b = zeros_arr[np.random.randint(len(zeros_arr))]

            h[b] = 1
            c5 = eval_c5(h)

                improved = True

                zeros_arr = np.where(h == 0)[0]

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_soft_max(h, t, dx):
    """Returns (value, gradient, distribution p)"""
    n = len(h)
    full = np.correlate(h, 1.0 - h, 'full')
    vals = full * dx
    max_v = vals.max()
    shifted = (vals - max_v) / t
    exps = np.exp(shifted)
    p = exps / exps.sum()
    val = max_v + t * np.log(exps.sum())
    
    # Gradient: dx * (1 - B - C)
    # B[j] = sum_m p[m] * h[j+m-(n-1)] = correlate(h, p, 'full')[j+n-1]
    # C[j] = sum_m p[m] * h[j-m+(n-1)] = convolve(p, h, 'full')[j+n-1]
    B = np.correlate(h, p, 'full')[n-1:2*n-1]
    C = np.convolve(p, h, 'full')[n-1:2*n-1]
    grad = dx * (1.0 - B - C)
    return val, grad, p
