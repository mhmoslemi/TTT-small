import numpy as np
import time
from scipy.optimize import minimize

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_gradient(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(-(n - 1), n):
        w = p[k + n - 1]
        if w < 1e-18:
            continue
        lo = max(0, k)
        hi = min(n - 1, n - 1 + k)
        if lo <= hi:
            grad[lo:hi + 1] += w * omh[lo - k:hi + 1 - k]
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + k:hi2 + 1 + k]
    return dx * grad

def project_feasible(h, target):
    h = np.asarray(h, dtype=np.float64)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = float(hp.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(hp.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            m = hp < 1.0
            c = int(m.sum())
            if c > 0:
                hp[m] = np.minimum(hp[m] + diff / c, 1.0)
        else:
            m = hp > 0.0
            c = int(m.sum())
            if c > 0:
                hp[m] = np.maximum(hp[m] + diff / c, 0.0)
    return hp

def smooth_max(c, beta):
    m = float(np.max(c))
    return m + float(np.log(np.sum(np.exp(c - m)))) / beta

def pg_optimize(h0, n, beta, max_iter, t_limit):

    target = 0.5 * n
    h = h0.copy()
    lr = 4.0 / (beta * dx + 1e-10)
    lr = min(lr, 50.0)
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        omh = 1.0 - h
        c = np.correlate(h, omh, mode='full') * dx
        cs = c - np.max(c)
        ec = np.exp(beta * cs)
        p = ec / ec.sum()
        g = compute_gradient(h, p, dx)
        obj_cur = smooth_max(c, beta)
        alpha = lr
        accepted = False
        for _ in range(50):
            h_try = project_feasible(h - alpha * g, target)
            c_try = np.correlate(h_try, 1.0 - h_try, mode='full') * dx
            obj_try = smooth_max(c_try, beta)
            if obj_try <= obj_cur + 1e-16:
                h = h_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            h = project_feasible(h - alpha * g, target)
        if it % 200 == 199 and it > 0:
            lr = min(lr * 1.05, 100.0 / (beta * dx + 1e-10))
    return h

def upsample_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, 0.5 * n_dst)

def gen_random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(np.float64)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h > 0.5)[0]

        add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[add] = 1.0
    return h

def slsqp_constraint_sum(hv, n):
    return float(hv.sum()) - 0.5 * n

def slsqp_constraint_jac(hv, n):
    return np.ones(len(hv))

def slsqp_obj(hv, dx, beta):
    c = np.correlate(hv, 1.0 - hv, mode='full') * dx
    return smooth_max(c, beta)

def slsqp_obj_grad(hv, dx, beta):

    cs = c - np.max(c)
    ec = np.exp(beta * cs)
    p = ec / ec.sum()

             'jac': slsqp_constraint_jac}]
    try:
        res = minimize(slsqp_obj, h0, jac=slsqp_obj_grad,
                       args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None and time.time() < t_limit:
            return project_feasible(res.x, 0.5 * n)
    except Exception:
        pass
    return h0

                    o1 = c5_value(h, dx)
                    h[j] = m2
                    o2 = c5_value(h, dx)

                    else:
                        lo_b = m1

            h[j] = best_j
            new_sum = float(h.sum())
            if abs(new_sum - target) > 1e-10:
                h = project_feasible(h, target)
            if abs(h[j] - old_val) > 1e-12:
                improved = True
        if not improved:

    return project_feasible(h, target)
