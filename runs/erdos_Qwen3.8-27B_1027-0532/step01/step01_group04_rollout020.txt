```python
def projected_gradient_step(h, grad, target_sum, step_size):
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0, 1)
    # Project onto sum constraint
    diff = target_sum - np.sum(h_new)
    if diff > 0:
        mask = h_new < 1.0
    else:
        mask = h_new > 0.0
    count = np.sum(mask)
    if count > 0:
        h_new[mask] += diff / count
    h_new = np.clip(h_new, 0, 1)
    return h_new
```
