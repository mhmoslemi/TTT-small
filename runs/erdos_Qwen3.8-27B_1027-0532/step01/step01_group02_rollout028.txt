import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.3)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.3)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def compute_subgradient(h):
    """Compute subgradient of C5 at h. Returns (gradient, max_corr_value, max_index)"""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)  # the shift
    
    grad = np.zeros(n)
    if k >= 0:
        # f_k(h) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        for j in range(n):
            if j < n - k:
                grad[j] += (1.0 - h[j + k])
            if j >= k:
                grad[j] -= h[j - k]
    else:
        kp = -k
        for j in range(n):
            if 0 <= j - kp < n - kp:
                grad[j] += (1.0 - h[j - kp])
            if 0 <= j < n - kp:
                grad[j] -= h[j + kp]
    
    return grad * dx, max(corr) * dx, max_idx

def subgradient_step(h, alpha, seed=0):
    """Take a subgradient step and project back to feasible set."""
    n = len(h)
    grad, c5_val, _ = compute_subgradient(h)
    h_new = h - alpha * grad
    h_new = repair_h(h_new, n)
    return h_new, c5_val

def projection_onto_feasible(h, n):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = n/2} using bisection."""
    target = n / 2.0
    # h = clamp(h_raw - lambda, 0, 1) for some lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        mask = (x > 0) & (x < 1)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_optimize(h, n, maxiter=500, ftol=1e-12):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)
    
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for j in range(n):
                if j < n - k:
                    grad[j] += (1.0 - x[j + k])
                if j >= k:
                    grad[j] -= x[j - k]
        else:
            kp = -k
            for j in range(n):
                if 0 <= j - kp < n - kp:
                    grad[j] += (1.0 - x[j - kp])
                if 0 <= j < n - kp:
                    grad[j] -= x[j + kp]
        return grad * dx
    
    try:
        r = minimize(
            obj, h, jac=obj_grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0,
                         'jac': lambda x, n=n: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if r.success or r.fun < evaluate_c5(h):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = repair_h(r.x, n)
            return r.x, evaluate_c5(r.x)
    except Exception:
        pass
    return h, evaluate_c5(h)

    try:

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def grad_at_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g[:n-k] = 1.0 - h[k:]
            g[k:] -= h[:n-k]
    else:
        kp = -k
        g[kp:] += 1.0 - h[:n-kp]
        g[:n-kp] -= h[kp:]
    return g * dx

def smooth_obj(x, beta, n):
    corr = get_corr(x)
    m = np.max(corr)
    return m / beta + np.log(np.sum(np.exp(corr - m))) / beta

def smooth_grad(x, beta, n):
    corr = get_corr(x)
    m = np.max(corr)
    w = np.exp(corr - m)
    w /= w.sum()
    g = np.zeros(n)
    significant = np.where(w > 1e-8)[0]
    for idx in significant:
        k = int(idx) - (n - 1)
        g += w[idx] * grad_at_lag(x, k)
    return g / beta

def exact_obj(x, n):
    return evaluate_c5(x)

def exact_grad(x, n):
    corr = get_corr(x)
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    return grad_at_lag(x, k)

def slsqp_smooth(h, n, beta, maxiter=100):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(smooth_obj, h, args=(beta, n), jac=lambda x: smooth_grad(x, beta, n),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = repair_h(r.x, n)
        c5 = evaluate_c5(h_new)
        if c5 <= evaluate_c5(h) + 1e-8:
            return h_new, c5
    except Exception:
        pass
    return h, evaluate_c5(h)

def slsqp_exact(h, n, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(exact_obj, h, args=(n,), jac=lambda x: exact_grad(x, n),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = repair_h(r.x, n)
        c5 = evaluate_c5(h_new)
        if c5 <= evaluate_c5(h) + 1e-8:
            return h_new, c5
    except Exception:
        pass
    return h, evaluate_c5(h)

def subgradient_momentum(h, n, max_iter, time_limit, top_m=5, mu=0.9):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    v = np.zeros(n)
    t0 = time.time()
    alpha = 0.05
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = get_corr(h_curr)
        top_indices = np.argsort(corr)[-top_m:]
        g = np.zeros(n)
        for idx in top_indices:
            k = int(idx) - (n - 1)
            g += grad_at_lag(h_curr, k)
        g /= top_m
        v = mu * v + (1.0 - mu) * g
        h_new = h_curr - alpha * v
        h_new = repair_h(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_curr = h_new
            alpha = max(0.001, alpha * 0.9995)
        else:
            accept_prob = np.exp(-(c5_new - h_curr and best_c5) * 30.0) if c5_new > best_c5 else 1.0
            if np.random.random() < accept_prob:
                h_curr = h_new
            else:
                if it % 100 == 50:
                    h_curr = repair_h(best_h + np.random.normal(0, 0.02, n), n)
            alpha = max(0.001, alpha * 0.99)
    return best_h, best_c5
