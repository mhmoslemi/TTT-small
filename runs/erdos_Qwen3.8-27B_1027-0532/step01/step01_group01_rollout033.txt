    best_h = pop[best_idx].copy()
    best_f = fitness[best_idx]

        # Update F and CR adaptively
        F = 0.3 + 0.4 * (1 - (time.time() - start_time) / time_limit)

                c = rng.randint(pop_size)

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0

    best_f = fitness[best_idx]

    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx
    
    def cons(x):
        return np.sum(x) - n / 2.0

        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': cons}],
            options={'maxiter': max_iter, 'ftol': 1e-12}

        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass

    lo = -2.0
    hi = 2.0

        else:

        return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx
