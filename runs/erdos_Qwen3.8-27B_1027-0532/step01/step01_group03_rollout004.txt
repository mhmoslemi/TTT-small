import time
import numpy as np
from scipy.optimize import minimize

_N = 86
_DX = 2.0 / _N
_TARGET_SUM = _N / 2.0
_ALPHA = 100.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * _DX)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _fix_sum_binary(h, target_sum, rng):
    h = h.copy()
    n_ones = int(np.sum(h))
    target = int(target_sum)
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        to_remove = n_ones - target
        if to_remove > 0 and len(ones) > 0:
            idx = rng.choice(ones, to_remove, replace=False)
            h[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        to_add = target - n_ones
        if to_add > 0 and len(zeros) > 0:
            idx = rng.choice(zeros, to_add, replace=False)
            h[idx] = 1.0
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _con_fun(x):
    return x.sum() - _TARGET_SUM

def _con_jac(x):
    return np.ones(_N)

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _smooth_max(corr, _ALPHA) * _DX

def _do_slsqp(h_start, alpha, maxiter, ftol):
    global _ALPHA
    _ALPHA = alpha
    bounds = [(0.0, 1.0)] * _N
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _run_binary_sa(h_init, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    T_init = 0.03
    T_final = 0.0003
    t0 = time.time()
    deadline = t0 + time_limit
    no_improve = 0
    patience = 500000
    while time.time() < deadline:
        elapsed = time.time() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress
        if no_improve > patience:
            T = max(T, 0.01)
            no_improve = 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _c5_of(h)
        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            delta = c5_new - best_c5
            if delta < 50.0 * T:
                if np.exp(-delta / max(T, 1e-12)) > rng.rand():
                    pass
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            else:
                h[i] = 1.0
                h[j] = 0.0
    return best_h, best_c5

def _smooth_binary(h_binary, sigma):
    kernel_size = int(2 * 3 * sigma) | 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel = kernel / kernel.sum()
    h_smooth = np.convolve(h_binary, kernel, mode='same')
    return h_smooth
