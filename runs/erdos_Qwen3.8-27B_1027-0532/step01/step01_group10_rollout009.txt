```python
    """Project h onto {x in [0,1]^n : sum(x) = n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0 - 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = diff / cnt
        h[mask] = np.minimum(h[mask] + step, 1.0)

        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = diff / cnt
        h[mask] = np.maximum(h[mask] + step, 0.0)
    return np.clip(h, 0.0, 1.0)
```

```python
def proj_sum(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2} using bisection"""
    target = n / 2.0
    def sum_at_offset(offset):
        return np.sum(np.clip(h + offset, 0.0, 1.0))
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_offset(mid)
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
```

```python
            h_pert = repair_h(best_h + noise, n_best)
            for beta in [10.0, 5.0]:
                if elapsed() > budget_s * 0.95:
                    break
                h_opt = opt_slsqp(h_pert, n_best, dx_best, beta, maxiter=800)
                c = c5_value(h_opt, dx_best)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
                h_pert = h_opt

    # Phase 6: Final SLSQP polish with very low beta
    if elapsed() < budget_s * 0.97:
        h_final = opt_slsqp(best_h, best_n, dx_best, 1.0, maxiter=3000)
        c = c5_value(h_final, dx_best)
        if c < best_c5:
            best_c5 = c
            best_h = h_final.copy()

    # Phase 7: Nelder-Mead on the raw (non-smooth) max for final squeeze
    if elapsed() < budget_s * 0.99:
        def nm_obj(x):
            return c5_value(x, dx_best)
        try:
            res = minimize(nm_obj, best_h, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_nm = repair_h(res.x, best_n)
                c = c5_value(h_nm, dx_best)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_nm.copy()
        except Exception:
            pass

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
```
