   Let me just implement a general 1D minimizer for the convex piecewise-linear function. I'll evaluate at O(n) candidate points.

   Actually, the cleanest approach: the minimum of max_k (a_k + x*b_k) over x ∈ [0,1] is the smallest t such that a_k + x*b_k ≤ t for all k. For a given t, this is feasible iff:
   - For b_k > 0: x ≤ (t - a_k) / b_k
   - For b_k < 0: x ≥ (t - a_k) / b_k = (a_k - t) / (-b_k)
   - For b_k = 0: need a_k ≤ t
   
   So the feasible x range for a given t is:
   [max over k with b_k < 0 of (a_k - t)/(-b_k), min over k with b_k > 0 of (t - a_k)/b_k] ∩ [0, 1]
   
   And we need this to be non-empty.
   
   The minimum t is the smallest value where this range is non-empty. This can be found by bisection on t!

   Let me binary search on t:
   - For a given t, compute L(t) = max(0, max_{k: b_k<0} (a_k-t)/(-b_k)) and R(t) = min(1, min_{k: b_k>0} (t-a_k)/b_k)
   - Feasible iff L(t) ≤ R(t) and a_k ≤ t for all k with b_k = 0
   - Binary search for the smallest feasible t

def _coord_descent(h, n, max_sweeps, target_sum, time_limit):
    """Cyclic coordinate descent for the Erdős problem."""
    t_start = time.time()
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):

            break
        for j in range(n):
            new_val = _coord_update(h, j, n)
            h[j] = new_val
        # Project to fix sum and bounds
        h = _project_onto_simplex_box(h, target_sum)
    
    return h

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(v, target_sum):
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if diff > 1e-10:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    elif diff < -1e-10:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _coord_update(h, j, n):
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    idx_pos = j + ks
    idx_neg = j - ks
    valid_pos = (idx_pos >= 0) & (idx_pos < n)
    valid_neg = (idx_neg >= 0) & (idx_neg < n)
    h_pos = np.where(valid_pos, h[idx_pos], 0.0)
    h_neg = np.where(valid_neg, h[idx_neg], 0.0)
    coef = (1.0 - h_neg) * valid_neg - h_pos * valid_pos
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = (corr - h[j] * coef) * dx
    B = coef * dx
    t_lo = float(np.max(np.minimum(A, A + B)))
    t_hi = float(np.max(np.maximum(A, A + B)))
    if t_hi - t_lo < 1e-14:
        return h[j]
    best_x = h[j]
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        pos_mask = B > 1e-14
        neg_mask = B < -1e-14
        zero_mask = ~(pos_mask | neg_mask)
        if np.any(zero_mask):
            if np.max(A[zero_mask]) > t_mid + 1e-14:
                t_lo = t_mid
                continue
        x_lo = 0.0
        x_hi = 1.0
        if np.any(neg_mask):
            bounds = (A[neg_mask] - t_mid) / (-B[neg_mask])
            x_lo = max(x_lo, float(np.max(bounds)))
        if np.any(pos_mask):
            bounds = (t_mid - A[pos_mask]) / B[pos_mask]
            x_hi = min(x_hi, float(np.min(bounds)))
        if x_lo <= x_hi + 1e-12:
            t_hi = t_mid
            best_x = np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0)
        else:
            t_lo = t_mid
    return best_x

def _coord_descent(h, n, max_sweeps, target_sum, time_limit):
    t_start = time.time()
    prev_c5 = _c5_of(h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        for j in range(n):
            h[j] = _coord_update(h, j, n)
        h = _project(h, target_sum)
        cur_c5 = _c5_of(h)
        if cur_c5 < prev_c5 - 1e-10:
            prev_c5 = cur_c5
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _sum_constraint(x, target_sum):
    return x.sum() - target_sum

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_refine(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (target_sum,)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def _binary_cd(h, n, target_sum, max_sweeps, time_limit):
    t_start = time.time()
    h = np.round(h)
    h = _fix_sum(h, target_sum)
    best_local = _c5_of(h)
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            if old < 0.5:
                h[j] = 1.0
            else:
                h[j] = 0.0
            h_try = _fix_sum(h.copy(), target_sum)
            c5_try = _c5_of(h_try)
            if c5_try < best_local - 1e-14:
                h = h_try
                best_local = c5_try
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):

    deadline = t_start + budget_s - 10
