def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_at(h, n, idx_star):
    """Compute subgradient at the lag index idx_star."""
    dx = 2.0 / n
    lag = idx_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] += -h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] += -h[:n - l]
    grad *= dx
    return grad

def nesterov_subgradient(h, n, n_iter=500, beta=0.9):

    sum_exp = np.sum(exp_vals)
    return (max_corr + np.log(sum_exp) / t) * dx

def smoothed_grad(h, n, t):

    corr = np.correlate(h, 1.0 - h, mode="full")

    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for idx in range(len(corr)):

        lag = idx - (n - 1)

        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])

        sh = np.clip(t * (corr - mx), -400, 400)
        return (mx + np.log(np.sum(np.exp(sh))) / t) * dx
    def grad(h):
        corr = compute_corr(h, n)
        mx = np.max(corr)
        sh = np.clip(t * (corr - mx), -400, 400)
        ev = np.exp(sh)
        sw = np.sum(ev)
        w = ev / sw
        g = np.zeros(n)
        for idx in range(len(corr)):
            wi = w[idx]
            if wi < 1e-13:
                continue
            lag = idx - (n - 1)
            gl = grad_single_lag(h, n, lag)
            g += wi * gl
        return g
    def cfun(h):
        return np.sum(h) - n / 2.0
    def cjac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

def nesterov_sg(h0, n, n_iter, beta, alpha0, decay, K=1):
    h = h0.copy().astype(float)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(n_iter):
        corr = compute_corr(y, n)
        if K <= 1:
            idx_star = np.argmax(corr)
            lag = idx_star - (n - 1)
            g = grad_single_lag(y, n, lag)
        else:
            top_idx = np.argsort(corr)[-K:]
            g = np.zeros(n)
            for idx in top_idx:
                lag = idx - (n - 1)
                g += grad_single_lag(y, n, lag)
            g /= K
        v = beta * v + (1.0 - beta) * g
        alpha = alpha0 / (1.0 + decay * it)
        y = y - alpha * v
        y = project_onto_constraint(y, n)
        c5 = compute_c5(y, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
    return best_h, best_c5
