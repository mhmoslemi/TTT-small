def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h

            if diff <= 1e-12:

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 12
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_init = len(initial_h_values)
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 86
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = _c5_of(best_h)

    n_list = [n_init]
    if n_init == 86:
        n_list = [86, 100]

    for n in n_list:
        target_sum = n / 2.0
        if time.time() > deadline - 60:
            break

        starts = []
        if n == n_init and 'initial_h_values' in globals() and initial_h_values is not None:
            starts.append(_fix_sum(h_init.copy(), target_sum))

        for freq in [0.5, 0.7, 1.0, 1.3, 1.7, 2.3]:
            amp = 0.3 + 0.1 * rng.rand()
            starts.append(_make_cosine(n, freq, rng.uniform(0, 6.28), amp))
            if time.time() > deadline - 60:
                break

        for _ in range(2):
            starts.append(_make_smooth_random(n, rng))

        if n == n_init and 'initial_h_values' in globals() and initial_h_values is not None:
            for _ in range(2):
                pert = h_init + 0.08 * rng.randn(n)
                starts.append(_fix_sum(pert, target_sum))

        for h_start in starts:
            if time.time() > deadline - 40:
                break
            h_opt, c5_opt = _optimize_one(h_start, n, target_sum, deadline, rng)
            h_opt = _directed_perturb(h_opt, n, target_sum, 0.02)
            c5_opt2 = _c5_of(h_opt)
            if c5_opt2 < c5_opt:
                c5_opt = c5_opt2
            if time.time() < deadline - 40:
                h_opt, c5_opt = _optimize_one(h_opt, n, target_sum, deadline, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    best_h = _fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, len(best_h))
