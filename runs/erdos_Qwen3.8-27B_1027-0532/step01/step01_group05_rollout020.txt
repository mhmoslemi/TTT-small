from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def overlap_lags(h, n):
    """Compute all overlap values at each lag."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def pnrm_smooth_obj(h, n, p):
    """Compute p-norm smoothed max of overlaps."""
    overlaps = overlap_lags(h, n)
    # Clip to avoid numerical issues
    overlaps = np.maximum(overlaps, 0.0)
    if p == np.inf:
        return np.max(overlaps)
    return np.sum(overlaps**p)**(1.0/p)

def pnrm_smooth_grad(h, n, p):
    """Gradient of p-norm smoothed max."""
    dx = 2.0 / n
    overlaps = overlap_lags(h, n)
    overlaps = np.maximum(overlaps, 0.0)
    pterm = overlaps**(p-1)
    pnorm = np.sum(overlaps**p)**(1.0/p)
    weights = pterm / (pnorm**(p-1)) if pnorm > 0 else np.zeros_like(overlaps)
    
    grad = np.zeros(n)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def overlap_lags(h, n):

def ls_exp_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]
    grad *= dx
    return obj, grad

        h_pol, c5_pol = nesterov_polish(best_h, best_n, n_iter=3000, t_soft=200.0, alpha0=0.02, decay=0.9999)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 9: Ultra-final subgradient with very small steps
    if time_left() > 15:
        h_final = best_h.copy()
        dx = 2.0 / best_n
        for it in range(500):
            corr = np.correlate(h_final, 1.0 - h_final, mode="full")
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            # Use top-3 lags
            sorted_idx = np.argsort(corr)[::-1][:3]
            grad = np.zeros(best_n)
            for idx in sorted_idx:
                lag = idx - (best_n - 1)
                w = 0.5 if idx == max_idx else 0.25
                if lag >= 0:
                    grad[lag:] += w * (1.0 - h_final[:best_n - lag])
                    grad[:best_n - lag] += -w * h_final[lag:]
                else:
                    l = -lag
                    grad[:best_n - l] += w * (1.0 - h_final[l:])
                    grad[l:] += -w * h_final[:best_n - l]
            grad *= dx
            step = 0.01 * (1.0 - it / 500.0) + 0.001
            h_final = project_onto_constraint(h_final - step * grad, best_n)
            c5_chk = compute_c5(h_final, best_n)
            if c5_chk < best_c5:
                best_c5 = c5_chk
                best_h = h_final.copy()

    # Final verification
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
