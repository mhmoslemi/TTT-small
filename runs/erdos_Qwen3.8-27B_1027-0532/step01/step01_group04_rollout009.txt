_g_n = 0
_g_h_full = None
_g_start = 0
_g_end = 0
_g_bt = 0.0
_g_deadline = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def obj_smooth(h, alpha):
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def full_objective(h):
    return obj_smooth(h, _g_alpha)


def full_con_eq(h):
    return np.sum(h) - _g_target


def full_con_jac(h):
    return np.ones(_g_n)


def block_objective(x):
    h_t = _g_h_full.copy()
    h_t[_g_start:_g_end] = x
    return obj_smooth(h_t, _g_alpha)


def block_con_eq(x):
    return np.sum(x) - _g_bt


def block_con_jac(x):
    return np.ones(_g_end - _g_start)


def time_remaining():
    return _g_deadline - time.time()


def try_optimize(h, n, ts, alpha, max_iter, ftol, bounds):
    _g_alpha = alpha
    _g_n = n
    _g_target = ts
    cons = [{'type': 'eq', 'fun': full_con_eq, 'jac': full_con_jac}]
    try:
        res = minimize(full_objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        h_new = project_feasible(res.x, ts)
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target, _g_n, _g_h_full, _g_start, _g_end, _g_bt, _g_deadline

    start_time = time.time()
    _g_deadline = start_time + budget_s - 12.0
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    n = len(best_h)
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Progressive alpha refinement from initial solution
    alphas = [5, 10, 20, 40, 80, 160, 320, 640]
    for alpha in alphas:
        if time_remaining() < 35:
            break
        max_iter = 4000 if alpha <= 40 else 2500
        h_new = try_optimize(best_h, n, ts, alpha, max_iter, 1e-16, bounds)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new

    # Phase 2: Block coordinate descent for local escape
    if time_remaining() > 180:
        _g_h_full = best_h.copy()
        block_size = 16
        for alpha_bcd in [20, 50, 100]:
            if time_remaining() < 50:
                break
            _g_alpha = alpha_bcd
            for sweep in range(4):
                if time_remaining() < 30:
                    break
                for start_idx in range(0, n - block_size, block_size):
                    end_idx = min(start_idx + block_size, n)
                    b = end_idx - start_idx
                    rest_sum = np.sum(_g_h_full) - np.sum(_g_h_full[start_idx:end_idx])
                    bt = ts - rest_sum
                    bt = max(0.0, min(float(b), bt))
                    _g_start = start_idx
                    _g_end = end_idx
                    _g_bt = bt
                    bounds_b = [(0.0, 1.0)] * b
                    cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
                    try:
                        res = minimize(block_objective, _g_h_full[start_idx:end_idx].copy(),
                                       method='SLSQP', bounds=bounds_b, constraints=cons_b,
                                       options={'maxiter': 60, 'ftol': 1e-12})
                        _g_h_full[start_idx:end_idx] = np.clip(res.x, 0.0, 1.0)
                    except Exception:
                        pass
            h_new = project_feasible(_g_h_full, ts)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new

    # Phase 3: Multi-grid exploration
    for n_try in [257, 401, 501, 663]:
        if time_remaining() < 70:
            break
        ts_t = n_try / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new_g = np.linspace(0.0, 1.0, n_try)
        h_cur = np.interp(x_new_g, x_old, best_h)
        h_cur = project_feasible(h_cur, ts_t)
        bounds_t = [(0.0, 1.0)] * n_try
        for alpha_t in [30, 80, 160, 320]:
            if time_remaining() < 25:
                break
            h_new = try_optimize(h_cur, n_try, ts_t, alpha_t, 1200, 1e-15, bounds_t)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
                h_cur = h_new

    # Phase 4: Diverse restarts from multiple starting structures
    n_use = len(best_h)
    ts_use = n_use / 2.0
    bounds_use = [(0.0, 1.0)] * n_use

    # Generate diverse starting points
    start_points = [best_h.copy()]

    # Circular shift variations
    for shift_amt in [n_use // 7, n_use // 4, n_use // 3]:
        shifted = np.roll(best_h, shift_amt)
        shifted = project_feasible(shifted, ts_use)
        start_points.append(shifted)

    # Smooth noise
    smooth_noise = best_h + rng.randn(n_use) * 0.06
    smooth_noise = project_feasible(smooth_noise, ts_use)
    start_points.append(smooth_noise)

    # Block flip
    bf = best_h.copy()
    bf_sz = max(5, n_use // 12)
    bf_start = rng.randint(0, max(1, n_use - bf_sz))
    bf[bf_start:bf_start + bf_sz] = 1.0 - bf[bf_start:bf_start + bf_sz]
    bf = project_feasible(bf, ts_use)
    start_points.append(bf)

    for sp in start_points:
        if time_remaining() < 40:
            break
        for alpha_sp in [30, 80, 200]:
            if time_remaining() < 20:
                break
            h_new = try_optimize(sp, n_use, ts_use, alpha_sp, 1000, 1e-14, bounds_use)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
            sp = h_new

    # Phase 5: Aggressive restart loop
    while time_remaining() > 30:
        strat = rng.randint(5)
        if strat == 0:
            h_pert = best_h + rng.randn(n_use) * 0.03
        elif strat == 1:
            h_pert = best_h.copy()
            n_flip = max(3, n_use // 20)
            idx = rng.choice(n_use, size=n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif strat == 2:
            h_pert = best_h.copy()
            bsz = rng.randint(8, max(9, n_use // 5))
            start_b = rng.randint(0, max(1, n_use - bsz))
            h_pert[start_b:start_b + bsz] = rng.rand(bsz) * 0.6 + 0.2
        elif strat == 3:
            h_pert = 0.85 * best_h + 0.15 * rng.rand(n_use)
        else:
            h_pert = best_h.copy()
            n_seg = rng.randint(3, 7)
            seg_len = n_use // n_seg
            for si in range(n_seg):
                s1 = si * seg_len
                e1 = s1 + seg_len if si < n_seg - 1 else n_use
                h_pert[s1:e1] += rng.randn(e1 - s1) * 0.08

        h_pert = project_feasible(h_pert, ts_use)
        alpha_r = int(rng.choice([15, 30, 60, 120, 200]))
        h_new = try_optimize(h_pert, n_use, ts_use, alpha_r, 400, 1e-13, bounds_use)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new

    # Phase 6: Final high-alpha polish
    if time_remaining() > 18:
        for alpha_f in [500, 1000, 2000]:
            if time_remaining() < 12:
                break
            h_new = try_optimize(best_h, n_use, ts_use, alpha_f, 3000, 1e-16, bounds_use)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
