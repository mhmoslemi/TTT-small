    dx = 2.0 / n

    tau_corr = tau * corr
    max_tc = np.max(tau_corr)

    weights = np.exp(tau_corr - max_tc)

        if weights[m] < 1e-14:

        w = weights[m]
        j_lo = max(0, lag)

    obj = log_sum_exp / tau

    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

    mu = 50.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = log_sum_exp / tau
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m] * dx
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    x_new = np.linspace(0, 2, n_new, endpoint=False)

    if time.time() > time_limit:
        return h0, compute_c5(h0)

    def obj_fn(x):
        xc = np.clip(x, 0.0, 1.0)
        sm, _ = soft_max_obj_and_grad(xc, tau)
        pen = 0.5 * mu * (xc.sum() - target_sum) ** 2
        return sm + pen

    def grad_fn(x):
        xc = np.clip(x, 0.0, 1.0)
        _, g = soft_max_obj_and_grad(xc, tau)
        return g + mu * (xc.sum() - target_sum)

    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-10})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, target_sum)
        c5 = compute_c5(h_opt)
        return h_opt, c5
    except Exception:
        return h0, compute_c5(h0)
