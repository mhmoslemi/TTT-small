    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    scaled = alpha * (corr - m)
    scaled = np.clip(scaled, -700, 700)
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    val = m + np.log(exp_vals.sum()) / alpha
    return val, w

    corr = compute_correlations(h)
    val, _ = smooth_max_and_weights(corr, alpha)
    return val

    n = len(h)
    corr = compute_correlations(h)
    val, w = smooth_max_and_weights(corr, alpha)
    # Gradient via FFT-free convolution
    g1 = np.correlate(h, w, 'full')[n-1:2*n-1]
    g2 = np.convolve(w, h, 'full')[n-1:2*n-1]
    grad = (1.0 - g1) - g2
    return val, grad

        return smooth_obj(x, alpha)
    def grad(x):
        _, g = smooth_grad(x, alpha)
        return g

        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)

                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return res.x
