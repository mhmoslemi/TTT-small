import numpy as np
import time

def project_feasible(x):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}"""
    n = len(x)
    target = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    """Compute correlation and max"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, np.max(corr)

def subgradient_of_max_corr(h, tol_frac=0.05):
    """Compute subgradient of max_k corr[k] for h"""
    n = len(h)
    corr, cmax = compute_corr(h)
    if cmax <= 0:
        return np.zeros(n), cmax, corr
    active = corr >= cmax * (1.0 - tol_frac)
    n_active = np.sum(active)
    
    grad = np.zeros(n)
    # For each active lag j, the gradient contribution at position i is:
    # (1 - h[i + shift]) if i + shift valid, minus h[i - shift] if i - shift valid
    # where shift = j - (n-1)
    for j in np.where(active)[0]:
        shift = j - (n - 1)
        if shift >= 0:
            # corr[j] = sum_i h[i] * (1-h[i+shift]) for i in [0, n-1-shift]
            # d/dh[i] = (1-h[i+shift]) for i in [0, n-1-shift]
            #           - h[i-shift] for i in [shift, n-1]
            i_max = n - 1 - shift
            grad[:i_max+1] += (1.0 - h[shift:])  # (1-h[i+shift]) for i=0..n-1-shift
            grad[shift:] -= h[:i_max+1]  # -h[i-shift] for i=shift..n-1
        elif shift < 0:
            s = -shift
            # corr[j] = sum_i h[i] * (1-h[i-s]) for i in [s, n-1]
            # d/dh[i] = (1-h[i-s]) for i in [s, n-1]
            #           - h[i+s] for i in [0, n-1-s]
            i_min = s
            grad[i_min:] += (1.0 - h[:n-s])  # (1-h[i-s]) for i=s..n-1
            grad[:n-s] -= h[s:]  # -h[i+s] for i=0..n-1-s
    
    grad /= n_active
    return grad, cmax, corr

def projected_gradient_descent(h, n_steps, step_size, tol_frac=0.05, verbose=False):
    """Projected gradient descent on the subgradient of max corr"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    n = len(h)
    dx = 2.0 / n
    
    for step in range(n_steps):
        grad, cmax, corr = subgradient_of_max_corr(h, tol_frac)
        c5 = cmax * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Projected gradient step
        h_new = project_feasible(h - step_size * grad)
        h = h_new
    
    # Final check
    _, cmax = compute_corr(h)
    if cmax * dx < best_c5:
        best_c5 = cmax * dx
        best_h = h.copy()
    
    return best_h, best_c5

def compute_subgradient(h, tol_frac=0.1):
    n = len(h)
    corr = compute_corr(h)
    cmax = np.max(corr)
    if cmax <= 1e-15:
        return np.zeros(n), cmax, corr
    
    active = np.where(corr >= cmax * (1.0 - tol_frac))[0]
    grad = np.zeros(n)
    
    for j in active:
        shift = j - (n - 1)
        # First term: (1-h[p-shift]) for p where 0<=p-shift<=n-1

        p_lo2 = max(0, -shift)

        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= h[p_lo2+shift:p_hi2+1+shift]
    
    grad /= len(active)
    return grad, cmax, corr

    n = len(x)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def score_c5(h):
    n = len(h)
    dx = 2.0 / n

def subgradient(h, tol_frac):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    if cmax < 1e-15:

    grad = np.zeros(n)

        p_lo = max(0, shift)
        p_hi = min(n - 1, n - 1 + shift)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += 1.0 - h[p_lo - shift:p_hi + 1 - shift]
        p_lo2 = max(0, -shift)
        p_hi2 = min(n - 1, n - 1 - shift)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= h[p_lo2 + shift:p_hi2 + 1 + shift]
    grad /= float(len(active))
    return grad, cmax

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    logsum = max_scaled + np.log(np.sum(exps))
    obj = dx * (cmax + T * logsum)
    # Gradient
    weights = exps / np.sum(exps)  # softmax weights
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for j in active:
        shift = int(j) - (n - 1)
        w = weights[j] * dx
        p_lo = max(0, shift)
        p_hi = min(n - 1, n - 1 + shift)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += w * (1.0 - h[p_lo - shift:p_hi + 1 - shift])
        p_lo2 = max(0, -shift)
        p_hi2 = min(n - 1, n - 1 - shift)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= w * h[p_lo2 + shift:p_hi2 + 1 + shift]
    return obj, grad

    return np.ones(len(h))
