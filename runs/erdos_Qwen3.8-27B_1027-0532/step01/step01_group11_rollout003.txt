def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target_sum:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # returns array of length 2n-1

def binary_sa(h_bin, n, target, rng, max_steps, T0, T_end, time_limit, start_time):
    """Simulated annealing on binary solutions."""
    C = compute_C_array(h_bin)
    best_c5 = float(np.max(C))
    best_h = h_bin.copy()
    T = T0
    cooling = (T_end / T0) ** (1.0 / max_steps)
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        i = rng.integers(n)
        sign = 1.0 if h_bin[i] == 0 else -1.0
        
        # Compute delta_C
        delta = np.zeros(2 * n - 1)
        # k ranges from -(n-1) to (n-1), mapped to index k + (n-1)
        # First factor: k such that 0 <= i+k < n, i.e., -i <= k <= n-1-i

            if k == 0:

            idx = k + (n - 1)
            delta[idx] += sign * (1 - h_bin[i + k])
        
        # Second factor: k such that 0 <= i-k < n, i.e., i-n+1 <= k <= i
        k_min_s = max(-(n-1), i - (n-1))
        k_max_s = min(n-1, i)
        for k in range(k_min_s, k_max_s + 1):
            if k == 0:
                continue
            idx = k + (n - 1)
            delta[idx] -= sign * h_bin[i - k]
        
        new_C = C + delta
        new_max = float(np.max(new_C))
        old_max = float(np.max(C))
        
        if new_max <= old_max or rng.random() < np.exp(-(new_max - old_max) / T):
            h_bin[i] = 1 - h_bin[i]
            C = new_C
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h_bin.copy()
        
        T *= cooling
    
    return best_h, best_c5
