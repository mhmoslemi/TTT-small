        start = rng.randint(0, n - 3)
        length = rng.randint(3, min(25, n - start))
        if rng.random() < 0.3:
            h_p[start:start + length] = rng.random(length)
        else:
            h_p[start:start + length] += 0.15 * rng.randn(length)
    return h_p

def wave_perturb(h, rng):
    n = len(h)
    h_p = h.copy()
    n_waves = rng.randint(1, 3)
    for _ in range(n_waves):
        freq = rng.uniform(2.0, 15.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        t = np.linspace(0, 2, n)
        h_p += amp * np.sin(freq * t + phase)
    return h_p

def swap_perturb(h, rng):
    n = len(h)
    h_p = h.copy()
    n_swaps = rng.randint(2, 6)
    for _ in range(n_swaps):
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        h_p[i], h_p[j] = h_p[j], h_p[i]
    return h_p

def run(seed=42, budget_s=1300, **kwargs):
    global _g_eps, _g_n
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    _g_n = n
    _g_eps = 100.0

    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Progressive SLSQP from parent
    for eps in [20.0, 50.0, 100.0, 300.0, 1000.0]:
        if remaining() < 200:
            break
        h_new = slsqp_optimize(best_h, eps, maxiter=500, ftol=1e-13)
        if np.all(np.isfinite(h_new)):
            h_new = project_feasible(h_new)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 2: Extensive basin hopping with diverse perturbations
    n_pert = 150
    for i in range(n_pert):
        if remaining() < 80:
            break
        pert_type = i % 4
        if pert_type == 0:
            sigma = 0.02 + 0.04 * (i / float(n_pert))
            h_p = best_h + sigma * rng.randn(n)
        elif pert_type == 1:
            h_p = block_perturb(best_h, rng)
        elif pert_type == 2:
            h_p = wave_perturb(best_h, rng)
        else:
            h_p = swap_perturb(best_h, rng)

        h_p = project_feasible(h_p)
        eps_use = 200.0 if i % 2 == 0 else 500.0
        h_new = slsqp_optimize(h_p, eps_use, maxiter=200, ftol=1e-12)
        if np.all(np.isfinite(h_new)):
            h_new = project_feasible(h_new)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 3: Refinement from current best with multiple eps
    for eps in [500.0, 2000.0, 5000.0]:
        if remaining() < 60:
            break
        h_new = slsqp_optimize(best_h, eps, maxiter=500, ftol=1e-14)
        if np.all(np.isfinite(h_new)):
            h_new = project_feasible(h_new)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Coordinate descent
    for rd in range(5):
        if remaining() < 30:
            break
        improved = True
        while improved and remaining() > 30:
            improved = False
            for j in range(n):
                for delta in [0.0005, -0.0005, 0.002, -0.002, 0.005, -0.005]:
                    h_t = best_h.copy()
                    h_t[j] = np.clip(h_t[j] + delta, 0.0, 1.0)
                    h_t = project_feasible(h_t)
                    c5 = evaluate_c5(h_t)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_t.copy()
                        improved = True
                        break

    # Phase 5: More diverse starts from slightly perturbed best
    for i in range(30):
        if remaining() < 30:
            break
        h_start = best_h + 0.005 * rng.randn(n)
        h_start = project_feasible(h_start)
        h_new = slsqp_optimize(h_start, 1000.0, maxiter=300, ftol=1e-14)
        if np.all(np.isfinite(h_new)):
            h_new = project_feasible(h_new)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 6: Final ultra-high precision
    if remaining() > 40:
        h_new = slsqp_optimize(best_h, 10000.0, maxiter=500, ftol=1e-15)
        if np.all(np.isfinite(h_new)):
            h_new = project_feasible(h_new)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    best_h = project_feasible(best_h)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
