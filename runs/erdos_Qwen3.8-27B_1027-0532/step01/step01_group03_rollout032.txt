```python
def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr is indexed 0 to 2n-2, corresponding to lag -(n-1) to n-1
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if 0 <= i + lag < n:
            g += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            g -= h[i - lag]
        grad[i] = g * dx
    
    h_new = h - eta * grad
    h_new = _fix_sum(h_new, n / 2.0)
    return h_new
```

This is O(n) per step (plus O(n log n) or O(n²) for the correlation). Very fast!

```python
def _c5_of(h):

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()

        return h

        order = np.argsort(1.0 - h)

            diff -= add
```
