    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp

            grad[s:] -= wkdx * h[:n-s]
        else:
            r = -s
            grad[r:] += wkdx * omh[:n-r]
            grad[:n-r] -= wkdx * h[r:]
    return grad

def simulated_annealing(h_init, start_temp, end_temp, n_iter, target_sum, rng):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    for i in range(n_iter):
        # Temperature schedule
        t = start_temp * (end_temp / start_temp) ** (i / n_iter)
        
        # Generate move: pick two indices, transfer small amount
        i_idx = rng.integers(n)
        j_idx = rng.integers(n)
        if i_idx == j_idx:
            continue
        delta = (rng.random() - 0.5) * 0.1
        
        # Check bounds
        if h[i_idx] + delta < 0 or h[i_idx] + delta > 1:
            continue
        if h[j_idx] - delta < 0 or h[j_idx] - delta > 1:
            continue
        
        h[i_idx] += delta
        h[j_idx] -= delta
        
        new_c5 = compute_c5(h)
        
        # Accept?
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / t):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[i_idx] -= delta
            h[j_idx] += delta
    
    return best_h, best_c5
