def project_feasible(x, target):
    n = len(x)

        val2 = 0.0

        d[k + n - 1] = val1 - val2
    return d

def lse_optimize(h0, t=50, max_iter=200, penalty=50):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    
    def obj_and_grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        exp_corr = np.exp(corr - m)
        w = exp_corr / exp_corr.sum()
        lse_val = m + np.log(exp_corr.sum()) / t
        
        # Gradient of LSE
        grad = np.zeros(n)
        for k in range(-(n-1), n):
            wk = w[k + n - 1]
            if wk < 1e-10:
                continue
            # ∂corr_k/∂h[p] for all p
            p = np.arange(n)
            # Term 1: (1-h[p-k]) where 0 <= p-k < n
            idx1 = p - k
            m1 = (idx1 >= 0) & (idx1 < n)
            # Term 2: -h[p+k] where 0 <= p+k < n
            idx2 = p + k
            m2 = (idx2 >= 0) & (idx2 < n)
            g = np.zeros(n)
            g[m1] += (1 - x[idx1[m1]])
            g[m2] -= x[idx2[m2]]

        s = x.sum() - tgt
        lse_val += penalty * s * s
        grad += 2 * penalty * s
        
        return lse_val, grad
    
    from scipy.optimize import minimize
    result = minimize(obj_and_grad, h, method='L-BFGS-B',
                      jac=True, bounds=[(0,1)]*n,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    return project_feasible(result.x, tgt)

from scipy.optimize import minimize

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_direction(h, p):
    n = len(h)
    k_vals = np.arange(-(n - 1), n)
    idx1 = p - k_vals
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = p + k_vals
    mask2 = (idx2 >= 0) & (idx2 < n)
    d = np.zeros(2 * n - 1)
    d[mask1] = 1.0 - h[idx1[mask1]]
    d[mask2] -= h[idx2[mask2]]
    return d

def lse_objective_and_grad(x, h, t, penalty, n, dx, tgt):
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    exp_corr = np.exp(corr - m)
    w = exp_corr / exp_corr.sum()
    lse_val = m + np.log(exp_corr.sum()) / t
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if wk < 1e-12:
            continue
        p_arr = np.arange(n)
        idx1 = p_arr - k
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = p_arr + k
        mask2 = (idx2 >= 0) & (idx2 < n)
        g = np.zeros(n)
        g[mask1] += (1.0 - x[idx1[mask1]])
        g[mask2] -= x[idx2[mask2]]
        grad += wk * g
    grad *= dx / t
    s = x.sum() - tgt
    lse_val += penalty * s * s
    grad += 2.0 * penalty * s
    return lse_val, grad

def lse_lbfgs(h0, t=50, max_iter=200, penalty=50, t_budget=None, t_start=None):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.astype(float).copy(), tgt)
    
    def obj_wrapper(x):
        val, g = lse_objective_and_grad(x, h0, t, penalty, n, dx, tgt)
        return val, g
    
    try:
        result = minimize(obj_wrapper, h, method='L-BFGS-B',
                          jac=True, bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = project_feasible(result.x, tgt)
    except Exception:
        h_out = h
    return h_out

    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
