def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_obj_raw(corr, beta):
    m = np.max(corr)
    if beta > 5e4:
        return m
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def lse_obj(h, beta, active_indices=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    if active_indices is not None:
        corr = corr[active_indices]
    return lse_obj_raw(corr, beta)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0.0)
            total_room = room.sum()
            if total_room < 1e-14:
                break
            h += room * (diff / total_room)
        else:
            avail = np.maximum(h, 0.0)
            total_avail = avail.sum()
            if total_avail < 1e-14:
                break
            h -= avail * (-diff / total_avail)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if abs(diff) <= 1e-14:
                    break
    return h

    # Phase 6: Random restarts at current best n
    n_cur = len(best_h)
    for restart in range(15):
        if tl() < 50:
            break
        noise = 0.03 + 0.04 * rng.random()
        h_pert = best_h + rng.normal(0, noise, n_cur)
        h_pert = repair_h(h_pert, n_cur)
        h_opt, c5_opt = slsqp_free(h_pert, n_cur, 200, maxiter=100)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if tl() > 30:
            h_opt2, c5_opt2 = slsqp_free(h_opt, n_cur, 2000, maxiter=100)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 7: Binary-like starts
    if tl() > 60:
        for period in [3, 4, 5, 7]:
            if tl() < 40:
                break
            h_bin = np.zeros(n_cur)
            for i in range(n_cur):
                phase = (i * 2.0 / n_cur * period) % 1.0
                h_bin[i] = 1.0 if phase < 0.5 else 0.0
            h_bin = repair_h(h_bin, n_cur)
            h_opt, c5_opt = slsqp_free(h_bin, n_cur, 100, maxiter=150)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 8: Final polish
    for beta in [5000, 20000, 50000]:
        if tl() < 20:
            break
        h_new, c5_new = slsqp_free(best_h, len(best_h), beta, maxiter=200)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if tl() > 20:
        h_new, c5_new = active_set_optimize(best_h, len(best_h), n_active=3, beta=50000, maxiter=200)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final coordinate descent
    if tl() > 30:
        h_new, c5_new = coord_descent_1d(best_h, len(best_h), n_grid=15, max_sweeps=2)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        if tl() > 10:
            h_new2, c5_new2 = slsqp_free(best_h, len(best_h), 10000, maxiter=100)
            if c5_new2 < best_c5:
                best_h, best_c5 = h_new2, c5_new2

    return best_h, best_c5, len(best_h)
