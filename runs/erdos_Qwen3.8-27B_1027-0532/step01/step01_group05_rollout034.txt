from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_smoothed(h, n, temperature):
    """Compute gradient of smoothed max overlap."""
    overlaps = get_overlaps(h, n)
    max_ov = np.max(overlaps)
    shifted = (overlaps - max_ov) / temperature
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    dx = 2.0 / n
    n_lags = len(overlaps)
    grad = np.zeros(n)
    
    for idx in range(n_lags):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            # O_k = sum_{i=0}^{n-1-k} h[i+k](1-h[i])
            # grad[j] for j >= k: (1-h[j-k]) * w
            # grad[j] for j < n-k: -h[j+k] * w
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += -w * h[k:]
        else:
            l = -k
            # O_{-l} = sum_{i=l}^{n-1} h[i](1-h[i-l])
            # grad[j] for j >= l: (1-h[j-l]) * w  (from h[i] term, i=j)
            # grad[j] for j < n-l: -h[j+l] * w  (from (1-h[i-l]) term, i-l=j)
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] += -w * h[l:]
    
    grad *= dx
    obj = max_ov + temperature * np.log(sum_exp)
    return obj, grad

def nesterov_pgd(h, n, temperatures, max_iters, momentum=0.9, step_init=1.0):
    """Nesterov accelerated projected gradient descent."""
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    velocity = np.zeros(n)
    step = step_init
    
    for it in range(max_iters):
        frac = it / max(1, max_iters - 1)
        temp = temperatures[0] * (temperatures[-1] / temperatures[0]) ** frac
        
        obj, grad = grad_smoothed(h, n, temp)
        
        # Nesterov step
        y = h + momentum * velocity
        obj_y, grad_y = grad_smoothed(y, n, temp)
        
        velocity = momentum * velocity - step * grad_y
        h_new = h + velocity
        h_new = project_onto_constraint(h_new, n)
        
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive step
        if it % 100 == 0 and it > 0:
            step *= 0.99
    
    return best_h, best_c5

    def objective_and_grad(x):

        sum_penalty = 100.0 * (np.sum(x) - n/2.0)**2

    obj = max_ov + temperature * np.log(sum_exp)
    return obj, grad

def optimize_pgd_momentum(h0, n, temps, max_iters, step=0.1, decay=0.999):
    """PGD with momentum and annealed temperature."""
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    vel = np.zeros(n)
    
    for it in range(max_iters):
        frac = it / max(1, max_iters - 1)
        temp = temps[0] * (temps[-1] / temps[0]) ** frac
        
        obj, grad = grad_smoothed(h, n, temp)
        
        # Momentum step
        vel = 0.9 * vel - step * grad
        h_new = h + vel
        h_new = project_onto_constraint(h_new, n)
        
        c5 = compute_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        
        # Decay step
        step *= decay
    
    return best_h, best_c5

def optimize_slsqp(h0, n, temperature, max_iter=300):
    """SLSQP optimization with smoothed objective."""
    h0 = h0.copy().astype(float)
    bounds = [(0.0, 1.0)] * n
    
    def obj_fn(x):
        o, _ = grad_smoothed(x, n, temperature)
        return o
    
    def grad_fn(x):
        _, g = grad_smoothed(x, n, temperature)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj_fn, h0, method='SLSQP', jac=grad_fn,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(np.clip(res.x, 0.0, 1.0), n)
        return h_opt
    except Exception:
        return h0.copy()

def binary_sa(h0, n, max_iters, t_start, t_end, rng):
    """Binary swap simulated annealing."""
    # Create initial binary solution
    h_bin = np.zeros(n)
    order = np.argsort(-h0)
    h_bin[order[:n//2]] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin, n)
    current_h = h_bin.copy()
    current_c5 = best_c5
    
    dx = 2.0 / n
    
    for it in range(max_iters):
        temp = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        
        ones = np.where(current_h == 1.0)[0]
        zeros = np.where(current_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        current_h[i] = 0.0
        current_h[j] = 1.0
        new_c5 = compute_c5(current_h, n)
        
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            current_h[i] = 1.0
            current_h[j] = 0.0
    
    return best_h, best_c5
