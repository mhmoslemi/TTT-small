    return np.max(corr) * dx

    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_max_weights(x, alpha):
    m = np.max(x)
    exps = np.exp(alpha * (x - m))
    return exps / np.sum(exps)

def compute_gradient(h, alpha):

    dx = 2.0 / n
    corr = compute_correlations(h)
    s = corr * dx
    weights = smooth_max_weights(s, alpha)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        js = max(0, lag)
        je = min(n, lag + n)
        if js < je:
            grad[js:je] += w * (1.0 - h[js - lag:je - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + lag:je2 + lag]

        if abs(diff) < 1e-12:

        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)

        x = x_new
        t = t_new

def compute_correlations(h):

    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = np.max(x)
    vals = alpha * (x - m)
    return m + np.log(np.sum(np.exp(vals))) / alpha

def get_softmax_weights(x, alpha):
    m = np.max(x)
    vals = alpha * (x - m)
    exps = np.exp(vals)
    return exps / np.sum(exps)

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    s = corr * dx
    weights = get_softmax_weights(s, alpha)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        js = max(0, lag)
        je = min(n, lag + n)
        if js < je:

        je2 = min(n, n - lag)

    )
    return res.x

        x = x_new
        t = t_new

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    vals = alpha * (x - m)
    return m + np.log(np.sum(np.exp(vals))) / alpha


def get_softmax_weights(x, alpha):
    m = np.max(x)
    vals = alpha * (x - m)
    exps = np.exp(vals)
    return exps / np.sum(exps)


def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    s = corr * dx
    weights = get_softmax_weights(s, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        js = max(0, lag)
        je = min(n, lag + n)
        if js < je:
            grad[js:je] += w * (1.0 - h[js - lag:je - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + lag:je2 + lag]
    return grad * dx


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def obj_pen(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    s = corr * dx
    val = smooth_max_val(s, alpha)
    sd = np.sum(h) - n / 2.0
    val += mu * sd * sd
    return val


def grad_pen(h, alpha, mu):
    n = len(h)
    grad = compute_gradient(h, alpha)
    sd = np.sum(h) - n / 2.0
    grad += 2.0 * mu * sd * np.ones(n)
    return grad


def lbfgsb_run(h0, alpha, mu, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj_pen, h0, jac=grad_pen,
        args=(alpha, mu),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50}
    )
    return res.x


def fista_run(h0, alpha, max_iter, lr_init=None):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0)
    z = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = smooth_max_val(compute_correlations(x) * dx, alpha)
    if lr_init is None:
        lr = 0.5 / (alpha * 2.0 * dx * n + 1.0)
    else:
        lr = lr_init
    for i in range(max_iter):
        grad = compute_gradient(z, alpha)
        step = lr
        x_new = project_feasible(z - step * grad)
        val_new = smooth_max_val(compute_correlations(x_new) * dx, alpha)
        backtracks = 0
        while val_new > best_obj + 1e-12 and backtracks < 15 and step > 1e-12:
            step *= 0.5
            x_new = project_feasible(z - step * grad)
            val_new = smooth_max_val(compute_correlations(x_new) * dx, alpha)
            backtracks += 1
        if val_new < best_obj:
            best_obj = val_new
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = project_feasible(x_new + ((t - 1.0) / t_new) * (x_new - x))
        x = x_new
        t = t_new
        lr = step
    return best_x


def make_random_binary(h_base, rng, n, frac):
    h = h_base.copy()
    mask = rng.rand(n) < frac
    n_mask = int(np.sum(mask))
    if n_mask > 0:
        h[mask] = rng.rand(n_mask)
    return project_feasible(h)

    shift = rng.randint(1, max(2, n // 10))
    h = np.roll(h_base, shift)
    return project_feasible(h)


def make_gauss_perturb(h_base, rng, n, scale):
    h = h_base + rng.randn(n) * scale
    return project_feasible(h)
