import numpy as np
import time


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def get_weights(corr, alpha):
    scaled = alpha * corr
    m = float(np.max(scaled))
    shifted = scaled - m
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return e / s


def compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = get_weights(corr, alpha)
    grad = np.zeros(n)
    for m_val in range(-(n - 1), n):
        wm = w[m_val + n - 1]
        if wm < 1e-15:
            continue
        j0 = max(0, -m_val)
        j1 = min(n, n - m_val)
        if j0 < j1:
            grad[j0:j1] += wm * (1.0 - h[j0 + m_val:j1 + m_val])
        j0b = max(0, m_val)
        j1b = min(n, n + m_val)
        if j0b < j1b:
            grad[j0b:j1b] -= wm * h[j0b - m_val:j1b - m_val]
    return grad * dx


def project_feasible(h, target):
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
