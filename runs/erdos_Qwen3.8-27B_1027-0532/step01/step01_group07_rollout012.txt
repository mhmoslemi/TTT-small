Note that c[l] = dx * sum_j h[j] * (1-h[j+l])

If we fix all of h except h[i], then c[l] is linear in h[i]:
c[l] = a_l + b_l * h[i]

where:
- a_l = dx * sum_{j≠i, j≠i±l valid} h[j] * (1-h[j+l]) (the part not involving h[i])
- b_l = dx * (contribution of h[i] to c[l])

The contribution of h[i] to c[l] comes from:
- h[i] appears as h[j] when j=i: contributes dx * (1-h[i+l])
- h[i] appears as h[j+l] when j+l=i, i.e., j=i-l: contributes dx * (-h[i-l])

So b_l = dx * [(1-h[i+l]) * (i+l valid) - h[i-l] * (i-l valid)]

This is exactly the gradient component we computed above!

So the problem is: minimize max_l (a_l + b_l * h[i]) subject to 0 ≤ h[i] ≤ 1

For a fixed i, the optimal h[i] is the one that minimizes the maximum of these linear functions, which is a 1D problem (can be solved by bisection or finding the intersection points).

This suggests an **alternating coordinate descent** approach:
1. For each i (cycling through), compute the optimal h[i] given all others fixed
2. Repeat until convergence

The 1D subproblem: Given a_l and b_l for all l, find h[i] ∈ [0,1] that minimizes max_l (a_l + b_l * h[i]).

This is a classic "minimize the maximum of linear functions in 1D" problem. The optimal point is either at the boundary (0 or 1) or at an intersection point of two of the lines.

We can solve this by:
1. Compute all lines f_l(t) = a_l + b_l * t for t = h[i]
2. Find t ∈ [0,1] that minimizes max_l f_l(t)
3. This can be done by finding the "upper envelope" of the lines and then finding the minimum of that envelope

For n=44 and 87 lags, we have 87 lines. The upper envelope of 87 lines can be found in O(87 log 87) time. Then finding the minimum of the envelope is O(87).

    n = len(h)

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_smooth_and_grad(h, alpha):
    """Compute smooth approximation to max correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1
    
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(alpha * shifted)
    w = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[center + l]
        if wl < 1e-12:
            continue
        if l >= 0:
            i1 = np.arange(0, n - l)
            p1 = (1.0 - h[i1 + l])
            i2 = np.arange(l, n)
            p2 = h[i2 - l]
        else:
            i1 = np.arange(-l, n)
            p1 = (1.0 - h[i1 + l])
            i2 = np.arange(0, n + l)
            p2 = h[i2 - l]
        grad[i1] += wl * dx * p1
        grad[i2] -= wl * dx * p2
    
    return smooth_val, grad, w

    """Nesterov accelerated gradient descent with projection."""

                alpha = min(alpha * 1.5, 10000.0)
                stale = 0
            elif stale == 100:
                alpha = min(alpha * 2.0, 20000.0)
                stale = 0
            elif stale > 200:
                break
        
        # Gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        
        v = momentum * v - lr * grad_normed
        h_new = project_feasible(h + v, target)
        
        # Backtracking line search
        for _ in range(10):

            if c5_try <= c5_current or np.linalg.norm(h_try - h) < 1e-10:
                h = h_try
                break
            v = 0.5 * v
        else:
            h = h_new
        
    return best_h, best_c5

def check_feasibility(T, h_init, n, maxiter=500):

    dx = 2.0 / n
    
    def obj(val):
        h = val
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        violations = np.maximum(0, c - T)
        return np.sum(violations**2) + 100.0 * (np.sum(h) - target)**2

        h = val
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        violations = np.maximum(0, c - T)

        # d/dh_i sum_k max(0, c_k - T)^2 = 2 * sum_k max(0, c_k-T) * dc_k/dh_i
        active = np.where(c > T)[0]

                i2 = np.arange(0, n + l)
                g[i2] -= coeff * dx * h[i2 - l]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _obj_func(val, alpha, n, pen):
    h = val
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = c - c_max
    val_obj = c_max + np.log(np.exp(alpha * shifted).sum()) / alpha
    target = n / 2.0
    val_obj += pen * (h.sum() - target) ** 2
    return val_obj

def _grad_func(val, alpha, n, pen):
    h = val
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(alpha * shifted)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    center = n - 1
    for l in range(-(n - 1), n):
        wl = w[center + l]
        if wl < 1e-14:
            continue
        ip = i_p_valid = (l >= 0)
        if l >= 0:
            if n - l > 0:
                idx1 = np.arange(0, n - l)
                grad[idx1] += wl * dx * (1.0 - h[idx1 + l])
            if l < n:
                idx2 = np.arange(l, n)
                grad[idx2] -= wl * dx * h[idx2 - l]
        else:
            il = -l
            if il < n:
                idx1 = np.arange(il, n)
                grad[idx1] += wl * dx * (1.0 - h[idx1 + l])
            if n + l > 0:
                idx2 = np.arange(0, n + l)
                grad[idx2] -= wl * dx * h[idx2 - l]
    target = n / 2.0
    grad += 2.0 * pen * (h.sum() - target)
    return grad

def _run_lbfgs(h0, n, alpha, maxiter, pen):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_func, h0, args=(alpha, n, pen),
                   jac=_grad_func, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    return res.x

def _solve_1d(a, s):
    def _slope_range(t):
        vals = a + s * t
        max_val = vals.max()
        tol = 1e-10
        active = np.where(np.abs(vals - max_val) <= tol)[0]
        return s[active].min(), s[active].max()
    min_s0, max_s0 = _slope_range(0.0)
    if min_s0 >= 0:
        return 0.0
    min_s1, max_s1 = _slope_range(1.0)
    if max_s1 <= 0:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        mn, mx = _slope_range(mid)
        if mn <= 0.0 <= mx:
            break
        elif mx < 0.0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) * 0.5

def _coord_descent(h, n, max_cycles, deadline):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    c_curr = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = float(c_curr.max())
    best_h = h.copy()
    lam = 0.0
    for _ in range(max_cycles):
        if deadline is not None and time.time() > deadline:
            break
        changed = False
        for i in range(n):
            b = np.zeros(2 * n - 1)
            for l in range(-(n - 1), n):
                v = 0.0
                ip = i + l
                im = i - l
                if 0 <= ip < n:
                    v += (1.0 - h[ip])
                if 0 <= im < n:
                    v -= h[im]
                b[center + l] = dx * v
            a = c_curr - b * h[i]
            s = b + lam
            t_star = _solve_1d(a, s)
            delta = t_star - h[i]
            if abs(delta) > 1e-12:
                h[i] = t_star
                c_curr = c_curr + delta * b
                changed = True
        sum_err = h.sum() - target
        lam += 0.4 * sum_err
        lam = np.clip(lam, -3.0, 3.0)
        c5_new = float(c_curr.max())
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
        if not changed:
            break
    best_h = project_feasible(best_h, target)
    return best_h, compute_c5(best_h)

def _make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 7)
    for b in range(n_blocks):
        start = int(n * b / n_blocks)
        length = max(1, n // (2 * n_blocks) + rng.randint(-2, 3))
        end = min(n, start + length)
        h[start:end] = 1.0
    return project_feasible(h, n / 2.0)

def _make_interleave_init(n, rng):
    h = np.zeros(n)
    period = rng.randint(2, 5)
    duty = rng.randint(1, period)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)
