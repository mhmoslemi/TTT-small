```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def project_onto_constraint(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(y - tau, 0.0, 1.0)

def compute_gradient(h, weights, n):
    """Compute gradient of smooth max objective."""
    dx = 2.0 / n
    n2 = 2 * n - 1
    lags = np.arange(n2) - (n - 1)
    grad = np.zeros(n)
    for p in range(n):
        idx1 = p + lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.sum(weights[mask1] * (1.0 - h[idx1[mask1]]))
        idx2 = p - lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.sum(weights[mask2] * h[idx2[mask2]])
        grad[p] = dx * (t1 - t2)
    return grad

def pgd_step(h, n, beta, lr, max_line_search=10):
    """One projected gradient descent step with line search."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    
    # Compute smooth max and weights
    scaled = beta * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    sum_exps = np.sum(exps)
    weights = exps / sum_exps
    smooth_val = (m + np.log(sum_exps)) / beta * dx
    
    # Compute gradient
    grad = compute_gradient(h, weights, n)
    
    # Line search
    best_step_h = h.copy()
    best_val = smooth_val
    lr_try = lr
    for _ in range(max_line_search):
        new_h = project_onto_constraint(h - lr_try * grad, n)
        new_corr = np.correlate(new_h, 1.0 - new_h, mode="full")
        new_scaled = beta * new_corr
        new_m = np.max(new_scaled)
        new_exps = np.exp(new_scaled - new_m)
        new_val = (new_m + np.log(np.sum(new_exps))) / beta * dx
        if new_val < best_val:
            best_step_h = new_h
            best_val = new_val
            lr_try *= 1.5
        else:
            lr_try *= 0.5
            break
    
    return best_step_h, best_val

def pgd_optimize(h, n, betas, iters_per_beta, initial_lr=None):
    """Run projected gradient descent with increasing beta."""
    h_curr = h.copy()
    for beta in betas:
        if initial_lr is None:
            lr = 2.0 / beta
        else:
            lr = initial_lr
        for i in range(iters_per_beta):
            h_curr, val = pgd_step(h_curr, n, beta, lr)
        # Evaluate actual C5
        c5 = evaluate_c5(h_curr)
    return h_curr, c5
```

```python
    h = np.clip(h, 0.0, 1.0)
```

```python
def topk_subgradient_optimize(h, n, k=5, iters=500, lr=0.01):
    h_curr = h.copy()
    best_c5 = evaluate_c5(h_curr)
    best_h = h_curr.copy()
    for i in range(iters):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        # Find top-k lags
        top_indices = np.argsort(corr)[-k:]
        # Compute subgradient: average of gradients of top-k correlations
        grad = np.zeros(n)
        for idx in top_indices:
            lag = idx - (n-1)
            for p in range(n):
                # d(corr[idx])/d(h[p])
                term = 0.0
                if 0 <= p + lag < n:
                    term += (1.0 - h_curr[p + lag])
                if 0 <= p - lag < n:
                    term -= h_curr[p - lag]
                grad[p] += term

        new_h = project_onto_constraint(h_curr - lr * grad, n)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        h_curr = new_h
```

```python
    g = np.zeros(n)
    if j >= 0:
        # (1-h[p+j]) for p=0..n-1-j
        g[:n-j] += (1.0 - h[j:])
        # -h[p-j] for p=j..n-1
        g[j:] -= h[:n-j]
    else:
        jp = -j
        # (1-h[p-jp]) for p=jp..n-1
        g[jp:] += (1.0 - h[:n-jp])
        # -h[p+jp] for p=0..n-1-jp
        g[:n-jp] -= h[jp:]
    return g
```
