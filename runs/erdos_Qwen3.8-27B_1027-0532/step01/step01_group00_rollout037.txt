def compute_gradient(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    scaled = corr * dx
    # softmax
    max_s = np.max(scaled)
    exp_s = np.exp(alpha * (scaled - max_s))
    w = exp_s / np.sum(exp_s)
    
    # grad[i] = sum_m w[m] * ((1-h[i-(m-(n-1))]) - h[i+(m-(n-1))])
    # Let k = m-(n-1), range -(n-1) to (n-1)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(-(n-1), n):
            m = k + (n - 1)
            # term 1: 1-h[i-k] if 0 <= i-k < n
            t1 = 0.0
            if 0 <= i - k < n:

Now let me think about the FISTA implementation more carefully.

For FISTA, I need:
1. The smooth objective f(h) = (1/alpha) * log(sum exp(alpha * c_k * dx))
2. The gradient ∇f(h)
3. The projection onto the feasible set
4. A step size (1/L where L is the Lipschitz constant)

For the Lipschitz constant, I'll use backtracking:
- Start with L = initial guess
- While f(x - (1/L)∇f(y)) > f(y) + ∇f(y)ᵀ(x-y) + (L/2)||x-y||²:
  - L *= 2

This is the standard backtracking line search for L.

Let me just use direct numpy operations (correlate, convolve) which are efficient enough.

from scipy.optimize import minimize
import time

def compute_corr(h):
    """Compute cross-correlation h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    """Compute C5 = max_k corr_k * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def softmax_weights(corr, dx, alpha):
    """Compute softmax weights for the log-sum-exp."""
    scaled = corr * dx
    m = np.max(scaled)
    exp_s = np.exp(alpha * (scaled - m))
    return exp_s / np.sum(exp_s)

def smooth_obj(h, alpha, dx):
    """Log-sum-exp smooth approximation to max(corr*dx)."""
    corr = compute_corr(h)
    scaled = corr * dx
    m = np.max(scaled)
    return m + np.log(np.sum(np.exp(alpha * (scaled - m)))) / alpha

def smooth_grad(h, alpha, dx):
    """Gradient of the smooth objective."""
    n = len(h)
    corr = compute_corr(h)
    w = softmax_weights(corr, dx, alpha)
    S1 = np.convolve(1.0 - h, w, mode='full')
    S2 = np.correlate(h, w, mode='full')
    offset = n - 1
    return dx * (S1[offset:offset+n] - S2[offset:offset+n])

    """Project y onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(y)

            lo = mid
        else:
            hi = mid

    """FISTA (accelerated projected gradient) for the smooth problem."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    L = 10.0 / dx  # initial guess for Lipschitz constant
    best_local = h0.copy()
    best_local_obj = smooth_obj(h0, alpha, dx)
    
    for it in range(max_iter):
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = project_feasible(y, target_sum)
        
        # Gradient at y
        gy = smooth_grad(y, alpha, dx)
        
        # Backtracking line search
        while True:
            step = 1.0 / L
            h_new = project_feasible(y - step * gy, target_sum)
            obj_y = smooth_obj(y, alpha, dx)
            obj_new = smooth_obj(h_new, alpha, dx)
            # Armijo condition
            if obj_new <= obj_y + np.dot(gy, h_new - y) + 0.5 * L * np.sum((h_new - y)**2) + 1e-15:
                break
            L *= 2.0
            if L > 1e15:
                break
        
        h_prev = h
        h = h_new
        t = t_new
        
        # Track best
        obj_h = smooth_obj(h, alpha, dx)
        if obj_h < best_local_obj:
            best_local_obj = obj_h
            best_local = h.copy()
        
        # Convergence check
        if it > 100 and abs(obj_y - obj_new) < tol:
            break
    
    return best_local
