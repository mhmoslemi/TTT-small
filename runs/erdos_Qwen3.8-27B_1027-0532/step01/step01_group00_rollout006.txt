```python
def project_onto_simplex_box(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Use bisection on lambda
    # We want sum(clip(h - lambda, 0, 1)) = target_sum
    # f(lambda) = sum(clip(h - lambda, 0, 1)) is decreasing in lambda
    
    # Find range for lambda
    lo = h.min() - 1  # all zeros
    hi = h.max() + 1  # all ones
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return proj
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx


def project_box_sum(h, target):
    n = len(h)
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def project_sum_simple(h, target):
    n = len(h)
    h = h.astype(float).copy()
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_grad(h, alpha, active_indices):
    corr = compute_corr(h)
    n = len(h)
    vals = corr[active_indices]
    m = np.max(vals)
    shifted = alpha * (vals - m)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / np.sum(exp_vals)
    smooth_val = m + (max_shifted + np.log(np.sum(exp_vals))) / alpha

    grad = np.zeros(n)
    for idx in range(len(active_indices)):
        if w[idx] < 1e-12:
            continue
        k_idx = active_indices[idx]
        k = k_idx - (n - 1)
        if k == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[k:] += w[idx] * (1.0 - h[:n - k])
                grad[:n - k] += -w[idx] * h[k:]
        else:
            p = -k
            if p < n:
                grad[:n - p] += w[idx] * (1.0 - h[p:])
                grad[p:] += -w[idx] * h[:n - p]
    return smooth_val, grad


def _penalty_wrap(h, alpha, mu, n_offset, active_indices):
    n = len(h)
    val, grad = smooth_max_grad(h, alpha, active_indices)
    s = h.sum()
    target = n / 2.0
    diff = s - target
    penalty = mu * diff * diff
    grad_pen = 2.0 * mu * diff
    return val + penalty, grad + grad_pen


def _obj_func(h, alpha, mu, n_offset, active_indices):
    return _penalty_wrap(h, alpha, mu, n_offset, active_indices)[0]


def _grad_func(h, alpha, mu, n_offset, active_indices):
    return _penalty_wrap(h, alpha, mu, n_offset, active_indices)[1]


def run_bfgs_obj(h0, alpha, mu, maxiter, n, active_indices):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        _obj_func, h0, method='L-BFGS-B', bounds=bounds,
        jac=_grad_func, args=(alpha, mu, n, active_indices),
        options={'maxiter': maxiter, 'gtol': 1e-9, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x


def try_improve(h, score, alpha, mu, maxiter, n, active_indices, best_h, best_score):
    res_x = run_bfgs_obj(h, alpha, mu, maxiter, n, active_indices)
    if res_x is not None:
        h_new = project_box_sum(res_x, n / 2.0)
        if not np.all(np.isfinite(h_new)):
            return best_h, best_score, False
        s = evaluate_score(h_new)
        if np.isfinite(s) and s < best_score:
            return h_new, s, True
    return best_h, best_score, False


def binarize_and_refine(h, best_score, mu, n_active, budget_remaining):
    n = len(h)
    h_bin = h.copy()
    h_bin[h_bin < 0.05] = 0.0
    h_bin[h_bin > 0.95] = 1.0
    free_mask = (h_bin > 0.0) & (h_bin < 1.0)
    n_free = np.sum(free_mask)
    if n_free < 5:
        return h, best_score
    target_free = n / 2.0 - np.sum(h_bin[~free_mask])
    if target_free < 0 or target_free > n_free:
        return h, best_score
    h_work = h.copy()
    h_work[free_mask] = target_free / n_free
    h_work = project_box_sum(h_work, n / 2.0)
    corr = compute_corr(h_work)
    top_k = np.argsort(corr)[-n_active:]
    for alpha in [200, 500, 1000]:
        if budget_remaining() < 30:

        h_work, s, improved = try_improve(

        if improved:

    s_final = evaluate_score(h_work)
    if np.isfinite(s_final) and s_final < best_score:
        return h_work, s_final
    return h, best_score
```
