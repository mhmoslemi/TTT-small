    a = h[i]
    b = h[j]
    # t ranges so that 0 ≤ a+t ≤ 1 and 0 ≤ b-t ≤ 1
    t_lo = max(-a, -(1-b))  # a+t ≥ 0 and b-t ≤ 1
    t_hi = min(1-a, b)      # a+t ≤ 1 and b-t ≥ 0

        h[j] = b - t
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    
    # Ternary search on [t_lo, t_hi]

    for _ in range(n_ternary):

        m2 = hi - (hi - lo) / 3.0

        else:

    x_old = np.linspace(0.0, 1.0, len(h))
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)


def slsqp_optimize(h, n, target_sum, alpha, maxiter, ftol):
    global _g_target_sum, _g_n
    _g_n = n
    _g_target_sum = target_sum
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(obj_lse, h, args=(alpha,), method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target_sum)


def slsqp_optimize_soft(h, n, target_sum, alpha, maxiter, ftol):
    global _g_target_sum, _g_n
    _g_n = n
    _g_target_sum = target_sum
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(obj_top_soft, h, args=(alpha,), method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target_sum)


def coordinated_perturb(h, n, target_sum, rng, strength):
    h_new = h.copy()
    for _ in range(5):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-strength, strength)
        h_new[i] += delta
        h_new[j] -= delta
    return project_feasible(h_new, target_sum)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target_sum, _g_n
    start = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - 25.0 - (time.time() - start)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    ts_parent = n_parent / 2.0
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick optimize at parent resolution
    for alpha in [50, 200, 500]:
        if _tl() < 30:
            break
        try:
            h_new = slsqp_optimize(best_h, n_parent, ts_parent, alpha, 2000, 1e-15)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Try higher resolutions with annealed optimization
    resolutions = [500, 800, 1200]
    for n_r in resolutions:
        if _tl() < 150:
            break
        ts_r = n_r / 2.0
        h_r = upsample_h(best_h, n_r)
        h_r = project_feasible(h_r, ts_r)
        score_r = compute_c5(h_r)

        alpha_schedule = [10, 50, 200, 500, 1000]
        for alpha in alpha_schedule:
            if _tl() < 60:
                break
            try:
                maxiter = 3000 if n_r <= 800 else 1500
                h_new = slsqp_optimize(h_r, n_r, ts_r, alpha, maxiter, 1e-14)
                s = compute_c5(h_new)
                if s < score_r:
                    score_r = s
                    h_r = h_new
            except Exception:
                pass

        if score_r < best_score:
            best_score = score_r
            best_h = h_r

    # Phase 3: Try the软装 (soft top) objective at best resolution
    n_best = len(best_h)
    ts_best = n_best / 2.0
    for alpha in [100, 300, 800]:
        if _tl() < 40:
            break
        try:
            h_new = slsqp_optimize_soft(best_h, n_best, ts_best, alpha, 2000, 1e-14)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    # Phase 4: Random restarts with local optimization
    bounds_best = [(0.0, 1.0)] * n_best
    _g_n = n_best
    _g_target_sum = ts_best
    constraints_best = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    noise_scales = [0.02, 0.04, 0.06, 0.1, 0.15]
    alpha_choices = [50, 100, 300, 500]

    while _tl() > 80:
        noise = float(rng.choice(noise_scales))
        h_pert = best_h + rng.randn(n_best) * noise
        h_pert = project_feasible(h_pert, ts_best)

        # Also try coordinated perturbation sometimes
        if rng.random() < 0.3:
            h_pert = coordinated_perturb(best_h, n_best, ts_best, rng, 0.1)

        alpha = float(rng.choice(alpha_choices))
        try:
            maxiter = 800 if _tl() > 200 else 400
            res = minimize(obj_lse, h_pert, args=(alpha,), method='SLSQP',
                           bounds=bounds_best, constraints=constraints_best,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_feasible(res.x, ts_best)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    # Phase 5: Try upsampled random starts at higher resolution
    if _tl() > 120:
        n_high = min(n_best + 400, 2000)
        if n_high % 2 == 0:
            n_high += 1
        ts_high = n_high / 2.0
        for attempt in range(3):
            if _tl() < 60:
                break
            h_try = upsample_h(best_h, n_high)
            h_try = project_feasible(h_try, ts_high)
            # Add small random perturbation
            h_try += rng.randn(n_high) * 0.02
            h_try = project_feasible(h_try, ts_high)

            for alpha in [50, 200, 500]:
                if _tl() < 30:
                    break
                try:
                    _g_n = n_high
                    _g_target_sum = ts_high
                    bnd_h = [(0.0, 1.0)] * n_high
                    cons_h = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
                    res = minimize(obj_lse, h_try, args=(alpha,), method='SLSQP',
                                   bounds=bnd_h, constraints=cons_h,
                                   options={'maxiter': 1500, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_high)
                    s = compute_c5(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new
                    h_try = h_new
                except Exception:
                    pass

    # Phase 6: Final high-alpha polish
    n_final = len(best_h)
    ts_final = n_final / 2.0
    _g_n = n_final
    _g_target_sum = ts_final
    bnd_f = [(0.0, 1.0)] * n_final
    cons_f = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    for alpha in [500, 1000, 2000, 5000]:
        if _tl() < 15:
            break
        try:
            res = minimize(obj_lse, best_h, args=(alpha,), method='SLSQP',
                           bounds=bnd_f, constraints=cons_f,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts_final)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    # Phase 7: Very targeted - minimize only the top correlations
    if _tl() > 30:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        _g_n = n_f
        _g_target_sum = ts_f
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_k = np.sum(corr > np.max(corr) * 0.99)
        top_k = max(int(top_k), 5)

        def _obj_focused(h_f):
            c = np.correlate(h_f, 1.0 - h_f, mode='full')
            c_sorted = np.sort(c)
            return np.sum(c_sorted[-top_k:]) / float(top_k)

        bnd_ff = [(0.0, 1.0)] * n_f
        cons_ff = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_focused, best_h, method='SLSQP',
                           bounds=bnd_ff, constraints=cons_ff,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts_f)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    h_final = project_feasible(best_h, ts_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
