        if remaining(0.82) > 20:
            h_sp2 = selpolish(h_res, tm=min(50, remaining(0.82) * 0.4), n_restart=1)
            c_sp2 = eval_c5(h_sp2)
            if c_sp2 < c_res:
                h_res, c_res = h_sp2, c_sp2

        if c_res < best_c5:
            best, best_c5 = h_res, c_res

        # Also try structured inits at this resolution
        if remaining(0.82) > 30:
            for init_i in range(2):
                if remaining(0.82) < 20:
                    break
                h_si = make_structured_init(n_t, rng)
                h_si_opt = smooth_optimize(h_si, tm=min(30, remaining(0.82) * 0.3),
                                           tau0=10.0, tau_end=300.0)
                c_si = eval_c5(h_si_opt)
                if remaining(0.82) > 15 and c_si < best_c5 + 0.01:
                    h_si_pol = selpolish(h_si_opt, tm=min(40, remaining(0.82) * 0.4), n_restart=1)
                    c_si_pol = eval_c5(h_si_pol)
                    if c_si_pol < best_c5:
                        best, best_c5 = h_si_pol, c_si_pol

    # Phase 6: Perturbation search on best
    for attempt in range(10):
        if remaining(0.92) < 8:
            break
        h_p = perturbed_h(best, rng, scale=0.03)
        h_po = smooth_optimize(h_p, tm=min(20, remaining(0.92) * 0.3),
                               tau0=50.0, tau_end=500.0, lr_init=0.03)
        c_po = eval_c5(h_po)
        if c_po < best_c5:
            best, best_c5 = h_po, c_po
            if remaining(0.92) > 15:
                h_po2 = selpolish(best, tm=min(20, remaining(0.92) * 0.3), n_restart=1)
                c_po2 = eval_c5(h_po2)
                if c_po2 < best_c5:
                    best, best_c5 = h_po2, c_po2

    # Phase 7: Final intensive polish
    if remaining(0.99) > 20:
        h_f = selpolish(best, tm=min(100, remaining(0.97)), n_restart=3)
        cf = eval_c5(h_f)
        if cf < best_c5:
            best, best_c5 = h_f, cf

    if remaining(0.995) > 15:
        h_f2 = smooth_optimize(best, tm=min(40, remaining(0.99)),
                               tau0=100.0, tau_end=800.0, lr_init=0.01, momentum=0.95)
        cf2 = eval_c5(h_f2)
        if cf2 < best_c5:
            best, best_c5 = h_f2, cf2

    if remaining(0.998) > 10:
        h_f3 = selpolish(best, tm=min(20, remaining(0.995)), n_restart=1)
        cf3 = eval_c5(h_f3)
        if cf3 < best_c5:
            best, best_c5 = h_f3, cf3

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
