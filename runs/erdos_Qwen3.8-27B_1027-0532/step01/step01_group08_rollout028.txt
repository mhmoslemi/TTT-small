import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5_softmax(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    max_tc = np.max(tc)
    weights = np.exp(tc - max_tc)
    weights /= np.sum(weights)
    obj = (max_tc / tau + np.log(np.sum(weights)) / tau) * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-15:
            continue
        lag = m - (n - 1)
        w = weights[m] / dx  # will multiply by dx at end
        # d/dh_i of corr[m] = ...
        if lag >= 0:
            # corr[m] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag]
            #           -h[i-lag] for i in [lag, n-1]
            j_lo = 0
            j_hi = n - 1 - lag
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo+lag:j_hi+lag+1])
            j2_lo = lag
            j2_hi = n - 1
            if j2_lo <= j2_hi:
                grad[j2_lo:j2_hi+1] += w * (-h[j2_lo-lag:j2_hi-lag+1])
        else:
            # lag < 0, let r = -lag
            # corr[m] = sum_{i=r}^{n-1} h[i]*(1-h[i-r])
            # d/dh[i] = (1-h[i-r]) for i in [r, n-1]
            #           -h[i+r] for i in [0, n-1-r]
            r = -lag
            j_lo = r
            j_hi = n - 1
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo-r:j_hi-r+1])
            j2_lo = 0
            j2_hi = n - 1 - r
            if j2_lo <= j2_hi:
                grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+r:j2_hi+r+1])
    return obj, grad * dx

    beta2 = 0.999

    base_lr = 0.02
    tau = tau0
    t = 0
    stall_count = 0
    stall_thresh = 50
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        obj, grad = grad_c5_softmax(h, tau)

        v = beta2 * v + (1 - beta2) * (grad ** 2)

def nesterov_optimize(h0, target_sum, deadline, max_iter=10000):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()  # look-ahead point
    v = np.zeros(n)  # velocity
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    mu = 0.9  # momentum
    base_lr = 0.05
    tau = 30.0
    t = 0
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        momentum = (1 + mu) / 2  # or use (1+sqrt(1+4*mu^2))/2
        # Nesterov step
        obj, grad = grad_c5_softmax(y, tau)
        v = mu * v + grad
        h_new = h - base_lr / (np.sqrt(t) + 1) * v
        y = h_new + momentum * (h_new - h)
        h = project_onto_constraint(h_new, target_sum)
        y = project_onto_constraint(y, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 80:
                base_lr *= 0.8
                tau = min(tau * 1.3, 200.0)
                stall_count = 0
                if base_lr < 0.0005:
                    break
    
    return best_h, best_c5

    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_single_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        k = lag
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] += (-h[:n-k])
    else:
        r = -lag
        grad[r:] += (1.0 - h[:n-r])
        grad[:n-r] += (-h[r:])
    return grad

def soft_max_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = corr * tau
    max_tc = np.max(tc)
    exp_tc = np.exp(tc - max_tc)
    sum_exp = np.sum(exp_tc)
    obj = (max_tc + np.log(sum_exp)) / tau * dx
    weights = exp_tc / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        g = grad_single_lag(h, lag)
        grad += w * g
    return obj, grad * dx

def multi_lag_subgrad(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_indices:
        lag = idx - (n - 1)
        grad += grad_single_lag(h, lag)
    return corr[idx.max()] * dx, grad * dx / top_k

def adam_optimize(h0, target_sum, deadline, tau0=10.0, tau_max=400.0, base_lr=0.015, max_iter=50000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    tau = tau0
    t = 0
    stall = 0
    stall_thresh = 60
    lr_scale = 1.0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        obj, grad = soft_max_obj_grad(h, tau)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = base_lr * lr_scale / (1.0 + 0.001 * t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall >= stall_thresh:
            tau = min(tau * 1.4, tau_max)
            lr_scale *= 0.65
            stall = 0
            if lr_scale < 0.01:
                break
    return best_h, best_c5

def nesterov_optimize(h0, target_sum, deadline, tau=40.0, max_iter=30000):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = 0.85
    base_lr = 0.03
    t = 0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        obj, grad = soft_max_obj_grad(y, tau)
        v = mu * v + (1.0 - mu) * grad
        lr = base_lr / (1.0 + 0.005 * t)
        h_new = h - lr * v
        h_new = project_onto_constraint(h_new, target_sum)
        y = h_new + 0.5 * (h_new - h)
        y = project_onto_constraint(y, target_sum)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                tau = min(tau * 1.3, 300.0)
                base_lr *= 0.7
                stall = 0
                if base_lr < 0.001:
                    break
    return best_h, best_c5

def multi_lag_polish(h0, target_sum, deadline, top_k=3, max_iter=20000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.005
    m = np.zeros(n)
    beta = 0.9
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        val, grad = multi_lag_subgrad(h, top_k)
        m = beta * m + (1.0 - beta) * grad
        h_new = h - lr * m
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.8
                top_k = min(top_k + 1, 8)
        h = h_new
        if lr < 1e-5:
            break
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, deadline, tau=60.0):
    def obj_fun(x):
        o, _ = soft_max_obj_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            c5 = compute_c5(h_opt)
            return h_opt, c5
    except Exception:
        pass
    return h0, compute_c5(h0)

def perturb_restart(h0, target_sum, deadline, scale=0.03, n_restarts=8):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project_onto_constraint(h_pert, target_sum)
        h_opt, c5_opt = adam_optimize(h_pert, target_sum, deadline, tau0=30.0,
                                       tau_max=200.0, base_lr=0.01, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        scale *= 0.8
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 10
    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0

    if best_h is not None and time.time() < deadline - 300:
        n = len(best_h)
        h_opt, c5_opt = adam_optimize(best_h, n / 2.0, deadline - 200,
                                       tau0=10.0, tau_max=100.0, base_lr=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 200:
        n = len(best_h)
        h_opt, c5_opt = nesterov_optimize(best_h, n / 2.0, deadline - 100, tau=50.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 150:
        n = len(best_h)
        h_opt, c5_opt = multi_lag_polish(best_h, n / 2.0, deadline - 80, top_k=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        h_opt, c5_opt = perturb_restart(best_h, n / 2.0, deadline - 50, scale=0.02, n_restarts=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    for n_try in [160, 240]:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        target = n_try / 2.0
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h0 = np.interp(x_new, x_old, best_h)
        else:
            h0 = np.full(n_try, 0.5)
        h0 = project_onto_constraint(h0, target)
        h_opt, c5_opt = adam_optimize(h0, target, deadline - 40, tau0=20.0,
                                       tau_max=200.0, base_lr=0.01, max_iter=8000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        h_opt, c5_opt = adam_optimize(best_h, n / 2.0, deadline - 25,
                                       tau0=50.0, tau_max=400.0, base_lr=0.005, max_iter=10000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_opt, c5_opt = multi_lag_polish(best_h, n / 2.0, deadline - 2, top_k=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is None:
        n = 120
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
