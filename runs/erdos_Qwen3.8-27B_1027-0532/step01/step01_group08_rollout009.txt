import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    max_tc = np.max(tc)
    lse = max_tc + np.log(np.sum(np.exp(tc - max_tc)))
    obj = (lse / tau) * dx
    weights = np.exp(tc - max_tc)
    weights /= np.sum(weights)
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        lag = m - (n - 1)
        j_start = max(0, lag)
        j_end = min(n - 1, n - 1 + lag)
        
        # Term 1: (1 - h[i-lag]) for i in [j_start, j_end]
        idx1_start = j_start - lag  # = max(0, lag) - lag = max(-lag, 0) → if lag≥0: 0, if lag<0: -lag
        idx1_end = j_end - lag + 1
        # Verify: 0 ≤ idx1_start, idx1_end-1 < n
        # If lag ≥ 0: idx1_start = 0, idx1_end-1 = j_end - lag ≤ n-1. OK.
        # If lag < 0: idx1_start = -lag, idx1_end-1 = j_end - lag = n-1+lag-lag = n-1. OK.
        grad[j_start:j_end+1] += w * (1.0 - h[idx1_start:idx1_end])
        
        # Term 2: -h[i+lag] for valid i
        if lag >= 0:
            i2_start = max(j_start, 0)  # j_start = lag ≥ 0
            i2_end = min(j_end, n - 1 - lag)
        else:
            i2_start = max(j_start, -lag)  # = -lag (since j_start=0 and -lag>0)
            i2_end = min(j_end, n - 1)  # j_end = n-1+lag < n-1
        if i2_start <= i2_end:
            idx2_start = i2_start + lag
            idx2_end = i2_end + lag + 1
            grad[i2_start:i2_end+1] += w * (-h[idx2_start:idx2_end])
    
    grad *= dx
    return obj, grad

                   tau_growth=1.05, max_iter=5000):

    y = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = 0.1  # step size (will be adapted)
    tau = tau0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        obj, grad = soft_max_obj_and_grad(y, tau)
        
        # Gradient step
        z = y - mu * grad
        
        # Proximal step (projection)
        h_new = project_onto_constraint(z, target_sum)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, -0.5, 1.5)  # keep y in reasonable range
        t = t_new
        h = h_new
        
        # Check improvement
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive tau
        if it % 50 == 0 and tau < tau_max:
            tau *= tau_growth
        
        # Adaptive step size (simple backtracking)
        # Check if the step reduced the objective
        # (simplified: just reduce mu if we're oscillating)
    
    return best_h, best_c5
