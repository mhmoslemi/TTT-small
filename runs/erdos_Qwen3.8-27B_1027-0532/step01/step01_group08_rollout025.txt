import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    tc = tau * corr
    m = np.max(tc)
    ev = np.exp(tc - m)
    sv = np.sum(ev)
    obj = (m + np.log(sv)) / tau * dx
    w = ev / sv
    grad = np.zeros(n)
    for idx in np.where(w > 1e-14)[0]:
        lag = idx - (n - 1)
        wi = w[idx]
        jlo = max(0, lag)
        jhi = min(n - 1, n - 1 + lag)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += wi * (1.0 - h[jlo - lag:jhi - lag + 1])
        jlo2 = max(0, -lag)
        jhi2 = min(n - 1, n - 1 - lag)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] += wi * (-h[jlo2 + lag:jhi2 + lag + 1])
    return obj, grad * dx

def exact_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    grad = np.zeros(n)
    jlo = max(0, lag)
    jhi = min(n - 1, n - 1 + lag)
    if jlo <= jhi:
        grad[jlo:jhi + 1] += (1.0 - h[jlo - lag:jhi - lag + 1])
    jlo2 = max(0, -lag)
    jhi2 = min(n - 1, n - 1 - lag)
    if jlo2 <= jhi2:
        grad[jlo2:jhi2 + 1] += (-h[jlo2 + lag:jhi2 + lag + 1])
    return corr[idx] * dx, grad * dx

def napg_optimize(h0, n, target_sum, deadline, tau0=8.0, tau_end=400.0, lr0=0.12):
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    elapsed_est = max(deadline - time.time(), 1.0)
    max_iter = int(min(30000, elapsed_est * 40))
    max_iter = max(max_iter, 100)
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        progress = (it + 1) / max_iter
        tau = tau0 * (tau_end / tau0) ** (progress ** 0.6)
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            mom = (t_k - 1.0) / t_new
            y = h + mom * (h - h_prev)
            t_k = t_new
        else:
            y = h.copy()
        _, grad = smooth_max_and_grad(y, tau)
        lr = lr0 / (1.0 + tau * 0.003)
        h_new = y - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                h = best_h + 0.01 * np.random.randn(n)
                h = project_onto_constraint(h, target_sum)
                h_prev = h.copy()
                t_k = 1.0
                stall_count = 0
                continue
        h_prev = h
        h = h_new
    return best_h, best_c5

def slsqp_polish(h0, n, target_sum, deadline, tau=200.0):
    def obj_fun(x):
        o, _ = smooth_max_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = smooth_max_and_grad(x, tau)
        return g
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: np.sum(x) - target_sum,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def subgradient_polish(h0, n, target_sum, deadline, n_steps=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.08
    avg_h = h.copy()
    for it in range(n_steps):
        if time.time() > deadline:
            break
        _, grad = exact_subgrad(h)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        t_curr = it + 1
        avg_h = (avg_h * (t_curr - 1) + h_new) / t_curr
        avg_h = project_onto_constraint(avg_h, target_sum)
        c5_avg = compute_c5(avg_h)
        c5_new = compute_c5(h_new)
        if c5_avg < best_c5 - 1e-14:
            best_c5 = c5_avg
            best_h = avg_h.copy()
        elif c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.997, 0.003)
    return best_h, best_c5

def interleave_optimize(h0, n, target_sum, deadline, tau0=5.0, tau_end=600.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    elapsed_est = max(deadline - time.time(), 1.0)
    max_iter = int(min(25000, elapsed_est * 35))
    max_iter = max(max_iter, 50)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        progress = (it + 1) / max_iter
        if progress < 0.3:
            tau = tau0 + (tau0 * 3 - tau0) * (progress / 0.3)
        elif progress < 0.7:
            tau = tau0 * 3 + (tau_end * 0.3 - tau0 * 3) * ((progress - 0.3) / 0.4)
        else:
            tau = tau_end * 0.3 + (tau_end - tau_end * 0.3) * ((progress - 0.7) / 0.3)
        _, grad = smooth_max_and_grad(h, tau)
        lr = 0.10 / (1.0 + tau * 0.004)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
