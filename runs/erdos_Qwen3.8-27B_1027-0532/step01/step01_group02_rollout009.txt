```python
            continue

        if lag >= 0:

            start = lag
            end = n
            if end > start:
                grad[start:end] += w * (1.0 - h[0:end-start])
        else:

            start = 0
            end = n + lag
            if end > start:
                grad[start:end] += w * (1.0 - h[0:end])

        # Valid when 0 <= i+lag < n

            # i < n - lag
            start = 0
            end = n - lag
            if end > start:
                grad[start:end] += w * (-h[lag:n])
        else:
            # i >= -lag
            start = -lag
            end = n
            if end > start:
                grad[start:end] += w * (-h[0:n+lag])
    
    grad *= dx  # because corr = corr_full * dx, so d(corr)/dh = dx * d(corr_full)/dh
    
    # Gradient of penalty term
    grad += 2 * penalty * constraint_violation * np.ones(n)
    
    return obj, grad
```

```python
def coordinate_descent_fast(h_init, n_passes=500):
    n = len(h_init)
    dx = 2.0 / n
    n_full = 2 * n - 1
    h = h_init.copy()
    
    # Precompute lags
    lags = np.arange(-(n-1), n)
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            old_val = h[i]
            
            # Compute d_k = dA_k/dh[i] = dx * [(1-h[i-lag]) * 1_{0<=i-lag<n} - h[i+lag] * 1_{0<=i+lag<n}]
            d_k = np.zeros(n_full)
            for idx, lag in enumerate(lags):
                val = 0.0
                # First term
                j = i - lag
                if 0 <= j < n:
                    val += (1.0 - h[j])
                # Second term
                j2 = i + lag
                if 0 <= j2 < n:
                    val -= h[j2]
                d_k[idx] = dx * val
            
            # Compute A_k at h[i] = 0 (base)
            h[i] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode="full") * dx
            h[i] = old_val
            
            # A_k(t) = corr_base[k] + t * d_k[k]
            # But wait, corr_base is computed with h[i]=0, so A_k(0) = corr_base[k]
            # And A_k(1) = corr_base[k] + d_k[k] (since d_k = A_k(1) - A_k(0))
            # Actually, d_k = A_k(1) - A_k(0) only if A_k is linear in h[i], which it is!
            
            # Find t in [0,1] minimizing max_k (corr_base[k] + t * d_k[k])
            # Use the fact that this is a convex piecewise-linear function
            # The minimum is at a point where two lines intersect, or at a boundary
            
            # Simple approach: use minimize_scalar (fast for 1D)
            # Or even simpler: the minimum of max of lines is where the
            # upper envelope changes direction, which we can find by bisection
            # on the derivative (which is the max of d_k for the active set)
            
            # For simplicity, use minimize_scalar
            from scipy.optimize import minimize_scalar
            def obj_1d(t):
                return np.max(corr_base + t * d_k)
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = res.x
            new_obj = res.fun
            
            if new_obj < np.max(corr_base + old_val * d_k) - 1e-14:
                h[i] = new_val
                improved = True
            # else keep old_val

            break
```

```python
def coordinate_descent_v2(h_init, n_passes=500):
    n = len(h_init)
    dx = 2.0 / n
    n_full = 2 * n - 1
    h = h_init.copy()
    lags = np.arange(-(n-1), n)
    
    # Initial correlation
    A = np.correlate(h, 1.0 - h, mode="full") * dx
    current_max = np.max(A)
    
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            old_val = h[i]
            
            # Compute d_k = dA_k/dh[i]
            d_k = np.zeros(n_full)
            for idx, lag in enumerate(lags):
                val = 0.0
                j = i - lag
                if 0 <= j < n:
                    val += (1.0 - h[j])
                j2 = i + lag
                if 0 <= j2 < n:
                    val -= h[j2]
                d_k[idx] = dx * val
            
            # A_k(t) = A_k(old_val) + (t - old_val) * d_k[k]
            # = (A - old_val * d_k) + t * d_k
            base = A - old_val * d_k
            
            # Find t in [0,1] minimizing max_k (base[k] + t * d_k[k])
            def obj_1d(t):
                return np.max(base + t * d_k)
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = res.x
            new_obj = res.fun
            old_obj = np.max(base + old_val * d_k)
            
            if new_obj < old_obj - 1e-14:
                # Update A incrementally
                A += (new_val - old_val) * d_k
                h[i] = new_val
                improved = True
        
        if not improved:
            break
    
    return h
```

  - 0 <= i - idx + (n-1) < n
  - -(i + n - 1) <= -idx < n - i - n + 1 = 1 - i
  - i + n - 1 >= idx > i - 1
  - So idx ranges from i to i + n - 1 (inclusive)? Let me check.
    - j >= 0: i - idx + n - 1 >= 0 => idx <= i + n - 1
    - j < n: i - idx + n - 1 < n => idx > i - 1 => idx >= i
  - So idx from i to i + n - 1.

  The value at idx is (1 - h[j]) where j = i - idx + (n-1).
  When idx = i: j = n-1
  When idx = i+1: j = n-2
  ...
  When idx = i+n-1: j = 0
  
  So d_k[first_part] = (1 - h[n-1]), (1 - h[n-2]), ..., (1 - h[0]) for idx = i, i+1, ..., i+n-1.
  Which is: d_k[i:i+n] += (1 - h[::-1])

- Second term valid when 0 <= i + lag < n
  - i + lag >= 0 => lag >= -i => idx >= n-1-i
  - i + lag < n => lag < n-i => idx < n-1+n-i = 2n-1-i => idx <= 2n-2-i
  - So idx from n-1-i to 2n-2-i.
  
  The value at idx is -h[i+lag] where i+lag = i + idx - (n-1) = idx + i - n + 1.
  When idx = n-1-i: i+lag = 0
  When idx = n-i: i+lag = 1
  ...
  When idx = 2n-2-i: i+lag = n-1
  
  So d_k[idx] += -h[idx + i - n + 1] for idx from n-1-i to 2n-2-i.
  Which is: d_k[n-1-i : 2n-1-i] += -h[0:n]

So:
```python
d_k = np.zeros(n_full)
d_k[i : i+n] += (1.0 - h[::-1])  # first term
d_k[n-1-i : 2*n-1-i] += -h  # second term
d_k *= dx
```

- First term: d_k[i : i+n] has length n. i+n <= i+n-1+1 = n_full? i <= n-1, so i+n <= 2n-1 = n_full. ✓
- Second term: d_k[n-1-i : 2n-1-i] has length n. 2n-1-i <= 2n-1 = n_full. ✓ and n-1-i >= 0. ✓

This is O(n) per coordinate with vectorized numpy operations. 

Now, for the 1D minimization, instead of using minimize_scalar (which has Python overhead), I can compute the minimum of a max of linear functions more directly.

The function `f(t) = max_k (base[k] + t * d_k[k])` is convex and piecewise linear. Its minimum occurs at:
- t = 0 (if the minimum is at the left boundary)
- t = 1 (if at the right boundary)
- Or at an intersection point of two lines

The derivative of f at t (where differentiable) is `max_k d_k[k]` for the active k (those achieving the max). The minimum is where this derivative changes from negative to positive.

A simple approach: bisection on the derivative.
- f'(t) = max_{k in active(t)} d_k[k] where active(t) = {k : base[k] + t*d_k[k] = f(t)}
- If f'(t) < 0, the minimum is to the right
- If f'(t) > 0, the minimum is to the left
- If f'(t) = 0, we're at the minimum (or a tie)

But computing the active set and the derivative is O(n) per evaluation, and bisection takes O(log(1/ε)) iterations. Total: O(n * log(1/ε)) per coordinate. For n=350 and ε=1e-10, that's about 350 * 34 = 12K operations per coordinate. For 350 coordinates, that's 4.2M per pass. Very fast!

Actually, there's an even simpler approach. Since f(t) = max_k (a_k + b_k * t) where a_k = base[k], b_k = d_k[k], the minimum of f on [0,1] is:

The minimum is at the point where the upper envelope of these lines is minimized. This can be found by:
1. Computing all intersection points of pairs of lines (O(n²) - too slow)
2. Or using a more clever algorithm

Actually, for the 1D case, the simplest efficient approach is just ternary search or bisection on the derivative, since f is convex. Let me use bisection:

```python
def min_max_linear(base, d_k, lo=0.0, hi=1.0, tol=1e-10):
    # Find t in [lo, hi] minimizing max_k (base[k] + t * d_k[k])
    # Bisection on the "subgradient" which contains 0 at the minimum
    for _ in range(80):  # enough for double precision
        mid = (lo + hi) / 2
        vals = base + mid * d_k
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-15
        # subgradient at mid is the set of weighted averages of d_k[active]
        # For a simple check: if all active d_k are > 0, go left; if all < 0, go right
        d_active = d_k[active]
        if np.all(d_active >= -1e-15):
            lo = mid
        elif np.all(d_active <= 1e-15):
            hi = mid
        else:
            # 0 is in the subgradient, we're at the minimum
            break
    t_opt = (lo + hi) / 2
    return t_opt, np.max(base + t_opt * d_k)
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def min_max_linear(base, d_k, lo=0.0, hi=1.0):
    for _ in range(80):
        if hi - lo < 1e-14:
            break
        mid = (lo + hi) / 2.0
        vals = base + mid * d_k
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-15
        d_active = d_k[active]
        if np.all(d_active >= -1e-15):
            lo = mid
        elif np.all(d_active <= 1e-15):
            hi = mid
        else:
            break
    t_opt = (lo + hi) / 2.0
    return t_opt

def coordinate_descent(h_init, n_passes=200, tol=1e-14):
    n = len(h_init)
    dx = 2.0 / n
    n_full = 2 * n - 1
    h = h_init.copy()
    A = np.correlate(h, 1.0 - h, mode="full") * dx

    for _ in range(n_passes):
        total_improvement = 0.0
        for i in range(n):
            old_val = h[i]
            d_k = np.zeros(n_full)
            d_k[i:i+n] += (1.0 - h[::-1])
            sl = n - 1 - i
            d_k[sl:sl+n] += -h
            d_k *= dx
            base = A - old_val * d_k
            t_opt = min_max_linear(base, d_k)
            new_val = t_opt
            if abs(new_val - old_val) > 1e-14:
                new_obj = np.max(base + new_val * d_k)
                old_obj = np.max(base + old_val * d_k)
                if new_obj < old_obj - tol:
                    A += (new_val - old_val) * d_k
                    h[i] = new_val
                    total_improvement += (old_obj - new_obj)
        if total_improvement < tol * 10:
            break
    return h

def smooth_max_obj_and_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    n_full = 2 * n - 1
    lags = np.arange(-(n-1), n)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_corr = np.exp(corr - m)
    sum_exp = np.sum(exp_corr)
    obj = (m + np.log(sum_exp)) / beta

    constraint_violation = np.sum(h) - target
    obj += penalty * constraint_violation ** 2

    p = exp_corr / sum_exp
    grad = np.zeros(n)

    for idx in range(n_full):
        w = p[idx]
        if w < 1e-15:
            continue
        lag = lags[idx]
        if lag >= 0:
            s = lag
            e = n
            if e > s:
                grad[s:e] += w * (1.0 - h[0:e-s])
            s2 = 0
            e2 = n - lag
            if e2 > s2:
                grad[s2:e2] += w * (-h[lag:n])
        else:
            s = 0
            e = n + lag
            if e > s:
                grad[s:e] += w * (1.0 - h[0:e])
            s2 = -lag
            e2 = n
            if e2 > s2:
                grad[s2:e2] += w * (-h[0:n+lag])

    grad *= dx
    grad += 2.0 * penalty * constraint_violation
    return obj, grad

def smooth_max_optimize(h, betas, penalty=1e5, maxiter=300):
    h_curr = h.copy()
    n = len(h_curr)
    for beta in betas:
        try:
            r = minimize(
                smooth_max_obj_and_grad,
                h_curr,
                jac=True,
                args=(beta, penalty),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter, 'maxls': 30, 'ftol': 1e-15}
            )
            h_curr = np.clip(r.x, 0.0, 1.0)
        except Exception:
            pass
        h_curr = repair_h(h_curr, n)
    return h_curr, evaluate_c5(h_curr)

def slsqp_optimize(h, n, maxiter=300, ftol=1e-12):
    try:
        def obj(x):
            return evaluate_c5(x)
        def obj_grad(x):
            dx = 2.0 / n
            n_full = 2 * n - 1
            lags = np.arange(-(n-1), n)
            corr = np.correlate(x, 1.0 - x, mode="full") * dx
            k_max = np.argmax(corr)
            lag = lags[k_max]
            grad = np.zeros(n)
            if lag >= 0:
                s = lag
                e = n
                if e > s:
                    grad[s:e] += (1.0 - x[0:e-s])
                s2 = 0
                e2 = n - lag
                if e2 > s2:
                    grad[s2:e2] += (-x[lag:n])
            else:
                s = 0
                e = n + lag
                if e > s:
                    grad[s:e] += (1.0 - x[0:e])
                s2 = -lag
                e2 = n
                if e2 > s2:
                    grad[s2:e2] += (-x[0:n+lag])
            return obj(x), grad * dx
        r = minimize(
            obj_grad, h, jac=True, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = repair_h(r.x, n)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    h_init = initial_h_values.copy().astype(float)
```
