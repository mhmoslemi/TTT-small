```python
def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
```

```python
def compute_delta_plus(h, n):
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # i + k: shape (n, 2n-1)
    ik = np.arange(n)[:, None] + lags[None, :]  # (n, 2n-1)
    # i - k: shape (n, 2n-1)
    ik2 = np.arange(n)[:, None] - lags[None, :]  # (n, 2n-1)
    
    valid1 = (ik >= 0) & (ik < n)  # (n, 2n-1)
    valid2 = (ik2 >= 0) & (ik2 < n)  # (n, 2n-1)
    
    # h[i+k] for valid: use np.take with clipped indices
    h_ik = np.take(h, np.clip(ik, 0, n-1))  # (n, 2n-1)
    h_ik2 = np.take(h, np.clip(ik2, 0, n-1))  # (n, 2n-1)
    
    deltas = (1.0 - h_ik) * valid1 + (-h_ik2) * valid2
    return deltas
```
