def compute_c5(h, n):

def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj(h, n, t, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (max_corr + log_sum / t) * dx
    violation = np.sum(h) - n / 2.0
    obj += 0.5 * penalty * violation * violation
    return obj


def smooth_grad(h, n, t, penalty):

    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)

    return grad


def lbfgs_optimize(h0, n, t, penalty, max_iter=200):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    h0 = project_onto_constraint(h0, n)

    def obj_fn(h):
        return smooth_obj(h, n, t, penalty)

    def grad_fn(h):
        return smooth_grad(h, n, t, penalty)

    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            obj_fn, h0.copy(), method='L-BFGS-B',
            jac=grad_fn, bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-15,
                     'gtol': 1e-10, 'maxfun': max_iter * 3, 'maxls': 40}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def subgradient_polish(h, n, n_iter=300, initial_step=0.04, anneal=True):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = initial_step
    sa_temp = 0.02
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += -h[lag:]
        else:
            l = -lag
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] += -h[:n - l]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = h - step * grad / gnorm
        h_new = project_onto_constraint(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            if anneal:
                sa_temp *= 0.998
        else:
            if anneal and np.random.random() < np.exp(-(c5_new - best_c5) / max(sa_temp, 1e-8)):
                h = h_new
            else:
                step *= 0.93
            if anneal:
                sa_temp *= 0.995
    return best_h, best_c5


def coarse_to_fine(h_coarse, n_coarse, n_fine):
    x_src = np.linspace(0, 2, n_coarse, endpoint=False)
    x_new = np.linspace(0, 2, n_fine, endpoint=False)
    h_interp = np.interp(x_new, x_src, h_coarse)
    return project_onto_constraint(h_interp, n_fine)


def fine_to_coarse(h_fine, n_fine, n_coarse):
    x_src = np.linspace(0, 2, n_fine, endpoint=False)
    x_new = np.linspace(0, 2, n_coarse, endpoint=False)
    h_interp = np.interp(x_new, x_src, h_fine)
    return project_onto_constraint(h_interp, n_coarse)


def slsqp_optimize(h0, n, t, max_iter=300):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    h0 = project_onto_constraint(h0, n)

    def obj_fn(h):
        return smooth_obj(h, n, t, 0.0)

    def grad_fn(h):
        return smooth_grad(h, n, t, 0.0)

    def cons_fun(h):
        return np.sum(h) - n / 2.0

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        result = minimize(
            obj_fn, h0.copy(), method='SLSQP',
            jac=grad_fn, bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining_frac():
        return max(0.0, 1.0 - (time.time() - start_time) / budget_s)

    parent = np.array(initial_h_values, dtype=float)
    n_parent = len(parent)
    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n_parent)
    best_n = n_parent

    penalty = 50.0

    # Phase 0: Coarse optimization at n/4
    n_coarse = max(n_parent // 4, 20)
    x_src = np.linspace(0, 2, n_parent, endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_coarse = np.interp(x_coarse, x_src, parent)
    h_coarse = project_onto_constraint(h_coarse, n_coarse)

    for t in [40, 25, 15, 10, 7, 5, 3]:
        if remaining_frac() < 0.55:
            break
        h_opt = lbfgs_optimize(h_coarse, n_coarse, t, penalty, max_iter=200)
        c5_opt = compute_c5(h_opt, n_coarse)
        if c5_opt < compute_c5(h_coarse, n_coarse):
            h_coarse = h_opt

    h_from_coarse = coarse_to_fine(h_coarse, n_coarse, n_parent)
    c5_coarse = compute_c5(h_from_coarse, n_parent)

    if c5_coarse < best_c5:
        best_c5 = c5_coarse
        best_h = h_from_coarse.copy()
    h_work = best_h.copy()

    # Phase 1: L-BFGS-B on parent grid with descending temperatures
    temps = [300, 150, 100, 60, 40, 30, 20, 15, 10, 8, 6, 4, 3, 2]
    for t in temps:
        if remaining_frac() < 0.30:
            break
        h_opt = lbfgs_optimize(h_work, n_parent, t, penalty, max_iter=250)
        c5_opt = compute_c5(h_opt, n_parent)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
        if c5_opt <= compute_c5(h_work, n_parent) + 1e-8:
            h_work = h_opt
        else:
            h_work = best_h.copy()

    # Phase 2: SLSQP refinement (handles constraint exactly)
    if remaining_frac() > 0.28:
        for t in [8, 5, 3]:
            if remaining_frac() < 0.25:
                break
            h_opt = slsqp_optimize(best_h, best_n, t, max_iter=300)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Subgradient with simulated annealing
    if remaining_frac() > 0.22:
        h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=400, initial_step=0.04, anneal=True)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: L-BFGS-B after subgradient
    if remaining_frac() > 0.18:
        h_opt = lbfgs_optimize(best_h, best_n, 5, penalty, max_iter=300)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Finer grid
    if remaining_frac() > 0.18:
        n_fine = n_parent * 2
        h_fine = coarse_to_fine(best_h, best_n, n_fine)
        c5_fine = compute_c5(h_fine, n_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine

        for t in [30, 20, 12, 8, 5, 3]:
            if remaining_frac() < 0.10:
                break
            h_opt = lbfgs_optimize(h_fine, n_fine, t, penalty, max_iter=150)
            c5_opt = compute_c5(h_opt, n_fine)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_fine
            if c5_opt < compute_c5(h_fine, n_fine):
                h_fine = h_opt

        if remaining_frac() > 0.10:
            h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=200, initial_step=0.03, anneal=True)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Structured restarts targeting worst lag
    if remaining_frac() > 0.08:
        for i in range(20):
            if remaining_frac() < 0.04:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            idx_max = np.argmax(corr)
            lag_max = idx_max - (best_n - 1)

            pert = np.zeros(best_n)
            if lag_max >= 0:
                pert[lag_max:] += (1.0 - best_h[lag_max:]) * 0.08
                pert[:best_n - lag_max] -= best_h[:best_n - lag_max] * 0.08
            else:
                l = -lag_max
                pert[:best_n - l] += (1.0 - best_h[:best_n - l]) * 0.08
                pert[l:] -= best_h[l:] * 0.08

            pert += rng.randn(best_n) * 0.015
            h_pert = project_onto_constraint(best_h + pert, best_n)

            t_use = [8, 5, 3, 15, 10][i % 5]
            h_opt = lbfgs_optimize(h_pert, best_n, t_use, penalty, max_iter=150)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Random restarts with varied noise
    if remaining_frac() > 0.05:
        for i in range(10):
            if remaining_frac() < 0.03:
                break
            noise_scale = 0.01 + 0.03 * (i % 3)
            h_pert = project_onto_constraint(best_h + rng.randn(best_n) * noise_scale, best_n)
            t_use = [5, 10, 20][i % 3]
            h_opt = lbfgs_optimize(h_pert, best_n, t_use, penalty, max_iter=100)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 8: Final polish
    if remaining_frac() > 0.02:
        h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=300, initial_step=0.02, anneal=False)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

        h_opt = lbfgs_optimize(best_h, best_n, 2, penalty, max_iter=200)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final verification
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
