def fourier_perturb(h, rng, n_modes, amplitude):
    n = len(h)
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        a = amplitude * rng.standard_normal()
        b = amplitude * rng.standard_normal()
        pert += a * np.cos(2 * np.pi * m * np.arange(n) / n) + b * np.sin(2 * np.pi * m * np.arange(n) / n)
    return h + pert

def block_slsqp(h_init, eps, maxiter, time_limit, start_time):

    def obj_func(x):

        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full') * dx
        cmax = np.max(corr)
        xv = (corr - cmax) / eps
        xv -= np.max(xv)
        return cmax + eps * np.log(np.sum(np.exp(xv)))

    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
    try:

                          bounds=bounds, constraints=cons,

        return h0, compute_c5(h0)

        h_opt2, c5_opt2 = simulated_anneal(best_h, eps, 3000, 0.02, 0.001, 0.1, tl2, start_time, rng)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2

        if time_left() < 60:
            break
        if eps_idx < len(eps_schedule) - 3:
            eps_idx += 1

    if time_left() > 60:
        n_restarts = min(8, int(time_left() // 80))
        for ri in range(n_restarts):
            if time_left() < 80:
                break
            strat = ri % 4
            if strat == 0:
                h_pert = best_h + 0.08 * rng.standard_normal(n0)
            elif strat == 1:
                h_pert = fourier_perturb(best_h, rng, 5, 0.15)
            elif strat == 2:
                h_bin = (rng.random(n0) < 0.5).astype(float)
                h_bin = project_onto_constraint(h_bin, n0 / 2.0)
                alpha = 0.6 + 0.3 * rng.random()
                h_pert = alpha * best_h + (1.0 - alpha) * h_bin
            else:
                h_pert = best_h.copy()
                nflip = max(3, n0 // 8)
                idx = rng.choice(n0, size=nflip, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_onto_constraint(h_pert, n0 / 2.0)
            tl = min(time_left() - 20, 60)
            h_opt, c5_opt = pgd_momentum(h_pert, 0.001, 30000, 0.2, 0.9, tl, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    if time_left() > 40:
        tl = min(time_left() - 10, 200)
        h_opt, c5_opt = block_slsqp(best_h, 0.0005, 500, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time_left() > 30:
        tl = min(time_left() - 5, 100)
        h_opt, c5_opt = coordinate_polish(best_h, 5, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time_left() > 80:
        n1 = 2 * n0
        h_up = np.interp(np.linspace(0, n0 - 1, n1), np.linspace(0, n0 - 1, n0), best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        tl = min(time_left() - 20, 300)
        h_opt, c5_opt = pgd_momentum(h_up, 0.001, 50000, 0.2, 0.9, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

        if time_left() > 30:
            tl2 = min(time_left() - 5, 60)
            h_opt2, c5_opt2 = coordinate_polish(best_h, 3, tl2, start_time)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    if time_left() > 20:
        n2 = len(best_h)
        h_fin = best_h + 0.005 * rng.standard_normal(n2)
        h_fin = project_onto_constraint(h_fin, n2 / 2.0)
        h_opt, c5_opt = pgd_momentum(h_fin, 0.0002, 50000, 0.1, 0.9, min(time_left() - 5, 60), start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
