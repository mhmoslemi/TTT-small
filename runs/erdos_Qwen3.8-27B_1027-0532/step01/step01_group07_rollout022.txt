import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    """Project h to {0≤x≤1, sum(x)=target} via bisection on shift."""
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def binary_hill_climb(h, n, deadline, rng, max_no_improve=200):
    """
    Binary hill climbing with random restarts.
    h is a binary array with sum = n/2.
    Returns the best h and c5 found.
    """
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    cur_h = h.copy()
    cur_c5 = best_c5
    
    while time.time() < deadline:
        improved = False
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        
        # Try all swaps, find the best improvement
        best_swap_c5 = cur_c5
        best_swap = None
        
        # Shuffle for diversity
        ones = rng.permutation(ones)
        zeros = rng.permutation(zeros)
        
        for i in ones:
            for j in zeros:
                cur_h[i] = 0
                cur_h[j] = 1
                c5_new = compute_c5(cur_h)
                cur_h[i] = 1
                cur_h[j] = 0
                if c5_new < best_swap_c5 - 1e-15:
                    best_swap_c5 = c5_new
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            cur_h[i] = 0
            cur_h[j] = 1
            cur_c5 = best_swap_c5
            no_improve = 0
            improved = True
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Random restart: make a random perturbation
                k = rng.randint(1, 4)
                for _ in range(k):
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    cur_h[i] = 0
                    cur_h[j] = 1
                    ones = np.where(cur_h == 1)[0]
                    zeros = np.where(cur_h == 0)[0]
                cur_c5 = compute_c5(cur_h)
                no_improve = 0
            
            # Check if we should just stop
            if time.time() > deadline - 5:
                break
    
    return best_h, best_c5
