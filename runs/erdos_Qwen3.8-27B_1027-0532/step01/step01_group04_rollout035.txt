def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    count = int(np.sum(mask))
    if count == 0:
        return h
    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    d_star = int(np.argmax(corr)) - (n - 1)
    
    # Compute subgradient for lag d_star
    grad = np.zeros(n)
    # First factor: (1 - h[j + d_star]) for valid j + d_star

        jm = j - d_star

        if 0 <= jm < n:
            grad[j] -= h[jm]
    
    h_new = h - alpha * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new, d_star

If the max is at lag 0, we want to decrease Σ h_i(1-h_i). The gradient of h_i(1-h_i) with respect to h_i is 1-2h_i. So the gradient of the sum is (1-2h). We subtract this: h_new = h - alpha*(1-2h). If h_i > 0.5, then 1-2h_i < 0, so h_new_i = h_i - alpha*(negative) = h_i + positive, pushing h_i further from 0.5. If h_i < 0.5, similarly pushed further from 0.5. This is correct! We're making h more "bimodal" (closer to 0 or 1), which reduces the pointwise product h(1-h).

But this conflicts with the sum constraint (sum must be n/2, i.e., mean must be 0.5). The projection step handles this.

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)

    else:
