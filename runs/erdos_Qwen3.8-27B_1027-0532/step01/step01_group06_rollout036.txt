def project_to_simplex_box(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum} using bisection on threshold."""
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

    Returns gradient vector of same shape as h.

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)

    if active_lags is None:

    else:
        active = active_lags

    weights = corr[active] - (cmax - epsilon)  # positive weights
    weights = np.maximum(weights, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)

        s = idx - (n - 1)
