```python
def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _proj_sum_box(v, target_sum):
    """Project v onto {h: sum(h)=target_sum, 0<=h<=1}"""
    # Bisection on tau
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) / 2
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    h = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # Fix remaining numerical error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among interior points
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
            term2 = 0.0
            if 0 <= j - s <= n - 1:
                term2 = -h[j - s]
            g += p[m] * (term1 + term2)
        grad[j] = dx * g
    
    return grad
```

```python
def _compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * corr * dx
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    p = exp_x / exp_x.sum()
    
    # s = m - n + 1, ranges from -(n-1) to (n-1)
    s = np.arange(-(n-1), n)  # length 2n-1
    # m = s + n - 1
    # p indexed by s: p_s = p[s+n-1]
    p_s = p  # already in order s=-(n-1),...,(n-1) since corr is in that order
    
    # For each j: grad[j] = dx * sum_s p_s[s] * [(1-h[j+s])*1_{0≤j+s≤n-1} - h[j-s]*1_{0≤j-s≤n-1}]
    
    # Vectorize over j:
    # j+s for j in 0..n-1, s in -(n-1)..(n-1): create matrix of shape (n, 2n-1)
    # But that's n*(2n-1) = 86*171 = 14706 elements. Fine.
    
    J = np.arange(n)[:, None]  # (n, 1)
    S = s[None, :]  # (1, 2n-1)
    
    js = J + S  # (n, 2n-1): j+s values
    js_valid = (js >= 0) & (js < n)
    
    jsm = J - S  # (n, 2n-1): j-s values  
    jsm_valid = (jsm >= 0) & (jsm < n)
    
    # h[j+s] where valid, else 0
    # Need to safely index h
    h_js = np.where(js_valid, h[np.clip(js, 0, n-1)], 0.0)
    h_jsm = np.where(jsm_valid, h[np.clip(jsm, 0, n-1)], 0.0)
    
    # term1 = (1-h[j+s])*1_{valid}, term2 = -h[j-s]*1_{valid}
    term1 = np.where(js_valid, 1.0 - h_js, 0.0)
    term2 = np.where(jsm_valid, -h_jsm, 0.0)
    
    # grad[j] = dx * sum_s p[s] * (term1[j,s] + term2[j,s])
    grad = dx * (term1 + term2) @ p_s
    
    return grad
```

This is fully vectorized! Memory is n*(2n-1) which for n=86 is about 14706 floats ≈ 118KB. Very manageable.

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _proj_sum_box(v, target_sum):
    lo = v.min() - 1.1
    hi = v.max() + 0.1
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h = np.clip(v - tau, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * corr * dx
    x_max = np.max(x)
    if x_max < -500:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        exp_x = np.exp(x - x_max)
        p = exp_x / exp_x.sum()
    
    s = np.arange(-(n - 1), n)
    J = np.arange(n)[:, None]
    S = s[None, :]
    
    jsm = J - S
    jsm_valid = (jsm >= 0) & (jsm < n)
    h_jsm = np.where(jsm_valid, h[np.clip(jsm, 0, n - 1)], 0.0)
    term1 = np.where(jsm_valid, 1.0 - h_jsm, 0.0)
    
    js = J + S
    js_valid = (js >= 0) & (js < n)
    h_js = np.where(js_valid, h[np.clip(js, 0, n - 1)], 0.0)
    term2 = np.where(js_valid, -h_js, 0.0)
    
    return dx * (term1 + term2) @ p
