import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def smooth_max_obj(x, n, beta, pen_w):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) * beta))
    smooth_val = m + np.log(s) / beta
    penalty = pen_w * (np.sum(h) - n / 2.0) ** 2
    return smooth_val + penalty

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def local_lbfgs(h, n, beta, pen_w, maxiter):
    try:
        r = minimize(
            smooth_max_obj, h,
            args=(n, beta, pen_w),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 80}
        )
        h_new = np.clip(r.x, 0.0, 1.0)
        h_new = repair_h(h_new, n)
        return h_new, evaluate_c5(h_new)
    except Exception:
        return h, evaluate_c5(h)

def slsqp_opt(h, n, maxiter):
    try:
        cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
        r = minimize(
            evaluate_c5, h,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if np.all(r.x >= -1e-8) and np.all(r.x <= 1 + 1e-8):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = repair_h(r.x, n)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

                    old_val = new_val
