One idea: **difference set construction**. A (v, k, λ) difference set is a k-subset D of a group of order v such that every non-identity element can be expressed as d1*d2^{-1} for exactly λ pairs (d1, d2) ∈ D×D. For cyclic groups, this gives sequences with constant autocorrelation.

For a cyclic group of order n, a (n, n/2, λ) difference set would give a binary sequence of length n with n/2 ones and constant autocorrelation. This would be optimal for our problem!

Such difference sets exist for specific parameters. For example, the Paley difference set exists when n ≡ 1 (mod 4).

For n ≡ 1 (mod 4): Let n = q where q is a prime power ≡ 1 (mod 4). The set of quadratic residues mod q forms a (q, (q-1)/2, (q-3)/4) difference set.

So for n = q (prime ≡ 1 mod 4), we can construct a sequence with:
- n/2 ones (approximately, since (q-1)/2 ≈ q/2)
- Constant autocorrelation

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))


def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def make_binary_balanced(h_interp, n):
    h_bin = (h_interp > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        remove = ones_idx[np.argsort(h_interp[ones_idx])[:n_ones - target]]
        h_bin[remove] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        add = zeros_idx[np.argsort(-h_interp[zeros_idx])[:target - n_ones]]
        h_bin[add] = 1
    return h_bin


def sa_binary_search(h_init, n, rng, time_limit, t_start):
    h = h_init.copy()
    current_score = max_corr(h)
    best_h = h.copy()
    best_score = current_score

    T = 4.0
    T_min = 0.003
    cool_rate = 0.99995
    best_stall_limit = 5000
    stall_count = 0

    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_arr = np.where(h == 0)[0].copy()
    zeros_count = len(zeros_arr)

    while time.time() - t_start < time_limit:
        if zeros_count == 0 or len(ones_set) == 0:
            break

        i = rng.choice(list(ones_set))
        j_idx = rng.randint(zeros_count)
        j = zeros_arr[j_idx]

        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_score = max_corr(h)
        delta = new_score - current_score

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_score = new_score
            ones_set.discard(i)
            ones_set.add(j)
            zeros_arr[j_idx] = i
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
                stall_count = 0
            else:
                stall_count += 1
            if delta < 0:
                T = max(T * 0.9999, T_min)
        else:
            h[i] = 1.0
            h[j] = 0.0

        if stall_count > best_stall_limit:
            T = min(T * 2.5, 5.0)
            stall_count = 0
            best_stall_limit = int(best_stall_limit * 1.3)

        if T > T_min:
            T = max(T * cool_rate, T_min)

    return best_h, best_score


def slsqp_refine(h, n, alpha, maxiter):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj_smooth, h, args=(alpha,),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, target_sum)
