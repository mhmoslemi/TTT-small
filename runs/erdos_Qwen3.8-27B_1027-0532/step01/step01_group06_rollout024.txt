def score_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(x):
    n = len(x)
    dx = 2.0 / n
    T = _T
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    # Softmax weights
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    p = exp_scaled / sum_exp  # probability distribution over k
    
    obj = dx * (cmax + T * np.log(sum_exp))
    
    # Gradient: dx * sum_k p_k * dcorr[k]/dx[i]
    # dcorr[k]/dx[i] = (1-x[i+s]) - x[i-s] where s = k - (n-1)
    s_vals = np.arange(len(corr)) - (n - 1)
    
    # For each i, dcorr/dx[i] = sum_k p_k * [(1-x[i+s_k])_valid - x[i-s_k]_valid]

        # x[i-s] for valid i-s

        term2 = np.sum(p[valid2] * x[i - s_vals[valid2]])
        grad[i] = dx * (term1 - term2)

    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))

import numpy as np
from scipy.optimize import minimize
import time

_T = 0.1

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / _T
    return dx * (cmax + _T * np.log(np.sum(np.exp(scaled))))

def _grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / _T
    exp_s = np.exp(scaled)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    s_vals = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        v1 = (i + s_vals >= 0) & (i + s_vals < n)
        v2 = (i - s_vals >= 0) & (i - s_vals < n)
        grad[i] = dx * (np.dot(p[v1], 1.0 - x[i + s_vals[v1]]) - np.dot(p[v2], x[i - s_vals[v2]]))
    return grad

    return np.sum(x) - len(x) / 2.0
