    dx = 2.0 / n

    m = np.max(corr)
    weights = np.exp(corr - m)
    weights /= np.sum(weights)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    weights = np.exp(corr - m)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - n + 1
        w_b = w / beta
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w_b * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= w_b * h[lag:]
        else:
            if n + lag > 0:
                grad[:n + lag] += w_b * (1.0 - h[-lag:])
            if n + lag > 0:
                grad[-lag:] -= w_b * h[:n + lag]
    return grad

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def mom_projected_grad(h, target_sum, beta, n_iter, alpha0, momentum):
    h_curr = h.copy()
    v = np.zeros_like(h_curr)
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(best_h)
    alpha = alpha0
    obj_prev = smooth_obj(h_curr, beta)
    
    for i in range(n_iter):
        grad = compute_smooth_grad(h_curr, beta)
        v = momentum * v - alpha * grad
        h_new = project_constraint(h_curr + v, target_sum)
        obj_new = smooth_obj(h_new, beta)
        
        if obj_new <= obj_prev + 1e-14:
            h_curr = h_new
            obj_prev = obj_new
            if i % 15 == 0 and alpha < 0.3:
                alpha *= 1.3
        else:
            alpha *= 0.5
            v = momentum * v * 0.5
            h_curr = project_constraint(h_curr + v, target_sum) if alpha > 1e-8 else h_curr
        
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
    
    return best_h, best_c5

def subgrad_step(h, target_sum, alpha, top_frac=0.1):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # Find active k values (within 5% of max)
    threshold = m - 0.05 * m
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for k in active:
        lag = k - n + 1
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= h[lag:]
        else:
            if n + lag > 0:
                grad[:n + lag] += (1.0 - h[-lag:])
            if n + lag > 0:
                grad[-lag:] -= h[:n + lag]
    
    grad /= len(active)
    h_new = project_constraint(h - alpha * grad, target_sum)
    return h_new

def subgrad_optimize(h, target_sum, n_iter=2000, alpha0=0.1):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(n_iter):
        alpha = alpha0 / (1.0 + 0.01 * i)

        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_curr = h_new
    
    return best_h, best_c5

def slsqp_optimize(h, maxiter=500, ftol=1e-12):
    n = len(h)
    target_sum = n / 2.0
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_smooth_grad(h, beta, top_k=200):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    weights = np.exp(corr - m)
    total_w = np.sum(weights)
    if total_w < 1e-300:
        return np.zeros(n)
    weights /= total_w
    if top_k < 2 * n - 1:
        idx = np.argsort(weights)[::-1][:top_k]
        weights_s = np.zeros_like(weights)
        weights_s[idx] = weights[idx]
        weights = weights_s
    grad = np.zeros(n)
    one_minus_h = 1.0 - h

def slsqp_optimize(h, maxiter=500, ftol=1e-12):
    n = len(h)
    target_sum = n / 2.0
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = project_constraint(r.x, target_sum)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)
