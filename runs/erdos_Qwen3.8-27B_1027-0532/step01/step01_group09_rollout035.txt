```python
def smooth_c5_and_grad(h, beta, penalty=0.0):
    """Returns (smooth_c5, gradient) where smooth_c5 approximates max_k f_k."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute all lag values
    lags = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    
    # Softmax weights
    bl = beta * lags
    bl_max = bl.max()
    exp_bl = np.exp(bl - bl_max)
    w = exp_bl / exp_bl.sum()
    
    # Smooth max value
    smooth_val = (bl_max + np.log(exp_bl.sum())) / beta
    
    # Gradient computation
    # For each lag k (0 to 2n-2), m_k = k - (n-1)
    # f_k = sum_{i: valid} h[i] * (1-h[i+m_k]) * dx
    # df_k/dh[j] = dx * [(1-h[j+m_k]) * 1{j+m_k in [0,n)} - h[j-m_k] * 1{j-m_k in [0,n)}]
    
    j = np.arange(n)
    grad = np.zeros(n)
    num_lags = 2*n - 1
    
    for k in range(num_lags):
        wk = w[k]
        if wk < 1e-10:
            continue
        m = k - (n - 1)
        # Term 1: (1-h[j+m]) for j+m in [0,n)
        a = j + m
        mask1 = (a >= 0) & (a < n)
        grad[mask1] += wk * (1.0 - h[a[mask1]]) * dx
        # Term 2: -h[j-m] for j-m in [0,n)
        b = j - m
        mask2 = (b >= 0) & (b < n)
        grad[mask2] -= wk * h[b[mask2]] * dx
    
    # Add penalty gradient if needed
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - target)
    
    return smooth_val, grad
```

```python
def equioscilation_gradient(h, n_active=5, tolerance=0.01):
    """Compute gradient using all lags near the maximum."""
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = lags.max()
    
    # Find active lags
    active = np.where(lags >= max_val - tolerance)[0]
    
    # Compute gradient as average over active lags (or all, with weights)
    j = np.arange(n)
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        a = j + m
        mask1 = (a >= 0) & (a < n)
        grad[mask1] += (1.0 - h[a[mask1]]) * dx
        b = j - m
        mask2 = (b >= 0) & (b < n)
        grad[mask2] -= h[b[mask2]] * dx
    grad /= len(active)
    return grad
```

from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def smooth_c5_and_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    bl = beta * lags
    bl_max = bl.max()
    exp_bl = np.exp(bl - bl_max)
    w = exp_bl / exp_bl.sum()
    smooth_val = (bl_max + np.log(exp_bl.sum())) / beta
    j = np.arange(n)
    grad = np.zeros(n)
    num_lags = 2 * n - 1
    for k in range(num_lags):
        wk = w[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        a = j + m
        mask1 = (a >= 0) & (a < n)
        grad[mask1] += wk * (1.0 - h[a[mask1]]) * dx
        b = j - m
        mask2 = (b >= 0) & (b < n)
        grad[mask2] -= wk * h[b[mask2]] * dx
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - target)
    return smooth_val, grad

def smooth_obj(x, beta, penalty):
    v, _ = smooth_c5_and_grad(x, beta, penalty)
    return v

def smooth_grad_func(x, beta, penalty):
    _, g = smooth_c5_and_grad(x, beta, penalty)
    return g

def polish_smooth(h0, beta_start, beta_end, n_betas, tm, penalty):
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    t0 = time.time()
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.time() - t0 > tm * 0.92:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj, h, jac=smooth_grad_func,
                          args=(beta, penalty), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10})
            h = project_feasible(res.x, target)
        except Exception:
            pass
    return h

def subgrad_step(h, beta_sub, tolerance):
    """One subgradient step using softmax-weighted active lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    j = np.arange(n)
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    bl = beta_sub * lags
    bl_max = bl.max()
    w = np.exp(bl - bl_max)
    w /= w.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        m = k - (n - 1)
        a = j + m

        grad[mask1] += wk * (1.0 - h[a[mask1]]) * dx
        b = j - m
        mask2 = (b >= 0) & (b < n)
        grad[mask2] -= wk * h[b[mask2]] * dx
    return grad
