def project_feasible(x, n):
    """Project x to [0,1]^n with sum = n/2 using bisection on threshold."""
    target = n / 2.0
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_max_value(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0/eps) * np.log(np.sum(np.exp(shifted)))) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = np.min(x) - 2.0
    hi = np.max(x) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _full_obj(h):
    corr = _corr_full(h)
    m = np.max(corr)
    shifted = _g_eps * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return (m + np.log(s) / _g_eps) * _g_dx

def _full_grad(h):
    corr = _corr_full(h)
    m = np.max(corr)
    shifted = _g_eps * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    n = _g_n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        l = k - n + 1
        if l >= 0:
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] += wk * (-h[:n - l])
        else:
            nl = -l
            grad[nl:] += wk * (1.0 - h[:n - nl])
            grad[:n - nl] += wk * (-h[nl:])
    grad *= _g_dx
    return grad
