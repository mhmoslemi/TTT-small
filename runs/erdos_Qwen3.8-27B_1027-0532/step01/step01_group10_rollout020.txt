def solve_1d_convex(corr, delta, u_min, u_max, n_iter=60):
    lo = u_min
    hi = u_max
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * delta
        max_val = np.max(vals)
        # Subgradient: if all delta of max-achieving lines > 0, go left; if all < 0, go right
        # Use a tolerance-based active set
        tol = max_val * 1e-10 + 1e-15
        active = vals >= max_val - tol
        if np.all(delta[active] > tol):
            hi = mid
        elif np.all(delta[active] < -tol):
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)
