import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def threshold_to_binary(h):
    n = len(h)
    n_ones = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def compute_P(h):
    n = len(h)
    M = 2 * n - 1
    m = np.arange(M)
    idx = np.arange(n)[:, None] + m[None, :] - (n - 1)
    valid = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n - 1)
    P = np.where(valid, h[idx_clipped], 0.0)
    return P

def compute_full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def update_P_swap(P, h, i, j, n):
    M = 2 * n - 1
    p_arr = np.arange(n)
    m_i = i - p_arr + n - 1
    valid_i = (m_i >= 0) & (m_i < M)
    P[p_arr[valid_i], m_i[valid_i]] = h[i]
    m_j = j - p_arr + n - 1
    valid_j = (m_j >= 0) & (m_j < M)
    P[p_arr[valid_j], m_j[valid_j]] = h[j]
    return P

def best_swap_improve(h, corr, P):
    n = len(h)
    M = 2 * n - 1
    R = corr[None, :] - P
    C = P
    new_max_full = np.max(R[:, :, None] + C[None, None, :].T, axis=1)
    zeros = h < 0.5
    ones = h >= 0.5
    valid_mask = zeros[:, None] & ones[None, :]
    new_max_masked = np.where(valid_mask, new_max_full, np.inf)
    flat_idx = int(np.argmin(new_max_masked))
    i, j = np.unravel_index(flat_idx, (n, n))
    best_max = new_max_masked[i, j]
    current_max = np.max(corr)
    if np.isinf(best_max) or best_max >= current_max - 1e-12:
        return h, corr, P, 0.0
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    new_corr = corr + P[j] - P[i]
    P_new = P.copy()
    update_P_swap(P_new, h_new, i, j, n)
    return h_new, new_corr, P_new, current_max - best_max

def coordinate_descent(h_init, time_limit=60.0, max_iter=200000):
    start = time.time()
    n = len(h_init)
    h = h_init.copy()
    corr = compute_full_corr(h)
    P = compute_P(h)
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        h, corr, P, improvement = best_swap_improve(h, corr, P)
        if improvement < 1e-12:
            break
        if current_max > np.max(corr):
            current_max = np.max(corr)
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
    return best_h, best_max

def simulated_annealing(h_init, T0, T_min, alpha, time_limit=300.0):
    start = time.time()
    n = len(h_init)
    M = 2 * n - 1
    h = h_init.copy()
    corr = compute_full_corr(h)
    P = compute_P(h)
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    T = T0
    rng = np.random.RandomState(12345)
    while T > T_min and time.time() - start < time_limit:
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        n_zeros = len(zeros)
        n_ones = len(ones)
        n_batch = max(1, int(2000 * T / (T0 + 1e-9)))
        for _ in range(n_batch):
            if time.time() - start > time_limit:
                break
            i = zeros[rng.randint(n_zeros)]
            j = ones[rng.randint(n_ones)]
            delta = P[j] - P[i]
            new_max = np.max(corr + delta)
            d = new_max - current_max
            if d < 0 or (d < 50.0 * T and rng.random() < np.exp(-d / T)):
                h[i], h[j] = h[j], h[i]
                update_P_swap(P, h, i, j, n)
                corr = corr + delta
                current_max = new_max
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
        T *= alpha
    return best_h, best_max

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_polish(h_init, eps, maxiter, time_limit=120.0):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    start = time.time()
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() - start > time_limit:
            return h0, compute_c5(h0)
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
