import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -10.0, 10.0
    for _ in range(300):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def smooth_max_corr(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if m * beta > 500:
        return m
    shifted = corr - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def smooth_corr_grad(h, beta):
    """Gradient of the log-sum-exp smoothed max correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = corr - m
    if m * beta > 500:
        # Effectively hard max
        ki = np.argmax(corr)
        w = np.zeros(len(corr))
        w[ki] = 1.0
    else:
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    nfull = len(corr)
    offset = n - 1
    for ki in range(nfull):
        if w[ki] < 1e-14:
            continue
        k = ki - offset
        wk = w[ki] * dx
        # (1-h[j+k]) for valid j+k
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1+k:hi1+k])
        # -h[j-k] for valid j-k
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2-k:hi2-k]
    return grad

def selpolish_smooth(h0, beta, tm=60):
    """Optimize using SLSQP with smoothed objective"""
    n = len(h0)
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()
    
    def f(x):
        if time.time() - t0 > tm * 0.9:
            raise Exception("timeout")
        return smooth_max_corr(x, beta)
    
    def g(x):
        return x.sum() - tgt
    
    def fprime(x):
        if time.time() - t0 > tm * 0.9:
            raise Exception("timeout")
        return smooth_corr_grad(x, beta)
    
    try:
        r = minimize(f, h0, jac=fprime, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': g, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': 5000, 'ftol': 1e-15})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def beta_anneal(h0, beta_start=20, beta_end=300, n_steps=10, tm_total=300):
    """Progressively increase beta for sharper optimization"""
    h = h0.copy()
    t0 = time.time()
    for i in range(n_steps):
        frac = i / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        tm_step = tm_total / n_steps - (time.time() - t0)
        if tm_step < 2:
            break
        h_new = selpolish_smooth(h, beta, tm=min(tm_step, 60))
        if eval_c5(h_new) < eval_c5(h):
            h = h_new
    return h

def coordinate_descent(h0, tm=30, n_sweeps=100):
    """Coordinate descent with pair swaps to maintain constraint"""
    n = len(h0)

```python
def hard_max_grad(h):
    """Exact gradient of the max correlation (subgradient, using the argmax)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    ki = np.argmax(corr)
    k = ki - (n - 1)
    
    grad = np.zeros(n)
    lo1 = max(0, -k)
    hi1 = min(n, n - k)
    if hi1 > lo1:
        grad[lo1:hi1] += dx * (1.0 - h[lo1+k:hi1+k])
    lo2 = max(0, k)
    hi2 = min(n, n + k)
    if hi2 > lo2:
        grad[lo2:hi2] -= dx * h[lo2-k:hi2-k]
    return grad
```

def smooth_grad(h, beta=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = corr - m
    w = np.exp(beta * shifted) / np.sum(np.exp(beta * shifted))
    
    grad = np.zeros(n)
    nfull = len(corr)
    offset = n - 1
    for ki in range(nfull):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if m * beta > 500:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        shifted = corr - m
        exps = np.exp(beta * shifted)
        w = exps / exps.sum()
    return corr, w

def obj_smooth(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if m * beta > 500:
        return m
    shifted = corr - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def grad_smooth(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr, w = corr_and_weights(h, beta)
    grad = np.zeros(n)
    nfull = len(corr)
    offset = n - 1
    for ki in range(nfull):
        if w[ki] < 1e-15:
            continue
        k = ki - offset
        wk = w[ki] * dx
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2-k:hi2-k]
    return grad

def slsqp_analytic(h0, beta=100.0, tm=60):
    """SLSQP with analytic gradient"""
    n = len(h0)
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()
    
    def f(x):
        if time.time() - t0 > tm * 0.95:
            raise Exception("timeout")
        return obj_smooth(x, beta)
    
    def fprime(x):
        if time.time() - t0 > tm * 0.95:
            raise Exception("timeout")
        return grad_smooth(x, beta)
    
    def g(x):
        return x.sum() - tgt
    
    def gprime(x):
        return np.ones(n)
    
    try:
        r = minimize(f, h0, jac=fprime, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': g, 'jac': gprime}],
                     options={'maxiter': 50000, 'ftol': 1e-20})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def multi_beta_optimize(h0, tm=120):
    """Optimize with increasing beta for progressive sharpening"""
    h = h0.copy()
    t0 = time.time()
    alphas = [(50, 0.2), (100, 0.25), (200, 0.25), (500, 0.15), (1000, 0.15)]
    for beta, frac in alphas:
        tm_step = tm * frac - (time.time() - t0)
        if tm_step < 3:
            break
        h_new = slsqp_analytic(h, beta=beta, tm=tm_step)
        if eval_c5(h_new) < eval_c5(h):
            h = h_new
    return h

def stochastic_polish(h0, tm=30, scale=0.02):
    """Stochastic local search with small perturbations"""
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best = h0.copy()
    best_c5 = eval_c5(h0)
    t0 = time.time()
    rng = np.random.RandomState(777)
    
    for it in range(200000):
        if time.time() - t0 > tm:
            break
        # Perturb 1-5 random coordinates
        n_pert = rng.randint(1, 6)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idx] = np.clip(h_try[idx] + rng.uniform(-scale, scale, size=n_pert), 0, 1)
        h_try = project_feasible(h_try, tgt)
        c5_try = eval_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best = h_try.copy()
            best_c5 = c5_try
            h = h_try
            scale = max(0.001, scale * 0.999)
    return best

def pair_swap_polish(h0, tm=30):
    """Try swapping values between pairs of indices"""
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(4242)
    
    for it in range(100000):
        if time.time() - t0 > tm:
            break
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        # Try various transfers from i to j
        for delta in [0.01, 0.05, 0.1, 0.2, 0.5]:
            if h[i] < delta or h[j] < delta:
                continue
            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta
            c5_try = eval_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best = h_try.copy()
                best_c5 = c5_try
                h = h_try
                break
            if h[i] - delta >= 0 and h[j] + delta <= 1:
                continue
    return best
