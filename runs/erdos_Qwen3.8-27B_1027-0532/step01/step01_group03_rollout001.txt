```python
def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h
```

So: ∂c[n-1-k]/∂h[j] = (1-h[j+k])·1_{0≤j+k<n} - h[j-k]·1_{0≤j-k<n}

This confirms my earlier formula. Let me vectorize:

```python
def _grad_lag(h, k):
    """Gradient of correlate[h,1-h][n-1-k] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[j+k]) for j where 0 <= j+k < n
    if k >= 0:
        # j+k in [0,n-1] means j in [0, n-1-k]
        if n - 1 - k >= 0:
            grad[:n-k] += (1.0 - h[k:])
        # Second term: -h[j-k] for j where 0 <= j-k < n
        if k > 0:
            grad[k:] -= h[:n-k]
    else:
        kk = -k
        # j+k = j-kk in [0,n-1] means j in [kk, n-1]
        grad[kk:] += (1.0 - h[:n-kk])
        # j-k = j+kk in [0,n-1] means j in [0, n-1-kk]
        grad[:n-kk] -= h[kk:]
    return grad
```

Let me verify with a small example. n=3, h=[a,b,c], k=1:
c[n-1-k] = c[1] (since n-1-k = 3-1-1 = 1)
c[1] = sum_i h[i] * (1-h[i+1]) = h[0]*(1-h[1]) + h[1]*(1-h[2])
     = a*(1-b) + b*(1-c)

∂c[1]/∂a = 1-b ✓ (first term: 1-h[0+1] = 1-b; second term: h[0-1] = h[-1] = 0)
∂c[1]/∂b = -a + 1-c ✓ (first term: 1-h[1+1] = 1-c; second term: -h[1-1] = -h[0] = -a)
∂c[1]/∂c = -b ✓ (first term: h[2+1] = h[3] = 0; second term: -h[2-1] = -h[1] = -b)

Using my vectorized code for k=1, h=[a,b,c], n=3:
- First term: grad[:n-k] = grad[:2] += (1-h[k:]) = (1-h[1:]) = (1-b, 1-c)
  So grad[0] += 1-b, grad[1] += 1-c
- Second term: k>0, so grad[k:] = grad[1:] -= h[:n-k] = h[:2] = (a, b)
  So grad[1] -= a, grad[2] -= b

Result: grad = [1-b, 1-c-a, -b] ✓

```python
import time
import numpy as np
from scipy.optimize import minimize, linprog


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    shifted = alpha * (x - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def _smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr * (2.0 / n), alpha)


def _grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[:n - k] += (1.0 - h[k:])
        if k > 0:
            grad[k:] -= h[:n - k]
    else:
        kk = -k
        grad[kk:] += (1.0 - h[:n - kk])
        grad[:n - kk] -= h[kk:]
    return grad


def _slsqp_box(h_start, alpha, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start


def _lp_cutting_plane(h, n, target_sum, max_iters=150):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_of(h)
    margin_base = 0.003

    for it in range(max_iters):
        overlaps = _get_overlaps(best_h)
        c5_cur = float(np.max(overlaps))

        margin = max(margin_base, 0.03 * c5_cur)
        active = np.where(overlaps >= c5_cur - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(overlaps))])

        n_corr = 2 * n - 1
        grads = np.zeros((len(active), n))
        for a_idx, m in enumerate(active):
            k = m - (n - 1)
            grads[a_idx] = _grad_lag(best_h, k) * dx

        nv = n + 1
        c_obj = np.zeros(nv)
        c_obj[n] = 1.0

        A_ub = np.zeros((len(active), nv))
        b_ub = np.zeros(len(active))
        for a_idx in range(len(active)):
            A_ub[a_idx, :n] = grads[a_idx]
            A_ub[a_idx, n] = -1.0
            b_ub[a_idx] = -overlaps[active[a_idx]]

        lower = np.concatenate([-best_h, [-1e10]])
        upper = np.concatenate([1.0 - best_h, [1e10]])

        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lower, upper)), method='highs')
            if res.status != 0:
                break
            delta = res.x[:n]
        except Exception:
            break

        if np.allclose(delta, 0):
            break

        improved = False
        for step in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.02, 0.005]:
            h_new = best_h + step * delta
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                break

        if not improved:
            break

        if best_c5 < 0.38080:
            break

    return best_h, best_c5


def _subgrad_descent(h, n, target_sum, max_iters=200):
    best_h = h.copy()
    best_c5 = _c5_of(h)
    step_size = 0.01
    dx = 2.0 / n

    for it in range(max_iters):
        overlaps = _get_overlaps(best_h)
        c5_cur = float(np.max(overlaps))
        margin = max(0.002, 0.05 * c5_cur)
        active = np.where(overlaps >= c5_cur - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(overlaps))])

        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            grad += _grad_lag(best_h, k) * dx
        grad /= len(active)

        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _fix_sum(h_new, target_sum)
        c5_new = _c5_of(h_new)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            step_size = min(step_size * 1.1, 0.05)
        else:
            step_size *= 0.5

        if step_size < 1e-8:
            step_size = 0.005

    return best_h, best_c5


def _coordinate_descent(h, n, target_sum, max_sweeps=20):
    best_h = h.copy()
    best_c5 = _c5_of(h)
    dx = 2.0 / n
    steps = [0.05, 0.02, 0.01, -0.05, -0.02, -0.01]

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            best_local = best_c5
            best_val = best_h[i]
            for s in steps:
                candidate = best_h[i] + s
                if candidate < 0 or candidate > 1:
                    continue
                h_test = best_h.copy()
                h_test[i] = candidate
                h_test = _fix_sum(h_test, target_sum)
                c5_test = _c5_of(h_test)
                if c5_test < best_local - 1e-12:
                    best_local = c5_test
                    best_val = candidate
            if best_val != best_h[i]:
                best_h[i] = best_val
                diff = best_val - best_h[i]
                # Redistribute to maintain sum
                best_h = _fix_sum(best_h, target_sum)
                best_c5 = best_local
                improved = True
        if not improved:
            break

    return best_h, best_c5


def _optimize_one(h0, n, target_sum, t_deadline):
    best_h = h0.copy()
```
