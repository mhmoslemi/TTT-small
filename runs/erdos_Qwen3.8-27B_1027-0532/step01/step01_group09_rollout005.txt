import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, tau, target):
    corr = compute_corr(h)
    m = corr.max()
    return m + np.log(np.sum(np.exp(corr - m))) / tau

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    exp_vals = np.exp(corr - m)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    j = np.arange(n)
    for ki in range(2*n - 1):
        k = ki - (n - 1)
        w = weights[ki]
        if w < 1e-12:
            continue
        # First term: (1-h[j+k]) for j in [max(0,-k), min(n, n-k))
        a_lo = max(0, -k)
        a_hi = min(n, n - k)
        if a_lo < a_hi:
            grad[a_lo:a_hi] += w * (1.0 - h[a_lo+k:a_hi+k]) * dx
        # Second term: -h[j-k] for j in [max(0,k), min(n, n+k))
        b_lo = max(0, k)
        b_hi = min(n, n + k)
        if b_lo < b_hi:
            grad[b_lo:b_hi] -= w * h[b_lo-k:b_hi-k] * dx
    return grad

def sles_optimize(h0, tau, tm, target):
    """Optimize the log-sum-exp smoothed objective with SLSQP"""
    n = len(h0)
    t0 = time.time()
    
    def obj(x):
        if time.time() - t0 > tm:
            raise TimeoutError()
        return smooth_objective(x, tau, target)
    
    def obj_grad(x):
        if time.time() - t0 > tm:
            raise TimeoutError()
        return smooth_gradient(x, tau)
    
    def con_fun(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)]*n,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': 5000, 'ftol': 1e-14})
        return project_feasible(result.x, target)
    except Exception:
        return h0

def block_descent(h0, tm, block_size=20, tau=50):
    """Block coordinate descent with smoothed subproblems"""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best = h0.copy()
    best_c5 = eval_c5(h0)
    t0 = time.time()
    
    n_blocks = n // block_size
    remainder = n % block_size
    
    # Precompute block ranges
    blocks = []
    for bi in range(n_blocks):
        blocks.append((bi * block_size, (bi+1) * block_size))
    if remainder > 0:
        blocks.append((n_blocks * block_size, n))
    
    for iteration in range(200):
        if time.time() - t0 > tm:
            break
        for bi, (start, end) in enumerate(blocks):
            if time.time() - t0 > tm:
                break
            B = end - start
            # Remaining sum for this block
            remaining = target - (h.sum() - h[start:end].sum())
            # remaining must be in [0, B]
            remaining = np.clip(remaining, 0.0, float(B))
            
            # Optimize this block
            def block_obj(x_block):
                x_tmp = h.copy()
                x_tmp[start:end] = x_block
                return smooth_objective(x_tmp, tau, target)
            
            def block_obj_grad(x_block):
                x_tmp = h.copy()
                x_tmp[start:end] = x_block
                full_grad = smooth_gradient(x_tmp, tau)
                return full_grad[start:end]
            
            def block_con_fun(x_block):
                return x_block.sum() - remaining
            
            def block_con_jac(x_block):
                return np.ones(B)
            
            x_init = h[start:end].copy()
            try:
                result = minimize(block_obj, x_init, jac=block_obj_grad, method='SLSQP',
                                  bounds=[(0.0, 1.0)]*B,
                                  constraints=[{'type': 'eq', 'fun': block_con_fun, 'jac': block_con_jac}],
                                  options={'maxiter': 1000, 'ftol': 1e-12})
                h[start:end] = project_feasible(result.x, remaining)
            except Exception:
                pass
        
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def active_set_subgrad(h0, tm, top_k=5, lr_init=0.1, beta=0.8):

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best = h0.copy()
    best_c5 = eval_c5(h0)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)
    
    for it in range(50000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        # Get top-k indices
        top_indices = np.argsort(corr)[::-1][:top_k]
        exp_vals = np.exp(corr[top_indices] - corr[top_indices[0]])
        weights = exp_vals / exp_vals.sum()
        
        grad = np.zeros(n)
        for idx, ki in enumerate(top_indices):
            k = ki - (n - 1)
            w = weights[idx]
            a_lo = max(0, -k)
            a_hi = min(n, n - k)
            if a_lo < a_hi:
                grad[a_lo:a_hi] += w * (1.0 - h[a_lo+k:a_hi+k]) * dx
            b_lo = max(0, k)
            b_hi = min(n, n + k)
            if b_lo < b_hi:
                grad[b_lo:b_hi] -= w * h[b_lo-k:b_hi-k] * dx
        
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9997
        if lr < 1e-8:
            break
    
    return best

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_value(corr, tau):
    m = corr.max()
    return m + np.log(np.sum(np.exp(corr - m))) / tau

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = corr.max()
    exp_vals = np.exp(corr - m)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        w = weights[ki]
        if w < 1e-14:
            continue
        a_lo = max(0, -k)
        a_hi = min(n, n - k)
        if a_lo < a_hi:
            grad[a_lo:a_hi] += w * (1.0 - h[a_lo + k:a_hi + k]) * dx
        b_lo = max(0, k)
        b_hi = min(n, n + k)
        if b_lo < b_hi:
            grad[b_lo:b_hi] -= w * h[b_lo - k:b_hi - k] * dx
    return grad

def sles_optimize(h0, tau, tm, target):
    n = len(h0)
    t0 = time.time()

    def obj(x):
        if time.time() - t0 > tm * 0.95:
            raise TimeoutError()
        corr = compute_corr(x)
        return smooth_obj_value(corr, tau)

    def obj_grad(x):
        if time.time() - t0 > tm * 0.95:
            raise TimeoutError()
        return smooth_gradient(x, tau)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': 3000, 'ftol': 1e-14})
        return project_feasible(result.x, target)
    except Exception:
        return h0

def block_descent(h0, tm, block_size=20, tau=100):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best = h0.copy()
    best_c5 = eval_c5(h0)
    t0 = time.time()

    n_full_blocks = n // block_size
    blocks = []
    for bi in range(n_full_blocks):
        blocks.append((bi * block_size, (bi + 1) * block_size))
    rem = n - n_full_blocks * block_size
    if rem > 0:
        blocks.append((n_full_blocks * block_size, n))

    for sweep in range(300):
        if time.time() - t0 > tm:
            break
        for bi in range(len(blocks)):
            if time.time() - t0 > tm:
                break
            start, end = blocks[bi]
            B = end - start
            remaining = target - (h.sum() - h[start:end].sum())
            if remaining < -1e-6 or remaining > B + 1e-6:
                continue
            remaining = np.clip(remaining, 0.0, float(B))

            def blk_obj(xb):
                xt = h.copy()
                xt[start:end] = xb
                corr = compute_corr(xt)
                return smooth_obj_value(corr, tau)

            def blk_grad(xb):
                xt = h.copy()
                xt[start:end] = xb
                fg = smooth_gradient(xt, tau)
                return fg[start:end]

            def blk_con(xb):
                return xb.sum() - remaining

            def blk_con_j(xb):
                return np.ones(B)

            x_init = h[start:end].copy()
            try:
                res = minimize(blk_obj, x_init, jac=blk_grad, method='SLSQP',
                               bounds=[(0.0, 1.0)] * B,
                               constraints=[{'type': 'eq', 'fun': blk_con, 'jac': blk_con_j}],
                               options={'maxiter': 1000, 'ftol': 1e-12})
                h[start:end] = project_feasible(res.x, remaining)
                h = project_feasible(h, target)
            except Exception:
                pass

        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def active_set_subgrad(h0, tm, top_k=5, lr_init=0.08, beta=0.85):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best = h0.copy()
    best_c5 = eval_c5(h0)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()

    for it in range(80000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        top_idx = np.argsort(corr)[::-1][:top_k]
        ev = np.exp(corr[top_idx] - corr[top_idx[0]])
        ws = ev / ev.sum()
        grad = np.zeros(n)
        for idx, ki in enumerate(top_idx):
            k = ki - (n - 1)
            w = ws[idx]
            a_lo = max(0, -k)
            a_hi = min(n, n - k)
            if a_lo < a_hi:
                grad[a_lo:a_hi] += w * (1.0 - h[a_lo + k:a_hi + k]) * dx
            b_lo = max(0, k)
            b_hi = min(n, n + k)
            if b_lo < b_hi:
                grad[b_lo:b_hi] -= w * h[b_lo - k:b_hi - k] * dx
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9996
        if lr < 1e-8:
            break
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def nesterov_subgrad(h0, tm, top_k=3, lr_init=0.05):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    best = h0.copy()
    best_c5 = eval_c5(h0)
    bet = 0.9
    t0 = time.time()
    for it in range(50000):
        if time.time() - t0 > tm:
            break
        h_corr = project_feasible(h - bet * v, target)
        corr = np.correlate(h_corr, 1.0 - h_corr, 'full') * dx
        top_idx = np.argsort(corr)[::-1][:top_k]
        ev = np.exp(corr[top_idx] - corr[top_idx[0]])
        ws = ev / ev.sum()
        grad = np.zeros(n)
        for idx, ki in enumerate(top_idx):
            k = ki - (n - 1)
            w = ws[idx]
            a_lo = max(0, -k)
            a_hi = min(n, n - k)
            if a_lo < a_hi:
                grad[a_lo:a_hi] += w * (1.0 - h_corr[a_lo + k:a_hi + k]) * dx
            b_lo = max(0, k)
            b_hi = min(n, n + k)
            if b_lo < b_hi:
                grad[b_lo:b_hi] -= w * h_corr[b_lo - k:b_hi - k] * dx
        v = bet * v + (1.0 - bet) * grad
        h = project_feasible(h - lr_init * v, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        bet = 0.5 * (bet + np.sqrt(1.0 + 4.0 * bet * bet))
    return best
