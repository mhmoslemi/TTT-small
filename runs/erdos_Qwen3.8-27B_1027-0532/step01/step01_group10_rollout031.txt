def grad_corr_lag(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        L = -lag
        m = n - L
        if m > 0:
            g[:m] += (1.0 - h[L:])
            g[L:] -= h[:m]
    return g

def topk_grad(h, n, topk=5):
    """Weighted subgradient using top-k lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # Softmax weights for top-k
    idx = np.argsort(corr)[::-1][:topk]
    weights = np.exp(corr[idx] - m)
    weights /= np.sum(weights)
    g = np.zeros(n)
    for i, k in enumerate(idx):
        lag = k - (n - 1)
        g += weights[i] * grad_corr_lag(h, n, lag)
    return g * dx

def run_subgradient(h, n, max_iter=10000, alpha_init=0.1, decay=0.001, 
                    momentum=0.9, topk=1, restart_every=500):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    h = h.copy()
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iter):
        if topk == 1:
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = np.argmax(corr)
            lag = k_star - (n - 1)
            g = grad_corr_lag(h, n, lag) * dx
        else:
            g = topk_grad(h, n, topk)
        
        alpha = alpha_init / (1.0 + decay * it)
        v = momentum * v - alpha * g
        h_new = proj_constraint(h + v, n)
        
        c5_new = c5_value(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve >= restart_every:
            # Reset momentum, try from best
            v = np.zeros(n)
            no_improve = 0
    
    return best_h, best_c5

def smooth_obj_grad(x, n, beta, penalty):
    dx = 2.0 / n
    h = x  # L-BFGS-B keeps x in bounds
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_t = np.exp(corr - m)
    w = exp_t / np.sum(exp_t)
    val = m + np.log(np.sum(exp_t)) / beta
    
    # Penalty for sum constraint
    sum_diff = np.sum(h) - n * 0.5
    val += 0.5 * penalty * sum_diff ** 2
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        g += wk * grad_corr_lag(h, n, lag)
    g *= dx
    g += penalty * sum_diff * np.ones(n)
    
    return val, g

    return x0

    rng = np.random.RandomState(seed)

    h_parent = np.asarray(initial_h_values, dtype=np.float64)
    n_parent = len(h_parent)
    dx_parent = 2.0 / n_parent

    best_h = h_parent.copy()
    best_c5 = c5_value(best_h, dx_parent)
    best_n = n_parent

    n = n_parent
    dx = 2.0 / n

    # Phase 1: Annealed L-BFGS-B from parent
    for beta in [20.0, 50.0, 100.0, 200.0]:
        if time.time() - t0 > budget_s - 200:
            break
        h_opt = run_lbf(best_h, n, beta, 1000.0, maxiter=800)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Phase 2: Subgradient with momentum from current best
    elapsed = time.time() - t0
    if budget_s - elapsed > 80:
        n_iter = min(30000, int((budget_s - elapsed - 200) * 60))
        n_iter = max(n_iter, 5000)
        h_sg, c5_sg = run_subgradient(best_h, n, n_iter,
                                       alpha_init=0.05, decay=0.0003,
                                       momentum=0.95, topk=1, restart_every=400)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 3: Top-k subgradient for diversity
    elapsed = time.time() - t0
    if budget_s - elapsed > 50:
        h_sg, c5_sg = run_subgradient(best_h, n, 8000,
                                       alpha_init=0.04, decay=0.0008,
                                       momentum=0.9, topk=3, restart_every=300)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 4: L-BFGS-B from diverse starts
    starts = []
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.3, 0.4, 0.45]:
            starts.append(gen_cosine(n, freq, amp, 0.0))
    starts.append(gen_random_init(rng, n))
    starts.append(gen_random_init(rng, n))

    for h_start in starts:
        if time.time() - t0 > budget_s - 100:
            break
        h_opt = run_lbf(h_start, n, 30.0, 1000.0, maxiter=500)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_h, best_c5 = h_opt, c
        if time.time() - t0 < budget_s - 30:
            h_sg, c5_sg = run_subgradient(h_opt, n, 4000,
                                           alpha_init=0.05, decay=0.0008,
                                           momentum=0.93, topk=1, restart_every=300)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    # Phase 5: Higher resolution n=512
    n_high = 512
    dx_high = 2.0 / n_high
    if time.time() - t0 < budget_s - 120:
        h_high = interpolate_h(best_h, best_n, n_high)
        h_high = proj_constraint(h_high, n_high)
        h_opt = run_lbf(h_high, n_high, 50.0, 1000.0, maxiter=600)
        c = c5_value(h_opt, dx_high)
        if c < best_c5:
            best_h, best_c5, best_n = h_opt, c, n_high
            dx = dx_high

        if time.time() - t0 < budget_s - 40:
            h_sg, c5_sg = run_subgradient(best_h, best_n, 15000,
                                           alpha_init=0.03, decay=0.0003,
                                           momentum=0.96, topk=1, restart_every=500)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    # Phase 6: Final refinement at best resolution
    n = best_n
    dx = 2.0 / n
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        n_iter = int(remaining * 70)
        n_iter = min(n_iter, 50000)
        h_sg, c5_sg = run_subgradient(best_h, n, n_iter,
                                       alpha_init=0.015, decay=0.0002,
                                       momentum=0.97, topk=1, restart_every=600)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 7: One more L-BFGS-B polish at high beta
    if time.time() - t0 < budget_s - 10:
        h_opt = run_lbf(best_h, best_n, 500.0, 2000.0, maxiter=300)
        c = c5_value(h_opt, 2.0 / best_n)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
