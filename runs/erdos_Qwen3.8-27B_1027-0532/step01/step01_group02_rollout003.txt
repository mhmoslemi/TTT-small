            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = corr * dx
    bs = beta * s
    bs_max = np.max(bs)
    exp_s = np.exp(bs - bs_max)
    sum_exp = np.sum(exp_s)
    obj = (bs_max + np.log(sum_exp)) / beta
    p = exp_s / sum_exp
    n_full = 2 * n - 1
    mid = n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        if p[k] < 1e-12:
            continue
        d = k - mid
        if d >= 0:
            if d < n:
                grad[d:] += p[k] * (1.0 - h[:n - d])
                grad[:n - d] -= p[k] * h[d:]
        else:
            d_abs = -d
            if d_abs < n:
                grad[:n - d_abs] += p[k] * (1.0 - h[d_abs:])
                grad[d_abs:] -= p[k] * h[:n - d_abs]
    grad *= dx
    return obj, grad

def slsqp_smooth(h, n, beta, maxiter=300):
    target = n / 2.0
    def obj(x):
        o, _ = smooth_obj_and_grad(x, beta)
        return o
    def grad(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g
    def con(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    try:
        r = minimize(
            obj, h, jac=grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_opt = np.clip(r.x, 0.0, 1.0)
        h_opt = repair_h(h_opt, n)
        return h_opt, evaluate_c5(h_opt)
    except Exception:
        return h, evaluate_c5(h)

def subgradient_optimize(h, n, max_iter=3000, lr0=0.004, momentum=0.85, decay=0.0005):
    dx = 2.0 / n
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    n_minus_1 = n - 1
    for t in range(max_iter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        s = corr * dx
        k_star = int(np.argmax(s))
        d = k_star - n_minus_1
        grad = np.zeros(n)
        if d >= 0:
            if d < n:
                grad[d:] += (1.0 - h_curr[:n - d])
                grad[:n - d] -= h_curr[d:]
        else:
            d_abs = -d
            if d_abs < n:
                grad[:n - d_abs] += (1.0 - h_curr[d_abs:])
                grad[d_abs:] -= h_curr[:n - d_abs]
        grad *= dx
        lr = lr0 / (1.0 + decay * t)
        v = momentum * v + lr * grad
        h_new = h_curr - v
        h_new = repair_h(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

def anneal_smooth(h, n, betas, maxiter_each=150):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    for beta in betas:
        h_opt, c5 = slsqp_smooth(h_curr, n, beta, maxiter=maxiter_each)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        h_curr = h_opt
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    def remaining():
        return budget_s - (time.time() - t0)
    h_init = initial_h_values.copy().astype(float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    if remaining() > 80:
        betas = [8, 15, 30, 60, 120, 250, 500]
        best_h, best_c5 = anneal_smooth(h_init, n_init, betas, maxiter_each=120)
    if remaining() > 120:
        h_sg, c5_sg = subgradient_optimize(best_h, len(best_h), max_iter=2000, lr0=0.003, momentum=0.88, decay=0.0003)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
    if remaining() > 60:
        n = len(best_h)
        h_fin, c5_fin = slsqp_smooth(best_h, n, 500, maxiter=500)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
    for i in range(20):
        if remaining() < 60:
            break
        n = len(best_h)
        scale = 0.03 + 0.01 * (i % 5)
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = repair_h(h_pert, n)
        betas_sub = [40, 100, 250]
        h_opt, c5_opt = anneal_smooth(h_pert, n, betas_sub, maxiter_each=80)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    if remaining() > 90:
        n2 = min(n_init * 2, 700)
        if n2 != len(best_h):
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n2)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = repair_h(h_up, n2)
            h_opt, c5_opt = anneal_smooth(h_up, n2, [30, 80, 150, 300], maxiter_each=80)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    if remaining() > 30:
        h_final, c5_final = slsqp_smooth(best_h, len(best_h), 800, maxiter=400)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
