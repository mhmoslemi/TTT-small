def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def gradient_descent_momentum(h, n, alpha_start, alpha_end, max_iter, deadline, 
                               lr_start=0.1, lr_end=0.005, momentum=0.9):
    """Gradient descent with Nesterov momentum and adaptive alpha."""
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    v = np.zeros(n)  # momentum velocity
    h_ = h.copy()  # for Nesterov
    
    n_total = max_iter
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Adaptive alpha and learning rate
        frac = it / max(n_total - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Nesterov step
        y = h_ + momentum * v
        y = project_feasible(y, target)
        
        val, grad = compute_smooth_grad(y, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Gradient step
        v = momentum * v - lr * grad / max(gnorm, 1e-10)
        h_new = h_ + v
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_ = h_new
    
    return best_h, best_c5

def gradient_descent_backtrack(h, n, alpha, max_iter, deadline, lr_init=0.1):
    """Gradient descent with backtracking line search."""
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        val, grad = compute_smooth_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad_normed = grad / max(gnorm, 1e-10)
        lr = lr_init
        
        # Backtracking line search
        for _ in range(30):
            h_try = project_feasible(h - lr * grad_normed, target)
            c5_try = compute_c5(h_try)
            if c5_try <= cur_c5 + 1e-12:
                break
            lr *= 0.5
        
        if lr < 1e-12:
            # Try increasing alpha instead
            alpha *= 2.0
            continue
        
        h = h_try
        cur_c5 = c5_try
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Gradually increase alpha
        if it % 500 == 499:
            alpha = min(alpha * 1.5, 5000.0)
    
    return best_h, best_c5

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 8)

        length = max(1, n // (2 * n_blocks) + rng.randint(-1, 2))
        end = min(n, start + length)
        h[start:end] = 1.0

def make_smooth_init(n, rng):

    h = 0.5 + h / (np.max(np.abs(h)) + 1e-10)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500, 500)
    smooth_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -500.0, 500.0)
    smooth_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad

def project_feasible(h, target):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    v = np.zeros(n)
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(float(it) / float(max(max_iter - 1, 1)), 1.0)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        lr = lr0 * (lr1 / lr0) ** frac

        v = mu * v - lr * grad / gn

            best_c5 = c5_new
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        h = h_new
        if stall_count > patience:
            break
    return best_h, best_c5

def gd_backtrack(h, n, deadline, alpha=50.0, lr_init=0.1, max_iter=15000, patience=4000):
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = compute_smooth_grad(h, alpha)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        g = grad / gn
        lr = lr_init
        improved = False
        for _ in range(40):
            h_try = project_feasible(h - lr * g, target)
            c5_try = compute_c5(h_try)
            if c5_try <= cur_c5 + 1e-13:
                h = h_try
                cur_c5 = c5_try
                improved = True
                break
            lr *= 0.5
        if not improved:

            if alpha > 20000.0:

            stall += 50

        if cur_c5 < best_c5 - 1e-10:

            stall = 0
        else:

    return best_h, best_c5

def coord_descent(h, n, deadline, max_iter=20000, rng=None):
    target = n / 2.0
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    cands = np.array([0.0, 0.2, 0.5, 0.8, 1.0])
    for it in range(max_iter):
        if time.time() > deadline:
            break
        p = rng.randint(n)
        old = h[p]
        best_trial_c5 = cur_c5
        best_trial_h = None
        for cv in cands:
            if abs(cv - old) < 0.05:
                continue
            h[p] = cv
            h_proj = project_feasible(h, target)
            c5_try = compute_c5(h_proj)
            if c5_try < best_trial_c5:
                best_trial_c5 = c5_try
                best_trial_h = h_proj.copy()
        if best_trial_h is not None:
            h = best_trial_h
            cur_c5 = best_trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[p] = old
            if abs(h.sum() - target) > 1e-6:
                h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to_n(h_from, n_new):
    n_old = len(h_from)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_from)
    return project_feasible(h_new, n_new / 2.0)

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 8)
    for b in range(n_blocks):
        start = int(n * b / n_blocks)
        length = max(1, n // (2 * n_blocks) + rng.randint(-1, 2))
        end = min(n, start + length)
        h[start:end] = 1.0
    return project_feasible(h, n / 2.0)

def make_binary_init(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h

def make_smooth_init(n, rng):
    t_arr = np.linspace(0.0, 2.0, n)
    n_freqs = rng.randint(2, 5)
    h = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.uniform(1.0, 6.0)
        p = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.sin(2.0 * np.pi * f * t_arr / 2.0 + p)
    h = 0.5 + h / (np.max(np.abs(h)) + 1e-10)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)
