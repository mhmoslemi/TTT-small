                h = np.clip(h, 0.0, 1.0)

        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Phase 2: Binary search - SA with multiple restarts
    h_bin = to_binary(best_h, n)
    c_bin = c5_value(h_bin, dx)
    h_binary_best = h_bin.copy()
    if c_bin < best_c5:
        best_h, best_c5, best_n = h_bin.copy(), c_bin, n

    sa_deadline = t0 + min(300.0, budget_s * 0.3)
    n_sa_restarts = 8
    for restart in range(n_sa_restarts):
        if time.time() > sa_deadline:
            break
        if restart == 0:
            h_sa = h_binary_best.copy()
            T0_sa = 0.005
            steps = 200000
        elif restart == 1:
            h_sa = h_binary_best.copy()
            T0_sa = 0.002
            steps = 150000
        elif restart == 2:
            h_sa = h_parent.copy()
            T0_sa = 0.004
            steps = 100000
        else:
            h_sa = perturb_binary(h_binary_best, n, rng, rng.randint(5, 30))
            T0_sa = 0.006
            steps = 80000
        h_sa_best, c_sa = sa_binary(h_sa, n, dx, rng, max_steps=steps, T0=T0_sa, alpha=0.9999, t_deadline=sa_deadline)
        if c_sa < c_bin:
            c_bin = c_sa
            h_binary_best = h_sa_best.copy()
        if c_sa < best_c5:
            best_h, best_c5, best_n = h_sa_best.copy(), c_sa, n

    # Phase 3: Hill climbing on best binary
    if remaining() > 60:
        deadline_hc = time.time() + min(60.0, remaining() * 0.5)
        # Local hill climbing (limited passes for time)
        h_hc, c_hc = hill_climb_binary(h_binary_best, n, dx, max_passes=50)
        if c_hc < best_c5:
            best_h, best_c5, best_n = h_hc, c_hc, n
        if c_hc < c_bin:
            c_bin = c_hc
            h_binary_best = h_hc.copy()

    # Phase 4: L-BFGS-B from best (continuous, allowing fractional values)
    n = best_n
    dx = 2.0 / n
    for beta in [100, 50, 20, 10, 5, 3, 2, 1]:
        if remaining() < 20:
            break
        h_opt = opt_lbfgsb(best_h, n, dx, beta, 100000, maxiter=8000)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Phase 5: Multi-resolution exploration
    n_candidates = [128, 160, 192, 224, 256, 288, 320, 384, 448, 512, 576, 640]
    for n_try in n_candidates:
        if remaining() < 40:
            break
        if n_try == best_n:
            continue
        dx_try = 2.0 / n_try
        h_start = repair_h(interp_h(best_h, best_n, n_try), n_try)
        for beta in [80, 30, 10, 5]:
            if remaining() < 15:
                break
            h_opt = opt_lbfgsb(h_start, n_try, dx_try, beta, 20000, maxiter=3000)
            c = c5_value(h_opt, dx_try)
            if c < best_c5:
                best_h, best_c5, best_n = h_opt, c, n_try
                h_start = h_opt

    # Phase 6: Second round of binary SA at best resolution
    n = best_n
    dx = 2.0 / n
    if remaining() > 100:
        h_bin2 = to_binary(best_h, n)
        deadline2 = time.time() + min(150.0, remaining() * 0.3)
        for restart in range(4):
            if time.time() > deadline2:
                break
            if restart == 0:
                h_sa = h_bin2.copy()
            else:
                h_sa = perturb_binary(h_binary_best, n, rng, rng.randint(5, 25))
            h_sa_best, c_sa = sa_binary(h_sa, n, dx, rng, max_steps=150000, T0=0.003, alpha=0.9999, t_deadline=deadline2)
            if c_sa < best_c5:
                best_h, best_c5, best_n = h_sa_best, c_sa, n

    # Phase 7: Final L-BFGS-B polish with very high penalty
    n = best_n
    dx = 2.0 / n
    for beta in [50, 20, 10, 5, 3, 2, 1, 0.5]:
        if remaining() < 15:
            break
        h_opt = opt_lbfgsb(best_h, n, dx, beta, 1000000, maxiter=10000)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_h, best_c5 = h_opt, c

    # Phase 8: Additional diversification - try different starting structures
    if remaining() > 60:
        for period in [4, 6, 8, 10, 14, 16, 20, 24, 32]:
            if remaining() < 20:
                break
            h_pat = block_pattern(n, [period // 2])
            h_pat = repair_h(h_pat, n)
            h_opt = opt_lbfgsb(h_pat, n, dx, 20, 50000, maxiter=3000)
            c = c5_value(h_opt, dx)
            if c < best_c5:
                best_h, best_c5 = h_opt, c

    # Final repair and evaluation
    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
