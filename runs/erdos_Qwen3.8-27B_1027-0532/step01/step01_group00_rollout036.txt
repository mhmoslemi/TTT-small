def smooth_grad_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-12:
            continue
        L = i - (n - 1)
        if L >= 0:
            # Part 1: j in [L, n-1], value = 1-h[j-L]
            if L < n:
                j1_start, j1_end = L, n
                grad[j1_start:j1_end] += w * (1.0 - h[j1_start - L:j1_end - L])
            # Part 2: j in [0, n-1-L], value = -h[j+L]
            if n - 1 - L >= 0:
                j2_start, j2_end = 0, n - L
                grad[j2_start:j2_end] += w * (-h[j2_start + L:j2_end + L])
        else:
            p = -L
            # Part 1: j in [0, n-1-p], value = 1-h[j+p]
            if n - 1 - p >= 0:
                j1_start, j1_end = 0, n - p
                grad[j1_start:j1_end] += w * (1.0 - h[j1_start + p:j1_end + p])
            # Part 2: j in [p, n-1], value = -h[j-p]
            if p < n:
                j2_start, j2_end = p, n
                grad[j2_start:j2_end] += w * (-h[j2_start - p:j2_end - p])
    return grad
```

def coord_step(h, j, c, n_grid=200):
    n = len(h)
    B = compute_B_for_j(h, j)
    d_lo = -h[j]
    d_hi = 1.0 - h[j]
    if d_lo == d_hi:
        return 0.0
    deltas = np.linspace(d_lo, d_hi, n_grid)
    # f(delta) = max_m (c[m] + delta * B[m])
    vals = c[np.newaxis, :] + deltas[:, np.newaxis] * B[np.newaxis, :]
    f_vals = np.max(vals, axis=1)
    best_idx = np.argmin(f_vals)
    return deltas[best_idx]
```
