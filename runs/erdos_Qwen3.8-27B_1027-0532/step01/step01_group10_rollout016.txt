```python
def compute_sca_gradient(h, n, dx, beta):
    """Compute gradient of beta-smoothed max of f_k"""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx  # shape (2n-1,)
    # softmax weights
    w = np.exp(beta * (corr - corr.max()))
    w /= w.sum()
    
    # For each lag m (0 to 2n-2), lag = m - (n-1)
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for idx, lag in enumerate(lags):
        if w[idx] < 1e-10:
            continue
        # A[j] = (1-h[j-lag]) if 0 <= j-lag < n
        # B[j] = h[j+lag] if 0 <= j+lag < n
        j = np.arange(n)
        valid_a = (j - lag >= 0) & (j - lag < n)
        valid_b = (j + lag >= 0) & (j + lag < n)
        A = np.where(valid_a, 1.0 - h[np.clip(j-lag, 0, n-1)], 0.0)
        B = np.where(valid_b, h[np.clip(j+lag, 0, n-1)], 0.0)
        grad += w[idx] * dx * (A - B)
    
    return grad
```

The range of λ is [min(v) - 1, max(v)] (roughly).

```python
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
            w = np.zeros(len(corr))
            w[top_idx] = np.exp(gamma * (corr[top_idx] - corr[top_idx].max()))
            w[top_idx] /= w[top_idx].sum()
        else:
            w = np.exp(gamma * (corr - corr.max()))
            w /= w.sum()
        
        grad = weighted_grad(h, n, dx, w)
        
        # Step size
        lr = base_lr / np.sqrt(count + 1.0)
        
        h_new = h - lr * grad
        h_new = project_box_simplex(h_new, S)
        
        c = float(np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx)
        
        if c < best_c5:
            best_h = h_new.copy()
            best_c5 = c
        
        count += 1
        avg_h = (avg_h * (count - 1) + h_new) / count
        h = h_new
    
    return best_h, best_c5, avg_h

def run_slsqp_refine(h0, n, dx, beta=30.0, maxiter=200):
    def obj_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = corr.max()
        val = m + np.log(np.sum(np.exp(corr - m))) / beta
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        grad = weighted_grad(h, n, dx, w)
        return val, grad
    
    val, grad = obj_and_grad(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: float(np.sum(h) - n * 0.5)}]
    res = minimize(lambda h: obj_and_grad(h)[0], h0,
                   jac=lambda h: obj_and_grad(h)[1],
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None:
        return project_box_simplex(res.x, n * 0.5)
    return h0
```

Wait, one more idea: **use the "proximal" gradient method** (also known as ISTA/FISTA for the projection problem). The key difference from the plain subgradient method is that proximal gradient uses the gradient of a smooth approximation (like the smooth max) and takes a "proximal step" which is the projection. This converges faster (O(1/T^2) for FISTA vs O(1/T) for plain subgradient).

For FISTA (Fast Iterative Shrinkage-Thresholding Algorithm):
1. y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})  (momentum step)
2. x_{t+1} = project(y_t - lr * grad(y_t))  (proximal step)

The momentum helps a lot with convergence speed.

```python
    h_prev = h.copy()
    t_idx = 1
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    avg_h = h.copy()
    count = 0

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_idx * t_idx)) / 2.0
        y = h + (t_idx - 1.0) / t_next * (h - h_prev)

        m = corr_y.max()
        w = np.exp(gamma * (corr_y - m))
        w /= w.sum()

        h = project_box_simplex(y - lr * grad, S)

            best_h = h.copy()

        t_idx = t_next
```

```python
def compute_delta_corr(n, j, k, h):
    """Compute change in correlation when h[j]: 0→1 and h[k]: 1→0"""
    L = np.arange(-(n-1), n)  # lag values
    # A_j: contribution from h[j] going 0→1
    valid_j_minus = (j - L) >= 0 & (j - L) < n
    valid_j_plus = (j + L) >= 0 & (j + L) < n
    A_j = np.zeros(2*n-1)
    A_j[valid_j_minus] = 1.0 - h[np.clip(j - L, 0, n-1)[valid_j_minus]]
    A_j[valid_j_plus] -= h[np.clip(j + L, 0, n-1)[valid_j_plus]]
    
    # A_k: contribution from h[k] going 1→0
    valid_k_minus = (k - L) >= 0 & (k - L) < n
    valid_k_plus = (k + L) >= 0 & (k + L) < n
    A_k = np.zeros(2*n-1)
    A_k[valid_k_minus] = -(1.0 - h[np.clip(k - L, 0, n-1)[valid_k_minus]])
    A_k[valid_k_plus] += h[np.clip(k + L, 0, n-1)[valid_k_plus]]
    
    return A_j + A_k
```

```python
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_simplex(v, S):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def weighted_gradient(h, n, dx, w):
    """Compute sum_k w[k] * grad(f_k) where f_k is the k-th correlation coefficient * dx"""
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for i in range(len(w)):
        wi = w[i]
        if wi < 1e-14:
            continue
        lag = int(lags[i])
        if lag == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * dx * (1.0 - h[:n-lag])
            grad[:n-lag] -= wi * dx * h[lag:]
        else:
            l = -lag
            grad[:n-l] += wi * dx * (1.0 - h[l:])
            grad[l:] -= wi * dx * h[:n-l]
    return grad

def fista_optimize(h_init, n, dx, S, t0, time_limit, gamma=20.0, lr=0.05):

    best_c5 = c5_value(h, dx)
    avg_h = h.copy()
    count = 0
    
    while time_limit - (time.time() - t0) > 5:
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_idx * t_idx)) / 2.0
        y = h + (t_idx - 1.0) / t_next * (h - h_prev)
        
        corr = np.correlate(y, 1.0 - y, mode="full") * dx
        m = corr.max()
        w = np.exp(gamma * (corr - m))
        w /= w.sum()
        grad = weighted_gradient(y, n, dx, w)
        
        h_prev = h.copy()
        h = project_box_simplex(y - lr * grad, S)

            best_h = h.copy()
        
        count += 1

        t_idx = t_next

    c_avg = c5_value(avg_h, dx)
    if c_avg < best_c5:

    return best_h, best_c5

        # From h[i+L] term: 0*(1-h[k-L]) = 0
        # From (1-h[i]) term: h[k+L]*(1-0) = h[k+L]
        # Total: h[k+L] (if k+L valid)
        
        # New contribution of h[k]=1 to corr[m]:
        # From h[i+L] term: 1*(1-h[k-L]) = 1-h[k-L]
        # From (1-h[i]) term: h[k+L]*(1-1) = 0
        # Total: 1-h[k-L] (if k-L valid)
        
        # Change (new - old) for h[k]:
        # = (1-h[k-L])*[k-L valid] - h[k+L]*[k+L valid]
        
        # Total delta:
        # delta = [h[j+L]*[j+L valid] - (1-h[j-L])*[j-L valid]] + [(1-h[k-L])*[k-L valid] - h[k+L]*[k+L valid]]
        
        # But wait, if j-L == k or k-L == j, there's an interaction.
        # Since h[j] and h[k] are in different positions, and we're computing the correlation
        # BEFORE the swap (using the old h), the interaction terms should be fine.
        # Actually, the correlation is:
        # corr[m] = sum_i h_old[i+L] * (1-h_old[i])
        # After swap:
        # corr_new[m] = sum_i h_new[i+L] * (1-h_new[i])
        # The difference only involves terms where i or i+L equals j or k.
        
        # Let me just compute it directly:
        # corr_new[m] - corr_old[m] = sum over affected terms
        # A term h[i+L](1-h[i]) is affected if i∈{j,k} or i+L∈{j,k}
        
        # This is getting complex. Let me just recompute the full correlation for safety.
        # It's only O(n^2) ≈ 130K ops, which is fine.
        
        h[j] = 0.0
        h[k] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(new_corr.max() * dx)
        
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()
        
        delta_c = new_max - best_c5  # Wait, this should be new_max - old_max
        # Actually, let me track old_max separately
        old_max = float(corr.max() * dx)
        delta_c = new_max - old_max
        
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / T):
            corr = new_corr
            # Update ones and zeros lists
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[j] = 1.0
            h[k] = 0.0
            # Don't update corr (it's still the old one)
        
        T = max(T * cool, T_min)
    
    return best_h, best_c5
```

Wait, I'm recomputing the full correlation for every swap attempt. That's O(n^2) per attempt. With n=256, that's ~130K ops per swap. At maybe 20μs per correlation (numpy optimized), I get ~50K swaps per second. In 1300s: 65M swaps. That's plenty.

But the `np.where(h==1.0)[0]` and `np.where(h==0.0)[0]` calls inside the loop are expensive (O(n) each). Over 65M swaps, that's 65M * 2 * O(n) = 65M * 512 = 33B operations. That might be too slow.

One more thought: the SA temperature schedule is crucial. Too high → random walk, too low → stuck in local optima. I'll use a geometric cooling: T = T_init * cool^t, with T_init chosen so that the initial acceptance rate is about 50% for "bad" moves.

For the initial temperature: ΔC5 is typically on the order of 0.001-0.01 for a bad move. So T_init ≈ 0.005 should give ~50% acceptance for moves with ΔC5 = 0.005.
