    n = len(h)

But it also means that to change the s=0 term, we need a different move (like mass transfer). So a combined approach might be better:

Actually, let me reconsider the smooth_gradient function. There might be a more efficient way to compute it.

The gradient of the smoothed objective is:
grad[h[j]] = sum_k w[k] * (d c[k] / d h[j])

where c[k] is the k-th correlation value and w[k] is the softmax weight.

For shift s = k - (n-1):
If s ≥ 0: c[k] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
  d c[k] / d h[j] = (1-h[j-s]) [if s ≤ j ≤ n-1] + (-h[j+s]) [if 0 ≤ j ≤ n-1-s]

If s < 0 (let r = -s): c[k] = sum_{j=0}^{n-1-r} h[j] * (1-h[j+r])
  d c[k] / d h[j] = (1-h[j+r]) [if 0 ≤ j ≤ n-1-r] + (-h[j-r]) [if r ≤ j ≤ n-1]

This is what the code does. It's O(n) per shift, O(n²) total. For n=90, that's fine.

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                total = n - s
                grad[s:s + total] += wk * dx * omh[:total]
                grad[:total] -= wk * dx * h[s:s + total]
        else:
            r = -s
            if r < n:
                total = n - r
                grad[:total] += wk * dx * omh[r:r + total]
                grad[r:r + total] -= wk * dx * h[:total]
    return grad

def constraint_fun_auto(x):
    return float(np.sum(x) - len(x) / 2.0)

def constraint_jac_auto(x):
    return np.ones(len(x))

def do_slsqp(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun_auto, 'jac': constraint_jac_auto}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def swap_sa_local_search(h, n_iter, rng, T_start=0.0005, T_end=1e-8):
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def mass_transfer_search(h, n_iter, rng):

def generate_random_h(n, rng, method):
    if method == 0:
        h = 0.5 + 0.4 * rng.standard_normal(n)
    elif method == 1:
        h = rng.random(n)
    elif method == 2:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            blen = int(rng.integers(2, max(3, n // 10)))
            h[pos:pos + blen] = 0.5 + 0.3 * rng.standard_normal(blen)
            pos += blen
    elif method == 3:
        t = np.arange(n) / n
        freq = 2.0 + 6.0 * rng.random()
        h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * t + rng.random() * 2 * np.pi)
        h += 0.15 * rng.standard_normal(n)
    elif method == 4:
        half = n // 2
        h_first = 0.5 + 0.35 * rng.standard_normal(half)
        h = np.zeros(n)
        h[:half] = h_first
        h[half:] = 1.0 - h_first
        if n % 2 == 1:
            h[n - 1] = 0.5
    else:
        h = np.zeros(n)
        n_ones = n // 2
        idx = rng.choice(n, size=n_ones, replace=False)
        h[idx] = 1.0
    h = np.clip(h, 0.0, 1.0)
    return project_onto_constraint(h, n / 2.0)

def optimize_schedule(h_init, eps_list, maxiter_per_eps, deadline):
    h = h_init.copy()
    for eps in eps_list:
        if time.time() > deadline:
            break
        h, c5 = do_slsqp(h, eps, maxiter_per_eps)
    return h, compute_c5(h)
