```python
import numpy as np
from scipy.optimize import minimize
import time

_CURRENT_T = 1.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj_grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    T = _CURRENT_T
    scaled = (corr - cmax) / T
    scaled = np.clip(scaled, -200, 0)
    exp_vals = np.exp(scaled)
    obj = dx * (cmax + T * np.log(np.sum(exp_vals) + 1e-300))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = exp_vals[k]
        if w < 1e-18:
            continue
        if s >= 0:
            m = n - s
            if m > 0:
                grad[s:s+m] += w * (1.0 - x[0:m])
                grad[0:m] -= w * x[s:s+m]
        else:
            p = -s
            m = n - p
            if m > 0:
                grad[0:m] += w * (1.0 - x[p:p+m])
                grad[p:p+m] -= w * x[0:m]
    grad *= dx / T
    return obj, grad

def _obj(x):
    return lse_obj_grad(x)[0]

def _grad(x):
    return lse_obj_grad(x)[1]

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def sum_con(x):
    return float(np.sum(x) - len(x) / 2.0)

def sum_con_jac(x):
    return np.ones(len(x))

def try_slsqp(h, T, max_iter):
    global _CURRENT_T
    _CURRENT_T = T
    n = len(h)
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            s = score_c5(x_new)
            return x_new, s
    except Exception:
        pass
    return h, score_c5(h)

def upsample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_pad = np.concatenate([h[-1:], h, h[:1]])
    x_pad = np.concatenate([[-1.0/n], x_old, [2.0 + 1.0/n]])
    h_new = np.interp(x_new, x_pad, h_pad)
    return project_feasible(h_new, n_new)

def local_search(h, n, t_start, time_limit, mode='perturb'):
    best = h.copy()
    best_score = score_c5(best)
    
    if mode == 'binary':
        candidates_fn = lambda i, v: [0.0, 1.0]
    else:
        candidates_fn = lambda i, v: [v-0.3, v-0.1, v+0.1, v+0.3, 0.0, 1.0]
    
    pass_count = 0
    max_passes = 200
    while time.time() - t_start < time_limit and pass_count < max_passes:
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                return best, best_score
            current = best[i]
            cands = candidates_fn(i, current)
            for cv in cands:
                cv = np.clip(cv, 0.0, 1.0)
                if abs(cv - current) < 1e-10:
                    continue
                trial = best.copy()
                trial[i] = cv
                trial = project_feasible(trial, n)
                s = score_c5(trial)
                if s < best_score - 1e-11:
                    best = trial
                    best_score = s
                    improved = True
                    break
        if not improved:
            break
        pass_count += 1
    return best, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global
