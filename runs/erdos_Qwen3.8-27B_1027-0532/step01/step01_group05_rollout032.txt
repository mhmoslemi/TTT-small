def project_onto_constraint(h, n):

    lo = -2.0
    hi = 2.0

        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    shifted = (corr - max_val) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)

        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    shifted = (corr - max_val) / T
    shifted = np.clip(shifted, -400.0, 400.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = (max_val + T * np.log(sum_exp)) * dx
    grad = np.zeros(n)
    threshold = 3.0 / (2 * n - 1)
    active = weights > threshold
    if not np.any(active):
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    active_idx = np.where(active)[0]

        lag = idx - (n - 1)
        if lag >= 0:

            grad[:n - l] += w * (1.0 - h[l:])

        if elapsed() > budget_s * 0.35:
            break
        h_opt, c5_opt = adam_optimize(h_curr, n_parent, iters_per_T, T=T, lr=0.12)
        c5_curr = compute_c5(h_curr, n_parent)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
        if c5_opt < c5_curr:
            h_curr = h_opt
        else:
            h_curr = best_h.copy()

    if elapsed() < budget_s * 0.45:
        h_mlg, c5_mlg = multi_lag_subgrad_hinge(best_h, best_n, n_iter=400, top_k=10, step=0.12)
        if c5_mlg < best_c5:
            best_c5 = c5_mlg
            best_h = h_mlg.copy()
            best_n = best_n

    n_restarts = 15
    for trial in range(n_restarts):
        if elapsed() > budget_s * 0.65:
            break
        scale = 0.04 + 0.06 * (trial % 4)
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = project_onto_constraint(h_pert, best_n)
        T_use = [4, 3, 2, 5][trial % 4]
        h_opt, c5_opt = adam_optimize(h_pert, best_n, 200, T=T_use, lr=0.08)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n

    if elapsed() < budget_s * 0.72:
        h_bin = binary_round(best_h, best_n)
        h_opt, c5_opt = adam_optimize(h_bin, best_n, 300, T=3, lr=0.06)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n

    if elapsed() < budget_s * 0.75:
        h_mlg2, c5_mlg2 = multi_lag_subgrad_hinge(best_h, best_n, n_iter=300, top_k=8, step=0.1)
        if c5_mlg2 < best_c5:
            best_c5 = c5_mlg2
            best_h = h_mlg2.copy()

    if elapsed() < budget_s * 0.78:
        n_new = n_parent * 2
        x_src = np.linspace(0.0, 2.0, n_parent, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_src, best_h)
        h_interp = project_onto_constraint(h_interp, n_new)
        c5_interp = compute_c5(h_interp, n_new)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_new
        h_opt, c5_opt = adam_optimize(h_interp, n_new, 300, T=4, lr=0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_new
        h_mlg3, c5_mlg3 = multi_lag_subgrad_hinge(h_opt, n_new, n_iter=200, top_k=10, step=0.08)
        if c5_mlg3 < best_c5:
            best_c5 = c5_mlg3
            best_h = h_mlg3.copy()
            best_n = n_new

    if elapsed() < budget_s * 0.90:
        h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=400)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = best_n

    if elapsed() < budget_s * 0.95:
        h_mlg4, c5_mlg4 = multi_lag_subgrad_hinge(best_h, best_n, n_iter=300, top_k=6, step=0.06)
        if c5_mlg4 < best_c5:
            best_c5 = c5_mlg4
            best_h = h_mlg4.copy()

    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
