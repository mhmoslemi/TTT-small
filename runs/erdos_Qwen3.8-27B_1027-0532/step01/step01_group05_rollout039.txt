def compute_gradient_coeffs(h, j, n):
    """Compute a[k] = d(corr[k])/d(h[j]) for all k, vectorized."""
    n_lags = 2 * n - 1
    a = np.zeros(n_lags)
    
    # Positive lags: s = 0, 1, ..., n-1, corresponding to k = n-1+s
    # a[n-1+s] = (1-h[j-s])*1_{s≤j} - h[j+s]*1_{j+s≤n-1}
    s_pos = np.arange(0, n)
    idx_pos = n - 1 + s_pos  # indices into a
    
    # term1: (1-h[j-s]) where s ≤ j
    valid1 = s_pos <= j
    if np.any(valid1):
        s_valid = s_pos[valid1]
        a[idx_pos[valid1]] += 1.0 - h[j - s_valid]
    
    # term2: -h[j+s] where j+s ≤ n-1
    valid2 = j + s_pos <= n - 1
    if np.any(valid2):
        s_valid = s_pos[valid2]
        a[idx_pos[valid2]] -= h[j + s_valid]
    
    # Negative lags: l = 1, ..., n-1, corresponding to k = n-1-l
    l_neg = np.arange(1, n)
    idx_neg = n - 1 - l_neg  # indices into a
    
    # term1: (1-h[j+l]) where j+l < n
    valid1 = j + l_neg < n
    if np.any(valid1):
        l_valid = l_neg[valid1]
        a[idx_neg[valid1]] += 1.0 - h[j + l_valid]
    
    # term2: -h[j-l] where j-l ≥ 0
    valid2 = j - l_neg >= 0
    if np.any(valid2):
        l_valid = l_neg[valid2]
        a[idx_neg[valid2]] -= h[j - l_valid]
    
    return a

def slsqp_optimize(h0, n, t, max_iter=300):
    dx = 2.0 / n
    
    def obj_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        shifted = t * (corr - max_corr)
        shifted = np.clip(shifted, -300.0, 300.0)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        obj = (max_corr + np.log(sum_exp) / t) * dx
        weights = exp_vals / sum_exp
        
        grad = np.zeros(n)
        for idx in range(len(corr)):
            w = weights[idx]
            if w < 1e-10:
                continue
            lag = idx - (n - 1)

                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] += -w * h[lag:]
            else:
                l = -lag
                grad[:n - l] += w * (1.0 - h[l:])
                grad[l:] += -w * h[:n - l]
        
        grad *= dx
        return obj, grad
    
    def obj_f(h):
        return obj_and_grad(h)[0]
    
    def grad_f(h):
        return obj_and_grad(h)[1]
    
    bounds = [(0.0, 1.0)] * n

             'jac': lambda h: np.ones(n)}]

def slsqp_obj_func(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -300.0, 300.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    return obj


def slsqp_grad_func(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -300.0, 300.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]
    
    grad *= dx
    return grad
