def binary_sa(h0, n, deadline, seed=42):
    np.random.seed(seed)
    h = h0.copy()
    # Make it binary
    # Keep the n/2 largest values as 1
    indices = np.argsort(h)[::-1][:n//2]
    h = np.zeros(n)
    h[indices] = 1.0
    
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    T = 0.1  # initial temperature
    cooling = 0.999
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Pick a random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        # Accept or reject
        delta = new_c5 - c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1
            h[j] = 0
        
        T *= cooling
    
    return best_h, best_c5

Actually, one idea that might help: since the problem is a min-max, I can use the **Jensen's inequality** trick. The optimal value t* satisfies:
t* = min_h max_k f_k(h) = max_{p ∈ Δ} min_h sum_k p_k * f_k(h)

where Δ is the simplex and f_k(h) = sum_i h_i(1-h_{i+k}) * dx.

The inner minimization (for fixed p) is a quadratic programming problem!
min_h sum_k p_k * f_k(h) subject to 0 ≤ h ≤ 1, sum h = n/2

The objective is: sum_k p_k * sum_i h_i(1-h_{i+k}) * dx
= dx * sum_i h_i * (1 - sum_k p_k * h_{i+k})  (not quite, let me be more careful)

Actually:
sum_k p_k * sum_i h_i(1-h_{i+k})
= sum_k p_k * (sum_i h_i - sum_i h_i * h_{i+k})
= sum_k p_k * sum_i h_i - sum_k p_k * sum_i h_i * h_{i+k}
= (sum_i h_i) * (sum_k p_k) - sum_i h_i * (sum_k p_k * h_{i+k})
= (n/2) * 1 - sum_i h_i * (sum_k p_k * h_{i+k})

So we want to MAXIMIZE: sum_i h_i * (sum_k p_k * h_{i+k})
= sum_i sum_k p_k * h_i * h_{i+k}
= sum_k p_k * (h^T S_k h_vector)  (not exactly, due to the range restrictions)

This is a quadratic form! And we want to maximize it subject to 0 ≤ h ≤ 1 and sum h = n/2.

Maximizing a quadratic form is generally non-convex (unless the matrix is positive semidefinite). So this doesn't directly help.

But the dual formulation suggests an algorithm:
1. Start with p = uniform distribution over k
2. Solve: min_h sum_k p_k * f_k(h) (a quadratic program)
3. Update p: p_k = 1/K_{max} (put all weight on the worst constraint)
   Actually, the update is: p_k ∝ f_k(h*) where h* is the solution to step 2
4. Repeat

This is the **dual ascent** method for min-max problems. But step 2 is still a non-convex QP.

However, if we relax the 0 ≤ h ≤ 1 constraint (or handle it with penalties), we get a QP that can be solved with standard methods.

Actually, for the continuous relaxation (just sum h = n/2, no box constraints), the QP is:
min_h dx * [n/2 - sum_i h_i * (sum_k p_k * h_{i+k})]
= min_h -dx * sum_i h_i * (A p)_i  where A is the correlation matrix

This is a quadratic program with a linear constraint. If the quadratic part is convex (i.e., the Hessian is PSD), we can solve it exactly.

The Hessian of sum_i sum_k p_k * h_i * h_{i+k} with respect to h is:
H_{ij} = sum_k p_k * (δ_{i+k, j} + δ_{i, j+k}) (approximately)

This is a banded matrix. Whether it's PSD depends on p.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def binary_sa(h0, n, deadline, seed, max_iter=2000000):
    np.random.seed(seed)
    h = to_binary(h0, n)
    dx = 2.0 / n
    corr = compute_corr(h)
    c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.08
    cooling = 0.99999
    no_improve = 0
    patience = 80000
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        if it % 50000 == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_c5 = np.max(new_corr) * dx
        delta = new_c5 - c5
        if delta < 0 or (T > 1e-10 and np.random.random() < np.exp(-delta / T)):
            c5 = new_c5
            corr = new_corr
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve % 5000 == 0:
            T *= cooling
        if no_improve > patience:
            h = best_h.copy()
            for _ in range(15):
                o = np.where(h == 1)[0]
                z = np.where(h == 0)[0]
                ii = o[np.random.randint(len(o))]
                jj = z[np.random.randint(len(z))]
                h[ii] = 0.0
                h[jj] = 1.0
            corr = compute_corr(h)
            c5 = np.max(corr) * dx
            T = 0.06
            no_improve = 0
    return best_h, best_c5

def pgd_objective_and_grad(h, n, tau):
    dx = 2.0 / n
    corr = compute_corr(h)
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    wsum = np.sum(weights)
    weights /= wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_min1 = max(0, lag)
        j_max1 = min(n - 1, n - 1 + lag)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1 + 1] += w * (1.0 - h[j_min1 - lag:j_max1 - lag + 1])
        j_min2 = max(0, -lag)
        j_max2 = min(n - 1, n - 1 - lag)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2 + 1] += w * (-h[j_min2 + lag:j_max2 + lag + 1])
    return obj, grad * dx

def pgd_optimize(h0, n, target_sum, deadline, max_iter=10000, tau_start=40.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.03
    tau = tau_start
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, g = pgd_objective_and_grad(h, n, tau)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            tau = max(tau * 1.05, 40.0)
        else:
            stall += 1
            if stall > 30:
                lr *= 0.75
                if lr < 0.003:
                    tau *= 1.6
                    lr = 0.03
                    stall = 0
                    if tau > 500:
                        break
        h = h_new
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, deadline, tau=60.0):
    def obj_fun(x):
        o, _ = pgd_objective_and_grad(x, n, tau)
        return o
    def grad_fun(x):
        _, g = pgd_objective_and_grad(x, n, tau)
        return g
    def cons_fun(x):
        return x.sum() - target_sum
    def cons_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def interpolate_to(h_src, n_target):

    x_src = np.linspace(0, 2, n_src, endpoint=False)

    h_tgt = np.interp(x_tgt, x_src, h_src)
    h_tgt = project_onto_constraint(h_tgt, n_target / 2.0)
    return h_tgt
