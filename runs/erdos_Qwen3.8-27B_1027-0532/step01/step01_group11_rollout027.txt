import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient_nesterov(h_init, eps, lr, n_steps, target_sum):
    h = h_init.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        grad = smooth_gradient(h, eps)
        v = 0.9 * v - lr * grad
        h_new = project_onto_constraint(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def coordinate_descent(h, target_sum, n_sweeps=10, delta=0.05):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            for d in [delta, -delta, delta/2, -delta/2, delta/4, -delta/4]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + d, 0, 1)
                # Fix sum by adjusting another element
                diff = target_sum - np.sum(h_try)
                if abs(diff) < 1e-10:
                    pass
                else:
                    # Find an element to adjust
                    if diff > 0:
                        # Need to increase sum, find element below 1
                        candidates = np.where(h_try < 1 - 1e-10)[0]
                        candidates = candidates[candidates != i]
                    else:
                        candidates = np.where(h_try > 1e-10)[0]
                        candidates = candidates[candidates != i]
                    if len(candidates) == 0:
                        continue
                    j = candidates[np.argmax(np.abs(h_try[candidates] - 0.5))]
                    h_try[j] = np.clip(h_try[j] + diff, 0, 1)
                    # Re-project if needed
                    if abs(np.sum(h_try) - target_sum) > 1e-8:
                        h_try = project_onto_constraint(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h_init, target_sum, t_start, t_end, n_iter, rng, slsqp_every=100):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    for i in range(n_iter):
        temperature = t_start * (t_end / t_start) ** (i / max(n_iter - 1, 1))
        
        # Perturbation: random pair swap
        h_new = h.copy()
        n = len(h)
        n_pairs = max(1, n // 15)
        for _ in range(n_pairs):
            idx = rng.choice(n, size=2, replace=False)
            i1, i2 = idx[0], idx[1]
            max_delta = min(h[i1], 1.0 - h[i2]) * 0.5
            min_delta = max(-h[i1], h[i2] - 1.0) * 0.5
            if max_delta > abs(min_delta):
                delta = rng.uniform(min_delta, max_delta)
            else:
                continue
            h_new[i1] += delta
            h_new[i2] -= delta
        
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        # Metropolis acceptance
        if c5_new < current_c5 or rng.random() < np.exp(-(c5_new - current_c5) / max(temperature, 1e-10)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Periodic SLSQP

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                h = h_opt.copy()
                current_c5 = c5_opt
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def nesterov_projected_grad(h_init, eps, lr, n_steps, target_sum, momentum=0.95):
    h = h_init.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    min_lr = lr * 0.01
    for step in range(n_steps):
        grad = smooth_gradient(h, eps)
        v = momentum * v - lr * grad
        h_new = project_onto_constraint(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if step % 50 == 49:
            lr = max(min_lr, lr * 0.9)
    return best_h, best_c5

def coord_descent(h, target_sum, n_sweeps=30, deltas=(0.05, 0.03, 0.02, 0.01, 0.005, 0.002)):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            for d in deltas:
                for sign in (1, -1):
                    h_try = h.copy()
                    h_try[i] = np.clip(h[i] + sign * d, 0.0, 1.0)
                    diff = target_sum - np.sum(h_try)
                    if abs(diff) < 1e-12:
                        pass
                    else:
                        if diff > 0:
                            cand = np.where(h_try < 1.0 - 1e-10)[0]
                        else:
                            cand = np.where(h_try > 1e-10)[0]
                        mask = np.delete(np.ones(n, dtype=bool), i)
                        cand = cand[mask[cand]] if len(cand) else np.array([], dtype=int)
                        if len(cand) == 0:
                            continue
                        med_idx = cand[np.argmin(np.abs(h_try[cand] - 0.5))]
                        h_try[med_idx] = np.clip(h_try[med_idx] + diff, 0.0, 1.0)
                        if abs(np.sum(h_try) - target_sum) > 1e-8:
                            h_try = project_onto_constraint(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try.copy()
                        best_c5 = c5_try
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h_init, target_sum, t_start, t_end, n_iter, rng, slsqp_every):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    n = len(h)
    for i in range(n_iter):
        temp = t_start * (t_end / t_start) ** (float(i) / float(max(n_iter, 1)))
        h_new = h.copy()
        n_pairs = max(1, n // 12)
        for _ in range(n_pairs):
            idx = rng.choice(n, size=2, replace=False)
            i1, i2 = int(idx[0]), int(idx[1])
            lo_d = max(-h[i1], h[i2] - 1.0)
            hi_d = min(1.0 - h[i1], h[i2])
            if hi_d > lo_d:
                delta = rng.uniform(lo_d, hi_d)
                h_new[i1] += delta
                h_new[i2] -= delta
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < current_c5 or rng.random() < np.exp(-(c5_new - current_c5) / max(temp, 1e-12)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        if slsqp_every > 0 and (i + 1) % slsqp_every == 0:
            h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 80)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                h = h_opt.copy()
                current_c5 = c5_opt
    return best_h, best_c5
