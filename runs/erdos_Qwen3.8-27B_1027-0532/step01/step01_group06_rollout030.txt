    """Compute gradient of the log-sum-exp approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    log_sum = np.log(np.sum(np.exp((corr - cmax) / T)))
    obj = cmax + T * log_sum

    w = np.exp((corr - cmax) / T)
    w /= w.sum()

    grad = np.zeros(n)

        wk = w[k]
        if wk < 1e-15:
            continue
        # Term 1: (1 - h[i+m]) for valid i+m
        i_start = max(0, -m)
        i_end = min(n, n - m)
        if i_end > i_start:
            grad[i_start:i_end] += wk * (1.0 - h[i_start + m:i_end + m])
        # Term 2: -h[i-m] for valid i-m
        i_start2 = max(0, m)
        i_end2 = min(n, n + m)
        if i_end2 > i_start2:
            grad[i_start2:i_end2] -= wk * h[i_start2 - m:i_end2 - m]
    
    grad *= dx
    return obj, grad

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def projected_gradient_descent(h, n, T, time_limit, t_start, lr=0.01, momentum=0.9):
    """Projected gradient descent with momentum on the LSE objective."""
    best = h.copy()
    best_score = score_c5(best)
    v = np.zeros(n)
    it = 0
    
    while time.time() - t_start < time_limit:

                it += 1

import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def lse_objective_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum = max_shifted + np.log(sum_exp)
    obj = cmax + T * log_sum
    
    w = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m == 0:
            continue
        j1_start = max(0, -m)
        j1_end = min(n, n - m)
        if j1_end > j1_start:
            grad[j1_start:j1_end] += wk * (1.0 - h[j1_start + m:j1_end + m])
        j2_start = max(0, m)
        j2_end = min(n, n + m)
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= wk * h[j2_start - m:j2_end - m]
    
    grad *= dx
    return obj, grad

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def pgd_solve(h_init, n, time_limit, t_start):
    """Projected gradient descent with adaptive step size and momentum."""
    best = h_init.copy()
    best_score = score_c5(best)
    v = np.zeros(n)
    lr = 0.1
   .mu = 0.9
    T = 1.0
    count = 0
    
    while time.time() - t_start < time_limit:
        # Decrease temperature
        T = max(0.001, T * 0.995)
        
        obj, grad = lse_objective_and_grad(best, T)
        v = 0.9 * v - lr * grad
        candidate = project_feasible(best + v, n)
        s = score_c5(candidate)
        
        if s < best_score - 1e-14:
            best = candidate
            best_score = s
            count += 1
        else:
            lr = lr * 0.9
            if lr < 1e-6:
                lr = 0.1
                T = max(0.001, T * 0.5)
                v = np.zeros(n)
    
    return best, best_score, count

        sorted_idx = np.argsort(h)[::-1]
        h_b[sorted_idx[:n_ones]] = 1.0
        h_b[sorted_idx[n_ones]] = 0.5

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0

        s = np.sum(np.clip(x - mid, 0.0, 1.0))

        else:

def lse_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)

        if m > 0:
            grad[m:] -= wk * h[:n-m]

            grad[:n+m] -= wk * h[-m:]
