def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid

def corr_and_grad(h, n, T):
    """Compute smoothed max objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_k = corr * dx
    max_c = np.max(c_k)
    shifted = T * (c_k - max_c)
    shifted = np.clip(shifted, -300.0, 300.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = max_c + np.log(sum_exp) / T
    
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1

        w = weights[idx]
        if w < 1e-12:
            continue
        w_dx = w * dx
        lag = idx - (n - 1)
        if lag >= 0:
            upper = n - lag
            grad[:upper] += w_dx * (1.0 - h[lag:lag + upper])
            grad[lag:] += -w_dx * h[:upper]
        else:
            l = -lag
            upper = n - l
            grad[l:] += w_dx * (1.0 - h[:upper])
            grad[:upper] += -w_dx * h[l:n]
    
    return obj, grad

    lo = -2.0

            lo = mid
        else:

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    shifted = T * (corr - max_c)

    ev = np.exp(shifted)
    se = np.sum(ev)
    obj = max_c + np.log(se) / T
    w = ev / se

    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue

        lag = idx - (n - 1)
        if lag >= 0:

            grad[:u] += wi_dx * (1.0 - h[lag:lag+u])
            grad[lag:] += -wi_dx * h[:u]

            l = -lag
            u = n - l
