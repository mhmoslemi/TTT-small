```python
        grad[:n-lag] += (1.0 - h[lag:n])
        grad[lag:] -= h[:n-lag]
    elif 0 < -lag < n:
        lag2 = -lag
        grad[lag2:] += (1.0 - h[:n-lag2])
        grad[:n-lag2] -= h[lag2:]
    return grad

def _topk_gradient(h, k, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-k:]
    topk_vals = corr[topk_idx]
    weights = np.exp(-alpha * (topk_vals - topk_vals.max()))
    weights /= weights.sum()
    grad = np.zeros(n)
    for weight, idx in zip(weights, topk_idx):
        lag = idx - n + 1
        grad += weight * _grad_at_lag(h, lag)
    return grad, corr.max()

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / len(h))

def _slsqp_optimize(h_start, n, target_sum, alpha, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start

def _pgd_optimize(h_start, n, target_sum, t_end, topk=5):
    """Projected gradient with adaptive step size."""
    h = h_start.copy()
    best_h = h.copy()

                best_c5 = c5_new
            if c5_new < _c5_of(h):
                h = h_new
            else:
                lr *= 0.7
            lr = min(lr, 0.1)
        lr = max(lr, 0.01)
    
    return best_h, best_c5
```

- subject to corr_k(h) ≤ t for all k
- 0 ≤ h_i ≤ 1

corr_k(h0 + δ) ≈ corr_k(h0) + ∇_k(h0)·δ

- min t
- subject to corr_k(h0) + ∇_k(h0)·δ ≤ t for all k
- 0 ≤ h0_i + δ_i ≤ 1
- sum (h0_i + δ_i) = n/2

This LP can be solved efficiently! And we can iterate: solve LP, update h0, repeat.

```python
import time
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[:n-lag] += (1.0 - h[lag:n])
        grad[lag:] -= h[:n-lag]
    elif 0 < -lag < n:
        lag2 = -lag
        grad[lag2:] += (1.0 - h[:n-lag2])
        grad[:n-lag2] -= h[lag2:]
    return grad

def _all_grads(h):
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - n + 1
        if 0 <= lag < n:
            grads[k, :n-lag] = (1.0 - h[lag:n])
            grads[k, lag:] -= h[:n-lag]
        elif 0 < -lag < n:
            l2 = -lag
            grads[k, l2:] = (1.0 - h[:n-l2])
            grads[k, :n-l2] -= h[l2:]
    return grads

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / len(h))

def _smooth_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grads = _all_grads(h)
    return (w @ grads) * (2.0 / len(h))

def _slsqp_solve(h_start, n, target_sum, alpha, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, method='SLSQP', args=(alpha,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start

def _slsqp_grad_solve(h_start, n, target_sum, alpha, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, method='SLSQP', args=(alpha,),
                       bounds=bounds, constraints=cons,
                       jac=lambda x: _smooth_grad(x, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start

def _lp_solve(h, n, target_sum, timeout=10):
    """Solve the linearized problem using cvxpy."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    grads = _all_grads(h)
    
    # Build constraint matrix: grads @ h_var - t_var <= grads @ h - corr
    A = np.hstack([grads, -np.ones((n_corr, 1))])
    b = grads @ h - corr
    
    h_var = cp.Variable(n)
```

```python
def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _grad_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[:n - lag] += (1.0 - h[lag:n])
        grad[lag:] -= h[:n - lag]
    elif 0 < -lag < n:
        l2 = -lag
        grad[l2:] += (1.0 - h[:n - l2])
        grad[:n - l2] -= h[l2:]
    return grad
```
