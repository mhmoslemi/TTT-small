Let me think about the subgradient approach more carefully:

The objective is f(h) = max_k corr_k(h) * dx where corr_k(h) = sum_i h[i]*(1-h[i+k]) for valid i.

The subgradient of f with respect to h[j] is:
- Find the active set A = {k : corr_k(h) = max}
- For each active k, the gradient of corr_k with respect to h[j] is:
  - For the term h[j]*(1-h[j+k]): derivative w.r.t. h[j] is (1-h[j+k])
  - For the term h[j-k]*(1-h[j]): derivative w.r.t. h[j] is -h[j-k]
  - So total gradient for h[j] from lag k is: (1-h[j+k]) - h[j-k] (with appropriate bounds)

The subgradient is the average (or any convex combination) of gradients over the active set.

This is much more efficient than SLSQP because:
- We can compute correlations in O(n) using FFT (or np.correlate which is O(n²) but fast in practice)
- The gradient computation is also O(n) for a given active k
- We don't need to solve a QP at each iteration

def project_onto_feasible(h, target_sum):
    # Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}
    # Using bisection on Lagrange multiplier
    lo, hi = -10.0, 10.0
    for _ in range(200):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    return projected

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        if time.time() - t0 > budget_s - 40:
            break
        scale = 0.04 + 0.12 * (r / 50.0)
        h_r = project_feasible(best_h + rng.randn(n) * scale, target)
        vel_r = np.zeros(n)
        for it2 in range(4000):
            if time.time() - t0 > budget_s - 25:
                break
            h_r, vel_r = gradient_step(h_r, lags, 600.0, 0.04, 0.90, vel_r, target, top_k=10)
            sc_r = evaluate_c5(h_r)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r.copy()
        if time.time() - t0 > budget_s - 25:
            break
        ht = slsqp_polish(h_r, target, 5000, 200)
        sc = evaluate_c5(ht)
        if sc < best_score:
            best_score = sc
            best_h = ht

    # Phase 4: Try different resolutions
    for n_try in [n + 1, 2 * n, 3 * n]:
        if n_try > 2000 or time.time() - t0 > budget_s - 80:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        tgt_t = n_try / 2.0
        h_interp = project_feasible(h_interp, tgt_t)
        sc = evaluate_c5(h_interp)
        if sc < best_score:
            best_score = sc
            best_h = h_interp
        dx_t = 2.0 / n_try
        lags_t = np.arange(-(n_try - 1), n_try, dtype=np.int64)
        vel_t = np.zeros(n_try)
        max_it4 = max(2000, 800000 // n_try)
        for it3 in range(max_it4):
            if time.time() - t0 > budget_s - 40:
                break
            alpha_t = 300.0 + 500.0 * min(1.0, it3 / max_it4)
            step_t = 0.06 * (0.9998 ** it3)
            h_interp, vel_t = gradient_step(h_interp, lags_t, alpha_t, step_t, 0.88, vel_t, tgt_t, top_k=12)
            sc_t = evaluate_c5(h_interp)
            if sc_t < best_score:
                best_score = sc_t
                best_h = h_interp.copy()
        if time.time() - t0 > budget_s - 40:
            break
        ht = slsqp_polish(h_interp, tgt_t, 5000, max(100, 400000 // n_try))
        sc = evaluate_c5(ht)
        if sc < best_score:
            best_score = sc
            best_h = ht

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 15:
        n_f = len(best_h)
        tgt_f = n_f / 2.0
        for a_f in [5000, 15000, 30000]:
            if time.time() - t0 > budget_s - 5:
                break
            ht = slsqp_polish(best_h, tgt_f, a_f, 500)
            sc = evaluate_c5(ht)
            if sc < best_score:
                best_score = sc
                best_h = ht

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_c5(best_h)
    return (best_h, best_score, len(best_h))
