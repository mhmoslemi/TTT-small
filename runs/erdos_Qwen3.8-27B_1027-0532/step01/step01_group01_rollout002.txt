import numpy as np
from scipy.optimize import minimize, differential_evolution, minimize_scalar

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-x[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

        obj, h0.copy(), jac=grad, method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constr}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    if np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h0.copy()

            best_c = best_c5
            for val in np.linspace(0.0, 1.0, n_grid):
                if abs(val - h[i]) < 1e-6:
                    continue
                h_trial = h.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, n)
                c = evaluate_c5(h_trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_val = val
            if best_val != h[i]:
                h[i] = best_val
                h = project_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
