a[0] = 1-x[1], b[0] = 0, so grad[0] += w*(1-x[1]) ✓
a[n-1] = 0, b[n-1] = x[n-2], so grad[n-1] += w*(0-x[n-2]) = -w*x[n-2] ✓



Now let me also think about whether I should try a completely different approach. 

**Alternative: Use the fact that the problem is a DC (Difference of Convex) optimization**

f_k(h) = sum h[j] - sum h[j]h[j-k]

The first term is convex (linear), the second term sum h[j]h[j-k] is... let's see. The matrix for this quadratic form has 1/2 on the k-th off-diagonal. This matrix is positive semi-definite if and only if the eigenvalues are non-negative. For a circulant-like matrix with 1/2 on the k-th off-diagonal, the eigenvalues are cos(2πmk/n) for m=0,...,n-1. Some of these are negative, so the matrix is NOT positive semi-definite.

So f_k is not a DC function in a useful way.

**Alternative: Projected subgradient method**

Since the objective (max of differentiable functions) is subdifferentiable, we can use:
1. Compute the subgradient (which is a convex combination of the gradients of the active functions)
2. Take a step in the negative subgradient direction
3. Project back to the feasible set

With a decreasing step size (like 1/sqrt(t)), this converges (sublinearly).

The advantage is that each iteration is O(n^2) (for the subgradient) + O(n) (for the projection), and we don't need SLSQP's overhead.

Let me try this:
```python
def projected_subgradient(h, n, T=10000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for t in range(1, T+1):
        alpha = 0.1 / np.sqrt(t)
        # Compute subgradient
        corr = np.correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr)
        l = k_max - (n-1)
        # Subgradient of corr[k_max]
        sg = np.zeros(n)
        if l >= 0:
            sg[:n-l] += (1-h[l:])
            if l > 0:
                sg[l:] -= h[:n-l]
        else:
            sg[-l:] += (1-h[:n+l])
            sg[:n+l] -= h[-l:]
        sg *= dx
        # Take step
        h_new = h - alpha * sg
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
```

This is simple but might be slow to converge. The subgradient method has O(1/sqrt(T)) convergence, so to get 1e-4 accuracy we need T ~ 1e8 iterations, which is too many.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, eps):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-10:
            continue
        a = np.zeros(n)
        b = np.zeros(n)
        if l >= 0:
            a[:n - l] = 1.0 - x[l:]
            if l > 0:
                b[l:] = x[:n - l]
        else:
            a[-l:] = 1.0 - x[:n + l]
            b[:n + l] = x[-l:]
        grad += w * (a - b)
    grad *= dx
    return grad

def pnorm_obj(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    return (np.sum(np.abs(corr) ** p) * dx ** p) ** (1.0 / p)

def pnorm_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    obj = pnorm_obj(x, p)
    if obj < 1e-15:
        return np.zeros(n)
    weights = np.abs(corr) ** (p - 1) * np.sign(corr) / (np.sum(np.abs(corr) ** p) * dx ** (p - 1))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-12 and w > -1e-12:
            continue
        a = np.zeros(n)
        b = np.zeros(n)
        if l >= 0:
            a[:n - l] = 1.0 - x[l:]
            if l > 0:
                b[l:] = x[:n - l]
        else:
            a[-l:] = 1.0 - x[:n + l]
            b[:n + l] = x[-l:]
        grad += w * (a - b)
    grad *= dx
    return grad

def slsqp_optimize(h_init, n, eps, maxiter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj_fn(x):
        return smooth_max_obj(x, eps)
    def jac_fn(x):
        return smooth_max_grad(x, eps)
    def con_fn(x):
        return np.sum(x) - target
    try:
        res = minimize(obj_fn, h_init.copy(), jac=jac_fn, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fn}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if np.all(np.isfinite(res.x)) else None
    except Exception:
        return None

def pnorm_slsqp(h_init, n, p, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj_fn(x):
        return pnorm_obj(x, p)
    def jac_fn(x):
        return pnorm_grad(x, p)
    def con_fn(x):
        return np.sum(x) - target
    try:
        res = minimize(obj_fn, h_init.copy(), jac=jac_fn, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fn}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x if np.all(np.isfinite(res.x)) else None
    except Exception:
        return None

def make_init(n, kind, rng):

    else:
        return 0.5 * np.ones(n)
