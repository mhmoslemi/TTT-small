from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def monos_constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def monos_constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)

        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient(h_init, eps, n_steps, target_sum, lr_init=0.005):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.85
    for i in range(n_steps):
        h_look = project_onto_constraint(h - momentum * v, target_sum)
        grad = smooth_max_grad(h_look, eps)
        h_new = project_onto_constraint(h - lr * grad, target_sum)
        v = momentum * v + (h_new - h) * 0.5
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:

            best_h = h.copy()
