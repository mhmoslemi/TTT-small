```python
def compute_grad_logsumexp(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    # corr index k corresponds to shift s = k - (n-1)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        # For shift s, the valid range is i from max(0,-s) to min(n-1, n-1-s)
        # corr contribution: sum h[i]*(1-h[i+s]) over valid i
        # d/dh[j]: 
        #   if j is in range as first factor: 1 - h[j+s] (if j+s valid)
        #   if j is in range as second factor: -h[j-s] (if j-s valid)
        
        i_start = max(0, -s)
        i_end = min(n-1, n-1-s)
        
        # First factor: h[i] where i ranges from i_start to i_end
        # h[j] appears as first factor if j in [i_start, i_end] and j+s in [0, n-1]
        # h[j] appears as second factor if j = i+s for some i in [i_start, i_end], i.e., j-s in [i_start, i_end]
        
        # Simplify: for s >= 0, i ranges from 0 to n-1-s
        # First factor: j in [0, n-1-s], contribution 1 - h[j+s]
        # Second factor: j-s in [0, n-1-s], i.e., j in [s, n-1], contribution -h[j-s]
        
        if s >= 0:
            # First factor: j from 0 to n-1-s
            lo1, hi1 = 0, n - 1 - s
            grad[lo1:hi1+1] += weights[k] * (1.0 - h[lo1+s:hi1+1+s])
            # Second factor: j from s to n-1
            lo2, hi2 = s, n - 1
            grad[lo2:hi2+1] -= weights[k] * h[lo2-s:hi2+1-s]
        else:
            # s < 0, let s' = -s > 0
            # i ranges from s' to n-1
            # corr = sum_{i=s'}^{n-1} h[i] * (1-h[i-s'])
            # First factor: j in [s', n-1], contribution 1 - h[j-s']
            # Second factor: j = i - s' where i in [s', n-1], so j in [0, n-1-s'], contribution -h[j+s']
            sp = -s
            lo1, hi1 = sp, n - 1
            grad[lo1:hi1+1] += weights[k] * (1.0 - h[lo1-sp:hi1+1-sp])
            lo2, hi2 = 0, n - 1 - sp
            grad[lo2:hi2+1] -= weights[k] * h[lo2+sp:hi2+1+sp]
    
    return grad
```

```python
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)
```
