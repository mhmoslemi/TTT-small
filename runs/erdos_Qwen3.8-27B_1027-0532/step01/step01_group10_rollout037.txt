        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[idx] = 1.0
    return h_bin

def thue_morse_h(n):
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        pc = bin(i).count('1')
        if pc % 2 == 0:
            h[i] = 1.0
    return h

def evaluate_max_A(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr), np.argmax(corr)

def greedy_local_search(h, n, max_rounds=50, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    curr_A = np.correlate(h, 1.0 - h, mode="full")
    curr_max = np.max(curr_A)
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        tried = 0
        max_tries = 2000
        while tried < max_tries:
            tried += 1
            pi = rng.randint(len(ones))
            qi = rng.randint(len(zeros))
            p = ones[pi]
            q = zeros[qi]
            h[p] = 0.0
            h[q] = 1.0
            new_A = np.correlate(h, 1.0 - h, mode="full")
            new_max = np.max(new_A)
            if new_max < curr_max:
                curr_max = new_max
                curr_A = new_A
                ones[pi] = q
                zeros[qi] = p
                improved = True
                tried = 0
            else:
                h[p] = 1.0
                h[q] = 0.0
    return h, curr_max

def simulated_annealing(h, n, T_start, T_end, n_iter, rng, time_limit_s=None, t0=None):
    h = h.copy()
    curr_A = np.correlate(h, 1.0 - h, mode="full")
    curr_max = float(np.max(curr_A))
    best_h = h.copy()
    best_max = curr_max
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    cooling = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    T = T_start
    for it in range(n_iter):
        if time_limit_s is not None and t0 is not None:
            if time.time() - t0 > time_limit_s:
                break
        pi = rng.randint(n_ones)
        qi = rng.randint(n_zeros)
        p = ones_list[pi]
        q = zeros_list[qi]
        h[p] = 0.0
        h[q] = 1.0
        ones_list[pi] = q
        zeros_list[qi] = p
        new_A = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_A))
        delta = new_max - curr_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_max = new_max
            curr_A = new_A
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
        else:
            h[p] = 1.0
            h[q] = 0.0
            ones_list[pi] = p
            zeros_list[qi] = q
        T *= cooling
    return best_h, best_max

def sa_with_restart(h, n, T_start, T_end, n_iter_per_run, n_restarts, rng, time_limit_s, t0):
    best_h = h.copy()
    best_max = None
    for r in range(n_restarts):
        if time.time() - t0 > time_limit_s:
            break
        if r == 0:
            h_start = h.copy()
        else:
            h_start = best_h.copy()
            for _ in range(3 + r):
                ones = np.where(h_start == 1.0)[0]
                zeros = np.where(h_start == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                pi = rng.randint(len(ones))
                qi = rng.randint(len(zeros))
                h_start[ones[pi]] = 0.0
                h_start[zeros[qi]] = 1.0
        h_res, m_res = simulated_annealing(h_start, n, T_start, T_end, n_iter_per_run, rng,
                                           time_limit_s=time_limit_s - (time.time() - t0), t0=t0)
        if m_res < (best_max if best_max is not None else float('inf')):
            best_max = m_res
            best_h = h_res
    return best_h, best_max

def non_binary_refine(h, n, max_rounds=50):
    h = h.copy()
    target = n * 0.5
    curr_c5 = c5_value(h, 2.0 / n)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            orig = h[i]
            for val in [0.0, 1.0, 0.5]:
                if abs(val - orig) < 1e-10:
                    continue
                h[i] = val
                adj = target - np.sum(h)
                if abs(adj) > 1e-10:
                    mask = (h < 1.0 - 1e-10) if adj > 0 else (h > 1e-10)
                    if np.sum(mask) == 0:
                        h[i] = orig
                        continue
                    h[mask] = np.clip(h[mask] + adj / np.sum(mask), 0.0, 1.0)
                new_c5 = c5_value(h, 2.0 / n)
                if new_c5 < curr_c5 - 1e-12:
                    curr_c5 = new_c5
                    improved = True
                else:
                    h[i] = orig
        if not improved:
            break
    h = repair_h(h, n)
    return h

def slsqp_refine(h0, n, dx, maxiter=200):
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(corr - m))) / 30.0
    def constraint_sum(h):
        return np.sum(h) - n * 0.5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        return repair_h(res.x, n)
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.asarray(initial_h_values, dtype=np.float64)
    n_parent = len(h_parent)
    dx_parent = 2.0 / n_parent
    best_c5 = c5_value(h_parent, dx_parent)
    best_h = h_parent.copy()
    best_n = n_parent

    def time_left():
        return budget_s - (time.time() - t0)

    n = 256
    dx = 2.0 / n

    if n_parent == n:
        h_bin = to_binary(h_parent, n)
    else:
        x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False) + dx_parent / 2
        x_new = np.linspace(0.0, 2.0, n, endpoint=False) + dx / 2
        h_interp = np.interp(x_new, x_parent, h_parent)
        h_bin = to_binary(h_interp, n)

    h_bin_ls, m_ls = greedy_local_search(h_bin, n, max_rounds=30, rng=rng)
    c5_ls = float(m_ls) * dx
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_bin_ls
        best_n = n

    tm = thue_morse_h(n)
    tm_ls, tm_m = greedy_local_search(tm, n, max_rounds=20, rng=rng)
    c5_tm = float(tm_m) * dx
    if c5_tm < best_c5:
        best_c5 = c5_tm
        best_h = tm_ls
        best_n = n

    h_sa_start = best_h if best_n == n else h_bin_ls

    if time_left() > 200:
        h_sa, m_sa = sa_with_restart(
            h_sa_start, n,
            T_start=4.0, T_end=0.1,
            n_iter_per_run=300000, n_restarts=5,
            rng=rng,
            time_limit_s=min(time_left() - 60, 700),
            t0=t0
        )
        c5_sa = float(m_sa) * dx
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            best_n = n

    if time_left() > 30:
        h_final, m_final = greedy_local_search(best_h, best_n, max_rounds=50, rng=rng)
        c5_final = float(m_final) * (2.0 / best_n)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = best_n

    if time_left() > 60:
        h_nb = non_binary_refine(best_h, best_n, max_rounds=10)
        c5_nb = c5_value(h_nb, 2.0 / best_n)
        if c5_nb < best_c5:
            best_c5 = c5_nb
            best_h = h_nb

    if time_left() > 40:
        h_slsqp = slsqp_refine(best_h, best_n, 2.0 / best_n, maxiter=100)
        c5_sl = c5_value(h_slsqp, 2.0 / best_n)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_slsqp

    if time_left() > 150:
        n2 = 512
        dx2 = 2.0 / n2
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False) + (2.0 / best_n) / 2
        x_new2 = np.linspace(0.0, 2.0, n2, endpoint=False) + dx2 / 2
        h_interp2 = np.interp(x_new2, x_old, best_h)
        h_bin2 = to_binary(h_interp2, n2)
        h_bin2_ls, m2_ls = greedy_local_search(h_bin2, n2, max_rounds=20, rng=rng)
        c5_2 = float(m2_ls) * dx2
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_bin2_ls
            best_n = n2

        if time_left() > 80:
            h_sa2, m_sa2 = sa_with_restart(
                best_h, best_n,
                T_start=4.0, T_end=0.2,
                n_iter_per_run=200000, n_restarts=3,
                rng=rng,
                time_limit_s=min(time_left() - 20, 100),
                t0=t0
            )
            c5_sa2 = float(m_sa2) * (2.0 / best_n)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2
                best_n = best_n

    if time_left() > 20:
        h_final2, m_final2 = greedy_local_search(best_h, best_n, max_rounds=30, rng=rng)
        c5_f2 = float(m_final2) * (2.0 / best_n)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_final2

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
