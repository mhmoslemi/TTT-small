    w = exp_s / np.sum(exp_s)
    
    obj = (m + (1.0/eps) * np.log(np.sum(exp_s))) * dx
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n-1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - x[:n-lag])
            grad[:n-lag] -= wk * x[lag:]
        else:
            p = -lag
            grad[:n-p] += wk * (1.0 - x[p:])
            grad[p:] -= wk * x[:n-p]
    grad *= dx
    return obj, grad

Now let me think about the simulated annealing phase. The key idea:
- Current solution: h
- Temperature: T (starts high, decreases geometrically)
- Perturbation: pick 1-3 random indices, change their values by small random amounts, then project to feasibility
- Accept if new C₅ ≤ old C₅, or with probability exp(-(new-old)/T) if worsening

So to increase R(k*), I should increase h at positions that are paired with other h values under shift k*.

5. Check if min_k R(k) improved (i.e., C₅ decreased)
6. If yes, accept; if no, try a smaller step or different direction

This is essentially a **gradient ascent on min_k R(k)** using the subgradient at the worst k.

_global_eps = 100.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x):
    n = _global_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    eps = _global_eps
    shifted = eps * (corr - m)
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return (m + (1.0/eps) * log_sum) * dx

def smooth_max_grad(x):
    n = _global_n
    dx = 2.0 / n
    eps = _global_eps
    corr = np.correlate(x, 1 - x, mode="full")

    exp_s = np.exp(shifted - max_shifted)

    grad = np.zeros(n)
    ncorr = 2 * n - 1

        lag = k - (n - 1)

        else:
            p = -lag
            grad[:n-p] += wk * (1.0 - x[p:])
            grad[p:] -= wk * x[:n-p]

        # Need: h_new[j] + delta <= 1 → delta <= 1 - h_new[j]
        # Need: h_new[i] - delta <= 1 → delta >= h_new[i] - 1 (always true if delta > 0 and h_new[i] <= 1)
        # Need: h_new[j] + delta >= 0 → delta >= -h_new[j] (always true if delta > 0)
        # If delta < 0: h_new[i] + |delta|, h_new[j] - |delta|
        # Need: h_new[i] + |delta| <= 1 → |delta| <= 1 - h_new[i]
        # Need: h_new[j] - |delta| >= 0 → |delta| <= h_new[j]
        
        if delta > 0:
            delta = min(delta, h_new[i], 1.0 - h_new[j])
        else:
            delta = max(delta, -(1.0 - h_new[i]), -h_new[j])
        
        if abs(delta) > 1e-12:
            h_new[i] -= delta
            h_new[j] += delta
    return h_new

    best_h = h_work.copy()
    best_c5 = evaluate_c5(h_work)

    # Phase 1: SLSQP from parent with increasing smoothness
    for eps in [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]:
        if time.time() - start_time > budget_s - 600:
            break
        h_new = slsqp_run(h_work, n, bounds, eps, 5000)
        if h_new is not None:
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_work = h_new.copy()

    # Phase 2: Multi-restart SLSQP from perturbed points
    n_restarts = 200
    for r in range(n_restarts):
        if time.time() - start_time > budget_s - 300:
            break
        scale = 0.01 + 0.03 * (r / n_restarts)
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project_feasible(h_pert, n)
        eps_choice = [200.0, 500.0, 1000.0][r % 3]
        h_new = slsqp_run(h_pert, n, bounds, eps_choice, 1500)
        if h_new is not None:
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 3: Simulated annealing with sum-preserving perturbations
    h_sa = best_h.copy()
    c5_sa = best_c5
    T = 0.002
    n_sa = 50000
    for it in range(n_sa):
        if time.time() - start_time > budget_s - 150:
            break
        T *= 0.99985
        h_trial = perturb_sum_preserving(h_sa, n, 0.015)
        c5_trial = evaluate_c5(h_trial)
        delta = c5_trial - c5_sa
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-9)):
            h_sa = h_trial
            c5_sa = c5_trial
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: SLSQP from best SA point with high smoothness
    if time.time() - start_time < budget_s - 80:
        for eps in [2000.0, 5000.0]:
            h_new = slsqp_run(best_h, n, bounds, eps, 3000)
            if h_new is not None:
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Phase 5: Targeted subgradient on worst lag
    h_work = best_h.copy()
    n_sub = 40000
    step_size = 0.003
    dx_val = 2.0 / n
    for it in range(n_sub):
        if time.time() - start_time > budget_s - 50:
            break
        corr = np.correlate(h_work, 1 - h_work, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        c5_work = np.max(corr) * dx_val

        g = np.zeros(n)
        if lag >= 0:
            g[lag:] = 1.0 - h_work[:n - lag]
            g[:n - lag] -= h_work[lag:]
        else:
            p = -lag
            g[:n - p] = 1.0 - h_work[p:]
            g[p:] -= h_work[:n - p]

        h_trial = h_work - step_size * g
        h_trial = project_feasible(h_trial, n)
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode="full")
        c5_trial = np.max(corr_trial) * dx_val

        if c5_trial < c5_work - 1e-14:
            h_work = h_trial
            step_size = max(0.0005, step_size * 0.995)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_work.copy()
        else:
            step_size = max(0.0003, step_size * 0.97)

    # Phase 6: SLSQP from subgradient result
    if time.time() - start_time < budget_s - 30:
        h_new = slsqp_run(h_work, n, bounds, 5000.0, 3000)
        if h_new is not None:
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 7: More restarts from best solution if time permits
    if time.time() - start_time < budget_s - 30:
        for r in range(50):
            if time.time() - start_time > budget_s - 15:
                break
            h_pert = best_h + 0.005 * np.random.randn(n)
            h_pert = project_feasible(h_pert, n)
            h_new = slsqp_run(h_pert, n, bounds, 3000.0, 1000)
            if h_new is not None:
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Phase 8: Fine coordinate descent polish
    for sweep in range(200):
        if time.time() - start_time > budget_s - 3:
            break
        improved = False
        for idx in range(n):
            for delta in [-0.0005, 0.0005, -0.001, 0.001]:
                h_trial = best_h.copy()
                h_trial[idx] += delta
                h_trial = project_feasible(h_trial, n)
                c5 = evaluate_c5(h_trial)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h_trial.copy()
                    improved = True
                    break
        if not improved:
            break

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
