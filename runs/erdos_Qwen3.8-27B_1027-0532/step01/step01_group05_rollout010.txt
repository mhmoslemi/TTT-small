            g[0:n-l] += -w * h[l:n]

    cv = np.sum(h) - n / 2.0

    return g

                  pgtol=pgtol, maxiter=max_iter, iprint=0, factr=1e6)

    best_h = h.copy()

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a = alpha
        cur = compute_c5(h, n)

            v = compute_c5(h_try, n)

            a *= 0.5
        else:
            break
        h = h_try
        if v < best_val:
            best_val = v
            best_h = h.copy()
        alpha = min(a * 1.3, 0.05)
    return best_h, best_val

        else:

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
