    # A[j] = sum_m w[m]*h[j+n-1-m]
    # This is: for each j, sum over m of w[m]*h[j+n-1-m]
    # = convolution of h and w (reversed)
    
    # Let me define w_rev[m] = w[2n-2-m] (reverse of w)
    # Then A[j] = sum_m w[m]*h[j+n-1-m] = sum_m w_rev[2n-2-m]*h[j+n-1-m]
    # Let p = 2n-2-m, so m = 2n-2-p
    # A[j] = sum_p w_rev[p]*h[j+n-1-(2n-2-p)] = sum_p w_rev[p]*h[j+1+n-1-2n+2+p]
    #      = sum_p w_rev[p]*h[j-p+1]  ... this is getting confusing
    
    # Let me just use scipy.signal or manual convolution
    
    # Actually, the simplest: A[j] = (convolve)(h, w_padded)
    # where w_padded is w placed with appropriate offset
    
    # Let me think differently. Define vector v of length 2n-1 where v[k] = w[k+n-1] for k=-(n-1),...(n-1)
    # Then A[j] = sum_k v[k] * h[j+k]
    
    # This is the cross-correlation of h with v (conjugate): np.correlate(h, v_conj, 'full')
    # np.correlate(h, v, 'full')[j] = sum_i h[i] * conj(v[i + (len(h)-1) - j])
    # Hmm, let me just use the direct formula.
    
    # For small n, just do it directly with numpy broadcasting:
    # j = 0,...,n-1; m = 0,...,2n-2; k = n-1-m
    # A[j] = sum_m w[m] * h[j+k_m] (with h[min(k_m, n-1+j)] if valid)
    
    # Create index matrix
    j_idx = np.arange(n)[:, None]  # (n, 1)
    k_vals = np.arange(n-1, -n, -1)  # n-1, n-2, ..., -(n-1), length 2n-1
    # k_vals[m] = n-1-m
    valid1 = (j_idx + k_vals[None, :] >= 0) & (j_idx + k_vals[None, :] < n)
    valid2 = (j_idx - k_vals[None, :] >= 0) & (j_idx - k_vals[None, :] < n)
    
    # A[j] = sum_m w[m] * h[j+k_m] * valid1
    h_idx1 = np.clip(j_idx + k_vals[None, :], 0, n-1)
    A = (w[None, :] * h[h_idx1] * valid1).sum(axis=1)
    
    # B[j] = sum_m w[m] * h[j-k_m] * valid2
    h_idx2 = np.clip(j_idx - k_vals[None, :], 0, n-1)
    B = (w[None, :] * h[h_idx2] * valid2).sum(axis=1)
    
    grad = dx * (w.sum() - A - B)
    return grad

    c = np.correlate(h, one_minus_h, mode='full')
    n1 = n - 1

        else:
            a = m1

    f1 = np.max(base + d)
    ft = np.max(base + t_opt * d)

        return 0.0

        return 1.0

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))

    valid2 = (j_idx - k_vals >= 0) & (j_idx - k_vals < n)

    h_idx2 = np.clip(j_idx - k_vals, 0, n - 1).astype(int)
    
    A = (w[None, :] * h[h_idx1] * valid1).sum(axis=1)
    B = (w[None, :] * h[h_idx2] * valid2).sum(axis=1)
    
    grad = dx * (w.sum() - A - B)
    return grad

def _slsqp_obj(val):
    return val[-1]

def _slsqp_ineq(val, n, m_idx):
    h = val[:n]
    t = val[n]
    m = m_idx - (n - 1)
    dx = 2.0 / n
    c_val = 0.0
    for j in range(n):
        if 0 <= j + m < n:
            c_val += h[j + m] * (1.0 - h[j])
    return t - c_val * dx

def _slsqp_eq(val, n, _):
    return np.sum(val[:n]) - n / 2.0

def run_slsqp(h_init, n, maxiter=200):
    target = n / 2.0
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = []
    for k in range(2 * n - 1):
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'args': (n, k)})
    cons.append({'type': 'eq', 'fun': _slsqp_eq, 'args': (n, 0)})
    res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_feasible(h_opt, target)
    return h_opt

def coord_descent_1d(h, j, n):
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')
    n1 = n - 1

    # k = n1 - m for m = 0, ..., 2n-2
    # d[m] = (1-h[j+k]) if 0<=j+k<n, minus h[j-k] if 0<=j-k<n
    k_vals = n1 - np.arange(2 * n - 1)  # k for each m

    jk = j + k_vals
    term1 = np.where((jk >= 0) & (jk < n), 1.0 - h[jk[(jk >= 0) & (jk < n)]], 0.0)
    
    # For term -h[j-k]: valid where 0 <= j-k < n
    jk2 = j - k_vals
    term2 = np.where((jk2 >= 0) & (jk2 < n), -h[jk2[(jk2 >= 0) & (jk2 < n)]], 0.0)
    
    d = term1 + term2
    base = c - h[j] * d
    
    # Ternary search
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(base + m1 * d) < np.max(base + m2 * d):
            b = m2
        else:
            a = m1
    
    t_opt = (a + b) / 2.0
    f0 = np.max(base)
    f1 = np.max(base + d)
    ft = np.max(base + t_opt * d)
    
    if f0 <= f1 and f0 <= ft:
        return 0.0
    elif f1 <= f0 and f1 <= ft:
        return 1.0
    else:
        return t_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_vec(h, w, n):
    dx = 2.0 / n
    n1 = n - 1
    j_idx = np.arange(n)[:, None].astype(float)
    k_vals = (n1 - np.arange(2 * n - 1)).astype(float)[None, :]
    valid1 = (j_idx + k_vals >= 0) & (j_idx + k_vals < n)
    valid2 = (j_idx - k_vals >= 0) & (j_idx - k_vals < n)
    h_idx1 = np.clip(j_idx + k_vals, 0, n - 1).astype(int)
    h_idx2 = np.clip(j_idx - k_vals, 0, n - 1).astype(int)
    A = (w[None, :] * h[h_idx1] * valid1).sum(axis=1)
    B = (w[None, :] * h[h_idx2] * valid2).sum(axis=1)
    return dx * (w.sum() - A - B)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    s = exp_shifted.sum()
    w = exp_shifted / s
    log_sum = max_shifted + np.log(s)
    smooth_val = c_max + log_sum / alpha
    grad = compute_grad_vec(h, w, n)
    return smooth_val, grad

def phases_adam(h_init, n, deadline, alphas, lrs, max_iters_per_phase):
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for phase in range(len(alphas)):
        if time.time() > deadline:
            break
        alpha = alphas[phase]
        lr = lrs[phase]
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t = 0
        stuck = 0
        for t in range(1, max_iters_per_phase + 1):
            if time.time() > deadline:
                break
            smooth_val, grad = smooth_max_and_grad(h, alpha)
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-10:
                best_c5 = c5_val
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
            if stuck > 3000:
                break
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_val + 1e-12:
                h = h_new
            else:
                lr *= 0.7
                if lr < 1e-12:
                    break
    return best_h, best_c5

def coord_descent_1d(h, j, n):
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')
    n1 = n - 1
    k_vals = n1 - np.arange(2 * n - 1)
    jk = j + k_vals
    valid1 = (jk >= 0) & (jk < n)
    jk2 = j - k_vals
    valid2 = (jk2 >= 0) & (jk2 < n)
    term1 = np.zeros(2 * n - 1)
    term2 = np.zeros(2 * n - 1)
    term1[valid1] = 1.0 - h[jk[valid1]]
    term2[valid2] = -h[jk2[valid2]]
    d = term1 + term2
    base = c - h[j] * d
    a, b = 0.0, 1.0
    for _ in range(45):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(base + m1 * d) < np.max(base + m2 * d):
            b = m2
        else:
            a = m1
    t_opt = (a + b) / 2.0
    f0 = np.max(base)
    f1 = np.max(base + d)
    ft = np.max(base + t_opt * d)
    if f0 <= f1 and f0 <= ft:
        return 0.0
    elif f1 <= f0 and f1 <= ft:
        return 1.0
    else:
        return t_opt

def coord_descent(h, n, n_sweeps, deadline):

    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

            best_h = h.copy()
    return best_h, best_c5

def _slsqp_obj(val):
    return val[-1]

def _slsqp_ineq(val, n, m_idx):
    h = val[:n]
    t = val[n]
    m = m_idx - (n - 1)
    dx = 2.0 / n
    c_val = 0.0
    for j in range(n):
        if 0 <= j + m < n:
            c_val += h[j + m] * (1.0 - h[j])
    return t - c_val * dx

def _slsqp_eq(val, n, _):
    return np.sum(val[:n]) - n / 2.0

def run_slsqp_active(h_init, n, top_k=10, maxiter=300, deadline=None):
    target = n / 2.0
    dx = 2.0 / n
    c = np.correlate(h_init, 1.0 - h_init, mode='full') * dx
    top_indices = np.argsort(c)[-top_k:]
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = []
    for k in top_indices:
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'args': (n, k)})
    cons.append({'type': 'eq', 'fun': _slsqp_eq, 'args': (n, 0)})
    res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-11})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_feasible(h_opt, target)
    return h_opt

        if _time_left() > 80 and idx < 4:
            try:
                h_sl = run_slsqp_active(best_h, n, top_k=10, maxiter=300)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()
            except Exception:
                pass

    if _time_left() > 40 and best_h is not None:
        for _ in range(8):
            if _time_left() < 30:
                break
            noise_scale = 0.02 + 0.03 * rng.random()
            h_pert = best_h + rng.normal(0, noise_scale, n)
            h_pert = project_feasible(h_pert, target)
            h_cd_p, c5_cd_p = coord_descent(h_pert, n, 400, deadline)
            if c5_cd_p < best_c5:
                best_c5 = c5_cd_p
                best_h = h_cd_p.copy()
            if _time_left() > 30:
                h_ad_p, c5_ad_p = phases_adam(h_cd_p, n, deadline,
                                              alphas[2:], lrs[2:], max_iters[2:])
                if c5_ad_p < best_c5:
                    best_c5 = c5_ad_p
                    best_h = h_ad_p.copy()

    if _time_left() > 60 and best_h is not None:
        try:
            h_sl = run_slsqp_active(best_h, n, top_k=15, maxiter=400)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
        except Exception:
            pass

    if _time_left() > 20 and best_h is not None:
        h_cd_final, c5_final = coord_descent(best_h, n, 300, deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cd_final.copy()

    if best_h is None:
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
