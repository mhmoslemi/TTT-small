        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    exps = np.exp(tau_corr - max_tc)
    sum_exps = np.sum(exps)
    obj = (max_tc + np.log(sum_exps)) / tau * dx
    weights = exps / sum_exps
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            idx = j_lo - lag
            grad[j_lo:j_hi + 1] += w * (1.0 - h[idx:idx + (j_hi - j_lo + 1)])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            idx2 = j2_lo + lag
            grad[j2_lo:j2_hi + 1] += w * (-h[idx2:idx2 + (j2_hi - j2_lo + 1)])
    return obj, grad * dx

        o, _ = lse_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = lse_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]

def coordinate_descent(h0, n, target_sum, deadline, seed=42, n_sweeps=0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    k_vals = np.arange(-(n-1), n)
    no_iter = 0
    sweep_count = 0
    thresh = 30
    
    while time.time() < deadline and (n_sweeps == 0 or sweep_count < n_sweeps):
        sweep_count += 1
        order = rng.permutation(n)
        for i in order:
            hi_val = h[i]
            i1 = i + k_vals
            i2 = i - k_vals
            valid1 = (i1 >= 0) & (i1 < n)
            valid2 = (i2 >= 0) & (i2 < n)
            b = np.zeros(2*n - 1)
            if np.any(valid1):
                b[valid1] += (1.0 - h[i1[valid1]])
            if np.any(valid2):
                b[valid2] -= h[i2[valid2]]
            
            a = corr - hi_val * b
            
            # Grid + refine
            t_grid = np.linspace(0.0, 1.0, 41)
            f_vals = np.max(a[None, :] + t_grid[:, None] * b[None, :], axis=1)
            t_best = t_grid[np.argmin(f_vals)]

            hi_t = min(1.0, (t_best + 1.0/41.0))
            for _ in range(12):
                mid1 = lo + (hi_t - lo) / 3.0
                mid2 = hi_t - (hi_t - lo) / 3.0
                f1 = np.max(a + mid1 * b)
                f2 = np.max(a + mid2 * b)
                if f1 < f2:
                    hi_t = mid2
                else:
                    lo = mid1
            t_opt = (lo + hi_t) / 2.0
            
            if abs(t_opt - hi_val) > 1e-12:
                delta = t_opt - hi_val
                corr += delta * b
                h[i] = t_opt
        
        h = project_onto_constraint(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

def worst_case_pgd(h0, n, target_sum, deadline, max_iter=5000, lr_init=0.02):

            break

        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)

        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            idx = j_lo - lag
            grad[j_lo:j_hi + 1] += w * (1.0 - h[idx:idx + (j_hi - j_lo + 1)])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            idx2 = j2_lo + lag
            grad[j2_lo:j2_hi + 1] += w * (-h[idx2:idx2 + (j2_hi - j2_lo + 1)])
    return obj, grad * dx

def slsqp_optimize(h0, n, target_sum, deadline, tau=50.0, maxiter=200):
    def obj_fun(x):
        o, _ = lse_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = lse_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            c5 = compute_c5(h_opt)
            if c5 < 1.0:
                return h_opt, c5
    except Exception:
        pass
    return h0, compute_c5(h0)

def coordinate_descent(h0, n, target_sum, deadline, seed=42, stabil_threshold=40):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    k_vals = np.arange(-(n - 1), n)
    no_improve = 0
    t_grid = np.linspace(0.0, 1.0, 33)

    while time.time() < deadline:
        order = rng.permutation(n)
        for i in order:
            hi_val = h[i]
            i1 = i + k_vals
            i2 = i - k_vals
            valid1 = (i1 >= 0) & (i1 < n)
            valid2 = (i2 >= 0) & (i2 < n)
            b = np.zeros(2 * n - 1)
            if np.any(valid1):
                b[valid1] += (1.0 - h[i1[valid1]])
            if np.any(valid2):
                b[valid2] -= h[i2[valid2]]
            a = corr - hi_val * b
            f_vals = np.max(a[None, :] + t_grid[:, None] * b[None, :], axis=1)
            idx_best = np.argmin(f_vals)
            t_best = t_grid[idx_best]
            if idx_best > 0:
                lo = t_grid[idx_best - 1]
            else:
                lo = 0.0
            if idx_best < len(t_grid) - 1:
                hi_t = t_grid[idx_best + 1]
            else:
                hi_t = 1.0
            for _ in range(14):
                mid1 = lo + (hi_t - lo) / 3.0
                mid2 = hi_t - (hi_t - lo) / 3.0
                f1 = np.max(a + mid1 * b)
                f2 = np.max(a + mid2 * b)
                if f1 < f2:
                    hi_t = mid2
                else:
                    lo = mid1
            t_opt = (lo + hi_t) / 2.0
            if abs(t_opt - hi_val) > 1e-12:
                delta = t_opt - hi_val
                corr += delta * b
                h[i] = t_opt
        h = project_onto_constraint(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > stabil_threshold:
                break
    return best_h, best_c5

def cd_with_perturbation(h0, n, target_sum, deadline, seed=42, n_restarts=5, perturb_mag=0.05):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    rng = np.random.RandomState(seed)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + perturb_mag * rng.randn(n)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_onto_constraint(h_pert, target_sum)
        c5_pert = compute_c5(h_pert)
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) / (n_restarts - r))
        h_opt, c5_opt = coordinate_descent(h_pert, n, target_sum, sub_deadline, seed=seed + r * 100,
                                           stabil_threshold=20)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            perturb_mag *= 0.7
    return best_h, best_c5

def worst_case_pgd(h0, n, target_sum, deadline, max_iter=8000, lr_init=0.025):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.97
            if lr < 0.001:
                lr = lr_init * 0.6
                h = best_h + 0.004 * np.random.randn(n)
                h = np.clip(h, 0, 1)
                h = project_onto_constraint(h, target_sum)
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def binary_init(n, target_sum, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    positions = rng.choice(n, size=int(target_sum), replace=False)
    h[positions] = 1.0
    return h

def sinusoidal_init(n, target_sum, freq, phase, seed=0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_onto_constraint(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 12

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0

    n_base = len(best_h) if best_h is not None else 120

    # Phase 1: Coordinate descent (exact 1D subproblems)
    if best_h is not None and time.time() < deadline - 250:
        h_opt, c5_opt = coordinate_descent(best_h, n_base, n_base / 2.0, deadline - 250, seed=seed,
                                           stabil_threshold=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: SLSQP with moderate tau
    if best_h is not None and time.time() < deadline - 200:
        h_opt, c5_opt = slsqp_optimize(best_h, n_base, n_base / 2.0, deadline - 200, tau=40.0, maxiter=150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: CD after SLSQP
    if best_h is not None and time.time() < deadline - 150:
        h_opt, c5_opt = coordinate_descent(best_h, n_base, n_base / 2.0, deadline - 150, seed=seed + 1,
                                           stabil_threshold=40)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: SLSQP with higher tau
    if best_h is not None and time.time() < deadline - 120:
        h_opt, c5_opt = slsqp_optimize(best_h, n_base, n_base / 2.0, deadline - 120, tau=80.0, maxiter=150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: CD with perturbation (escape local optima)
    if best_h is not None and time.time() < deadline - 100:
        h_opt, c5_opt = cd_with_perturbation(best_h, n_base, n_base / 2.0, deadline - 100,
                                             seed=seed + 10, n_restarts=4, perturb_mag=0.08)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Try higher resolution
    if best_h is not None and time.time() < deadline - 80:
        n_high = n_base + 24
        h_high = interpolate_h(best_h, n_high)
        h_high = project_onto_constraint(h_high, n_high / 2.0)
        h_opt, c5_opt = coordinate_descent(h_high, n_high, n_high / 2.0, deadline - 40, seed=seed + 20,
                                           stabil_threshold=30)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: SLSQP on current best
    if best_h is not None and time.time() < deadline - 50:
        n_cur = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, n_cur, n_cur / 2.0, deadline - 30, tau=100.0, maxiter=120)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: More CD with different perturbation
    if best_h is not None and time.time() < deadline - 40:
        n_cur = len(best_h)
        h_opt, c5_opt = cd_with_perturbation(best_h, n_cur, n_cur / 2.0, deadline - 40,
                                             seed=seed + 50, n_restarts=3, perturb_mag=0.03)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final CD
    if best_h is not None and time.time() < deadline - 20:
        n_cur = len(best_h)
        h_opt, c5_opt = coordinate_descent(best_h, n_cur, n_cur / 2.0, deadline - 10, seed=seed + 100,
                                           stabil_threshold=30)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 10: Worst-case PGD final polish
    if best_h is not None and time.time() < deadline - 3:
        n_cur = len(best_h)
        h_opt, c5_opt = worst_case_pgd(best_h, n_cur, n_cur / 2.0, deadline, max_iter=8000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
