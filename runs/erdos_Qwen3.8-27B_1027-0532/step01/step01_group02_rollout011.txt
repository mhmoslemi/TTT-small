def evaluate_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break

            mask = h < 1.0
            if mask.sum() == 0:

def binary_anneal(b, n, n_iter, t_start, t_end, rng):
    """Simulated annealing on binary sequences via swaps."""
    current = b.copy()
    current_c5 = evaluate_c5(current)
    best = current.copy()
    best_c5 = current_c5
    
    ones = np.where(current == 1.0)[0]
    zeros = np.where(current == 0.0)[0]
    
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter - 1))
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

            continue

        old_bi, old_bj = current[i], current[j]
        current[i] = old_bj
        current[j] = old_bi

        delta = new_c5 - current_c5

            # Update ones/zeros lists

def min_of_max_linear(b, m, L, U):
