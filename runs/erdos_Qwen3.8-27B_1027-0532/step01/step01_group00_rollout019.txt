def subgradient(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L >= 0:
        grad[:n-L] += (1 - h[L:])
        grad[L:] -= h[:n-L]
    else:
        M = -L
        grad[M:] += (1 - h[:n-M])
        grad[:n-M] -= h[M:]
    return grad

1. Start with T = current best
2. Find the worst lag k*
3. Minimize overlap(k*) subject to all other overlaps ≤ T
4. Decrease T slightly
5. Repeat

This is like a "dichotomic" or "cutting plane" approach.

Also, the problem is invariant under time reversal: h(x) → h(2-x) (or in discrete form, reversing the array). This is because the correlation is symmetric.

            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 3: Subgradient optimization (handles non-smooth max directly)
    if time_left() > 90:
        n_sub = min(10000, int(time_left() * 8))
        h_sub, raw_sub = subgrad_optimize(best_h, n_sub, 0.8, n / 2.0, 5)
        s_sub = raw_sub * (2.0 / n)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()
        # Polish the subgradient result
        if time_left() > 60:
            alpha_p = 2000
            penalty_p = alpha_p * 50.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_sub, alpha_p, penalty_p, 500, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 4: Use symmetry h -> 1-h for a new starting point
    if time_left() > 90:
        h_sym = 1.0 - best_h
        h_sym = project_box_sum(h_sym, n / 2.0)
        for alpha in [200, 800, 2000]:
            if time_left() < 60:
                break
            penalty = alpha * 30.0
            bounds = [(0.0, 1.0)] * n
            res_x = lbfgsb_optimize(h_sym, alpha, penalty, 600, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                h_sym = h_new

    # Phase 5: Binary local search
    if time_left() > 60:
        n_iters = min(8000, int(time_left() * 6))
        h_bin, s_bin = binary_local_search(best_h, n_iters, rng)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
        # Also try perturbing first
        if time_left() > 40:
            h_pert = best_h + rng.randn(n) * 0.08
            h_pert = project_box_sum(h_pert, n / 2.0)
            h_bin2, s_bin2 = binary_local_search(h_pert, min(5000, int(time_left() * 4)), rng)
            if s_bin2 < best_score:
                best_score = s_bin2
                best_h = h_bin2.copy()

    # Phase 6: Multi-resolution refinement
    if time_left() > 120:
        for n_try in [n * 2, n * 3]:
            if n_try > 2500 or time_left() < 60:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_box_sum(h_init, n_try / 2.0)
            for alpha in [100, 500, 2000]:
                if time_left() < 30:
                    break
                penalty = alpha * 25.0
                bounds = [(0.0, 1.0)] * n_try
                maxiter = max(50, 250000 // n_try)
                res_x = lbfgsb_optimize(h_init, alpha, penalty, maxiter, bounds)
                if res_x is not None:
                    h_fine = project_box_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    s = evaluate_score(h_fine)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_fine.copy()
                        h_init = h_fine.copy()

    # Phase 7: Final high-precision refinement
    if time_left() > 30:
        n_final = len(best_h)
        for alpha in [5000, 10000]:
            if time_left() < 20:
                break
            penalty = alpha * 100.0
            bounds = [(0.0, 1.0)] * n_final
            res_x = lbfgsb_optimize(best_h, alpha, penalty, 1000, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
            bounds = [(0.0, 1.0)] * n_final
            res_x = slsqp_optimize(best_h, alpha, 500, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Final projection
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
