The full correlation is:
FullCorr[m] = sum_{valid i} h[i] * (1-h[i + m - (n-1)])

Let lag = m - (n-1). Then:
FullCorr[m] = sum_{i: 0<=i<n, 0<=i+lag<n} h[i] * (1-h[i+lag])
            = sum_{i: max(0,-lag) <= i < min(n, n-lag)} h[i] * (1-h[i+lag])

d(FullCorr[m])/d(h[j]) = 
  - sum over i where i+lag=j and i is valid: h[i] = h[j-lag] (from the -(h[i+lag]) term)
  + sum over i where i=j and i+lag is valid: (1-h[i+lag]) = (1-h[j+lag]) (from the h[i] term)

So:
d(FullCorr[m])/d(h[j]) = (1-h[j+lag]) * [0 <= j+lag < n] - h[j-lag] * [0 <= j-lag < n]

where [condition] is 1 if condition is true, 0 otherwise.

Wait, let me be more careful. Let f(h) = sum_i h[i] * (1-h[i+lag]) where i ranges over valid indices.

df/dh[j] = sum_i [d/dh[j] (h[i])] * (1-h[i+lag]) + h[i] * [d/dh[j](1-h[i+lag])]
         = (1-h[j+lag]) * [j+lag valid] - h[j-lag] * [j-lag valid]

where "j+lag valid" means 0 <= j < n and 0 <= j+lag < n (so that h[j] appears as h[i] with i=j)
and "j-lag valid" means 0 <= j-lag < n (so that h[j] appears as h[i+lag] with i=j-lag)

So the subgradient of max_k FullCorr[k] at the maximizing shift k* (with lag* = k* - (n-1)) is:
sg[j] = (1-h[j+lag*]) * [0<=j+lag*<n] - h[j-lag*] * [0<=j-lag*<n]

And we can do a projected subgradient step:

where eta is a step size.

```python
def subgradient_direction(h, k_star):
    """Compute the subgradient of corr[k_star] w.r.t. h"""
    n = len(h)
    lag = k_star - (n - 1)
    sg = np.zeros(n)
    for j in range(n):
        # Term 1: h[j] appears as h[i] with i=j, contributes (1-h[j+lag]) if j+lag is valid
        if 0 <= j + lag < n:
            sg[j] += (1.0 - h[j + lag])
        # Term 2: h[j] appears as h[i+lag] with i=j-lag, contributes -h[j-lag] if j-lag is valid
        if 0 <= j - lag < n:
            sg[j] -= h[j - lag]
    return sg
```

import numpy as np
from scipy.optimize import minimize
import time

_current_n = 0
_current_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def cons_fun(h):
    return np.sum(h) - _current_target


def cons_jac(h):
    return np.ones(_current_n)


def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def obj_power(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr ** p) ** (1.0 / p))


def subgradient_at_shift(h, k_index):
    n = len(h)
    lag = k_index - (n - 1)
    sg = np.zeros(n)
    if lag == 0:
        sg = 1.0 - h.copy()
    else:
        valid_plus = np.arange(max(0, -lag), min(n, n - lag))
        if len(valid_plus) > 0:
            sg[valid_plus] += (1.0 - h[valid_plus + lag])
        valid_minus = np.arange(max(0, lag), min(n + lag, n))
        if len(valid_minus) > 0:
            sg[valid_minus] -= h[valid_minus - lag]
    return sg


def weighted_subgradient(h, top_k, alpha_w):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[idx]
    w = np.exp(alpha_w * (vals - np.max(vals)))
    w = w / np.sum(w)
    sg_total = np.zeros(len(h))
    for i in range(len(idx)):
        sg_total += w[i] * subgradient_at_shift(h, idx[i])
    return sg_total, corr


def obj_topk_focus(h, k, alpha_extra):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[idx]
    base = np.max(top_vals)
    shifted = alpha_extra * (top_vals - base)
    shifted = np.clip(shifted, -500.0, 0.0)
    smooth_part = base + np.log(np.sum(np.exp(shifted))) / alpha_extra
    rest_sum = np.sum(corr) - np.sum(top_vals)
    n_total = len(corr)
    penalty = 0.01 * rest_sum / (n_total - k)
    return float(smooth_part + penalty)

    _current_n = n_init
    _current_target = ts_init
    bounds = [(0.0, 1.0)] * n_init
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    h_sg = best_h.copy()

    base_eta = 0.05
    for t in range(n_sg_steps):
        if time_left() < 60:
            break
        eta = base_eta / (1.0 + 0.05 * t)
        top_k = 5

        sg, corr = weighted_subgradient(h_sg, top_k, alpha_w)
        h_candidate = h_sg - eta * sg
        h_candidate = project_feasible(h_candidate, ts_init)
        s_candidate = compute_c5(h_candidate)
        if s_candidate < best_score:

            best_h = h_candidate
        h_sg = h_candidate

                          options={'maxiter': 4000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts_init)

    _current_target = ts_use
