**Faster coordinate descent:**

Instead of computing the full correlation each time, I can update only the affected parts.

When I change h[i] from old to new (Δ = new - old), which f(s) values change?

f(s) = dx * sum_j h[j]*(1-h[j+s]) (for s ≥ 0, j from 0 to n-1-s)

If h[i] changes by Δ:
- For s ≥ 0: f(s) changes by dx * Δ * (1-h[i+s]) if i+s < n, and by -dx * Δ * h[i-s] if i-s ≥ 0
- For s < 0 (let r = -s > 0): f(-r) = dx * sum_j h[j]*(1-h[j-r]) changes by dx * Δ * (1-h[i-r]) if i-r ≥ 0, and by -dx * Δ * h[i+r] if i+r < n

So changing h[i] affects ALL shifts s. The update is:
Δf(s) = dx * Δ * [(1-h[i+s]) * 1_{i+s < n} - h[i-s] * 1_{i-s ≥ 0}] (for s ≥ 0)
Δf(s) = dx * Δ * [(1-h[i-r]) * 1_{i-r ≥ 0} - h[i+r] * 1_{i+r < n}] (for s = -r < 0)

This is O(n) per change, so we can maintain the f array incrementally.

```python
def fast_coordinate_descent(h, n_sweeps=50):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    
    f = np.correlate(h, 1.0 - h, mode='full') * dx  # current f values
    c5 = np.max(f)
    
    best_h = h.copy()
    best_c5 = c5
    best_f = f.copy()
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = best_h[i]
            # Precompute the "impact" vector: how f changes per unit change in h[i]
            impact = np.zeros(2 * n - 1)
            for s in range(2 * n - 1):
                shift = s - (n - 1)
                if shift >= 0:
                    # f[shift] includes h[i]*(1-h[i+shift]) if i+shift < n
                    # and -h[i-shift]*h[i] if i-shift >= 0 (from the h[i-shift]*(1-h[i]) term)
                    if i + shift < n:
                        impact[s] += dx * (1 - best_h[i + shift])
                    if i - shift >= 0:
                        impact[s] -= dx * best_h[i - shift]
                else:
                    r = -shift
                    # f[-r] includes h[i]*(1-h[i-r]) if i-r >= 0
                    # and -h[i+r]*h[i] if i+r < n
                    if i - r >= 0:
                        impact[s] += dx * (1 - best_h[i - r])
                    if i + r < n:
                        impact[s] -= dx * best_h[i + r]
            
            # Now, if we change h[i] by delta, f becomes f + delta * impact
            # We want to minimize max(f + delta * impact) subject to:
            # 0 ≤ h[i] + delta ≤ 1 and sum(h) = n/2
            # (For now, ignore the sum constraint and handle it separately)
            
            # The optimal delta minimizes max(f + delta * impact)
            # This is a 1-D problem. The max changes at "kinks" where:
            # f[k] + delta * impact[k] = f[k'] + delta * impact[k']
            # delta = (f[k'] - f[k]) / (impact[k] - impact[k'])
            
            # For a simpler approach, just try a few delta values
            best_delta = 0
            best_max = c5
            
            # Try delta values that would make the "most constrained" f(s) decrease
            # The most constrained are those with f[s] close to c5
            hot_indices = np.where(f >= 0.95 * c5)[0]
            
            for delta in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                new_f = f + delta * impact
                new_c5 = np.max(new_f)
                if new_c5 < best_max:
                    best_max = new_c5
                    best_delta = delta
            
            if best_delta != 0:
                best_h[i] = old_val + best_delta
                f = f + best_delta * impact
                best_h = project_onto_constraint(best_h, target_sum)
                # After projection, f has changed, so recompute
                f = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
                c5 = np.max(f)
                if c5 < best_c5:
                    best_c5 = c5
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
```

```python
def paired_swap_search(h, n_trials=5000):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2
    
    f = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = np.max(f)
    best_h = h.copy()
    best_c5 = c5
    best_f = f.copy()
    
    for trial in range(n_trials):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Impact of changing h[i] by +δ and h[j] by -δ
        # Impact of h[i] += δ:
        impact_i = np.zeros(2*n-1)
        for s in range(2*n-1):
            shift = s - (n-1)
            if shift >= 0:
                if i + shift < n: impact_i[s] += dx * (1 - best_h[i+shift])
                if i - shift >= 0: impact_i[s] -= dx * best_h[i-shift]
            else:
                r = -shift
                if i - r >= 0: impact_i[s] += dx * (1 - best_h[i-r])
                if i + r < n: impact_i[s] -= dx * best_h[i+r]
        
        # Impact of h[j] -= δ is -impact_j where impact_j is impact of h[j] += δ
        impact_j = np.zeros(2*n-1)
        for s in range(2*n-1):
            shift = s - (n-1)
            if shift >= 0:
                if j + shift < n: impact_j[s] += dx * (1 - best_h[j+shift])
                if j - shift >= 0: impact_j[s] -= dx * best_h[j-shift]
            else:
                r = -shift
                if j - r >= 0: impact_j[s] += dx * (1 - best_h[j-r])
                if j + r < n: impact_j[s] -= dx * best_h[j+r]
        
        # Total impact for h[i] += δ, h[j] -= δ:
        impact = impact_i - impact_j
        
        # Find δ that minimizes max(f + δ * impact)
        # Subject to 0 ≤ best_h[i] + δ ≤ 1 and 0 ≤ best_h[j] - δ ≤ 1
        delta_max = min(1 - best_h[i], best_h[j])
        delta_min = max(-best_h[i], -(1 - best_h[j]))
        
        # Try a few δ values
        candidates = np.linspace(delta_min, delta_max, 11)

            new_c5 = np.max(best_f + delta * impact)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_delta = delta

            best_h[i] += best_delta
            best_h[j] -= best_delta
            best_f = best_f + best_delta * impact
            c5 = np.max(best_f)
            if c5 < best_c5:
                best_c5 = c5
    
    return best_h, best_c5
```

```python
def compute_impact(i, h, n):
    dx = 2.0 / n
    impact = np.zeros(2*n - 1)
    shift = np.arange(2*n-1) - (n-1)  # from -(n-1) to n-1
    
    # For shift >= 0:
    pos_mask = shift >= 0
    s = shift[pos_mask]
    # (1-h[i+s]) * 1_{i+s < n}
    idx1 = i + s
    valid1 = idx1 < n
    term1 = np.zeros_like(s, dtype=float)
    term1[valid1] = 1 - h[idx1[valid1]]
    # h[i-s] * 1_{i-s >= 0}
    idx2 = i - s
    valid2 = idx2 >= 0
    term2 = np.zeros_like(s, dtype=float)
    term2[valid2] = h[idx2[valid2]]
    impact[pos_mask] = dx * (term1 - term2)
    
    # For shift < 0:
    neg_mask = shift < 0
    r = -shift[neg_mask]
    # (1-h[i-r]) * 1_{i-r >= 0}
    idx1 = i - r
    valid1 = idx1 >= 0
    term1 = np.zeros_like(r, dtype=float)
    term1[valid1] = 1 - h[idx1[valid1]]
    # h[i+r] * 1_{i+r < n}
    idx2 = i + r
    valid2 = idx2 < n
    term2 = np.zeros_like(r, dtype=float)
    term2[valid2] = h[idx2[valid2]]
    impact[neg_mask] = dx * (term1 - term2)
    
    return impact
```

```python
import numpy as np

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_impact(i, h):
    n = len(h)
    dx = 2.0 / n
    impact = np.zeros(2*n - 1)
    shift = np.arange(2*n-1) - (n-1)
    
    pos_mask = shift >= 0
    if np.any(pos_mask):
        s = shift[pos_mask]
        idx1 = i + s
        term1 = np.zeros(len(s))
        valid1 = idx1 < n
        term1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = i - s
        term2 = np.zeros(len(s))
        valid2 = idx2 >= 0
        term2[valid2] = h[idx2[valid2]]
        impact[pos_mask] = dx * (term1 - term2)
    
    neg_mask = shift < 0
    if np.any(neg_mask):
        r = -shift[neg_mask]
        idx1 = i - r
        term1 = np.zeros(len(r))
        valid1 = idx1 >= 0
        term1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = i + r
        term2 = np.zeros(len(r))
        valid2 = idx2 < n
        term2[valid2] = h[idx2[valid2]]
        impact[neg_mask] = dx * (term1 - term2)
    
    return impact

    n = len(h)
    f = compute_f(h)
    c5 = np.max(f)
    best_h = h.copy()
    best_c5 = c5
    best_f = f.copy()

    shift = np.arange(n_corr) - (n - 1)
    
    for trial in range(n_trials):
        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Compute impact for i and j
        imp_i = np.zeros(n_corr)
        imp_j = np.zeros(n_corr)
        
        # Vectorized impact computation
        s_pos = shift[shift >= 0]
        s_neg = -shift[shift < 0]
        
        # For index i, positive shifts:
        pos_mask = shift >= 0
        if np.any(pos_mask):
            idx1 = i + s_pos
            t1 = np.zeros(len(s_pos))
            v1 = idx1 < n
            t1[v1] = 1.0 - best_h[idx1[v1]]
            idx2 = i - s_pos
            t2 = np.zeros(len(s_pos))
            v2 = idx2 >= 0
            t2[v2] = best_h[idx2[v2]]
            imp_i[pos_mask] = (2.0/n) * (t1 - t2)
            
            idx1 = j + s_pos
            t1j = np.zeros(len(s_pos))
            v1j = idx1 < n
            t1j[v1j] = 1.0 - best_h[idx1[v1j]]
            idx2 = j - s_pos

def binary_init(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2

        h[to_flip] = 0

        zero_indices = np.where(h == 0)[0]
        to_flip = rng.choice(zero_indices, size=target - n_ones, replace=False)
        h[to_flip] = 1
    return h
```

```python
def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
```
