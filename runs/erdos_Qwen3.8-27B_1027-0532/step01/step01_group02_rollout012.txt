    """Compute the subgradient of C5 at h in the direction of shift k_star."""
    n = len(h)
    L = k_star - (n - 1)
    dx = 2.0 / n

        # -h[j-L'] for j in [L', n)

import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Use water-filling projection
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = np.clip(h, 0.0, 1.0)
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def project_feasible(x, n):
    """Project x onto {h: 0<=h_i<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # Find lambda such that sum(clip(x_i - lambda, 0, 1)) = target
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(x - lam, 0.0, 1.0)

def subgradient_c5(h, top_k=None):
    """Compute (sub)gradient of C5 for h. If top_k given, use smooth max of top_k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):
        # Use log-sum-exp of top_k values
        # First find the top_k threshold
        sorted_corr = np.sort(corr)[::-1]
        threshold = sorted_corr[top_k - 1]
        # Weights: for values above threshold, weight proportional to value - threshold
        # Actually, for the "trimmed" objective (avg of top k), the subgradient is:
        # average of gradients of the top k
        # For simplicity, use log-sum-exp with a high beta on all values
        # but that's expensive. Let's just use the top_k average subgradient.
        top_indices = np.argsort(corr)[::-1][:top_k]
    else:
        top_indices = np.array([np.argmax(corr)])
    
    grad = np.zeros(n)
    for k in top_indices:
        L = k - (n - 1)
        if L >= 0:

            grad[:n-L] -= h[L:]
        else:
            Lp = -L
            grad[:n-Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n-Lp]
    
    grad /= len(top_indices)
    return dx * grad

def subgradient_descent(h, n, maxiter=5000, top_k=3, initial_step=0.01, decay=0.995):

    best_h = h.copy()
    best_c5 = evaluate_c5(h_curr)
    step = initial_step
    
    for it in range(maxiter):
        g = subgradient_c5(h_curr, top_k=top_k)
        h_new = project_feasible(h_curr - step * g, n)

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        h = np.clip(h, 0.0, 1.0)
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h


def compute_subgrad_from_top(h, top_indices, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in top_indices:
        L = k - (n - 1)
        if L >= 0:
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n - Lp]
    grad /= len(top_indices)
    return dx * grad


def lazy_subgradient_descent(h, n, maxiter, top_k, initial_step, refresh_interval, deadline):
    h_curr = h.astype(float).copy()
    best_h = h.astype(float).copy()
    best_c5 = evaluate_c5(h_curr)
    step = initial_step
    top_indices = None
    stagnant = 0

    for it in range(maxiter):
        if time.time() > deadline:
            break
        if it % refresh_interval == 0 or top_indices is None:
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
            top_indices = np.argsort(corr)[::-1][:top_k]
        g = compute_subgrad_from_top(h_curr, top_indices, n)
        h_curr = project_feasible(h_curr - step * g, n)
        if (it + 1) % refresh_interval == 0:
            c5 = evaluate_c5(h_curr)
            if c5 < best_c5 - 1e-12:
                best_h = h_curr.copy()
                best_c5 = c5
                stagnant = 0
                step = min(step * 1.01, 0.4)
            else:
                stagnant += 1
                if stagnant % 30 == 0:
                    step *= 0.92
            if stagnant > 500:
                break
    return best_h, best_c5


def slsqp_refine(h, n, maxiter):
    try:
        def obj(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full")
            return float(np.max(corr) * dx)
        result = minimize(
            obj, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        h_opt = project_feasible(result.x, n)
        c5 = evaluate_c5(h_opt)
        if c5 < evaluate_c5(h) + 1e-8:
            return h_opt, c5
    except Exception:
        pass
    return h, evaluate_c5(h)


def sa_with_local(h, n, rng, maxiter, t_init, t_final, deadline):
    h_curr = h.astype(float).copy()
    best_h = h.astype(float).copy()
    best_c5 = evaluate_c5(h_curr)
    T = t_init
    cool = (t_final / t_init) ** (1.0 / max(1, maxiter))
    stag = 0

    for it in range(maxiter):
        if time.time() > deadline:
            break
        T *= cool
        n_p = rng.randint(1, 4)
        idx = rng.choice(n, size=n_p, replace=False)
        h_new = h_curr.copy()
        h_new[idx] += rng.uniform(-0.15, 0.15, size=n_p)
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_curr = h_new
            stag = 0
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            h_curr = h_new
            stag += 1
        else:
            stag += 1
        if (it + 1) % 300 == 0:

                                                      initial_step=0.04, refresh_interval=3,
                                                      deadline=time.time() + 5.0)
            if c5_sub < best_c5:

                best_c5 = c5_sub
                h_curr = h_sub.copy()
            stag = 0
        if stag > max(maxiter // 3, 400):
            break
    return best_h, best_c5
