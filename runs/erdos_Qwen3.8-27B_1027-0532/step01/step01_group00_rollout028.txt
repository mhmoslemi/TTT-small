def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h_proj = np.clip(h - t, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        interior = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h_proj[interior] += diff / n_int
            np.clip(h_proj, 0.0, 1.0, out=h_proj)
    return h_proj


def smooth_max(x, alpha):
    m = np.max(x)
    with np.errstate(over='ignore'):
        return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    val = smooth_max(corr, alpha)
    s = np.sum(h)
    val += penalty * (s - target) ** 2
    return val
