    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    # Project h onto {sum(h) = target_sum, 0 <= h[i] <= 1}
    # using bisection on the Lagrange multiplier
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_and_grad(h, tau):
    """Compute soft-max approximation of max(corr) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    logsumexp = max_s + np.log(sum_exp)
    obj = logsumexp / tau * dx
    
    weights = exp_s / sum_exp
    
    # Gradient of corr[m] with respect to h[j]
    # corr[m] = sum_i h[i] * (1 - h[i + (m-n+1)])

        if w < 1e-15:

        lag = m - (n - 1)

        if 0 <= j + lag < n:  # this is per j, so we need to do it differently
            pass

    #   d(corr[m])/dh[j] = (1-h[j+L]) [if 0<=j+L<n and j is in valid range for "i"]
    #                      - h[j-L] [if 0<=j-L<n and j is in valid range for "i+L"]

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, tau):
    """Soft-max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    sum_exp = np.sum(exp_tc)
    obj = (tc_max + np.log(sum_exp)) / tau * dx
    w = exp_tc / sum_exp
    
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] < 1e-15:
            continue
        L = m - (n - 1)
        # First term: j as "i" (h[j] * (1-h[j+L]))
        j_lo = max(0, -L)
        j_hi = min(n-1, n-1-L)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w[m] * (1.0 - h[j_lo+L:j_hi+L+1])
        # Second term: j as "i+L" (h[j-L] * (1-h[j]))
        j2_lo = max(0, L)
        j2_hi = min(n-1, n-1+L)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w[m] * (-h[j2_lo-L:j2_hi-L+1])
    
    return obj, grad * dx

def continuation_pgd(h0, n, target_sum, deadline):
    """Continuation method: start with small tau, increase over time."""
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    tau = 5.0
    lr = 0.05
    it = 0
    no_improve = 0
    
    while time.time() < deadline:
        _, g = soft_max_obj_grad(h, tau)
        h_new = h - lr * g
        h_new = project_onto_box_sum(h_new, target_sum)
        c5 = compute_c5(h_new)
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        it += 1
        
        # Adapt tau and learning rate
        if no_improve > 30:
            tau *= 1.5
            lr = max(lr * 0.7, 0.005)
            no_improve = 0
            if lr < 0.005:
                # Try to jump: add noise and restart
                h = best_h + 0.01 * np.random.randn(n)
                h = project_onto_box_sum(h, target_sum)
                lr = 0.03
                tau = max(tau * 0.5, 5.0)
                no_improve = 0
        
        # Slowly increase tau for better approximation
        if it % 100 == 0:
            tau = min(tau * 1.2, 200.0)
    
    return best_h, best_c5

def slsqp_from(h0, n, target_sum, deadline, tau=100.0):
    """SLSQP optimization with soft-max objective."""
    def obj(x):
        o, _ = soft_max_obj_grad(x, tau)
        return o
    
    def grad(x):
        _, g = soft_max_obj_grad(x, tau)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_box_sum(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    except:
        return h0, compute_c5(h0)

def binary_local_search(h_bin, n, max_rounds=50):
    """Local search on binary solution: try swaps."""
    h = h_bin.copy().astype(float)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for round in range(max_rounds):
        improved = False
        np.random.shuffle(ones)
        for i in ones:
            np.random.shuffle(zeros)
            for j in zeros[:20]:  # try at most 20 swaps per 1-position
                # Swap: h[i] 1->0, h[j] 0->1
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    # Update ones and zeros
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5
