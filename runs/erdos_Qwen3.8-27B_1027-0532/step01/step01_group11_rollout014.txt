def smooth_max_weight(corr, eps):
    m = np.max(corr)
    x = (corr - m) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    return ex / np.sum(ex)

def compute_grad_smooth_max(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    w = smooth_max_weight(corr, eps)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            # corr[s] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[j] = (1-h[j+s]) for j < n-s
            #           -h[j-s] for j >= s (and s>0)
            #           additional -h[j] for s=0
            for j in range(n - s):
                grad[j] += wk * (1.0 - h[j + s])
            if s == 0:
                for j in range(n):
                    grad[j] -= wk * h[j]
            else:
                for j in range(s, n):
                    grad[j] -= wk * h[j - s]
        else:
            r = -s  # r > 0
            # corr[-r] = sum_{j=0}^{n-1-r} h[j+r]*(1-h[j])
            # d/dh[t]:
            #   If t = j+r for j in [0,n-1-r], i.e., t in [r, n-1]: (1-h[t-r])
            #   If t = j for j in [0,n-1-r], i.e., t in [0, n-1-r]: -h[t+r]
            for j in range(n - r):
                grad[j + r] += wk * (1.0 - h[j])
                grad[j] -= wk * h[j + r]
    return grad

    grad = compute_grad_smooth_max(h, eps)
    h_new = h - lr * grad
    return project_onto_box_sum(h_new, target_sum)

At the optimum, suppose lags k1, k2, ..., km are active (all achieving the max C). The KKT conditions require that the gradient of the objective is a nonnegative combination of the gradients of the active constraints. In practice, this means:
- All active lags should have the same value (= C)
- The gradient should be "balanced" across the active lags

1. Identify the top-K lags (K=5 or so)
2. Compute their values
3. If they're not all equal, find a direction in h-space that equalizes them
4. Take a step in that direction
5. Repeat

This is essentially a "water-filling" approach applied to the optimization.

- Find the h-direction that most decreases the maximum while increasing the minimum of the top-K

def project_onto_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x = x - np.max(x)
    return m + eps * np.log(np.sum(np.exp(x)))

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad += dx * wk * (1.0 - 2.0 * h)
            else:
                grad[:n - s] += dx * wk * (1.0 - h[s:])
                grad[s:] -= dx * wk * h[:n - s]
        else:
            r = -s
            grad[r:] += dx * wk * (1.0 - h[:n - r])
            grad[:n - r] -= dx * wk * h[r:]

        # Quick optimization
        h_cur = h_p.copy()
        for eps in [0.01, 0.005, 0.001]:
            if tl() < 120:
                break
            h_cur, c5_cur = projected_gradient(h_cur, eps, n0, target, 80, 0.003, tl, 120)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

        if tl() > 120:
            h_opt = constrained_minimize(h_cur, 0.0005, n0, target, 200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 5: Upscale to 2n
    if tl() > 250:
        n1 = 2 * n0
        t1 = n1 / 2.0
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x1 = np.linspace(0, 2, n1, endpoint=False)
        h_ext = np.concatenate([best_h, best_h[:1]])
        h_up = np.interp(x1, x0, h_ext)
        h_up = project_onto_box_sum(h_up, t1)

        for eps in [0.02, 0.01, 0.005, 0.002, 0.001]:
            if tl() < 150:
                break
            h_up, c5_up = projected_gradient(h_up, eps, n1, t1, 100, 0.003, tl, 150)

        if tl() > 150:
            h_opt = constrained_minimize(h_up, 0.0005, n1, t1, 400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                if tl() > 100:
                    h_opt2 = constrained_minimize(h_up, 0.0001, n1, t1, 400)
                    c5_opt2 = compute_c5(h_opt2)
                    if c5_opt2 < best_c5:
                        best_c5 = c5_opt2
                        best_h = h_opt2

    # Phase 6: Final polish on best solution
    if tl() > 60:
        nb = len(best_h)
        tb = nb / 2.0
        h_fin = constrained_minimize(best_h, 0.0001, nb, tb, 800)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final projection to ensure exact feasibility
    final_h = project_onto_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
