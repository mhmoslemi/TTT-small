import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_grad_direct(h, p, dx, n, thresh=1e-8):
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < thresh:
            continue
        k = m - (n - 1)
        mk = abs(k)
        if mk == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n - mk] += w * (1.0 - h[mk:])
            grad[mk:] -= w * h[:n - mk]
    return dx * grad

def nesterov_smoothed(h0, tm, beta_init=2.0, beta_end=60.0, step_scale=0.4, max_iter=80000):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    h_prev = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    L = 2.0 * dx / step_scale
    step = 1.0 / L if L > 1e-10 else 1.0
    stall_count = 0
    stall_limit = 3000
    last_best = best_c5
    for it in range(max_iter):
        if time.time() - t0 > tm:
            break
        frac = min(1.0, float(it) / max(1.0, max_iter * 0.6))
        beta = beta_init + (beta_end - beta_init) * frac
        t_val = float(it + 1)
        momentum = (t_val - 1.0) / (t_val + 2.0)
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        mx = corr.max()
        shifted = beta * (corr - mx)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        grad = compute_grad_direct(y, p, dx, n, thresh=1e-9)
        h_prev = h.copy()
        h_new = project_feasible(y - step * grad, tgt)
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > stall_limit:
            stall_count = 0
            break
        if it > 500:
            step *= 0.9999
    return best

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0.copy(), tgt)
    t0 = time.time()
    def f(x):
        if time.time() - t0 > tm * 0.9:
            raise Exception("timeout")
        return float(np.max(np.correlate(x, 1.0 - x, mode='full') * dx))
    def gfun(x):
        return x.sum() - tgt
    def gjac(x):
        return np.ones(n)
    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 8000, 'ftol': 1e-15})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), float(n_dst) / 2.0)

def perturb(h0, rng, scale=0.03):
    n = len(h0)
    h = h0 + rng.normal(0, scale, size=n)
    return project_feasible(h, float(n) / 2.0)
