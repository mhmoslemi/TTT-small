import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = 0.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binary_sa(h, n, max_iters=200000, t_start=0.5, t_end=0.01, deadline=None):
    """Simulated annealing on binary sequence."""
    h = h.copy()
    # Round to binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Ensure exact count
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = np.random.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = np.random.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iters)
        
        # Random move: flip one bit or swap two bits
        if np.random.random() < 0.7:
            # Single flip
            j = np.random.randint(n)
            h[j] = 1.0 - h[j]
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[j] = 1.0 - h[j]
        else:
            # Pair swap
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                j1 = ones[np.random.randint(len(ones))]
                j2 = zeros[np.random.randint(len(zeros))]
                h[j1], h[j2] = h[j2], h[j1]
                new_c5 = compute_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                else:
                    h[j1], h[j2] = h[j2], h[j1]
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = 0.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def round_to_binary(h, n):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = np.random.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = np.random.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def greedy_refine(h, n, max_rounds=200, deadline=None):

    best_h = h.copy()
    best_c5 = cur_c5

            break
        improved = False

def adam_optimize(h0, n, target_sum, deadline, max_iter=5000, tau=40.0):
    h = h0.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        tc = tau * corr
        max_tc = np.max(tc)
        log_sum_exp = max_tc + np.log(np.sum(np.exp(tc - max_tc)))
        obj = (log_sum_exp / tau) * dx
        weights = np.exp(tc - max_tc)
        weights /= np.sum(weights)
        g = np.zeros(n)
        for m_idx in range(2 * n - 1):
            if weights[m_idx] < 1e-14:
                continue
            lag = m_idx - (n - 1)
            w = weights[m_idx]
            j_lo = max(0, lag)
            j_hi = min(n - 1, n - 1 + lag)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                g[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_onto_constraint(h, target_sum)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_constraint(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if it % 500 == 0:
            tau *= 1.5
            lr *= 0.85
    return best_h, best_c5

def soft_max_grad(h, tau, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    max_tc = np.max(tc)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tc - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tc - max_tc)
    weights /= np.sum(weights)
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if weights[m_idx] < 1e-14:
            continue
        lag = m_idx - (n - 1)
        w = weights[m_idx]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, g * dx

def slsqp_optimize(h0, n, target_sum, deadline, tau=60.0):
    def obj_fun(x):
        o, _ = soft_max_grad(x, tau, n)
        return o
    def grad_fun(x):
        _, g = soft_max_grad(x, tau, n)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def alternating_init(n):

def block_init(n, period=4, ones_in_period=2):

            if c5_g < best_c5:

                best_h = h_g.copy()

    # Final validation
    if best_h is None:
        best_h = np.full(120, 0.5)
        best_c5 = compute_c5(best_h)

    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
