import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_all_c5_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def active_set_objective(h, active_indices, eps):
    vals = compute_all_c5_values(h)
    active_vals = vals[active_indices]
    cmax = np.max(active_vals)
    x = (active_vals - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def compute_gradient_for_lag(h, k_idx, n, dx):
    """Compute gradient of f_k with respect to h."""
    omh = 1.0 - h
    grad = np.zeros(n)
    s = k_idx - (n - 1)
    if s >= 0:
        if s < n:
            grad[s:] += dx * omh[:n - s]
            grad[:n - s] -= dx * h[s:]
    else:
        r = -s
        if r < n:
            grad[:n - r] += dx * omh[r:]
            grad[r:] -= dx * h[:n - r]
    return grad

def active_set_gradient(h, active_indices, eps):
    n = len(h)
    dx = 2.0 / n
    vals = compute_all_c5_values(h)
    active_vals = vals[active_indices]
    cmax = np.max(active_vals)
    x = (active_vals - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    
    grad = np.zeros(n)
    for j, k_idx in enumerate(active_indices):
        if w[j] < 1e-15:
            continue
        grad += w[j] * compute_gradient_for_lag(h, k_idx, n, dx)
    return grad

def optimize_active_set(h_init, top_k_list, eps_list, time_limit, maxiter_per_eps=200):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.copy(), target_sum)
    start = time.time()
    
    for top_k in top_k_list:
        remaining_eps = [e for e in eps_list if time.time() - start < time_limit]
        for eps in remaining_eps:
            if time.time() - start > time_limit:
                break
            # Find current top-k active indices
            vals = compute_all_c5_values(h)
            active = np.argsort(vals)[-top_k:]
            
            # Optimize using L-BFGS-B
            bounds = [(0.0, 1.0)] * n
            # Use penalty for sum constraint
            def obj(x):
                x_proj = project_onto_constraint(x, target_sum)
                return active_set_objective(x_proj, active, eps)

                return active_set_gradient(x_proj, active, eps)

            h = project_onto_constraint(result.x, target_sum)

from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def gradient_for_lag_index(h, k_idx, n):
    k = k_idx - (n - 1)
    omh = 1.0 - h
    grad = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[:nk] += omh[k:]
            grad[k:] -= h[:nk]
    else:
        r = -k
        nr = n - r
        if nr > 0:
            grad[r:] += omh[:nr]
            grad[:nr] -= h[r:]
    return grad

def obj_pen(x, top_k_idx, eps, penalty, n, target_sum):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    active_vals = corr[top_k_idx]
    cmax = np.max(active_vals)
    xv = (active_vals - cmax) / eps
    xv -= np.max(xv)
    ex = np.exp(xv)
    base = cmax + eps * np.log(np.sum(ex))
    sv = np.sum(x) - target_sum
    return base + penalty * sv * sv

def grad_pen(x, top_k_idx, eps, penalty, n, target_sum):

        grad += w[j] * gradient_for_lag_index(x, top_k_idx[j], n)
    sv = np.sum(x) - target_sum
    grad += 2.0 * penalty * sv
    return grad

def optimize_lbfgsb(h_init, top_k_idx, eps, penalty, maxiter, bounds):
    n = len(h_init)
    target_sum = n / 2.0
    args = (top_k_idx, eps, penalty, n, target_sum)

                          bounds=bounds,
                          options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-14})
        return result.x
    except Exception:
        return h_init
