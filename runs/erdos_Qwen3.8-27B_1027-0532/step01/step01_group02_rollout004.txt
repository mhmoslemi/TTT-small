```python
def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * beta * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()  # length 2n-1
    
    grad = np.zeros(n)
    # For each shift m (0 to 2n-2), s = m - (n-1)
    # grad[j] += dx * w[m] * [(1-h[j+s]) if valid - h[j-s] if valid]
    
    # Vectorized: for shift s, the valid j range is [max(0,-s), min(n-1, n-1-s)]
    # and the contribution to grad[j] is w[s+(n-1)] * (1-h[j+s]) - w[s+(n-1)] * h[j-s]
    
    for m in range(len(w)):
        s = m - (n-1)
        wm = w[m]
        # Term 1: (1-h[j+s]) for j in [max(0,-s), min(n-1, n-1-s)]
        j1_lo = max(0, -s)
        j1_hi = min(n-1, n-1-s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += wm * (1 - h[j1_lo+s:j1_hi+s+1])
        # Term 2: -h[j-s] for j in [max(0,s), min(n-1, n-1+s)]
        j2_lo = max(0, s)
        j2_hi = min(n-1, n-1+s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] -= wm * h[j2_lo-s:j2_hi-s+1]
    
    return dx * grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_bounded_simplex(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    return project_bounded_simplex(h, n / 2.0)


def slsqp_optimize(h, n, maxiter=300, ftol=1e-12):
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if r.success or (np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9)):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = project_bounded_simplex(r.x, n / 2.0)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)


def compute_grad_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * beta * dx
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()

    grad = np.zeros(n)
    for mi in range(len(w)):
        s = mi - (n - 1)
        wm = w[mi]
        if wm < 1e-15:
            continue
        j1_lo = max(0, -s)
        j1_hi = min(n - 1, n - 1 - s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += wm * (1.0 - h[j1_lo + s:j1_hi + s + 1])
        j2_lo = max(0, s)
        j2_hi = min(n - 1, n - 1 + s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] -= wm * h[j2_lo - s:j2_hi - s + 1]
    return dx * grad


def smooth_grad_descent(h, n, beta_start=50, beta_end=2000, n_iter=3000,
                        base_step=0.01, t0=None, budget_s=0):
    h_curr = h.copy()
    best_local = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    for it in range(n_iter):
        if t0 is not None and (time.time() - t0) > budget_s - 30:
            break
        frac = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        g = compute_grad_smoothed(h_curr, beta)
        step = base_step * (1.0 - 0.5 * frac)
        h_new = h_curr - step * g
        h_new = project_bounded_simplex(h_new, n / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_local = h_new.copy()
        h_curr = h_new
    return best_local, best_c5


def slsqp_with_active_set(h, n, maxiter=500, ftol=1e-13):
    h_curr = h.copy()
    c5_curr = evaluate_c5(h_curr)
    for _ in range(3):
        dx = 2.0 / n
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full") * dx
        max_val = np.max(corr)
        active_mask = corr > max_val - 0.001
        h_opt, c5_opt = slsqp_optimize(h_curr, n, maxiter=maxiter, ftol=ftol)
        if c5_opt < c5_curr - 1e-10:
            h_curr = h_opt
            c5_curr = c5_opt
        else:
            break
    return h_curr, c5_curr
```
