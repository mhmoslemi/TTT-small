import time

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, dx, beta, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    
    m_max = np.max(c)
    exps = np.exp(beta * (c - m_max))
    S = m_max + np.log(np.sum(exps)) / beta
    w = exps / np.sum(exps)
    
    # Gradient of smooth max
    w_rev = w[::-1]
    conv1 = np.convolve(1.0 - h, w, mode='full')
    conv2 = np.convolve(h, w_rev, mode='full')
    s1 = conv1[n-1 : 2*n-1]
    s2 = conv2[n-1 : 2*n-1]
    grad_sm = dx * (s1 - s2)
    
    # Penalty gradient
    sum_h = np.sum(h)
    penalty_term = penalty * (sum_h - n*0.5) ** 2
    grad_pen = 2.0 * penalty * (sum_h - n*0.5) * np.ones(n)
    
    total = S + penalty_term
    total_grad = grad_sm + grad_pen
    
    return total, total_grad

def optimize_lbfgsb(h0, n, dx, beta, penalty, maxiter):
    h0 = repair_h(h0, n)
    res = minimize(
        lambda x: smooth_max_obj_grad(x, dx, beta, penalty)[0],
        lambda x: smooth_max_obj_grad(x, dx, beta, penalty)[1],
        h0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50}
    )
    return repair_h(res.x, n)
