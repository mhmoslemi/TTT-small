def project_feasible(h, target):
    """Project h to [0,1]^n with sum = target using bisection on shift."""
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    """After initial projection, do a few gradient steps to fine-tune the sum."""
    h = project_feasible(h, target)

def binary_local_search(h, n, max_iters, deadline):
    """Local search for binary h with exactly n/2 ones."""
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    c = np.correlate(h, 1-h, mode='full')
    current_max = np.max(c)
    best_max = current_max
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        improved = False
        # Try all possible swaps: move a 1 from position i to position j (where h[j]=0)
        for i_idx in range(len(ones)):
            if time.time() > deadline:
                break
            for j_idx in range(len(zeros)):
                i, j = ones[i_idx], zeros[j_idx]
                if i == j:
                    continue
                
                # Compute change in correlation
                # h[i] goes from 1 to 0, h[j] goes from 0 to 1
                # This affects correlation values at specific indices
                
                # Old: h[i]=1, contributes to c at positions where h[i] interacts with (1-h[j'])
                # New: h[j]=1, h[i]=0
                
                # For the full correlation c = correlate(h, 1-h):
                # c[m] = sum_k h[k] * (1-h[k+m-(n-1)])
                # Changing h[i]: if h[i] was 1, it contributed (1-h[i+m-(n-1)]) to c[m] for all m
                # Changing h[j]: (1-h[j]) was the "other" factor, so changing h[j] affects c[m] where j = k+m-(n-1), i.e., m = j-k+n-1 for various k
                
                # This is getting complex. Let me just recompute.
                # For n=44, correlate is fast.
                
                h[i] = 0
                h[j] = 1
                c_new = np.correlate(h, 1-h, mode='full')
                new_max = np.max(c_new)
                
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    # Update ones and zeros
                    ones[i_idx] = j  # the one that was at i is now at j
                    zeros[j_idx] = i  # the zero that was at j is now at i
                    current_max = new_max
                    improved = True
                else:
                    h[i] = 1
                    h[j] = 0
        
        if not improved:
            break
    
    return best_h, float(best_max) * 2.0 / n
