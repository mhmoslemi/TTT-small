import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_corr(h):
    """Compute all correlation values."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_smooth_val_and_weights(c, alpha):
    """Compute smooth max value and softmax weights."""
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    val = c_max + np.log(exp_c.sum()) / alpha
    return val, w

def compute_gradient(h, w):
    """Compute gradient of weighted sum of correlations.
    w has length 2n-1, corresponding to the full correlation output.
    Returns gradient of sum_m w[m] * corr[m] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    # corr[m] = sum_j h[j] * (1-h[j + m - (n-1)]) for valid j
    # dcorr[m]/dh[i] = (1-h[i + m - (n-1)]) * 1{valid} - h[i - m + (n-1)] * 1{valid}
    
    # Using matrix approach:
    # grad[i] = dx * (sum_j w[j-i+n-1]*(1-h[j]) - sum_j w[i-j+n-1]*h[j])
    #         = dx * (PartA[i] - PartB[i])
    # where PartA[i] = sum_j w[(n-1)+(j-i)]*(1-h[j])
    #       PartB[i] = sum_j w[(n-1)+(i-j)]*h[j]
    
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    
    # M2[i,j] = w[j-i+(n-1)] → index = cols - rows + (n-1)
    idx1 = cols - rows + (n - 1)
    idx1 = np.clip(idx1, 0, 2 * n - 2)
    M2 = w[idx1]
    # Zero out invalid entries
    M2[(cols - rows + (n-1) < 0) | (cols - rows + (n-1) > 2*n-2)] = 0
    
    # M1[i,j] = w[i-j+(n-1)] → index = rows - cols + (n-1)
    idx2 = rows - cols + (n - 1)
    idx2 = np.clip(idx2, 0, 2 * n - 2)
    M1 = w[idx2]
    M1[(rows - cols + (n-1) < 0) | (rows - cols + (n-1) > 2*n-2)] = 0
    
    PartA = M2.dot(1.0 - h)
    PartB = M1.dot(h)
    grad = dx * (PartA - PartB)
    
    return grad

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on offset."""
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def projected_gradient_descent(h, n, alpha_start, alpha_end, max_iter, deadline, lr_initial=0.02, momentum=0.9):
    """Projected gradient descent with backtracking and momentum."""
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    vel = np.zeros(n)
    alpha = alpha_start
    lr = lr_initial
    n_total = max_iter + 100  # total iterations including alpha scheduling
    it = 0
    
    while time.time() < deadline:
        if it >= max_iter:

        alpha = min(alpha_end, alpha * (1.001 + 0.5 * it / max_iter))

        grad = compute_gradient(h, w)
        
        # Momentum

        h_new = h + vel
        h_new = project_feasible(h_new, target)

        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        steps = 0

            lr *= 0.5
            vel = momentum * vel - lr * grad

            steps += 1

def upscale_nearest(h, factor):
    """Upscale by nearest neighbor, preserving integral constraint."""
    n = len(h)
    h_new = np.zeros(n * factor)
    for i in range(n):
        h_new[i*factor:(i+1)*factor] = h[i]
    return h_new

def upscale_linear(h, factor=2):
    """Upscale by linear interpolation (circular), preserving integral."""
    n = len(h)
    h_new = np.zeros(n * factor)
    for i in range(n):
        h_next = h[(i + 1) % n]
        for j in range(factor):
            t = j / (factor - 1) if factor > 1 else 0.0
            h_new[i*factor + j] = (1-t) * h[i] + t * h_next
    # Fix sum
    current_sum = h_new.sum()
    target_sum = n * factor / 2.0
    # Adjust by distributing the difference
    diff = target_sum - current_sum
    h_new += diff / len(h_new)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, target_sum)
