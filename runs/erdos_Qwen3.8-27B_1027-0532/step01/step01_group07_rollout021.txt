def solve_coord_fast(h, i, n):
    save = h[i]
    dx = 2.0 / n
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = save
    d = c1 - c0
    
    # f(t) = max(c0 + d*t), find min over t in [0,1]
    # Breakpoints: t = (c0[a]-c0[b])/(d[b]-d[a]) for all pairs where d[a]!=d[b]
    
    # Vectorized: compute all pairwise breakpoints
    D = d[:, None] - d[None, :]  # D[a,b] = d[a] - d[b]
    C = c0[None, :] - c0[:, None]  # C[a,b] = c0[b] - c0[a]
    
    valid = np.abs(D) > 1e-15
    t_vals = np.where(valid, C / np.where(valid, D, 1.0), -1.0)
    
    # Filter to [0,1]
    mask = (t_vals >= 0.0) & (t_vals <= 1.0)
    candidates = np.unique(t_vals[mask])
    
    # Add boundaries
    candidates = np.concatenate([candidates, [0.0, 1.0]])
    
    # Evaluate f at each candidate
    f_vals = np.max(c0[None, :] + d[None, :] * candidates[:, None], axis=1)
    
    best_idx = np.argmin(f_vals)
    return candidates[best_idx]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def solve_coord(h, i, n):
    save = h[i]
    dx = 2.0 / n
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = save
    d = c1 - c0
    m = len(c0)
    
    # Vectorized breakpoint search
    D = d[:, None] - d[None, :]
    C = c0[None, :] - c0[:, None]
    valid = np.abs(D) > 1e-15
    D_safe = np.where(valid, D, 1.0)
    t_vals = C / D_safe
    mask = valid & (t_vals >= 0.0) & (t_vals <= 1.0)
    candidates = np.unique(t_vals[mask])
    candidates = np.concatenate([candidates, [0.0, 1.0]])
    
    f_vals = np.max(c0[None, :] + d[None, :] * candidates[:, None], axis=1)
    best_idx = np.argmin(f_vals)
    return candidates[best_idx]

def coord_descent_opt(h_init, n, max_passes=500, deadline=None, rng=None):
    target = n / 2.0
    h = h_init.copy()
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)

            h[i] = new_val

            best_h = h.copy()
            no_improve = 0

def smooth_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_sum = np.exp(shifted).sum()
    val = c_max + np.log(exp_sum) / alpha
    w = np.exp(shifted) / exp_sum
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            lag = m - (n - 1)
            if 0 <= i + lag < n:
                g += w[m] * (1.0 - h[i + lag]) * dx
            if 0 <= i - lag < n:
                g += w[m] * (-h[i - lag]) * dx
        grad[i] = g
    return val, grad

def gradient_polish(h_init, n, alpha, max_iter, deadline, lr=0.01):

    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def solve_coord(h, i, n):
    save = h[i]
    dx = 2.0 / n
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = save
    d = c1 - c0
    m = len(c0)
    lag0_idx = n - 1

    S = c0[lag0_idx]
    d_quad = c1[lag0_idx] - c0[lag0_idx]

    c0_lin = np.concatenate([c0[:lag0_idx], c0[lag0_idx+1:]])
    d_lin = np.concatenate([d[:lag0_idx], d[lag0_idx+1:]])
    m_lin = len(c0_lin)

    D = d_lin[:, None] - d_lin[None, :]
    C = c0_lin[None, :] - c0_lin[:, None]
    valid = np.abs(D) > 1e-15
    D_safe = np.where(valid, D, 1.0)
    t_vals = C / D_safe
    mask = valid & (t_vals >= 0.0) & (t_vals <= 1.0)
    candidates = np.unique(t_vals[mask])

    A_coeffs = d_lin - 1.0
    B_coeffs = c0_lin - S
    disc = A_coeffs**2 - 4.0 * B_coeffs
    quad_mask = disc >= 0
    if np.any(quad_mask):
        sq = np.sqrt(disc[quad_mask])
        t1 = (-A_coeffs[quad_mask] + sq) / 2.0
        t2 = (-A_coeffs[quad_mask] - sq) / 2.0
        qt = np.concatenate([t1, t2])
        qt = qt[(qt >= 0.0) & (qt <= 1.0)]
        candidates = np.append(candidates, qt)

    candidates = np.concatenate([candidates, [0.0, 0.5, 1.0]])
    candidates = np.unique(candidates)
    candidates = candidates[(candidates >= 0.0) & (candidates <= 1.0)]

    if len(candidates) == 0:
        return save

    linear_vals = c0_lin[None, :] + d_lin[None, :] * candidates[:, None]
    linear_max = linear_vals.max(axis=1) if m_lin > 0 else np.zeros(len(candidates))
    quad_vals = S + d_quad * candidates - d_quad * candidates**2
    f_vals = np.maximum(linear_max, quad_vals)

    best_idx = np.argmin(f_vals)
    return candidates[best_idx]

def coord_descent_opt(h_init, n, max_passes=500, deadline=None, rng=None):
    target = n / 2.0
    h = h_init.copy()
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for p in range(max_passes):
        if deadline is not None and time.time() > deadline - 3:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            h[i] = solve_coord(h, i, n)
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
    return best_h, best_c5

def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_sum = np.exp(shifted).sum()
    val = c_max + np.log(exp_sum) / alpha
    w = np.exp(shifted) / exp_sum

    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            lag = m - (n - 1)
            if 0 <= i - lag < n:
                g += w[m] * (1.0 - h[i - lag]) * dx
            if 0 <= i + lag < n:
                g += w[m] * (-h[i + lag]) * dx
        grad[i] = g
    return val, grad

def gradient_polish(h_init, n, alpha, max_iter, deadline, lr=0.02):
    h = h_init.copy()
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_lr = lr
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 2:
            break
        val, grad = smooth_c5_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_new = h - cur_lr * grad / max(gnorm, 1e-10)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= val + 1e-10:
            h = h_new
            if it % 500 == 0 and it > 0:
                alpha = min(alpha * 1.5, 2000.0)
        else:
            cur_lr *= 0.7
            if cur_lr < 1e-10:
                break
    return best_h, best_c5

def interpolate_h(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_src, h_src, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s

    best_h = None
    best_c5 = float('inf')

    def _time_left():
        return deadline - time.time()

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        h0 = project_feasible(h0, n0 / 2.0)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)

        h0_sym = 1.0 - h0
        h0_sym = project_feasible(h0_sym, n0 / 2.0)
        c5_sym = compute_c5(h0_sym)
        if c5_sym < best_c5:
            best_c5 = c5_sym
            best_h = h0_sym.copy()

        if _time_left() > 80:
            h_cd, c5_cd = coord_descent_opt(best_h, n0, max_passes=400, deadline=deadline, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if _time_left() > 40:
            h_gp, c5_gp = gradient_polish(best_h, n0, alpha=80.0, max_iter=5000, deadline=deadline)
            if c5_gp < best_c5:
                best_c5 = c5_gp
                best_h = h_gp.copy()

        if _time_left() > 60:
            h_cd2, c5_cd2 = coord_descent_opt(best_h, n0, max_passes=400, deadline=deadline, rng=rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

        if _time_left() > 30:
            h_gp2, c5_gp2 = gradient_polish(best_h, n0, alpha=200.0, max_iter=4000, deadline=deadline)
            if c5_gp2 < best_c5:
                best_c5 = c5_gp2
                best_h = h_gp2.copy()

        perturb_scales = [0.12, 0.08, 0.18, 0.06, 0.25, 0.15]
        for ps in perturb_scales:
            if _time_left() < 35:
                break
            h_pert = best_h + rng.normal(0, ps, n0)
            h_pert = project_feasible(h_pert, n0 / 2.0)
            h_cd, c5_cd = coord_descent_opt(h_pert, n0, max_passes=300, deadline=deadline, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                if _time_left() > 20:
                    h_gp, c5_gp = gradient_polish(best_h, n0, alpha=100.0, max_iter=2000, deadline=deadline)
                    if c5_gp < best_c5:
                        best_c5 = c5_gp
                        best_h = h_gp.copy()
            elif _time_left() > 15:
                h_gp, c5_gp = gradient_polish(h_pert, n0, alpha=50.0, max_iter=2000, deadline=deadline)
                if c5_gp < best_c5:
                    best_c5 = c5_gp
                    best_h = h_gp.copy()

        if _time_left() > 30:
            h_cd3, c5_cd3 = coord_descent_opt(best_h, n0, max_passes=500, deadline=deadline, rng=rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()

        for n_try in [46, 48, 50, 52]:
            if _time_left() < 70:
                break
            h_interp = interpolate_h(best_h, n_try)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            h_cd, c5_cd = coord_descent_opt(h_interp, n_try, max_passes=350, deadline=deadline, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                if _time_left() > 25:
                    h_gp, c5_gp = gradient_polish(best_h, n_try, alpha=100.0, max_iter=3000, deadline=deadline)
                    if c5_gp < best_c5:
                        best_c5 = c5_gp
                        best_h = h_gp.copy()
            else:
                h_gp, c5_gp = gradient_polish(h_interp, n_try, alpha=60.0, max_iter=2000, deadline=deadline)
                if c5_gp < best_c5:
                    best_c5 = c5_gp
                    best_h = h_gp.copy()

    if best_h is None or _time_left() > 100:
        for n_try in [44, 48]:
            if _time_left() < 60:
                break
            inits = []
            h_bin = np.zeros(n_try)
            h_bin[::2] = 1.0
            inits.append(project_feasible(h_bin, n_try / 2.0))

            h_blk = np.zeros(n_try)
            h_blk[:n_try // 4] = 1.0
            h_blk[3 * n_try // 4:] = 1.0
            inits.append(project_feasible(h_blk, n_try / 2.0))

            t_arr = np.linspace(0, 2, n_try)
            h_smooth = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t_arr / 2.0)
            inits.append(project_feasible(np.clip(h_smooth, 0, 1), n_try / 2.0))

            h_rnd = rng.uniform(0, 1, n_try)
            inits.append(project_feasible(h_rnd, n_try / 2.0))

            for h_init in inits:
                if _time_left() < 30:
                    break
                h_cd, c5_cd = coord_descent_opt(h_init, n_try, max_passes=300, deadline=deadline, rng=rng)
                if best_h is None or c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), len(best_h)
