    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def solve_1d_max(a, b):
    """Solve min_{v in [0,1]} max_k (a[k] + b[k]*v)"""
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        g_val = np.max(vals)
        active = vals >= g_val - 1e-10
        if np.all(b[active] > 1e-14):
            hi = mid
        elif np.all(b[active] < -1e-14):
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def compute_b(h, p, n):
    """Compute b[k] such that corr[k] = a[k] + b[k]*h[p]"""
    b = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        k = lag + n - 1
        c = 0.0
        i1 = p - lag
        if 0 <= i1 < n:
            c += 1.0 - h[i1]
        i2 = p + lag
        if 0 <= i2 < n:
            c -= h[i2]
        b[k] = c
    return b

    target = n / 2.0

import numpy as np

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def solve_1d_max(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        g_val = np.max(vals)
        active = vals >= g_val - 1e-12
        if np.all(b[active] > 1e-14):
            hi = mid
        elif np.all(b[active] < -1e-14):
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def compute_b(h, p, n):
    b = np.zeros(2 * n - 1)
    for lag in range(-(n-1), n):
        k = lag + n - 1
        c = 0.0
        i1 = p - lag
        if 0 <= i1 < n:
            c += 1.0 - h[i1]
        i2 = p + lag
        if 0 <= i2 < n:
            c -= h[i2]
        b[k] = c
    return b

def coordinate_descent(h, n, time_limit=30, random_order=False):
    h = h.copy()
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    if random_order:
        order = np.random.permutation(n)
    else:
        order = np.arange(n)
    
    stuck_count = 0
    for sweep in range(100000):
        if time.time() - start > time_limit:
            break
        for p in order:
            old = h[p]
            b = compute_b(h, p, n)
            corr = np.correlate(h, 1 - h, mode="full")
            a = corr - old * b
            v = solve_1d_max(a, b)
            
            if abs(v - old) > 1e-12:
                h[p] = v
                h = project_feasible(h, n)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 50:
                break
    
    return best_h, best_c5

        h_pert = best_h + 0.01 * rng.randn(n)
    return project_feasible(h_pert, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _constraint_n, _current_eps
    np.random.seed(seed)
    _constraint_n = len(initial_h_values)
    n = _constraint_n
    start_time = time.time()

    h = np.array(initial_h_values, dtype=float)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Phase 1: Coordinate descent from initial
    h1, c5_1 = coordinate_descent(h, n, time_limit=30, random_order=False, seed=seed)
    if c5_1 < best_c5:
        best_h = h1.copy()
        best_c5 = c5_1

    # Phase 2: Multiple perturbation + coordinate descent
    rng = np.random.RandomState(seed + 1)
    phase2_deadline = start_time + min(550, budget_s - 120)
    attempt = 0
    while time.time() < phase2_deadline and attempt < 300:
        mode = attempt % 5
        h_pert = _make_perturbed(best_h, n, rng, mode)
        remaining = phase2_deadline - time.time()
        if remaining < 8:
            break
        tl = min(20, remaining - 3)
        h_cand, c5_cand = coordinate_descent(h_pert, n, time_limit=tl,
                                             random_order=(attempt % 2 == 1),
                                             seed=seed + attempt)
        if c5_cand < best_c5:
            best_h = h_cand.copy()
            best_c5 = c5_cand
        attempt += 1

    # Phase 3: SLSQP polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 80:
        tb = min(300, remaining - 60)
        h_sp, c5_sp = slsqp_polish(best_h, n, time_budget=tb)
        if c5_sp < best_c5:
            best_h = h_sp.copy()
            best_c5 = c5_sp

    # Phase 4: Final coordinate descent refinement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 15:
        h_final, c5_final = coordinate_descent(best_h, n,
                                               time_limit=max(5, remaining - 5),
                                               random_order=True, seed=seed + 9999)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final

    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
