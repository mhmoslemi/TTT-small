        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Coordinate descent for n=85
    deltas_c = [-0.005, -0.002, -0.001, -0.0005, 0.0005, 0.001, 0.002, 0.005]
    best_h, best_c5 = coord_descent(best_h, n, deltas_c, 800, time_limit(0.70))
    h_new, c5_new = slsqp_local(best_h, n, bounds_n, 10000.0, 600, 1e-15)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new

    # === Now try n=170 ===
    n2 = 170
    bounds_n2 = [(0.0, 1.0)] * n2
    h_up = project_feasible(upsample(best_h, n, n2), n2)
    best_h2 = h_up.copy()
    best_c52 = evaluate_c5(h_up)

    # Also try a structured init for n2
    h_struct = np.zeros(n2)
    for i in range(n2):
        h_struct[i] = 1.0 if ((i * 31) % 170) < 85 else 0.0
    h_struct = project_feasible(h_struct, n2)
    c5_struct = evaluate_c5(h_struct)
    if c5_struct < best_c52:
        best_c52 = c5_struct
        best_h2 = h_struct

    # Quick binary local search on n2
    h_bin = (h_up > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n2)
    h_bin, c5_bin = binary_local_search(h_bin, n2, 5000, time_limit(0.75))
    if c5_bin < best_c52:
        best_c52 = c5_bin
        best_h2 = h_bin

    # DE for n2
    if elapsed() < budget_s * 0.80:
        def de_obj2(x):
            return evaluate_c5(project_feasible(x, n2))
        try:
            de_res2 = differential_evolution(
                de_obj2, bounds_n2,
                maxiter=400, popsize=20,
                tol=1e-10, mutation=(0.3, 1.0),
                crossover=0.8, seed=seed + 100,
                polish=False, init='latinhypercube'
            )
            h_de2 = project_feasible(de_res2.x, n2)
            c5_de2 = evaluate_c5(h_de2)
            if c5_de2 < best_c52:
                best_c52 = c5_de2
                best_h2 = h_de2
        except Exception:
            pass

    # SLSQP refinement for n2
    for eps_val in [500.0, 2000.0, 5000.0, 10000.0]:
        if elapsed() > budget_s * 0.92:
            break
        h_new2, c5_new2 = slsqp_local(best_h2, n2, bounds_n2, eps_val, 500, 1e-15)
        if c5_new2 < best_c52:
            best_c52 = c5_new2
            best_h2 = h_new2

    # L-BFGS-B for n2
    if elapsed() < budget_s * 0.94:
        h_new2, c5_new2 = lbfgs_penalty(best_h2, n2, bounds_n2, 5000.0, 100000.0, 3000)
        if c5_new2 < best_c52:
            best_c52 = c5_new2
            best_h2 = h_new2
        h_new2, c5_new2 = slsqp_local(best_h2, n2, bounds_n2, 10000.0, 500, 1e-15)
        if c5_new2 < best_c52:
            best_c52 = c5_new2
            best_h2 = h_new2

    # Coordinate descent for n2
    if elapsed() < budget_s * 0.97:
        best_h2, best_c52 = coord_descent(best_h2, n2, deltas_c, 300, time_limit(0.97))

    # Final comparison
    if best_c52 < best_c5:
        final_h = best_h2
        final_c5 = best_c52
        final_n = n2
    else:
        final_h = best_h
        final_c5 = best_c5
        final_n = n

    final_h = project_feasible(final_h, final_n)
    final_c5 = evaluate_c5(final_h)
    return final_h, final_c5, final_n
