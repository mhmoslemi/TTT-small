def project_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Binary search for threshold that gives the right sum
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    n = len(h)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    return np.log(np.sum(np.exp(alpha * (corr - max_c)))) / alpha + max_c

def softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()

def project_feasible(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return log_sum / alpha + max_c

def softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    active = w > 1e-10
    for m in np.where(active)[0]:
        wm = w[m]
        k = m - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += wm * (1.0 - h[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= wm * h[idx + k]
    return grad * dx

            return np.ones((1, n))
