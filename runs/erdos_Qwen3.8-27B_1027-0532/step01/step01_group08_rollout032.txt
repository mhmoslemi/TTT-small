import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 - abs(lag))
        if lag >= 0:
            i_lo = max(0, lag)
            i_hi = min(n - 1, n - 1 - lag)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += w * (1.0 - h[i_lo+lag:i_hi+lag+1])
        else:
            i_lo = max(0, -lag)
            i_hi = min(n - 1, n - 1 + lag)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += w * (-h[i_lo+lag:i_hi+lag+1])
    return obj, grad * dx

def nesterov_pgd(h0, n, target_sum, deadline, max_iter=5000, tau_init=30.0, tau_max=500.0):

    y = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = tau_init
    t = 1.0
    lr = 0.02
    stall_count = 0
    stall_thresh = 60
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, g = soft_max_obj_and_grad(y, tau)

def coord_descent_1d(h, i, n, tau):
    dx = 2.0 / n
    a_coeffs = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        coef = 0.0
        if 0 <= i + lag < n:
            coef += (1.0 - h[i + lag])
        elif i + lag == i:
            pass
        if 0 <= i - lag < n and i - lag != i:
            coef += (-h[i - lag])
        a_coeffs[m] = coef
    corr = np.correlate(h, 1.0 - h, mode="full")
    b_coeffs = corr - h[i] * a_coeffs * 0 + (corr - h[i] * a_coeffs)
    # Actually b = corr - h[i]*a, but corr already includes h[i]'s contribution
    # When h[i] is set to t, C(ℓ) = a_ℓ * t + (corr - a_ℓ * h[i])
    # But corr = a_ℓ * h[i] + b_ℓ, so b_ℓ = corr - a_ℓ * h[i]
    b_coeffs = corr - h[i] * a_coeffs
    
    def neg_grad(t):
        tau_c = tau * (a_coeffs * t + b_coeffs) * dx
        mc = np.max(tau_c)
        exps = np.exp(tau_c - mc)
        s = np.sum(exps)
        if s < 1e-300:
            return 0.0
        pw = exps / s
        return np.sum(pw * a_coeffs) * dx
    
    lo_t = 0.0
    hi_t = 1.0
    g_lo = neg_grad(lo_t)
    g_hi = neg_grad(hi_t)
    if g_lo >= 0:
        return 0.0
    if g_hi <= 0:
        return 1.0
    for _ in range(50):
        mid = (lo_t + hi_t) / 2.0
        g_mid = neg_grad(mid)
        if g_mid > 0:
            lo_t = mid
        else:
            hi_t = mid
    return (lo_t + hi_t) / 2.0

def coordinate_descent(h0, n, target_sum, max_sweeps=200, tau=80.0, deadline=None):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    order = list(range(n))
    np.random.shuffle(order)

            diff = new_val - h[i]
            h[i] = new_val
            if abs(diff) > 1e-10:
                h = project_onto_constraint(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interpolate_h(h_old, n_new):

                                       max_iter=8000, tau_init=20.0, tau_max=300.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Coordinate descent polish
    remaining = deadline - time.time()
    if remaining > 30:
        alloc = min(remaining * 0.2, 150)
        dl = t_start + alloc
        n = len(best_h)
        h_opt, c5_opt = coordinate_descent(best_h, n, n/2.0, max_sweeps=300, tau=100.0, deadline=dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Try different n values
    n_candidates = [100, 140, 160, 200, 240]
    for n_try in n_candidates:
        remaining = deadline - time.time()
        if remaining < 80:
            break
        if n_try == len(best_h):
            continue
        
        h0 = interpolate_h(best_h, n_try)
        target = n_try / 2.0
        
        # Nesterov on this n
        alloc = min(remaining * 0.3, 120)
        dl = t_start + alloc
        h_opt, c5_opt = nesterov_pgd(h0, n_try, target, dl,
                                       max_iter=5000, tau_init=20.0, tau_max=300.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        
        # Coordinate descent polish
        remaining = deadline - time.time()
        if remaining < 30:
            break
        alloc = min(remaining * 0.2, 80)
        dl = t_start + alloc
        h_opt, c5_opt = coordinate_descent(best_h, len(best_h), len(best_h)/2.0,
                                            max_sweeps=200, tau=120.0, deadline=dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 4: More Nesterov with higher tau on best
    remaining = deadline - time.time()
    if remaining > 40:
        alloc = min(remaining * 0.3, 150)
        dl = t_start + alloc
        n = len(best_h)
        h_opt, c5_opt = nesterov_pgd(best_h, n, n/2.0, dl,
                                       max_iter=10000, tau_init=80.0, tau_max=800.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 5: Final coordinate descent with multiple tau values
    remaining = time.time() - t_start
    time_left = deadline - time.time()
    if time_left > 20:
        n = len(best_h)
        for tau_cd in [50.0, 100.0, 200.0, 400.0]:
            time_left = deadline - time.time()
            if time_left < 10:
                break
            alloc = min(time_left * 0.3, 60)
            dl = t_start + alloc
            h_opt, c5_opt = coordinate_descent(best_h, n, n/2.0,
                                                max_sweeps=500, tau=tau_cd, deadline=dl)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 6: Random restarts if time allows
    time_left = deadline - time.time()
    if time_left > 60:
        for attempt in range(6):
            time_left = deadline - time.time()
            if time_left < 20:
                break
            n_try = len(best_h)
            target = n_try / 2.0
            h_rand = diverse_init(n_try, target, seed + 100 + attempt * 7)
            
            alloc = min(time_left * 0.25, 40)
            dl = t_start + alloc
            h_opt, c5_opt = nesterov_pgd(h_rand, n_try, target, dl,
                                           max_iter=3000, tau_init=30.0, tau_max=200.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            
            time_left = deadline - time.time()
            if time_left < 10:
                break
            alloc = min(time_left * 0.2, 20)
            dl = t_start + alloc
            h_opt, c5_opt = coordinate_descent(h_opt, n_try, target,
                                                max_sweeps=100, tau=100.0, deadline=dl)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Final coordinate descent polish
    time_left = deadline - time.time()
    if time_left > 5:
        n = len(best_h)
        dl = t_start + time_left * 0.8
        h_opt, c5_opt = coordinate_descent(best_h, n, n/2.0,
                                            max_sweeps=1000, tau=200.0, deadline=dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Ensure feasibility
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n
