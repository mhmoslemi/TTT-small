def compute_gradient_efficient(h, w):
    """Compute gradient using the fact that only shifts with significant weight matter."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each shift m with weight w[m], k = m - (n-1)
    # dc[m]/dh[i] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    # We can vectorize this
    for m in range(2 * n - 1):

        if wm < 1e-10:

        # Contribution: (1-h[i-k])
        i_start = max(0, k)
        i_end = min(n, n + k)  # exclusive
        if i_start < i_end:
            grad[i_start:i_end] += wm * (1.0 - h[i_start - k:i_end - k])
        # Case 2: i+k valid means i ranges from max(0,-k) to min(n-1, n-1-k)
        # Contribution: -h[i+k]
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)  # exclusive
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= wm * h[i_start2 + k:i_end2 + k]
    return dx * grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def smooth_weights(c, alpha):
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / exp_s.sum()
    return w

def compute_gradient_efficient(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_start < i_end:
            grad[i_start:i_end] += wm * (1.0 - h[i_start - k:i_end - k])
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= wm * h[i_start2 + k:i_end2 + k]
    return dx * grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def gd_optimize(h_init, n, deadline, alpha_start=20.0, alpha_end=500.0,
                max_iters=6000, lr_start=0.4, momentum=0.85):
    target = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_start
    no_improve = 0
    patience = 500

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t = min(it / max(max_iters - 1, 1), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** t

        c = compute_c5_all(h)
        c5_val = float(np.max(c))
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience and it > 100:
            break

        w = smooth_weights(c, alpha)
        grad = compute_gradient_efficient(h, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        v = momentum * v + grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-14:
            break

        accepted = False
        for trial in range(12):
            step = lr * (0.5 ** trial)
            candidate = h - step * v / vnorm
            candidate = project_feasible(candidate, target)
            c5_trial = compute_c5(candidate)
            if c5_trial <= c5_val + 1e-12:
                h = candidate
                accepted = True
                break

        if not accepted:
            lr *= 0.8
            v = 0.5 * v

    return best_h, best_c5

def upscale(h_low, n_low, n_high):
    x_low = np.linspace(0, 2, n_low)
    x_high = np.linspace(0, 2, n_high)
    h_high = np.interp(x_high, x_low, h_low)
    return project_feasible(h_high, n_high / 2.0)

def make_anti_periodic_init(n, rng):
    half = n // 2
    h_half = np.clip(rng.uniform(0.1, 0.9, half), 0, 1)
    h = np.zeros(n)
    if n % 2 == 0:
        h[:half] = h_half
        h[half:] = 1.0 - h_half
    else:
        h[:half] = h_half
        h[half:n-1] = 1.0 - h_half
        h[n-1] = 0.5
    return project_feasible(h, n / 2.0)

def make_sine_init(n, freq, phase, amp):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

def make_multisine_init(n, rng):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    n_freqs = rng.randint(2, 5)
    for _ in range(n_freqs):
        f = rng.uniform(1, 8)
        p = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.15)
        h += amp * np.sin(2.0 * np.pi * f * x / 2.0 + p)
    h = 0.5 + h
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

def make_random_init(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n / 2.0)

def coordinate_refine(h, n, deadline, max_rounds=50):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            c5_before = compute_c5(h)
            save = h[i]
            for trial_val in [0.0, 1.0, save + 0.01, save - 0.01, save + 0.005, save - 0.005]:
                h[i] = trial_val
                h_adj = project_feasible(h, target)
                c5_trial = compute_c5(h_adj)
                if c5_trial < best_c5 - 1e-10:
                    best_c5 = c5_trial
                    best_h = h_adj.copy()
                    h = h_adj.copy()
                    improved = True
                    break
            h[i] = save
        if not improved:
            break

    return best_h, best_c5
