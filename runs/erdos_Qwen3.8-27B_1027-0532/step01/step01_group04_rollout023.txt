    corr = np.correlate(h, 1.0 - h, mode='full')
    excess = np.maximum(corr - target, 0.0)
    value = 0.5 * float(np.sum(excess * excess))
    weights = excess.copy()
    wsum = float(np.sum(weights))
    if wsum < 1e-15:
        return value, np.zeros(n)

    term1 = np.zeros(n)
    term2 = np.zeros(n)

    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        k = j - (n - 1)
        if k >= 0:
            length = n - k
            term1[:length] += w * (1.0 - h[k:])
            term2[k:] += w * h[:length]
        else:
            kk = -k
            length = n - kk
            term1[kk:] += w * (1.0 - h[:length])
            term2[:length] += w * h[kk:]

    grad = term1 - term2
    return value, grad


def pgd_step_smooth(h, S, alpha, lr):
    _, grad = smooth_max_and_grad(h, alpha)
    h_new = h - lr * grad
    return project_box_hyperplane(h_new, S)


def pgd_step_hinge(h, S, target, lr):
    _, grad = hinge_grad(h, target)
    h_new = h - lr * grad
    return project_box_hyperplane(h_new, S)


def binary_projection(h, S):
    n = len(h)
    S_int = int(round(S))
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:S_int]] = 1.0
    return h_bin


def slsqp_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = 150.0 * (corr - c_max)
    np.clip(shifted, -500.0, 0.0, out=shifted)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / 150.0


def slsqp_eq(h):
    return np.sum(h) - _g_S


def slsqp_jac(h):
    return np.ones(_g_n)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_S, _g_n
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    S = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Projected gradient with log-sum-exp annealing
    alphas_p1 = [5, 15, 40, 100, 250, 500, 1000, 2000]
    lrs_p1 = [0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003]

    for i in range(len(alphas_p1)):
        if time.time() > deadline:
            break
        alpha = alphas_p1[i]
        lr = lrs_p1[i]
        lr_max = lrs_p1[i] * 3.0
        stall = 0
        for it in range(8000):
            if time.time() > deadline:
                break
            h_new = pgd_step_smooth(best_h, S, alpha, lr)
            s_new = compute_c5(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new
                stall = 0
                lr = min(lr * 1.03, lr_max)
            else:
                stall += 1
                lr *= 0.85
                if stall > 150:
                    break

    # Phase 2: Hinge-based refinement
    if time.time() < deadline:
        target = best_score / (2.0 / n) * 0.95
        lr_h = 0.01
        stall = 0
        for it in range(5000):
            if time.time() > deadline:
                break
            h_new = pgd_step_hinge(best_h, S, target, lr_h)
            s_new = compute_c5(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new
                target = max(best_score / (2.0 / n) * 0.9, target * 0.99)
                lr_h = min(lr_h * 1.03, 0.03)
                stall = 0
            else:
                stall += 1
                lr_h *= 0.9
                if stall > 200:
                    break

    # Phase 3: Binary projection and refine
    if time.time() < deadline:
        h_bin = binary_projection(best_h, S)
        s_bin = compute_c5(h_bin)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()

        if time.time() < deadline:
            for alpha, lr0 in [(30, 0.01), (100, 0.005), (400, 0.001)]:
                if time.time() > deadline:
                    break
                lr = lr0
                stall = 0
                for it in range(5000):
                    if time.time() > deadline:
                        break
                    h_new = pgd_step_smooth(best_h, S, alpha, lr)
                    s_new = compute_c5(h_new)
                    if s_new < best_score - 1e-12:
                        best_h = h_new
                        best_score = s_new
                        lr = min(lr * 1.03, lr0 * 2.5)
                        stall = 0
                    else:
                        stall += 1
                        lr *= 0.87
                        if stall > 200:
                            break

    # Phase 4: Try larger grids
    n_use = len(best_h)
    S_use = n_use / 2.0

    for n_try in [440, 550, 700]:
        if time.time() > deadline:
            break
        S_try = n_try / 2.0
        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, len(best_h))
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_box_hyperplane(h_try, S_try)
        s_try = compute_c5(h_try)

        for alpha, lr0 in [(30, 0.01), (100, 0.004), (300, 0.001)]:
            if time.time() > deadline:
                break
            lr = lr0
            stall = 0
            for it in range(4000):
                if time.time() > deadline:
                    break
                h_try = pgd_step_smooth(h_try, S_try, alpha, lr)
                s_try = compute_c5(h_try)
                if s_try < best_score - 1e-12:
                    best_h = h_try.copy()
                    best_score = s_try
                    lr = min(lr * 1.03, lr0 * 2.0)
                    stall = 0
                else:
                    stall += 1
                    lr *= 0.87
                    if stall > 250:
                        break

    # Phase 5: Restarts with noise (use best grid)
    n_use = len(best_h)
    S_use = n_use / 2.0

    for r in range(40):
        if time.time() > deadline:
            break
        noise_scale = 0.03 + 0.04 * rng.rand()
        h_p = project_box_hyperplane(best_h + rng.randn(n_use) * noise_scale, S_use)
        alpha = int(rng.choice([30, 80, 200, 500]))
        lr = 0.005
        h_cur = h_p.copy()
        s_cur = compute_c5(h_cur)
        stall = 0
        for it in range(3000):
            if time.time() > deadline:
                break
            h_cur = pgd_step_smooth(h_cur, S_use, alpha, lr)
            s_new = compute_c5(h_cur)
            if s_new < s_cur - 1e-12:
                s_cur = s_new
                lr = min(lr * 1.03, 0.015)
                stall = 0
            else:
                stall += 1
                lr *= 0.88
                if stall > 150:
                    break
        if s_cur < best_score:
            best_score = s_cur
            best_h = h_cur.copy()

    # Phase 6: Fine-tune with alternating alpha and hinge
    if time.time() < deadline:
        n_use = len(best_h)
        S_use = n_use / 2.0
        for cycle in range(3):
            if time.time() > deadline:
                break
            alpha = 500 + 300 * cycle
            lr = 0.001 / (1.0 + cycle * 0.5)
            stall = 0
            for it in range(5000):
                if time.time() > deadline:
                    break
                h_new = pgd_step_smooth(best_h, S_use, alpha, lr)
                s_new = compute_c5(h_new)
                if s_new < best_score - 1e-12:
                    best_h = h_new
                    best_score = s_new
                    lr = min(lr * 1.02, 0.003 / (1.0 + cycle * 0.5))
                    stall = 0
                else:
                    stall += 1
                    lr *= 0.9
                    if stall > 300:
                        break
            if time.time() > deadline:
                break
            target = best_score / (2.0 / n_use) * 0.92
            lr_h = 0.003 / (1.0 + cycle * 0.5)
            stall = 0
            for it in range(3000):
                if time.time() > deadline:
                    break
                h_new = pgd_step_hinge(best_h, S_use, target, lr_h)
                s_new = compute_c5(h_new)
                if s_new < best_score - 1e-12:
                    best_h = h_new
                    best_score = s_new
                    lr_h = min(lr_h * 1.02, 0.008 / (1.0 + cycle * 0.5))
                    stall = 0
                else:
                    stall += 1
                    lr_h *= 0.9
                    if stall > 200:
                        break

    # Phase 7: SLSQP polish
    if time.time() < deadline:
        n_use = len(best_h)
        S_use = n_use / 2.0
        _g_S = S_use
        _g_n = n_use
        bounds = [(0.0, 1.0)] * n_use
        cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_jac}]
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 1000, 'ftol': 1e-15})
            h_pol = project_box_hyperplane(res.x, S_use)
            s_pol = compute_c5(h_pol)
            if s_pol < best_score:
                best_score = s_pol
                best_h = h_pol
        except Exception:
            pass

    # Final projection and evaluation
    n_use = len(best_h)
    S_use = n_use / 2.0
    h_final = project_box_hyperplane(best_h, S_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
