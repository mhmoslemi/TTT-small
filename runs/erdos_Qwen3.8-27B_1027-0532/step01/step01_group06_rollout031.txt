    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    val = dx * (cmax + T * (m + np.log(np.sum(exp_s))))
    one_minus_2h = 1.0 - 2.0 * h
    grad = np.zeros(n)
    for p in range(n):

        if p <= 2 * n - 2:
            grad[p] = dx * np.dot(w[p:end], one_minus_2h[:end - p])
    return val, grad, cmax * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def thue_morse(n):
    t = np.zeros(n, dtype=int)
    t[0] = 0
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def init_thue_morse(n):
    tm = thue_morse(n)
    h = tm.astype(float)
    # Adjust to satisfy sum = n/2
    # tm has roughly n/2 ones
    s = np.sum(h)
    if s > n / 2:
        # Need to reduce: set some 1s to 0
        excess = s - n / 2
        ones = np.where(h == 1)[0]
        for i in range(int(np.ceil(excess))):
            h[ones[i]] = 0
            excess -= 1
        if excess > 0:
            h[ones[int(np.ceil(excess)) - 1]] = 1 - excess
    elif s < n / 2:
        deficit = n / 2 - s
        zeros = np.where(h == 0)[0]
        for i in range(int(np.ceil(deficit))):
            h[zeros[i]] = 1
            deficit -= 1
        if deficit > 0:
            h[zeros[int(np.ceil(deficit)) - 1]] = deficit
    return project_feasible(h, n)

def projected_gradient_descent(h, n, t0, time_limit, T_start=1.0, T_end=0.01):

    # Generate Thue-Morse sequence
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        # Count trailing ones in i, mod 2
        val = 0
        x = i
        while x > 0:
            val ^= (x & 1)
            x >>= 1
        t[i] = val

    s = np.sum(t)
    if s > n / 2:
        excess = s - n / 2
        idx = np.where(t == 1)[0]
        for j in range(idx.size):
            if excess <= 0:
                break
            take = min(1.0, excess)
            t[idx[j]] -= take
            excess -= take
    elif s < n / 2:
        deficit = n / 2 - s
        idx = np.where(t == 0)[0]
        for j in range(idx.size):
            if deficit <= 0:
                break
            take = min(1.0, deficit)
            t[idx[j]] += take
            deficit -= take
    return t
