    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx  # shape (2n-1,)
    t_grid = np.linspace(0.0, 1.0, n_grid)

        jl = j - lags

        B = corr - A * h[j]

        vals = A[None, :] * t_grid[:, None] + B[None, :]

        if abs(new_val - h[j]) > 1e-10:
            # Update corr
            corr += A * (new_val - h[j])
            h[j] = new_val

        term1 = np.zeros(2 * n - 1)
        m1 = (jl >= 0) & (jl < n)
        term1[m1] = 1.0 - h[jl[m1]]
        term2 = np.zeros(2 * n - 1)
        m2 = (jr >= 0) & (jr < n)
        term2[m2] = h[jr[m2]]
        A = dx * (term1 - term2)
        B = corr - A * h[j]
        vals = A[None, :] * t_grid[:, None] + B[None, :]
        f_vals = vals.max(axis=1)
        best_idx = f_vals.argmin()
        new_val = t_grid[best_idx]
        if abs(new_val - h[j]) > 1e-12:
            delta = new_val - h[j]
            corr += A * delta
            h[j] = new_val
    return h

def smooth_max_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def slsqp_obj(h, dx, beta):
    return smooth_max_c5(h, dx, beta)

def slsqp_con(h, n):
    return np.sum(h) - n * 0.5

def slsqp_refine(h0, n, dx, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    res = minimize(slsqp_obj, h0, args=(dx, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if res.x is not None:
        return repair_h(res.x, n)
    return h0

def penalty_obj(h, dx, n, lam):
    base = c5_value(h, dx)
    viol = abs(np.sum(h) - n * 0.5)
    return base + lam * viol

def nelder_refine(h0, n, dx, maxiter):
    res = minimize(penalty_obj, h0, args=(dx, n, 500.0),
                   method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-14, 'fatol': 1e-16})
    if res.x is not None:
        return repair_h(res.x, n)
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

        return budget_s - elapsed()

    h_parent = np.asarray(initial_h_values, dtype=np.float64)

        if sweep % 50 == 0:
            h_work = repair_h(h_work, n)
            c = c5_value(h_work, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h_work.copy()
        if sweep % 500 == 0 and remaining() > 500:
            h_work = repair_h(h_work + rng.randn(n) * 0.02, n)

    h_work = repair_h(h_work, n)
    c = c5_value(h_work, dx)
    if c < best_c5:
        best_c5 = c
        best_h = h_work.copy()

    if remaining() > 200:

            h_opt = slsqp_refine(best_h, best_n, 2.0 / best_n, beta, maxiter=2000)

    if remaining() > 100:

    if remaining() > 250:

            h2 = coord_descent_sweep(h2, n2, dx2, 256, rng)
            if sweep % 100 == 0:
                h2 = repair_h(h2, n2)
                c = c5_value(h2, dx2)
                if c < best_c5:
                    best_c5 = c
                    best_h = h2.copy()
                    best_n = n2
        h2 = repair_h(h2, n2)
        c = c5_value(h2, dx2)
        if c < best_c5:
            best_c5 = c
            best_h = h2.copy()
            best_n = n2

        if remaining() > 100:
            h_opt = slsqp_refine(best_h, best_n, 2.0 / best_n, 50.0, maxiter=1000)
            c = c5_value(h_opt, 2.0 / best_n)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt

    if remaining() > 200:
        inits = []
        coeffs1 = [(1, 0.3), (2, 0.2), (3, 0.15), (4, 0.1), (5, 0.08)]
        inits.append(make_fourier_h(n, coeffs1))
        coeffs2 = [(1, 0.25), (3, 0.25), (5, 0.15), (7, 0.1)]
        inits.append(make_fourier_h(n, coeffs2))
        coeffs3 = [(2, 0.3), (4, 0.2), (6, 0.15), (8, 0.1)]
        inits.append(make_fourier_h(n, coeffs3))
        coeffs4 = [(1, 0.2), (2, 0.2), (3, 0.2), (4, 0.15), (5, 0.1), (6, 0.08)]
        inits.append(make_fourier_h(n, coeffs4))

        for h_init in inits:

                h_tmp = coord_descent_sweep(h_tmp, n, dx, 256, rng)
                if sweep % 200 == 0:
                    h_tmp = repair_h(h_tmp, n)
            h_tmp = repair_h(h_tmp, n)
            c = c5_value(h_tmp, dx)

    return best_h, best_c5, best_n
