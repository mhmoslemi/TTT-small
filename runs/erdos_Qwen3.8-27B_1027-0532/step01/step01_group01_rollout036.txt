```python
    # grad_j = dx * sum_k w_k * gamma_k(j)
    # gamma_k(j) = (1-x[j+l_k]) if valid - (x[j-l_k]) if valid
    # where l_k = k - (n-1)
    
    grad = np.zeros(n)
    l_vals = np.arange(-(n-1), n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        l = l_vals[k]
        for j in range(n):
            g = 0.0
            if 0 <= j + l < n:
                g += 1.0 - x[j + l]
            if 0 <= j - l < n:
                g -= x[j - l]
            grad[j] += w[k] * g
    grad *= dx
    return grad
```

This is O(n * (2n-1)) = O(n^2) due to the double loop. For n=85, that's 85 * 169 = 14365 operations. That's fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_eps = 500.0
_g_n = 85

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_gamma_vec(h, j, n):
    n_corr = 2 * n - 1
    l_vals = np.arange(-(n - 1), n)
    ja = j + l_vals
    jb = j - l_vals
    gamma = np.zeros(n_corr)
    va = (ja >= 0) & (ja < n)
    vb = (jb >= 0) & (jb < n)
    gamma[va] = 1.0 - h[ja[va]]
    gamma[vb] -= h[jb[vb]]
    return gamma

def paired_best_delta(corr, d, lo, hi):
    if lo >= hi:
        return 0.0
    a, b = lo, hi
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * d)
        f2 = np.max(corr + m2 * d)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def slsqp_objective(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = _g_eps * (corr - m)
    return (m + (1.0 / _g_eps) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_gradient(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = _g_eps * (corr - m)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    j_range = np.arange(n)
    l_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    sig = w > 1e-12
    for k in np.where(sig)[0]:
        l = l_vals[k]
        ja = j_range + l
        jb = j_range - l
        g_k = np.zeros(n)
        va = (ja >= 0) & (ja < n)
        vb = (jb >= 0) & (jb < n)
        g_k[va] = 1.0 - x[ja[va]]
        g_k[vb] -= x[jb[vb]]
        grad += w[k] * g_k
    grad *= dx
    return grad

def slsqp_eq(x):
    return np.sum(x) - _g_n / 2.0

def do_slsqp(h, n, maxiter=300, eps=500.0):
    global _g_n, _g_eps
    _g_n = n
    _g_eps = eps
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_objective, h.copy(), jac=slsqp_gradient,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': slsqp_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def paired_descent(h, n, max_time, rng):
    """Run paired coordinate descent for max_time seconds."""
    start = time.time()
    corr = compute_corrs(h)
    n_corr = 2 * n - 1
    j_range = np.arange(n)
    
    while time.time() - start < max_time:
        # Random pairs
        for _ in range(300):
            j1 = rng.randint(n)
            j2 = rng.randint(n)
            if j1 == j2:
                continue
            g1 = compute_gamma_vec(h, j1, n)
            g2 = compute_gamma_vec(h, j2, n)
            d = g1 - g2
            lo = max(-h[j1], h[j2] - 1.0)
            hi = min(1.0 - h[j1], h[j2])
            if lo >= hi - 1e-15:
                continue
            opt = paired_best_delta(corr, d, lo, hi)
            if abs(opt) > 1e-14:
                new_corr = corr + opt * d
                if np.max(new_corr) < np.max(corr) - 1e-13:
                    h[j1] += opt
                    h[j2] -= opt
                    corr = new_corr
        
        # Adjacent sweep
        for j in range(n):
            j1 = j
            j2 = (j + 1) % n
            g1 = compute_gamma_vec(h, j1, n)
            g2 = compute_gamma_vec(h, j2, n)
            d = g1 - g2
            lo = max(-h[j1], h[j2] - 1.0)
            hi = min(1.0 - h[j1], h[j2])
            if lo >= hi - 1e-15:
                continue
            opt = paired_best_delta(corr, d, lo, hi)
            if abs(opt) > 1e-14:
                new_corr = corr + opt * d
                if np.max(new_corr) < np.max(corr) - 1e-13:
                    h[j1] += opt
                    h[j2] -= opt
                    corr = new_corr
    
    return h
```
