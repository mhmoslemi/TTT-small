def compute_subgrad(h, k, n):
    """Compute subgradient of f_k at h"""
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    g = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    g[m1] += (1.0 - h[a[m1]])
    g[m2] -= h[b[m2]]
    return g * dx

    return g * dx, corr
