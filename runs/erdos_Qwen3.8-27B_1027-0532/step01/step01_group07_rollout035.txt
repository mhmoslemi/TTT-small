import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_B(h):
    """Compute B(k) = A(k) - S(k) for k = 0, ..., n-1
    where A(k) = sum h[i]*h[i+k] and S(k) = sum h[i] over valid i
    B(k) being large means good (S(k)-A(k) is small)"""
    n = len(h)
    B = np.zeros(n)
    for k in range(n):
        valid = n - k
        S = np.sum(h[:valid])
        A = np.sum(h[:valid] * h[k:k+valid])
        B[k] = A - S
    return B

def compute_B_grad(h, beta):
    """Compute gradient of f(h) = -(1/beta) * logsumexp(beta * B)
    Which approximates -min_k B(k), so minimizing it maximizes min B(k)."""
    n = len(h)
    B = compute_B(h)
    # Weight toward minimum B(k)
    shifted = beta * B - beta * B.max()  # for numerical stability
    w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-10:
            continue
        valid = n - k
        # d!k B(k)/dh[i]:
        # For i < valid: contributes h[i+k] - 1
        # For i >= k: contributes h[i-k]
        vec = np.zeros(n)
        vec[:valid] += h[k:k+valid] - 1.0
        vec[k:] += h[:valid]
        grad += wk * vec
    
    return grad  # This is -d f/dh (since f = -(1/beta)*logsumexp, -df/dh = sum w_k * dBk/dh)

def smooth_gradient_descent(h, n, beta_schedule, lr, deadline):
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_lr = lr
    
    for beta in beta_schedule:
        for it in range(5000):
            if time.time() > deadline:
                break
            grad = compute_B_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step in direction of increasing B (i.e., decreasing f)
            # f = -(1/beta)*logsumexp(beta*B), -df/dh = sum w_k * dBk/dh = grad (as computed)
            # We want to minimize f, so step in -grad direction... 
            # Wait: we want to MAXIMIZE min B(k), which means minimize f = -min B(k) approx
            # df/dh = -sum w_k * dBk/dh = -grad
            # So to minimize f, step in -df/dh direction = +grad direction
            h_new = h + cur_lr * grad / max(gnorm, 1e-10)
            h_new = project_feasible(h_new, target)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_val:
                best_val = c5_new
                best_h = h_new.copy()
            
            # Backtracking
            if c5_new <= compute_c5(h) + 1e-12:
                h = h_new
            else:
                cur_lr *= 0.7
                if cur_lr < 1e-8:
                    break
        # Reset lr for next beta
        cur_lr = max(lr * 0.5, 0.01)
    
    return best_h, best_val
