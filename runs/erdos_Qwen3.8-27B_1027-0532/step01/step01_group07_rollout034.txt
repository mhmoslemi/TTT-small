def compute_smooth_grad(h, alpha):

    c = np.correlate(h, 1.0 - h, mode='full') * dx

    Part2 = M2.dot(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(alpha * c_shifted)
    w = exp_c / exp_c.sum()
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return c_max + np.log(exp_c.sum()) / alpha, grad

def project_feasible(h, target):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

            h10, c5_10 = gd_nesterov(best_h, 100.0, 5000, deadline, alpha_max=10000.0, lr_init=0.05)
            if c5_10 < best_c5:
                best_c5, best_h = c5_10, h10.copy()

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_h = project_feasible(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
