```python
def lse_obj_and_grad(x):
    n = len(x)
    dx = 2.0 / n
    T = _CURRENT_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled)
    weights = exp_scaled / exp_scaled.sum()  # softmax
    obj = dx * (cmax + T * np.log(exp_scaled.sum()))
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(len(corr)):
        s = m - (n - 1)  # shift
        w = weights[m]
        # For this shift s, d(corr[m])/d(h[i]) = (1-h[i+s])*I[0<=i+s<n] - h[i-s]*I[0<=i-s<n]
        # First term: (1-h[i+s]) for i such that 0 <= i+s < n, i.e., -s <= i < n-s
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            grad[i_start:i_end] += w * (1.0 - x[i_start+s:i_end+s])
        # Second term: -h[i-s] for i such that 0 <= i-s < n, i.e., s <= i < n+s
        i_start2 = max(0, s)
        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w * x[i_start2-s:i_end2-s]
    
    grad *= dx
    return obj, grad
```

_CURRENT_MU = 100.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_gradient(h, n):
    dx = 2.0 / n
    T = _CURRENT_T
    mu = _CURRENT_MU
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    weights = exp_scaled / total
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    s_range = np.arange(-(n-1), n)

        w = weights[idx]
        if w < 1e-15:
            continue

        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w * h[i_start2-s:i_end2-s]

    grad += mu * (np.sum(h) - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_CURRENT_T = 1.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    T = _CURRENT_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    return dx * (cmax + T * np.log(np.sum(np.exp(scaled))))

def lse_grad(x):
    n = len(x)
    dx = 2.0 / n
    T = _CURRENT_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_scaled / total
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        w = weights[idx]
        if w < 1e-14:
            continue
        s = idx - (n - 1)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            grad[i_start:i_end] += w * (1.0 - x[i_start + s:i_end + s])
        i_start2 = max(0, s)
        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= w * x[i_start2 - s:i_end2 - s]
    return grad * dx

def lse_obj_grad(x):
    return lse_obj(x), lse_grad(x)

def run_lbfgsb(h, n, T, max_iter, t_start, time_limit):
    global _CURRENT_T
    _CURRENT_T = T
    try:
        res = minimize(lse_obj_grad, h, method='L-BFGS-B', jac=True,
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'gtol': 1e-10, 'iprint': -1})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)

def compute_a_matrix(h, n):
    n_corr = 2 * n - 1
    a = np.zeros((n, n_corr))
    for idx in range(n_corr):
        s = idx - (n - 1)
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            a[i_start:i_end, idx] = (1.0 - h[i_start + s:i_end + s])
        i_start2 = max(0, s)
        i_end2 = min(n, n + s)
        if i_start2 < i_end2:
            a[i_start2:i_end2, idx] -= h[i_start2 - s:i_end2 - s]
    return a
