def nesterov_optimize(h_init, eps, time_limit, target_sum):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_val = ls_exp_obj(h, eps)
    best_true = compute_c5(h)
    lr = 1.0
    momentum = 0.9
    start = time.time()
    
    while time.time() - start < time_limit:
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0, 1)  # soft clip for gradient eval
        
        grad = ls_exp_grad_fast(y, eps)
        
        # Projected gradient step with backtracking
        for _ in range(50):
            h_new = project_onto_constraint(y - lr * grad, target_sum)
            val_new = ls_exp_obj(h_new, eps)
            if val_new <= ls_exp_obj(y, eps) - 0.5 * lr * np.sum(grad * (h_new - y)):
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-8)
            h_new = project_onto_constraint(y - lr * grad, target_sum)
        
        v = h_new - h
        h = h_new
        
        true_val = compute_c5(h)

            best_true = true_val
            best_h = h.copy()
            momentum = min(momentum * 1.01, 0.95)
        elif np.random.random() < 0.1:
            # Occasionally reset momentum
            momentum = 0.8
    
    return best_h

    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)

    w = ex / sexp
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def ls_exp_val(h, eps):
    corr = compute_corr_dx(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def ls_exp_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * omh[:n - s]
            grad[:n - s] -= wk * h[s:]
        else:
            r = -s
            grad[:n - r] += wk * omh[r:]
            grad[r:] -= wk * h[:n - r]
    grad *= dx
    return grad

def eq_constraint_fun(h, target_sum):
    return float(np.sum(h) - target_sum)

def eq_constraint_jac(h, target_sum):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, target_sum):
    n = len(h_init)
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: eq_constraint_fun(x, target_sum),
             'jac': lambda x: eq_constraint_jac(x, target_sum)}]
    try:
        result = minimize(ls_exp_val, h0, args=(eps,),
                          jac=lambda x: ls_exp_grad(x, eps),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def nesterov_pg(h_init, eps, time_limit, target_sum):
    n = len(h_init)
    h = h_init.copy()
    m = np.zeros(n)
    best_h = h.copy()
    best_true = compute_c5(h)
    lr = 0.02
    mu = 0.85
    start = time.time()
    
    for iteration in range(200000):
        if time.time() - start > time_limit:
            break
        y = h + mu * m
        y_safe = np.clip(y, 0.0, 1.0)
        g = ls_exp_grad(y_safe, eps)
        val_y = ls_exp_val(y_safe, eps)
        step = lr
        h_new = None
        for _ in range(30):
            cand = project_onto_constraint(y_safe - step * g, target_sum)
            val_c = ls_exp_val(cand, eps)
            if val_c <= val_y - 0.01 * step * np.dot(g, y_safe - cand):
                h_new = cand
                break
            step *= 0.5
        if h_new is None:
            h_new = project_onto_constraint(y_safe - step * g, target_sum)
        m = h_new - h
        h = h_new
        lr = max(step * 1.5, 1e-8)
        true_val = compute_c5(h)
        if true_val < best_true - 1e-12:
            best_true = true_val
            best_h = h.copy()
    return best_h, best_true
