    resolutions = sorted(set([n_init, 500, 700, 900, 1200]))

    for n in resolutions:
        if tl() < 120:
            break

        if n != len(best_h):
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            h_curr = np.interp(x_new, x_old, best_h)
        else:
            h_curr = best_h.copy()
        h_curr = repair_h(h_curr, n)
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")

        n_passes_min = 80
        n_passes_max = 400
        pass_count = 0
        stall_count = 0
        prev_c5 = float('inf')

        for it in range(n_passes_max):
            if tl() < 60:
                break
            h_curr, corr = coord_descent_pass(h_curr, n, corr)
            h_curr = repair_h(h_curr, n)
            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
            pass_count += 1
            c5 = evaluate_c5(h_curr)
            if c5 < best_c5:
                best_h = h_curr.copy()
                best_c5 = c5
                stall_count = 0
            else:
                if abs(prev_c5 - c5) < 1e-8:
                    stall_count += 1
                else:
                    stall_count = max(0, stall_count - 1)
            prev_c5 = c5
            if pass_count >= n_passes_min and stall_count > 30:
                break

        if tl() > 60:
            n_final = len(best_h)
            if n_final != n:
                h_cur = repair_h(np.interp(np.linspace(0, 2, n),
                                           np.linspace(0, 2, n_final), best_h), n)
            else:
                h_cur = best_h.copy()
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            for it in range(100):
                if tl() < 40:
                    break
                h_cur, corr = coord_descent_pass(h_cur, n, corr, momentum=0.95)
                h_cur = repair_h(h_cur, n)
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
                c5 = evaluate_c5(h_cur)
                if c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5

    n_best = len(best_h)

    for restart in range(30):
        if tl() < 45:
            break
        x = np.linspace(0, 2, n_best)
        r = restart
        if r % 5 == 0:
            h_s = best_h + rng.normal(0, 0.03, n_best)
        elif r % 5 == 1:
            h_s = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * x) + 0.15 * np.sin(2 * np.pi * 5 * x + 0.7)
        elif r % 5 == 2:
            h_s = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x + 1.2)
        elif r % 5 == 3:
            h_s = rng.uniform(0, 1, n_best)
        else:
            nblk = 6 + rng.randint(0, 8)
            h_s = np.zeros(n_best)
            bs = max(1, n_best // (2 * nblk))
            for b in range(0, n_best, 2 * bs):
                h_s[b:b + bs] = 1.0
        h_s = np.clip(h_s, 0, 1)
        h_s = repair_h(h_s, n_best)
        corr = np.correlate(h_s, 1.0 - h_s, mode="full")

        n_inner = 30
        for it in range(n_inner):
            if tl() < 30:
                break
            h_s, corr = coord_descent_pass(h_s, n_best, corr)
            h_s = repair_h(h_s, n_best)
            corr = np.correlate(h_s, 1.0 - h_s,

mode="full")
            c5 = evaluate_c5(h_s)
            if c5 < best_c5:
                best_h = h_s.copy()
                best_c5 = c5

    if tl() > 25:
        h_pol = best_h.copy()
        corr = np.correlate(h_pol, 1.0 - h_pol, mode="full")
        for it in range(50):
            if tl() < 10:
                break
            h_pol, corr = coord_descent_pass(h_pol, len(h_pol), corr, momentum=0.98)
            h_pol = repair_h(h_pol, len(h_pol))
            corr = np.correlate(h_pol, 1.0 - h_pol, mode="full")
            c5 = evaluate_c5(h_pol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pol.copy()

    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
