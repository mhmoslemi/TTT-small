def score_c5(h):

    return corr * dx

        else:

    hi = 5.0

            if len(interior) > 0:
                params[interior] += diff / len(interior)
                params = np.clip(params, 0.0, 1.0)

        try:
            res = minimize(periodic_obj, params, args=(period, n), method='SLSQP',
                           bounds=bounds, constraints=constraint,
                           options={'maxiter': 3000, 'ftol': 1e-16})
            if res.x is not None and res.success:
                s = periodic_obj(res.x, period, n)
                if s < best_score:
                    best_score = s
                    best_params = res.x.copy()
        except Exception:
            pass

    if best_params is not None:
        h = make_periodic_h(best_params, period, n)
        return h, best_score
    return None, np.inf


def pairwise_cd(h, n, max_sweeps=20, time_limit=None, t_start=None, n_cands=8):
    best = h.copy()
    best_score = score_c5(best)
    rng = np.random.RandomState(12345)
    stall_count = 0

    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)

        for ii, i in enumerate(order):
            if time_limit is not None and time.time() - t_start > time_limit:
                return best, best_score
            for j in range(i + 1, n):
                if time_limit is not None and time.time() - t_start > time_limit:
                    return best, best_score
                pair_sum = best[i] + best[j]
                t_min = max(0.0, pair_sum - 1.0)
                t_max = min(1.0, pair_sum)
                if t_min >= t_max - 1e-14:
                    continue

                span = t_max - t_min
                candidates = np.concatenate([
                    [best[i]],
                    [t_min + 1e-8, t_max - 1e-8],
                    np.linspace(t_min, t_max, n_cands + 2)[1:-1]
                ])

                best_t = best[i]
                best_val = best_score

                for t in candidates:
                    t = np.clip(t, 0.0, 1.0)
                    vj = pair_sum - t
                    if vj < -1e-10 or vj > 1.0 + 1e-10:
                        continue
                    vj = np.clip(vj, 0.0, 1.0)
                    trial = best.copy()
                    trial[i] = t
                    trial[j] = vj
                    s = score_c5(trial)
                    if s < best_val - 1e-14:
                        best_val = s
                        best_t = t

                if best_val < best_score - 1e-14:
                    vj = np.clip(pair_sum - best_t, 0.0, 1.0)
                    best[i] = best_t
                    best[j] = vj
                    best = project_feasible(best, n)
                    best_score = score_c5(best)
                    improved = True

        if not improved:
            stall_count += 1
            if stall_count >= 3:
                break
        else:
            stall_count = 0

    return best, best_score
