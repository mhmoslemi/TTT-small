import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_smooth_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(alpha * c_shifted)
    m = exp_c.sum()
    w = exp_c / m
    part1 = np.correlate(1.0 - h, w, mode='full')[2 * n - 2:3 * n - 2][::-1]
    part2 = np.convolve(w, h, mode='full')[n - 1:2 * n - 1]
    grad = dx * (part1 - part2)
    smooth_val = c_max + np.log(m) / alpha
    return smooth_val, grad

def make_even_binary(n, phase=0):
    h = np.zeros(n)
    h[phase::2] = 1.0
    s = h.sum()
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        for i in ones[:s - target]:
            h[i] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        for i in zeros[:target - s]:
            h[i] = 1
    return h

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 9)
    for b in range(n_blocks):
        start = int(n * b / n_blocks)
        length = max(1, n // (2 * n_blocks) + rng.randint(-3, 4))
        end = min(n, start + length)
        h[start:end] = 1.0
    return project_feasible(h, n / 2.0)

def make_fourier_init(n, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 * np.ones(n)
    n_harm = rng.randint(2, 6)
    for _ in range(n_harm):
        f = rng.randint(1, 5)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.05, 0.2)
        h += a * np.sin(2 * np.pi * f * t / 2.0 + p)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

def make_random_binary(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h

def make_resampled(orig_h, n, rng):
    n_orig = len(orig_h)
    t_orig = np.linspace(0, 2, n_orig, endpoint=False)
    t_new = np.linspace(0, 2, n, endpoint=False)
    h_new = np.interp(t_new, t_orig, orig_h, period=2.0)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n / 2.0)

def adam_optimize(h_init, n, deadline, rng, alpha_start, lr_init, max_iter):
    target = n / 2.0
    h = h_init.copy()
    alpha = alpha_start
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    lr_cur = lr_init
    while time.time() < deadline and t < max_iter:
        t += 1
        _, grad = compute_smooth_and_grad(h, alpha)
        m_t = 0.9 * m_t + 0.1 * grad
        v_t = 0.999 * v_t + 0.001 * grad * grad
        m_hat = m_t / (1.0 - 0.9 ** t)
        v_hat = v_t / (1.0 - 0.999 ** t)
        step = lr_cur * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= cur_c5:
            h = h_new
            cur_c5 = c5_new
        else:
            T = 0.002 * (0.997 ** t)
            if T > 1e-10 and rng.random() < np.exp(-(c5_new - cur_c5) / T):
                h = h_new
                cur_c5 = c5_new
        if t % 150 == 0:
            alpha = min(alpha * 1.35, 50000.0)
        if t % 500 == 0:
            lr_cur = max(lr_cur * 0.93, lr_init * 0.05)
    return best_h, best_c5

def slsqp_polish(h_init, n, maxiter=200):
    target = n / 2.0
    dx = 2.0 / n
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def obj(val):
        return val[-1]

    cons = []
    k_max = min(n, 2 * n - 1)
    for k in range(k_max):
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'args': (n, k, dx)})
    cons.append({'type': 'eq', 'fun': _slsqp_eq, 'args': (n, target)})
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_feasible(h_opt, target)
        return h_opt
    except Exception:
        return h_init

def _slsqp_ineq(val, n, k, dx):
    h = val[:n]
    t = val[n]
    m = k - (n - 1)
    c_val = 0.0
    j_min = max(0, -m)
    j_max = min(n, n - m)
    for j in range(j_min, j_max):
        c_val += h[j] * (1.0 - h[j + m])
    return t - c_val * dx

def _slsqp_eq(val, n, target):
    return np.sum(val[:n]) - target
