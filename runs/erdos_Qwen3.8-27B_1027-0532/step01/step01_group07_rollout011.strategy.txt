    def sbx(u, v, eta, rng=None):
        if rng is None:
            rng = np.random.RandomState()
        u = np.array(u)
        v = np.array(v)
        if u.size == 1:
            u = [u]
            v = [v]
        if len(u) != len(v):
            raise ValueError("Vectors must be same length")
        child = np.zeros(len(u))
        for i in range(len(u)):
            u_val = u[i]; v_val = v[i]
            # If both are at boundaries, we keep one
            check = (u_val == 0 and v_val==0) or (u_val==1 and v_val==1) or (u_val==0 and v_val==1) or (u_val==1 and v_val==0)
            if check:
                child[i] = u_val if rng.rand()<0.5 else v_val
            else:
                # we can safely do the crossover
                beta = np.zeros(2)
                if rng.rand() < 0.5:
                    beta = (2 * rng.rand())**(1/(eta+1)) - 1
                else:
                    beta = 1 - (2*(1-rng.rand()))**(1/(eta+1))
                # Clamp beta in [-1,1]
                if beta < -1: beta=-1
                if beta > 1: beta=1
                child[i] = 0.5 * (u_val + v_val) + beta * (u_val - v_val)
        return child

    def mutate_sigma(vector, sigma=0.1, rng=None):
        if rng is None:
            rng = np.random.RandomState()
        u = rng.normal(0, sigma, size=len(vector))
        child = vector + u
        child = np.clip(child, 0.0, 1.0)
        return child
