```python
import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_obj_and_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled - np.max(scaled))
    w = exp_scaled / np.sum(exp_scaled)
    obj = dx * (cmax + T * np.log(np.sum(exp_scaled)))
    
    # Gradient
    indices = np.arange(n)
    lags = np.arange(2*n-1) - (n-1)
    im = indices[:, None] + lags[None, :]
    im2 = indices[:, None] - lags[None, :]
    valid1 = (im >= 0) & (im < n)
    valid2 = (im2 >= 0) & (im2 < n)
    partA = np.where(valid1, 1.0 - x[im.clip(0, n-1)], 0.0)
    partA = np.sum(partA * w[None, :], axis=1)
    partB = np.where(valid2, x[im2.clip(0, n-1)], 0.0)
    partB = np.sum(partB * w[None, :], axis=1)
    grad = dx * (partA - partB)
    return obj, grad

def projected_grad_descent(h, n, T, lr, max_iter, t_start, time_limit):
    best = h.copy()
    best_score = score_c5(best)
    x = h.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        obj, grad = lse_obj_and_grad(x, T)
        
        # Backtracking line search
        step = lr
        x_new = project_feasible(x - step * grad, n)
        s_new = score_c5(x_new)
        
        if s_new < best_score:
            best = x_new.copy()
            best_score = s_new
        
        for _ in range(20):
            if s_new < obj:  # sufficient decrease (use actual score as proxy)
                break
            step *= 0.5
            x_new = project_feasible(x - step * grad, n)
            s_new = score_c5(x_new)
        
        x = x_new
    
    return best, best_score

def binary_2opt(b, n, t_start, time_limit):
    """2-opt local search on binary sequence with exactly n/2 ones."""
    best = b.copy()
    best_score = score_c5(best)
    ones = np.where(best == 1.0)[0]
    zeros = np.where(best == 0.0)[0]
    
    improved = True
    while improved and time.time() - t_start < time_limit:
        improved = False
        # For each 1, try swapping with each 0
        for i in ones:
            for j in zeros:
                if time.time() - t_start > time_limit:
                    return best, best_score
                trial = best.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                s = score_c5(trial)
                if s < best_score - 1e-12:
                    best = trial
                    best_score = s
                    improved = True
                    ones = np.where(best == 1.0)[0]
                    zeros = np.where(best == 0.0)[0]
                    break
            if improved:
                break
    
    return best, best_score
```

Wait, let me redo this.

corr[m] = sum_{valid i'} b[i'] * (1-b[i'+m])

When b[i] changes from 1 to 0:
- Terms where i' = i: was b[i]*(1-b[i+m]) = 1*(1-b[i+m]), now 0. Change: -(1-b[i+m]) = b[i+m]-1
- Terms where i'+m = i, i.e., i' = i-m: was b[i-m]*(1-b[i]) = b[i-m]*1, now b[i-m]*0. Change: -b[i-m]
So total change from b[i]: 1→0 is: (b[i+m]-1) - b[i-m] = b[i+m] - b[i-m] - 1

When b[j] changes from 0 to 1:
- Terms where i' = j: was 0, now 1*(1-b[j+m]). Change: (1-b[j+m])
- Terms where i'+m = j, i.e., i' = j-m: was b[j-m]*(1-0) = b[j-m], now b[j-m]*(1-1) = 0. Change: -b[j-m]
So total change from b[j]: 0→1 is: (1-b[j+m]) - b[j-m]

Total delta[m] = (b[i+m] - b[i-m] - 1) + (1-b[j+m] - b[j-m])
= b[i+m] - b[i-m] - b[j+m] - b[j-m]

(without boundary conditions; out-of-range terms are 0)

So for each pair (i,j), I can compute delta[m] for all m in O(n), then check if max(corr + delta) < current_max. That's O(n) per pair, O(n²) pairs, O(n³) total per pass. For n=85: 85³ ≈ 614,000. Much better!

```python
def binary_swap_delta(b, i, j, n):
    """Compute delta[m] for swapping b[i]: 1->0, b[j]: 0->1."""
    lags = np.arange(2*n-1) - (n-1)
    delta = np.zeros(2*n-1)
    for k, m in enumerate(lags):
        d = 0.0
        ip = i + m
        im = i - m
        jp = j + m
        jm = j - m
        if 0 <= ip < n:
            d += b[ip]
        if 0 <= im < n:
            d -= b[im]
        d -= 1.0
        if 0 <= jp < n:
            d -= b[jp]
        if 0 <= jm < n:
            d -= b[jm]
        delta[k] = d
    return delta
```

```python
def binary_swap_delta_vec(b, i, j, n):
    lags = np.arange(2*n-1) - (n-1)
    # i+m, i-m, j+m, j-m
    ip = i + lags
    im = i - lags
    jp = j + lags
    jm = j - lags
    
    b_ip = np.where((ip >= 0) & (ip < n), b[np.clip(ip, 0, n-1)], 0.0)
    b_im = np.where((im >= 0) & (im < n), b[np.clip(im, 0, n-1)], 0.0)
    b_jp = np.where((jp >= 0) & (jp < n), b[np.clip(jp, 0, n-1)], 0.0)
    b_jm = np.where((jm >= 0) & (jm < n), b[np.clip(jm, 0, n-1)], 0.0)
    
    delta = b_ip - b_im - 1.0 - b_jp - b_jm
    return delta
```

```python
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_obj_and_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled - np.max(scaled))
    w = exp_scaled / np.sum(exp_scaled)
    obj = dx * (cmax + T * np.log(np.sum(exp_scaled)))
    
    indices = np.arange(n)
    lags = np.arange(2*n-1) - (n-1)
    im = indices[:, None] + lags[None, :]
    im2 = indices[:, None] - lags[None, :]
    valid1 = (im >= 0) & (im < n)
    valid2 = (im2 >= 0) & (im2 < n)
    xc = np.clip(im, 0, n-1)
    xc2 = np.clip(im2, 0, n-1)
    partA = np.where(valid1, 1.0 - x[xc], 0.0)
    partA = np.sum(partA * w[None, :], axis=1)
    partB = np.where(valid2, x[xc2], 0.0)
    partB = np.sum(partB * w[None, :], axis=1)
    grad = dx * (partA - partB)
    return obj, grad

def grad_descent(h, n, T, n_iter, t_start, time_limit):
    best = h.copy()
    best_score = score_c5(best)
    x = h.copy()
    lr = 1.0
    
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        obj, grad = lse_obj_and_grad(x, T)
        
        # Try the step
        step = lr
        for _ in range(10):
            x_new = project_feasible(x - step * grad, n)
            s_new = score_c5(x_new)
            if s_new < best_score or step < 1e-10:
                break
            step *= 0.5
        
        x = x_new
        s_new = score_c5(x)
        if s_new < best_score:
            best = x.copy()
            best_score = s_new
        # Adaptive lr
        if s_new < best_score + 1e-10:
            lr = min(lr * 1.1, 5.0)
        else:
            lr = max(lr * 0.8, 0.01)
    
    return best, best_score

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    b = (h > 0.5).astype(float)
    s = np.sum(b)
    target = n / 2.0
    if s > target:
        # Remove excess ones
        excess = int(s - target)
        ones = np.where(b == 1.0)[0]
        # Remove the ones with lowest h values (least "committed" to being 1)
        order = np.argsort(h[ones])
        for idx in order[:excess]:
            b[ones[idx]] = 0.0
    elif s < target:
        deficit = int(target - s)
        zeros = np.where(b == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:deficit]:
            b[zeros[idx]] = 1.0
    return b

def binary_2opt(b, n, t_start, time_limit):
    """Efficient 2-opt for binary sequences."""
    best = b.astype(float).copy()
    dx = 2.0 / n
    best_corr = np.correlate(best, 1.0 - best, mode='full').astype(float)
    best_max = np.max(best_corr)
    
    lags = np.arange(2*n-1) - (n-1)
    
    passes = 0
    max_passes = 50
    while time.time() - t_start < time_limit and passes < max_passes:

        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]

                im = i - lags
                jp = j + lags
                jm = j - lags

                b_im = np.where((im >= 0) & (im < n), best[np.clip(im, 0, n-1)], 0.0)
                b_jp = np.where((jp >= 0) & (jp < n), best[np.clip(jp, 0, n-1)], 0.0)
                b_jm = np.where((jm >= 0) & (jm < n), best[np.clip(jm, 0, n-1)], 0.0)

def interpolate_to_n(h, n_old, n_new):

    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
```

When best[i] changes from 1 to 0:
- For the term where i is the "source": was best[i]*(1-best[i+m]) = 1*(1-best[i+m]), now 0. Change: -(1-best[i+m]) = best[i+m]-1

```python
def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_obj_and_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    exp_scaled = np.exp(scaled - np.max(scaled))
    w = exp_scaled / np.sum(exp_scaled)
    obj = dx * (cmax + T * np.log(np.sum(exp_scaled)))
    indices = np.arange(n)
    lags = np.arange(2*n-1) - (n-1)
    im = indices[:, None] + lags[None, :]
    im2 = indices[:, None] - lags[None, :]
    valid1 = (im >= 0) & (im < n)
    valid2 = (im2 >= 0) & (im2 < n)
    xc = np.clip(im, 0, n-1)
    xc2 = np.clip(im2, 0, n-1)
    partA = np.where(valid1, 1.0 - x[xc], 0.0)
    partA = np.sum(partA * w[None, :], axis=1)
    partB = np.where(valid2, x[xc2], 0.0)
    partB = np.sum(partB * w[None, :], axis=1)
    grad = dx * (partA - partB)
    return obj, grad

def grad_descent(h, n, T, n_iter, t_start, time_limit):
    best = h.copy()
    best_score = score_c5(best)
    x = h.copy()
    lr = 1.0
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        obj, grad = lse_obj_and_grad(x, T)
        step = lr
        x_new = x  # fallback
        s_new = best_score  # fallback
        for _ in range(15):
            x_candidate = project_feasible(x - step * grad, n)
            s_candidate = score_c5(x_candidate)
            if s_candidate <= obj + 1e-10 or step < 1e-15:
                x_new = x_candidate
                s_new = s_candidate
                break

        x = x_new
        if s_new < best_score:
            best = x.copy()
            best_score = s_new
            lr = min(lr * 1.05, 10.0)
        else:
            lr = max(lr * 0.9, 0.001)
    return best, best_score

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    s = int(np.sum(b))
    target = n // 2
    if s > target:
        excess = s - target
        ones = np.where(b == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in order[:excess]:
            b[ones[idx]] = 0.0
    elif s < target:
        deficit = target - s
        zeros = np.where(b == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:deficit]:
            b[zeros[idx]] = 1.0
    return b
```
