The projection onto {0≤h≤1, sum(h)=n/2}:
- Given a point y, find the closest point in the feasible set
- This is: minimize ||h-y||^2 subject to 0≤h≤1, sum(h)=n/2
- Solution: h = clip(y - λ, 0, 1) for the right λ
- Find λ by bisection (since sum(clip(y-λ, 0, 1)) is decreasing in λ)

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def project_feasible(y, n):
    target = n * 0.5
    lo, hi = -2.0, 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += residual / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, dx, beta):
    C = np.correlate(h, 1.0 - h, mode="full")
    bC = beta * dx * C
    m = np.max(bC)
    exp_bC = np.exp(bC - m)
    sum_exp = np.sum(exp_bC)
    p = exp_bC / sum_exp
    n1 = n - 1
    j_idx = np.arange(n)
    # Use broadcasting to compute indices
    # idx1[j,i] = j - i + n1, idx2[j,i] = i - j + n1
    # Since 0<=j,i<n, we have 0 <= j-i+n1 <= 2n-2, always valid
    idx = np.subtract.outer(j_idx, j_idx) + n1  # idx1: shape (n,n), idx[j,i] = j-i+n1
    p1 = p[idx]
    p2 = p[idx.T]  # p2[j,i] = p[i-j+n1] = p1[i,j] = p[idx[i,j]]
    # Wait: p2[j,i] should be p[i-j+n1]. idx[i,j] = i-j+n1. So p2[j,i] = p[idx[i,j]] = p[idx.T[j,i]]
    # Yes! p2 = p1.T in terms of the p values, but not exactly because p1[j,i] = p[j-i+n1] and p2[j,i] = p[i-j+n1].
    # p1.T[j,i] = p1[i,j] = p[i-j+n1] = p2[j,i]. Yes!
    
    T1 = p1 @ (1.0 - h)
    T2 = p2 @ h
    grad = dx * (T1 - T2)
    smooth_val = (m + np.log(sum_exp)) / beta
    return smooth_val, grad

def projected_gradient_step(h, grad, step_size, n):
    y = h - step_size * grad
    return project_feasible(y, n)

def barzilai_borwein(h_prev, h_curr, grad_prev, grad_curr):
    """Compute BB step size."""
    s = h_curr - h_prev
    y = grad_curr - grad_prev
    sTs = np.dot(s, s)
    if sTs < 1e-30:
        return 1.0
    sTy = np.dot(s, y)
    # BB1: sTy / sTs (for minimization)
    # BB2: sTs / yTy
    yTy = np.dot(y, y)
    if yTy < 1e-30:
        return sTy / sTs if sTy > 0 else 1.0
    # Alternate between BB1 and BB2
    # For safety, clip to reasonable range
    bb1 = np.clip(sTy / sTs, 1e-6, 10.0) if sTy > 0 else 1.0
    bb2 = np.clip(sTs / yTy, 1e-6, 10.0)
    return bb1  # Use BB1 primarily

def run_pgdescent(h0, n, dx, beta, max_iter, time_limit):
    """Projected gradient descent with BB steps and backtracking."""
    t_start = time.time()
    h_curr = h0.copy()
    f_curr, g_curr = smooth_max_and_grad(h_curr, n, dx, beta)
    best_h = h_curr.copy()
    best_f = f_curr
    best_real = c5_value(h_curr, dx)
    
    h_prev = h_curr.copy()
    g_prev = g_curr.copy()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        # BB step size
        step = barzilai_borwein(h_prev, h_curr, g_prev, g_curr)
        
        # Backtracking line search
        alpha = step
        improved = False
        for bt in range(20):
            h_new = projected_gradient_step(h_curr, g_curr, alpha, n)
            f_new, g_new = smooth_max_and_grad(h_new, n, dx, beta)
            if f_new < f_curr - 1e-12 * np.abs(f_curr):
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try smaller beta
            break
        
        h_prev = h_curr
        g_prev = g_curr
        h_curr = h_new
        f_curr = f_new
        g_curr = g_new
        
        real_f = c5_value(h_curr, dx)
        if real_f < best_real:
            best_real = real_f
            best_h = h_curr.copy()
    
    return best_h, best_real

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def generate_fourier_h(n, freqs, amplitudes, phases, rng):
    x = np.arange(n)
    h = np.full(n, 0.5)
    for f, a, ph in zip(freqs, amplitudes, phases):
        h += a * np.sin(2.0 * np.pi * f * x / n + ph)
    return repair_h(h, n)

    np.random.shuffle(block_sizes)  # wait, need to use rng
```
