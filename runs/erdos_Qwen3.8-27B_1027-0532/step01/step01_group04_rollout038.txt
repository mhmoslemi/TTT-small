def project_sum_box(y, target_sum):
    """Project y onto {0 <= x <= 1, sum(x) = target_sum}."""
    n = len(y)
    # Water filling
    lo, hi = -2.0, 2.0  # τ is in this range
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Fix any numerical issues
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = (x > 0) & (x < 1)
        else:
            mask = (x > 0) & (x < 1)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
    return np.clip(x, 0.0, 1.0)

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_sum_box(y, target_sum):
    n = len(y)

    lo = y.min() - target_sum / n - 2.0
    hi = y.max() - target_sum / n + 2.0

    n = len(h)
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)

    best_h = h.copy()
    best_obj = compute_c5(h)
    for t in range(1, n_steps + 1):
        if time.time() - start_time > budget_s - 30:
            break
        obj, grad = smooth_obj_grad(h, alpha)
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * grad ** 2
        m_hat = m_arr / (1 - b1 ** t)
        v_hat = v_arr / (1 - b2 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - update
        h_new = project_sum_box(h_new, target_sum)
        new_obj = compute_c5(h_new)
        if new_obj < best_obj:
            best_obj = new_obj
            best_h = h_new.copy()
        h = h_new
    return best_h, best_obj

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum_box(y, target_sum):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
        else:
            x += diff / n
    return np.clip(x, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = alpha * (corr - max_c)

    sum_exp = exp_vals.sum()
    obj = dx * (max_c + np.log(sum_exp) / alpha)
    w = exp_vals / sum_exp
    grad = np.zeros(n)

            length = n - kp
            grad[:length] -= wm * h[kp:]
            grad[kp:] += wm * (1.0 - h[:length])
    grad *= dx
