```python
def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h
```

```python
def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * (2.0 / n)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    zeros = np.where(h_b == 0.0)[0]
```

```python
def _steepest_local_search(h_b, n, target_sum, max_iter=1000):
    """Steepest descent local search on 0-1 solutions."""
    dx = 2.0 / n
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    
    current_c5 = _c5_of(h_b)
    
    for it in range(max_iter):
        best_delta = 0
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                h_b[i] = 0.0
                h_b[j] = 1.0
                c5 = _c5_of(h_b)
                delta = c5 - current_c5
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
                h_b[i] = 1.0
                h_b[j] = 0.0
        
        if best_i == -1:
            break  # Local optimum
        
        h_b[best_i] = 0.0
        h_b[best_j] = 1.0
        current_c5 += best_delta
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
    
    return h_b, current_c5
```

```python
    if m < -500:
        return m * (2.0 / len(x))
    return (m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha) * (2.0 / len(x))

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha)

def _do_slsqp(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _sum_constraint(x):
    return x.sum() - x.size / 2.0

def _sum_jac(x):
    return np.ones(x.size)

def _subgradient_step(h, n, alpha):
    """One step of the subgradient method with optimal subgradient."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    # Active set
    threshold = max_val - 1e-8
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        # Compute gradient of g_lag w.r.t. h
        # g_lag = sum_i h[i]*(1-h[i+lag]) for valid i
        for i in range(n):
            val = 0.0
            j = i + lag
            if 0 <= j < n:
                val += (1.0 - h[j])
            j2 = i - lag
            if 0 <= j2 < n:
                val += (-h[j2])
            grad[i] += val
    
    grad *= (2.0 / n) / len(active)
    return grad, max_val * (2.0 / n)
```

```python
        grad[L:] += (-h[:n-L])
    else:
        # m ranges from -L to n-1
        grad[-L:] += (1.0 - h[:n+L])
        # m ranges from 0 to n+L-1
        grad[:n+L] += (-h[-L:])
    return grad
```

def _c5_of(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * (2.0 / n)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

            grad[L:] += (-h[:n-L])

            grad[:n+L] += (-h[-L:])
    return grad
