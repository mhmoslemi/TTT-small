    # c[m] in full mode corresponds to lag (m - (n-1))

    # = (1-h[i-k]) * I[0≤i-k<n] - h[i+k] * I[0≤i+k<n]

        if weight < 1e-15:
            continue

        # For term (1-h[i-k]): valid when 0≤i-k<n, i.e., k≤i<n+k
        # For term -h[i+k]: valid when 0≤i+k<n, i.e., -k≤i<n-k

            for i in range(k, n):
                grad[i] += weight * (1.0 - h[i-k])
            # -h[i+k]: i from 0 to n-1-k
            for i in range(0, n-k):

        else:

            # (1-h[i+ kpos]): i from 0 to n-1-kpos
            for i in range(0, n-kpos):
                grad[i] += weight * (1.0 - h[i+kpos])
            # -h[i-kpos]: i from kpos to n-1
            for i in range(kpos, n):
                grad[i] += weight * (-h[i-kpos])

            h_opt, c5_opt = gd_optimize(child, n, gc, 20, 400, phase2_deadline, lr_init=0.04)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                no_improve_gens = 0
            else:
                no_improve_gens += 1

            pop_h.append(h_opt)
            pop_f.append(c5_opt)

        # Truncate population
        if len(pop_h) > pop_size + 15:
            order = np.argsort(pop_f)
            keep = order[:pop_size + 15]
            pop_h = [pop_h[i] for i in keep]
            pop_f = [pop_f[i] for i in keep]

        # Ensure best is in population
        if best_h is not None:
            if best_c5 < pop_f[0] + 1e-8:
                pop_h[0] = best_h
                pop_f[0] = best_c5

        # If stuck, try bigger jumps
        if no_improve_gens > 6:
            for _ in range(2):
                if time.time() > phase2_deadline:
                    break
                h_jump = best_h + rng.choice([-1, 0, 1], n) * 0.3
                h_jump = np.clip(h_jump, 0, 1)
                h_jump = project_feasible(h_jump, target)
                h_opt, c5_opt = gd_optimize(h_jump, n, gc, 20, 300, phase2_deadline, lr_init=0.05)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    no_improve_gens = 0
                pop_h.append(h_opt)
                pop_f.append(c5_opt)

    # Phase 3: Intensive polish
    phase3_deadline = deadline - 20
    if best_h is not None and time.time() < phase3_deadline:
        h_polish = best_h.copy()
        for alpha in [50, 100, 300, 500, 1000, 2000]:
            if time.time() > phase3_deadline:
                break
            h_polish, c5_p = gd_optimize(h_polish, n, gc, alpha, 1500, phase3_deadline, lr_init=0.03)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_polish.copy()

        # Fine perturbations
        for _ in range(15):
            if time.time() > phase3_deadline:
                break
            h_pert = best_h + rng.normal(0, 0.015, n)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = project_feasible(h_pert, target)
            h_opt, c5_opt = gd_optimize(h_pert, n, gc, 200, 800, phase3_deadline, lr_init=0.02)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Binary perturbation on a few positions
        for _ in range(20):
            if time.time() > phase3_deadline:
                break
            h_bin = best_h.copy()
            n_flip = rng.randint(1, 4)
            flip_idx = rng.choice(n, n_flip, replace=False)
            h_bin[flip_idx] = 1.0 - h_bin[flip_idx]
            h_bin = project_feasible(h_bin, target)
            h_opt, c5_opt = gd_optimize(h_bin, n, gc, 50, 600, phase3_deadline, lr_init=0.03)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final evaluation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    return best_h, best_c5, n
