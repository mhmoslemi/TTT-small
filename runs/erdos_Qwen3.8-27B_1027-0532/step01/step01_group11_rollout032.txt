import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0

    return float(dx * np.max(corr))

def smooth_objective(h, eps):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = ex.sum()

    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

        if time.time() - start_time > time_limit - 30:
            break
        h = best_h + 0.05 * rng.standard_normal(n)
        h = project_onto_constraint(h, target_sum)
        T = 0.02
        T_min = 0.0001
        cooling = 0.995
        cur_c5 = compute_c5(h)
        cur_best = h.copy()
        cur_best_c5 = cur_c5
        for step in range(5000):
            if time.time() - start_time > time_limit - 30:
                break
            idx = rng.integers(0, n)
            delta = (rng.random() - 0.5) * 0.1
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
            h_new = project_onto_constraint(h_new, target_sum)
            new_c5 = compute_c5(h_new)
            if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                h = h_new
                cur_c5 = new_c5
                if new_c5 < cur_best_c5:
                    cur_best_c5 = new_c5
                    cur_best = h.copy()
            T = max(T * cooling, T_min)
        if cur_best_c5 < best_c5:
            best_c5 = cur_best_c5
            best_h = cur_best
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 90
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    h0 = project_onto_constraint(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Fine SLSQP optimization at n0
    eps_fine = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    for eps in eps_fine:
        remaining = budget_s - (time.time() - start_time) - 120
        if remaining < 20:
            break
        h_opt, c5_opt = optimize_slsqp(best_h, eps, 400, remaining, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: PGD refinement at n0
    remaining = budget_s - (time.time() - start_time) - 100
    if remaining > 30:
        h_opt, c5_opt = pgd_refine(best_h, n0 / 2.0, 1e-5, 0.001, 3000, start_time,
                                     start_time + remaining - 20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Random restarts at n0
    remaining = budget_s - (time.time() - start_time) - 100
    n_restarts = 0
    while remaining > 80 and n_restarts < 15:
        strategy = n_restarts % 5
        if strategy == 0:
            h_pert = best_h + 0.03 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_h + 0.08 * (rng.random(n0) - 0.5)
        elif strategy == 2:
            h_bin = (rng.random(n0) < 0.5).astype(float)
            h_bin = project_onto_constraint(h_bin, n0 / 2.0)
            alpha = 0.6 + 0.3 * rng.random()
            h_pert = alpha * best_h + (1.0 - alpha) * h_bin
        elif strategy == 3:
            h_pert = best_h.copy()
            nflip = max(1, n0 // 8)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            h_pert = best_h.copy()
            shift = rng.integers(1, n0 // 2)
            h_pert = np.roll(best_h, shift)
            h_pert = project_onto_constraint(h_pert, n0 / 2.0)

        h_pert = project_onto_constraint(h_pert, n0 / 2.0)
        tl = min(remaining - 40, 80)
        h_opt, c5_opt = optimize_slsqp(h_pert, 0.001, 200, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        n_restarts += 1
        remaining = budget_s - (time.time() - start_time) - 100

    # Phase 4: Simulated annealing
    remaining = budget_s - (time.time() - start_time) - 80
    if remaining > 60:
        h_sa, c5_sa = simulated_annealing(best_h, n0 / 2.0, start_time,
                                           start_time + remaining - 40, rng, n_restarts=3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        remaining = budget_s - (time.time() - start_time) - 80

    # Phase 5: Upsample to 2*n0 and optimize
    remaining = budget_s - (time.time() - start_time) - 40
    if remaining > 120:
        n1 = 2 * n0
        h_up = upsample(best_h, n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        eps_up = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        for eps in eps_up:
            remaining = budget_s - (time.time() - start_time) - 40
            if remaining < 30:
                break
            h_opt, c5_opt = optimize_slsqp(h_up, eps, 300, remaining, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n1 = len(best_h)
            else:
                h_up = h_opt

        # PGD at higher resolution
        remaining = budget_s - (time.time() - start_time) - 40
        if remaining > 60:
            h_opt, c5_opt = pgd_refine(best_h, len(best_h) / 2.0, 5e-5, 0.0005,
                                        2000, start_time, start_time + remaining - 20)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 6: Final restarts at highest resolution
    n_final = len(best_h)
    remaining = budget_s - (time.time() - start_time) - 30
    n_final_restarts = 0
    while remaining > 50 and n_final_restarts < 8:
        if n_final_restarts % 2 == 0:
            h_pert = best_h + 0.02 * rng.standard_normal(n_final)
        else:
            h_pert = best_h.copy()
            nflip = max(1, n_final // 10)
            idx = rng.choice(n_final, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = project_onto_constraint(h_pert, n_final / 2.0)
        tl = min(remaining - 20, 60)
        h_opt, c5_opt = optimize_slsqp(h_pert, 1e-5, 200, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        n_final_restarts += 1
        remaining = budget_s - (time.time() - start_time) - 30

    # Phase 7: Final polish
    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 30:
        h_polish = best_h + 0.005 * rng.standard_normal(len(best_h))
        h_polish = project_onto_constraint(h_polish, len(best_h) / 2.0)
        h_opt, c5_opt = optimize_slsqp(h_polish, 5e-6, 300, remaining - 5, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final validation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
