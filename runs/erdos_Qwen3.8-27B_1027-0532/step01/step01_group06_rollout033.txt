from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def gradient_at_lag(h, k_lag, n):
    """Gradient of corr[k_lag] w.r.t. h."""
    shift = k_lag - n + 1
    g = np.zeros(n)
    n_start = max(0, -shift)
    n_end = min(n, n - shift)
    if n_end <= n_start:
        return g
    if shift == 0:
        g[n_start:n_end] = 1.0 - 2.0 * h[n_start:n_end]
    else:
        # First term: j = i + shift, i in [n_start, n_end)
        j_start = n_start + shift
        j_end = n_end + shift
        # Clip to [0, n)
        j_start_c = max(0, j_start)
        j_end_c = min(n, j_end)
        i_start_c = j_start_c - shift
        i_end_c = j_end_c - shift
        g[j_start_c:j_end_c] += (1.0 - h[i_start_c:i_end_c])
        
        # Second term: j = i, i in [n_start, n_end)
        g[n_start:n_end] -= h[n_start + shift:n_end + shift]
    return g

def multi_lag_gradient(h, corr, n, top_m=5, temperature=0.1):
    """Weighted gradient using top m lags."""
    # Use softmax weighting over top m lags

    weights = weights / weights.sum()
    g = np.zeros(n)

        g += w * gradient_at_lag(h, k, n)
    return g

        constraints = [cp.Variable(n) >= 0, cp.Variable(n) <= 1, cp.sum(cp.Variable(n)) == n/2]

import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_at_lag(h, k_lag, n):
    shift = k_lag - n + 1
    g = np.zeros(n)
    n_start = max(0, -shift)
    n_end = min(n, n - shift)
    if n_end <= n_start:
        return g
    if shift == 0:
        g[n_start:n_end] = 1.0 - 2.0 * h[n_start:n_end]
    else:
        j_start = max(0, n_start + shift)
        j_end = min(n, n_end + shift)
        g[j_start:j_end] += (1.0 - h[j_start - shift:j_end - shift])
        g[n_start:n_end] -= h[n_start + shift:n_end + shift]
    return g

def projected_gradient_descent(h, n, t_end, top_m=5, use_multi=True):
    best_h = h.copy()
    best_score = score_c5(best_h)
    alpha = 0.05
    
    while time.time() < t_end:
        corr = compute_corr(best_h)
        
        if use_multi and top_m > 1:
            # Use top m lags with softmax weighting
            idx = np.argsort(corr)[-top_m:]
            temps = 0.05
            weights = np.exp((corr[idx] - corr[idx[-1]]) / temps)
            weights /= weights.sum()
            g = np.zeros(n)
            for w, k in zip(weights, idx):
                g += w * gradient_at_lag(best_h, int(k), n)
        else:
            k_max = np.argmax(corr)
            g = gradient_at_lag(best_h, k_max, n)
        
        g_norm = np.linalg.norm(g)

            break
        
        # Backtracking line search
        improved = False

            h_new = project_feasible(best_h - alpha * g, n)
            s_new = score_c5(h_new)
            if s_new < best_score - 1e-15:
                best_h = h_new

                    best_h = h_new
                    best_score = s_new
            alpha = 0.05
