def project_onto_constraint(h, target_sum):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def grad_of_corr_m(h, m):
    """Gradient of (dx * corr[m]) with respect to h (full correlation index m)"""
    n = len(h)
    dx = 2.0 / n
    k = m - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += dx * (1.0 - h[:n-k])
        grad[:n-k] -= dx * h[k:]
    else:
        r = -k
        grad[:n-r] += dx * (1.0 - h[r:])
        grad[r:] -= dx * h[:n-r]
    return grad

def subgradient_c5(h, n_active=5, temp=0.005):
    """Compute subgradient of C5 using top-n_active correlation lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top indices
    top_idx = np.argsort(corr)[-n_active:][::-1]
    top_vals = corr[top_idx]
    # Softmax weights
    w = np.exp((top_vals - top_vals[0]) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wt in zip(top_idx, w):
        grad += wt * grad_of_corr_m(h, idx)
    return grad, float(corr.max())

    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    y = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.9
    
    for step in range(n_steps):
        # Nesterov: evaluate gradient at y = h + momentum * (h - h_prev)
        # Simplified: just use momentum on the step
        grad, c5_val = subgradient_c5(y, n_active, temp)

        lr = base_lr / (1.0 + step * 0.01)
        
        # Gradient step
        h_new = project_onto_constraint(y - lr * grad, target_sum)
        
        # Check improvement, backtrack if needed
        c5_new = compute_c5(h_new)
        if c5_new > c5_val * 1.01 + 1e-10:
            # Halve lr
            for _ in range(5):
                lr *= 0.5
                h_new = project_onto_constraint(y - lr * grad, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new <= c5_val + 1e-10:

        y = project_onto_constraint(y, target_sum)
        h = h_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

We have C5 = 1/2 - min_k R_g(k) where g = h - 1/2.

To minimize C5, we need to maximize min_k R_g(k).

The autocorrelation R_g(k) = ∫g(x)g(x+k)dx (linear, not circular).

For the step function with n points, this is:
R_g(k) = dx * sum_{valid i} g[i] * g[i+k] (for k ≥ 0)
R_g(k) = dx * sum_{valid i} g[i] * g[i-k] (for k < 0, by symmetry R_g(-k) = R_g(k))

Wait, is R_g symmetric? R_g(k) = ∫g(x)g(x+k)dx. R_g(-k) = ∫g(x)g(x-k)dx = ∫g(x+k)g(x)dx = R_g(k). Yes!

So we only need to consider k ≥ 0.

R_g(0) = ∫g(x)²dx = sum(g²) * dx (this is always ≥ 0)
R_g(k) for k > 0: this can be positive or negative.

We want min_k R_g(k) to be as large as possible. Since R_g(0) = ∫g² ≥ 0, the minimum will be at some k > 0 (or -k).

For the minimum to be positive, we need ALL autocorrelation values (at all non-zero lags) to be positive. This means g should be "positively autocorrelated" at all lags, which means g should be "smooth" (low-frequency).

A smooth function like g(x) = sin(πx) (one period on [0,2]) would have positive autocorrelation at most lags. Let's check:
g(x) = sin(πx/2) (a half-sine on [0,2]... no, let's use g(x) = cos(πx) on [0,2])

R_g(k) = ∫₀^{2-k} cos(πx)cos(π(x+k))dx = ∫₀^{2-k} cos(πx)cos(πx+πk)dx
= 1/2 ∫₀^{2-k} [cos(2πx+πk) + cos(πk)]dx
= 1/2 [sin(2π(2-k)+πk)/(2π) - sin(πk)/(2π) + (2-k)cos(πk)/2]
= 1/2 [(2-k)cos(πk)/2 + (sin(2π(2-k)+πk) - sin(πk))/(2π)]
= (2-k)cos(πk)/4 + [sin(4π-2πk+πk) - sin(πk)]/(4π)

= (2-k)cos(πk)/4 - sin(πk)/(2π)

For k=1: R_g(1) = cos(π)/4 - sin(π)/(2π) = -1/4. Negative!

def corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr_m(h, m):
    n = len(h)
    dx = 2.0 / n
    k = m - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += dx * (1.0 - h[:n - k])
            grad[:n - k] -= dx * h[k:]
    else:
        r = -k
        if r < n:
            grad[:n - r] += dx * (1.0 - h[r:])
            grad[r:] -= dx * h[:n - r]
    return grad

def subgradient_c5(h, tol=0.001):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    active = np.where(c >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for m in active:
        grad += grad_corr_m(h, int(m))
    grad /= len(active)
    return grad

def pgd_optimize(h_init, time_limit, start_time, start_lr=3.0, mu=0.9):
    n = len(h_init)
    target = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    tol_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    tol_idx = 0

    for it in range(10000000):
        if time.time() - start_time > time_limit:
            break
        if it % 3000 == 0 and tol_idx < len(tol_schedule) - 1:
            tol_idx += 1

        grad = subgradient_c5(h, tol=tol_schedule[tol_idx])
        lr = start_lr / (1.0 + 0.001 * it) ** 0.5

        v = mu * v - lr * grad
        h_new = project_onto_constraint(h + v, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            v = np.zeros(n)
            no_improve = 0

        h = h_new

    return best_h, best_c5

def sm_objective(h, eps):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    x = (c - cmax) / eps
    x -= x.max()
    return cmax + eps * np.log(np.exp(x).sum())

def sm_gradient(h, eps):
    n = len(h)
    c = corr_full(h)
    cmax = c.max()
    x = (c - cmax) / eps
    x -= x.max()
    ex = np.exp(x)
    w = ex / ex.sum()
    threshold = 0.005 / len(c)
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] > threshold:
            grad += w[m] * grad_corr_m(h, m)
    return grad

def slqp_polish(h_init, time_limit, eps=0.003):
    n = len(h_init)
    target = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target)

    def obj(h):
        return sm_objective(h, eps)

    def grad(h):
        return sm_gradient(h, eps)

    def con_fun(h):
        return np.sum(h) - target

    def con_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        t_start = time.time()
        maxiter = max(50, int((time_limit - 5) / 0.5))
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': min(maxiter, 1000), 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 90
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0, 1)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target = n0 / 2.0
    h0 = project_onto_constraint(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    remaining = budget_s - (time.time() - start) - 20
    t1 = min(remaining * 0.25, 200)
    h1, c5_1 = pgd_optimize(h0, t1, time.time(), start_lr=4.0, mu=0.9)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    remaining = budget_s - (time.time() - start) - 20
    if remaining > 60:
        t2 = min(remaining * 0.2, 180)
        h2, c5_2 = slqp_polish(best_h, t2, eps=0.003)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    remaining = budget_s - (time.time() - start) - 20
    n_restarts = 0
    while remaining > 70 and n_restarts < 30:
        strategy = n_restarts % 5
        if strategy == 0:
            h_p = best_h + 0.02 * rng.standard_normal(n0)
        elif strategy == 1:
            h_p = best_h + 0.04 * (rng.random(n0) - 0.5)
        elif strategy == 2:
            x = np.arange(n0) * 2.0 / n0
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            pert = 0.04 * np.cos(freq * x + phase)
            h_p = best_h + pert
        elif strategy == 3:
            h_p = best_h.copy()
            n_flip = max(2, n0 // 20)
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            h_bin = (rng.random(n0) < 0.5).astype(float)
            h_bin = project_onto_constraint(h_bin, target)
            alpha = 0.8 + 0.1 * rng.random()
            h_p = alpha * best_h + (1.0 - alpha) * h_bin

        h_p = project_onto_constraint(h_p, target)
        t_attempt = remaining - 20
        t_pgd = min(t_attempt * 0.45, 50)
        hp, c5p = pgd_optimize(h_p, t_pgd, time.time(), start_lr=2.5, mu=0.85)

        t_slqp_rem = t_attempt - (time.time() - (time.time() - t_pgd))
        t_slqp_rem = t_attempt - t_pgd - (time.time() - (start + budget_s - remaining + 20))
        elapsed_in_attempt = time.time() - (start + budget_s - remaining - 20 + 20)
        t_slqp_rem = t_attempt - (time.time() - (time.time() - t_pgd)) - 5

        t_rem_now = budget_s - (time.time() - start) - 20
        t_slqp_rem = min(t_rem_now - 10, 50)
        if t_slqp_rem > 15:
            hs, c5s = slqp_polish(hp, t_slqp_rem, eps=0.001)
            if c5s < c5p:
                hp, c5p = hs, c5s

        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

        n_restarts += 1
        remaining = budget_s - (time.time() - start) - 20

    remaining = budget_s - (time.time() - start) - 5
    if remaining > 30:
        h_final, c5_final = slqp_polish(best_h, remaining, eps=0.001)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    remaining = budget_s - (time.time() - start) - 2
    if remaining > 20:
        h_polish, c5_polish = pgd_optimize(best_h, remaining, time.time(), start_lr=1.0, mu=0.8)
        if c5_polish < best_c5:
            best_h, best_c5 = h_polish, c5_polish

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
