    # Compute all R(k)

            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

        )

            c5 = evaluate_c5(r.x)
            return r.x, c5

        pass
    return h, evaluate_c5(h)

def subgradient_ascent(h, n, n_iters, alpha_init, alpha_decay, momentum=0.9):
    """Subgradient ascent to maximize min_k R(k)"""
    h_curr = h.copy()
    velocity = np.zeros(n)
    alpha = alpha_init
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    
    for it in range(n_iters):
        R = compute_R(h_curr)
        k_star = np.argmin(R)
        
        if k_star == 0:
            grad = 2.0 * h_curr
        else:
            grad = np.zeros(n)
            grad[:n-k_star] += h_curr[k_star:]
            grad[k_star:] += h_curr[:n-k_star]
        
        velocity = momentum * velocity + grad
        h_new = h_curr + alpha * velocity
        h_new = repair_h(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_curr = h_new
        alpha *= alpha_decay
    
    return best_h, best_c5

def simulated_annealing(h, n, t_start, t_end, n_iters, rng):

    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)

def smoothed_subgradient(h, n, beta=20.0):
    """Compute smoothed subgradient using softmax weighting of R(k) values"""
    R = compute_R(h)
    # Use the minimum few values with softmax weighting
    # Lower R(k) means larger weight (they're the "worst")
    weights = np.exp(-beta * R)  # higher weight for lower R

            h7, c5_7 = slsqp_optimize(best_h, n_init, maxiter=400)
            if c5_7 < best_c5:
                best_h, best_c5 = h7, c5_7

    # Phase 6: Multi-resolution
    for n_try in [280, 420, 500]:
        if time_remaining(t0, budget_s) < 150:
            break
        if n_try == n_init:
            continue
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = repair_h(h_interp, n_try)

        mi = 200 if n_try >= 420 else 300
        h8, c5_8 = slsqp_optimize(h_interp, n_try, maxiter=mi)
        if c5_8 < best_c5:
            best_h, best_c5 = h8, c5_8

        if time_remaining(t0, budget_s) > 120:
            src = best_h if len(best_h) == n_try else h_interp
            h9, c5_9 = subgradient_ascent(src, n_try, 5000, 0.04, 0.9995, 5.0, 0.7, t0, budget_s)
            if c5_9 < best_c5:
                best_h, best_c5 = h9, c5_9
            if time_remaining(t0, budget_s) > 60:
                h10, c5_10 = slsqp_optimize(best_h, len(best_h), maxiter=200)
                if c5_10 < best_c5:
                    best_h, best_c5 = h10, c5_10

    # Phase 7: Final local perturbations
    while time_remaining(t0, budget_s) > 40:
        n_cur = len(best_h)
        noise = 0.01 + 0.025 * rng.random()
        h_pert = best_h + rng.normal(0, noise, n_cur)
        h_pert = repair_h(h_pert, n_cur)
        h_opt, c5_opt = slsqp_optimize(h_pert, n_cur, maxiter=150)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final SLSQP polish
    if time_remaining(t0, budget_s) > 30:
        h_final, c5_final = slsqp_optimize(best_h, len(best_h), maxiter=500, ftol=1e-14)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return best_h, best_c5, len(best_h)
