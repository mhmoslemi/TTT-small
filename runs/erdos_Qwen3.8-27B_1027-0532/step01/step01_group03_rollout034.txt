def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _compute_delta(h, i, j, lags, n):
    ik = i - lags
    ip = i + lags
    jk = j - lags
    jp = j + lags

    m2 = (ip >= 0) & (ip < n)

    delta[m2] += h[ip[m2]]
    delta[m3] -= h[jk[m3]]
    delta[m4] -= h[jp[m4]]
    return delta

def _slsqp_refine(h_start, n, target_sum, deadline, alpha=500):

        return c.max()
