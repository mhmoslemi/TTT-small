def active_set_lp(h, n, tol=0.005):
    """Solve the linearized active-set problem."""
    corr = np.correlate(h, 1-h, mode='full')
    dx = 2.0/n
    c_max = np.max(corr)
    
    # Identify active set: lags where corr is within tol of max
    active = np.where(corr >= c_max - tol)[0]
    
    if len(active) == 0:
        return h
    
    # Build LP: min t s.t. c_k + grad_k . (h' - h) <= t, sum(h') = n/2, 0 <= h' <= 1
    # Variables: h'[0..n-1], t -> n+1 variables
    # Constraints:
    #   For each active k: grad_k . h' - t <= -c_k + grad_k . h
    #   sum(h') = n/2
    #   0 <= h' <= 1
    #   t is free (or t >= 0)
    
    n_active = len(active)
    # c = [0, ..., 0, 1] (n zeros then 1 for t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k_idx:
    #   sum_j grad_k[j] * h'[j] - t <= -c_k + sum_j grad_k[j] * h[j]
    #   i.e., [grad_k, -1] . [h', t] <= -c_k + grad_k . h
    
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    
    for ii, k_idx in enumerate(active):
        lag = k_idx - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = 1.0
            if 0 <= j + lag < n:
                grad[j] -= h[j + lag]
            if 0 <= j - lag < n:
                grad[j] -= h[j - lag]
        A_ub[ii, :n] = grad
        A_ub[ii, n] = -1.0
        b_ub[ii] = -corr[k_idx] + np.dot(grad, h)
    
    # Equality constraint: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h'[j] <= 1, t free (or t >= -1)
    bounds_lp = [(0, 1)] * n + [(-1, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')

    x_old = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n_to)
    return h_new

def downsample(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n_to)
    return h_new

def coordinate_descent(h, n, max_rounds=50):
    h_cur = h.copy()
    c5_cur = evaluate_c5(h_cur)
    for rd in range(max_rounds):
        improved = False
        for j in range(n):
            for delta in [-0.01, -0.005, 0.005, 0.01]:
                h_trial = h_cur.copy()
                h_trial[j] += delta
                if h_trial[j] < -0.1 or h_trial[j] > 1.1:
                    continue
                h_trial = project_feasible(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5_cur - 1e-10:
                    h_cur = h_trial.copy()
                    c5_cur = c5_trial
                    improved = True
        if not improved:
            break
    return h_cur, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    best_n = n

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP polish with various eps
    for eps in [20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]:
        if time_left() < 120:
            break
        h_new, c5_new = slsqp_optimize(best_h, n, eps, maxiter=300, time_limit=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: LP-based local search
    if time_left() > 300:
        h_lp, c5_lp = lp_local_search(best_h, n, max_iter=40, time_limit=200)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 3: More SLSQP after LP
    for eps in [100.0, 500.0, 2000.0]:
        if time_left() < 100:
            break
        h_new, c5_new = slsqp_optimize(best_h, n, eps, maxiter=400, time_limit=120)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Another LP round
    if time_left() > 200:
        h_lp, c5_lp = lp_local_search(best_h, n, max_iter=30, time_limit=150)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 5: Upscale to 2*n and optimize
    if time_left() > 400:
        n2 = 2 * n
        h2 = upsample(best_h, n, n2)
        c5_2 = evaluate_c5(h2)
        # Optimize at higher resolution
        for eps in [50.0, 200.0, 500.0]:
            if time_left() < 200:
                break
            h2_new, c5_2_new = slsqp_optimize(h2, n2, eps, maxiter=200, time_limit=120)
            if c5_2_new < c5_2:
                h2 = h2_new.copy()
                c5_2 = c5_2_new
        if time_left() > 150:
            h2_lp, c5_2_lp = lp_local_search(h2, n2, max_iter=20, time_limit=120)
            if c5_2_lp < c5_2:
                h2 = h2_lp.copy()
                c5_2 = c5_2_lp
        # Compare: use higher-res if better, otherwise downsample to n
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
        else:
            # Try downsampling the optimized high-res back to n
            h_ds = downsample(h2, n2, n)
            c5_ds = evaluate_c5(h_ds)
            if c5_ds < best_c5:
                best_c5 = c5_ds
                best_h = h_ds.copy()
                best_n = n

    # Phase 6: Try n=100 from scratch (interpolate from best)
    if time_left() > 350:
        n3 = 100
        if n3 != best_n:
            h3 = upsample(best_h, best_n, n3) if n3 > best_n else downsample(best_h, best_n, n3)
            c5_3 = evaluate_c5(h3)
            for eps in [50.0, 200.0, 500.0, 2000.0]:
                if time_left() < 200:
                    break
                h3_new, c5_3_new = slsqp_optimize(h3, n3, eps, maxiter=300, time_limit=100)
                if c5_3_new < c5_3:
                    h3 = h3_new.copy()
                    c5_3 = c5_3_new
            if time_left() > 120:
                h3_lp, c5_3_lp = lp_local_search(h3, n3, max_iter=20, time_limit=100)
                if c5_3_lp < c5_3:
                    h3 = h3_lp.copy()
                    c5_3 = c5_3_lp
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

    # Phase 7: Final coordinate descent + LP
    if time_left() > 60:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_rounds=30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 60:
        h_lp, c5_lp = lp_local_search(best_h, best_n, max_iter=15, time_limit=50)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Final polish
    if time_left() > 30:
        h_new, c5_new = slsqp_optimize(best_h, best_n, 2000.0, maxiter=200, time_limit=25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, best_n
