            else:
                amt = min(steps[si], best[j] * 0.5, 1.0 - best[i])
                i, j = j, i
            if amt < 1e-10:
                continue
            trial = best.copy()
            trial[i] -= amt
            trial[j] += amt
            trial = project_feasible(trial, n)
            s = score_c5(trial)
            if s < best_score - 1e-13:
                best = trial
                best_score = s
                improved = True
                stall = 0
                si = max(0, si - 1)
                break
        if not improved:
            stall += 1
            si = min(si + 1, len(steps) - 1)
    return best, best_score


def multi_start_slsqp(t0, time_limit, rng, best_h, best_c5):
    n_list = [85, 64, 100, 128]
    for n_try in n_list:
        if time.time() - t0 > time_limit:
            break
        for attempt in range(2):
            if time.time() - t0 > time_limit:
                break
            r = rng.random()
            if r < 0.3:
                h_try = random_binary_init(n_try, rng)
            elif r < 0.6:
                h_try = thue_morse_init(n_try, rng)
            else:
                h_try = fourier_init(n_try, rng)
            x_new, s_new = try_slsqp(h_try, 1.0, 800)
            if s_new < best_c5 or s_new < best_c5 + 0.005:
                x_new, s_new = try_slsqp(x_new, 0.3, 1500)
                if s_new < best_c5:
                    x_new, s_new = try_slsqp(x_new, 0.1, 2000)
                    if s_new < best_c5:
                        best_h = x_new.copy()
                        best_c5 = s_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _CURRENT_T
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = thue_morse_init(85, rng)

    n_init = len(h_init)
    best_h = project_feasible(h_init, n_init)
    best_c5 = score_c5(best_h)

    phase1_end = t0 + budget_s * 0.22
    temps = [1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for T in temps:
        if time.time() > phase1_end:
            break
        x_new, s_new = try_slsqp(best_h, T, 2500)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()

    phase2_end = t0 + budget_s * 0.42
    best_h, best_c5 = multi_start_slsqp(t0, phase2_end, rng, best_h, best_c5)

    phase3_end = t0 + budget_s * 0.72
    tl3 = phase3_end - time.time()
    if tl3 > 30:
        best_h, best_c5 = simulated_annealing(best_h, t0, tl3, rng)

    if time.time() < phase3_end - 10:
        n = len(best_h)
        x_new, s_new = try_slsqp(best_h, 0.02, 2000)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()

    phase4_end = t0 + budget_s * 0.88
    tl4 = phase4_end - time.time()
    if tl4 > 15:
        best_h, best_c5 = targeted_search(best_h, t0, tl4, rng)

    if time.time() < t0 + budget_s - 20:
        n = len(best_h)
        x_new, s_new = try_slsqp(best_h, 0.008, 3000)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()
        x_new, s_new = try_slsqp(best_h, 0.003, 2000)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = score_c5(best_h)
    return best_h, best_c5, n
