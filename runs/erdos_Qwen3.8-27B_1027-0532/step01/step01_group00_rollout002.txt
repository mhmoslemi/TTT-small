import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def soft_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def soft_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    return w

def grad_soft_max_corr(h, alpha, w_thresh=1e-10):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    active = np.where(w > w_thresh)[0]
    for k in active:
        wk = w[k]
        lag = k - n + 1
        if lag == 0:
            grad[:] += wk
            grad[:] -= wk * 2.0 * h
        elif lag > 0:
            grad[:n - lag] += wk
            grad[:n - lag] -= wk * h[lag:]
            grad[lag:] -= wk * h[:n - lag]
        else:
            L = -lag
            grad[L:] += wk
            grad[:n - L] -= wk * h[L:]
            grad[L:] -= wk * h[:n - L]
    return grad * (2.0 / n)

def penalty_obj(h, alpha, pen_w):
    corr = compute_corr(h)
    sm = soft_max_val(corr, alpha) * (2.0 / len(h))
    pen = pen_w * (np.sum(h) - len(h) / 2.0) ** 2
    return sm + pen

def penalty_grad(h, alpha, pen_w):
    n = len(h)
    g = grad_soft_max_corr(h, alpha)
    g += 2.0 * pen_w * (np.sum(h) - n / 2.0)
    return g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_step(h, alpha, pen_w, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        penalty_obj, h, grad=penalty_grad,
        method='L-BFGS-B', bounds=bounds,
        args=(alpha, pen_w),
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10}
    )
    return project_feasible(res.x, n)

def slsqp_polish(h, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return val * (2.0 / n)

    def con_eq(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    res = minimize(
        obj, h, method='SLSQP', bounds=bounds, args=(),
        constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16}
    )
    if res.x is not None:
        return project_feasible(res.x, n)
    return project_feasible(h, n)

def generate_binary_pattern(n, rng, style=0):
    if style == 0:
        h = (rng.rand(n) < 0.5).astype(float)
    elif style == 1:
        period = max(3, n // 50)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i // period) % 2 == 0 else 0.0
        h += rng.randn(n) * 0.3
    elif style == 2:
        freq = rng.randint(3, 30)
        phase = rng.rand() * 2 * np.pi
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        h = np.clip(h, 0, 1)
        h = (h > 0.5).astype(float)
        h += rng.randn(n) * 0.2
    elif style == 3:
        shift = rng.randint(1, n)
        h = np.zeros(n)
        idx = np.arange(n)
        h[(idx * shift) % n] = 1.0
        seen = np.zeros(n, dtype=bool)
        for i in range(n):
            pos = (i * shift) % n
            if not seen[pos]:
                seen[pos] = True
                h[pos] = 1.0
        h += rng.randn(n) * 0.15
    else:
        h = rng.rand(n)
        h = (h > np.median(h)).astype(float)
    return project_feasible(h, n)
