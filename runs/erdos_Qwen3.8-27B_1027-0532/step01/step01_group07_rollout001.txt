def compute_c5(h):

def project_feasible(h, target):

def sbx_crossover(p1, p2, rng, eta=30.0):
    """Simulated binary crossover"""
    n = len(p1)
    c1 = p1.copy()
    c2 = p2.copy()
    mask = rng.random(n) < 0.5
    for i in np.where(mask)[0]:
        if p1[i] == p2[i]:
            continue
        u = rng.random()
        if u <= 0.5:
            beta = (2*u) ** (1.0/(eta+1))
        else:
            beta = (1.0/(2*(1-u))) ** (1.0/(eta+1))
        c1[i] = 0.5*((1+beta)*p1[i] + (1-beta)*p2[i])
        c2[i] = 0.5*((1-beta)*p1[i] + (1+beta)*p2[i])
    return c1, c2

def gaussian_mutate(h, rng, rate=0.1, sigma=0.1):
    """Gaussian mutation"""
    h_mut = h.copy()
    mask = rng.random(len(h)) < rate
    h_mut[mask] += rng.normal(0, sigma, mask.sum())
    return h_mut

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    idx = np.arange(n)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        shift = m - (n - 1)
        valid1 = (idx + shift >= 0) & (idx + shift < n)
        grad[valid1] += wm * (1.0 - h[idx[valid1] + shift])
        valid2 = (idx - shift >= 0) & (idx - shift < n)
        grad[valid2] += wm * (-h[idx[valid2] - shift])
    grad *= dx
    return grad


def refine_grad_descent(h, n, deadline, max_outer=6):

    best_c5 = compute_c5(h)
    best_h = h.copy()

        if time.time() > deadline - 5:
            break
        alpha = alphas[outer]
        h_cur = best_h.copy()
        lr = 0.3 / np.sqrt(n)
        for it in range(8000):
            if time.time() > deadline - 5:
                break
            c5_val = compute_c5(h_cur)
            if c5_val < best_c5 - 1e-10:
                best_c5 = c5_val
                best_h = h_cur.copy()
            grad = smooth_gradient(h_cur, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_val + 1e-13:
                h_cur = h_new
                if it % 500 == 499:
                    lr *= 0.95
            else:
                lr *= 0.75
                if lr < 1e-14:
                    break
    return best_h, best_c5


def refine_subgradient(h, n, deadline):
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idx = np.arange(n)
    lr = 0.4
    for it in range(20000):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        k_worst = int(np.argmax(c))
        c_val = c[k_worst]
        if c_val < best_c5 - 1e-10:
            best_c5 = c_val
            best_h = h.copy()
        shift = k_worst - (n - 1)
        grad = np.zeros(n)
        valid1 = (idx + shift >= 0) & (idx + shift < n)
        grad[valid1] = (1.0 - h[idx[valid1] + shift])
        valid2 = (idx - shift >= 0) & (idx - shift < n)
        grad[valid2] += -h[idx[valid2] - shift]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = lr / np.sqrt(it + 10)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new <= c_val + 1e-12:
            h = h_new
        else:
            lr *= 0.92
            if lr < 1e-10:
                break
    return best_h, best_c5


def sbx_crossover(p1, p2, rng, eta):
    n = len(p1)
    c1 = p1.copy()
    c2 = p2.copy()
    mask = rng.random(n) < 0.5
    idx = np.where(mask)[0]
    for i in idx:
        if abs(p1[i] - p2[i]) < 1e-12:
            continue
        u = rng.random()
        if u <= 0.5:
            beta = (2.0 * u) ** (1.0 / (eta + 1.0))
        else:
            beta = (1.0 / (2.0 * (1.0 - u))) ** (1.0 / (eta + 1.0))
        c1[i] = 0.5 * ((1.0 + beta) * p1[i] + (1.0 - beta) * p2[i])
        c2[i] = 0.5 * ((1.0 - beta) * p1[i] + (1.0 + beta) * p2[i])
    return c1, c2


def tournament_select(pop, fitness, rng, k=3):
    idx = rng.randint(0, len(pop), size=k)
    best_local = np.argmin(fitness[idx])
    return pop[idx[best_local]].copy()


def init_population(n, rng, h_init, pop_size):
    target = n / 2.0
    pop = []
    if h_init is not None and len(h_init) == n:
        for _ in range(40):
            h_p = h_init + rng.normal(0, 0.04, n)
            pop.append(project_feasible(h_p, target))
        for _ in range(15):
            h_r = h_init[::-1] + rng.normal(0, 0.04, n)
            pop.append(project_feasible(h_r, target))
        for _ in range(10):
            scale = rng.uniform(0.3, 0.7)
            h_s = scale * h_init / np.clip(h_init.sum() / target, 0.1, 10.0)
            pop.append(project_feasible(h_s, target))
    for _ in range(30):
        h = np.zeros(n)
        n_blocks = rng.randint(2, 7)
        block_widths = rng.randint(2, max(3, n // n_blocks), n_blocks)
        positions = rng.choice(n, n_blocks, replace=False)
        for pos, bw in zip(positions, block_widths):
            end = min(n, pos + bw)
            h[pos:end] = 1.0
        kernel = np.ones(3) / 3.0
        h = np.convolve(h, kernel, mode='same')
        pop.append(project_feasible(h, target))
    for _ in range(25):
        t = np.linspace(0, 2, n)
        h = np.zeros(n)
        n_freqs = rng.randint(2, 6)
        for _ in range(n_freqs):
            f = rng.uniform(0.5, 8.0)
            p = rng.uniform(0, 2.0 * np.pi)
            a = rng.uniform(0.1, 0.4)
            h += a * np.sin(2.0 * np.pi * f * t / 2.0 + p)
        h = 0.5 + h
        pop.append(project_feasible(h, target))
    for _ in range(15):
        h = rng.uniform(0, 1, n)
        pop.append(project_feasible(h, target))
    while len(pop) < pop_size:
        h = h_init + rng.normal(0, 0.08, n) if h_init is not None else rng.uniform(0, 1, n)
        pop.append(project_feasible(h, target))
    return pop[:pop_size]
