    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

                                      (alpha, n2), maxiter=200, ftol=1e-14)
                    h_new = _fix_sum(h_new, target_sum2)
                    c5_new = _c5_of(h_new)
                    if c5_new < h2_c5 - 1e-14:
                        h2_c5 = c5_new
                        h2_best = h_new.copy()
            if h2_c5 < best_c5 - 1e-14:
                best_c5 = h2_c5
                best_h = h2_best.copy()
                best_n = n2

    # Phase 3: Try n=120
    if elapsed_ok(budget_s * 0.25) and time.time() < phase3_end:
        n3 = 120
        target_sum3 = n3 / 2.0
        if best_n == n:
            h3 = _interpolate_h(best_h, n, n3)
        else:
            h3 = _interpolate_h(best_h, best_n, n3)
        h3 = _fix_sum(h3, target_sum3)
        h3_best = h3.copy()
        h3_c5 = _c5_of(h3)
        if h3_c5 < 1.0:
            for alpha in [50, 150, 400]:
                if not elapsed_ok(30):
                    break
                h_new = _do_slsqp(h3_best, n3, target_sum3, _obj_lse,
                                  (alpha, n3), maxiter=250, ftol=1e-14)
                h_new = _fix_sum(h_new, target_sum3)
                c5_new = _c5_of(h_new)
                if c5_new < h3_c5 - 1e-14:
                    h3_c5 = c5_new
                    h3_best = h_new.copy()
            if elapsed_ok(40):
                corr3 = np.correlate(h3_best, 1.0 - h3_best, mode='full')
                for top_m in [3, 5]:
                    for _ in range(2):
                        if not elapsed_ok(15):
                            break
                        active_idx = np.argsort(corr3)[-top_m:]
                        h_new = _do_slsqp(h3_best, n3, target_sum3, _obj_lse_active,
                                          (active_idx, 400, n3), maxiter=200, ftol=1e-14)
                        h_new = _fix_sum(h_new, target_sum3)
                        c5_new = _c5_of(h_new)
                        if c5_new < h3_c5 - 1e-14:
                            h3_c5 = c5_new
                            h3_best = h_new.copy()
                        corr3 = np.correlate(h3_best, 1.0 - h3_best, mode='full')
                        active_idx = np.argsort(corr3)[-top_m:]
            if elapsed_ok(30):
                h_pgd3 = _projected_subgradient(h3_best.copy(), n3, target_sum3,
                                                n_iter=3000, step_init=0.03)
                c5_pgd3 = _c5_of(h_pgd3)
                if c5_pgd3 < h3_c5 - 1e-14:
                    h3_c5 = c5_pgd3
                    h3_best = h_pgd3.copy()
            if h3_c5 < best_c5 - 1e-14:
                best_c5 = h3_c5
                best_h = h3_best.copy()
                best_n = n3

    # Try different starting points for n=86
    if elapsed_ok(60):
        starts = []
        for ph in [0.5, 1.0, 1.5, 2.0, 2.5]:
            h_s = _make_scan_start(n, target_sum, phase=ph)
            starts.append(h_s)
        for sv in [1, 7, 13]:
            h_b = _make_binary_start(n, target_sum, seed_val=sv)
            starts.append(h_b)

        for h_s in starts:
            if not elapsed_ok(30):
                break
            h_s = _fix_sum(h_s, target_sum)
            h_cand = h_s.copy()
            for alpha in [80, 200]:
                if not elapsed_ok(15):
                    break
                h_new = _do_slsqp(h_cand, n, target_sum, _obj_lse,
                                  (alpha, n), maxiter=200, ftol=1e-13)
                h_new = _fix_sum(h_new, target_sum)
                c5_new = _c5_of(h_new)
                if c5_new < _c5_of(h_cand) - 1e-14:
                    h_cand = h_new.copy()
            c5_cand = _c5_of(h_cand)
            if c5_cand < best_c5 - 1e-14:
                best_c5 = c5_cand
                best_h = h_cand.copy()
                best_n = n

    # Final polish: very high alpha on the best candidate
    if elapsed_ok(60):
        tn = best_n
        ts = tn / 2.0
        for alpha in [800, 1500, 3000]:
            if not elapsed_ok(20):
                break
            h_new = _do_slsqp(best_h, tn, ts, _obj_lse, (alpha, tn),
                              maxiter=400, ftol=1e-15)
            h_new = _fix_sum(h_new, ts)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = tn

        # Active set final
        if best_n == n:
            corr_final = np.correlate(best_h, 1.0 - best_h, mode='full')
            active = np.argsort(corr_final)[-5:]
            for alpha in [1000, 2000]:
                if not elapsed_ok(15):
                    break
                h_new = _do_slsqp(best_h, n, target_sum, _obj_lse_active,
                                  (active, alpha, n), maxiter=300, ftol=1e-15)
                h_new = _fix_sum(h_new, target_sum)
                c5_new = _c5_of(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()

        # Projected subgradient final
        if elapsed_ok(30):
            h_pgd_final = _projected_subgradient(best_h.copy(), best_n, best_n / 2.0,
                                                 n_iter=10000, step_init=0.02)
            c5_pf = _c5_of(h_pgd_final)
            if c5_pf < best_c5 - 1e-14:
                best_c5 = c5_pf
                best_h = h_pgd_final.copy()

    # Final SLSQP polish
    if elapsed_ok(30):
        tn = best_n
        ts = tn / 2.0
        h_new = _do_slsqp(best_h, tn, ts, _obj_lse, (2000, tn),
                          maxiter=500, ftol=1e-15)
        h_new = _fix_sum(h_new, ts)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = tn

    # Ensure feasibility
    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)

    return (best_h, best_c5, best_n)
