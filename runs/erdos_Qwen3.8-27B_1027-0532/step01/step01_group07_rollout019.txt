        inits.append(('sym', make_init_symmetric(n, rng, 3)))
        inits.append(('rand1', make_init_random(n, rng, 2)))
        inits.append(('rand2', make_init_random(n, rng, 4)))

        inits.sort(key=lambda x: compute_c5(x[1]))
        max_inits = min(len(inits), 7)

        for label, h_init in inits[:max_inits]:
            if n_deadline - time.time() < 12:
                break
            cur_h = h_init.copy()
            cur_c5 = compute_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

            if n_deadline - time.time() > 20:
                h1, c5_1 = gd_backtrack(cur_h, n, 8.0, 4000, n_deadline, lr_start=0.08, alpha_growth=1.02, alpha_max=100.0)
                if c5_1 < best_c5:
                    best_c5 = c5_1
                    best_h = h1.copy()
                cur_h = h1

            if n_deadline - time.time() > 15:
                for tau in [0.06, 0.12, 0.2]:
                    if n_deadline - time.time() < 8:
                        break
                    h_push = soft_threshold(cur_h, n, tau)
                    h2, c5_2 = gd_backtrack(h_push, n, 20.0, 2500, n_deadline, lr_start=0.05, alpha_growth=1.015, alpha_max=150.0)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        cur_h = h2

            if n_deadline - time.time() > 15:
                h3, c5_3 = gd_backtrack(cur_h, n, 80.0, 3000, n_deadline, lr_start=0.02, alpha_growth=1.01, alpha_max=400.0)
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = h3.copy()
                cur_h = h3

            if n_deadline - time.time() > 20:
                for sigma in [0.015, 0.04, 0.08]:
                    if n_deadline - time.time() < 8:
                        break
                    h_pert = cur_h + rng.normal(0, sigma, n)
                    h_pert = project_feasible(h_pert, n / 2.0)
                    h4, c5_4 = gd_backtrack(h_pert, n, 12.0, 2500, n_deadline, lr_start=0.06, alpha_growth=1.02, alpha_max=120.0)
                    if c5_4 < best_c5:
                        best_c5 = c5_4
                        best_h = h4.copy()
                        cur_h = h4

        if deadline - time.time() > 30 and best_h is not None and len(best_h) == n:
            h_fin, c5_fin = gd_backtrack(best_h.copy(), n, 150.0, 5000, deadline, lr_start=0.01, alpha_growth=1.008, alpha_max=600.0)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
