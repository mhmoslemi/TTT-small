            g = gradient_for_lag_vec(h, lag)
            A_ub[row, :n] = g
            A_ub[row, -1] = -1.0
            b_ub[row] = float(g @ h) - float(lags[lag_idx])

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if res.status != 0:
            if stall_count > 30:
                break
            continue

        h_lp = res.x[:n]

        alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]
        taken = False
        for alpha in alphas:
            h_trial = (1.0 - alpha) * h + alpha * h_lp
            h_trial = project_onto_constraint(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial <= worst_val + 1e-9:
                h = h_trial
                taken = True
                break

        if not taken:
            if stall_count > 50:
                break
            h = h + 0.05 * (h_lp - h)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_constraint(h, target_sum)

        lags_new = compute_all_lags(h)
        if float(np.max(lags_new)) >= worst_val - 1e-10 and it > 20:
            if stall_count > 80:
                break

    return best_h, best_c5

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = float(log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        k = abs(lag)
        if k >= n:
            continue
        if lag >= 0:
            i_range = np.arange(n - k)
            grad[i_range] += w * (1.0 - h[i_range + k])
            grad[i_range + k] += w * (-h[i_range])
        else:
            i_range = np.arange(k, n)
            grad[i_range] += w * (1.0 - h[i_range - k])
            grad[i_range - k] += w * (-h[i_range])
    return obj, grad * dx

def lbfgsb_optimize(h0, n, target_sum, deadline, tau=100.0, maxiter=300):
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    h_best = h0.copy()
    best_c5 = compute_c5(h0)
    n_outer = 5
    cur_tau = tau / (2.0 ** (n_outer - 1))
    for outer in range(n_outer):
        if time.time() > deadline - 5:
            break
        def obj_fun(x):
            o, _ = soft_max_obj_and_grad(x, cur_tau)
            return o
        def grad_fun(x):
            _, g = soft_max_obj_and_grad(x, cur_tau)
            return g
        try:
            res = minimize(obj_fun, h0, jac=grad_fun, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter // (n_outer - outer),
                                    'ftol': 1e-14, 'gtol': 1e-8})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = project_onto_constraint(h_opt, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    h_best = h_opt.copy()
                    h0 = h_opt.copy()
        except Exception:
            pass
        cur_tau *= 2.0
    return h_best, best_c5

def pgd_optimize(h0, n, target_sum, deadline, max_iter=5000, tau=20.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.04
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        _, g = soft_max_obj_and_grad(h, tau)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 30:
                tau *= 1.5
                lr = max(lr * 0.7, 0.003)
                stall = 0
                if lr < 0.003:
                    break
    return best_h, best_c5

def worst_case_pgd(h0, n, target_sum, deadline, max_iter=3000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.03
    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = gradient_for_lag_vec(h, lag)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.97
            if lr < 0.002:
                lr = 0.03
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old / n_new)
    return np.interp(x_new, x_old, h_old)

def structured_init(n, target_sum, seed_offset=0):
    np.random.seed(42 + seed_offset)
    x = np.linspace(0, 2, n, endpoint=False)
    candidates = []
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0]:
        for phase in [0.0, np.pi / 2, np.pi]:
            h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x + phase)
            h = np.clip(h, 0, 1)
            h = project_onto_constraint(h, target_sum)
            candidates.append(h)
    if len(candidates) > 0:
        best = min(candidates, key=compute_c5)
        return best
    return np.full(n, 0.5)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 20

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0
    else:
        n0 = 120
        best_h = structured_init(n0, n0 / 2.0)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    target = n / 2.0

    # Phase 1: Quick SLP to improve initial point
    if time.time() < deadline - 800:
        h_slp, c5_slp = slp_optimize(best_h, n, target, deadline - 700, max_iters=200, max_active=30)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    # Phase 2: Main SLP optimization
    if time.time() < deadline - 500:
        h_slp, c5_slp = slp_optimize(best_h, n, target, deadline - 400, max_iters=1000, max_active=80)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    # Phase 3: L-BFGS-B refinement
    if time.time() < deadline - 300:
        n = len(best_h)
        target = n / 2.0
        h_lb, c5_lb = lbfgsb_optimize(best_h, n, target, deadline - 200, tau=80.0, maxiter=200)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Phase 4: PGD polish
    if time.time() < deadline - 200:
        n = len(best_h)
        target = n / 2.0
        h_pgd, c5_pgd = pgd_optimize(best_h, n, target, deadline - 100, max_iter=3000, tau=15.0)
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()

    # Phase 5: SLP again after PGD (exploit new structure)
    if time.time() < deadline - 150:
        n = len(best_h)
        target = n / 2.0
        h_slp, c5_slp = slp_optimize(best_h, n, target, deadline - 80, max_iters=300, max_active=60)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    # Phase 6: Try different resolutions
    n_alternatives = [100, 110, 130, 140, 150, 160]
    for n_alt in n_alternatives:
        if time.time() > deadline - 60:
            break
        target_alt = n_alt / 2.0
        h_interp = interpolate_h(best_h, n_alt)
        h_interp = project_onto_constraint(h_interp, target_alt)
        # Quick SLP on this resolution
        h_tmp, c5_tmp = slp_optimize(h_interp, n_alt, target_alt, min(deadline - 30, time.time() + 80), max_iters=100, max_active=40)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()

    # Phase 7: Final multi-start SLP from best
    if time.time() < deadline - 30:
        n = len(best_h)
        target = n / 2.0
        for attempt in range(3):
            if time.time() > deadline - 10:
                break
            if attempt == 0:
                h_start = best_h + 0.01 * np.random.randn(n)
            elif attempt == 1:
                h_start = best_h.copy()
                idx = np.random.choice(n, size=n // 4, replace=False)
                h_start[idx] = 1.0 - h_start[idx]
            else:
                h_start = structured_init(n, target, seed_offset=attempt + 10)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = project_onto_constraint(h_start, target)
            h_tmp, c5_tmp = slp_optimize(h_start, n, target, min(deadline - 5, time.time() + 40), max_iters=80, max_active=40)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_tmp.copy()

    # Phase 8: Final worst-case PGD
    if time.time() < deadline - 3:
        n = len(best_h)
        target = n / 2.0
        h_wc, c5_wc = worst_case_pgd(best_h, n, target, deadline + 10, max_iter=5000)
        if c5_wc < best_c5:
            best_c5 = c5_wc
            best_h = h_wc.copy()

    # Final verify and project
    n = len(best_h)
    target = n / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
