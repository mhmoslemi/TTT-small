import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def sharpen(h, alpha=5.0):
    """Push h towards 0 or 1 using a sigmoid"""
    n = len(h)
    tgt = n / 2.0
    # Re-center h around 0.5
    h_centered = (h - 0.5) * alpha
    h_sharp = 0.5 + 0.5 * np.tanh(h_centered)
    return project_feasible(h_sharp, tgt)

def subgradient_step(h, step_size, k_star):
    """Take one subgradient step in direction of reducing lag k_star"""
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    k = k_star
    a = j - k
    b = j + k
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad = np.zeros(n)
    grad[m1] += (1.0 - h[a[m1]])
    grad[m2] -= h[b[m2]]
    grad *= dx
    return h - step_size * grad

def smooth_grad_descent(h0, tau=0.05, tm=60, lr=0.1):
    """Gradient descent on the smooth max objective"""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    
    for it in range(50000):
        if time.time() - t0 > tm:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = corr.max()
        weights = np.exp((corr - m) / tau)
        weights /= weights.sum()
        
        # Compute smooth gradient
        j = np.arange(n)
        grad = np.zeros(n)
        for idx in range(len(weights)):
            w = weights[idx]
            if w < 1e-8:
                continue
            k = idx - (n - 1)
            k_abs = abs(k)
            if k_abs >= n:
                continue
            # First term: (1-h[j-k_abs]) for j >= k_abs
            mask1 = j >= k_abs
            grad[mask1] += w * dx * (1.0 - h[j[mask1] - k_abs])
            # Second term: -h[j+k_abs] for j+k_abs < n
            mask2 = j + k_abs < n
            grad[mask2] -= w * dx * h[j[mask2] + k_abs]
        
        h = project_feasible(h - lr * grad, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        
        # Adaptive learning rate
        lr *= 0.9999
    
    return best

def local_search_binary(h0, tm=60):
    """Local search treating h as approximately binary"""
    n = len(h0)
    h = h0.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    tgt = n / 2.0
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h
    
    # Make it exactly binary
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, tgt)
    best = h_bin.copy()
    best_c5 = eval_c5(h_bin)
    h_bin = best.copy()
    
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    
    no_improve = 0
    while time.time() - t0 < tm and no_improve < 200:
        # Pick a random one and zero to swap
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Try the swap
        h_trial = h_bin.copy()
        h_trial[i1] = 0.0
        h_trial[i0] = 1.0
        c5_trial = eval_c5(h_trial)
        
        if c5_trial < best_c5 - 1e-10:
            h_bin = h_trial
            best = h_bin.copy()
            best_c5 = c5_trial
            # Update ones and zeros
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
            no_improve = 0
        else:
            no_improve += 1
    
    return best

Let me also think about a key issue: the smooth gradient descent might converge to a solution where the "soft max" is spread over many lags, while the true max is at one specific lag. After the smooth phase, we should switch to the hard subgradient.

Also, I should think about the learning rate schedule. A common approach is:
- Start with a moderate learning rate
- Use backtracking line search to ensure descent
- Gradually decrease the learning rate

Let me also consider using the `project_feasible` function more carefully. The binary search in `project_feasible` is exact but might be slow for large n. An alternative is to use a water-filling algorithm, which is O(n log n).

Actually, for n up to a few thousand, the binary search (100 iterations * O(n)) is O(100n) which is fine.

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_W(weights, n):
    """Compute the symmetrized weight array W from the softmax weights"""
    W = np.zeros(n)
    W[0] = weights[n-1]
    for m in range(1, n):
        W[m] = weights[n-1+m] + weights[n-1-m]
    return W

def smooth_gradient(h, tau, dx):
    """Compute the gradient of the smooth max objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    weights = np.exp((corr - m) / tau)
    weights /= weights.sum()
    
    W = compute_W(weights, n)
    first = np.convolve(W, 1.0 - h, mode='full')[:n]
    second = np.correlate(h, W, mode='full')[n-1:2*n-1]
    return dx * (first - second)

def smooth_grad_descent(h0, tau=0.05, tm=60, lr_init=0.1, lr_decay=0.9999):
    """Gradient descent on the smooth max objective with projection"""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.astype(float).copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    lr = lr_init
    
    for it in range(100000):
        if time.time() - t0 > tm:
            break
        
        grad = smooth_gradient(h, tau, dx)
        h_new = project_feasible(h - lr * grad, tgt)
        
        # Check if we improved
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
        else:
            # Backtrack: reduce learning rate
            lr *= 0.9
            if lr < 1e-10:
                break
        
        lr = max(lr * lr_decay, 1e-10)
    
    return best

def subgrad_opt(h0, tm=60, lr_init=0.3, beta=0.8):
    """Subgradient optimization with momentum"""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.astype(float).copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)
    
    for it in range(100000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        k_abs = abs(k)
        if k_abs >= n:
            break
        a = j - k_abs
        b = j + k_abs
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += (1.0 - h[a[m1]])
        grad[m2] -= h[b[m2]]
        grad *= dx
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9998
        if lr < 1e-7:
            break
    
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def sharpen(h, alpha=5.0):
    """Push h towards 0 or 1"""
    n = len(h)
    tgt = n / 2.0
    h_c = (h - h.mean()) * alpha
    h_s = 0.5 + 0.5 * np.tanh(h_c)
    return project_feasible(h_s, tgt)

def selpolish(h0, tm=60):
    """SLSQP polish"""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0.astype(float), tgt)
    t0 = time.time()
    
    def f(x):
        if time.time() - t0 > tm * 0.85:
            raise Exception("tl")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()
    
    def gfun(x):
        return x.sum() - tgt
    
    def gjac(x):
        return np.ones(n)
    
    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 20000, 'ftol': 1e-16})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def make_structured_init(n, rng):
    """Create various structured initializations"""
    r = rng.randint(8)
    if r == 0:
        # Random
        h = rng.random(n)
    elif r == 1:
        # Binary random
        h = (rng.random(n) > 0.5).astype(float)
    elif r == 2:
        # Sine wave
        x = np.linspace(0.0, 2.0, n)
        freq = float(rng.choice([1, 2, 3, 4, 5, 6, 7]))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + 0.5 * np.sin(np.pi * freq * x + phase)
    elif r == 3:
        # Block structure
        nblocks = rng.randint(2, 6)
        block_sizes = np.full(nblocks, n // nblocks)
        block_sizes[:n - block_sizes.sum()] += 1
        h = np.zeros(n)
        pos = 0
        for i, bs in enumerate(block_sizes):
            if i % 2 == 0:
                h[pos:pos+bs] = 1.0
            pos += bs
    elif r == 4:
        # Quadratic phase (chirp)
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.5 * np.sin(np.pi * (x**2) * rng.uniform(1, 3))
    elif r == 5:
        # Sawtooth
        x = np.linspace(0.0, 2.0, n)
        h = np.abs(x * rng.uniform(2, 5) % 2.0 - 1.0)
    elif r == 6:
        # Two-tone
        x = np.linspace(0.0, 2.0, n)
        f1, f2 = rng.choice([1,2,3,4,5], size=2)
        h = 0.5 + 0.3 * np.sin(np.pi * f1 * x) + 0.2 * np.sin(np.pi * f2 * x)
    else:
        # Gold sequence-like (maximally flat autocorrelation)
        m = rng.randint(4, 8)
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.5 * np.sign(np.sin(np.pi * m * x + rng.uniform(0, 2*np.pi)))

        k_abs = abs(k)

        # The subgradient tells us which direction to go
        j = np.arange(n)
        a = j - k_abs
        b = j + k_abs

        m2 = (b >= 0) & (b < n)
        grad = np.zeros(n)

        grad[m2] -= h[b[m2]]

        # and the coordinate with the most positive gradient (most beneficial decrease)
        # But we need to maintain the constraint

        neg_mask = grad < 0
        pos_mask = grad > 0
        
        if not neg_mask.any() or not pos_mask.any():
            break
        
        neg_indices = np.where(neg_mask)[0]
        pos_indices = np.where(pos_mask)[0]
        
        # Pick the most beneficial pair
        i_inc = neg_indices[np.argmin(grad[neg_mask])]  # most negative
        i_dec = pos_indices[np.argmax(grad[pos_mask])]  # most positive
        
        # Try transferring a small amount
        for delta in [0.01, 0.05, 0.1, 0.2, 0.5]:
            h_trial = h.copy()
            h_trial[i_inc] = min(1.0, h_trial[i_inc] + delta)

            h_trial[i_dec] -= actual_delta
            h_trial = np.clip(h_trial, 0.0, 1.0)

            if c5_trial < best_c5 - 1e-12:
                h = h_trial
                best = h.copy()
                best_c5 = c5_trial
                break
    
    return best
