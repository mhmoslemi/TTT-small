import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights /= w_sum
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = weights[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
        else:
            k2 = -k
            grad[:n-k2] += wm * (1.0 - h[k2:])
            grad[k2:] -= wm * h[:n-k2]
    return grad * dx

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    res = minimize(
        obj_smooth, h0, args=(alpha,),
        jac=lambda x: grad_smooth(x, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res.x
