        h[remaining[0]] = float(extras)
    return project_feasible(h, n)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = random_binary_init(85, rng)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = score_c5(best_h)

    deadline1 = t0 + budget_s * 0.15
    h_opt, s = lambda_coord_descent(h0, n0, max_outer=100, max_inner_sweeps=300, tol=1e-10, t_deadline=deadline1)
    if s < best_c5:
        best_c5 = s
        best_h = h_opt.copy()

    deadline2 = t0 + budget_s * 0.45
    h_opt, s = perturbed_optimize(best_h, n0, rng, n_perturb=15, t_deadline=deadline2)
    if s < best_c5:
        best_c5 = s
        best_h = h_opt.copy()

    deadline3 = t0 + budget_s * 0.80
    n_values = [40, 50, 60, 70, 85, 95, 110, 130]
    for n_try in n_values:
        if time.time() > deadline3:
            break
        tl = deadline3 - time.time()
        if tl < 10:
            break

        h_init = None
        if n_try == n0:
            h_init = best_h.copy()
        elif n_try != len(best_h):
            h_init = interpolate_h(best_h, n_try)

        for attempt in range(3):
            if time.time() > deadline3:
                break
            if h_init is not None and attempt > 0:
                h_try = project_feasible(h_init + 0.05 * rng.randn(n_try), n_try)
            elif h_init is not None:
                h_try = h_init.copy()
            else:
                h_try = random_binary_init(n_try, rng)

            h_opt, s = lambda_coord_descent(h_try, n_try, max_outer=60, max_inner_sweeps=150, tol=1e-10, t_deadline=deadline3)
            if s < best_c5:
                best_c5 = s
                best_h = h_opt.copy()
                h_init = h_opt.copy()

    deadline4 = t0 + budget_s * 0.95
    if time.time() < deadline4:
        nl = len(best_h)
        h_opt, s = perturbed_optimize(best_h, nl, rng, n_perturb=8, t_deadline=deadline4)
        if s < best_c5:
            best_c5 = s
            best_h = h_opt.copy()

    if time.time() < t_deadline - 5:
        nl = len(best_h)
        h_opt, s = lambda_coord_descent(best_h, nl, max_outer=50, max_inner_sweeps=200, tol=1e-11, t_deadline=t_deadline - 2)
        if s < best_c5:
            best_c5 = s
            best_h = h_opt.copy()

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = score_c5(best_h)
    return best_h, best_c5, len(best_h)
