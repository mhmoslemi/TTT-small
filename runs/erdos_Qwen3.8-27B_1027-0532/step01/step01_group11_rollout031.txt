def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def optimize_slsqp(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def simulated_annealing(h_init, rng, n_steps, start_temp, cool_rate):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    temp = start_temp
    for step in range(n_steps):
        n_pert = max(1, int(n * (0.03 + 0.07 * rng.random())))
        idx = rng.choice(n, size=n_pert, replace=False)
        pert = rng.uniform(-0.3, 0.3, size=n_pert)
        h_new = h_cur.copy()
        h_new[idx] += pert
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h_cur)
        delta = c5_new - c5_cur
        if delta < 0 or (temp > 1e-10 and rng.random() < np.exp(-delta / temp)):
            h_cur = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        temp *= cool_rate
    return best_h, best_c5
