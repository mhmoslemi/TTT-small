def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_objective_and_grad(h, n, t):
    """Compute log-sum-exp smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = max_corr + np.log(sum_exp) / t
    weights = exp_vals / sum_exp
    
    # Compute gradient
    grad = np.zeros(n)
    n_c = len(corr)  # 2n-1
    for idx in range(n_c):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = max_corr + np.log(sum_exp) / t
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    nc = len(corr)
    for idx in range(nc):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[0:n - lag] += w * (1.0 - h[lag:n])
            if lag < n:
                grad[lag:n] -= w * h[0:n - lag]
        else:
            l = -lag
            grad[l:n] += w * (1.0 - h[0:n - l])
            if n - 1 - l >= 0:
                grad[0:n - l] -= w * h[l:n]
    grad *= dx
    return obj, grad


def fista_optimize(h0, n, t, max_iter=3000):
    h = project_onto_constraint(h0.astype(float).copy(), n)
    y = h.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    step = 0.08
    beta_hist = 1.0

    for it in range(max_iter):
        obj, grad = lse_grad(y, n, t)
        h_new = y - step * grad
        h_new = project_onto_constraint(h_new, n)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_hist * beta_hist)) / 2.0
        y = h_new + ((beta_hist - 1.0) / beta_new) * (h_new - h)
        y = project_onto_constraint(y, n)
        beta_hist = beta_new
        h = h_new

        if it % 100 == 0 or it == max_iter - 1:
            true_obj = compute_c5(h, n)
            if true_obj < best_obj:
                best_obj = true_obj
                best_h = h.copy()
    return best_h, best_obj


def slsqp_solve(h0, n, t, max_iter=150):
    h0 = project_onto_constraint(h0.astype(float).copy(), n)
    bounds = [(0.0, 1.0)] * n

    def obj_fn(h):
        val, _ = lse_grad(h, n, t)
        return val

    def grad_fn(h):
        _, g = lse_grad(h, n, t)
        return g

    def con_fun(h):
        return np.sum(h) - n / 2.0

    def con_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(np.clip(res.x, 0.0, 1.0), n)
        return h_opt
    except Exception:
        return h0.copy()


def subgradient_polish(h, n, n_iter=2000, step_init=0.02):
    dx = 2.0 / n
    h = project_onto_constraint(h.astype(float).copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step_init

    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[0:n - lag] += (1.0 - h[lag:n])
            if lag < n:
                grad[lag:n] -= h[0:n - lag]
        else:
            l = -lag
            grad[l:n] += (1.0 - h[0:n - l])
            if n - 1 - l >= 0:
                grad[0:n - l] -= h[l:n]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_try = h - step * grad
        h_try = project_onto_constraint(h_try, n)
        c5_try = compute_c5(h_try, n)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = h_try
        step *= 0.9995
        if step < 0.0005:
            break
    return best_h, best_c5
