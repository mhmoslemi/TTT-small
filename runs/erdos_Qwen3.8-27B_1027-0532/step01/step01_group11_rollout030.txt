import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_raw(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_raw(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * dx * omh[:n - s]
            grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            grad[:n - r] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:n - r]
    return grad

def slsqp_solve(h_init, eps, maxiter=300):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def constraint_fun(x):
        return float(np.sum(x) - target_sum)
    
    def constraint_jac(x):
        return np.ones(len(x))
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def binary_local_search(h_bin, max_passes=100, time_limit=None, start_time=None):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    corr_raw = compute_corr_raw(h).astype(float)
    best_c5 = dx * np.max(corr_raw)
    
    # Precompute delta vectors
    # delta_plus[i]: effect on corr_raw of changing h[i] from 0 to 1
    # delta_minus[i]: effect on corr_raw of changing h[i] from 1 to 0
    delta = np.zeros((n, 2*n-1))
    one_minus_h = 1.0 - h
    for i in range(n):
        k_start = n - 1 - i
        delta[i, k_start:k_start+n] = one_minus_h
    delta_minus = -delta  # removing a 1
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for iteration in range(max_passes):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        best_improvement = 0
        best_flip = None
        current_max = np.max(corr_raw)
        
        for i in ones:
            di = delta_minus[i]  # shape (2n-1,)
            for j in zeros:
                dj = delta[j]  # shape (2n-1,)
                new_max = np.max(corr_raw + di + dj)
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_flip = (i, j)
        
        if best_flip is None:
            break
        
        i, j = best_flip
        h[i] = 0.0
        h[j] = 1.0
        one_minus_h = 1.0 - h
        # Update delta
        for idx in range(n):
            delta[idx, n-1-idx : n-1-idx+n] = one_minus_h
        delta_minus = -delta
        
        corr_raw = compute_corr_raw(h).astype(float)
        new_c5 = dx * np.max(corr_raw)
        if new_c5 < best_c5:
            best_c5 = new_c5
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return h, best_c5
