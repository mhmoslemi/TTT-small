def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

    """Find x in [0,1] minimizing max_k(a[k]*x + b[k])."""
    # Check boundaries
    f0 = np.max(b)
    f1 = np.max(a + b)

    # Compute subgradient at 0 and 1
    vals0 = b

        return 0.0  # Function increasing at 0, so min at 0
    
    vals1 = a + b
    active1 = np.where(vals1 >= f1 - 1e-14)[0]
    if len(active1) > 0 and np.max(a[active1]) < 0:
        return 1.0  # Function decreasing at 1, so min at 1
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a * mid + b

        active = np.where(vals >= f_mid - 1e-14)[0]

        sg_max = np.max(a[active])
        if sg_min <= 1e-14 and sg_max >= -1e-14:
            return mid
        elif sg_min > 1e-14:
            hi = mid
        else:

So c_k = sum_n a[n+k] * v[n] (ignoring conjugate for real values)

In 'full' mode, k ranges from -(len(v)-1) to len(a)-1.

The output has length len(a) + len(v) - 1 = 2n-1 (when both have length n).

The output index i (0 to 2n-2) corresponds to lag k = i - (len(v) - 1) = i - (n-1).

So: output[i] = sum_n a[n + (i-(n-1))] * v[n] for valid n.

With a = h and v = 1-h:
output[i] = sum_n h[n + i - n + 1] * (1-h[n]) for valid n
           = sum_n h[n + s] * (1-h[n]) where s = i - n + 1

Let me redo with s = i - (n-1):
output[i] = sum_n h[n+s] * (1-h[n]) where s = i-(n-1), and n ranges so that 0 ≤ n < n and 0 ≤ n+s < n.

So for our problem:
c_s = sum_n h[n+s] * (1-h[n]) for valid n (where s = i-(n-1))

Now, how does h[i_idx] (where i_idx is the coordinate index) appear in c_s?

h[i_idx] appears as h[n+s] when n+s = i_idx, i.e., n = i_idx - s. Valid if 0 ≤ i_idx-s < n and 0 ≤ i_idx < n (always true for 0≤i_idx<n).
Contribution: (1-h[i_idx-s]) * h[i_idx] → coefficient of h[i_idx] is (1-h[i_idx-s])

h[i_idx] appears in (1-h[n]) when n = i_idx. Valid if 0 ≤ i_idx < n (always) and 0 ≤ i_idx+s < n.
Contribution: h[i_idx+s] * (1-h[i_idx]) → coefficient of h[i_idx] is -h[i_idx+s]

So: a_s = (1-h[i_idx - s])*[0 ≤ i_idx-s < n] - h[i_idx + s]*[0 ≤ i_idx+s < n]

Let me double-check with the small example:
n=3, h = [h0, h1, h2], v = 1-h = [1-h0, 1-h1, 1-h2]

output has 5 elements, s ranges from -2 to 2:
- s=-2 (i=0): c = sum_n h[n-2]*(1-h[n]), valid n: n=2 (since n-2≥0 → n≥2, and n<3). So c = h[0]*(1-h[2])
- s=-1 (i=1): c = sum_n h[n-1]*(1-h[n]), valid n: n=1,2. So c = h[0]*(1-h[1]) + h[1]*(1-h[2])
- s=0 (i=2): c = sum_n h[n]*(1-h[n]), valid n: n=0,1,2. So c = h[0]*(1-h[0]) + h[1]*(1-h[1]) + h[2]*(1-h[2])
- s=1 (i=3): c = sum_n h[n+1]*(1-h[n]), valid n: n=0,1. So c = h[1]*(1-h[0]) + h[2]*(1-h[1])
- s=2 (i=4): c = sum_n h[n+2]*(1-h[n]), valid n: n=0. So c = h[2]*(1-h[0])

Now let's compute a_s for i_idx=1 (h[1]):
- s=-2: a = (1-h[1-(-2)])*[0≤3<3] - h[1+(-2)]*[0≤-1<3] = (1-h[3])*[F] - h[-1]*[F] = 0
  Check: c_{-2} = h[0]*(1-h[2]), no h[1]. ✓
- s=-1: a = (1-h[1-(-1)])*[0≤2<3] - h[1+(-1)]*[0≤0<3] = (1-h[2]) - h[0]
  Check: c_{-1} = h[0]*(1-h[1]) + h[1]*(1-h[2]) = h[0] - h[0]*h[1] + h[1] - h[1]*h[2]
  = h[1]*(1-h[2]) + h[0] - h[0]*h[1]
  Coefficient of h[1]: (1-h[2]) - h[0]. ✓
- s=0: a = (1-h[1-0])*[0≤1<3] - h[1+0]*[0≤1<3] = (1-h[1]) - h[1] = 1 - 2*h[1]
  Check: c_0 = h[0]*(1-h[0]) + h[1]*(1-h[1]) + h[2]*(1-h[2])
  Coefficient of h[1]: 1-2*h[1]. ✓
- s=1: a = (1-h[1-1])*[0≤0<3] - h[1+1]*[0≤2<3] = (1-h[0]) - h[2]
  Check: c_1 = h[1]*(1-h[0]) + h[2]*(1-h[1])
  Coefficient of h[1]: (1-h[0]) - h[2]. ✓
- s=2: a = (1-h[1-2])*[0≤-1<3] - h[1+2]*[0≤3<3] = 0
  Check: c_2 = h[2]*(1-h[0]), no h[1]. ✓

So the formula is:
a_s = (1-h[i-s])*[0≤i-s<n] - h[i+s]*[0≤i+s<n]

In my code, s_values = np.arange(2*n-1) - (n-1) gives s from -(n-1) to n-1.

For coordinate i_idx:
- valid_first (h[i_idx] as h[n+s], i.e., n+s=i_idx, n=i_idx-s): 0≤i_idx-s<n
- valid_second (h[i_idx] in (1-h[n]), i.e., n=i_idx, need 0≤i_idx+s<n): 0≤i_idx+s<n

a_s = (1-h[i_idx - s_values[valid_first]])*[valid_first] - h[i_idx + s_values[valid_second]]*[valid_second]

```python
def compute_a_s(h, i, s_values):
    n = len(h)
    a_s = np.zeros(len(s_values))
    
    # h[i] appears as h[n+s] when n = i-s, valid if 0 <= i-s < n
    nf = i - s_values  # n = i - s
    valid_f = (nf >= 0) & (nf < n)
    a_s[valid_f] += (1.0 - h[nf[valid_f]])
    
    # h[i] appears in (1-h[n]) when n = i, valid if 0 <= i+s < n
    ns = i + s_values  # need i+s in [0, n)
    valid_s = (ns >= 0) & (ns < n)
    a_s[valid_s] -= h[ns[valid_s]]
    
    return a_s
```

Let me also think about whether there's a way to use the structure of the problem more effectively.

Key observation: The overlap c_s = sum_n h[n+s]*(1-h[n]) can be rewritten as:
c_s = sum_n h[n+s] - sum_n h[n+s]*h[n] = (sum of h[n+s] over valid n) - (autocorrelation at lag s)

The first term depends only on s (it's the sum of h over a sub-range), and the second term is the autocorrelation.

Actually, for the full correlate, the sum of h[n+s] over valid n is:
- For s ≥ 0: sum_{n=0}^{n-1-s} h[n+s] = sum_{m=s}^{n-1} h[m]
- For s < 0: sum_{n=-s}^{n-1} h[n+s] = sum_{m=0}^{n-1+s} h[m]

So c_s = (partial sum of h) - (autocorrelation at lag s)

The partial sum is fixed (given h), so minimizing max c_s is equivalent to maximizing min (autocorrelation at lag s) - which is a different way to look at it but doesn't immediately help.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def find_optimal_1d(a, b):
    n_lags = len(a)
    f0 = np.max(b)
    f1 = np.max(a + b)
    vals0 = b
    tol = 1e-12 * max(1.0, np.max(np.abs(vals0)))
    active0 = np.where(vals0 >= f0 - tol)[0]
    if len(active0) > 0 and np.min(a[active0]) > 1e-10:
        return 0.0
    vals1 = a + b
    tol1 = 1e-12 * max(1.0, np.max(np.abs(vals1)))
    active1 = np.where(vals1 >= f1 - tol1)[0]
    if len(active1) > 0 and np.max(a[active1]) < -1e-10:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        f_mid = np.max(vals)
        tol_m = 1e-12 * max(1.0, f_mid)
        active = np.where(vals >= f_mid - tol_m)[0]
        if len(active) == 0:
            active = np.where(vals >= f_mid - 1e-10)[0]
        sg_min = np.min(a[active])
        sg_max = np.max(a[active])
        if sg_min <= 1e-10 and sg_max >= -1e-10:
            return mid
        elif sg_min > 1e-10:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coordinate_descent(h, n_sweeps, time_limit, seed):
    n = len(h)
    target_sum = n / 2.0
    start_time = time.time()
    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    s_values = np.arange(2 * n - 1) - (n - 1)
    c = np.correlate(h, 1.0 - h, mode='full')
    no_improve_count = 0
    for sweep in range(n_sweeps):
        if time.time() - start_time > time_limit:
            break
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            a_s = np.zeros(2 * n - 1)
            nf = i - s_values
            valid_f = (nf >= 0) & (nf < n)
            if np.any(valid_f):
                a_s[valid_f] += (1.0 - h[nf[valid_f]])
            ns = i + s_values
            valid_s = (ns >= 0) & (ns < n)
            if np.any(valid_s):
                a_s[valid_s] -= h[ns[valid_s]]
            b_s = c - a_s * h[i]
            x_opt = find_optimal_1d(a_s, b_s)
            delta = x_opt - h[i]
            if abs(delta) > 1e-14:
                h[i] = x_opt
                c += delta * a_s
        h = project_feasible(h, target_sum)
        c = np.correlate(h, 1.0 - h, mode='full')
        score = compute_c5(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                break
    return best_h, best_score

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_global_n = 0
_global_target = 0.0

def _constraint_eq(h):
    return np.sum(h) - _global_target

def _constraint_eq_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h, n_points, target_sum, alpha, maxiter):
    global _global_n, _global_target
    _global_n = n_points
    _global_target = target_sum
    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': _constraint_eq, 'jac': _constraint_eq_jac}]
    try:
        res = minimize(obj_smooth, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return h

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target_sum = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: Coordinate descent from initial (long run)
    tl = time_left()
    if tl > 50:
        h_cd, score_cd = coordinate_descent(best_h, n_sweeps=2000, time_limit=min(300, tl * 0.35), seed=seed)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd

    # Phase 2: Binarize and re-optimize
    tl = time_left()
    if tl > 50:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, target_sum)
        h_bin, score_bin = coordinate_descent(h_bin, n_sweeps=1000, time_limit=min(150, tl * 0.3), seed=seed + 1)
        if score_bin < best_score:
            best_score = score_bin
            best_h = h_bin

    # Phase 3: SLSQP polish on current best
    tl = time_left()
    if tl > 60:
        for alpha in [50, 100, 200]:
            if time_left() < 30:
                break
            h_sl = slsqp_optimize(best_h, n, target_sum, alpha, 1000)
            s_sl = compute_c5(h_sl)
            if s_sl < best_score:
                best_score = s_sl
                best_h = h_sl

    noise_levels = [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]
    while time_left() > 80:
        noise = rng.choice(noise_levels)
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target_sum)
        remaining = time_left()

        if score_new < best_score:

            best_h = h_new
        # Also try binarized perturbation
        tl2 = time_left()
        if tl2 > 40:
            h_pert_bin = np.where(h_pert > 0.5, 1.0, 0.0)
            h_pert_bin = project_feasible(h_pert_bin, target_sum)
            hb, sb = coordinate_descent(h_pert_bin, n_sweeps=300, time_limit=min(60, tl2 * 0.3), seed=rng.randint(100000))
            if sb < best_score:
                best_score = sb
                best_h = hb

    # Phase 5: Try different grid sizes
    tl = time_left()
    if tl > 100:
        for n_try in [169, 257, 211]:
            if time_left() < 80:
                break
            ts = n_try / 2.0
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, n)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, ts)
            h_interp, score_interp = coordinate_descent(h_interp, n_sweeps=500, time_limit=min(80, time_left() * 0.4), seed=rng.randint(100000))
            if score_interp < best_score:
                best_score = score_interp
                best_h = h_interp
                n = n_try
                target_sum = ts

    # Phase 6: Final coordinate descent polish
    tl = time_left()
    if tl > 30:
        best_h, best_score = coordinate_descent(best_h, n_sweeps=2000, time_limit=max(20, tl * 0.5), seed=seed + 9999)

    # Phase 7: SLSQP final polish
    tl = time_left()
    if tl > 40:
        for alpha in [100, 300, 500]:
            if time_left() < 30:
                break
            h_sl = slsqp_optimize(best_h, n, target_sum, alpha, 2000)
            s_sl = compute_c5(h_sl)
            if s_sl < best_score:

    # Final coordinate descent
    tl = time_left()
    if tl > 20:
        best_h, best_score = coordinate_descent(best_h, n_sweeps=1000, time_limit=max(10, tl), seed=seed + 77777)

    best_h = project_feasible(best_h, target_sum)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
