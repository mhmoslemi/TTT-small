def pairwise_update(h, p, q, nx=50):
    """Optimize h[p] and h[q] with h[p]+h[q] constant using golden section."""
    h_save = (h[p], h[q])
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-10:
        return 0.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h[p] += c
    h[q] -= c
    fc = (np.correlate(h, 1.0 - h, mode='full')).max()
    
    h[p] = h_save[0] + d
    h[q] = h_save[1] - d
    fd = (np.correlate(h, 1.0 - h, mode='full')).max()

        if fc < fd:
            b = d
            d = c
            fd = fc

            h[q] = h_save[1] - c
            fc = (np.correlate(h, 1.0 - h, mode='full')).max()

            c = d
            fc = fd

            h[q] = h_save[1] - d
            fd = (np.correlate(h, 1.0 - h, mode='full')).max()

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def eval_c5(h):

def golden_pair(h, p, q, nx=35):
    hp, hq = h[p], h[q]
    t_lo = max(-hp, hq - 1.0)
    t_hi = min(1.0 - hp, hq)
