def project_onto_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    # Use bisection on lambda: h = clip(x - lambda, 0, 1)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    # Adjust the largest non-bound component
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Find interior points
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(ov)

    p = exp_vals / np.sum(exp_vals)

    """
    Compute gradient of sum_k p_k * c_k where c_k = corr[k]*dx
    corr[k] = sum_i h[i+d_k]*(1-h[i]) where d_k = k-(n-1)
