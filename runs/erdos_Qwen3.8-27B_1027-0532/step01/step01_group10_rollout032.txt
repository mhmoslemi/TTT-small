            best_h = h_new.copy()
        h = h_new
        if it % 5000 == 0:

import numpy as np

import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_budget(v, budget):
    """Project v onto {x : 0<=x<=1, sum(x)=budget}"""
    lo, hi = -np.max(np.abs(v)) - 1.0, 1.0 + np.max(np.abs(v))
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > budget:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(v - t, 0.0, 1.0)

def compute_gradient(h, n, weights):
    """Compute weighted gradient of sum_k w[k] * corr_full[k]"""
    grad = np.zeros(n)
    n1 = n - 1
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-12:
            continue
        d = m - n1
        if d == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif d > 0:
            # k = d > 0
            # grad[j] = (1-h[j-d]) * [d <= j] - h[j+d] * [j <= n-1-d]
            grad[d:] += wm * (1.0 - h[:-d])  # (1-h[j-d]) for j=d..n-1, h[j-d] is h[0..n-1-d]
            grad[:n-d] -= wm * h[d:]          # -h[j+d] for j=0..n-1-d, h[j+d] is h[d..n-1]
        else:
            d2 = -d
            # k = -d2 < 0
            # grad[j] = (1-h[j+d2]) * [j <= n-1-d2] - h[j-d2] * [d2 <= j]
            grad[:n-d2] += wm * (1.0 - h[d2:])  # (1-h[j+d2]) for j=0..n-1-d2
            grad[d2:] -= wm * h[:n-d2]           # -h[j-d2] for j=d2..n-1
    return grad

def subgrad_descent(h0, n, dx, n_iter, alpha0, beta_sm=50.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        w = np.exp((corr - m) * beta_sm)
        sw = np.sum(w)
        if sw < 1e-300:
            break
        w /= sw
        
        alpha = alpha0 / (1.0 + 0.01 * it)
        grad = compute_gradient(h, n, w)
        h_new = h - alpha * grad
        h_new = project_budget(h_new, n * 0.5)
        
        c = c5_value(h_new, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def lsumexp(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def lbfgsb_optimize(h0, n, dx, beta, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(lsumexp, h0, args=(dx, beta), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    if res.x is not None:
        h = np.clip(res.x, 0.0, 1.0)
        h = project_budget(h, n * 0.5)

def thue_morse_binary(n):
    """Generate Thue-Morse sequence of length n, with exactly n/2 ones"""
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Ensure exactly n/2 ones
    ones = np.sum(t)
    if ones > n / 2:
        # Set excess ones to zero (from the end)
        idx = np.where(t == 1)[0]
        for i in idx[-(ones - n//2):]:
            t[i] = 0
    elif ones < n / 2:
        idx = np.where(t == 0)[0]
        for i in idx[:n//2 - int(ones)]:
            t[i] = 1
    return t
