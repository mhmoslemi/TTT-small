def smooth_max_objective(h, beta, n):
    """Compute smooth approximation of max_k corr[k] * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    max_val = np.max(scaled)
    # For numerical stability
    log_sum = max_val + np.log(np.sum(np.exp(scaled - max_val)))
    return log_sum / beta

def aug_lagrangian_smooth(h, n, beta, max_outer=20, max_inner=200):
    """Minimize F_beta(h) subject to 0<=h<=1, sum(h)=n/2 using augmented Lagrangian"""
    target = n / 2.0
    dx = 2.0 / n
    h = h.astype(float).copy()
    lambda_ = 0.0
    mu = 10.0
    
    for outer in range(max_outer):
        # Inner: minimize F_beta(h) + lambda*(sum(h)-target) + mu/2*(sum(h)-target)^2
        #       subject to 0<=h<=1
        def obj_and_grad(x):

            scaled = beta * c * dx
            m = np.max(scaled)
            val = (m + np.log(np.sum(np.exp(scaled-m))))/beta
            # Penalty terms
            s = np.sum(x) - target
            val += lambda_ * s + 0.5 * mu * s * s
            return val
        
        def grad(x):
            c = np.correlate(x, 1.0-x, mode="full")
            scaled = beta * c * dx
            m = np.max(scaled)
            exps = np.exp(scaled-m)

            s = np.sum(x) - target
            g += (lambda_ + mu * s) * np.ones(n)
            return g

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
