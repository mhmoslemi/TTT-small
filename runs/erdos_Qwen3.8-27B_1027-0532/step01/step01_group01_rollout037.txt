import time

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_obj(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-x[jpl])
            grad[j] += w * term
    grad *= dx
    return grad
