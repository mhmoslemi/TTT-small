def project_feasible(x, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_value_grad(h, n, eps):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_shift = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shift)
    sum_exp = np.sum(exp_shifted)
    value = (m + max_shift / eps + (1.0 / eps) * np.log(sum_exp)) * dx
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += w * term
    grad *= dx
    return value, grad

def slsqp_refine(h, n, bounds, eps, maxiter):
    target = n / 2.0
    def obj(x):
        v, _ = smooth_max_value_grad(x, n, eps)
        return v
    def grad(x):
        _, g = smooth_max_value_grad(x, n, eps)
        return g
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(
            obj, h.copy(),
            jac=grad,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_local_search(h, n, max_passes, time_limit):
    h_best = h.copy()
    best_c5 = evaluate_c5(h_best)
    for _pass in range(max_passes):
        if time.time() > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_trial = h_best.copy()
            if h_best[i] > 0.5:
                h_trial[i] = 0.0
            else:
                h_trial[i] = 1.0
            h_trial = project_feasible(h_trial, n)
            c5 = evaluate_c5(h_trial)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                h_best = h_trial.copy()
                improved = True
        if not improved:
            break
        improved = False
        n_swaps = min(200, n * (n - 1) // 2)
        for _ in range(n_swaps):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            h_trial = h_best.copy()
            h_trial[i], h_trial[j] = h_best[j], h_best[i]
            if np.any(h_trial < -1e-10) or np.any(h_trial > 1 + 1e-10):
                continue
            c5 = evaluate_c5(h_trial)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                h_best = h_trial.copy()
                improved = True
        if not improved:
            break
    return h_best, best_c5

def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 0.01 * max_val)[0]
    grad = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += term
    grad /= (len(active) * n)
    grad *= dx
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def make_binary_start(n, seed_offset=0):
    rng = np.random.RandomState(42 + seed_offset)
    h = np.zeros(n)
    n_ones = n // 2
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: SLSQP with increasing eps
    for eps in [50.0, 100.0, 300.0]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 200:
            break
        h_new = slsqp_refine(best_h, n, bounds, eps, 200)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: Iterate binary local search + SLSQP
    num_iter = 150
    for iteration in range(num_iter):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 150:
            break
        time_limit = start_time + budget_s - 120

        h_bin, c5_bin = binary_local_search(best_h, n, max_passes=4, time_limit=time_limit)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        elapsed = time.time() - start_time
        if budget_s - elapsed < 100:
            break
        h_new = slsqp_refine(best_h, n, bounds, 200.0, 150)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        if iteration % 20 == 19:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 200:
                break
            h_new = slsqp_refine(best_h, n, bounds, 500.0, 200)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 3: Multi-start with different initializations
    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        for s_off in range(6):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 100:
                break
            h_alt = make_binary_start(n, s_off)
            h_alt = slsqp_refine(h_alt, n, bounds, 100.0, 150)
            h_alt, c5_alt = binary_local_search(h_alt, n, max_passes=3, time_limit=start_time + budget_s - 60)
            c5_alt = evaluate_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                h_new = slsqp_refine(best_h, n, bounds, 300.0, 200)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 4: Subgradient fine-tuning
    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 40:
                break
            for _ in range(50):
                h_sub = subgradient_step(best_h, n, step)
                c5_sub = evaluate_c5(h_sub)
                if c5_sub < best_c5 - 1e-12:
                    best_c5 = c5_sub
                    best_h = h_sub.copy()
                else:
                    break

    # Phase 5: Final refinement
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_new = slsqp_refine(best_h, n, bounds, 500.0, 200)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 6: Tiny perturbations
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        for _ in range(30):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 10:
                break
            h_pert = best_h + 0.0005 * np.random.randn(n)
            h_pert = project_feasible(h_pert, n)
            h_new = slsqp_refine(h_pert, n, bounds, 300.0, 50)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
