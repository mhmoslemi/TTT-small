            ones = np.where(h == 1)[0]

        cooling = 0.999
        current_c5 = compute_c5(h)
        
        for it in range(1000000):
            if time.time() > deadline - 2:
                break
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Pick random swap
            ones = np.where(h == 1)[0]

            a = ones[np.random.randint(len(ones))]

    stall = 0

            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        dx = 2.0 / n

        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)

            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])

        j2_hi = min(n - 1, n - 1 - lag)
