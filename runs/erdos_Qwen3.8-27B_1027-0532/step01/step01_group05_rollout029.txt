def binary_round_and_smooth(h, n):
    """Round to binary then smooth back."""
    dx = 2.0 / n
    # Round to binary: keep the n/2 largest values as 1

    h_bin = (h >= threshold_idx).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Set the smallest of the ones to zero
        one_indices = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_indices])
        for idx in order:
            h_bin[one_indices[idx]] = 0
            n_ones -= 1
            if n_ones == n // 2:
                break
    elif n_ones < n // 2:
        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_indices])

            n_ones += 1

def binary_local_search(h0_bin, n, n_iter=10000):
    """Local search on binary patterns to minimize max R(k)."""
    h = h0_bin.copy()
    dx = 2.0 / n
    
    def eval_binary(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr)
    
    current_max = eval_binary(h)
    
    for it in range(n_iter):
        # Find the lag with maximum R(k)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(corr)
        lag_max = idx_max - (n - 1)
        
        # Try to reduce R(lag_max) by flipping a bit
        # R(lag) depends on pairs (i, i+lag)
        # If h[i]=1 and h[i+lag]=0, flipping h[i] to 0 reduces R(lag) by 1
        # If h[i+lag]=0 and h[i]=1, flipping h[i+lag] to 1 reduces R(lag) by 1
        # But we need to maintain exactly n/2 ones, so we do a swap
        
        # Find candidates: positions where h[i]=1 and h[i+lag]=0 (or vice versa)
        if lag_max >= 0:
            # R(lag) = sum over i of h[i]*(1-h[i+lag]) for i in [0, n-1-lag]
            # To reduce: find i where h[i]=1, h[i+lag]=0, and swap with some j where h[j]=0, h[j+lag]=1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # For each one position i, check if h[i+lag]=0
            # For each zero position j, check if h[j+lag]=1
            # We want to swap i (one) with j (zero) such that:
            # - This reduces R(lag_max): h[i]=1,h[i+lag]=0 and h[j]=0,h[j+lag]=1
            #   After swap: h[i]=0 (removes from R), h[j]=1 (adds to R if h[j+lag]=0)
            #   Net change in R(lag): -1 + [h[j+lag]==0 ? 1 : 0] = -1 + (1 - h[j+lag])
            #   Since h[j+lag]=1 (our condition), net change = -1 + 0 = -1. Good!
            # - But we need to check effect on other lags
            
            # Simple approach: try random swaps and accept if max R(k) decreases
            if len(ones) > 0 and len(zeros) > 0:
                improved = False
                for _ in range(100):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if i == j:
                        continue
                    h[i] = 0
                    h[j] = 1
                    new_max = eval_binary(h)
                    if new_max < current_max:
                        current_max = new_max
                        improved = True
                        ones[i] = j  # Update (rough)
                        break
                    else:
                        h[i] = 1
                        h[j] = 0

                    # Try random swap anyway (simulated annealing)

                        # Accept with some probability
                        if new_max <= current_max + rng.randint(0, 2):
                            current_max = new_max
                        else:
                            h[i] = 1
                            h[j] = 0
    
    return h

When we swap h[i] from 1 to 0 and h[j] from 0 to 1:
- For each lag k, R(k) changes by:

  - Adding contribution of position j as a "1": for each k where j+k < n: R(k) increases by (1-h[j+k])
  - Also the (1-h[.]) terms change: for each k where i-k ≥ 0: R(k) changes by...

When we add a 1 at position p:

  - R(k) += h[p-k] if p-k ≥ 0 [new 1 at p means 1-h[p] goes from 1 to 0, so R(k) decreases by h[p-k]... wait]

If we change h[p] from 0 to 1:
- For lag k ≥ 0: term i=p changes from 0 to (1-h[p+k]) if p ≤ n-1-k
  → R(k) increases by (1-h[p+k]) for k ≤ n-1-p
- For lag k ≥ 0: term i=p-k changes from h[p-k]*(1-0) to h[p-k]*(1-1) = 0 if p-k ≥ 0
  → R(k) decreases by h[p-k] for k ≤ p

So net change in R(k) when adding 1 at position p:
ΔR(k) = (1-h[p+k]) * [k ≤ n-1-p] - h[p-k] * [k ≤ p] (for k ≥ 0)

By symmetry, similar formula.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_max_corr(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)), corr

def smoothed_obj_grad(h, n, t):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted = np.clip(shifted, -400.0, 400.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    obj = (max_c + np.log(sum_exp) / t) * dx
    w = exp_v / sum_exp
    grad = np.zeros(n)
    n_corr = len(corr)
    center = n - 1
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - center
        if lag >= 0:
            grad[lag:] += wi * omh[:n - lag]
            grad[:n - lag] -= wi * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wi * omh[l:]
            grad[l:] -= wi * h[:n - l]
    grad *= dx
    return obj, grad

def subgradient_anneal(h, n, n_iter, t_start, t_end, step_scale=1.0):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    # Polyak averaging
    avg_h = h * 0.0
    gamma = step_scale / np.sqrt(n_iter)
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac if t_start > 0 else t_end
        t = max(t, 1.0)
        obj, grad = smoothed_obj_grad(h, n, t)
        step = gamma / (1.0 + it * 0.01)
        h_new = h - step * grad
        h_new = project_onto_constraint(h_new, n)
        avg_h += h_new
        h = h_new
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Also check the average
    avg_h /= n_iter
    avg_h = project_onto_constraint(avg_h, n)
    c5_avg = compute_c5(avg_h, n)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = avg_h.copy()
    
    return best_h, best_c5

def binary_local_search(h_bin, n, rng, n_swaps):
    h = h_bin.copy()
    current_max, _ = get_max_corr(h, n)
    best_h = h.copy()
    best_max = current_max
    n_ones = n // 2
    
    for it in range(n_swaps):

        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        new_max, _ = get_max_corr(h, n)
        if new_max <= current_max:

                best_max = new_max
                best_h = h.copy()
        else:

            h[j] = 0

                zeros = np.where(h == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i != j:
                    h[i], h[j] = h[j], h[i]
            current_max, _ = get_max_corr(h, n)
    
    return best_h, best_max

    """Create a structured binary pattern based on a sinusoid with random phase."""
    x = np.arange(n)
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(1.0, 2.0)
    vals = np.sin(2 * np.pi * freq * x / n + phase)
    h = (vals > 0).astype(float)

        for k in range(n_ones - n // 2):

        order = np.argsort(-vals[zero_idx])
        for k in range(n // 2 - n_ones):
            h[zero_idx[order[k]]] = 1
    return h
