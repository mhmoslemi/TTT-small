    if n % 2 == 0:

        ones = np.where(h == 1)[0].copy()

            h[ones[0]] = 0
            ones = ones[1:]
        while len(ones) < target:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros)] = 1
            ones = np.where(h == 1)[0].copy()
    else:
        target = n // 2
        h = np.where(h >= 0.5, 1.0, 0.0)
        ones = np.where(h == 1)[0].copy()
        while len(ones) > target:
            h[ones[0]] = 0
            ones = ones[1:]
        while len(ones) < target:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros)] = 1
            ones = np.where(h == 1)[0].copy()
        # Set one 1 to 0.5 to get sum = n/2 = (n-1)/2 + 0.5
        ones = np.where(h == 1)[0]
        if len(ones) > 0:
            h[rng.choice(ones)] = 0.5

    best_score = score_c5(h)
    t_start = time.time()

        improved = False

        if k_offset > 0:
            k = k_offset
            order = rng.permutation(n - k)
            for i in order:
                if h[i] > 0.5 and h[i + k] < 0.5:
                    h[i], h[i + k] = h[i + k], h[i]
                    s = score_c5(h)
                    if s < best_score - 1e-12:
                        best_score = s
                        improved = True
                        break
                    h[i], h[i + k] = h[i + k], h[i]
        elif k_offset < 0:
            k = -k_offset
            order = rng.permutation(n - k) + k
            for i in order:
                if h[i] > 0.5 and h[i - k] < 0.5:
                    h[i], h[i - k] = h[i - k], h[i]
                    s = score_c5(h)
                    if s < best_score - 1e-12:
                        best_score = s
                        improved = True
                        break
                    h[i], h[i - k] = h[i - k], h[i]

            low = np.where(h < 0.5)[0]

                break

                    best_pair = (i, j)

                h[i], h[j] = h[j], h[i]
                best_score = best_s
                improved = True

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

def lse_obj(x, T):

    dx = 2.0 / n

    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    return dx * (cmax + T * np.log(np.sum(np.exp(scaled))))

                    if s < best_score - 1e-12:
                        best_score = s
                        improved = True
                        break
                    h[i], h[i - k] = h[i - k], h[i]

        if not improved:
            high = np.where(h > 0.5)[0]
            low = np.where(h < 0.5)[0]
            if len(high) == 0 or len(low) == 0:
                break
            nh, nl = len(high), len(low)
            n_pairs = nh * nl
            order = rng.permutation(n_pairs)
            best_s = best_score
            best_pair = None
            for p in order:
                i = high[p // nl]
                j = low[p % nl]
                h[i], h[j] = h[j], h[i]
                s = score_c5(h)
                h[i], h[j] = h[j], h[i]
                if s < best_s - 1e-12:
                    best_s = s
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i], h[j] = h[j], h[i]
                best_score = best_s
                improved = True

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

    return h, best_score


def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / _CURRENT_T
    return dx * (cmax + _CURRENT_T * np.log(np.sum(np.exp(scaled))))


def sum_con(x):
    return np.sum(x) - len(x) / 2.0


def sum_con_jac(x):
    return np.ones(len(x))


def slsqp_polish(h, T, max_iter=3000):
    global _CURRENT_T
    _CURRENT_T = T
    n = len(h)
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            x_new = project_feasible(res.x)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _CURRENT_T
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    n_list = [86, 84, 88, 82, 90]
    phase1_end = t0 + budget_s * 0.40

    for n in n_list:
        if time.time() > phase1_end:
            break
        time_left = phase1_end - time.time()
        if time_left < 10:
            break
        n_starts = max(2, int(time_left / 20))
        per_start = time_left / n_starts
        for _ in range(n_starts):
            if time.time() > phase1_end:
                break
            h_init = rng.random(n)
            h_try, s_try = binary_local_search(h_init, n, per_start, rng)
            if s_try < best_c5:
                best_c5 = s_try
                best_h = h_try.copy()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        h_parent = project_feasible(h_parent)
        s_parent = score_c5(h_parent)
        if s_parent < best_c5:
            best_c5 = s_parent
            best_h = h_parent.copy()

        if time.time() < t0 + budget_s * 0.60:
            h_cur = h_parent.copy()
            for T in [1.0, 0.4, 0.15, 0.05, 0.02]:
                if time.time() > t0 + budget_s * 0.60:
                    break
                h_new, s_new = slsqp_polish(h_cur, T, 3000)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = h_new.copy()
                h_cur = h_new

    if best_h is not None:
        n_best = len(best_h)
        phase3_end = t0 + budget_s * 0.85
        hops_done = 0
        while time.time() < phase3_end and hops_done < 50:
            hops_done += 1
            h_pert = best_h.copy()
            n_pert = rng.randint(2, 7)
            idx = rng.choice(n_best, size=n_pert, replace=False)
            h_pert[idx] += rng.uniform(-0.35, 0.35, size=n_pert)
            h_pert = project_feasible(h_pert)
            h_new, s_new = slsqp_polish(h_pert, 0.1, 800)
            if n_best % 2 == 0:
                h_bin, s_bin = binary_local_search(h_pert, n_best, 3.0, rng)
                if s_bin < s_new:
                    h_new, s_new = h_bin, s_bin
            if s_new < best_c5:
                best_c5 = s_new
                best_h = h_new.copy()

    if best_h is not None and time.time() < t0 + budget_s - 15:
        n_best = len(best_h)
        unique_vals = np.unique(best_h)
        is_binary = all(np.abs(v) < 1e-10 or np.abs(v - 1) < 1e-10 or np.abs(v - 0.5) < 1e-10 for v in unique_vals)
        if is_binary:
            tl = min(80, t0 + budget_s - 20 - time.time())
            if tl > 5:
                h_final, s_final = binary_local_search(best_h, n_best, tl, rng)
                if s_final < best_c5:
                    best_c5 = s_final
                    best_h = h_final.copy()
                    n_best = len(best_h)
        if time.time() < t0 + budget_s - 8:
            for T in [0.05, 0.01, 0.005]:
                if time.time() > t0 + budget_s - 8:
                    break
                h_new, s_new = slsqp_polish(best_h, T, 2000)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = h_new.copy()

    if best_h is None:
        n = 85
        best_h = project_feasible(rng.random(n))
        best_c5 = score_c5(best_h)
    else:
        best_h = project_feasible(best_h)
        best_c5 = score_c5(best_h)

    return best_h, best_c5, len(best_h)
