    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):

        if abs(diff) < 1e-9:

            count = mask.sum()
            if count == 0:
                break

            h[mask] = np.maximum(0.0, h[mask] - step)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            count = int(mask.sum())
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            count = int(mask.sum())
            if count == 0:
                break
            step = -diff / count
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

    T_min = 0.05

        if w_sum < 1e-30:
