Let me also think about a key trick: **alternating minimization over the active set**.

At the optimum, multiple lags k achieve the maximum simultaneously. If I can identify which lags are active, I can solve a simpler subproblem: minimize the maximum of just those few linear combinations.

For a fixed active set {k1, k2, ..., km}, the subproblem is:
minimize max_i f_{ki}(h)
s.t. 0 ≤ h ≤ 1, sum(h) = n/2

This can be solved by:
minimize t
s.t. f_{ki}(h) ≤ t for all i
     0 ≤ h ≤ 1
     sum(h) = n/2

But f_k is not linear in h (it's quadratic), so this isn't a LP.

However, I can linearize: f_k(h) = sum_j h[j](1-h[j-s]) 
If I fix h[j-s] (from the previous iteration), then f_k becomes linear in h.

This gives an alternating approach:
1. Fix all h[j-s] terms
2. Solve the linear subproblem (now it's a min of linear functions, which is a LP)
3. Update
4. Repeat

This is essentially the **Successive Linear Programming** or **cutting plane** approach.

If I change h[i] by δ:

import numpy as np
from scipy.signal import fftconvolve
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], 'full') * dx
    return corr

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = ex.sum()
    obj = cmax + eps * np.log(sexp)
    w = ex / sexp
    
    omh = 1.0 - h
    s_vals = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i - s_vals
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.dot(w[valid1], omh[idx1[valid1]])
        idx2 = i + s_vals
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.dot(w[valid2], h[idx2[valid2]])
        grad[i] = dx * (term1 - term2)
    
    return obj, grad

def optimize_pg(h_init, target_sum, eps_list, max_iters_per_eps, time_limit):
    n = len(h_init)
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for eps in eps_list:
        if time.time() - start > time_limit:
            break
        
        h_curr = h.copy()
        obj_curr = compute_c5(h_curr)
        
        for it in range(max_iters_per_eps):
            if time.time() - start > time_limit:
                break
            
            obj, grad = smooth_obj_grad(h_curr, eps)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            
            # Backtracking line search
            alpha = 1.0 / (gnorm + 1e-10)
            for _ in range(30):
                h_try = project_onto_constraint(h_curr - alpha * grad, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < obj_curr:
                    break
                alpha *= 0.5
            else:
                break  # No improvement possible
            
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            
            h_curr = h_try
            obj_curr = c5_try
    
    return best_h, best_c5

def upsample(h, factor=2):
    """Upsample h by interpolation."""

import numpy as np
from scipy.signal import fftconvolve

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], 'full') * dx
    return corr

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = ex.sum()
    obj = cmax + eps * np.log(sexp)
    w = ex / sexp
    omh = 1.0 - h
    s_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i - s_vals
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.dot(w[valid1], omh[idx1[valid1]])
        idx2 = i + s_vals
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.dot(w[valid2], h[idx2[valid2]])
        grad[i] = dx * (term1 - term2)
    return obj, grad

            h_init = gen_blocks(n0, rng)
            alpha = 0.3 + 0.3 * rng.random()
            h_init = alpha * best_h + (1.0 - alpha) * h_init
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        eps_local = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        tl = min(hard_limit - time.time() - 180, 120)
        if tl < 20:
            break
        h_opt, c5_opt = optimize_multi_eps(h_init, n0 / 2.0, eps_local, 2000, time.time() + tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time.time() - start_time < hard_limit - 250:
        n1 = 2 * n0
        h_up = upsample(best_h, 2)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        eps_up = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        tl = min(hard_limit - time.time() - 200, 400)
        h_opt, c5_opt = optimize_multi_eps(h_up, n1 / 2.0, eps_up, 2000, time.time() + tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time.time() - start_time < hard_limit - 100:
        n_now = len(best_h)
        eps_fine = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        tl = min(hard_limit - time.time() - 50, 300)
        h_opt, c5_opt = optimize_multi_eps(best_h, n_now / 2.0, eps_fine, 5000, time.time() + tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time.time() - start_time < hard_limit - 20:
        h_pert = gen_perturb(best_h, rng, 0.02)
        h_pert = project_onto_constraint(h_pert, len(h_pert) / 2.0)
        eps_polish = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        tl = hard_limit - time.time()
        h_opt, c5_opt = optimize_multi_eps(h_pert, len(h_opt) / 2.0, eps_polish, 5000, time.time() + max(tl, 10), start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
