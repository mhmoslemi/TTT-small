            alpha_start=200.0, alpha_end=600.0
        )
        if score_res < best_score:
            best_score = score_res
            best_h = h_res

    h_final = _project(best_h, target_sum)
    final_score = _compute_c5(h_final)

    return h_final, final_score, len(h_final)
