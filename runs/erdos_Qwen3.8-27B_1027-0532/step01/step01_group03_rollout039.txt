import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def _slsqp_optimize(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _coordinate_descent(h, n, target_sum, n_iter):
    """Coordinate descent with sum constraint maintained by adjusting a pair."""
    best_c5 = _c5_of(h)
    for it in range(n_iter):
        improved = False
        for i in range(n):
            # Try to adjust h[i] slightly, compensating with h[j] for some j
            # Find j with most room for compensation
            for direction in [1, -1]:
                # Try moving h[i] by delta in direction
                delta = 0.01 * direction
                if 0 <= h[i] + delta <= 1:
                    h[i] += delta
                    # Compensate: move h[j] by -delta in the same direction
                    # Pick j = (i+1) % n or the one with most room
                    j = (i + 1) % n
                    if direction == 1 and h[j] >= delta:  # h[i] increased, h[j] decreases
                        h[j] -= delta
                        new_c5 = _c5_of(h)
                        if new_c5 < best_c5 - 1e-12:
                            best_c5 = new_c5
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                    elif direction == -1 and h[j] <= 1 - delta:  # h[i] decreased, h[j] increases
                        h[j] += delta
                        new_c5 = _c5_of(h)
                        if new_c5 < best_c5 - 1e-12:
                            best_c5 = new_c5
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
        if not improved:
            break
    return h, best_c5

        target_sum = n / 2.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def _slsqp_optimize(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _harmonic_h(params, x, n_harm):
    h = np.full(len(x), 0.5)
    for m in range(1, n_harm + 1):
        if 2*(m-1) < len(params):
            h += params[2*(m-1)] * np.cos(m * np.pi * x)
        if 2*(m-1)+1 < len(params):
            h += params[2*(m-1)+1] * np.sin(m * np.pi * x)
    return h

    viol = np.maximum(0, -h) + np.maximum(0, h - 1)
    c5 = _c5_of(np.clip(h, 0, 1))
    return c5 + penalty * np.sum(viol**2)

    else:

def _coordinate_descent(h, n, target_sum, n_sweeps, t_start, budget_s):
    """Simple coordinate descent with pair swaps to maintain sum."""
    def elapsed_ok(margin=10):
        return time.time() < t_start + budget_s - margin

    for sweep in range(n_sweeps):
        if not elapsed_ok(5):
            break
        improved = False
        for i in range(n):
            if not elapsed_ok(5):
                break
            # Try small adjustments to h[i], compensated by h[j]
            for j in [(i+1) % n, (i-1) % n, (i + n//2) % n]:
                if j == i:
                    continue
                for delta in [0.005, 0.001, -0.005, -0.001]:
                    new_hi = h[i] + delta
                    new_hj = h[j] - delta
                    if 0 <= new_hi <= 1 and 0 <= new_hj <= 1:
                        h[i] = new_hi
                        h[j] = new_hj
                        c5_try = _c5_of(h)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            improved = True
                        else:
                            h[i] = new_hi - delta
                            h[j] = new_hj + delta
        if not improved:
            break
    return h, best_c5

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(corr, alpha):
    m = np.max(corr)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    except Exception:
        return h_start

def _penalty_c5_obj(h, n, target_C, sum_pen):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    overlaps = corr * dx
    excess = np.maximum(0.0, overlaps - target_C)
    obj = np.sum(excess * excess)
    sv = h.sum() - target_sum_global
    obj += sum_pen * sv * sv
    return obj

target_sum_global = 43.0

def _penalty_slsqp(h_start, n, target_sum, target_C, sum_pen, maxiter):
    global target_sum_global
    target_sum_global = target_sum
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        overlaps = corr * dx
        excess = np.maximum(0.0, overlaps - target_C)
        o = np.sum(excess * excess)
        sv = x.sum() - target_sum
        return o + sum_pen * sv * sv
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10})
        return res.x
    except Exception:
        return h_start

def _subgrad_step(h, n, target_sum, step_size):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    max_idx = np.argmax(corr)
    k = max_idx
    sg = np.zeros(n)
    if k < n:
        s = k
        for i in range(n - s):
            sg[i] += (1.0 - h[i + s]) - h[i]
            sg[i + s] += -h[i]
    else:
        s = k - (n - 1)
        i_min = max(0, -s)
        i_max = min(n - 1, n - 1 - s)
        for i in range(i_min, i_max + 1):
            if s == 0:
                sg[i] += (1.0 - h[i]) - h[i]
            else:
                sg[i] += (1.0 - h[i + s])
            if s != 0 and i + s < n:
                sg[i + s] += -h[i]
    h_new = h - step_size * dx * sg
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _fix_sum(h_new, target_sum)
    return h_new

def _coord_desc(h, n, target_sum, n_sweeps, deadline):
    best_c5 = _c5_of(h)
    deltas = [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]
    for sweep in range(n_sweeps):
        if time.time() > deadline - 10:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 10:
                break
            j = (i + 1) % n
            if j == i:
                continue
            for d in deltas:
                hi_new = h[i] + d
                hj_new = h[j] - d
                if 0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0:
                    h[i] = hi_new
                    h[j] = hj_new
                    c5_try = _c5_of(h)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        improved = True
                    else:
                        h[i] = hi_new - d
                        h[j] = hj_new + d
        if not improved:
            break
    return h, best_c5

def _refine_slsqp(h, n, target_sum, deadline):
    h_cur = h.copy()
    alphas = [50, 100, 200, 400, 800, 1500, 3000]
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        h_new = _slsqp_smooth(h_cur, n, target_sum, alpha, maxiter=400, ftol=1e-15)
        h_new = _fix_sum(h_new, target_sum)
        if _c5_of(h_new) <= _c5_of(h_cur) + 1e-12:
            h_cur = h_new
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 86

    n_candidates = [86]
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_init = len(initial_h_values)
        if n_init not in n_candidates:
            n_candidates.insert(0, n_init)

    for n in n_candidates:
        if time.time() > deadline - 80:
            break
        target_sum = n / 2.0

        if 'initial_h_values' in globals() and initial_h_values is not None and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float).copy()
        else:
            h0 = np.full(n, 0.5)
        h0 = _fix_sum(h0, target_sum)

        # Phase 1: Main SLSQP optimization
        h_cur = _refine_slsqp(h0, n, target_sum, deadline)
        c5_cur = _c5_of(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

        # Phase 2: Subgradient steps to escape local optima
        if time.time() < deadline - 60:
            for step_sz in [0.01, 0.005, 0.002]:
                if time.time() > deadline - 40:
                    break
                for it in range(200):
                    if time.time() > deadline - 30:
                        break
                    h_sg = _subgrad_step(h_cur, n, target_sum, step_sz)
                    c5_sg = _c5_of(h_sg)
                    if c5_sg < _c5_of(h_cur):
                        h_cur = h_sg
                    elif it > 50:
                        break
            # Re-polish after subgradient
            if time.time() < deadline - 30:
                h_cur = _refine_slsqp(h_cur, n, target_sum, deadline)
                c5_cur = _c5_of(h_cur)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n

        # Phase 3: Penalty method on active set
        if time.time() < deadline - 50:
            current_c5 = _c5_of(h_cur)
            target_C = current_c5 * 0.995
            for pen_round in range(10):
                if time.time() > deadline - 20:
                    break
                for sum_pen in [10.0, 100.0]:
                    if time.time() > deadline - 20:
                        break
                    h_pen = _penalty_slsqp(h_cur, n, target_sum, target_C, sum_pen, maxiter=300)
                    h_pen = _fix_sum(h_pen, target_sum)
                    c5_pen = _c5_of(h_pen)
                    if c5_pen < _c5_of(h_cur):
                        h_cur = h_pen
                    else:
                        break
                # Re-refine
                if time.time() < deadline - 20:
                    h_cur = _refine_slsqp(h_cur, n, target_sum, deadline)
                    c5_cur = _c5_of(h_cur)
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h_cur.copy()
                        best_n = n
                    target_C = c5_cur * 0.998

        # Phase 4: Coordinate descent polish
        if time.time() < deadline - 20:
            h_cd, c5_cd = _coord_desc(h_cur, n, target_sum, 30, deadline)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n
                h_cur = h_cd
                # Final SLSQP polish
                if time.time() < deadline - 10:
                    h_final = _refine_slsqp(h_cur, n, target_sum, deadline)
                    c5_final = _c5_of(h_final)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final.copy()
                        best_n = n

        # Phase 5: Random restarts
        for r in range(3):
            if time.time() > deadline - 30:
                break
            h_rand = np.clip(0.5 + 0.15 * rng.randn(n), 0.0, 1.0)
            h_rand = _fix_sum(h_rand, target_sum)
            h_rand = _refine_slsqp(h_rand, n, target_sum, deadline)
            c5_rand = _c5_of(h_rand)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h_rand.copy()
                best_n = n

    # Try alternative n values
    for n_alt in [84, 88, 90]:
        if time.time() > deadline - 60:
            break
        if n_alt == (len(initial_h_values) if 'initial_h_values' in globals() and initial_h_values is not None else -1):
            continue
        target_sum = n_alt / 2.0
        h0 = np.full(n_alt, 0.5)
        h_alt = _refine_slsqp(h0, n_alt, target_sum, deadline)
        c5_alt = _c5_of(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt
        # Subgradient + polish
        if time.time() < deadline - 30:
            for step_sz in [0.005, 0.002]:
                if time.time() > deadline - 15:
                    break
                for it in range(100):
                    if time.time() > deadline - 15:
                        break
                    h_sg = _subgrad_step(h_alt, n_alt, target_sum, step_sz)
                    if _c5_of(h_sg) < _c5_of(h_alt):
                        h_alt = h_sg
            if time.time() < deadline - 15:
                h_alt = _refine_slsqp(h_alt, n_alt, target_sum, deadline)
                c5_alt = _c5_of(h_alt)
                if c5_alt < best_c5:
                    best_c5 = c5_alt
                    best_h = h_alt.copy()
                    best_n = n_alt

    if best_h is None:
        best_h = np.full(86, 0.5)
        best_c5 = _c5_of(best_h)
        best_n = 86

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
