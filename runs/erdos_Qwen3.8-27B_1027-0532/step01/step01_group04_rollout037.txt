Actually, for the smooth max objective, I can compute the gradient analytically:
∂/∂h[i] smooth_max(corr, alpha) = Σ_k softmax_k * ∂corr[k]/∂h[i]

where softmax_k = exp(alpha*(corr[k] - max)) / Σ_j exp(alpha*(corr[j] - max))

And ∂corr[k]/∂h[i] can be computed:
corr[k_offset] = Σ_j h[j](1-h[j+k])
∂corr[k_offset]/∂h[i] = (1-h[i+k]) if the term h[i] appears as first factor
                        + h[i-k]*(−1) if the term (1-h[i]) appears (i.e., h[i] is the second factor)

More precisely, for corr at lag k (where corr is the 'full' correlation):
corr[n-1+k] = Σ_{j: 0≤j<n, 0≤j+k<n} h[j] * (1-h[j+k]) for k≥0
corr[n-1-k] = Σ_{j: 0≤j<n, 0≤j-k<n} h[j] * (1-h[j-k]) for k>0

∂corr[n-1+k]/∂h[i] for k > 0:
= (1-h[i+k]) if i+k < n (h[i] is first factor at position i)
  + (-1)*h[i-k] if i-k ≥ 0 (h[i] appears as (1-h[i]) when j+k = i, i.e., j = i-k)
= (1-h[i+k])_{i+k<n} - h[i-k]_{i-k≥0}

Similarly for k < 0.

For k=0: corr[n-1] = Σ_j h[j](1-h[j]), ∂/∂h[i] = 1-2h[i]

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def interp_and_project(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def compute_grad_corr(h_ref):
    """Compute the Jacobian of the correlation vector with respect to h.
    Returns a matrix G where G[k, i] = d corr_full[k] / d h[i].
    corr_full has 2n-1 entries."""
    n = len(h_ref)
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        l = m - n + 1
        if l >= 0:
            # g[i] = (1-h[i+l]) if i < n-l, else 0, minus h[i-l] if i >= l, else 0
            for i in range(n-l):
                G[m, i] += (1.0 - h_ref[i+l])
            for i in range(l, n):
                G[m, i] -= h_ref[i-l]
        elif l < 0:
            lp = -l
            for i in range(lp, n):
                G[m, i] += (1.0 - h_ref[i-lp])
            for i in range(n-lp):
                G[m, i] -= h_ref[i+lp]
        else:
            G[m, :] = 1.0 - 2.0 * h_ref
    return G
```

```python
def run_lp_iteration(h_ref, solver='CLARABEL'):
    n = len(h_ref)
    corr = np.correlate(h_ref, 1.0 - h_ref, mode='full')
    G = compute_grad_corr(h_ref)
    a = corr - G @ h_ref  # a[m] = corr[m] - G[m,:] @ h_ref
    
    h_var = cp.Variable(n)
    t_var = cp.Variable(scalar=True)

        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        G @ h_var <= t_var - a
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
```
