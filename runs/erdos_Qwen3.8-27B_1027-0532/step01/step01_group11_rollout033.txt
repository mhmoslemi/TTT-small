def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    """Compute slopes of corr[k] w.r.t. h[i] for all k (linear terms only, exclude k=n-1)"""
    n = len(h)

        s = k - (n-1)
        if s == 0:
            continue  # handle separately
        sl = 0.0

        jp = i + s
        if 0 <= js < n:
            sl += (1.0 - h[js])
        if 0 <= jp < n:
            sl -= h[jp]
        slopes[k] = sl

def optimize_coordinate(h, i):
    """Optimize h[i] given all other coordinates fixed."""
    n = len(h)
    h_old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute slopes (linear terms, exclude k=n-1)
    s_vals = np.arange(2*n-1) - (n-1)
    # slope_k = (1-h[i-s_k])*[0<=i-s_k<n] - h[i+s_k]*[0<=i+s_k<n]
    # for s_k != 0
    
    slopes = np.zeros(2*n-1)
    for idx, s in enumerate(s_vals):
        if s == 0:
            continue
        js = i - int(s)
        jp = i + int(s)
        if 0 <= js < n:
            slopes[idx] += (1.0 - h[js])
        if 0 <= jp < n:
            slopes[idx] -= h[jp]
    
    # A[k] = corr[k] - slopes[k]*h_old (intercept when t=0)
    # B[k] = slopes[k] (coefficient of t)
    # For k=n-1, the term is quadratic: zl_const + t - t^2
    mask = np.ones(2*n-1, dtype=bool)
    mask[n-1] = False
    
    A = corr[mask] - slopes[mask] * h_old
    B = slopes[mask]
    zl_const = corr[n-1] - h_old * (1.0 - h_old)
    
    # Binary search on T
    val0 = max(np.max(A) if len(A) > 0 else 0.0, zl_const)
    val1 = max(np.max(A + B) if len(A+B) > 0 else 0.0, zl_const)
    T_lo = min(val0, val1, 0.0)
    T_hi = max(val0, val1)
    
    pos_mask = B > 1e-15
    neg_mask = B < -1e-15
    
    for _ in range(50):
        T_mid = (T_lo + T_hi) / 2.0
        # Feasibility check
        ub_val = 1.0
        lb_val = 0.0
        if np.any(pos_mask):
            ub_candidates = (T_mid - A[pos_mask]) / B[pos_mask]
            ub_val = min(1.0, np.min(ub_candidates))
        if np.any(neg_mask):
            lb_candidates = (T_mid - A[neg_mask]) / B[neg_mask]
            lb_val = max(0.0, np.max(lb_candidates))
        
        zero_mask = ~(pos_mask | neg_mask)
        if np.any(zero_mask) and np.max(A[zero_mask]) > T_mid + 1e-12:
            T_lo = T_mid
            continue
        
        if lb_val > ub_val + 1e-12:
            T_lo = T_mid
            continue
        
        t_lo = max(lb_val, 0.0)
        t_hi = min(ub_val, 1.0)
        if t_lo > t_hi:
            T_lo = T_mid
            continue
        
        # Check zero-lag
        t_peak = 0.5
        t_checks = [t_lo, t_hi, min(max(t_peak, t_lo), t_hi)]
        f_max = max(zl_const + tc - tc*tc for tc in t_checks)
        if f_max > T_mid + 1e-12:
            T_lo = T_mid
            continue
        
        T_hi = T_mid
    
    # Find t at T_hi
    T = T_hi + 1e-12
    ub_val = 1.0
    lb_val = 0.0
    if np.any(pos_mask):
        ub_candidates = (T - A[pos_mask]) / B[pos_mask]
        ub_val = min(1.0, np.min(ub_candidates))
    if np.any(neg_mask):
        lb_candidates = (T - A[neg_mask]) / B[neg_mask]
        lb_val = max(0.0, np.max(lb_candidates))
    
    t_lo = max(lb_val, 0.0)
    t_hi = min(ub_val, 1.0)
    t_opt = (t_lo + t_hi) / 2.0
    t_opt = np.clip(t_opt, 0.0, 1.0)
    
    # Verify and possibly refine
    h[i] = t_opt
    return t_opt

def coordinate_descent(h, start_time, time_limit, max_sweeps=2000):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        for i in range(n):
            optimize_coordinate(h, i)
        h = project_onto_constraint(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    # gradient of log-sum-exp with respect to h
    # d/dh[i] of corr[k] for k = n-1+s:
    #   = (1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n] (for s≠0)
    #   = 1-2*h[i] (for s=0, but this is for the linear part; actual is derivative of h*(1-h) = 1-2h)
    
    grad = np.zeros(n)
    s_vals = np.arange(2*n-1) - (n-1)
    for i in range(n):
        for idx in range(2*n-1):
            s = s_vals[idx]
            if w[idx] < 1e-15:
                continue
            sl = 0.0
            if s == 0:
                sl = 1.0 - 2.0 * h[i]
            else:
                si = int(s)
                js = i - si
                jp = i + si
                if 0 <= js < n:
                    sl += (1.0 - h[js])
                if 0 <= jp < n:
                    sl -= h[jp]
            grad[i] += w[idx] * dx * sl
    return grad

def slsqp_refine(h, eps, maxiter):
    n = len(h)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(len(x))}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
