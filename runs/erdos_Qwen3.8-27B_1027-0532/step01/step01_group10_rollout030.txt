    j_end = i + n - 1

def repair_h(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)

    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

        t_phase3 = t0 + min(880, budget_s * 0.68)
        h_cd512, c_cd512 = coord_desc(h512, n512, t_phase3, rng)
        if c_cd512 < best_c5:
            best_h = h_cd512
            best_c5 = c_cd512
            best_n = n512
    
    # Phase 4: Coordinate descent on best solution (remaining - 100s)
    t_phase4 = t0 + budget_s - 100
    h_cd2, c_cd2 = coord_desc(best_h.copy(), best_n, t_phase4, rng)
    if c_cd2 < best_c5:
        best_h = h_cd2
