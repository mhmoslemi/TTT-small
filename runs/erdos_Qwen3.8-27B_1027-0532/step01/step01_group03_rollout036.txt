The objective f(h) = max_k c_k * dx is convex in h? No, it's not convex because c_k is bilinear.

But for a fixed h, the subgradient with respect to h is well-defined (it's the gradient of the active constraint). We can use:
1. Compute c_k for all k
2. Find the set of "active" lags (those with c_k close to max)
3. Compute the average gradient over active lags
4. Take a step in the negative gradient direction
5. Project back to the feasible set (0 ≤ h ≤ 1, sum = n/2)

The projection onto {h: 0 ≤ h ≤ 1, sum(h) = n/2} can be done efficiently:
- Project h onto the hyperplane sum(h) = n/2 (by subtracting the mean difference)
- Clip to [0, 1]
- Iterate until convergence (this is the Euclidean projection onto the intersection of a box and a hyperplane)

This is similar to "binary search on the Lagrange multiplier" for the equality constraint.

∂c_k/∂h[j] = sum over terms where j appears:

For the max of c_k over k, the subgradient is the set of convex combinations of the active gradients. We can use the average of the top m active gradients.

```python
    """Euclidean projection onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo = -1.0 - float(h.max())
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(100):

        if s < target_sum:

    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _subgrad(h, top_m):
    """Compute average subgradient of max c_k at the top_m active lags."""
    n = len(h)
    corr = _corr_of(h)
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - n + 1
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        i_range = np.arange(i_lo, i_hi)
        if len(i_range) == 0:
            continue
        grad[i_range] -= h[i_range + lag]
        grad[i_range + lag] += (1.0 - h[i_range])
    return grad / float(top_m)

def _subgrad_descent(h, n, target_sum, lr_init, momentum, n_iters, top_m=5, deadline=None):
    """Run subgradient descent with momentum and projection."""
    best_h = h.copy()
    best_c5 = _c5_of(h)
    prev_h = h.copy()
    
    for t in range(1, n_iters + 1):
        if deadline is not None and time.time() > deadline:
            break
        
        lr = lr_init / np.sqrt(float(t))
        g = _subgrad(h, top_m)
        y = h + momentum * (h - prev_h)
        h_new = y - lr * g
        h_new = _project_box_sum(h_new, target_sum)
        
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        h = h_new
```

- For each coordinate i, fix all other coordinates.
- The objective becomes: max_k [c_k_base + (delta_i * contribution)]

For coordinate i, the lags that include it are:
- k such that 0 ≤ i < n and 0 ≤ i+lag < n (i appears as the "i" in the product)

2. Checking the endpoints x=0 and x=1
3. The minimum of a convex piecewise-linear function on [0,1] is at a point where the subgradient contains 0, or at the boundary.

```python
def _optimize_coordinate_i(h, i, n):
    """Optimize h[i] while keeping other coordinates fixed."""
    # Compute a_k and b_k for all k
    corr = _corr_of(h)  # current correlation values
    # For coordinate i, the coefficient a_k is:
    # a_k = (1-h[i-lag])*I[0<=i-lag<n] - h[i+lag]*I[0<=i+lag<n]
    # where lag = k - n + 1
    
    a = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - n + 1
        coeff = 0.0
        if 0 <= i - lag < n:
            coeff += (1.0 - h[i - lag])
        if 0 <= i + lag < n:
            coeff -= h[i + lag]
        a[k] = coeff
    
    # b_k = corr[k] - h[i] * a[k]
    b = corr - h[i] * a
    
    # Now minimize max_k (b[k] + x * a[k]) over x in [0, 1]
    # This is a 1D problem. The optimal x is where the subgradient contains 0.
    # Subgradient at x: conv{a[k] : b[k] + x*a[k] = max_j (b[j] + x*a[j])}
    
    # Find x* by bisection on the subgradient
    def subgrad_at_x(x):
        vals = b + x * a
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        return np.min(a[active]), np.max(a[active])
    
    # Bisection: find x where subgradient straddles 0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s_lo, s_hi = subgrad_at_x(mid)
        if s_lo > 0:
            # All active a_k > 0, so the function is increasing → move left
            hi = mid
        elif s_hi < 0:
            # All active a_k < 0, so the function is decreasing → move right
            lo = mid
        else:
            # 0 is in the subgradient → optimal
            lo = hi = mid
            break
    
    x_opt = (lo + hi) * 0.5
    h[i] = x_opt
    return h
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(h, target_sum):
    lo = -1.0 - float(np.max(h))
    hi = 1.0 - float(np.min(h)) + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _subgrad_step(h, top_m=5):
    """Compute one subgradient step direction (without projection)."""
    n = len(h)
    corr = _corr_of(h)
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - n + 1
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo >= i_hi:
            continue
        i_range = np.arange(i_lo, i_hi)
        grad[i_range] -= h[i_range + lag]
        grad[i_range + lag] += (1.0 - h[i_range])
    return grad / float(top_m)

def _subgrad_run(h, n, target_sum, n_iters, lr_init, momentum, top_m, deadline):
    """Run subgradient descent with momentum."""
    best_h = h.copy()
    best_c5 = _c5_of(h)
    prev = h.copy()

        if time.time() > deadline:

        lr = lr_init / np.sqrt(float(t))
        g = _subgrad_step(h, top_m)
        y = h + momentum * (h - prev)
        h_new = y - lr * g
        h_new = _project_box_sum(h_new, target_sum)
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        prev = h
        h = h_new
    
    return best_h, best_c5
```

Actually, let me reconsider one thing. The `_coord_coeff` function computes a_k for all 2n-1 = 171 lags. This involves a loop over 171 iterations with simple arithmetic. In Python, this is slow. Let me vectorize it:

For coordinate i:
- lag ranges from -(n-1) to n-1
- For each lag, a_k = (1-h[i-lag])*I(0≤i-lag<n) - h[i+lag]*I(0≤i+lag<n)

Let me define:
- For the first term: i-lag is valid when 0 ≤ i-lag < n, i.e., i < n and lag > i-n, and i-lag ≥ 0, i.e., lag ≤ i.
  So lag ranges from max(-(n-1), i-n+1) to min(n-1, i). Wait, let me be more careful.
  
  i-lag ≥ 0 → lag ≤ i
  i-lag < n → lag > i-n → lag ≥ i-n+1 (since lag is integer)
  
  Also lag ≥ -(n-1) and lag ≤ n-1.
  
  So lag ranges from max(-(n-1), i-n+1) to min(n-1, i).

- For the second term: i+lag is valid when 0 ≤ i+lag < n, i.e., lag ≥ -i and lag ≤ n-1-i.
  Combined with lag ∈ [-(n-1), n-1]:
  lag ranges from max(-(n-1), -i) to min(n-1, n-1-i).

Let me vectorize:

```python
def _coord_coeff_vec(h, i, n):
    a = np.zeros(2 * n - 1)
    k_indices = np.arange(2 * n - 1)
    lags = k_indices - (n - 1)
    
    # First term: (1-h[i-lag]) * I(0 <= i-lag < n)
    idx1 = i - lags
    mask1 = (idx1 >= 0) & (idx1 < n)
    a[mask1] += (1.0 - h[idx1[mask1]])
    
    # Second term: -h[i+lag] * I(0 <= i+lag < n)
    idx2 = i + lags
    mask2 = (idx2 >= 0) & (idx2 < n)
    a[mask2] -= h[idx2[mask2]]
    
    return a
```
