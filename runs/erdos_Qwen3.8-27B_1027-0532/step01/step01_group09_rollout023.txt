    m = corr.max()
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

        h_new = subgrad_opt(best, tm=min(30, tleft(0.18)), lr_init=0.5, beta_mom=0.9)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    # Phase 3: SLSQP polish
    if tleft(0.30) > 20:
        h_new = selpolish(best, tm=min(50, tleft(0.30)))
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    # Phase 4: Smooth gradient again after SLSQP (different basin)
    if tleft(0.35) > 15:
        h_new = projected_grad_descent(best, 50, tm=min(25, tleft(0.35)), lr0=0.02, momentum=0.4)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    # Phase 5: Try higher resolutions
    n_targets = [n_orig + 20, n_orig + 50, n_orig + 100, int(n_orig * 1.5), int(n_orig * 2), int(n_orig * 2.5), int(n_orig * 3)]
    n_targets = sorted(set(max(50, x) for x in n_targets))
    for n_t in n_targets:
        if tleft(0.78) < 20:
            break
        if n_t == n_orig:
            continue
        if n_t > 2000:
            continue
        h_res = resample_h(best, n_t)
        if tleft(0.78) > 15:
            h_new = anneal_smooth(h_res, n_betas=3, tm_total=min(30, tleft(0.78)))
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best, best_c5 = h_new, c5_new
        if tleft(0.78) > 10:
            h_new2 = subgrad_opt(best, tm=min(15, tleft(0.78)), lr_init=0.3, beta_mom=0.88)
            c5_new2 = eval_c5(h_new2)
            if c5_new2 < best_c5:
                best, best_c5 = h_new2, c5_new2
        if tleft(0.78) > 15 and len(best) <= 400:
            h_new3 = selpolish(best, tm=min(25, tleft(0.78)))
            c5_new3 = eval_c5(h_new3)
            if c5_new3 < best_c5:
                best, best_c5 = h_new3, c5_new3

    # Phase 6: Random restarts at best resolution
    n_best = len(best)
    n_restarts = max(1, int(tleft(0.88) // 18))
    for r in range(n_restarts):
        if tleft(0.88) < 12:
            break
        h_rand = make_structured_init(n_best, rng)
        h_new = anneal_smooth(h_rand, n_betas=3, tm_total=min(14, tleft(0.88)))
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new
            if tleft(0.88) > 12:
                h_new2 = selpolish(best, tm=min(12, tleft(0.88)))
                c5_new2 = eval_c5(h_new2)
                if c5_new2 < best_c5:
                    best, best_c5 = h_new2, c5_new2

    # Phase 7: Local perturbation search
    for attempt in range(10):
        if tleft(0.93) < 8:
            break
        scale = 0.03 + 0.01 * attempt
        h_pert = project_feasible(best + rng.normal(0, scale, best.size), len(best) / 2.0)
        h_new = projected_grad_descent(h_pert, 80, tm=min(10, tleft(0.93)), lr0=0.015, momentum=0.3)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    # Phase 8: Final intensive polish
    if tleft(0.99) > 20:
        h_new = selpolish(best, tm=min(50, tleft(0.97)))
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    if tleft(0.99) > 15:
        h_new = subgrad_opt(best, tm=min(25, tleft(0.99)), lr_init=0.25, beta_mom=0.93)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    if tleft(0.99) > 10:
        h_new = projected_grad_descent(best, 150, tm=min(15, tleft(0.99)), lr0=0.008, momentum=0.5)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    if tleft(0.99) > 8 and len(best) <= 300:
        h_new = selpolish(best, tm=min(10, tleft(0.99)))
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
