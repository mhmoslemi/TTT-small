    neg = slopes < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, np.min((v - intercepts[pos]) / slopes[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.max((v - intercepts[neg]) / slopes[neg]))
    
    if t_lo > t_hi:
        t_opt = (t_lo + t_hi) / 2.0
    else:
        t_opt = (t_lo + t_hi) / 2.0
    
    return np.clip(t_opt, lo, hi)

def coord_descent(h0, max_sweeps=500, deadline=None):
    """Coordinate descent to minimize max_k c_k(h)."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        for i in range(n):
            # Compute c_k with h[i] = 0
            hi_save = h[i]
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hi_save
            
            # slopes[m] for lag k = m - (n-1):
            # c_k = slope * h[i] + intercept
            # slope = 1 - h[i-k] - h[i+k] (with 0 for out-of-bounds)
            # intercept = corr0[m] (which is c_k with h[i]=0)
            
            slopes = np.ones(2*n - 1)
            for k in range(-(n-1), n):
                m = k + (n-1)
                s = 0.0
                if 0 <= i - k < n:
                    s += h[i - k]
                if 0 <= i + k < n:
                    s += h[i + k]
                slopes[m] = 1.0 - s
            
            intercepts = corr0
            
            t_opt = solve_1d_minmax(slopes, intercepts, 0.0, 1.0)
            h[i] = t_opt
        
        # Project to maintain sum constraint
        h = project_onto_constraint(h, n / 2.0)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def pgd_optimize(h0, n, target_sum, deadline, max_iter=5000):
    """Projected gradient descent with soft-max surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = 15.0
    lr = 0.03
    stall = 0
    
    def soft_max_obj_and_grad(h, tau):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        tau_corr = tau * corr
        max_tc = np.max(tau_corr)
        log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
        obj = (log_sum_exp / tau) * dx
        weights = np.exp(tau_corr - max_tc)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if weights[m] < 1e-14:
                continue
            lag = m - (n - 1)
            w = weights[m]
            j_lo = max(0, lag)
            j_hi = min(n - 1, n - 1 + lag)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
        return obj, grad * dx
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = soft_max_obj_and_grad(h, tau)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 40:
                tau *= 1.8
                lr = max(lr * 0.6, 0.002)
                stall = 0
                if lr < 0.002:
                    break
    return best_h, best_c5

def worst_case_pgd(h0, n, target_sum, deadline, max_iter=5000):
    """PGD targeting the worst lag."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.95
            if lr < 0.001:
                lr = 0.02
                h = best_h + 0.005 * np.random.randn(n)
                h = project_onto_constraint(h, target_sum)
    return best_h, best_c5
