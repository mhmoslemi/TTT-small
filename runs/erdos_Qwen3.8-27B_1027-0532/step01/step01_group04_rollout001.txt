    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]
    try:
        res = minimize(obj_smooth, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, N / 2.0)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    global _target_sum, _n_pts
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    N = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on parent grid (sequential)
    if time_left() > 30:
        cd_h, cd_c5 = coordinate_descent(best_h, n_sweeps=60, lam=0.0, random_order=False)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h

    # Phase 2: Coordinate descent with random order
    if time_left() > 30:
        cd_h, cd_c5 = coordinate_descent(best_h, n_sweeps=80, lam=0.0, random_order=True)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h

    # Phase 3: SLSQP polishing
    if time_left() > 30:
        for alpha in [50, 150, 400]:
            if time_left() < 30:
                break
            h_new, c5_new = slsqp_refine(best_h, alpha, maxiter=3000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Try different grid sizes
    n_candidates = [101, 121, 151, 181, 211, 251, 281, 331, 361, 421]
    for n_try in n_candidates:
        if time_left() < 80:
            break
        target_s = n_try / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, target_s)

        cd_h, cd_c5 = coordinate_descent(h_try, n_sweeps=60, lam=0.0, random_order=True)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h

        if time_left() > 30:
            h_new, c5_new = slsqp_refine(best_h, 150, maxiter=1500)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: More coordinate descent on best grid
    n_use = len(best_h)
    ts_use = n_use / 2.0
    _n_pts = n_use
    _target_sum = ts_use
    if time_left() > 40:
        cd_h, cd_c5 = coordinate_descent(best_h, n_sweeps=100, lam=0.0, random_order=True)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h

    # Phase 6: Random restarts
    noise_levels = [0.02, 0.04, 0.07, 0.1, 0.15]
    while time_left() > 50:
        noise = rng.choice(noise_levels)
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, ts_use)
        cd_h, cd_c5 = coordinate_descent(h_pert, n_sweeps=25, lam=0.0, random_order=True)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h
            n_use = len(best_h)
            ts_use = n_use / 2.0

    # Phase 7: Final SLSQP polish
    if time_left() > 20:
        for alpha in [200, 500, 1000]:
            if time_left() < 20:
                break
            h_new, c5_new = slsqp_refine(best_h, alpha, maxiter=3000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 8: Final coordinate descent polish
    if time_left() > 15:
        cd_h, cd_c5 = coordinate_descent(best_h, n_sweeps=40, lam=0.0, random_order=False)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = cd_h

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(h_final)
    return h_final, final_c5, len(h_final)
