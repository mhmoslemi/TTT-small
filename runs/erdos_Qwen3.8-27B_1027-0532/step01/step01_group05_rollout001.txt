import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def corr_all(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def lag_gradient_single(j, h, n):
    grad = np.zeros(n)
    lag = j - (n - 1)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] += -h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] += -h[:n - l]
    return grad

def subgradient_polish(h, n, n_iter=300, step_init=0.05):

def interpolate_to_grid(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    h_interp = project_onto_constraint(h_interp, n_new)
    return h_interp


def slsqp_optimize(h0, n, t, max_iter=300, top_k=30):
    dx = 2.0 / n
    h0 = project_onto_constraint(np.clip(h0, 0.0, 1.0), n)

    def obj_fun(hv):
        corr = corr_all(hv, n)
        max_corr = np.max(corr)
        K = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-K:]
        top_corr = corr[top_idx]
        shifted = t * (top_corr - max_corr)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        return (max_corr + np.log(np.sum(exp_vals)) / t) * dx

    def grad_fun(hv):
        corr = corr_all(hv, n)
        max_corr = np.max(corr)
        K = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-K:]
        top_corr = corr[top_idx]
        shifted = t * (top_corr - max_corr)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        grad = np.zeros(n)
        for k_idx in range(K):
            w = weights[k_idx]
            if w < 1e-8:
                continue
            j = top_idx[k_idx]
            lag = j - (n - 1)
            if lag == 0:
                grad += w * (1.0 - 2.0 * hv)
            elif lag > 0:
                grad[lag:] += w * (1.0 - hv[:n - lag])
                grad[:n - lag] += -w * hv[lag:]
            else:
                l = -lag
                grad[:n - l] += w * (1.0 - hv[l:])
                grad[l:] += -w * hv[:n - l]
        grad *= dx
        return grad

    bounds = [(0.0, 1.0)] * n
    eq_cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                'jac': lambda x: np.ones(n)}]
    try:
        result = minimize(obj_fun, h0.copy(), method='SLSQP', jac=grad_fun,
                          bounds=bounds, constraints=eq_cons,
                          options={'maxiter': max_iter, 'ftol': 1e-10})
        h_opt = project_onto_constraint(np.clip(result.x, 0.0, 1.0), n)
        return h_opt
    except Exception:
        return h0.copy()
