def slsqp_smooth(h0, eps, maxiter):

    n = len(h0)
    target_sum = n / 2.0

def make_binary_pattern(n, rng, strategy=0):
    """Generate a structured binary pattern with exactly n/2 ones"""
    if strategy == 0:
        # Random
        h = (rng.random(n) < 0.5).astype(float)
    elif strategy == 1:
        # Alternating blocks
        block = n // 10
        h = np.zeros(n)
        for i in range(n):
            if (i // block) % 2 == 0:
                h[i] = 1.0
    elif strategy == 2:
        # Low-frequency
        h = (0.5 + 0.4 * np.sin(2 * np.pi * 3 * np.arange(n) / n) + 
             0.3 * rng.standard_normal(n)).astype(float)
        h = (h > 0.5).astype(float)
    else:
        # Pairs
        h = np.zeros(n)
        for i in range(0, n, 4):
            if i + 1 < n:
                h[i] = 1.0
                h[i+1] = 1.0
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    if len(ones) > n // 2:
        excess = rng.choice(ones, size=len(ones) - n//2, replace=False)
        h[excess] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h == 0)[0]
        need = n // 2 - len(ones)
        fill = rng.choice(zeros, size=min(need, len(zeros)), replace=False)
        h[fill] = 1
    return h

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    return float(np.max(compute_corr(h)))

            idx1 = np.arange(s, n)
            grad[idx1] += (1.0 - h[idx1 - s])
            idx2 = np.arange(0, n - s)
            grad[idx2] -= h[idx2 + s]
    else:
        r = -s
        if r < n:
            idx1 = np.arange(0, n - r)
            grad[idx1] += (1.0 - h[idx1 + r])
            idx2 = np.arange(r, n)
            grad[idx2] -= h[idx2 - r]
    return grad * dx

def smooth_gradient(h, eps):

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * lag_gradient(h, k)
    return grad
