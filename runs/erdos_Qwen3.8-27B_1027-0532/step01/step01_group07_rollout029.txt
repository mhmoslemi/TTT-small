```python
def subgradient_step(h, n, deadline):
    """One subgradient step on the hard max."""
    c = compute_c5_all(h)
    idx_max = np.argmax(c)
    # subgradient is the gradient of c[idx_max]
    # c[idx_max] = dx * sum_k h[k] * (1-h[k+m]) where m = idx_max - (n-1)
    m = idx_max - (n - 1)
    grad = np.zeros(n)
    for k in range(n):
        # dc/dh[k] from first factor: dx * (1-h[k+m]) if valid
        if 0 <= k + m < n:
            grad[k] += dx * (1 - h[k+m])
        # dc/dh[k] from second factor: -dx * h[k-m] if valid
        if 0 <= k - m < n:
            grad[k] -= dx * h[k-m]
    return grad
```

```python
import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def project_feasible_warm(h, target, t_prev):
    t = t_prev
    proj = np.clip(h + t, 0.0, 1.0)
    s = proj.sum()
    if abs(s - target) < 1e-10:
        return proj, t
    lo = t - 0.5
    hi = t + 0.5
    for _ in range(50):
        t = (lo + hi) * 0.5
        proj = np.clip(h + t, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = t
        else:
            hi = t
        if hi - lo < 1e-12:
            break
    proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    return proj, (lo + hi) * 0.5


def compute_gradient(h, w):
    n = len(h)
    dx = 2.0 / n
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    m1 = rows + n - 1 - cols
    m2 = cols - rows + n - 1
    M1 = np.where((m1 >= 0) & (m1 < 2 * n - 1), w[m1], 0.0)
    M2 = np.where((m2 >= 0) & (m2 < 2 * n - 1), w[m2], 0.0)
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    return dx * (Part1 - Part2)


def smooth_max_weights(c, alpha):
    c_max = np.max(c)
    shifted = c - c_max
    if alpha > 50:
        shifted = np.clip(shifted, -0.5, 0.0)
    exp_c = np.exp(alpha * shifted)
    w = exp_c / np.sum(exp_c)
    return w


def gd_optimize(h_init, n, deadline, alpha_start, max_iter, lr_init):
    target = n / 2.0
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_start
    lr = lr_init
    t_prev = 0.0
    stagnant = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        c = compute_c5_all(h)
        c_cur = float(np.max(c))
        w = smooth_max_weights(c, alpha)
        grad = compute_gradient(h, w)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            break

        step = lr / max(gnorm, 1e-10)
        improved = False
        for _ls in range(25):
            h_try = h - step * grad
            h_try, t_new = project_feasible_warm(h_try, target, t_prev)
            c5_try = compute_c5(h_try)
            if c5_try < c_cur - 1e-15:
                h = h_try
                t_prev = t_new
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                improved = True
                break
            step *= 0.6

        if not improved:
            stagnant += 1
            alpha = min(alpha * 1.35, 10000.0)
            if alpha >= 10000 and stagnant > 200:
                break
        elif stagnant > 50:
            alpha = min(alpha * 1.1, 10000.0)
            stagnant = max(0, stagnant - 10)

        if (it + 1) % 500 == 0:
            alpha = min(alpha * 1.2, 10000.0)

    return best_h, best_c5


def upsample_linear(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)


def binary_local_search(h_bool, n, deadline, max_rounds):

        if len(ones) == 0 or len(zeros) == 0:

                    h[i] = 1.0
                    h[j] = 0.0

                        best_pair = (i, j)

            c5_new = compute_c5(h)
            if c5_new < best_c5:

        else:
```
