import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")


def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx


def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        for i in range(n):
            idx2 = i + k - (n - 1)
            if 0 <= idx2 < n:
                grad[i] += w * (1.0 - h[idx2])
            idx1 = i - k + (n - 1)
            if 0 <= idx1 < n:
                grad[i] -= w * h[idx1]
    grad *= dx
    return grad


def de_mutation(pop, i, F, n):
    N = len(pop)
    while True:
        a, b, c = np.random.choice(N, 3, replace=False)
        if a != i and b != i and c != i:
            break
    return pop[a] + F * (pop[b] - pop[c])


def de_crossover(donor, current, CR, n):
    r = np.random.random()
    j_rand = np.random.randint(n)
    mask = (np.random.random(n) < CR) | (np.arange(n) == j_rand)
    trial = np.where(mask, donor, current)
    return trial


def de_phase(h_start, n, budget_s, seed):
    np.random.seed(seed)
    pop_size = 60
    pop = np.zeros((pop_size, n))
    fitness = np.zeros(pop_size)

def subgradient_phase(h_start, n, budget_s, seed):
    np.random.seed(seed)
    h = h_start.copy()
    start_time = time.time()
    max_time = budget_s
    
    eps_vals = [50, 100, 200, 500]
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for eps in eps_vals:
        if time.time() - start_time > max_time - 10:
            break
        
        # Barzilai-Borwein step size
        step = 0.1
        h_prev = h.copy()
        c5_prev = evaluate_c5(h)
        g_prev = smooth_max_grad(h, eps)
        
        for it in range(2000):
            if time.time() - start_time > max_time - 5:
                break
            
            g = smooth_max_grad(h, eps)
            
            # BB step size
            s = h - h_prev
            y = g - g_prev
            sy = np.dot(s, y)
            nn = np.dot(s, s)
            yn = np.dot(y, y)
            if sy > 1e-12 and nn > 1e-12 and yn > 1e-12:
                step = sy / yn
                step = np.clip(step, 0.001, 0.5)
            
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Backtracking if needed
            obj_new = smooth_max_obj(h_new, eps)
            obj = smooth_max_obj(h, eps)
            if obj_new > obj + 0.5 * step * np.dot(g, h_new - h):
                step *= 0.5
                h_new = h - step * g
                h_new = project_feasible(h_new, n)
            
            h_prev = h
            g_prev = g
            h = h_new
            c5_prev = c5_new
            
            # Convergence check
            if np.max(np.abs(g)) < 1e-8:
                break
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx


def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w_k = weights[k]
        if w_k < 1e-12:
            continue
        for i in range(n):
            idx2 = i + k - (n - 1)
            if 0 <= idx2 < n:
                grad[i] += w_k * (1.0 - h[idx2])
            idx1 = i - k + (n - 1)
            if 0 <= idx1 < n:
                grad[i] -= w_k * h[idx1]
    grad *= dx
    return grad

Actually, I realize I should think more carefully about what makes this problem hard and what approach is most likely to succeed.

The key challenge is that the objective max_k c_k is non-smooth (it's the max of many smooth functions). At the optimum, multiple k values may be "active" (achieving the maximum). The solution needs to balance all of them.

The smooth-max approximation (log-sum-exp) is a good approach but the resolution depends on eps. With large eps, it approximates the true max but the landscape becomes more rugged. With small eps, it's smoother but may not capture the true optimum.

A key insight: **at the optimum, we want all active k values to have the same c_k value** (otherwise we could shift weight to reduce the maximum). This suggests using an approach that explicitly encourages equalizing the active c_k values.

One approach: **minimize max(c_k) + λ * range(c_k)** where range = max - min over active k. This encourages the active k values to be equal.

But this is essentially what the smooth-max already does (it spreads the weight across nearby-max k values).

Another approach: **alternating minimization**. Fix a target T, then minimize sum_k max(0, c_k - T)². This is a softly constrained optimization. Binary search on T.

Let me try this "water-filling" approach:
1. For a given T, define f(h) = sum_k max(0, c_k - T)²
2. This is smooth in h (for T > 0)
3. Minimize f(h) subject to constraints
4. Binary search on T to find the smallest T for which f(h*) ≈ 0

This might give a better solution because it explicitly handles the "active set" structure.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    return np.clip(x - lam, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")


def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx


def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if w_k < 1e-12:

        for i in range(n):

            idx1 = i - k + (n - 1)

    return grad
