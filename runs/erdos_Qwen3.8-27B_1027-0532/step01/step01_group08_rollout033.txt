def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    h = h_bin.copy()
    best_h = h.copy()
    best_overlaps = compute_overlaps(h)
    best_c5 = np.max(best_overlaps)
    
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    
    T = 0.5  # initial temperature

  - h[i-d]*(1-h[i]): was 0 (since 1-h[i]=0), now h[i-d]. Change: +h[i-d]

Terms involving h[j]:
- h[j] itself: was 0, now 1. This affects: h[j]*(1-h[j+d]) and h[j-d]*(1-h[j])
  - h[j]*(1-h[j+d]): was 0, now (1-h[j+d]). Change: +(1-h[j+d])
  - h[j-d]*(1-h[j]): was h[j-d], now 0. Change: -h[j-d]

Total change in O(d):
ΔO(d) = -(1-h[i+d]) + h[i-d] + (1-h[j+d]) - h[j-d]
= -1 + h[i+d] + h[i-d] + 1 - h[j+d] - h[j-d]
= h[i+d] + h[i-d] - h[j+d] - h[j-d]

(Where we interpret h[k] = 0 for k < 0 or k >= n.)

The function f(h) = max_k O_k(h) where O_k(h) = Σ_j h[j](1-h[j+k]) dx is the max of bilinear functions.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binary_sa(h_bin, n, target, deadline, max_iter=1000000):
    """Simulated annealing on binary sequences."""
    h = h_bin.copy()
    best_h = h.copy()
    best_overlaps = compute_overlaps(h)

            break
        if it % 200 == 0:

        i = ones[np.random.randint(len(ones))]

            ones = np.where(h == 1)[0]

def nesterov_subgrad(h0, n, target, deadline, max_iter=100000):
    """Nesterov-accelerated subgradient method for the max overlap."""
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    alpha = 0.1  # step size
    mu = 0.9     # momentum parameter
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        overlaps = compute_overlaps(h)
        k_star = np.argmax(overlaps)
        lag = k_star - (n - 1)
        dx = 2.0 / n
        
        # Compute subgradient of the worst overlap
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi+1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        
        # Nesterov step
        y = h + mu * (h - h_prev)
        h_prev = h.copy()
        h = y - alpha * g
        h = project_onto_constraint(h, target)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay
        alpha *= 0.9999
        mu = min(mu + 0.0001 * (it % 1000) / 1000, 0.95)
    
    return best_h, best_c5

def smooth_max_opt(h0, n, target_sum, deadline, tau=50.0, max_iter=50000):
    """Optimize using soft-max (smooth approximation of max)."""
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = 0.03
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        tau_corr = tau * corr
        max_tc = np.max(tau_corr)
        weights = np.exp(tau_corr - max_tc)
        weights /= np.sum(weights)
        
        # Gradient of soft-max
        g = np.zeros(n)
        for m in range(len(weights)):
            if weights[m] < 1e-12:
                continue
            lag = m - (n - 1)
            w = weights[m]
            j_lo = max(0, lag)
            j_hi = min(n - 1, n - 1 + lag)
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                g[j2_lo:j2_hi+1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        
        h_new = h - alpha * g
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = log_sum_exp / tau
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    n = len(h)
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, g * dx

def lbfgsb_smooth_opt(h0, n, target_sum, deadline, tau_seq=None):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()

            break

            return o
        def grad_fn(x):
            _, g = soft_max_obj_and_grad(x, tau, dx)
            return g
        try:
            res = minimize(obj_fn, x, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                                   'maxfun': 5000})
            h_new = project_onto_constraint(np.clip(res.x, 0.0, 1.0), float(target_sum))
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new
        except Exception:
