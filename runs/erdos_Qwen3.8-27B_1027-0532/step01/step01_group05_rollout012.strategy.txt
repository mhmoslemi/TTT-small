2. Mathematical specification:

We discreteize h as n-dimensional vector with:
- Each component h[i] ∈ [0,1]
- ∑h[i] = n/2 (exactly)

Objective: Minimize C_5 = max_k ∑ₖ h(x) (1-h(x+k)) dx

Represented as:
C_5 = (1/dx) * max_k ∑ₖ h(x) (1-h(x+k)) 

With dx = 2/n

Constraints are:
- h[i] ∈ [0,1] ∀i
- ∑h[i] = n/2

Optimization variables: h = (h₁, h₂, ..., h_n)

Grid interpolation: For grid expansion, we use RBF interpolation:
φ(||x-y||) = exp(-γ||x-y||²)

Gradient calculation:
For each spatial shift Δ:
∇h(x) = ∑ₖ (δ_{x,i} * (1 - h(x+k) - h(x) * δ_{x+k,i})

This gradient is computed efficiently using time-shifts

```python
def interpolate_rbf(x_coarse, y_coarse, n_fine):
    # Calculate coefficients for RBF interpolation
    x_fine = np.linspace(0,2,n_fine)
    gamma = 0.1  # chosen empirically
    basis_matrix = np.zeros((n_fine, len(x_coarse)))
    for i,xi in enumerate(x_coarse):
        corr = np.abs(xi - x_fine.reshape(-1,1))
        basis_matrix[:,i] = np.exp(-gamma*corr*corr)
    
    # Adjust for rank-deficiency
    A = np.vstack([basis_matrix, np.ones(len(x_coarse))])
    x_block = np.concatenate([y_coarse, np.ones(len(y_coarse))])
    coeff, _, _, _ = np.linalg.lstsq(A, x_block, rcond=1e-10)
    w = coeff[:-1]
    return w[0][np.newaxis,:]  # First row is the actual interpolation
```

```python
def optimize_with_cvxpy(h0, n_points, max_time=300, epsilon=1e-6):
    n = n_points
    dx = 2.0/n
    
    from cvxpy import Variable, Problem, Minimize, Constraint, Epigraph
    from cvxpy.expressions import eps_parameter
    
    # Define variables
    h = Variable(n)
    
    # Objective: max_k ∑h(x)(1-h(x+k)) with max over shifts
    h_vec = np.reshape(h, (n,), order='C')
    x = np.linspace(0,2,n)
    
    # Prepare all cyclic shifts
    H = np.zeros((n),dtype=float)
    delays = np.arange(-n+1, n)
    A = []
    for lag in delays:
        shift_h = np.roll(h_vec, lag)
        if lag < 0:
            shifted = np.concatenate([h_vec[-|lag|:], h_vec[:-|lag|]])
            A.append(shifted + 1 - h_vec)
        else:
            A.append(h_vec + 1 - np.roll(h_vec, lag))
    
    A = np.array(A)
    max_term = Epigraph(1.0 / dx * np.max( [np.dot(A[i], h.value) for i in range(len(A))] ), 
                        shape=(1,1))
    
    # Subject to:
    c = np.zeros(n)
    c[:n//2] = 1.0
    c[n//2] = 0 if n%2==0 else 0.5  # Handle odd symmetry
    c[n//2+1:] = -0.5 if n%2==0 else [-0.5]*(n//2)
    
    A_flat = np.concatenate([h.value, c])
    b_flat = np.concatenate([np.array([1.0]), np.array([1.0])])

    constraints = [h >= 0, h <= 1, h @ c == 1.0]

    prob = Problem(Minimize(max_term), constraints)
    
    # Set options with time limit
    options = {'verbose': False, 'max_iters': 1000, 'relative_gap': 1e-6}
    result = prob.solve(solver=CVXOPT, **options)
    
    return h.value
```
