def compute_grad_lse(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # gradient of f wrt h[j]
    # s_k = k - (n-1), k = 0, ..., 2n-2

        # first term: (1-h[j+s]) for valid j+s

def gradient_descent(h, n, T, elapsed, deadline):

def compute_grad_lse(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    # For each k, s = k - (n-1)
    # First term: (1-h[j+s]) for valid j+s, j = max(0,-s) to min(n-1, n-1-s)
    # Second term: -h[j-s] for valid j-s, j = max(0,s) to min(n-1, n-1+s)
    
    for k in range(2*n-1):
        s = k - (n-1)
        w = weights[k] * dx
        # First term
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start < j_end:
            grad[j_start:j_end] += w * (1.0 - h[j_start+s:j_end+s])
        # Second term
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2-s:j_end2-s]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def compute_grad_lse(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    scaled -= scaled.max()
    weights = np.exp(scaled)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = n - 1 - k
        w = weights[k] * dx
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start < j_end:
            grad[j_start:j_end] += w * (1.0 - h[j_start + s:j_end + s])
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= w * h[j_start2 - s:j_end2 - s]
    return grad


def compute_grad_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    s = n - 1 - k_star
    grad = np.zeros(n)
    j_start = max(0, -s)
    j_end = min(n, n - s)
    if j_start < j_end:
        grad[j_start:j_end] += dx * (1.0 - h[j_start + s:j_end + s])
    j_start2 = max(0, s)
    j_end2 = min(n, n + s)
    if j_start2 < j_end2:
        grad[j_start2:j_end2] -= dx * h[j_start2 - s:j_end2 - s]
    return grad


def gradient_descent(h, t_start, deadline, T, lr_init=0.05, max_iter=50000):
    n = len(h)
    h = h.copy()
    c5 = score_c5(h)
    lr = lr_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if T > 0:
            grad = compute_grad_lse(h, T)
        else:
            grad = compute_grad_subgrad(h)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        accepted = False
        for _bt in range(40):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c5_new = score_c5(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
                lr = min(lr * 1.03, 1.0)
                stall = 0
                accepted = True
                break
            lr *= 0.4
        if not accepted:
            stall += 1
            if stall > 80:
                break
    return h, c5


def basin_hopping(h, t_start, deadline, rng, n_inner_iters=5000):
    n = len(h)
    h = h.copy()
    c5 = score_c5(h)
    best_h = h.copy()
    best_c5 = c5
    noise_std = 0.08
    stall_count = 0
    max_stall = 40
    while time.time() < deadline:
        h_pert = h + rng.normal(0, noise_std, n)
        h_pert = project_feasible(h_pert, n)
        h_local, c5_local = gradient_descent(h_pert, t_start, min(time.time() + 10, deadline), 0.01, lr_init=0.02, max_iter=n_inner_iters)
        h_local = project_feasible(h_local, n)
        c5_local = score_c5(h_local)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            h = h_local.copy()
            c5 = c5_local
            noise_std = max(noise_std * 0.92, 0.005)
            stall_count = 0
        else:
            if rng.random() < np.exp(-(c5_local - c5) / (noise_std * 0.05 + 1e-10)):
                h = h_local.copy()
                c5 = c5_local
            noise_std = min(noise_std * 1.08, 0.3)
            stall_count += 1
        if stall_count > max_stall:
            noise_std = min(noise_std * 2.0, 0.4)
            stall_count = 0
            if noise_std > 0.4:
                break
    return best_h, best_c5


def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new)
    return h_new


def binary_local_search(h, t_start, deadline, rng):
    n = len(h)
    target = n / 2.0
    h_bin = h.copy()
    sorted_idx = np.argsort(h_bin)
    h_bin[:] = 0.0
    n_ones = int(target)
    h_bin[sorted_idx[-n_ones:]] = 1.0
    if target != n_ones:
        h_bin[sorted_idx[-n_ones - 1]] = target - n_ones
    c5 = score_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for delta in [0.1, -0.1, 0.25, -0.25, 0.5, -0.5, 1.0, -1.0]:
                trial = best_h.copy()
                trial[i] += delta
                trial = project_feasible(trial, n)
                c5_trial = score_c5(trial)
                if c5_trial < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c5_trial
                    improved = True
                    break
    return best_h, best_c5
