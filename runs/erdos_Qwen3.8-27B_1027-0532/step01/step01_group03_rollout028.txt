        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj_factory(n):
    dx = 2.0 / n
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
    return obj

def _do_slsqp(h_start, n, target_sum, alpha, maxiter=300, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    dx = 2.0 / n
    try:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            if m < -500:
                return 0.0
            return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _do_trust_constr(h_start, n, target_sum, alpha, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    dx = 2.0 / n
    try:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            if m < -500:
                return 0.0
            return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
        res = minimize(obj, h_start, method='trust-constr',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10})
        return res.x
    except Exception:
        return h_start

def _coordinate_descent(h, target_sum, rng, max_sweeps=30):
    n = len(h)
    h = h.copy()
    best_c5 = _c5_of(h)
    deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.002, -0.002]
    
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            best_delta = 0.0
            for d in deltas:
                h_trial = h.copy()
                h_trial[j] += d
                if h_trial[j] < -1e-9 or h_trial[j] > 1.0 + 1e-9:
                    continue
                h_trial = _fix_sum(h_trial, target_sum)
                c5 = _c5_of(h_trial)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_delta = d
            if best_delta != 0.0:
                h[j] += best_delta
                h = _fix_sum(h, target_sum)
                improved = True
        if not improved:
            break
    return h, best_c5

def _pair_swap(h, target_sum, rng, n_tries=2000):
    n = len(h)
    h = h.copy()
    best_c5 = _c5_of(h)
    
    for _ in range(n_tries):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
        c5 = _c5_of(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c5

def _interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def _subgradient_descent(h, target_sum, lr, n_steps, rng):
    n = len(h)
    h = h.copy()
    best_c5 = _c5_of(h)
    
    for step in range(n_steps):
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr_raw)
        tol = 1e-6
        active = np.where(corr_raw >= max_val - tol)[0]
        
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            if k >= 0:
                valid = slice(0, n - k)
                first = h[valid]
                second = h[k:k + (n - k)]
                grad[valid] += (1.0 - second)
                grad[k:k + (n - k)] += (-first)
            else:
                k2 = -k
                valid = slice(0, n - k2)
                first = h[k2:k2 + (n - k2)]
                second = h[valid]
                grad[k2:k2 + (n - k2)] += (1.0 - second)
                grad[valid] += (-first)
        
        grad /= len(active)
        h_new = h - lr * grad
        h_new = _fix_sum(h_new, target_sum)
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            h = h_new
            best_c5 = c5
        else:
            lr *= 0.7
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 12
    
    def elapsed_ok(margin=15):
        return time.time() < deadline - margin
    
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(86, 0.5)
    
    n0 = len(h_init)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h_init = _fix_sum(h_init, target0)
    
    # Phase 1: SLSQP refinement with increasing alpha on n=86
    alphas_p1 = [20, 50, 100, 200, 400, 800, 1500]
    h_cur = h_init.copy()
    c5_cur = _c5_of(h_cur)
    
    for alpha in alphas_p1:
        if not elapsed_ok(30):
            break
        h_new = _do_slsqp(h_cur, n0, target0, alpha, maxiter=300, ftol=1e-13)
        h_new = _fix_sum(h_new, target0)
        c5_new = _c5_of(h_new)
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n0
    
    # Phase 2: trust-constr with high alpha
    if elapsed_ok(60):
        for alpha in [500, 1000, 2000]:
            if not elapsed_ok(30):
                break
            h_new = _do_trust_constr(h_cur, n0, target0, alpha, maxiter=300)
            h_new = _fix_sum(h_new, target0)
            c5_new = _c5_of(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n0
    
    # Phase 3: Coordinate descent
    if elapsed_ok(60):
        h_cd, c5_cd = _coordinate_descent(h_cur, target0, rng, max_sweeps=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n0
        h_cur = h_cd.copy()
        c5_cur = c5_cd
    
    # Phase 4: Pair swaps
    if elapsed_ok(40):
        h_sw, c5_sw = _pair_swap(h_cur, target0, rng, n_tries=3000)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()
            best_n = n0
        h_cur = h_sw.copy()
        c5_cur = c5_sw
    
    # Phase 5: Subgradient descent
    if elapsed_ok(60):
        h_sg, c5_sg = _subgradient_descent(h_cur, target0, 0.01, 200, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n0
        h_cur = h_sg.copy()
        c5_cur = c5_sg
    
    # Phase 6: More SLSQP with very high alpha
    if elapsed_ok(60):
        for alpha in [2000, 5000, 10000]:
            if not elapsed_ok(30):
                break
            h_new = _do_slsqp(h_cur, n0, target0, alpha, maxiter=400, ftol=1e-14)
            h_new = _fix_sum(h_new, target0)
            c5_new = _c5_of(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n0
    
    # Phase 7: Final coordinate descent polish
    if elapsed_ok(40):
        h_cd2, c5_cd2 = _coordinate_descent(h_cur, target0, rng, max_sweeps=15)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n0
        h_cur = h_cd2.copy()
        c5_cur = c5_cd2
    
    # Phase 8: Try different n values
    for n_try in [100, 120, 150]:
        if not elapsed_ok(120):
            break
        target_try = n_try / 2.0
        h_interp = _interpolate_h(h_cur, n0, n_try)
        h_interp = _fix_sum(h_interp, target_try)
        
        # Also try block start
        h_block = np.zeros(n_try)
        h_block[:int(target_try)] = 1.0
        h_block = _fix_sum(h_block, target_try)
        
        starts = [h_interp, h_block]
        
        for h_start_try in starts:
            if not elapsed_ok(60):
                break
            h_try = h_start_try.copy()
            c5_try = _c5_of(h_try)
            for alpha in [50, 200, 500, 2000]:
                if not elapsed_ok(30):
                    break
                h_new = _do_slsqp(h_try, n_try, target_try, alpha, maxiter=250, ftol=1e-13)
                h_new = _fix_sum(h_new, target_try)
                c5_new = _c5_of(h_new)
                if c5_new < c5_try - 1e-14:
                    h_try = h_new
                    c5_try = c5_new
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
        
        # Coordinate descent on best n_try solution
        if elapsed_ok(30):
            h_try = best_h if best_n == n_try else h_interp
            h_cd3, c5_cd3 = _coordinate_descent(h_try, target_try, rng, max_sweeps=10)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n_try
    
    # Phase 9: Random restarts with structure
    if elapsed_ok(120):
        n_best = best_n
        t_best = n_best / 2.0
        for restart in range(8):
            if not elapsed_ok(30):
                break
            if restart == 0:
                h_r = np.zeros(n_best)
                h_r[:int(t_best)] = 1.0
            elif restart == 1:
                x = np.linspace(0, 2, n_best, endpoint=False)
                h_r = 0.5 + 0.3 * np.cos(2 * np.pi * x * 2)
            elif restart == 2:
                h_r = 0.5 + 0.4 * np.sin(2 * np.pi * x * 3)
            elif restart == 3:
                h_r = (np.arange(n_best) % 3 == 0).astype(float) * 1.0
                h_r = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * np.arange(n_best) / n_best * 5)) * 0.3
            else:
                h_r = 0.5 + 0.3 * rng.randn(n_best)
            h_r = _fix_sum(h_r, t_best)
            
            h_try = h_r.copy()
            c5_try = _c5_of(h_try)
            for alpha in [100, 500, 2000]:
                if not elapsed_ok(20):
                    break
                h_new = _do_slsqp(h_try, n_best, t_best, alpha, maxiter=200, ftol=1e-13)
                h_new = _fix_sum(h_new, t_best)
                c5_new = _c5_of(h_new)
                if c5_new < c5_try - 1e-14:
                    h_try = h_new
                    c5_try = c5_new
            
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_best
    
    # Phase 10: Final polish with very fine coordinate descent
    if elapsed_ok(60):
        t_final = best_n / 2.0
        h_final = best_h.copy()
        h_cd_final, c5_final = _coordinate_descent(h_final, t_final, rng, max_sweeps=25)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_cd_final.copy()
    
    # Phase 11: One more SLSQP pass on the best
    if elapsed_ok(60):
        t_final = best_n / 2.0
        h_new = _do_slsqp(best_h, best_n, t_final, 5000, maxiter=500, ftol=1e-15)
        h_new = _fix_sum(h_new, t_final)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final verification
    best_h = _fix_sum(best_h, best_n / 2.0)
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    
    return (best_h, best_c5, best_n)
