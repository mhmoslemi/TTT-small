import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _soft_subgradient(h, n, temperature):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = (corr - max_c) / temperature
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        k = m + (n-1)
        w = weights[k]
        if w < 1e-10:
            continue
        if m >= 0:
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
        else:
            am = -m
            grad[am:] += w * (1.0 - h[:n-am])
            grad[:n-am] -= w * h[am:]
    
    grad *= dx
    return grad, max_c * dx

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def _do_slsqp(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < _smooth_obj(h_start, alpha) + 1e-10:
            return res.x
        return h_start
    except Exception:
        return h_start

def _coordinate_descent(h, n, target_sum, n_sweeps=3, time_limit=60):
    best_c5 = _c5_of(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            # Try a range of values for h[j]
            lo = max(0.0, h[j] - 0.15)
            hi = min(1.0, h[j] + 0.15)
            candidates = np.linspace(lo, hi, 15)
            best_j_val = h[j]
            best_c5_j = best_c5
            for cand in candidates:
                h_trial = h.copy()
                h_trial[j] = cand
                h_trial = _fix_sum(h_trial, target_sum)
                c5_trial = _c5_of(h_trial)
                if c5_trial < best_c5_j - 1e-12:
                    best_c5_j = c5_trial
                    best_j_val = cand
            if best_c5_j < best_c5 - 1e-12:
                h[j] = best_j_val
                h = _fix_sum(h, target_sum)
                best_c5 = _c5_of(h)
    return h, best_c5

```python
def _soft_subgradient(h, n, temperature):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = (corr - max_c) / temperature
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        m = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-10:
            continue
        if m >= 0:
            grad[m:n] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            am = -m
            grad[:n-am] += w * (1.0 - h[am:n])
            grad[am:n] -= w * h[:n-am]
    
    grad *= dx
    return grad, max_c * dx
```
