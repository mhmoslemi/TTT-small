def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_h(h, alpha):
    corrs = compute_correlations(h)
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = m + np.log(sum_exp) / alpha
    return obj

    for alpha in [3000, 6000, 10000, 15000]:
        if time_left() < 30:
            break
        res_x = slsqp_optimize(best_h, alpha, 200, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 6: Very high alpha L-BFGS-B polish
    for alpha in [20000, 40000, 80000]:
        if time_left() < 15:
            break
        res_x = lbfgsb_optimize(best_h, alpha, 50000.0, 300, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), len(best_h) / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 7: Try slightly different resolutions around the best
    for n_adj in [len(best_h) + 5, len(best_h) - 5, len(best_h) + 15, len(best_h) - 15]:
        if time_left() < 20:
            break
        if n_adj < 100 or n_adj > 2000:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_adj, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_sum(h_init, n_adj / 2.0)
        bounds_adj = [(0.0, 1.0)] * n_adj
        for alpha in [1000, 3000]:
            if time_left() < 10:
                break
            res_x = lbfgsb_optimize(h_init, alpha, 10000.0, 200, bounds_adj)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_adj / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_init = h_new.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
