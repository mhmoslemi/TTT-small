from scipy.optimize import minimize

_gs = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def subgrad(h, tol=1e-4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    active = np.where(corr >= c_max - tol)[0]
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s >= 0:
            if s == 0:
                grad += (1.0 - 2.0 * h) * dx
            else:
                j1 = np.arange(0, n - s)
                grad[j1] += (1.0 - h[j1 + s]) * dx
                j2 = np.arange(s, n)
                grad[j2] -= h[j2 - s] * dx
        else:
            t = -s
            j1 = np.arange(t, n)
            grad[j1] += (1.0 - h[j1 - t]) * dx
            j2 = np.arange(0, n - t)
            grad[j2] -= h[j2 + t] * dx
    grad /= len(active)
    return grad, c_max * dx, corr

    return np.sum(x) - _gs['target_sum']

    return np.ones(_gs['n'])

    c_max = np.max(corr)

    _gs['target_sum'] = target_sum
    _gs['n'] = n

    cons = [{'type': 'eq', 'fun': _constraint_eq, 'jac': _constraint_jac}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h

    no_improve = 0

    for it in range(max_iter):

        grad, f, _ = subgrad(h_nest, tol=1e-4)
        f_nest = compute_c5(h_nest)[0]
        if f_nest < best_f:
            best_f = f_nest
            best_h = h_nest.copy()
            no_improve = 0
        else:
            no_improve += 1
