def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent(h, n, max_sweeps=50, time_limit=60):

    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    
    k_arr = np.arange(-(n-1), n)  # all lags
    
    for sweep in range(max_sweeps):

            break
        max_change = 0.0

            idx1 = j + k_arr  # j+k for each lag k
            idx2 = j - k_arr  # j-k for each lag k

            valid2 = (idx2 >= 0) & (idx2 < n)
            
            idx1_c = np.clip(idx1, 0, n-1).astype(int)
            idx2_c = np.clip(idx2, 0, n-1).astype(int)
            shift1 = h[idx1_c] * valid1
            shift2 = h[idx2_c] * valid2
            
            B = (1.0 - shift1) * valid1 - shift2 * valid2
            # For k=0 (index n-1): the linear approximation isn't valid (quadratic)
            # We'll handle k=0 separately
            B[n-1] = 0.0  # will be handled by the quadratic term

            f_other[n-1, :] = f0  # override k=0
            max_f = np.max(f_other, axis=0)
            best_idx = np.argmin(max_f)

            hi = x_grid[min(len(x_grid)-1, best_idx + 1)]

                m2 = hi - (hi - lo) / 3.0
                f_m1 = max(S0 + m1 - m1**2, np.max(A + B * m1))
                f_m2 = max(S0 + m2 - m2**2, np.max(A + B * m2))
                if f_m1 < f_m2:
                    hi = m2
                else:
                    lo = m1

            h[j] = new_val
            max_change = max(max_change, abs(new_val - old_val))
