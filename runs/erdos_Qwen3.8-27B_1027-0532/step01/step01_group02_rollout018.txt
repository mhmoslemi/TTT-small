import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = (-diff) / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def to_binary(h, n):
    """Convert to binary by thresholding, preserving sum."""
    target = n // 2
    h_sorted = np.sort(h)
    # Find the threshold
    # We want exactly target ones
    # Sort indices by h value descending
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return float(np.max(corr) * dx)

def binary_local_search(h_bin, n, rng, max_iters=100000, t0=1.0, t1=0.01, time_limit=60):
    """Simulated annealing on binary h."""
    best_h = h_bin.copy()
    best_c5 = evaluate_c5(h_bin)
    curr_h = h_bin.copy()
    curr_c5 = best_c5
    
    ones = np.where(curr_h == 1.0)[0]
    zeros = np.where(curr_h == 0.0)[0]

    n_zeros = len(zeros)
    
    start_time = time.time()

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            count = mask.sum()
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-15
            count = mask.sum()
            if count == 0:
                break
            step = (-diff) / count
            h[mask] = np.maximum(0.0, h[mask] - step)

        j = zeros_list[jj]

        return None  # Skip, let SLSQP use numerical gradients

    """Project g onto {h: 0≤h≤1, sum(h)=n/2}."""
    target = n / 2.0

    lo = -1.0
    hi = 1.0

        h = np.clip(g + mid, 0.0, 1.0)

        if s < target:

        else:

            np.clip(h, 0.0, 1.0, out=h)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_opt(h, n, maxiter=500, ftol=1e-12):
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        x = np.clip(r.x, 0.0, 1.0)
        if abs(np.sum(x) - n / 2.0) > 1e-6:
            x = repair_h(x, n)
        return x, evaluate_c5(x)
    except Exception:
        return h.copy(), evaluate_c5(h)

def softmax_slsqp(h, n, beta, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        b = beta * corr
        m = b.max()
        return float((m + np.log(np.sum(np.exp(b - m)))) / beta)
    try:
        r = minimize(
            obj, h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        x = np.clip(r.x, 0.0, 1.0)
        if abs(np.sum(x) - n / 2.0) > 1e-6:
            x = repair_h(x, n)
        return x, evaluate_c5(x)
    except Exception:
        return h.copy(), evaluate_c5(h)

def binary_sa(h_bin, n, rng, time_limit=120):
    best_h = h_bin.copy()
    best_c5 = evaluate_c5(h_bin)
    curr = h_bin.copy()
    curr_c5 = best_c5
    ones_list = list(np.where(curr == 1.0)[0])
    zeros_list = list(np.where(curr == 0.0)[0])
    T0 = 0.04
    T1 = 0.00005
    max_iter = 800000
    start = time.time()
    it = 0
    while it < max_iter:
        if time.time() - start > time_limit:
            break
        T = T0 * (T1 / T0) ** (it / max_iter)
        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))
        i = ones_list[ii]
        j = zeros_list[jj]
        curr[i] = 0.0
        curr[j] = 1.0
        ones_list[ii] = j
        zeros_list[jj] = i
        new_c5 = evaluate_c5(curr)
        delta = new_c5 - curr_c5
        if delta <= 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr.copy()
        else:
            curr[i] = 1.0
            curr[j] = 0.0
            ones_list[ii] = i
            zeros_list[jj] = j
        it += 1
    return best_h, best_c5

def cutting_plane_slsqp(h, n, max_rounds=10, maxiter_per=150):
    dx = 2.0 / n
    curr = h.copy()
    for _ in range(max_rounds):
        corr = np.correlate(curr, 1.0 - curr, mode="full") * dx
        mx = corr.max()
        active = np.where(corr >= mx - 1e-5)[0]
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full") * dx
            return float(np.max(c[active]))
        try:
            r = minimize(
                obj, curr, method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                options={'maxiter': maxiter_per, 'ftol': 1e-11}
            )
            new_h = np.clip(r.x, 0.0, 1.0)
            if abs(np.sum(new_h) - n / 2.0) > 1e-6:
                new_h = repair_h(new_h, n)
            if evaluate_c5(new_h) < evaluate_c5(curr):
                curr = new_h
        except Exception:
            pass
    return curr, evaluate_c5(curr)

def perturb_restart(h, n, rng, scale=0.05):
    h_pert = h + rng.normal(0, scale, n)
    return repair_h(h_pert, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    h_init = initial_h_values.copy().astype(float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if rem() > 80:
        h1, c51 = slsqp_opt(best_h, n_init, maxiter=1000)
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: Softmax SLSQP at initial resolution
    if rem() > 80:
        for beta in [40, 100, 250]:
            if rem() < 50:
                break
            h2, c52 = softmax_slsqp(best_h, n_init, beta=beta, maxiter=200)
            if c52 < best_c5:
                best_h, best_c5 = h2, c52

    # Phase 3: Binary SA at initial resolution
    h_bin = to_binary(best_h, n_init)
    bc = evaluate_c5(h_bin)
    if bc < best_c5:
        best_h, best_c5 = h_bin, bc

    if rem() > 130:
        sa_time = min(220.0, rem() - 100.0)
        h_bin2, c5b = binary_sa(h_bin.copy(), n_init, rng, time_limit=sa_time)
        if c5b < best_c5:
            best_h, best_c5 = h_bin2, c5b
        # Also try from a random binary start
        h_rand_bin = np.zeros(n_init)
        h_rand_bin[rng.choice(n_init, n_init // 2, replace=False)] = 1.0
        h_bin3, c5b3 = binary_sa(h_rand_bin, n_init, rng, time_limit=min(80.0, rem() - 60.0))
        if c5b3 < best_c5:
            best_h, best_c5 = h_bin3, c5b3

    # Phase 4: Cutting plane at initial resolution
    if rem() > 120:
        h_cp, c5cp = cutting_plane_slsqp(best_h, n_init, max_rounds=6, maxiter_per=100)
        if c5cp < best_c5:
            best_h, best_c5 = h_cp, c5cp

    # Phase 5: Higher resolution
    for n_new in [500, 700]:
        if rem() < 180:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair_h(h_interp, n_new)

        h3, c53 = slsqp_opt(h_interp, n_new, maxiter=500)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

        if rem() > 130:
            h4, c54 = softmax_slsqp(best_h, n_new, beta=80, maxiter=300)
            if c54 < best_c5:
                best_h, best_c5 = h4, c54

        if rem() > 100:
            h5, c55 = cutting_plane_slsqp(best_h, n_new, max_rounds=5, maxiter_per=80)
            if c55 < best_c5:
                best_h, best_c5 = h5, c55

    # Phase 6: Perturbation restarts at final resolution
    n_fin = len(best_h)
    for i in range(15):
        if rem() < 40:
            break
        scale = 0.02 + 0.03 * rng.random()
        h_pert = perturb_restart(best_h, n_fin, rng, scale)
        h_p, c5p = slsqp_opt(h_pert, n_fin, maxiter=300)
        if c5p < best_c5:
            best_h, best_c5 = h_p, c5p

    # Phase 7: Binary SA at final resolution if different from initial
    if n_fin != n_init and rem() > 80:
        h_bin_f = to_binary(best_h, n_fin)
        h_bin4, c5b4 = binary_sa(h_bin_f, n_fin, rng, time_limit=min(80.0, rem() - 40.0))
        if c5b4 < best_c5:
            best_h, best_c5 = h_bin4, c5b4

    # Phase 8: Final refinement
    if rem() > 50:
        h_f, c5f = slsqp_opt(best_h, len(best_h), maxiter=2000, ftol=1e-14)
        if c5f < best_c5:
            best_h, best_c5 = h_f, c5f

    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
