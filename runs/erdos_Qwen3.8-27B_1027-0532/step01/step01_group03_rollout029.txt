                stall_count = 0
            else:
                stall_count += 1
                if stall_count >= 8:
                    break

    return best_h, best_c5

def _slsqp_polish(h_start, n, target_sum, deadline):
    best_h = h_start.copy()
    best_c5 = _c5_of(h_start)
    for alpha in [50, 150, 400, 800]:
        if time.time() > deadline:
            break
        h_new = _do_slsqp(best_h, n, target_sum, alpha, 300, 1e-14)
        h_new = _fix_sum(h_new, target_sum)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
    return best_h, best_c5

def _make_structured_start(n, target_sum, variant):
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:
        h = np.full(n, 0.5)
    elif variant == 1:
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
        h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    elif variant == 2:
        h = (np.sin(2.0 * np.pi * x) ** 2)
        h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    elif variant == 3:
        h = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.2 * np.cos(6.0 * np.pi * x)
        h = np.clip(h, 0, 1)
    elif variant == 4:
        half = n // 2
        h = np.zeros(n)
        h[:half] = 1.0 + 0.1 * np.sin(4.0 * np.pi * np.arange(half) / half)
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n, 0.5)
    return _fix_sum(h, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    n = 86
    dx = 2.0 / n
    target_sum = n / 2.0
    t_start = time.time()
    deadline = t_start + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(n, 0.5)
    h0 = _fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_of(h0)

    rng = np.random.RandomState(seed)

    # Phase 1: LP-based optimization from initial point
    best_h, best_c5 = _lp_polish(best_h, n, target_sum, min(deadline, t_start + 300))

    # Phase 2: SLSQP polish
    if time.time() < deadline - 90:
        best_h, best_c5 = _slsqp_polish(best_h, n, target_sum, min(deadline, t_start + 500))

    # Phase 3: More LP from polished point
    if time.time() < deadline - 60:
        best_h, best_c5 = _lp_polish(best_h, n, target_sum, min(deadline, t_start + 700))

    # Phase 4: Try structured starting points
    if time.time() < deadline - 120:
        for variant in range(5):
            if time.time() > deadline - 60:
                break
            h_struct = _make_structured_start(n, target_sum, variant)
            h_lp, c5_lp = _lp_polish(h_struct, n, target_sum, min(deadline - 30, t_start + 1100), max_rounds=200)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 5: Random perturbations
    if time.time() < deadline - 60:
        for _ in range(30):
            if time.time() > deadline - 30:
                break
            scale = 0.03 + 0.07 * rng.rand()
            h_pert = best_h + scale * rng.randn(n)
            h_pert = _fix_sum(h_pert, target_sum)
            h_new, c5_new = _lp_polish(h_pert, n, target_sum, min(deadline - 15, time.time() + 25), max_rounds=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 10:
        best_h, best_c5 = _slsqp_polish(best_h, n, target_sum, deadline)

    # Final LP polish
    if time.time() < deadline - 5:
        best_h, best_c5 = _lp_polish(best_h, n, target_sum, deadline, max_rounds=50)

    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n)
