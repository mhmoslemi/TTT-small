from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c_lag(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        g[:n - lag] += dx * (1.0 - h[lag:])
        g[lag:] -= dx * h[:n - lag]
    else:
        L = -lag
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    return g

def soft_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    mx = np.max(tc)
    lse = mx + np.log(np.sum(np.exp(tc - mx)))
    obj = lse / tau * dx
    w = np.exp(tc - mx)
    w /= w.sum()

        if w[m] < 1e-13:
            continue
        lag = m - (n - 1)
        grad += w[m] * grad_c_lag(h, lag, dx)
    return obj, grad

    x = np.linspace(0, 2, n, endpoint=False)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_onto_constraint(h_new, n_new / 2.0)
    return h_new
