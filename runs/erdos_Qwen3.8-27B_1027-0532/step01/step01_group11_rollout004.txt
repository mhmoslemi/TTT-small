            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    n_restarts = min(25, max(5, int((budget_s - elapsed - 60) / 50)))
    for i in range(n_restarts):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 80:
            break
        strategy = i % 5
        if strategy == 0:
            h_pert = best_h + 0.06 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_h + 0.12 * (rng.random(n0) - 0.5)
        elif strategy == 2:
            h_bin = (rng.random(n0) < 0.5).astype(float)
            h_bin = project_onto_constraint(h_bin, n0 / 2.0)
            alpha = 0.5 + 0.4 * rng.random()
            h_pert = alpha * best_h + (1.0 - alpha) * h_bin
        elif strategy == 3:
            h_pert = best_h.copy()
            nflip = max(1, n0 // 5)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_onto_constraint(h_pert, n0 / 2.0)
        else:
            h_pert = best_h.copy()
            perm = rng.permutation(n0)
            h_pert[perm] = best_h
            h_pert = project_onto_constraint(h_pert, n0 / 2.0)

        h_pert = project_onto_constraint(h_pert, n0 / 2.0)
        tl = min(budget_s - elapsed - 50, 80)
        h_opt, c5_opt = multi_eps_optimize(h_pert, eps_fine[2:], 200, tl, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 250:
        n1 = 2 * n0
        x_old = np.linspace(0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2.0, n1, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        tl = min(budget_s - elapsed - 40, 600)
        h_opt, c5_opt = multi_eps_optimize(h_up, eps_fine, 200, tl, start_time, use_pnorm_first=True)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        elapsed = time.time() - start_time
        if budget_s - elapsed > 200:
            h_rand_up = np.clip(0.5 + 0.3 * rng.standard_normal(n1), 0.0, 1.0)
            h_rand_up = project_onto_constraint(h_rand_up, n1 / 2.0)
            tl2 = min(budget_s - elapsed - 40, 400)
            h_opt2, c5_opt2 = multi_eps_optimize(h_rand_up, eps_fine, 150, tl2, start_time, use_pnorm_first=True)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    elapsed = time.time() - start_time
    n_cur = len(best_h)
    if budget_s - elapsed > 80:
        tl = budget_s - elapsed - 15
        h_opt, c5_opt = optimize_slsqp_smooth(best_h, 0.0001, 800)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_opt, c5_opt = optimize_slsqp_smooth(best_h, 0.00005, 600)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        h_opt, c5_opt = optimize_slsqp_smooth(best_h, 0.00002, 600)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
