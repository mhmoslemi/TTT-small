    """Compute all lag correlations for binary h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr  # length 2n-1

def local_search(h, max_iters, deadline):

    dx = 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    best_h = h.copy()

Wait, one more thought. The G matrix update: when I swap h[i] from 1 to 0 and h[j] from 0 to 1, the G matrix changes. Let me think about how to update it efficiently.

G[p, d] = (1-h[p-d])_valid + h[p+d]_valid

After the swap:
- h[i] changed from 1 to 0
- h[j] changed from 0 to 1

For G[p, d]:
- The term (1-h[p-d]) changes if p-d = i (i.e., d = p-i): the old value was (1-h[i]) = 0, new value is (1-h[i]) = 1. So G[p, d] increases by 1 when d = p-i.
  Wait, p-d = i means d = p-i. The old h[i] = 1, so old (1-h[i]) = 0. New h[i] = 0, so new (1-h[i]) = 1. Change: +1.
- The term (1-h[p-d]) changes if p-d = j (i.e., d = p-j): old (1-h[j]) = 1, new (1-h[j]) = 0. Change: -1.
- The term h[p+d] changes if p+d = i (i.e., d = i-p): old h[i] = 1, new h[i] = 0. Change: -1.
- The term h[p+d] changes if p+d = j (i.e., d = j-p): old h[j] = 0, new h[j] = 1. Change: +1.

So for each p, G[p, d] changes at most at 4 specific values of d: d = p-i, d = p-j, d = i-p, d = j-p.

This means the G update is O(n) (one update per row p, at specific columns). Much faster than recomputing the entire row!

def update_G(G, h, i, j, n):
    """Update G matrix after swapping h[i]: 1->0 and h[j]: 0->1.
    G[p, d_idx] where d_idx = d + (n-1).
    """
    # For each p, check if d = p-i, p-j, i-p, or j-p is in range
    for p in range(n):
        # d = p - i: term (1-h[p-d]) = (1-h[i]) changes from 0 to 1, so G[p, (p-i)+(n-1)] += 1
        d1 = p - i
        if -(n-1) <= d1 <= n-1:
            G[p, d1 + n - 1] += 1.0
        # d = p - j: term (1-h[p-d]) = (1-h[j]) changes from 1 to 0, so G[p, (p-j)+(n-1)] -= 1
        d2 = p - j
        if -(n-1) <= d2 <= n-1:
            G[p, d2 + n - 1] -= 1.0
        # d = i - p: term h[p+d] = h[i] changes from 1 to 0, so G[p, (i-p)+(n-1)] -= 1
        d3 = i - p
        if -(n-1) <= d3 <= n-1:
            G[p, d3 + n - 1] -= 1.0
        # d = j - p: term h[p+d] = h[j] changes from 0 to 1, so G[p, (j-p)+(n-1)] += 1
        d4 = j - p
        if -(n-1) <= d4 <= n-1:
            G[p, d4 + n - 1] += 1.0

I need to be careful about one thing: the `update_G_fast` function. Let me verify it:

G[p, d_idx] = (1-h[p-d])_valid + h[p+d]_valid where d = d_idx - (n-1)

For G[p, d_idx] to change, we need either p-d = i or p-d = j (first term) or p+d = i or p+d = j (second term).

Case 1: p-d = i, i.e., d = p-i
Old: (1-h[i]) = 0 (since h[i]=1). New: (1-h[i]) = 1 (since h[i]=0). Change: +1.

Case 2: p-d = j, i.e., d = p-j
Old: (1-h[j]) = 1 (since h[j]=0). New: (1-h[j]) = 0 (since h[j]=1). Change: -1.
So G[p, (p-j)+(n-1)] -= 1.

Old: h[i] = 1. New: h[i] = 0. Change: -1.
So G[p, (i-p)+(n-1)] -= 1.

Case 4: p+d = j, i.e., d = j-p
Old: h[j] = 0. New: h[j] = 1. Change: +1.

- p-j ∈ [-(n-1), n-1] ✓
- i-p ∈ [-(n-1), n-1] ✓

- Case 3: d = i-p = 0, G[i, (0)+(n-1)] -= 1

- Case 4: d = j-p = 0, G[j, (0)+(n-1)] += 1

            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
        grad *= dx
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 20
    best_h = None
    best_c5 = float('inf')
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_c5 = c5_0
        h_bin = to_binary(h_init, n0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        else:
            best_h = h_init.copy()
    else:
        n0 = 120
        h_bin = np.zeros(n0)
        h_bin[:n0 // 2] = 1.0
        np.random.shuffle(h_bin)
        best_h = h_bin.copy()
        best_c5 = compute_c5(h_bin)
    n_current = len(best_h)
    phase1_deadline = t_start + 120
    h_greedy, c5_greedy = greedy_limited(best_h, phase1_deadline, max_iters=80000, top_k=40)
    if c5_greedy < best_c5:
        best_c5 = c5_greedy
        best_h = h_greedy.copy()
    phase2_deadline = t_start + 500
    h_sa, c5_sa = simulated_annealing(best_h, phase2_deadline, T0=0.6, T_min=0.002, cool=0.99997, seed=seed)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    h_greedy2, c5_greedy2 = greedy_limited(best_h, phase2_deadline, max_iters=50000, top_k=50)
    if c5_greedy2 < best_c5:
        best_c5 = c5_greedy2
        best_h = h_greedy2.copy()
    phase3_deadline = t_start + 800
    h_sa2, c5_sa2 = simulated_annealing(best_h, phase3_deadline, T0=0.3, T_min=0.001, cool=0.99996, seed=seed + 100)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()
    h_full, c5_full = full_greedy(best_h, phase3_deadline, max_iters=3000)
    if c5_full < best_c5:
        best_c5 = c5_full
        best_h = h_full.copy()
    n_target_list = [100, 110, 130, 140, 160, 80]
    phase4_deadline = t_start + 1100
    for n_try in n_target_list:
        if time.time() > phase4_deadline:
            break
        if n_try == len(best_h):
            continue
        if n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = to_binary(h_interp, n_try)
        c5_try = compute_c5(h_bin_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_bin_try.copy()
        h_g, c5_g = greedy_limited(h_bin_try, min(phase4_deadline, time.time() + 60), max_iters=20000, top_k=30)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        h_s, c5_s = simulated_annealing(best_h, min(phase4_deadline, time.time() + 80), T0=0.4, T_min=0.002, cool=0.99995, seed=seed + n_try)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    phase5_deadline = t_start + budget_s - 40
    n_best = len(best_h)
    h_cont = best_h.astype(float)
    h_pgd, c5_pgd = pgd_refine(h_cont, phase5_deadline, max_iter=1000)
    if c5_pgd < best_c5:
        best_c5 = c5_pgd
        best_h = h_pgd.copy()
    if time.time() < deadline - 5:
        h_bin_final = to_binary(best_h, len(best_h))
        c5_final = compute_c5(h_bin_final)
        if c5_final <= best_c5 + 1e-9:
            h_g3, c5_g3 = greedy_limited(h_bin_final, deadline, max_iters=10000, top_k=40)
            if c5_g3 < best_c5:
                best_c5 = c5_g3
                best_h = h_g3.copy()
    return best_h, best_c5, len(best_h)
