def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgradient_max_corr(h, n, tol=1e-6):
    """Compute subgradient of max_k corr_k with respect to h"""
    corr = get_correlations(h)
    max_v = np.max(corr)
    active = np.where(corr >= max_v - tol)[0]
    
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        # For each j, compute partial derivative
        # From left factor: (1 - h[j+shift]) if 0 <= j+shift < n
        # From right factor: -h[j-shift] if 0 <= j-shift < n
        # j+shift range: j in [0, n-1] and j+shift in [0, n-1]
        # j-shift range: j in [0, n-1] and j-shift in [0, n-1]
        
        if shift >= 0:
            # j+shift valid when j in [0, n-1-shift]
            left_idx = np.arange(0, n - shift)
            grad[left_idx] += (1.0 - h[left_idx + shift])
            # j-shift valid when j in [shift, n-1]
            right_idx = np.arange(shift, n)
            grad[right_idx] -= h[right_idx - shift]
        else:
            s = -shift
            # j+shift = j-s valid when j in [s, n-1]
            left_idx = np.arange(s, n)
            grad[left_idx] += (1.0 - h[left_idx - s])
            # j-shift = j+s valid when j in [0, n-1-s]
            right_idx = np.arange(0, n - s)
            grad[right_idx] -= h[right_idx + s]
    
    grad /= len(active)
    return grad, max_v

    """Projected subgradient method for minimizing max correlation"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, n, tol):
    """Compute subgradient of max_k corr_k(h) w.r.t. h"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_v = np.max(corr)
    active = np.where(corr >= max_v - tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        if shift >= 0:
            j1 = np.arange(0, n - shift)
            grad[j1] += (1.0 - h[j1 + shift])
            j2 = np.arange(shift, n)
            grad[j2] -= h[j2 - shift]
        else:
            s = -shift
            j1 = np.arange(s, n)
            grad[j1] += (1.0 - h[j1 - s])
            j2 = np.arange(0, n - s)
            grad[j2] -= h[j2 + s]
    
    grad /= float(len(active))
    return grad, max_v

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def subgradient_opt(h, n, dx, n_iter, lr_init, lr_decay, tol_init, tol_final):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    
    for it in range(n_iter):
        tol = tol_init * (tol_final / tol_init) ** (it / max(1, n_iter - 1))
        grad, _ = compute_subgradient(h, n, tol)
        lr = lr_init * (lr_decay ** it)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = c5_value(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new

def block_cd(h, n, dx, n_passes=10, block_size=16):
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    n_blocks = n // block_size
    
    for _ in range(n_passes):
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            end = min(start + block_size, n)
            bs = end - start
            block = best_h[start:end].copy()
            
            def blk_obj(params):
                h_try = best_h.copy()
                h_try[start:end] = np.clip(params, 0.0, 1.0)

            res = minimize(blk_obj, block, method='Nelder-Mead',

def binary_round(h, n):
    """Round to binary and adjust"""
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Remove excess ones: pick the ones with lowest h value
        order = np.argsort(h)  # ascending
        for idx in order:
            if n_ones <= target:
                break
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                n_ones -= 1
    elif n_ones < target:
        # Add missing ones: pick the zeros with highest h value
        order = np.argsort(-h)  # descending
        for idx in order:
            if n_ones >= target:
                break
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                n_ones += 1
    return h_bin

def simulated_annealing(h, n, dx, n_iter, T_init, T_end, rng):

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_v = np.max(corr)
        n_active = max(1, int((2*n-1) * active_frac))
        top_idx = np.argsort(corr)[-n_active:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()

                grad[j1] += (1.0 - h[j1 - s])

            a = np.zeros(2*n-1)
            b = corr.copy()  # will be adjusted

                shift = k - (n-1)
                a_k = 0.0

                if 0 <= i - shift < n:

                a[k] = a_k
                b[k] = corr[k] - h[i] * a_k

            # This is 1D convex minimization
            # Use ternary search
            lo, hi = 0.0, 1.0

                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a * m1 + b)
                f2 = np.max(a * m2 + b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            else:

                h = repair_h(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgradient_of_max(h, n, tol):
    corr = compute_corr(h)
    max_v = np.max(corr)
    active = np.where(corr >= max_v - tol)[0]
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        if shift >= 0:
            j1 = np.arange(0, n - shift)
            grad[j1] += (1.0 - h[j1 + shift])
            j2 = np.arange(shift, n)
            grad[j2] -= h[j2 - shift]
        else:
            s = -shift
            j1 = np.arange(s, n)
            grad[j1] += (1.0 - h[j1 - s])
            j2 = np.arange(0, n - s)
            grad[j2] -= h[j2 + s]
    grad /= float(len(active))
    return grad, max_v

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def subgradient_opt(h, n, dx, n_iter, lr_init, lr_decay, tol_init, tol_final, budget_fn, budget_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    for it in range(n_iter):
        if budget_fn() > budget_limit:
            break
        tol = tol_init * (tol_final / tol_init) ** (it / max(1, n_iter - 1))
        grad, _ = subgradient_of_max(h, n, tol)
        lr = lr_init * (lr_decay ** it)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = c5_value(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def block_cd(h, n, dx, n_passes, block_size, rng, budget_fn, budget_limit):
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    for _ in range(n_passes):
        if budget_fn() > budget_limit:
            break
        improved = False
        n_blocks = n // block_size
        for bi in range(n_blocks):
            if budget_fn() > budget_limit:
                break
            start = bi * block_size
            end = start + block_size
            block = best_h[start:end].copy()
            def blk_obj(params):
                h_try = best_h.copy()
                h_try[start:end] = np.clip(params, 0.0, 1.0)
                h_try = repair_h(h_try, n)
                return c5_value(h_try, dx)
            res = minimize(blk_obj, block, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-11, 'fatol': 1e-13})
            if res.x is not None and res.fun < best_c5:
                h_try = best_h.copy()
                h_try[start:end] = np.clip(res.x, 0.0, 1.0)
                h_try = repair_h(h_try, n)
                c5_try = c5_value(h_try, dx)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        order = np.argsort(h)
        for idx in order:
            if n_ones <= target:
                break
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                n_ones -= 1
    elif n_ones < target:
        order = np.argsort(-h)
        for idx in order:
            if n_ones >= target:
                break
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                n_ones += 1
    return h_bin

def simulated_annealing(h, n, dx, n_iter, T_init, T_end, rng, budget_fn, budget_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    c5_curr = best_c5
    for it in range(n_iter):
        if budget_fn() > budget_limit:
            break
        frac = it / max(1, n_iter - 1)
        T = T_init * (T_end / T_init) ** frac
        block_len = max(2, int(n / (8 + 30 * frac)))
        start = rng.randint(0, max(1, n - block_len + 1))
        h_new = h.copy()
        amp = 0.04 * (1.0 + 2.0 * frac)
        pert = rng.normal(0, amp, block_len)
        h_new[start:start + block_len] = np.clip(h_new[start:start + block_len] + pert, 0, 1)
        h_new = project_feasible(h_new, n)
        c5_new = c5_value(h_new, dx)
        if c5_new < c5_curr or rng.random() < np.exp(-min(50.0, (c5_new - c5_curr) / T)):
            h = h_new
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def slp_iterate(h0, n, dx, n_iter, active_frac, budget_fn, budget_limit):
    try:
        import cvxpy as cp
    except ImportError:
        return h0, c5_value(h0, dx)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    n_c = 2 * n - 1
    shifts = np.arange(n_c) - (n - 1)
    for it in range(n_iter):
        if budget_fn() > budget_limit:
            break
        corr = compute_corr(h)
        max_v = np.max(corr)
        n_active = max(5, int(n_c * active_frac))
        top_idx = np.argsort(corr)[-n_active:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n * 0.5, h_var >= 0, h_var <= 1]
        for k in top_idx:
            shift = shifts[k]
            grad = np.zeros(n)
            if shift >= 0:
                j1 = np.arange(0, n - shift)
                grad[j1] += (1.0 - h[j1 + shift])
                j2 = np.arange(shift, n)
                grad[j2] -= h[j2 - shift]
            else:
                s = -shift
                j1 = np.arange(s, n)
                grad[j1] += (1.0 - h[j1 - s])
                j2 = np.arange(0, n - s)
                grad[j2] -= h[j2 + s]
            rhs = -corr[k] + grad @ h
            cons.append(grad @ h_var - t_var / dx <= rhs)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            try:
                prob.solve(verbose=False)
            except Exception:
                break
        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0.0, 1.0)
            h_new = repair_h(h_new, n)
            c5_new = c5_value(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        else:
            break
    return best_h, best_c5

def smooth_max_fista(h, n, dx, beta, n_iter, budget_fn, budget_limit):
    h = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    n_c = 2 * n - 1
    shifts_arr = np.arange(n_c) - (n - 1)
    t_acc = 1.0
    for it in range(n_iter):
        if budget_fn() > budget_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc)) / 2.0
        theta = h + ((t_acc - 1.0) / t_new) * (h - h_prev)
        theta = project_feasible(theta, n)
        corr_theta = compute_corr(theta) * dx
        m = np.max(corr_theta)
        logsumexp = m + np.log(np.sum(np.exp(corr_theta - m)))
        p = np.exp(corr_theta - m) / np.sum(np.exp(corr_theta - m))
        grad = np.zeros(n)
        for k in range(n_c):
            if p[k] < 1e-8:
                continue
            shift = shifts_arr[k]
            if shift >= 0:
                j1 = np.arange(0, n - shift)
                grad[j1] += p[k] * (1.0 - theta[j1 + shift]) * dx
                j2 = np.arange(shift, n)
                grad[j2] -= p[k] * theta[j2 - shift] * dx
            else:
                s = -shift
                j1 = np.arange(s, n)
                grad[j1] += p[k] * (1.0 - theta[j1 - s]) * dx
                j2 = np.arange(0, n - s)
                grad[j2] -= p[k] * theta[j2 + s] * dx
        lr = 1.0 / (beta * n * 2.0)
        h_prev = h.copy()
        h = theta - lr * grad
        h = project_feasible(h, n)
        t_acc = t_new
        c5_curr = c5_value(h, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def budget_fn():
        return time.time() - t0

    h_parent = np.asarray(initial_h_values, dtype=np.float64)
    n_parent = len(h_parent)
    dx_parent = 2.0 / n_parent

    best_c5 = c5_value(best_h, dx_parent)
    best_n = n_parent

    n = best_n
    dx = 2.0 / n
    budget_limit = 0.15 * budget_s

    h_sg, c5_sg = subgradient_opt(best_h, n, dx, 8000, 0.3, 0.9997, 0.05, 1e-5, budget_fn, budget_limit)
    if c5_sg < best_c5:

    budget_limit = 0.30 * budget_s
    h_fista, c5_fista = smooth_max_fista(best_h, n, dx, beta=20.0, n_iter=3000, budget_fn=budget_fn, budget_limit=budget_limit)
    if c5_fista < best_c5:
        best_h, best_c5 = h_fista, c5_fista

    budget_limit = 0.50 * budget_s
    h_cd, c5_cd = coordinate_descent(best_h, n, dx, 30, rng, budget_fn, budget_limit)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd, c5_cd

    budget_limit = 0.65 * budget_s
    h_bcd, c5_bcd = block_cd(best_h, n, dx, 6, 16, rng, budget_fn, budget_limit)
    if c5_bcd < best_c5:
        best_h, best_c5 = h_bcd, c5_bcd

    budget_limit = 0.75 * budget_s
    h_bin = binary_round(best_h, n)
    h_bin, c5_bin = subgradient_opt(h_bin, n, dx, 5000, 0.2, 0.9998, 0.03, 1e-5, budget_fn, budget_limit)
    h_bin, c5_bin = coordinate_descent(h_bin, n, dx, 20, rng, budget_fn, budget_limit)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    budget_limit = 0.85 * budget_s
    h_sa, c5_sa = simulated_annealing(best_h, n, dx, 6000, 0.02, 1e-5, rng, budget_fn, budget_limit)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa

    budget_limit = 0.92 * budget_s
    h_cd2, c5_cd2 = coordinate_descent(best_h, n, dx, 20, rng, budget_fn, budget_limit)
    if c5_cd2 < best_c5:
        best_h, best_c5 = h_cd2, c5_cd2

    for n_try in [192, 320, 384]:
        if budget_fn() > 0.95 * budget_s:
            break
        dx_try = 2.0 / n_try
        x_src = np.linspace(0, 2, best_n, endpoint=False)
        x_dst = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_dst, x_src, best_h)
        h_try = repair_h(h_try, n_try)
        h_try, c5_try = coordinate_descent(h_try, n_try, dx_try, 15, rng, budget_fn, 0.97 * budget_s)
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try, c5_try, n_try

    n = best_n
    dx = 2.0 / n
    if budget_fn() < 0.98 * budget_s:
        h_final, c5_final = coordinate_descent(best_h, n, dx, 15, rng, budget_fn, budget_s)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
