from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_gradient_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_lp_minimax(h0, n, target_sum, active_lags, f_values):
    """Solve LP: min t s.t. grad_k·h - t <= c_k for active k, sum h = target, 0<=h<=1"""
    # Variables: [h(0), ..., h(n-1), t]
    nvar = n + 1
    
    # Objective: minimize t
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    
    # Inequality constraints: grad_k · h - t <= c_k for each active lag
    c_values = {}
    A_ub_rows = []
    b_ub_vals = []
    for idx in active_lags:
        lag = idx - (n - 1)  # convert from correlate index to actual lag
        grad = compute_gradient_for_lag(h0, lag)
        f_val = f_values[idx]
        c_k = f_val - np.dot(grad, h0)  # f(k;h0) - grad·h0
        row = np.zeros(nvar)
        row[:n] = grad
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(c_k)
    
    if A_ub_rows:
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_gradient_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        k = -lag
        if n - k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    return g

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_lp_for_step(h0, n, target_sum, active_indices, f_values):
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    
    A_ub_rows = []
    b_ub_vals = []
    for idx in active_indices:
        lag = idx - (n - 1)
        grad = compute_gradient_for_lag(h0, lag)
        f_val = f_values[idx]
        c_k = f_val - np.dot(grad, h0)
        row = np.zeros(nvar)
        row[:n] = grad
        row[-1] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(c_k)
    
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    if A_ub_rows:
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
    else:
        A_ub = None
        b_ub = None
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

def lp_minimax_optimize(h0, n, target_sum, deadline, max_iter=2000, margin=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        f_vals = compute_all_f(h)
        max_f = np.max(f_vals)
        
        if max_f < best_c5 - 1e-12:
            best_c5 = max_f
            best_h = h.copy()
        
        # Active lags
        active = np.where(f_vals > max_f - margin)[0]
        
        # Solve LP
        h_lp = solve_lp_for_step(h, n, target_sum, active, f_vals)
        if h_lp is None:
            break
        
        h_lp = np.clip(h_lp, 0.0, 1.0)
        h_lp = project_onto_constraint(h_lp, target_sum)
        
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < best_c5:
            h = h_lp
        else:
            # Backtrack
            improved = False
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_trial = h + alpha * (h_lp - h)
                h_trial = project_onto_constraint(h_trial, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h = h_trial
                    improved = True

            if not improved:

def pgd_polish(h0, n, target_sum, deadline, max_iter=10000):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.005
    stall = 0

        f_vals = compute_all_f(h)

        c5 = compute_c5(h_new)

            h = h_new
            stall = 0
        else:

            stall += 1

                    h = best_h + 0.001 * np.random.randn(n)
                    h = project_onto_constraint(h, target_sum)
                    lr = 0.005
    
    return best_h, best_c5

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-15:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def slsqp_optimize(h0, n, target_sum, deadline, tau=100.0):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except:
        pass
    return h0, compute_c5(h0)

    h_new = project_onto_constraint(h_new, n_new / 2.0)
    return h_new

def compute_gradient_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        k = -lag
        if n - k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    return g

        else:

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-15:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def slsqp_optimize(h0, n, target_sum, deadline, tau=150.0):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_onto_constraint(h_new, n_new / 2.0)
    return h_new

def binary_local_search(h0, n, target_sum, deadline, max_iter=50000):
    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    target = int(target_sum)
    if n_ones > target:
        ones = np.where(h == 1.0)[0]
        to_flip = np.random.choice(ones, size=n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        to_flip = np.random.choice(zeros, size=target - n_ones, replace=False)
        h[to_flip] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[j] = 1.0
        h[i] = 0.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or (T > 1e-6 and np.random.rand() < np.exp(-delta / T)):
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            h[j] = 0.0
            h[i] = 1.0
        T *= 0.9998
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 20
    
    best_h = None
    best_c5 = float('inf')
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None
    
    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0
    
    if best_h is None:
        n0 = 120
        best_h = np.full(n0, 0.5)
        best_c5 = compute_c5(best_h)
    
    n = len(best_h)
    
    # Phase 1: LP minimax on original resolution
    t_phase_end = min(t_start + 250, deadline - 200)
    if time.time() < t_phase_end:
        h_opt, c5_opt = lp_minimax_optimize(best_h, n, n / 2.0, t_phase_end)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: PGD polish on original resolution
    t_phase_end = min(time.time() + 100, deadline - 150)
    if time.time() < t_phase_end:
        n = len(best_h)
        h_opt, c5_opt = pgd_polish(best_h, n, n / 2.0, t_phase_end)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Higher resolution attempts
    for n_new in [200, 300, 400]:
        if time.time() > deadline - 100:
            break
        h_interp = interpolate_h(best_h, n_new)
        target = n_new / 2.0
        
        t_lp_end = min(time.time() + 180, deadline - 80)
        if time.time() < t_lp_end:
            h_opt, c5_opt = lp_minimax_optimize(h_interp, n_new, target, t_lp_end)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        t_pgd_end = min(time.time() + 60, deadline - 40)
        if time.time() < t_pgd_end:
            n_curr = len(best_h)
            h_opt, c5_opt = pgd_polish(best_h, n_curr, n_curr / 2.0, t_pgd_end)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 4: SLSQP polish
    if time.time() < deadline - 15:
        n = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, n, n / 2.0, deadline, tau=200.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 5: Final PGD
    if time.time() < deadline and best_h is not None:
        n = len(best_h)
        h_opt, c5_opt = pgd_polish(best_h, n, n / 2.0, deadline, max_iter=10000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 6: LP minimax one more time if time allows
    if time.time() < deadline - 5 and best_h is not None:
        n = len(best_h)
        h_opt, c5_opt = lp_minimax_optimize(best_h, n, n / 2.0, deadline, max_iter=1000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5, len(best_h)
