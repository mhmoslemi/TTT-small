```python
def compute_subgradient_vec(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    
    weights = np.exp((corr - max_val) / (alpha * max(1e-10, max_val)))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-8:
            continue
        L = idx - (n - 1)
        w = weights[idx]
        # ∂c_L/∂h[j] = [0≤j-L<n ? (1-h[j-L]) : 0] - [0≤j+L<n ? h[j+L] : 0]
        
        # First term: (1-h[j-L]) for j in [L, L+n-1] intersect [0, n-1]
        j_start = max(0, L)
        j_end = min(n, L + n)
        h_idx_start = j_start - L
        h_idx_end = j_end - L
        grad[j_start:j_end] += w * (1.0 - h[h_idx_start:h_idx_end])
        
        # Second term: -h[j+L] for j in [-L, -L+n-1] intersect [0, n-1]
        j2_start = max(0, -L)
        j2_end = min(n, -L + n)
        h2_idx_start = j2_start + L
        h2_idx_end = j2_end + L
        grad[j2_start:j2_end] -= w * h[h2_idx_start:h2_idx_end]
    
    return grad, max_val * dx
```

```python
def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    max_corr = np.max(corr)
    exp_args = (corr - max_corr) / T
    weights = np.exp(exp_args)
    weights /= np.sum(weights)
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-10:
            continue
        L = idx - (n - 1)
        w = weights[idx]
        
        # dc_L/dh[j] = [0≤j-L<n ? (1-h[j-L]) : 0] - [0≤j+L<n ? h[j+L] : 0]
        
        # First term: (1-h[j-L]) for j where 0≤j-L<n, i.e., L≤j≤L+n-1
        j_start = max(0, L)
        j_end = min(n, L + n)
        if j_start < j_end:
            h_idx = slice(j_start - L, j_end - L)
            grad[j_start:j_end] += w * (1.0 - h[h_idx])
        
        # Second term: -h[j+L] for j where 0≤j+L<n, i.e., -L≤j≤n-1-L
        j2_start = max(0, -L)
        j2_end = min(n, n - L)
        if j2_start < j2_end:
            h2_idx = slice(j2_start + L, j2_end + L)
            grad[j2_start:j2_end] -= w * h[h2_idx]
    
    return dx * grad
```

import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    max_corr = np.max(corr)
    exp_args = (corr - max_corr) / T
    weights = np.exp(exp_args)
    wsum = np.sum(weights)
    weights /= wsum
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-10:
            continue
        L = idx - (n - 1)
        w = weights[idx]
        
        # First term: +w*(1-h[j-L]) for j in [max(0,L), min(n, L+n))
        js = max(0, L)
        je = min(n, L + n)
        if js < je:
            grad[js:je] += w * (1.0 - h[js-L:je-L])
        
        # Second term: -w*h[j+L] for j in [max(0,-L), min(n, n-L))
        js2 = max(0, -L)
        je2 = min(n, n - L)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2+L:je2+L]
    
    return dx * grad, corr

def proj_grad_descent(h, T, max_iter, t_start, time_limit, initial_step=0.1):
    n = len(h)
    best = h.copy()
    best_score = score_c5(best)[0]
    step = initial_step
    stall = 0
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        grad, corr = lse_gradient(h, T)
        # Line search
        for lr in [step, step*0.5, step*0.25, step*0.1]:
            h_new = h - lr * grad
            h_new = project_feasible(h_new)
            s = score_c5(h_new)[0]
            if s < best_score:
                best = h_new.copy()
                best_score = s
                h = h_new
                stall = 0
                step = min(step * 1.05, 0.5)
                break
        else:
            # No improvement, still move (to escape)
            h_new = h - step * grad
            h_new = project_feasible(h_new)
            s = score_c5(h_new)[0]
            h = h_new
            step *= 0.8
            stall += 1
            if stall > 200:
                # Reset toward best with noise
                h = best + 0.05 * np.random.randn(n)
                h = project_feasible(h)
                stall = 0
    
    return best, best_score

def binary_tabu_search(n, t_start, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[:n//2] = 1.0
    rng.shuffle(h)
    
    best = h.copy()
    best_score = score_c5(h)[0]
    current_score = best_score
    tabu = {}
    tabu_len = 20
    it = 0
    temp = 1.0
    
    while time.time() - t_start < time_limit:
        it += 1
        # Generate neighbors: flip one 0 to 1 and one 1 to 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swaps
        best_move = None
        best_move_score = current_score
        n_tries = 50
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if tabu.get(i, 0) > it or tabu.get(j, 0) > it:
                continue
            h[i] = 0
            h[j] = 1
            s = score_c5(h)[0]
            h[i] = 1
            h[j] = 0
            if s < best_move_score:
                best_move_score = s
                best_move = (i, j)
        
        if best_move is None:
            # Aspiration criterion: accept if better than best
            best_move = (ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))])
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            best_move_score = score_c5(h)[0]
            tabu[best_move[0]] = it + tabu_len
            tabu[best_move[1]] = it + tabu_len
        else:
            # Accept the move
            tabu[best_move[0]] = it + tabu_len
            tabu[best_move[1]] = it + tabu_len
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            current_score = best_move_score
        
        if current_score < best_score:
            best = h.copy()
            best_score = current_score
            temp = max(temp * 0.99, 0.01)
        else:
            temp = max(temp * 0.995, 0.01)
        
        # Occasionally randomize
        if rng.random() < 0.01:
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
            current_score = score_c5(h)[0]
    
    return best, best_score

Let me also think about the 2-swap local search more carefully. Instead of random swaps, I should use informed swaps:

For the current h, identify the worst lags (those closest to the maximum). For each worst lag L, the "bad" positions are those i where h[i]=1 and h[i+L]=0. To reduce the correlation at lag L, I should:

def directed_swap_search(h, t_start, time_limit):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_score = score_c5(best)[0]
    
    while time.time() - t_start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_val = np.max(corr)
        
        # Find worst lags
        threshold = max_val - 0.5
        bad_lags = np.where(corr >= threshold)[0]
        
        improved = False
        for idx in bad_lags:
            L = idx - (n - 1)
            # Find "bad" positions: i where best[i] > 0.5 and best[i+L] < 0.5 (or vice versa)
            # Actually, the correlation counts h[i]*(1-h[i+L]), so "bad" means h[i]≈1 and h[i+L]≈0
            
            if L >= 0:
                valid_i = range(n - L)
            else:
                valid_i = range(-L, n)
            
            for i in valid_i:
                if L >= 0:
                    j = i + L
                else:
                    j = i + L
                if not (0 <= j < n):
                    continue

                # Try to fix by reducing h[i] or increasing h[j] (or vice versa)

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def lse_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_a = (corr - max_c) / T
    w = np.exp(exp_a)
    ws = np.sum(w)
    w /= ws
    obj = dx * (max_c + T * np.log(ws))
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-10:
            continue
        L = idx - (n - 1)
        wi = w[idx]
        js = max(0, L)
        je = min(n, L + n)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js - L:je - L])
        js2 = max(0, -L)
        je2 = min(n, n - L)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2 + L:je2 + L]
    return obj, dx * grad


def pd_optimize(h, T, t_start, time_limit, init_step=0.05):
    n = len(h)
    best = h.copy()
    best_obj = lse_obj_grad(best, T)[0]
    cur_obj = best_obj
    step = init_step
    stall = 0
    while time.time() - t_start < time_limit:
        obj, grad = lse_obj_grad(h, T)
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            rho = step * factor
            h_new = h - rho * grad
            h_new = project_feasible(h_new)
            new_obj = lse_obj_grad(h_new, T)[0]
            if new_obj < cur_obj - 1e-14:
                h = h_new
                cur_obj = new_obj
                if cur_obj < best_obj:
                    best = h.copy()
                    best_obj = cur_obj
                step = min(step * 1.05, 0.3)
                stall = 0
                accepted = True
                break
        if not accepted:
            step *= 0.8
            stall += 1
            h_new = h - step * grad
            h_new = project_feasible(h_new)
            h = h_new
            if stall > 150:
                break
    return best, best_obj


def compute_profiles(h):
    n = len(h)
    L_arr = np.arange(2 * n - 1) - (n - 1)
    profiles = np.zeros((n, 2 * n - 1))
    for p in range(n):
        idx1 = p + L_arr
        m1 = (idx1 >= 0) & (idx1 < n)
        profiles[p, m1] += h[idx1[m1]]
        idx2 = p - L_arr
        m2 = (idx2 >= 0) & (idx2 < n)
        profiles[p, m2] += h[idx2[m2]]
    return profiles


def binary_best_swap(h, t_start, time_limit):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_score = score_c5(best)
    rng = np.random.RandomState(12345)
    while time.time() - t_start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_corr = np.max(corr)
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        profiles = compute_profiles(best)
        best_new_max = max_corr
        best_p = -1
        best_q = -1
        for p in ones:
            A = corr + profiles[p]
            diffs = A[None, :] - profiles[zeros, :]
            new_maxs = np.max(diffs, axis=1)
            min_idx = np.argmin(new_maxs)
            if new_maxs[min_idx] < best_new_max:
                best_new_max = new_maxs[min_idx]
                best_p = p
                best_q = zeros[min_idx]
        if best_p >= 0 and best_new_max < max_corr:
            best[best_p] = 0.0
            best[best_q] = 1.0
            best_score = best_new_max * dx
        else:
            improved = False
            for _ in range(200):
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
                trial = best.copy()
                trial[p] = 0.0
                trial[q] = 1.0
                s = score_c5(trial)
                if s < best_score:
                    best = trial
                    best_score = s
                    improved = True
                    break
            if not improved:
                break
    return best, best_score


def binary_to_continuous(h_bin, t_start, time_limit):
    n = len(h_bin)
    h = h_bin.astype(float)
    h = project_feasible(h)
    best = h.copy()
    best_score = score_c5(best)
    temps = [0.3, 0.1, 0.03, 0.01]
    for T in temps:
        tl = time_limit - (time.time() - t_start)
        if tl < 1:
            break
        res, obj = pd_optimize(best, T, t_start, t_start + tl * 0.7, init_step=0.03)
        if obj < lse_obj_grad(best, T)[0]:
            best = res
            best_score = score_c5(best)
    return best, best_score


def random_binary(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h


def try_resolution(n, t_start, time_limit, seed):
    rng = np.random.RandomState(seed)
    best_h = None
    best_score = 1.0
    n_attempts = 3
    per_attempt = time_limit / n_attempts
    for att in range(n_attempts):
        tl = time_limit - (time.time() - t_start)
        if tl < 2:
            break
        h_bin = random_binary(n, rng)
        alloc = tl * 0.4
        h_bin, s_bin = binary_best_swap(h_bin, t_start, t_start + alloc)
        tl2 = time_limit - (time.time() - t_start)
        if tl2 < 1:
            if s_bin < best_score:
                best_score = s_bin
                best_h = h_bin.copy()
            continue
        alloc2 = tl2 * 0.5
        h_cont, s_cont = binary_to_continuous(h_bin, t_start, t_start + alloc2)
        if s_cont < best_score:
            best_score = s_cont
            best_h = h_cont.copy()
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
    if best_h is None:
        return None, 1.0
    return best_h, best_score


def local_refine(h, t_start, time_limit):
    n = len(h)
    best = h.copy()
    best_score = score_c5(best)
    step = 0.02
    stall = 0
    while time.time() - t_start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_idx = np.argmax(corr)
        L = max_idx - (n - 1)
        grad = np.zeros(n)
        js = max(0, L)
        je = min(n, L + n)
        if js < je:
            grad[js:je] += (1.0 - best[js - L:je - L])
        js2 = max(0, -L)
        je2 = min(n, n - L)
        if js2 < je2:
            grad[js2:je2] -= best[js2 + L:je2 + L]
        grad *= 2.0 / n
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.1]:
            rho = step * factor
            h_new = best - rho * grad
            h_new = project_feasible(h_new)
            s = score_c5(h_new)
            if s < best_score - 1e-14:
                best = h_new
                best_score = s
                step = min(step * 1.05, 0.15)
                stall = 0
                accepted = True
                break
        if not accepted:
            step *= 0.85
            stall += 1
            if stall > 100:
                break
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = random_binary(85, rng)
    n = len(h)
    h = project_feasible(h)
    best_h = h.copy()
    best_c5 = score_c5(best_h)

    phase1_end = t0 + budget_s * 0.30
    temps = [0.5, 0.15, 0.05, 0.02, 0.008]
    for T in temps:
        if time.time() >= phase1_end:
            break
        tl = phase1_end - time.time()
        if tl < 1:
            break
        res, obj = pd_optimize(h, T, t0, t0 + tl * 0.8, init_step=0.04)
        s = score_c5(res)
        if s < best_c5:
            best_c5 = s
            best_h = res.copy()
        h = res.copy()

    if time.time() >= phase1_end or time.time() - t0 >= budget_s * 0.35:
        pass

    phase2_end = t0 + budget_s * 0.65
    n_slots = 5
    per_slot = (phase2_end - time.time()) / n_slots
    for slot in range(n_slots):
        if time.time() >= phase2_end:
            break
        slot_end = time.time() + per_slot
        if slot == 0:
            h_bin = (h > 0.5).astype(float)
            h_bin = project_feasible(h_bin)
        else:
            h_bin = best_h.copy()
            n_flip = max(1, n // 10)
            flip_indices = rng.choice(n, size=n_flip, replace=False)
            h_bin[flip_indices] = 1.0 - h_bin[flip_indices]
            h_bin = project_feasible(h_bin)
        h_bin2, s_bin = binary_best_swap(h_bin, t0, slot_end * 0.5)
        if s_bin < best_c5:
            best_c5 = s_bin
            best_h = h_bin2.copy()
        if time.time() < slot_end:
            h_cont, s_cont = binary_to_continuous(h_bin2, t0, slot_end)
            if s_cont < best_c5:
                best_c5 = s_cont
                best_h = h_cont.copy()

    phase3_end = t0 + budget_s * 0.85
    if time.time() < phase3_end:
        n_options = [60, 90, 120]
        per_res = (phase3_end - time.time()) / len(n_options)
        for ni in n_options:
            if time.time() >= phase3_end:
                break
            alloc = min(per_res, phase3_end - time.time())
            if alloc < 5:
                break
            h_try, s_try = try_resolution(ni, t0, alloc, seed + ni)
            if h_try is not None and s_try < best_c5:
                best_c5 = s_try
                best_h = h_try.copy()

    if time.time() < t0 + budget_s - 10:
        final_tl = t0 + budget_s - 10 - time.time()
        if final_tl > 5:
            best_h, best_c5 = local_refine(best_h, t0, t0 + final_tl * 0.6)
            n = len(best_h)
            h_ref, obj_ref = pd_optimize(best_h, 0.005, t0, t0 + final_tl, init_step=0.02)
            s_ref = score_c5(h_ref)
            if s_ref < best_c5:
                best_c5 = s_ref
                best_h = h_ref.copy()

    best_h = project_feasible(best_h)
    best_c5 = score_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
