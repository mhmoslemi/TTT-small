def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    w = exp_shifted / s
    val = c_max + np.log(s) / alpha
    
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    m1 = rows + n - 1 - cols
    m2 = cols - rows + n - 1
    w_full = np.zeros(2 * n - 1)
    w_full[:] = w
    M1 = np.where((m1 >= 0) & (m1 < 2*n-1), w_full[np.clip(m1, 0, 2*n-2)], 0.0)
    M2 = np.where((m2 >= 0) & (m2 < 2*n-1), w_full[np.clip(m2, 0, 2*n-2)], 0.0)
    grad = dx * (M1.dot(1.0 - h) - M2.dot(h))
    return val, grad

def _obj_fn(val, alpha, target, lam):
    h = np.clip(val, 0, 1)
    v, _ = smooth_max_and_grad(h, alpha)
    return v + lam * (np.sum(h) - target) ** 2

def resample_parent(h_parent, n):
    if len(h_parent) == n:
        return h_parent.copy()
    x_old = np.linspace(0, 2, len(h_parent))
    x_new = np.linspace(0, 2, n)
    h_new = np.interp(x_new, x_old, h_parent)
