```python
def coord_best_t(A, B):
    """Find t in [0,1] minimizing max(A + B*t)"""
    # Try a grid
    ts = np.linspace(0, 1, 200)
    vals = np.max(A[None, :] + B[None, :] * ts[:, None], axis=1)
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    # Refine locally
    lo = max(0, ts[max(0, best_idx-1)])
    hi = min(1, ts[min(len(ts)-1, best_idx+1)])
    ts2 = np.linspace(lo, hi, 200)
    vals2 = np.max(A[None, :] + B[None, :] * ts2[:, None], axis=1)
    best_t = ts2[np.argmin(vals2)]
    return best_t
```

The problem is equivalent to: find a function h: [0,2] → [0,1] with ∫h=1 that minimizes max_k ∫h(x)(1-h(x+k))dx.

For the continuous version, there might be an optimal solution that's a "bang-bang" function (taking values 0 and 1 almost everywhere) with specific switching points. The discrete version approximates this.

One natural candidate: h(x) = 1 if x mod (2/n) is in some set S (of measure n/2 out of n slots). The optimal S is the one that maximizes min_k |S ∩ (S-k)|.

This is a well-studied problem in additive combinatorics. For |G| = n (a cyclic group), |S| = n/2, we want to maximize min_{k≠0} |S ∩ (S-k)|.

For n = 85: |S| = 42 or 43 (since we need sum = 42.5, we'll have 42 ones, one 0.5, and 42 zeros, or similar).

The theoretical best for |S| = n/2 in a group of size n: the minimum intersection is at least n/4 - O(√n) for random S, but structured sets can do much better.

Actually, for a "balanced" set (like the first half of the group), S = {0, 1, ..., n/2-1}:
|S ∩ (S-k)| = max(0, n/2 - |k|) for k > 0.
So min_k |S ∩ (S-k)| = 0 (at k = n/2). That's terrible.

For a "well-distributed" set like S = {j: j mod 4 ∈ {0,1}} (taking every other pair):
This is like h = 1,1,0,0,1,1,0,0,...
|S ∩ (S-k)| depends on k mod 4. For k=0: |S| = n/2. For k=1: |S ∩ (S-1)| ≈ n/2. For k=2: |S ∩ (S-2)| ≈ n/2 - 2 (loses 2 from the boundaries of each block). For k=3: |S ∩ (S-3)| ≈ n/4.

So the minimum is at k=3 (or k=1 in the other direction), giving about n/4. Then corr[k] ≈ n/2 - n/4 = n/4, and C₅ ≈ (n/4)*(2/n) = 1/2 = 0.5. Still not great.

What about a "quadratic" pattern? h[j] = 1 if j² mod p < p/2 for a prime p ≈ n? This is related to quadratic residue difference sets, which have the property that |S ∩ (S-k)| is almost constant for all k ≠ 0.

For a (v, k, λ) difference set in a group of order v: |S ∩ (S-k)| = λ for all k ≠ 0.
If v = n = 85 = 5 * 17, we'd need a difference set with v=85.
k = n/2 = 42.5 (not integer, so this doesn't directly apply).

For v = 84 (= 2² * 3 * 7), k = 42: we'd need a (84, 42, λ) difference set.
The parameters must satisfy k(k-1) = λ(v-1), so 42*41 = λ*83, λ = 1722/83 ≈ 20.75. Not an integer, so no perfect difference set exists.

For v = 85 (= 5 * 17): k = 42 or 43.
k=42: 42*41 = λ*84 → λ = 1722/84 = 20.5. Not integer.
k=43: 43*42 = λ*84 → λ = 1806/84 = 21.5. Not integer.

So perfect difference sets don't exist for these parameters. But we can get close.

The **closest to a difference set** would minimize the variance of |S ∩ (S-k)| over k, which is exactly what we want!

```python
_glob_eps = 100.0
_glob_n = 85
_glob_target = 42.5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def _sm_obj(h):
    n = _glob_n
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = _glob_eps * (corr - m)
    return (m + (1.0 / _glob_eps) * np.log(np.sum(np.exp(shifted)))) * dx

def _sm_grad(h):
    n = _glob_n
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = _glob_eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            g = 0.0
            if 0 <= j - lag < n:
                g += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                g += (-h[j + lag])
            grad[j] += w * g
    grad *= dx
    return grad

    return np.sum(h) - _glob_target

def coordinate_descent(h, max_sweeps=200, grid_size=300):
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        for i in range(n):
            corr = np.correlate(h, 1 - h, mode="full")
            k_offset = n - 1
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            qi = -1
            for idx in range(2 * n - 1):
                lag = idx - k_offset
                if lag == 0:
                    A[idx] = corr[idx] - h[i] * (1.0 - h[i])
                    qi = idx
                else:
                    D = 0.0
                    if 0 <= i - lag < n:
                        D += (1.0 - h[i - lag])
                    if 0 <= i + lag < n:
                        D += (-h[i + lag])
                    A[idx] = corr[idx] - h[i] * D
                    B[idx] = D
            ts = np.linspace(0.0, 1.0, grid_size)
            vals = A[None, :] + B[None, :] * ts[:, None]
            if qi >= 0:
                vals[:, qi] = A[qi] + ts - ts * ts
            max_vals = np.max(vals, axis=1)
            h[i] = ts[np.argmin(max_vals)]
        h = project_feasible(h, n)
    return h
```
