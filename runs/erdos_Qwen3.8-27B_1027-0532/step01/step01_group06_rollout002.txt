import numpy as np
from scipy.optimize import minimize
import time

_CURRENT_T = 1.0


def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip((corr - cmax) / _CURRENT_T, -500.0, 0.0)
    return dx * (cmax + _CURRENT_T * np.log(np.sum(np.exp(shifted))))


def project_feasible(x, n=None):
    if n is None:
        n = len(x)
    target = n / 2.0
    lo = -20.0
    hi = 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def sum_con(x):
    return np.sum(x) - len(x) / 2.0


def sum_con_jac(x):
    return np.ones(len(x))


def try_slsqp(h, n, T, max_iter):
    global _CURRENT_T
    _CURRENT_T = T
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None and res.fun < 1e10:
            x_new = project_feasible(res.x, n)
            s = score_c5(x_new)
            if s < 1e10:
                return x_new, s
    except Exception:
        pass
    return h, score_c5(h)

    h_new = np.interp(x_new, x_ext, h_ext)
    return project_feasible(h_new, n_new)
