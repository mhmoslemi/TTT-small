    ex = np.exp(alpha * (x - m))

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def project_feasible(h, target_sum):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def grad_obj(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad


def _fun_unconstrained(x, alpha):
    return smooth_max_val(compute_correlations(x), alpha)


def _jac_unconstrained(x, alpha):
    return grad_obj(x, alpha)


def _cons_fun_unconstrained(x):
    return np.sum(x) - len(x) / 2.0


def _cons_jac_unconstrained(x):
    return np.ones(len(x))


def slsqp_opt(h0, alpha, maxiter):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _fun_unconstrained, h0, method='SLSQP', bounds=bounds,
        jac=_jac_unconstrained, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': _cons_fun_unconstrained,
                      'jac': _cons_jac_unconstrained}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def proj_grad_fista(h0, alpha, target_sum, max_iter, lr):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        g = grad_obj(z, alpha)
        z_new = project_feasible(z - lr * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = z_new + ((t_k - 1.0) / t_new) * (z_new - x)
        x_new = project_feasible(x_new, target_sum)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
        z = z_new
        t_k = t_new
    return best_x, best_score


def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h


def _ap_fun(g, alpha, n_full):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)


def _ap_jac(g, alpha, n_full):
    h = antipal_h(g)
    grad_h = grad_obj(h, alpha)
    half = len(g)
    grad_g = grad_h[:half].copy()
    for i in range(half):
        j = n_full - 1 - i
        grad_g[i] += -grad_h[j]
    return grad_g
