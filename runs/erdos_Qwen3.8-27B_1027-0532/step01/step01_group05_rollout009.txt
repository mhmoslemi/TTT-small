    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_obj_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    
    nc = len(corr)
    grad = np.zeros(n)

        if w < 1e-12:
            continue
        s = m - (n - 1)

            grad[0:n-s] += w * (1.0 - h[s:n])
            grad[s:n] += -w * h[0:n-s]

            tp = -s
            grad[tp:n] += w * (1.0 - h[0:n-tp])
            grad[0:n-tp] -= w * h[tp:n]

def pure_subgrad(h0, n, n_iter=2000, step0=0.1):
    """Pure subgradient using only the worst lag."""
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step0
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        s = idx - (n - 1)
        
        grad = np.zeros(n)
        if s >= 0:
            grad[0:n-s] += (1.0 - h[s:n])
            grad[s:n] += -h[0:n-s]
        else:
            tp = -s
            grad[tp:n] += (1.0 - h[0:n-tp])
            grad[0:n-tp] += -h[tp:n]
        grad *= dx
        
        # Polyak-like step size
        h_new = h - step * grad
        h_new = project_onto_constraint(h_new, n)
        c5 = compute_c5(h_new, n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.95
        
        step = max(step * 0.9995, 0.0001)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def get_obj_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    nc = len(corr)
    grad = np.zeros(n)
    for m in range(nc):
        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[0:n-s] += w * (1.0 - h[s:n])
            grad[s:n] += -w * h[0:n-s]
        else:
            tp = -s
            grad[tp:n] += w * (1.0 - h[0:n-tp])
            grad[0:n-tp] += -w * h[tp:n]
    grad *= dx
    return obj, grad


def slsqp_solve(h0, n, t, max_iter=500):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)

    def fun(x):
        v, _ = get_obj_grad(x, n, t)
        return v

    def jac(x):
        _, g = get_obj_grad(x, n, t)
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

            h = h_new
            step = min(step * 1.02, step0)

            v *= 0.5
        if step < 0.001:

    return best_h, best_c5

        h_new[idx] = 1.0 - h_new[idx]
    else:
        h_new = h + rng.randn(n) * 0.08
