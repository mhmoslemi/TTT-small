    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = np.max(c)

        return m
    shifted = alpha * (c - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

        else:

        hp[j] += eps
        hm = h.copy()
        hm[j] -= eps
        grad[j] = (_smooth_obj(hp, alpha) - _smooth_obj(hm, alpha)) / (2.0 * eps)
    return grad

            return h_new, f_new
        step *= 0.4
    return h, f0

    except Exception:
        pass
    return h

            if tl() > 80.0:
                h_s = _do_slsqp(h_p, target_sum, 3000, 200, 1e-13)
                h_s = _fix_sum(h_s, target_sum)
                c5_s = _c5_of(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()

    # Phase 3: Fine refinement on best
    if tl() > 80.0:
        for alpha in [5000, 10000, 20000]:
            if tl() < 40.0:
                break
            base_step = max(0.01, 0.5 / alpha)
            for s in range(80):
                if tl() < 40.0:
                    break
                best_h, _ = _pg_step(best_h, alpha, target_sum, base_step)
            best_h = _fix_sum(best_h, target_sum)
            new_c5 = _c5_of(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5

    # Phase 4: SLSQP final polish
    if tl() > 50.0:
        for alpha in [5000, 15000, 30000]:
            if tl() < 30.0:
                break
            h_f = _do_slsqp(best_h, target_sum, alpha, 300, 1e-14)
            h_f = _fix_sum(h_f, target_sum)
            c5_f = _c5_of(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 5: Very fine PG polish
    if tl() > 30.0:
        for alpha in [10000, 30000, 50000]:
            if tl() < 20.0:
                break
            base_step = 0.005
            for s in range(100):
                if tl() < 20.0:
                    break
                best_h, _ = _pg_step(best_h, alpha, target_sum, base_step)
            best_h = _fix_sum(best_h, target_sum)
            new_c5 = _c5_of(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5

    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n)
