import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def softmax_obj_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    obj = max_c + log_sum / alpha
    w = exp_s / exp_s.sum()
    grad = np.ones(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-14:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] -= wk * h[s:]
            grad[s:] -= wk * h[:n - s]
        else:
            grad[-s:] -= wk * h[:n + s]
            grad[:n + s] -= wk * h[-s:]
    grad *= dx
    if penalty > 0:
        dev = h.sum() - target
        obj += penalty * dev * dev
        grad += 2.0 * penalty * dev
    return obj, grad


def lbfgs_soft(h0, alpha, penalty, max_iter):
    n = len(h0)

    def fun(x):
        return softmax_obj_grad(x, alpha, penalty)[0]

    def jac(x):
        return softmax_obj_grad(x, alpha, penalty)[1]

    res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'gtol': 1e-12, 'maxls': 40})
    return res.x


def softmax_anneal(h0, alphas, penalty, time_budget):
    h = project_feasible(h0, len(h0) / 2.0).astype(float)
    best = h.copy()
    best_c5 = eval_c5(best)
    t0 = time.time()
    for alpha in alphas:
        rem = time_budget - (time.time() - t0)
        if rem < 1:
            break
        max_iter = max(30, min(600, int(rem * 8)))
        h_raw = lbfgs_soft(h, alpha, penalty, max_iter)
        h = project_feasible(h_raw, len(h) / 2.0)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best


def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    t0 = time.time()

    def f(x):
        if time.time() - t0 > tm * 0.9:
            raise RuntimeError("timeout")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

    res = minimize(f, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': 8000, 'ftol': 1e-18})
    h1 = project_feasible(res.x, tgt)
    if eval_c5(h1) < eval_c5(h0):
        return h1
    return h0


defcoord_descent(h0, n_rounds=50, step=0.01):

    tgt = n / 2.0
    h = project_feasible(h0, tgt).astype(float)
    best_c5 = eval_c5(h)
    for _ in range(n_rounds):
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                h_new = h.copy()
                h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h_new = project_feasible(h_new, tgt)
                c5 = eval_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
    return h


def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)
