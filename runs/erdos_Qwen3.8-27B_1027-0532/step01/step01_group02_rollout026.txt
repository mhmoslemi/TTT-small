C(k*) = dx * Σ_i h[i] * (1 - h[i + k* - (n-1)])

- m appears in h[i+shift] when i+shift = m, i.e., i = m - shift: contributes dx * (-h[m-shift]) if m-shift is valid

So: ∂C(k*)/∂h[m] = dx * [(1 - h[m+shift]) * 1_{0≤m+shift<n} - h[m-shift] * 1_{0≤m-shift<n}]

        # 1 - h[m+shift] for m where m+shift < n
        valid1 = slice(0, n - shift)
        g[valid1] += (1.0 - h[shift:n])
        # -h[m-shift] for m where m-shift >= 0
        valid2 = slice(shift, n)
        g[valid2] -= h[0:n-shift]
    else:
        s = -shift
        # 1 - h[m+shift] = 1 - h[m-s] for m where m-s >= 0
        valid1 = slice(s, n)
        g[valid1] += (1.0 - h[0:n-s])
        # -h[m-shift] = -h[m+s] for m where m+s < n
        valid2 = slice(0, n - s)
        g[valid2] -= h[s:n]
    g *= dx
    return g

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def subgradient_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[0:n-shift] += (1.0 - h[shift:n])
        g[shift:n] -= h[0:n-shift]
    else:
        s = -shift
        g[s:n] += (1.0 - h[0:n-s])
        g[0:n-s] -= h[s:n]
    g *= dx
    return g

def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

        return logsumexp_obj(x, n, beta)
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    try:
        r = minimize(
            objective, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint_fun}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if r.success or np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            return r.x, evaluate_c5(r.x)
    except Exception:
        pass
    return h, evaluate_c5(h)

            best_h = h_new.copy()
        h_curr = h_new
        step = step_init / (1 + 0.001 * it)
    return best_h, best_c5

def multi_subgrad(h, n, max_iter=500, n_active=5):
    """Subgradient using top-n active lags."""
    h_curr = h.copy().astype(float)
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    step = 0.01
    for it in range(max_iter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full") * (2.0/n)
        top_k = np.argsort(corr)[-n_active:]
        g = np.zeros(n)
        for k in top_k:
            shift = k - (n - 1)
            if shift >= 0:
                g[0:n-shift] += (1.0 - h_curr[shift:n])
                g[shift:n] -= h_curr[0:n-shift]
            else:

            best_c5 = c5
            best_h = h_new.copy()

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def lse_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def lse_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights /= total
    g = np.zeros(n)
    significant = weights > 1e-10
    idx = np.where(significant)[0]
    for k in idx:
        w = weights[k]
        shift = k - (n - 1)
        if shift >= 0:
            g[0:n-shift] += w * (1.0 - h[shift:n])
            g[shift:n] -= w * h[0:n-shift]
        else:
            s = -shift
            g[s:n] += w * (1.0 - h[0:n-s])
            g[0:n-s] -= w * h[s:n]
    g *= dx
    return g

def lse_slsqp(h, n, beta, maxiter):
    def objective(x):
        return lse_obj(x, n, beta)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    try:
        r = minimize(
            objective, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con_fun}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            return r.x, evaluate_c5(r.x)
    except Exception:
        pass
    return h, evaluate_c5(h)

    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)

    step = step_init

        v = momentum * v - step * g

            best_h = h_new.copy()

        h_sg, c5_sg = lse_grad_descent(best_h, n_init, 200, 6000, 0.004, 0.85)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    elapsed = time.time() - t0
    if budget_s - elapsed > 90:
        h_sg, c5_sg = subgrad_multi(best_h, n_init, 4000, 3, 0.004)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    elapsed = time.time() - t0
    if budget_s - elapsed > 60:
        h_pol, c5_pol = lse_slsqp(best_h, n_init, 500, 300)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    elapsed = time.time() - t0
    if budget_s - elapsed > 240:
        max_perts = 8
        for p in range(max_perts):
            elapsed = time.time() - t0
            if budget_s - elapsed < 90:
                break
            noise_scale = 0.04 / (1.0 + p * 0.5)
            h_pert = best_h + rng.normal(0, noise_scale, n_init)
            h_pert = repair_h(h_pert, n_init)
            h_sg, c5_sg = lse_grad_descent(h_pert, n_init, 150, 3000, 0.003, 0.8)
            if c5_sg < best_c5:

            elapsed = time.time() - t0
            if budget_s - elapsed > 45:
                h_pol, c5_pol = lse_slsqp(best_h, n_init, 300, 200)
                if c5_pol < best_c5:
                    best_h, best_c5 = h_pol, c5_pol

    elapsed = time.time() - t0
    if budget_s - elapsed > 300:
        n_target = 500
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_target)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair_h(h_interp, n_target)
        for beta in [50, 150, 300]:
            elapsed = time.time() - t0
            if budget_s - elapsed < 60:
                break
            h_opt, c5_opt = lse_slsqp(h_interp, n_target, beta, 150)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        elapsed = time.time() - t0
        if budget_s - elapsed > 90:
            h_sg, c5_sg = lse_grad_descent(best_h, n_target, 200, 4000, 0.003, 0.85)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg
        elapsed = time.time() - t0
        if budget_s - elapsed > 45:
            h_pol, c5_pol = lse_slsqp(best_h, n_target, 400, 200)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol

    elapsed = time.time() - t0
    if budget_s - elapsed > 360:
        n_target = 700
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_target)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair_h(h_interp, n_target)
        for beta in [80, 200]:
            elapsed = time.time() - t0
            if budget_s - elapsed < 60:
                break
            h_opt, c5_opt = lse_slsqp(h_interp, n_target, beta, 100)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        elapsed = time.time() - t0
        if budget_s - elapsed > 90:
            h_sg, c5_sg = lse_grad_descent(best_h, n_target, 200, 3000, 0.002, 0.8)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg
        elapsed = time.time() - t0
        if budget_s - elapsed > 45:
            h_pol, c5_pol = lse_slsqp(best_h, n_target, 300, 150)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol

    elapsed = time.time() - t0
    if budget_s - elapsed > 120:
        n_cur = len(best_h)
        h_bin, c5_bin = binary_local_search(best_h, n_cur, 2000, rng)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin
        h_rel, c5_rel = lse_slsqp(best_h, n_cur, 100, 200)
        if c5_rel < best_c5:
            best_h, best_c5 = h_rel, c5_rel
        elapsed = time.time() - t0
        if budget_s - elapsed > 60:
            h_sg, c5_sg = lse_grad_descent(best_h, n_cur, 150, 3000, 0.003, 0.85)
            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    elapsed = time.time() - t0
    if budget_s - elapsed > 60:
        n_cur = len(best_h)
        h_sw, c5_sw = pair_swap_refine(best_h, n_cur, 5000, rng)
        if c5_sw < best_c5:
            best_h, best_c5 = h_sw, c5_sw
        h_pol, c5_pol = lse_slsqp(best_h, n_cur, 500, 200)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    elapsed = time.time() - t0
    if budget_s - elapsed > 45:
        n_cur = len(best_h)
        h_pol, c5_pol = lse_slsqp(best_h, n_cur, 800, 400)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        h_sg, c5_sg = lse_grad_descent(best_h, n_cur, 500, 3000, 0.001, 0.9)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
