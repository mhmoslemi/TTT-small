def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Add to positions with most room
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:

            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add

            if diff >= -1e-12:

            sub = min(-diff, h[i])

            diff += sub
    return h

f(h) = max_k g_k(h) where g_k(h) = corr[k] * dx

The subgradient of f at h is:
∂f/∂h_i = ∑_{k ∈ K*} (∂g_k/∂h_i) * λ_k

The gradient of g_k with respect to h_i:
g_k = ∑_j h[j](1-h[j+k]) (for the appropriate range)

∂g_k/∂h_i = (1-h[i+k]) - h[i-k] (with appropriate boundary handling)

So the update direction is:
h_i ← h_i - α · [(1-h[i+k*]) - h[i-k*]] (for the active lag k*)

This is a **subgradient step** toward reducing the active constraint.

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active lags
    active = np.where(np.abs(corr - max_val) < 1e-6)[0]
    
    # Compute subgradient (average over active lags)
    grad = np.zeros(n)
    for lag_idx in active:
        lag = lag_idx - (n - 1)
        for i in range(n):
            # d(corr[lag_idx])/d(h[i])
            # corr[lag_idx] = sum_j h[j](1-h[j+lag]) for valid j
            # If i is in position j: contributes (1-h[i+lag])
            # If i is in position j+lag: contributes -h[i-lag]
            if 0 <= i + lag < n:
                grad[i] += (1 - h[i + lag])
            if 0 <= i - lag < n:
                grad[i] -= h[i - lag]
    grad /= len(active)
    
    # Update
    h_new = h - alpha * grad
    h_new = _fix_sum(h_new, target_sum)
    return h_new

import time
import numpy as np
from scipy.optimize import minimize, linprog

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

             'jac': lambda x: np.ones(n)}]

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x

        return h
