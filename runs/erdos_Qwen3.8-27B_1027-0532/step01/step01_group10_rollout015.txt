def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            # Add diff/cnt to each masked element, but cap at 1.0
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(h, dx, beta):

    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

When m = m1: i-m = i-m1 (largest index in h)

Reversed: h[i-m2 : i-m1+1][::-1] gives [h[i-m1], ..., h[i-m2]]

So B[m1:m2+1] += 1 - h[i-m2:i-m1+1][::-1]

For the second term, m ranges from m3 to m4, and B[m] gets -h[m+i].
When m = m3: m+i = m3+i
When m = m4: m+i = m4+i
So B[m3:m4+1] -= h[m3+i : m4+i+1]

This is efficient! Each is an O(1) slice operation (view) + O(n) addition.

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def constraint_sum_fn(h, n):
    return float(np.sum(h) - n * 0.5)

def run_slsqp(h0, n, dx, beta, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum_fn, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return repair_h(res.x, n)
    except:
        pass
    return h0

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def coord_descent(h, n, n_sweeps=1000, seed=42, mu=0.0):
    """Coordinate descent for C5 minimization."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    target = n * 0.5
    
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_sum = float(np.sum(h))
        
        order = rng.permutation(n)
        changed = False
        
        for i in order:
            old = h[i]
            
            # Compute B vector
            # First term: B[m] += (1-h[i-m]) for valid m where 0 <= i-m < n
            # m range: max(0, i-(n-1)) to min(2n-2, i)
            m1 = max(0, i - (n-1))
            m2 = min(2*n-2, i)
            
            # Second term: B[m] -= h[m+i] for valid m where 0 <= m+i < n
            # m range: max(0, -i) to min(2n-2, n-1-i)
            # Since i >= 0: max(0,-i) = 0
            m3 = 0
            m4 = min(2*n-2, n-1-i)
            
            # B is non-zero only in [m1, m2] ∪ [m3, m4]
            # For the 1D optimization, we need B at all m where it's non-zero
            # and the corresponding corr[m]
            
            # For efficiency, work with the combined range
            lo = min(m1, m3) if m1 <= m2 and m3 <= m4 else max(m1, m3)
            hi = max(m2, m4) if m1 <= m2 and m3 <= m4 else min(m2, m4)
            
            # Actually, both ranges are always valid (m1<=m2 and m3<=m4 for valid i)
            lo = min(m1, m3)
            hi = max(m2, m4)
            
            B_seg = np.zeros(hi - lo + 1)
            
            # First term contributes to [m1, m2] within [lo, hi]
            idx_m1 = m1 - lo
            idx_m2 = m2 - lo
            # h[i-m] for m in [m1, m2]: h indices from i-m1 down to i-m2
            # h[i-m2 : i-m1+1][::-1]
            B_seg[idx_m1:idx_m2+1] += 1.0 - h[i-m2:i-m1+1][::-1]
            
            # Second term contributes to [m3, m4] within [lo, hi]
            idx_m3 = m3 - lo
            idx_m4 = m4 - lo
            B_seg[idx_m3:idx_m4+1] -= h[m3+i:m4+i+1]
            
            corr_seg = corr[lo:hi+1]
            
            # 1D optimization: min_t max(corr_seg + t * B_seg) [+ mu * (cur_sum + t - target)^2]
            t_lo = -old
            t_hi = 1.0 - old
            
            if mu > 0:
                for _ in range(40):
                    t_mid = 0.5 * (t_lo + t_hi)
                    vals = corr_seg + t_mid * B_seg
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-12
                    pg = 2.0 * mu * (cur_sum + t_mid - target)
                    if np.min(B_seg[active]) + pg > 1e-12:
                        t_hi = t_mid
                    elif np.max(B_seg[active]) + pg < -1e-12:
                        t_lo = t_mid
                    else:
                        break
            else:
                for _ in range(30):
                    t_mid = 0.5 * (t_lo + t_hi)
                    vals = corr_seg + t_mid * B_seg
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-12
                    if np.min(B_seg[active]) > 1e-12:
                        t_hi = t_mid
                    elif np.max(B_seg[active]) < -1e-12:
                        t_lo = t_mid
                    else:
                        break
            
            t_opt = 0.5 * (t_lo + t_hi)
            new_val = old + t_opt
            
            if abs(new_val - old) > 1e-14:
                h[i] = new_val
                corr[lo:hi+1] += t_opt * B_seg
                cur_sum += t_opt
                changed = True
        
        if not changed:
            break
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            sub = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def constraint_sum_fn(h, n):
    return float(np.sum(h) - n * 0.5)

def run_slsqp(h0, n, dx, beta, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum_fn, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h0, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return repair_h(res.x, n)
    except Exception:
        pass
    return h0

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def coord_descent(h, n, n_sweeps=1000, seed=42, mu=1.0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n * 0.5
    n2 = 2 * n - 1

    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_sum = float(np.sum(h))
        order = rng.permutation(n)
        changed = False

        for i in order:
            old = h[i]
            m1 = max(0, i - (n - 1))
            m2 = min(n2 - 1, i)
            m3 = 0
            m4 = min(n2 - 1, n - 1 - i)

            if m1 > m2 and m3 > m4:
                continue

            lo = min(m1, m3)
            hi = max(m2, m4)
            seg_len = hi - lo + 1
            B_seg = np.zeros(seg_len)

            if m1 <= m2:
                idx_a = m1 - lo
                idx_b = m2 - lo
                B_seg[idx_a:idx_b + 1] += 1.0 - h[i - m2:i - m1 + 1][::-1]

            if m3 <= m4:
                idx_c = m3 - lo
                idx_d = m4 - lo
                B_seg[idx_c:idx_d + 1] -= h[m3 + i:m4 + i + 1]

            corr_seg = corr[lo:hi + 1]

            if mu > 0.0:
                t_lo = -old
                t_hi = 1.0 - old
                for _ in range(50):
                    t_mid = 0.5 * (t_lo + t_hi)
                    vals = corr_seg + t_mid * B_seg
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-11
                    pg = 2.0 * mu * (cur_sum + t_mid - target)
                    if np.min(B_seg[active]) + pg > 1e-11:
                        t_hi = t_mid
                    elif np.max(B_seg[active]) + pg < -1e-11:
                        t_lo = t_mid
                    else:
                        break
            else:
                t_lo = -old
                t_hi = 1.0 - old
                for _ in range(35):
                    t_mid = 0.5 * (t_lo + t_hi)
                    vals = corr_seg + t_mid * B_seg
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-11
                    if np.min(B_seg[active]) > 1e-11:
                        t_hi = t_mid
                    elif np.max(B_seg[active]) < -1e-11:
                        t_lo = t_mid
                    else:
                        break

            t_opt = 0.5 * (t_lo + t_hi)
            new_val = old + t_opt

            if abs(new_val - old) > 1e-14:
                h[i] = new_val
                corr[lo:hi + 1] += t_opt * B_seg
                cur_sum += t_opt
                changed = True

        if not changed:
            break

    return h

def perturb_h(h, n, rng, scale=0.05):
    h_pert = h.copy()
    noise = rng.uniform(-scale, scale, n)
    h_pert = h_pert + noise
    return repair_h(h_pert, n)
