def subgradient_polish(h, n, n_iter=200):

    h = h.copy().astype(float)
    best_h = h.copy()

    step = 0.05
    momentum = 0.0
    velocity = np.zeros(n)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            m = n - lag
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
        else:
            l = -lag
            m = n - l
            grad[:m] += (1.0 - h[l:l+m])
            grad[l:l+m] += -h[:m]
        grad *= dx
        
        # Nesterov momentum
        scale = step / (max(np.linalg.norm(grad), 1e-12))
        velocity = momentum * velocity - scale * grad
        h = h + velocity
        h = project_onto_constraint(h, n)
        
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        step *= 0.998
        momentum = 0.9
    
    return best_h, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smoothed_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[lag:] += w * (1.0 - h[:m])
            grad[:m] += -w * h[lag:]
        else:
            l = -lag
            m = n - l
            grad[:m] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:m]
    grad *= dx
    return obj, grad
