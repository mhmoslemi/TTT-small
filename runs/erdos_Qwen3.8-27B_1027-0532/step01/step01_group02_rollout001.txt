def fast_full_correlate(a, b):
    n = len(a)
    N = 2 * n  # power of 2 for speed
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    # Circular cross-correlation: ifft(A * conj(B)) gives sum_i a[i] * b[i-k] (mod N)
    # We want: sum_i a[i] * b[i+k] for k = -(n-1), ..., n-1
    # ifft(conj(A) * B) gives sum_i conj(a[i-k]) * b[i] = sum_i a[i-k] * b[i] (for real)
    # Which is sum_j a[j] * b[j+k] where j = i-k
    # So ifft(conj(A) * B) gives the correlation we want, in circular form
    circ = np.fft.irfft(np.conj(A) * B, N)
    # The circular correlation has the linear correlation in positions:
    # k = 0, 1, ..., n-1 at positions 0, 1, ..., n-1
    # k = -(n-1), ..., -1 at positions N-(n-1), ..., N-1
    # np.correlate(a, b, 'full') gives k = -(n-1), ..., 0, ..., n-1
    # So we need to rearrange
    result = np.concatenate([circ[N-(n-1):], circ[:n]])
    return result

            h_bin[ones[order[i]]] = 0.0

        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - current_sum):
            h_bin[zeros[order[i]]] = 1.0
    return h_bin

def discrete_local_search(h, n, time_limit=30):
    h = h.copy()
    t_start = time.time()
    c5 = evaluate_c5(h)
    improved = True
    while improved and (time.time() - t_start) < time_limit:
        improved = False
        for i in range(n):
            if (time.time() - t_start) > time_limit:
                break
            old = h[i]
            h[i] = 1.0 - old
            h_fixed = repair_h(h, n)
            c5_new = evaluate_c5(h_fixed)
            if c5_new < c5 - 1e-12:
                h = h_fixed
                c5 = c5_new
                improved = True
            else:
                h[i] = old
                h = h_fixed
    return h, c5

            break

        h_opt = repair_h(h_opt, n_best)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

        if c5_ds < best_c5:
