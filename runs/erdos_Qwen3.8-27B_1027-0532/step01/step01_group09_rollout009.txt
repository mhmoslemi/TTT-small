import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, 'full') * dx).max()

            if np.any(valid1):
                a_vals[valid1] += dx * (1.0 - h[i + m[valid1]])
            if np.any(valid2):

                    lo_t = mid_t
                elif left_sg > 1e-10:
                    hi_t = mid_t
                else:

            delta = new_h_i - h[i]

def selpolish_smoothed(h0, alpha=100.0, tm=60):

        return x.sum() - target
    
    def gjac(x):
        return np.ones(n)

        c1 = eval_c5(h1)
        c0 = eval_c5(h0)
        if c1 < c0:

    except Exception:
        pass
    return h0

def subgrad_opt(h0, tm=30, lr_init=0.3, beta=0.8):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)
    for it in range(80000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        a = j - k
        b = j + k
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += 1.0 - h[a[m1]]
        grad[m2] -= h[b[m2]]
        grad *= dx
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9998
        if lr < 1e-7:
            break
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)

    xs = np.arange(n_src, dtype=float)

    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def ilds_search(h0, tm=60, seed=42):

        # Local optimize (randomly choose method)
        r = rng.random()

        c5_opt = eval_c5(h_opt)
        if c5_opt < best_c5:
            best = h_opt.copy()
            best_c5 = c5_opt
            num_restarts = 0
        else:
            num_restarts += 1
            # Occasionally accept worse solutions (simulated annealing)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, 'full') * dx).max()

def coord_desc(h0, max_sweeps=5000, tm=60, seed=0):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    lags = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    lam = 0.0
    mu = 1.0
    t0 = time.time()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        order = rng.permutation(n)
        for i in order:
            m = lags
            a_vals = np.zeros(len(m))
            v1 = (i + m >= 0) & (i + m < n)
            v2 = (i - m >= 0) & (i - m < n)
            if np.any(v1):
                a_vals[v1] += dx * (1.0 - h[i + m[v1]])
            if np.any(v2):
                a_vals[v2] -= dx * h[i - m[v2]]
            S = h.sum() - h[i] - target
            lo_t = 0.0
            hi_t = 1.0
            for _ in range(50):
                mid_t = 0.5 * (lo_t + hi_t)
                vals = a_vals * mid_t + corr - a_vals * h[i]
                max_val = vals.max()
                active = (vals >= max_val - 1e-9)
                left_sg = a_vals[active].min() + lam + mu * (S + mid_t)
                right_sg = a_vals[active].max() + lam + mu * (S + mid_t)
                if right_sg < -1e-10:
                    lo_t = mid_t
                elif left_sg > 1e-10:
                    hi_t = mid_t
                else:
                    break
            new_h_i = 0.5 * (lo_t + hi_t)
            delta = new_h_i - h[i]
            if abs(delta) > 1e-12:
                h[i] = new_h_i
                corr += a_vals * delta
        h = project_feasible(h, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
            stall = 0
        else:
            stall += 1
        violation = h.sum() - target
        lam += mu * violation
        if abs(violation) > target * 0.01:
            mu = min(mu * 1.3, 1e4)
        if abs(violation) < target * 0.001:
            lam *= 0.7
            mu = max(mu * 0.9, 1.0)

            h = project_feasible(h, target)
            lam = 0.0
            mu = 1.0
            stall = 0
    return best, best_c5

def selpolish_smoothed(h0, alpha=100.0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    t0 = time.time()

    def fobj(x):
        if time.time() - t0 > tm * 0.85:
            raise Exception("timeout")
        corr = np.correlate(x, 1.0 - x, 'full') * dx
        mc = corr.max()
        return mc + np.log(np.sum(np.exp(alpha * (corr - mc)))) / alpha

    def gfun(x):
        return x.sum() - target

    def gjac(x):
        return np.ones(n)

    try:
        r = minimize(fobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 3000, 'ftol': 1e-16})
        h1 = project_feasible(r.x, target)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0
