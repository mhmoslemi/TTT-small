    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def obj_power(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr = np.maximum(corr, 1e-30)
    return float(np.sum(corr ** p) ** (1.0 / p))


def constraint_fun(h):
    return np.sum(h) - _g_ts


def constraint_jac(h):
    return np.ones(_g_n)


def make_golden_h(n, phase=0.0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = (np.arange(n) + 0.5 + phase) * 2.0 / n
    frac = (x * phi) % 1.0
    h = np.where(frac < 0.5, 1.0, 0.0)
    return project_feasible(h, n / 2.0)


def make_block_h(n, block_len):
    h = np.zeros(n)
    for i in range(0, n, 2 * block_len):
        end = min(i + block_len, n)
        h[i:end] = 1.0
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_ts, _g_n
    start_time = time.time()
    rng = np.random.RandomState(seed)

    elapsed = time.time() - start_time
    remaining = budget_s - 25.0 - elapsed

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    ts0 = n0 / 2.0
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    _g_n = n0
    _g_ts = ts0
    bounds0 = [(0.0, 1.0)] * n0
    cons0 = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    for alpha in [30, 80, 200, 500, 1000, 2000, 4000]:
        if budget_s - 50.0 - (time.time() - start_time) < 0:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(alpha,),
                           method='SLSQP', bounds=bounds0, constraints=cons0,
                           options={'maxiter': 10000, 'ftol': 1e-17})
            h_new = project_feasible(res.x, ts0)
            s = compute_c5(h_new)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    for p in [50, 100, 200, 400, 800]:
        if budget_s - 40.0 - (time.time() - start_time) < 0:
            break
        try:
            res = minimize(obj_power, best_h, args=(p,),
                           method='SLSQP', bounds=bounds0, constraints=cons0,
                           options={'maxiter': 6000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts0)
            s = compute_c5(h_new)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    n_even = n0 + 1 if n0 % 2 != 0 else n0
    ts_e = n_even / 2.0
    _g_n = n_even
    _g_ts = ts_e
    bounds_e = [(0.0, 1.0)] * n_even
    cons_e = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
    x_new = (np.arange(n_even) + 0.5) * (2.0 / n_even)
    h_even = np.interp(x_new, x_old, best_h, period=2.0)
    h_even = project_feasible(h_even, ts_e)

    h_bin = (h_even > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = int(ts_e)
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1.0)[0]
        sort_order = np.argsort(h_even[ones_idx])
        h_bin[ones_idx[sort_order[:n_ones - target_ones]]] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0.0)[0]
        sort_order = np.argsort(-h_even[zeros_idx])
        h_bin[zeros_idx[sort_order[:target_ones - n_ones]]] = 1.0

    s_bin = compute_c5(h_bin)
    if s_bin < best_score:
        best_score = s_bin
        best_h = h_bin.copy()

    for alpha in [100, 300, 800, 2000]:
        if budget_s - 30.0 - (time.time() - start_time) < 0:
            break
        try:
            res = minimize(obj_smooth, h_bin, args=(alpha,),
                           method='SLSQP', bounds=bounds_e, constraints=cons_e,
                           options={'maxiter': 6000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, ts_e)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    for p in [100, 200, 500]:
        if budget_s - 30.0 - (time.time() - start_time) < 0:
            break
        try:
            res = minimize(obj_power, best_h, args=(p,),
                           method='SLSQP', bounds=bounds_e, constraints=cons_e,
                           options={'maxiter': 4000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts_e)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    if budget_s - 150.0 - (time.time() - start_time) > 0:
        n_sa = n_even
        h_sa = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_sa))
        target_ones = int(ts_e)
        if n_ones != target_ones:
            if n_ones > target_ones:
                oi = np.where(h_sa == 1.0)[0]
                h_sa[oi[:n_ones - target_ones]] = 0.0
            else:
                zi = np.where(h_sa == 0.0)[0]
                h_sa[zi[:target_ones - n_ones]] = 1.0

        best_bin = h_sa.copy()
        best_bin_score = compute_c5(h_sa)
        T = 0.005
        max_sa_iters = 100000
        sa_iter = 0
        stuck = 0

        while sa_iter < max_sa_iters:
            if budget_s - 80.0 - (time.time() - start_time) < 0:
                break
            ones = np.where(h_sa == 1.0)[0]
            zeros = np.where(h_sa == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_sa[i] = 0.0
            h_sa[j] = 1.0
            new_score = compute_c5(h_sa)
            delta = new_score - best_bin_score
            if delta < 0:
                best_bin_score = new_score
                best_bin = h_sa.copy()
                stuck = 0
            else:
                if rng.rand() < np.exp(-delta / T):
                    stuck = 0
                else:
                    h_sa[i] = 1.0
                    h_sa[j] = 0.0
                    stuck += 1
            sa_iter += 1
            if stuck > 5000:
                T *= 0.9
                stuck = 0
            if sa_iter % 10000 == 0:
                T = max(T * 0.95, 0.0001)

        if best_bin_score < best_score:
            best_score = best_bin_score
            best_h = best_bin.copy()

    if budget_s - 100.0 - (time.time() - start_time) > 0:
        n_alt = n_even
        ts_alt = n_alt / 2.0
        _g_n = n_alt
        _g_ts = ts_alt
        bounds_alt = [(0.0, 1.0)] * n_alt
        cons_alt = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for phase in [0.0, 0.1, 0.25, 0.3, 0.5, 0.7, 0.13]:
            if budget_s - 60.0 - (time.time() - start_time) < 0:
                break
            h_gold = make_golden_h(n_alt, phase)
            for alpha in [200, 500]:
                if budget_s - 40.0 - (time.time() - start_time) < 0:
                    break
                try:
                    res = minimize(obj_smooth, h_gold, args=(alpha,),
                                   method='SLSQP', bounds=bounds_alt, constraints=cons_alt,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_alt)
                    s = compute_c5(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new
                except Exception:
                    pass

        for bl in [2, 3, 4, 5, 7]:
            if budget_s - 50.0 - (time.time() - start_time) < 0:
                break
            h_blk = make_block_h(n_alt, bl)
            try:
                res = minimize(obj_smooth, h_blk, args=(300,),
                               method='SLSQP', bounds=bounds_alt, constraints=cons_alt,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts_alt)
                s = compute_c5(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new
            except Exception:
                pass

    n_use = len(best_h)
    ts_use = n_use / 2.0
    _g_n = n_use
    _g_ts = ts_use
    bounds_u = [(0.0, 1.0)] * n_use
    cons_u = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    alpha_pool = [200, 500, 1000, 2000, 4000]
    p_pool = [100, 200, 500, 1000]

    while budget_s - 50.0 - (time.time() - start_time) > 0:
        strategy = rng.randint(5)
        if strategy == 0:
            noise = 0.05 + 0.1 * rng.rand()
            h_p = best_h + rng.randn(n_use) * noise
        elif strategy == 1:
            shift = rng.randint(1, max(2, n_use // 10))
            h_p = np.roll(best_h, shift)
        elif strategy == 2:
            h_p = best_h.copy()
            nf = rng.randint(3, max(4, n_use // 6))
            idx = rng.choice(n_use, size=nf, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        elif strategy == 3:
            t_mix = 0.15 + 0.25 * rng.rand()
            h_p = (1.0 - t_mix) * best_h + t_mix * h0[:n_use] if n_use == n0 else best_h + rng.randn(n_use) * 0.1
        else:
            h_p = (best_h > 0.5).astype(float)
            nf = rng.randint(1, max(2, n_use // 15))
            flip_idx = rng.choice(n_use, size=nf, replace=False)
            h_p[flip_idx] = 1.0 - h_p[flip_idx]
            h_p = project_feasible(h_p, ts_use)
            h_p += rng.randn(n_use) * 0.03

        h_p = project_feasible(h_p, ts_use)

        if rng.rand() < 0.55:
            alpha = alpha_pool[rng.randint(len(alpha_pool))]
            try:
                res = minimize(obj_smooth, h_p, args=(alpha,),
                               method='SLSQP', bounds=bounds_u, constraints=cons_u,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts_use)
            except Exception:
                continue
        else:
            p_val = p_pool[rng.randint(len(p_pool))]
            try:
                res = minimize(obj_power, h_p, args=(p_val,),
                               method='SLSQP', bounds=bounds_u, constraints=cons_u,
                               options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts_use)
            except Exception:
                continue

        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new

    for p in [300, 600, 1000]:
        if budget_s - 25.0 - (time.time() - start_time) < 5:
            break
        try:
            res = minimize(obj_power, best_h, args=(p,),
                           method='SLSQP', bounds=bounds_u, constraints=cons_u,
                           options={'maxiter': 8000, 'ftol': 1e-17})
            h_new = project_feasible(res.x, ts_use)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    for alpha in [2000, 5000]:
        if budget_s - 20.0 - (time.time() - start_time) < 5:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(alpha,),
                           method='SLSQP', bounds=bounds_u, constraints=cons_u,
                           options={'maxiter': 8000, 'ftol': 1e-17})
            h_new = project_feasible(res.x, ts_use)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    h_final = project_feasible(best_h, ts_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
