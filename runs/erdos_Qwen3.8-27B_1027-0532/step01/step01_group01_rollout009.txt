    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def make_binary_like(h, n):
    target = n / 2.0
    sorted_idx = np.argsort(h)
    h_bl = np.zeros(n)
    num_zeros = int(np.floor(target))
    for i in range(n):
        if i < num_zeros:
            h_bl[sorted_idx[i]] = 0.0
        elif i == num_zeros:
            h_bl[sorted_idx[i]] = 0.5
        else:
            h_bl[sorted_idx[i]] = 1.0
    return h_bl


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(np.binary_repr(i).count('1'))
    parity = np.mod(h, 2.0)
    num_ones = int(np.sum(parity == 1))
    h_bt = np.zeros(n)
    if num_ones == n // 2 + 1:
        ones_idx = np.where(parity == 1)[0]
        zeros_idx = np.where(parity == 0)[0]
        h_bt[ones_idx] = 1.0
        h_bt[zeros_idx] = 0.0
        h_bt[ones_idx[-1]] = 0.5
    elif num_ones == n // 2:
        ones_idx = np.where(parity == 1)[0]
        zeros_idx = np.where(parity == 0)[0]
        h_bt[ones_idx] = 1.0
        h_bt[zeros_idx] = 0.0
        h_bt[zeros_idx[-1]] = 0.5
    else:
        h_bt = np.zeros(n)
        positions = np.random.choice(n, n // 2, replace=False)
        h_bt[positions] = 1.0
        remaining = np.setdiff1d(np.arange(n), positions)
        h_bt[remaining[0]] = 0.5
    return h_bt


def combinatorial_local_search(h_start, n, max_iters, time_limit, rng):
    h = h_start.copy()
    start_t = time.time()
    best_c5 = evaluate_c5(h)
    no_improve_count = 0

    for iteration in range(max_iters):
        if time.time() - start_t > time_limit:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        halves = np.where(h == 0.5)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        best_move = None
        best_move_c5 = best_c5

        num_swap_tries = min(80, len(ones) * len(zeros))
        for _ in range(num_swap_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5 = evaluate_c5(h)
            if c5 < best_move_c5 - 1e-14:
                best_move_c5 = c5
                best_move = (0, i, j, -1)
            h[i], h[j] = 1.0, 0.0

        if len(halves) > 0:
            p = halves[0]
            num_bc_tries = min(40, len(ones) * len(zeros))
            for _ in range(num_bc_tries):
                q = ones[rng.randint(len(ones))]
                r = zeros[rng.randint(len(zeros))]
                h[p] = 0.0
                h[q] = 0.5
                h[r] = 1.0
                c5 = evaluate_c5(h)
                if c5 < best_move_c5 - 1e-14:
                    best_move_c5 = c5
                    best_move = (1, p, q, r)
                h[p] = 0.5
                h[q] = 1.0
                h[r] = 0.0

            for _ in range(num_bc_tries):
                q = zeros[rng.randint(len(zeros))]
                r = ones[rng.randint(len(ones))]
                h[p] = 1.0
                h[q] = 0.5
                h[r] = 0.0
                c5 = evaluate_c5(h)
                if c5 < best_move_c5 - 1e-14:
                    best_move_c5 = c5
                    best_move = (2, p, q, r)
                h[p] = 0.5
                h[q] = 0.0
                h[r] = 1.0

        if best_move is not None:
            mtype, a, b, c = best_move
            if mtype == 0:
                h[a], h[b] = 0.0, 1.0
            elif mtype == 1:
                h[a] = 0.0
                h[b] = 0.5
                h[c] = 1.0
            elif mtype == 2:
                h[a] = 1.0
                h[b] = 0.5
                h[c] = 0.0
            best_c5 = best_move_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 80:
                if rng.random() < 0.3:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                    new_c5 = evaluate_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                    else:
                        accepted = rng.random() < np.exp(-(new_c5 - best_c5) / 0.001)
                        if not accepted:
                            h[i], h[j] = 1.0, 0.0
                    no_improve_count = 0
                else:
                    break

    return h, best_c5


def pnorm_objective(x, p, active, n):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode="full")
    vals = c[active]
    return (np.sum(vals ** p) / len(active)) ** (1.0 / p) * dx


def pnorm_gradient(x, p, active, n):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode="full")
    vals = c[active]
    base = (np.sum(vals ** p) / len(active)) ** (1.0 / p)
    if base < 1e-15:
        return np.zeros(n)
    g = np.zeros(n)
    n_corr = 2 * n - 1
    for idx in range(len(active)):
        k = active[idx]
        l = k - (n - 1)
        w = (vals[idx] ** (p - 1)) / (len(active) * base ** (p - 1))
        if abs(w) < 1e-15:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-x[jpl])
            g[j] += w * term
    return g * dx


def continuous_refine(h_start, n, start_time, budget_s, rng):
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n

    for p in [2, 4, 8, 16, 32, 64, 128, 256]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 90:
            break

        corr = np.correlate(h, 1 - h, mode="full")
        m = np.max(corr)
        thresh = 0.9 * m
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        try:
            res = minimize(
                pnorm_objective, h.copy(),
                args=(p, active, n),
                jac=pnorm_gradient,
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    h = h_new.copy()
        except Exception:
            pass

    return h, best_c5


def refined_slsqp(h_start, n, start_time, budget_s, eps_list):
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n
    n_corr = 2 * n - 1

    for eps in eps_list:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 60:
            break

        def obj(x):
            corr = np.correlate(x, 1 - x, mode="full")
            m = np.max(corr)
            shifted = eps * (corr - m)
            return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

        def grad(x):
            corr = np.correlate(x, 1 - x, mode="full")
            m = np.max(corr)
            shifted = eps * (corr - m)
            exp_s = np.exp(shifted)
            sum_exp = np.sum(exp_s)
            weights = exp_s / sum_exp
            g = np.zeros(n)
            for k in range(n_corr):
                l = k - (n - 1)
                w = weights[k]
                if w < 1e-13:
                    continue
                for j in range(n):
                    term = 0.0
                    jml = j - l
                    if 0 <= jml < n:
                        term += (1.0 - x[jml])
                    jpl = j + l
                    if 0 <= jpl < n:
                        term += (-x[jpl])
                    g[j] += w * term
            return g * dx

        try:
            res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                          options={'maxiter': 300, 'ftol': 1e-13})
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    h = h_new.copy()
        except Exception:
            pass

    return h, best_c5


def coordinate_polish(h_start, n, start_time, budget_s, rng):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)

    for _ in range(5000):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 10:
            break
        idx = rng.randint(n)
        best_delta = 0.0
        for delta in [-0.1, -0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_trial = h.copy()
            h_trial[idx] += delta
            h_trial = project_feasible(h_trial, n)
            c5 = evaluate_c5(h_trial)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                h = h_trial.copy()
                best_delta = delta
                break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Binary-like combinatorial search from parent
    h_bl = make_binary_like(h0, n)
    c5_bl = evaluate_c5(h_bl)
    if c5_bl < best_c5:
        best_c5 = c5_bl
        best_h = h_bl.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        t_alloc = min(250, (budget_s - 300) * 0.35)
        h_bl2, c5_bl2 = combinatorial_local_search(h_bl, n, 5000, t_alloc, rng)
        if c5_bl2 < best_c5:
            best_c5 = c5_bl2
            best_h = h_bl2.copy()

    # Phase 2: Try structured initializations
    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        h_tm = thue_morse_init(n)
        h_tm_bl = make_binary_like(h_tm, n)
        t_alloc = min(120, (budget_s - 200) * 0.15)
        h_tm2, c5_tm = combinatorial_local_search(h_tm_bl, n, 3000, t_alloc, rng)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm2.copy()

    # Phase 3: Random binary-like restarts
    elapsed = time.time() - start_time
    if budget_s - elapsed > 150:
        num_restarts = 4
        t_per = (budget_s - 150) * 0.15 / num_restarts
        for _ in range(num_restarts):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 120:
                break
            h_rand = np.zeros(n)
            positions = rng.choice(n, n // 2, replace=False)
            h_rand[positions] = 1.0
            remaining = np.setdiff1d(np.arange(n), positions)
            h_rand[remaining[rng.randint(len(remaining))]] = 0.5
            h_r, c5_r = combinatorial_local_search(h_rand, n, 2000, t_per, rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: Continuous p-norm optimization
    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        h_c, c5_c = pnorm_refine(best_h, n, start_time, budget_s, rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 5: Refined SLSQP with log-sum-exp (different eps schedule)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        eps_list = [300.0, 500.0, 1000.0, 2000.0]
        h_r, c5_r = refined_slsqp(best_h, n, start_time, budget_s, eps_list)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Coordinate polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        h_p, c5_p = coordinate_polish(best_h, n, start_time, budget_s, rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final: try one more combinatorial pass from the continuous solution
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_bl_final = make_binary_like(best_h, n)
        t_alloc = min(60, budget_s - elapsed - 10)
        h_f, c5_f = combinatorial_local_search(h_bl_final, n, 2000, t_alloc, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection
    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
