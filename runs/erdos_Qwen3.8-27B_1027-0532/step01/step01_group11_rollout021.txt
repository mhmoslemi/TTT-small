    lo, hi = -2.0, 2.0

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp

        wk = w[k]
        if wk < 1e-15:

        h0 = project_onto_constraint(0.5 + 0.3 * rng.standard_normal(n0), n0 / 2.0)
    else:
        h0 = project_onto_constraint(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    de_configs = [
        (50, 5000, 0.5, 0.8),
        (50, 4000, 0.8, 0.7),
        (45, 4000, 0.3, 0.9),
        (60, 3500, 0.6, 0.85),
        (45, 4000, 0.7, 0.55),
        (40, 3500, 0.4, 0.75),
        (55, 3000, 0.9, 0.65),
    ]

    for F, CR, pop_size, max_gen in de_configs:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 50:
            break
        deadline = start_time + min(remaining - 15, 250)
        h_de, f_de = run_de(best_h, n0, pop_size, max_gen, F, CR, rng, deadline)
        if f_de < best_c5:
            best_c5 = f_de
            best_h = h_de

    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        eps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        deadline = start_time + min(remaining - 15, 350)
        h_pol, c5_pol = polish(best_h, eps_list, 250, deadline)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    remaining = budget_s - (time.time() - start_time)
    if remaining > 70:
        n_cur = len(best_h)
        h_comp = 1.0 - best_h[::-1].copy()
        h_comp = project_onto_constraint(h_comp, n_cur / 2.0)
        h_avg = 0.5 * best_h + 0.5 * h_comp
        h_avg = project_onto_constraint(h_avg, n_cur / 2.0)
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg
        eps_list2 = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        deadline2 = start_time + min(remaining - 15, 180)
        h_pol2, c5_pol2 = polish(h_avg, eps_list2, 200, deadline2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

    for n_try in [60, 120, 80]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 120:
            break
        n_cur = len(best_h)
        if n_try == n_cur:
            continue
        x_orig = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_onto_constraint(h_interp, n_try / 2.0)

        deadline = start_time + min(remaining - 20, 250)
        h_de2, f_de2 = run_de(h_interp, n_try, 45, 3000, 0.5, 0.8, rng, deadline)
        if f_de2 < best_c5:
            best_c5 = f_de2
            best_h = h_de2

        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            eps_list3 = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
            deadline3 = start_time + min(remaining - 10, 200)
            h_pol3, c5_pol3 = polish(best_h, eps_list3, 200, deadline3)
            if c5_pol3 < best_c5:
                best_c5 = c5_pol3
                best_h = h_pol3

    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_final = len(best_h)
        h_nudge = best_h + 0.003 * rng.standard_normal(n_final)
        h_nudge = project_onto_constraint(h_nudge, n_final / 2.0)
        eps_fin = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        deadline_fin = start_time + remaining - 5
        h_fin, c5_fin = polish(h_nudge, eps_fin, 200, deadline_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h_nudge2 = best_h + 0.001 * rng.standard_normal(len(best_h))
        h_nudge2 = project_onto_constraint(h_nudge2, len(best_h) / 2.0)
        eps_fin2 = [0.0001, 0.00005, 0.00002, 0.00001]
        deadline_fin2 = start_time + remaining - 3
        h_fin2, c5_fin2 = polish(h_nudge2, eps_fin2, 150, deadline_fin2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
