def compute_gradient(h, w, n):
    # w is the softmax weights, length 2n-1
    # w[k] corresponds to lag k - (n-1)
    # Returns gradient of smooth max w.r.t. h
    g = np.zeros(n)
    for i in range(n):

        s2 = 0.0

            if 0 <= i + lag < n:

    return g

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx


def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def make_grad_indices(n):
    j = np.arange(n)
    i = np.arange(n)
    idx1 = (i[:, None] - j[None, :] + n - 1).astype(np.int32)
    idx2 = (j[None, :] - i[:, None] + n - 1).astype(np.int32)
    return idx1, idx2


def smooth_obj_and_grad(h, alpha, n, idx1, idx2):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = c_max + np.log(sum_exp) / alpha
    w = exp_s / sum_exp
    M1 = w[idx1]
    M2 = w[idx2]
    grad = M1 @ (1.0 - h) - M2 @ h
    return obj, grad


def smooth_obj_func(h, alpha, n, idx1, idx2):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 500.0)
    exp_s = np.exp(shifted)
    return c_max + np.log(np.sum(exp_s)) / alpha


def projected_gradient(h, n, alpha, max_iter, t0, budget_s, margin):
    idx1, idx2 = make_grad_indices(n)
    h_curr = h.copy()
    lr = 1.0 / (alpha * 5.0)
    best_local = evaluate_score(h_curr)
    best_h_local = h_curr.copy()

    for it in range(max_iter):
        if (time.time() - t0) > (budget_s - margin):
            break
        obj, grad = smooth_obj_and_grad(h_curr, alpha, n, idx1, idx2)
        if not np.all(np.isfinite(grad)):
            break
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-14:
            break

        step = lr
        h_new = None
        for bt in range(25):
            h_try = project_feasible(h_curr - step * grad, n)
            obj_try, _ = smooth_obj_and_grad(h_try, alpha, n, idx1, idx2)
            if obj_try <= obj - 1e-6 * step * norm_g * norm_g:
                h_new = h_try
                break
            step *= 0.5
        if h_new is None:
            h_new = project_feasible(h_curr - step * grad, n)
        h_curr = h_new

        actual = evaluate_score(h_curr)
        if actual < best_local - 1e-12:
            best_local = actual
            best_h_local = h_curr.copy()

        if it % 20 == 19:
            lr *= 1.15
            if lr > 1.0 / (alpha * 0.5):
                lr = 1.0 / (alpha * 0.5)

    return best_h_local, best_local


def slsqp_refine(h, n, alpha, t0, budget_s, margin):
    idx1, idx2 = make_grad_indices(n)
    bounds = [(0.0, 1.0)] * n

    def obj_fn(hv):
        return smooth_obj_func(hv, alpha, n, idx1, idx2)

    def obj_jac(hv):
        return smooth_obj_and_grad(hv, alpha, n, idx1, idx2)[1]

    def eq_fun(hv):
        return np.sum(hv) - n / 2.0

    def eq_jac_fn(hv):
        return np.ones(n)

    res = minimize(
        obj_fn, h, method='SLSQP', bounds=bounds, jac=obj_jac,
        constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac_fn}],
        options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0}
    )
    if res.x is not None:
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    return h


def binary_local_search(h, n, max_rounds, t0, budget_s, margin):
    h = np.round(np.clip(h, 0.0, 1.0))
    n_ones = int(np.sum(h))
    target = n // 2
    if n % 2 != 0:
        return h, evaluate_score(h)
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng_flip = np.random.RandomState(0)
        rng_flip.shuffle(idx)
        for i in idx[:n_ones - target]:
            h[i] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng_flip = np.random.RandomState(0)
        rng_flip.shuffle(idx)
        for i in idx[:target - n_ones]:
            h[i] = 1

    best = evaluate_score(h)
    for rnd in range(max_rounds):
        if (time.time() - t0) > (budget_s - margin):
            break
        improved = False
        order = np.arange(n)
        np.random.RandomState(rnd).shuffle(order)
        for i in order:
            if (time.time() - t0) > (budget_s - margin):
                break
            h[i] = 1.0 - h[i]
            s = np.sum(h)
            if s != target:
                h[i] = 1.0 - h[i]
                continue
            score = evaluate_score(h)
            if score < best - 1e-12:
                best = score
                improved = True
            else:
                h[i] = 1.0 - h[i]
        if not improved:
            break
    return h, best
