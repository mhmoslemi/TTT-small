import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlations(h)
    c_max = np.max(c)
    logits = alpha * (c - c_max)
    logits -= np.max(logits)
    exp_logits = np.exp(logits)
    w = exp_logits / np.sum(exp_logits)
    grad = np.zeros(n)
    for m in range(-(n - 1), n):
        wm = dx * w[m + n - 1]
        if wm < 1e-14:
            continue
        i_start = max(0, m)
        i_end = min(n, n + m)
        if i_end > i_start:
            j = np.arange(i_start, i_end) - m
            grad[i_start:i_end] += wm * (1.0 - h[j])
        i_start2 = max(0, -m)
        i_end2 = min(n, n - m)
        if i_end2 > i_start2:
            j2 = np.arange(i_start2, i_end2) + m
            grad[i_start2:i_end2] -= wm * h[j2]
    return grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def upsample_h(h, factor=2):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n):
        h_new[i * factor:(i + 1) * factor] = h[i]
    return project_feasible(h_new, n_new / 2.0)

def adam_descent(h, n, alpha_start, alpha_end, n_iter, lr, deadline,
                 beta1=0.85, beta2=0.999, eps=1e-8):
    target = n / 2.0
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        progress = it / max(n_iter, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        lr_eff = lr * (1.0 - 0.3 * progress)
        grad = compute_smooth_gradient(h, alpha)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        t += 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        step = lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def make_binary_init(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 7)
    block_size = max(1, n // (2 * n_blocks))
    for b in range(n_blocks):
        start = int(n * b / n_blocks) + rng.randint(-block_size // 2, block_size // 2 + 1)
        start = max(0, min(n - block_size, start))
        h[start:start + block_size] = 1.0
    return project_feasible(h, n / 2.0)

def make_sinusoidal_init(n, rng):
    t = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for _ in range(rng.randint(1, 4)):
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        h += amp * np.sin(2 * np.pi * freq * t / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

    h = h_base + rng.normal(0, sigma, n)
    return project_feasible(h, n / 2.0)
