    shifted = np.clip(shifted, -500, 500)

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        # d(corr[k])/d(x[j]) for each j
        # corr[k] = sum_i valid: x[i] * (1-x[i+l])
        # d/dx[j]: (1-x[j+l]) if j+l valid + (-x[j-l]) if j-l valid
        # Vectorized:
        jl = np.arange(n) + l
        valid_jl = (jl >= 0) & (jl < n)
        jml = np.arange(n) - l
        valid_jml = (jml >= 0) & (jml < n)
        contrib = np.zeros(n)
        contrib[valid_jl] += (1.0 - x[jl[valid_jl]])
        contrib[valid_jml] += (-x[jml[valid_jml]])
        grad += w * contrib
    grad *= dx
    return grad

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -1.5
    hi = 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        jl = idx + l
        valid_jl = (jl >= 0) & (jl < n)
        jml = idx - l
        valid_jml = (jml >= 0) & (jml < n)
        contrib = np.zeros(n)
        if np.any(valid_jl):
            contrib[valid_jl] += (1.0 - x[jl[valid_jl]])
        if np.any(valid_jml):
            contrib[valid_jml] += (-x[jml[valid_jml]])
        grad += w * contrib
    grad *= dx
    return grad

            return smooth_max_grad(x, n, eps)

            pass

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_g_n = 85

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        jl = idx + l
        valid_jl = (jl >= 0) & (jl < n)
        jml = idx - l
        valid_jml = (jml >= 0) & (jml < n)
        contrib = np.zeros(n)
        if np.any(valid_jl):
            contrib[valid_jl] += (1.0 - x[jl[valid_jl]])
        if np.any(valid_jml):
            contrib[valid_jml] += (-x[jml[valid_jml]])
        grad += w * contrib
    grad *= dx
    return grad

def equality_constraint(x):
    return np.sum(x) - _g_n / 2.0

def de_objective(x):
    h = project_feasible(x, _g_n)
    return evaluate_c5(h)

def nm_objective(x):
    h = project_feasible(x, _g_n)
    return evaluate_c5(h)

def slsqp_refine(h, n, max_time=30):
    global _g_n
    _g_n = n
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': equality_constraint}]
    for eps in [500.0, 200.0, 80.0, 30.0, 10.0, 5.0, 2.0, 1.0]:
        if time.time() - start > max_time:
            break
        try:
            res = minimize(
                smooth_max_obj, best_h.copy(),
                args=(n, eps),
                jac=smooth_max_grad,
                method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12}
            )
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def nm_refine(h, n, max_time=20):
    global _g_n
    _g_n = n
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    try:
        res = minimize(
            nm_objective, best_h.copy(),
            method='Nelder-Mead',
            options={'maxiter': 800, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
    except Exception:
        pass
    return best_h, best_c5

def generate_diverse_starts(best_h, h0, n, count, rng):
    starts = []
    for i in range(count):
        strategy = i % 8
        if strategy == 0:
            h_trial = best_h + 0.01 * rng.standard_normal(n)
        elif strategy == 1:
            h_trial = best_h + 0.04 * rng.standard_normal(n)
        elif strategy == 2:
            h_trial = best_h + 0.12 * rng.standard_normal(n)
        elif strategy == 3:
            h_trial = h0 + 0.08 * rng.standard_normal(n)
        elif strategy == 4:
            h_trial = rng.random(n)
        elif strategy == 5:
            h_trial = best_h.copy()
            bsz = rng.integers(5, max(6, n // 3))
            s = rng.integers(0, n - bsz)
            h_trial[s:s + bsz] += rng.choice([-0.25, -0.15, 0.15, 0.25])
        elif strategy == 6:
            h_trial = best_h.copy()
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            h_trial[i1], h_trial[i2] = h_trial[i2], h_trial[i1]
        else:
            h_trial = np.zeros(n)
            block = n // 3
            c = rng.integers(0, n - block)
            h_trial[c:c + block] = 3.0
            h_trial += 0.05 * rng.standard_normal(n)
        starts.append(project_feasible(h_trial, n))
    return starts

        except Exception:

    # Phase 2: Second DE run biased around current best
    elapsed = time.time() - start_time

        try:

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:

    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:

    # Phase 4: Multi-start with diverse perturbations
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 100
    per_start = max(5, int(remaining / 30))
    n_starts = min(30, max(5, int(remaining / per_start)))

    starts = generate_diverse_starts(best_h, h0, n, n_starts, rng)
    for i in range(n_starts):
        if time.time() - start_time > budget_s - 100:
            break
        h_trial = starts[i]
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()

        if c5_ref < best_c5:

            best_h = h_ref.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 35:

    if budget_s - elapsed > 15:

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
