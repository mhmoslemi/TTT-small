    exp_shifted = np.exp(shifted)
    return (m + (1.0 / eps) * np.log(np.sum(exp_shifted))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - (n - 1)
        # Contribution from (1-h[j-l]) term
        j_start = max(0, l)
        j_end = min(n - 1, n - 1 + l)
        if j_start <= j_end:
            grad[j_start:j_end+1] += w * (1.0 - h[j_start - l:j_end + 1 - l])

        j_start2 = max(0, -l)

        if j_start2 <= j_end2:
            grad[j_start2:j_end2+1] += w * (-h[j_start2 + l:j_end2 + 1 + l])
