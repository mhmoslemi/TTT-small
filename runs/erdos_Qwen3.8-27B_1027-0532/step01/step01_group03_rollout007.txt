def _pair_move(h, i, j, corr):
    """Find optimal δ for pair (i,j): increase h[i] by δ, decrease h[j] by δ."""
    n = len(h)
    num_shifts = 2 * n - 1
    
    # Compute d_k for all shifts
    # corr[k] corresponds to shift s = k - (n-1)
    # d_k = h[j+s] + h[j-s] - h[i+s] - h[i-s] (with boundary handling)
    
    d = np.zeros(num_shifts)
    for k in range(num_shifts):
        s = k - (n - 1)
        dk = 0.0
        # h[j+s] term
        if 0 <= j + s < n:
            dk += h[j + s]
        # h[j-s] term
        if 0 <= j - s < n:
            dk += h[j - s]
        # -h[i+s] term
        if 0 <= i + s < n:
            dk -= h[i + s]
        # -h[i-s] term
        if 0 <= i - s < n:
            dk -= h[i - s]
        d[k] = dk
    
    # δ range
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_hi - delta_lo < 1e-15:
        return 0.0
    
    # Minimize max_k (corr[k] + δ * d[k]) over δ ∈ [delta_lo, delta_hi]
    # Use ternary search (function is convex)
    
    def obj(delta):
        vals = corr + delta * d
        return np.max(vals)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = delta_lo, delta_hi
    c = b - gr * (b - a)
    dpt = a + gr * (b - a)
    fc = obj(c)
    fd = obj(dpt)
    
    for _ in range(60):
        if fc < fd:
            b = dpt
            dpt = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = dpt
            fc = fd
            dpt = a + gr * (b - a)
            fd = obj(dpt)
    
    return (a + b) / 2.0

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)

        return h

                break

            h[i] -= sub
            diff += sub

    n = len(h)
    corr = _corr_of(h)
    # Find all shifts that are (nearly) at the max
    max_val = np.max(corr)
    # Use all shifts within epsilon of max for a better gradient
    active = np.where(corr >= max_val - 0.01)[0]
    
    # Compute average gradient over active shifts
    s = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in active:
        shift = k - (n-1)
        # grad[i] += (1-h[i+shift]) - h[i-shift]
        i_plus = np.arange(n) + shift
        i_minus = np.arange(n) - shift
        mask_plus = (i_plus >= 0) & (i_plus < n)
        mask_minus = (i_minus >= 0) & (i_minus < n)
        term_plus = np.zeros(n)
        term_minus = np.zeros(n)
        term_plus[mask_plus] = 1.0 - h[i_plus[mask_plus]]
        term_minus[mask_minus] = h[i_minus[mask_minus]]
        grad += term_plus - term_minus
    
    grad /= len(active)
    
    # Take step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _fix_sum(h_new, n / 2.0)
    return h_new

                d = _compute_d(h, ii, jj, plus_c, minus_c, mask_plus, mask_minus)
                delta = _pair_optimal_delta(corr, d, h[ii], h[jj])
                if abs(delta) > 1e-14:
                    new_max = np.max(corr + delta * d)
                    if new_max * dx < best_c5 - 1e-14:
                        h[ii] += delta
                        h[jj] -= delta
                        corr += delta * d
                        best_c5 = new_max * dx
                        best_h = h.copy()
                        improved = True
                        total_improvements += 1
        
        if not improved:
            stagnation += 1
            if stagnation >= max_stagnation:
                break
        else:
            stagnation = 0
    
    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return best_h, best_c5

def _binary_local_search(h_binary, t_start, time_budget):
    """Local search on binary solutions: try all single moves (move a 1 to a 0 position)."""
    n = len(h_binary)
    dx = 2.0 / n
    h = h_binary.copy()
    best_c5 = _c5_of(h)
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - t_start < time_budget:
        improved = False
        best_gain = 0
        best_move = None
        
        for oi in ones:
            for zi in zeros:
                # Try moving 1 from oi to zi
                h[oi] = 0
                h[zi] = 1
                c5 = _c5_of(h)
                if c5 < best_c5 - 1e-14:
                    best_gain = best_c5 - c5
                    best_move = (oi, zi)
                h[oi] = 1
                h[zi] = 0
        
        if best_move is not None:
            oi, zi = best_move
            h[oi] = 0
            h[zi] = 1
            best_c5 = _c5_of(h)
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = True
        else:
            break
    
    return best_h, best_c5

def _simulated_annealing(h_start, t_start, time_budget, seed=42):
    """Simulated annealing on binary solutions."""
    n = len(h_start)
    dx = 2.0 / n
    # Round to binary
    h = (h_start > 0.5).astype(float)
    h = _fix_sum(h, n / 2.0)
    # Make it exactly binary
    ones = np.where(h == 1)[0]
    if len(ones) != n // 2:
        # Fix
        if len(ones) > n // 2:
            to_remove = ones[len(ones) - n//2:]
            h[to_remove] = 0
        else:
            zeros = np.where(h == 0)[0]
            need = n // 2 - len(ones)
            h[zeros[:need]] = 1
    
    best_h = h.copy()
    best_c5 = _c5_of(h)
    current_c5 = best_c5
    rng = np.random.RandomState(seed)
    
    T_start = 0.05
    T_end = 0.001
    total_iters = 200000
    
    while time.time() - t_start < time_budget:
        iters_this_round = min(10000, total_iters)
        for it in range(iters_this_round):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            oi = ones[rng.randint(len(ones))]
            zi = zeros[rng.randint(len(zeros))]
            if oi == zi:
                continue
            h[oi] = 0
            h[zi] = 1
            new_c5 = _c5_of(h)
            delta = new_c5 - current_c5
            T = T_start * (T_end / T_start) ** (it / iters_this_round)
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[oi] = 1
                h[zi] = 0
        
        total_iters -= iters_this_round
        if total_iters <= 0:
            break
    
    return best_h, best_c5
