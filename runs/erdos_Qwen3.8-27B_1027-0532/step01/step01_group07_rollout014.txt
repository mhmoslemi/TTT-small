def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = exp_shifted.sum()
    val = c_max + (max_shifted + np.log(sum_exp)) / alpha
    w = exp_shifted / sum_exp
    
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M2 = w[cols - rows + n - 1]
    M1 = w[rows - cols + n - 1]
    grad = dx * (M2 @ (1.0 - h) - M1 @ h)
    return val, grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):

        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid

        h = np.clip(x, 0.0, 1.0)

        s = h.sum() - target
        return val + penalty * s * s
