    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:K]
    top_vals = corr[top_idx]
    max_val = top_vals[0]
    weights = np.exp(t_smooth * (top_vals - max_val))
    weights /= weights.sum()

    grad = np.zeros(n)
    for wi, idx in zip(weights, top_idx):
        lag = idx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag + m])
            grad[lag:lag + m] += -wi * h[:m]
        else:
            l = -lag
            m = n - l
            grad[:m] += wi * (1.0 - h[l:l + m])
            grad[l:l + m] += -wi * h[:m]
    grad *= dx
    return grad


def nesterov_smooth(h0, n, t, n_iter, step0=0.005):
    h = h0.copy().astype(float)
    h_prev = h.copy()
    beta = 1.0
    step = step0
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)

    for it in range(n_iter):
        beta_next = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = h + ((beta - 1.0) / beta_next) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_onto_constraint(y, n)

        obj, grad = smooth_max_and_grad(y, n, t)

        alpha = step
        for _ in range(15):
            h_new = y - alpha * grad
            h_new = project_onto_constraint(h_new, n)
            obj_new, _ = smooth_max_and_grad(h_new, n, t)
            if obj_new <= obj + 1e-14:
                break
            alpha *= 0.5

        h_prev = h
        h = h_new
        beta = beta_next

        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5


def subgradient_polish_multilag(h, n, n_iter=500, step0=0.02, K=3, t_smooth=0.5):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step0

    for it in range(n_iter):
        grad = multilag_subgrad(h, n, K=K, t_smooth=t_smooth)
        h = h - step * grad
        h = project_onto_constraint(h, n)

        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        step *= 0.999

    return best_h, best_c5


def subgradient_polish_single(h, n, n_iter=500, step0=0.03):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step0

    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag == 0:
            grad = 1.0 - 2.0 * h
        elif lag > 0:
            m = n - lag
            grad[:m] += (1.0 - h[lag:lag + m])
            grad[lag:lag + m] += -h[:m]
        else:
            l = -lag
            m = n - l
            grad[:m] += (1.0 - h[l:l + m])
            grad[l:l + m] += -h[:m]
        grad *= dx

        h = h - step * grad
        h = project_onto_constraint(h, n)

        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        step *= 0.9995

    return best_h, best_c5


def binary_sa(h, n, n_iter=3000, T0=0.005, T_end=0.0005, seed=42):
    rng = np.random.RandomState(seed)
    target = int(n / 2)

    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h_bin == 1)[0]
            to_remove = rng.choice(ones, n_ones - target, replace=False)
            h_bin[to_remove] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            to_add = rng.choice(zeros, target - n_ones, replace=False)
            h_bin[to_add] = 1

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin, n)
    curr_c5 = best_c5

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (float(it) / max(n_iter - 1, 1))
        if T < 1e-12:
            T = 1e-12

        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin, n)

        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

    return best_h, best_c5


def slsqp_optimize(h0, n, t, max_iter=300):
    dx = 2.0 / n

    def obj_fn(x):
        v, _ = smooth_max_and_grad(x, n, t)
        return v

    def grad_fn(x):
        _, g = smooth_max_and_grad(x, n, t)
        return g

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        result = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-12})
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def in_budget(frac=1.0):
        return elapsed() < budget_s * frac

    parent = np.array(initial_h_values, dtype=float)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n_parent)
    best_n = n_parent

    # Phase 1: Nesterov optimization on parent grid with temperature schedule
    h_curr = parent.copy()
    temps_coarse = [300, 150, 80, 40, 25, 15, 10, 7, 5, 3.5, 2.5, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15]
    for t in temps_coarse:
        if not in_budget(0.30):
            break
        h_opt, c5_opt = nesterov_smooth(h_curr, n_parent, t, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
        h_curr = h_opt

    # SLSQP refinement on parent
    if in_budget(0.35):
        for t in [2.0, 1.0, 0.5]:
            if not in_budget(0.35):
                break
            h_opt = slsqp_optimize(best_h, best_n, t, max_iter=200)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Multi-lag subgradient polish
    if in_budget(0.40):
        h_pol, c5_pol = subgradient_polish_multilag(best_h, best_n, n_iter=800, step0=0.02, K=5, t_smooth=0.4)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Single-lag subgradient polish
    if in_budget(0.43):
        h_pol, c5_pol = subgradient_polish_single(best_h, best_n, n_iter=800, step0=0.015)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Binary SA on parent grid
    if in_budget(0.48):
        h_sa, c5_sa = binary_sa(best_h, best_n, n_iter=5000, T0=0.004, T_end=0.0004, seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 2: Grid refinement to 2x
    n_fine = n_parent * 2
    if in_budget(0.52) and n_fine <= 600:
        x_src = np.linspace(0, 2, best_n, endpoint=False)
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_src, best_h)
        h_fine = project_onto_constraint(h_fine, n_fine)

        fine_temps = [60, 30, 15, 10, 7, 5, 3.5, 2.5, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25]
        for t in fine_temps:
            if not in_budget(0.78):
                break
            h_opt, c5_opt = nesterov_smooth(h_fine, n_fine, t, 250)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_fine
            h_fine = h_opt

        if in_budget(0.80):
            h_pol, c5_pol = subgradient_polish_multilag(best_h, best_n, n_iter=800, step0=0.01, K=5, t_smooth=0.3)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if in_budget(0.82):
            h_pol, c5_pol = subgradient_polish_single(best_h, best_n, n_iter=800, step0=0.01)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if in_budget(0.85):
            h_sa, c5_sa = binary_sa(best_h, best_n, n_iter=4000, T0=0.003, T_end=0.0003, seed=seed + 100)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 3: Random restarts
    if in_budget(0.87):
        n_try = best_n
        for i in range(12):
            if not in_budget(0.93):
                break
            if i % 4 == 0:
                h_pert = best_h + rng.randn(n_try) * 0.04
            elif i % 4 == 1:
                h_pert = best_h + rng.randn(n_try) * 0.10
            elif i % 4 == 2:
                h_pert = best_h + rng.randn(n_try) * 0.15
            else:
                h_pert = (rng.rand(n_try) * 2.0 > 1.0).astype(float)
            h_pert = project_onto_constraint(h_pert, n_try)
            h_opt, c5_opt = nesterov_smooth(h_pert, n_try, 3.0, 250)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 4: Final polish
    if in_budget(0.95):
        h_pol, c5_pol = subgradient_polish_multilag(best_h, best_n, n_iter=2000, step0=0.005, K=3, t_smooth=0.3)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if in_budget(0.97):
        h_pol, c5_pol = subgradient_polish_single(best_h, best_n, n_iter=1000, step0=0.003)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final verification
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
