    return float(np.max(corr)) * dx

    h = np.clip(h, 0.0, 1.0)

    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_eq(h):
    return np.sum(h) - _ts


def _cons_jac(h):
    return np.ones(_n_pts)


def _do_slsqp(h, n, alpha, bounds, cons, maxiter=3000):
    try:
        res = minimize(obj_smooth, h, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h


def _try_update(h_new, ts, best_h, best_score):
    h_new = project_feasible(h_new, ts)
    s_new = compute_c5(h_new)
    if s_new < best_score:
        return h_new, s_new
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _n_pts, _ts
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - 20.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    ts = n / 2.0
    _n_pts = n
    _ts = ts

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent with increasing penalty
    for lam in [1.0, 5.0, 10.0, 30.0, 100.0, 300.0]:
        h_cd = best_h.copy()
        for sweep in range(40):
            if _tl() < 80:
                break
            h_cd, S = coord_descent_sweep(h_cd, n, ts, lam, n_ternary=35)
            best_h, best_score = _try_update(h_cd, ts, best_h, best_score)

    # Phase 2: SLSQP refinement
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]

    for alpha in [10, 20, 50, 100, 200, 500]:
        if _tl() < 40:
            break
        h_try = _do_slsqp(best_h, n, alpha, bounds, cons)
        best_h, best_score = _try_update(h_try, ts, best_h, best_score)

    # Phase 3: Fine-tuned coordinate descent
    for lam in [100.0, 300.0, 500.0, 1000.0]:
        for _ in range(15):
            if _tl() < 40:
                break
            h_cd = best_h.copy()
            h_cd, S = coord_descent_sweep(h_cd, n, ts, lam, n_ternary=45)
            best_h, best_score = _try_update(h_cd, ts, best_h, best_score)

    # Phase 4: SLSQP with higher alpha for sharper max
    for alpha in [200, 500, 1000, 2000]:
        if _tl() < 40:
            break
        h_try = _do_slsqp(best_h, n, alpha, bounds, cons, maxiter=5000)
        best_h, best_score = _try_update(h_try, ts, best_h, best_score)

    # Phase 5: Try higher resolution
    n_current = len(best_h)
    for n_try in [441, 529, 625, 729]:
        if _tl() < 150:
            break
        ts_try = n_try / 2.0
        _n_pts = n_try
        _ts = ts_try

        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, len(best_h))
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, ts_try)

        for lam in [10.0, 50.0, 200.0]:
            if _tl() < 60:
                break
            for _ in range(15):
                if _tl() < 30:
                    break
                h_try, S = coord_descent_sweep(h_try, n_try, ts_try, lam, n_ternary=30)
                s_try = compute_c5(project_feasible(h_try, ts_try))
                if s_try < best_score:
                    best_score = s_try
                    best_h = project_feasible(h_try, ts_try)

        bounds_t = [(0.0, 1.0)] * n_try
        cons_t = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
        for alpha in [50, 200]:
            if _tl() < 30:
                break
            h_try = _do_slsqp(best_h, n_try, alpha, bounds_t, cons_t)
            best_h, best_score = _try_update(h_try, ts_try, best_h, best_score)

    n_current = len(best_h)
    ts = n_current / 2.0
    _n_pts = n_current
    _ts = ts

    # Phase 6: Random restarts with coordinate descent
    bounds = [(0.0, 1.0)] * n_current
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]

    for noise in [0.02, 0.04, 0.06, 0.1, 0.15, 0.25]:
        for _ in range(20):
            if _tl() < 40:
                break
            h_pert = best_h + rng.randn(n_current) * noise
            h_pert = project_feasible(h_pert, ts)
            for lam in [50.0, 200.0, 500.0]:
                h_pert, S = coord_descent_sweep(h_pert, n_current, ts, lam, n_ternary=35)
                best_h, best_score = _try_update(h_pert, ts, best_h, best_score)

    # Phase 7: Final SLSQP polish
    for alpha in [100, 300, 500, 1000]:
        if _tl() < 25:
            break
        h_try = _do_slsqp(best_h, n_current, alpha, bounds, cons, maxiter=5000)
        best_h, best_score = _try_update(h_try, ts, best_h, best_score)

    # Phase 8: Final coordinate descent sweep
    for lam in [200.0, 500.0, 1000.0]:
        if _tl() < 25:
            break
        h_cd = best_h.copy()
        h_cd, S = coord_descent_sweep(h_cd, n_current, ts, lam, n_ternary=50)
        best_h, best_score = _try_update(h_cd, ts, best_h, best_score)

    best_h = project_feasible(best_h, ts)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
