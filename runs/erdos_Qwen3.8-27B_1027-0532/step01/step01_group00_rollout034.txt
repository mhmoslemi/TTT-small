    v = np.zeros(n)  # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01

        corr = compute_correlations(h)
        shifted = corr - corr.max()
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
