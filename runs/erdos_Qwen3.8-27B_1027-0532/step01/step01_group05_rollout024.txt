- We need: for all l in L*, h[a+l] = 0 and h[b+l] = 1
- Additionally, the swap shouldn't increase other lags too much

This gives us a filtering criterion! We can:
1. Find critical lags L*
2. For each position a with h[a]=1, check if h[a+l]=0 for all l in L*
3. For each position b with h[b]=0, check if h[b+l]=1 for all l in L*
4. Among valid (a,b) pairs, pick the one that minimizes the new max

This is much more efficient if |L*| is small!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_continuous(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(-h)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def smoothed_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]

    grad *= dx
    return obj, grad


def obj_wrapper(h, n, t):
    val, _ = smoothed_obj_and_grad(h, n, t)
    return val

def grad_wrapper(h, n, t):
    _, g = smoothed_obj_and_grad(h, n, t)
    return g

def constraint_fun(h, n):
    return np.sum(h) - n / 2.0

def constraint_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h0, n, t, max_iter=300):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{
        'type': 'eq',
        'fun': constraint_fun,
        'jac': constraint_jac,
        'args': (n,)
    }]
    try:
        result = minimize(
            obj_wrapper, h0.copy(), method='SLSQP',
            jac=grad_wrapper, args=(n, t),
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_continuous(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_continuous(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def to_binary(h, n):

def smoothed_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]
    grad *= dx
    return obj, grad


def obj_wrapper(h, n, t):
    val, _ = smoothed_obj_and_grad(h, n, t)
    return val


def grad_wrapper(h, n, t):
    _, g = smoothed_obj_and_grad(h, n, t)
    return g


def constraint_fun(h, n):
    return np.sum(h) - n / 2.0


def constraint_jac(h, n):
    return np.ones(n)


def slsqp_optimize(h0, n, t, max_iter=300):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
    try:
        result = minimize(
            obj_wrapper, h0.copy(), method='SLSQP',
            jac=grad_wrapper, args=(n, t),
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_continuous(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

    n_parent = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n_parent)
    best_n = n_parent

    def time_left():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Binarize parent and local search
    h_bin = to_binary(parent, n_parent)
    c5_bin = compute_c5(h_bin, n_parent)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    h_ls, c5_ls = local_search_binary(h_bin, n_parent, max_iters=30000,
                                      time_limit=time_left() * 0.10, seed=100)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = n_parent

    # Phase 2: SLSQP with multiple temperatures, then binarize + local search
    h_start = best_h.copy()
    slsqp_temps = [25, 60, 120, 250, 500]
    for t in slsqp_temps:
        if time_left() < budget_s * 0.35:
            break
        slsqp_time = min(time_left() * 0.06, 30.0)
        h_opt = slsqp_optimize(h_start, n_parent, t, max_iter=250)
        c5_opt = compute_c5(h_opt, n_parent)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
        h_bin2 = to_binary(h_opt, n_parent)
        ls_time = min(time_left() * 0.04, 20.0)
        h_ls2, c5_ls2 = local_search_binary(h_bin2, n_parent, max_iters=20000,
                                            time_limit=ls_time, seed=200 + t)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            best_n = n_parent
        if c5_opt < compute_c5(h_start, n_parent):
            h_start = h_opt.copy()

    # Phase 3: Iterated local search from best solution
    if time_left() > budget_s * 0.25:
        n_rounds = 8
        for r in range(n_rounds):
            if time_left() < budget_s * 0.12:
                break
            perturbation_intensity = 5 + r * 2
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(perturbation_intensity):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h_pert[a] = 0
                h_pert[b] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            ls_time = min(time_left() * 0.05, 25.0)
            h_ls3, c5_ls3 = local_search_binary(h_pert, n_parent, max_iters=20000,
                                                time_limit=ls_time, seed=500 + r * 7)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
                best_n = n_parent
            # Also try the unperturbed best again with different seed
            ls_time2 = min(time_left() * 0.03, 15.0)
            h_ls4, c5_ls4 = local_search_binary(best_h, n_parent, max_iters=15000,
                                                time_limit=ls_time2, seed=1000 + r * 13)
            if c5_ls4 < best_c5:
                best_c5 = c5_ls4
                best_h = h_ls4.copy()
                best_n = n_parent

    # Phase 4: Random restarts
    if time_left() > budget_s * 0.15:
        n_restarts = 12
        for i in range(n_restarts):
            if time_left() < budget_s * 0.08:
                break
            h_rand = np.zeros(n_parent)
            idx = rng.permutation(n_parent)[:n_parent // 2]
            h_rand[idx] = 1.0
            ls_time = min(time_left() * 0.04, 18.0)
            h_ls5, c5_ls5 = local_search_binary(h_rand, n_parent, max_iters=15000,
                                                time_limit=ls_time, seed=2000 + i * 3)
            if c5_ls5 < best_c5:
                best_c5 = c5_ls5
                best_h = h_ls5.copy()
                best_n = n_parent

    # Phase 5: Finer grid
    if time_left() > budget_s * 0.10:
        n_fine = n_parent * 2
        x_src = np.linspace(0, 2, best_n, endpoint=False)
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_src, best_h)
        h_bin_fine = to_binary(h_fine, n_fine)
        c5_fine = compute_c5(h_bin_fine, n_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_bin_fine.copy()
            best_n = n_fine
        if time_left() > budget_s * 0.06:
            ls_time = min(time_left() * 0.06, 30.0)
            h_ls6, c5_ls6 = local_search_binary(h_bin_fine, n_fine, max_iters=15000,
                                                time_limit=ls_time, seed=5000)
            if c5_ls6 < best_c5:
                best_c5 = c5_ls6
                best_h = h_ls6.copy()
                best_n = n_fine

    # Final SLSQP polish on best
    if time_left() > budget_s * 0.04:
        h_final = slsqp_optimize(best_h, best_n, t=100, max_iter=200)
        c5_final = compute_c5(h_final, best_n)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        h_bin_final = to_binary(best_h, best_n)
        c5_bin_final = compute_c5(h_bin_final, best_n)
        if c5_bin_final < best_c5:
            best_c5 = c5_bin_final
            best_h = h_bin_final.copy()

    # Ensure constraint
    best_h = project_continuous(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
