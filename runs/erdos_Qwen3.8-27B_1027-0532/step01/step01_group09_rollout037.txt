- h[i] appears as h_{j+k} when j+k = i, i.e., j = i-k: contributes h_{i-k} * (1-h_i)

def smooth_obj_and_grad(h, tau):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    cmax = corr.max()

    if max_shifted > 500:

    weights = exp_s / exp_s.sum()

            # -h[j+ak] for j+ak < n, i.e., j < n-ak
            grad[:n-ak] -= w * h[ak:]
    grad *= dx
    return value, grad

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    cmax = corr.max()
    shifted = tau * (corr - cmax)
    ms = shifted.max()
    if ms > 500:
        shifted = shifted - ms
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    value = cmax + log_sum / tau
    weights = exp_s / exp_s.sum()
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(n_full):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        ak = abs(k)
        if ak == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            grad[ak:] += w * (1.0 - h[:-ak])
            grad[:n - ak] -= w * h[ak:]
    grad *= dx
    return value, grad

def proj_grad_smooth(h0, tau_start, tau_end, n_iter, target, momentum=0.9, lr_init=0.05, tm=30):
    n = len(h0)
    h = project_feasible(h0.copy(), target)
    v = np.zeros(n)
    best = h.copy()
    best_val = eval_c5(h)
    lr = lr_init
    t0 = time.time()
    for t in range(n_iter):
        if time.time() - t0 > tm:
            break
        frac = t / max(1, n_iter - 1)
        tau = tau_start + (tau_end - tau_start) * frac
        val, grad = smooth_obj_and_grad(h, tau)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = project_feasible(h - lr * v, target)
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_val:
            best = h.copy()
            best_val = c5
        lr = lr_init * (1.0 + t / n_iter) ** (-1.5)
    return best, best_val

def subgrad_opt(h0, tm, target, lr_init=0.05, beta=0.9):
    n = len(h0)
    h = project_feasible(h0.copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    v = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    while time.time() - t0 < tm:
        corr = compute_corr(h)
        ki = np.argmax(corr)
        k = ki - (n - 1)
        ak = abs(k)
        grad = np.zeros(n)
        if ak == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[ak:] += 1.0 - h[:-ak]
            grad[:n - ak] -= h[ak:]
        grad *= 2.0 / n
        v = beta * v + (1.0 - beta) * grad
        h = project_feasible(h - lr * v, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best = h.copy()
            best_c5 = c5
        lr *= 0.9995
        if lr < 1e-12:
            break
    return best, best_c5

def coord_polish(h0, target, n_sweeps=3):
    n = len(h0)
    h = h0.copy()
    best_c5 = eval_c5(h)
    deltas = [0.02, 0.05, 0.1, 0.15, -0.02, -0.05, -0.1, -0.15]
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            best_nv = old_val
            best_local = best_c5
            best_h = None
            for d in deltas:
                nv = old_val + d
                if nv < -0.01 or nv > 1.01:
                    continue
                nv = np.clip(nv, 0.0, 1.0)
                h[i] = nv
                h_proj = project_feasible(h, target)
                c5 = eval_c5(h_proj)
                if c5 < best_local - 1e-10:
                    best_local = c5
                    best_nv = nv
                    best_h = h_proj.copy()
            if best_h is not None:
                h = best_h
                best_c5 = best_local
                improved = True
            else:
                h[i] = old_val
        if not improved:
            break
    return h, best_c5

def resample_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.linspace(0.0, n_src, n_src)
    xd = np.linspace(0.0, n_src, n_dst)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), target_dst)

def make_structured_init(n, rng):
    r = rng.randint(10)
    x = np.linspace(0.0, 2.0, n)
    if r == 0:
        bw = rng.uniform(0.3, 0.8)
        h = 0.5 + 0.5 * np.cos(np.pi * (x - 1.0) / bw)
        h = np.clip(h, 0, 1)
    elif r == 1:
        center = rng.uniform(0.4, 1.6)
        sigma = rng.uniform(0.2, 0.5)
        h = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif r == 2:
        h = 0.5 * np.ones(n)
        for _ in range(rng.randint(2, 5)):
            freq = rng.choice([1, 2, 3, 4, 5])
            phase = rng.uniform(0, 2.0 * np.pi)
            h += 0.15 * np.sin(np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif r == 3:
        nblocks = rng.randint(3, 9)
        block_vals = rng.uniform(0.15, 0.85, nblocks)
        h = np.repeat(block_vals, n // nblocks + 1)[:n]
    elif r == 4:
        a = rng.uniform(0.1, 0.4)
        b = rng.uniform(0.6, 0.9)
        h = np.linspace(a, b, n)
    elif r == 5:
        c1, c2 = sorted(rng.uniform(0.2, 1.8, 2))
        s1, s2 = rng.uniform(0.15, 0.4, 2)
        h = np.maximum(np.exp(-0.5 * ((x - c1) / s1) ** 2),
                        np.exp(-0.5 * ((x - c2) / s2) ** 2))
    elif r == 6:
        peak = rng.uniform(0.3, 1.7)
        width = rng.uniform(0.4, 1.0)
        h = np.maximum(0.0, 1.0 - np.abs(x - peak) / width)
    elif r == 7:
        h = 1.0 / (1.0 + np.exp(-rng.uniform(3, 8) * (x - rng.uniform(0.4, 1.6))))
    elif r == 8:
        h = rng.random(n)
        k = 5
        kern = np.ones(k) / k
        h = np.convolve(h, kern, mode='same')
    else:
        h = rng.random(n)
    return project_feasible(h, n / 2.0)

def refine_resolution(h_coarse, n_fine, target_fine, tm=20):
    h_start = resample_h(h_coarse, n_fine, target_fine)
    noise_scale = 0.01
    h_noisy = project_feasible(h_start + np.random.RandomState(0).normal(0, noise_scale, n_fine), target_fine)
    h_opt, c_opt = proj_grad_smooth(h_noisy, 20.0, 150.0, 2000, target_fine,
                                     momentum=0.92, lr_init=0.03, tm=tm)
    return h_opt, c_opt

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def tleft(frac=0.95):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    n_init = len(h_init)
    target_init = n_init / 2.0
    best = h_init.copy()
    best_c5 = eval_c5(best)

    # Phase 1: Smoothed gradient at initial resolution
    if tleft(0.75) > 30:
        h1, c1 = proj_grad_smooth(best, 5.0, 60.0, 4000, target_init,
                                   momentum=0.9, lr_init=0.06,
                                   tm=min(40, tleft(0.70)))
        if c1 < best_c5:
            best, best_c5 = h1, c1

    if tleft(0.70) > 20:
        h1b, c1b = proj_grad_smooth(best, 20.0, 150.0, 3000, target_init,
                                     momentum=0.93, lr_init=0.03,
                                     tm=min(25, tleft(0.65)))
        if c1b < best_c5:
            best, best_c5 = h1b, c1b

    # Subgradient pass
    if tleft(0.65) > 15:
        h2, c2 = subgrad_opt(best, min(20, tleft(0.65)), target_init,
                              lr_init=0.04, beta=0.9)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    # Phase 2: Try higher resolution
    if tleft(0.55) > 40:
        n_fine = n_init * 2
        target_fine = n_fine / 2.0
        h_fine, c_fine = refine_resolution(best, n_fine, target_fine,
                                            tm=min(50, tleft(0.50)))
        if c_fine < best_c5:
            best, best_c5 = h_fine, c1  # placeholder, recompute below
            best_c5 = eval_c5(best)
        # Additional optimization at fine resolution
        if tleft(0.50) > 30:
            h_f2, c_f2 = proj_grad_smooth(best, 30.0, 200.0, 3000, target_fine,
                                           momentum=0.93, lr_init=0.02,
                                           tm=min(40, tleft(0.45)))
            if c_f2 < best_c5:
                best, best_c5 = h_f2, c_f2
        if tleft(0.45) > 15:
            h_f3, c_f3 = subgrad_opt(best, min(20, tleft(0.45)), target_fine,
                                      lr_init=0.02, beta=0.92)
            if c_f3 < best_c5:
                best, best_c5 = h_f3, c_f3

    # Phase 3: Try intermediate resolution n=300
    if tleft(0.45) > 30 and len(best) != 300:
        n_int = 300
        target_int = n_int / 2.0
        h_start = resample_h(best, n_int, target_int)
        h_i, c_i = proj_grad_smooth(h_start, 10.0, 100.0, 3000, target_int,
                                     momentum=0.9, lr_init=0.04,
                                     tm=min(30, tleft(0.40)))
        if c_i < best_c5:
            best, best_c5 = h_i, c_i
        if tleft(0.40) > 15:
            h_i2, c_i2 = subgrad_opt(best, min(15, tleft(0.40)), target_int,
                                      lr_init=0.03, beta=0.88)
            if c_i2 < best_c5:
                best, best_c5 = h_i2, c_i2

    # Phase 4: Random restarts and perturbation search
    n_best = len(best)
    target_best = n_best / 2.0
    for attempt in range(40):
        if tleft(0.30) < 8:
            break
        scale = 0.015 + 0.04 * rng.random()
        h_pert = project_feasible(best + rng.normal(0, scale, size=n_best), target_best)
        h_opt, c_opt = proj_grad_smooth(h_pert, 30.0, 120.0, 1500, target_best,
                                         momentum=0.92, lr_init=0.03,
                                         tm=min(15, tleft(0.30)))
        if c_opt < best_c5:
            best, best_c5 = h_opt, c_opt
            # Continue optimizing from new best
            if tleft(0.30) > 10:
                h_opt2, c_opt2 = proj_grad_smooth(best, 50.0, 200.0, 2000, target_best,
                                                   momentum=0.95, lr_init=0.015,
                                                   tm=min(15, tleft(0.30)))
                if c_opt2 < best_c5:
                    best, best_c5 = h_opt2, c_opt2

    # Also try structured inits at best resolution
    for si in range(5):
        if tleft(0.25) < 10:
            break
        h_s = make_structured_init(n_best, rng)
        # Blend with current best
        alpha = rng.uniform(0.3, 0.7)
        h_blend = project_feasible(alpha * best + (1 - alpha) * h_s, target_best)
        h_s_opt, c_s_opt = proj_grad_smooth(h_blend, 10.0, 100.0, 2000, target_best,
                                             momentum=0.9, lr_init=0.04,
                                             tm=min(20, tleft(0.25)))
        if c_s_opt < best_c5:
            best, best_c5 = h_s_opt, c_s_opt

    # Phase 5: Coordinate descent polish
    if tleft(0.15) > 10:
        h_cd, c_cd = coord_polish(best, target_best, n_sweeps=4)
        if c_cd < best_c5:
            best, best_c5 = h_cd, c_cd

    # Phase 6: Final intensive optimization
    if tleft(0.10) > 25:
        h_final, c_final = proj_grad_smooth(best, 50.0, 300.0, 4000, target_best,
                                             momentum=0.95, lr_init=0.015,
                                             tm=min(60, tleft(0.08)))
        if c_final < best_c5:
            best, best_c5 = h_final, c_final

    if tleft(0.08) > 15:
        h_sf, c_sf = subgrad_opt(best, min(20, tleft(0.06)), target_best,
                                  lr_init=0.01, beta=0.95)
        if c_sf < best_c5:
            best, best_c5 = h_sf, c_sf

    if tleft(0.06) > 5:
        h_cd2, c_cd2 = coord_polish(best, target_best, n_sweeps=3)
        if c_cd2 < best_c5:
            best, best_c5 = h_cd2, c_cd2

    final_c5 = eval_c5(best)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best, best_c5, len(best)
