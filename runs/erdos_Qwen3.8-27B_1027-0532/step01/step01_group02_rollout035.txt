def smooth_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    scaled = corr * beta
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx)

def compute_c5_and_subgrad(h, n, top_m=3):
    """Compute C5 and subgradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Only consider non-negative lags (second half of full correlation)
    # Full correlation has 2n-1 elements, indexed 0 to 2n-2
    # Lag k (positive) is at index k + (n-1)
    # Lag 0 is at index n-1
    # We want lags 0, 1, ..., n-1 which are at indices n-1, n, ..., 2n-2
    
    half = corr[n-1:]  # lags 0 to n-1
    
    # Find top m lags
    m = min(top_m, len(half))
    top_idx = np.argsort(half)[-m:]
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in top_idx:
        p = idx  # positive lag p
        if p == 0:
            grad += 1 - 2 * h
        else:
            grad[:n-p] -= h[p:]
            grad[p:] += 1 - h[:n-p]
    
    grad /= m
    c5 = float(np.max(half) * dx)
    return c5, grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.01)
            h = h.copy()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.01)
            h = h.copy()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def compute_c5_and_subgrad(h, n, top_m=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    half = corr[n-1:]
    m = min(top_m, len(half))
    top_idx = np.argsort(half)[-m:]
    grad = np.zeros(n)
    for idx in top_idx:
        p = idx
        if p == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[:n-p] -= h[p:]
            grad[p:] += 1.0 - h[:n-p]
    grad /= m
    c5 = float(np.max(half) * dx)
    return c5, grad

def projected_subgrad(h, n, n_iter=1000, alpha_init=0.01, alpha_decay=0.999, top_m=3):
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h_curr)
    alpha = alpha_init
    
    for i in range(n_iter):
        c5, grad = compute_c5_and_subgrad(h_curr, n, top_m)
        h_new = h_curr - alpha * grad
        h_new = repair_h(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h_curr = h_new
        alpha *= alpha_decay
    
    return best_h, best_c5

def slsqp_optimize(h, n, maxiter=200, ftol=1e-12):
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        r.x = np.clip(r.x, 0.0, 1.0)
        r.x = repair_h(r.x, n)
        c5 = evaluate_c5(r.x)
        if c5 < evaluate_c5(h):
            return r.x, c5
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

def slsqp_smooth(h, n, beta, maxiter=100):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        half = corr[n-1:]
        scaled = half * beta
        m_val = np.max(scaled)
        return float((m_val + np.log(np.sum(np.exp(scaled - m_val)))) / beta * dx)
    try:
        r = minimize(
            obj, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        r.x = np.clip(r.x, 0.0, 1.0)
        r.x = repair_h(r.x, n)
        c5 = evaluate_c5(r.x)
        if c5 < evaluate_c5(h):
            return r.x, c5
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

**Idea: Alternating optimization**
The problem is: min_h max_k R_k(h) where R_k(h) = sum_i h[i+k](1-h[i]) * dx

We can think of this as a game: we choose h, and nature chooses k. By minimax theorem (if we relax to mixed strategies), we could:
1. Maintain a distribution over lags (the "dual" variable)
2. For a fixed distribution, the optimal h minimizes E_k[R_k(h)]
3. Update the distribution to put more weight on the lag that gives the maximum

This is like aMirror descent / Game theory approach.

Specifically:
- Let w be a probability distribution over lags {0, 1, ..., n-1}
- Inner problem: min_h sum_k w_k * R_k(h) subject to constraints
  This is a convex problem (sum of bilinear terms with fixed weights... wait, R_k is bilinear in h, not linear)

Actually R_k(h) = sum_i h[i+k](1-h[i]) = sum_i h[i+k] - sum_i h[i+k]*h[i]

This is NOT linear in h! It's quadratic (the second term). So the inner problem is non-convex.

Hmm, but if we linearize... Let's define g_i = 1-h[i]. Then R_k = sum h[i+k]*g[i]. This is bilinear in (h, g) with the constraint g = 1-h.

If we fix g (= 1-h), then R_k is linear in h: R_k = sum h[i+k] * g[i] = sum_j h[j] * g[j-k] (reindexing).

So we have an alternating minimization:
1. Fix g = 1-h. Minimize sum_k w_k * sum_j h[j]*g[j-k] over h.
   This is: min_h inner product of h with (sum_k w_k * shifted g)
   The minimum is achieved by putting h as small as possible where the "effective field" is large.
   But we have the constraint sum h = n/2 and 0 ≤ h ≤ 1.
   This is a linear program!

2. Fix h. Update g = 1-h (trivial).
3. Update w: increase weight on the lag that gives the maximum R_k.

Actually, step 1 is the key insight. For fixed g, the problem becomes:
min_h sum_j h[j] * F[j]  where F[j] = sum_k w_k * g[j-k] (with appropriate boundary handling)
subject to 0 ≤ h ≤ 1, sum h = n/2

This is a LP! The optimal solution is: set h[j] = 0 where F[j] is large (to minimize the objective), and h[j] = 1 where F[j] is small. Specifically, sort F and set the n/2 smallest to 1 and the rest to 0.

from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.01)
            h = h.copy()
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.01)
            h = h.copy()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def slsqp_optimize(h, n, maxiter=200, ftol=1e-12):
    try:
        r = minimize(
            lambda x: evaluate_c5(x), h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        r.x = np.clip(r.x, 0.0, 1.0)
        r.x = repair_h(r.x, n)
        c5 = evaluate_c5(r.x)
        if c5 < evaluate_c5(h):
            return r.x, c5
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

def compute_subgrad(h, n, top_m=5):

    corr = np.correlate(h, 1.0 - h, mode="full")
    half = corr[n-1:]
    m = min(top_m, len(half))
    top_idx = np.argsort(half)[-m:]
    grad = np.zeros(n)
    for idx in top_idx:
        p = int(idx)
        if p == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[:n-p] -= h[p:]
            grad[p:] += 1.0 - h[:n-p]
    grad /= m
    c5 = float(np.max(half) * dx)
    return c5, grad
