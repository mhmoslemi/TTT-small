        w = weights[k]
        if w < 1e-12:
            continue
        if l >= 0:
            # j - l in [0, n-1] means j in [l, l+n-1], intersect [0, n-1] = [l, n-1]
            # j + l in [0, n-1] means j in [-l, n-1-l], intersect [0, n-1] = [0, n-1-l]
            grad[l:] += w * (1.0 - x[:n-l])
            if l > 0:
                grad[:n-l] -= w * x[l:]
        else:
            # l < 0, let l' = -l
            lp = -l
            # j - l = j + lp in [0, n-1] means j in [-lp, n-1-lp], intersect [0, n-1] = [0, n-1-lp]
            # j + l = j - lp in [0, n-1] means j in [lp, lp+n-1], intersect [0, n-1] = [lp, n-1]
            grad[:n-lp] += w * (1.0 - x[lp:])
            grad[lp:] -= w * x[:n-lp]
    grad *= dx
    return grad

def smooth_max_obj(x):
    n = _curr_n
    eps = _curr_eps
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    return (m + max_s / eps + (1.0 / eps) * np.log(np.sum(np.exp(shifted - max_s)))) * dx

def smooth_max_grad(x):
    n = _curr_n
    eps = _curr_eps
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if l >= 0:
            grad[l:] += w * (1.0 - x[:n - l])
            if l > 0:
                grad[:n - l] -= w * x[l:]
        else:
            lp = -l
            grad[:n - lp] += w * (1.0 - x[lp:])
            grad[lp:] -= w * x[:n - lp]
    grad *= dx
