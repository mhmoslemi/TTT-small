Let me define influence_1_to_0[i] as a vector of length 2n-1:
For lag k (output index n-1+k):

For lag k (output index n-1+k):
- If 0 <= j-k < n: contribution is +(1-h[j-k]) [adding the new term]
- If 0 <= j+k < n: contribution is -h[j+k] [removing the old term]

Note that influence_0_to_1[j] = -influence_1_to_0[j] (by symmetry of the flip).

Alternative: Don't precompute influence vectors. Instead, for each candidate swap, compute the change directly in O(n):

```python
def evaluate_swap(corr, h, i, j, n):
    """Compute the new correlation after swapping i (1->0) and j (0->1). O(n)."""
    delta = np.zeros(2*n - 1)
    for k in range(-(n-1), n):
        ok = n - 1 + k
        d = 0.0
        # i: 1->0
        if 0 <= i - k < n:
            d -= (1.0 - h[i - k])
        if 0 <= i + k < n:
            d += h[i + k]
        # j: 0->1
        if 0 <= j - k < n:
            d += (1.0 - h[j - k])
        if 0 <= j + k < n:
            d -= h[j + k]
        delta[ok] = d
    new_corr = corr + delta
    return np.max(new_corr)
```

- delta at output index m (corresponding to lag k = m - (n-1)):
  - Term from i as first factor: -(1-h[i-k]) if 0 <= i-k < n
    i-k = i - (m-(n-1)) = n-1+i-m

def compute_delta_vectorized(h, i, j, n):
    m = np.arange(2*n - 1)
    delta = np.zeros(2*n - 1)
    
    # i: 1->0
    # Term 1: 0 <= n-1+i-m < n → i <= m <= i+n-1
    mask1 = (m >= i) & (m <= i + n - 1) & (m < 2*n - 1)
    if np.any(mask1):
        idx = n - 1 + i - m[mask1]
        delta[mask1] += -(1.0 - h[idx])
    
    # Term 2: 0 <= m-(n-1-i) < n → n-1-i <= m <= 2n-2-i
    mask2 = (m >= n-1-i) & (m <= 2*n-2-i) & (m >= 0) & (m < 2*n-1)
    if np.any(mask2):
        idx = m[mask2] - (n-1-i)
        delta[mask2] += h[idx]
    
    # j: 0->1
    # Term 1: 0 <= n-1+j-m < n → j <= m <= j+n-1
    mask3 = (m >= j) & (m <= j + n - 1) & (m < 2*n - 1)
    if np.any(mask3):
        idx = n - 1 + j - m[mask3]
        delta[mask3] += (1.0 - h[idx])
    
    # Term 2: 0 <= m-(n-1-j) < n → n-1-j <= m <= 2n-2-j
    mask4 = (m >= n-1-j) & (m <= 2*n-2-j) & (m >= 0) & (m < 2*n-1)
    if np.any(mask4):
        idx = m[mask4] - (n-1-j)
        delta[mask4] -= h[idx]
    
    return delta

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_objective(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    return (max_corr + np.log(sum_exp) / t) * dx

def smoothed_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]
    
    grad *= dx
    return grad

def slsqp_optimize(h0, n, t, max_iter=300):

        return smoothed_objective(x, n, t)

        return smoothed_grad(x, n, t)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:

                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-10})
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()
