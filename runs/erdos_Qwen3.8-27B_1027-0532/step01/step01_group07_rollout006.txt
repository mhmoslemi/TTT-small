    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n = 44
    if h0 is not None and len(h0) == n:
        h_start = project_feasible(h0.copy(), n / 2.0)
    else:
        h_start = _make_binary_init(n, seed)

    c5_start = compute_c5(h_start)
    best_c5 = c5_start
    best_h = h_start.copy()

    phase_deadline = deadline - 280
    h1, c5_1 = gd_phase(h_start, 5.0, 50.0, 4000, phase_deadline, 0.5)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    if time.time() < deadline - 240:
        phase_deadline = min(deadline - 240, time.time() + 60)
        h2, c5_2 = gd_phase(best_h, 50.0, 300.0, 4000, phase_deadline, 0.2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    if time.time() < deadline - 200:
        h_ref = active_set_refine(best_h, 8, 500)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if time.time() < deadline - 160:
        phase_deadline = min(deadline - 160, time.time() + 50)
        h3, c5_3 = gd_phase(best_h, 300.0, 2000.0, 3000, phase_deadline, 0.05)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    if time.time() < deadline - 120:
        h_ref = active_set_refine(best_h, 10, 500)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if time.time() < deadline - 80:
        h_cd, c5_cd = coordinate_descent(best_h, min(deadline - 80, time.time() + 40), max_rounds=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time.time() < deadline - 60:
        for rnd in range(10):
            if time.time() > deadline - 10:
                break
            h_pert = best_h + rng.normal(0, 0.02, n)
            h_pert = project_feasible(h_pert, n / 2.0)
            h4, c5_4 = gd_phase(h_pert, 100.0, 800.0, 1500, deadline - 10, 0.1)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
            if time.time() > deadline - 30:
                break

    for n_try in [42, 46]:
        if time.time() > deadline - 40:
            break
        if best_h is not None:
            t_old = np.linspace(0, 2, len(best_h))
            t_new = np.linspace(0, 2, n_try)
            h_try = np.interp(t_new, t_old, best_h)
        else:
            h_try = _make_binary_init(n_try, seed + n_try)
        h_try = project_feasible(h_try, n_try / 2.0)
        h5, c5_5 = gd_phase(h_try, 10.0, 200.0, 2000, deadline - 10, 0.3)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        h_ref = active_set_refine(best_h, 8, 300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if best_h is None:
        best_h = np.zeros(44)
        best_h[::2] = 1.0
        best_h = project_feasible(best_h, 22.0)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
