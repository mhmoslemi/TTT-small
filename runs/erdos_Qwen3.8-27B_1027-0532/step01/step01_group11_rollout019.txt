Let me re-read: `np.correlate(h, 1-h, mode='full')` computes the cross-correlation. For h = 0.5 * ones(n), 1-h = 0.5 * ones(n).

- At lag 0: sum of 0.25 * n = 0.25n
- At lag ±1: sum of 0.25 * (n-1) = 0.25(n-1)

            pop[i] = project_onto_constraint(pop[i], n0 / 2.0)
            pop_fit[i] = compute_c5(pop[i])
            if pop_fit[i] < best_c5:
                best_c5 = pop_fit[i]
                best_h = pop[i].copy()
                no_improve = 0

    # Final aggressive local refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 60:
        for eps_final in [0.001, 0.0005, 0.0002]:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 30:
                break
            h_f = local_refine(best_h, maxiter=400, eps=eps_final)
            h_f_c5 = compute_c5(h_f)
            if h_f_c5 < best_c5:
                best_c5 = h_f_c5
                best_h = h_f.copy()

    # Very tight final polish
    elapsed = time.time() - start_time
    if elapsed < budget_s - 20:
        h_p = local_refine(best_h, maxiter=500, eps=0.0001)
        h_p_c5 = compute_c5(h_p)
        if h_p_c5 < best_c5:
            best_c5 = h_p_c5
            best_h = h_p.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
