```python
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    return result

def lse_obj_wrapper(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = corr.max()
    scaled = (corr - cmax) / T
    scaled = np.clip(scaled, -50, 50)
    logsum = np.log(np.sum(np.exp(scaled)))
    return dx * (cmax + T * logsum)

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def lse_optimize(h, n, T, max_iter=5000):
    try:
        res = minimize(lse_obj_wrapper, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)

def simulated_annealing(h, n, t_start, time_limit, T_init=0.01, T_min=1e-5):
    best = h.copy()
    best_score = score_c5(best)
    current = h.copy()
    current_score = best_score
    T = T_init
    n_iter = 0
    while time.time() - t_start < time_limit:
        n_iter += 1
        # Pick two random indices and swap mass
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        amt = np.random.uniform(-0.1, 0.1)
        new_i = np.clip(current[i] + amt, 0, 1)
        new_j = np.clip(current[j] - (new_i - current[i]), 0, 1)
        # Project to maintain sum
        trial = current.copy()
        trial[i] = new_i
        trial[j] = new_j
        trial = project_feasible(trial, n)
        s_new = score_c5(trial)
        delta = s_new - current_score
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current = trial
            current_score = s_new
            if current_score < best_score:
                best = current.copy()
                best_score = current_score
        # Cool down
        T *= 0.999
        if T < T_min:
            T = T_min
    return best, best_score
```

```python
def corr_gradient(h, n):
    """Compute gradient of all correlations with respect to h. Returns shape (2n-1, n)."""
    # offset = m - n + 1, ranges from -(n-1) to (n-1)
    # ∂corr[m]/∂h[k] = (1-h[k+offset])*I[0≤k+offset<n] - h[k-offset]*I[0≤k-offset<n]
    offsets = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each offset, compute the contribution
    # (1-h[k+offset]) for valid k: this is a shifted version of (1-h)
    # h[k-offset] for valid k: this is a shifted version of h
    
    # Actually, let me just do it with the loop since n=85 is small
    grad = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for idx, off in enumerate(offsets):
        # First term: (1-h[k+off]) for k+off in [0,n-1]
        # k ranges from max(0, -off) to min(n-1, n-1-off)
        k_start = max(0, -off)
        k_end = min(n, n - off)  # exclusive
        if k_end > k_start:
            grad[idx, k_start:k_end] += one_minus_h[k_start + off : k_end + off]
        # Second term: -h[k-off] for k-off in [0,n-1]
        # k ranges from max(0, off) to min(n-1, n-1+off)
        k_start2 = max(0, off)
        k_end2 = min(n, n + off)  # exclusive
        if k_end2 > k_start2:
            grad[idx, k_start2:k_end2] -= h[k_start2 - off : k_end2 - off]
    return grad
```

```python
def compute_active_gradient(h, n, threshold_frac=0.02):
    """Compute average gradient of active correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    threshold = cmax * (1.0 - threshold_frac)
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in active:
        off = m - n + 1
        # First term: (1-h[k+off]) for k+off in [0,n-1]
        k_min = max(0, -off)
        k_max = min(n, n - off)
        if k_max > k_min:
            grad[k_min:k_max] += one_minus_h[k_min+off:k_max+off]
        # Second term: -h[k-off] for k-off in [0,n-1]
        k_min2 = max(0, off)
        k_max2 = min(n, n + off)
        if k_max2 > k_min2:
            grad[k_min2:k_max2] -= h[k_min2-off:k_max2-off]
    
    grad /= len(active)
    return grad
```

_T = 1.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def _lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = corr.max()
    scaled = (corr - cmax) / _T
    scaled = np.clip(scaled, -50, 50)
    logsum = np.log(np.sum(np.exp(scaled)))
    return dx * (cmax + _T * logsum)

def _sum_con(x):
    return np.sum(x) - len(x) / 2.0

def _sum_con_jac(x):
    return np.ones(len(x))

def lse_slsqp(h, n, T, max_iter):
    global _T
    _T = T

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})

            return x_new, score_c5(x_new)
    except Exception:

        if k_max2 > k_min2:
