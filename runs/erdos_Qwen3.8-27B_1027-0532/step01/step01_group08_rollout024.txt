def binary_local_search(h_init, n, time_budget=30.0):
    t0 = time.time()
    h = (h_init > 0.5).astype(float)
    # Fix to have exactly n//2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones = np.where(h == 1)[0]
        h[ones[n//2:]] = 0.0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        h[zeros[:n//2 - n_ones]] = 1.0
    
    # Precompute position arrays (don't change during search)
    n_lags = 2 * n - 1
    idx = np.arange(n)
    lags_arr = np.arange(-(n-1), n)
    positions_f = idx[:, None] + lags_arr[None, :]  # (n, n_lags): p + lag
    positions_s = idx[:, None] - lags_arr[None, :]  # (n, n_lags): p - lag
    mask_f = (positions_f >= 0) & (positions_f < n)
    mask_s = (positions_s >= 0) & (positions_s < n)
    
    def compute_delta(h_current):
        d = np.zeros((n, n_lags))
        d[mask_f] = (1.0 - h_current[positions_f[mask_f]])
        d[mask_s] += (-h_current[positions_s[mask_s]])
        return d
    
    delta_01 = compute_delta(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()
    
    while time.time() - t0 < time_budget:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        
        zd = delta_01[zeros]
        od = delta_01[ones]
        
        tensor = corr[None, None, :] + zd[:, None, :] - od[None, :, :]
        new_maxes = np.max(tensor, axis=2)
        
        best_pos = np.unravel_index(np.argmin(new_maxes), new_maxes.shape)
        best_val = new_maxes[best_pos]
        
        if best_val < current_max - 1e-12:
            i = zeros[best_pos[0]]
            j = ones[best_pos[1]]
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta_01[i] - delta_01[j]
            delta_01 = compute_delta(h)
            current_max = best_val
            c5 = current_max * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * corr * dx
    max_s = np.max(scaled)
    log_sum_exp = max_s + np.log(np.sum(np.exp(scaled - max_s)))
    obj = log_sum_exp / tau
    weights = np.exp(scaled - max_s)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + lag:i_hi + lag])
        j_lo = max(0, lag)
        j_hi = min(n, n + lag)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (-h[j_lo - lag:j_hi - lag])
    return obj, grad * dx

def binary_local_search(h_init, n, time_budget=30.0):
    t0 = time.time()
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones = np.where(h == 1)[0]
        h[ones[n // 2:]] = 0.0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        h[zeros[:n // 2 - n_ones]] = 1.0

    n_lags = 2 * n - 1
    idx = np.arange(n)
    lags_arr = np.arange(-(n - 1), n)
    positions_f = idx[:, None] + lags_arr[None, :]
    positions_s = idx[:, None] - lags_arr[None, :]
    mask_f = (positions_f >= 0) & (positions_f < n)
    mask_s = (positions_s >= 0) & (positions_s < n)
    flat_f_indices = np.where(mask_f)
    flat_s_indices = np.where(mask_s)
    flat_f_vals = positions_f[flat_f_indices]
    flat_s_vals = positions_s[flat_s_indices]

    def compute_delta(hc):
        d = np.zeros((n, n_lags))
        d[flat_f_indices] = (1.0 - hc[flat_f_vals])
        d[flat_s_indices] += (-hc[flat_s_vals])
        return d

    delta_01 = compute_delta(h)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    dx = 2.0 / n
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()

    while time.time() - t0 < time_budget:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        zd = delta_01[zeros]
        od = delta_01[ones]
        tensor = corr[None, None, :] + zd[:, None, :] - od[None, :, :]
        new_maxes = np.max(tensor, axis=2)
        best_pos = np.unravel_index(np.argmin(new_maxes), new_maxes.shape)
        best_val = float(new_maxes[best_pos])
        if best_val < current_max - 1e-12:
            i = zeros[best_pos[0]]
            j = ones[best_pos[1]]
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta_01[i] - delta_01[j]
            delta_01 = compute_delta(h)
            current_max = best_val
            c5 = current_max * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def lbfgs_continuous(h0, n, target_sum, deadline, tau=50.0):
    t0 = time.time()
    penalty = 100.0

    def obj_pen(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        s = np.sum(x) - target_sum
        return o + penalty * s * s

    def grad_pen(x):
        _, g = soft_max_obj_and_grad(x, tau)
        s = np.sum(x) - target_sum
        return g + 2.0 * penalty * s

    try:
        remaining = deadline - time.time()
        if remaining < 10:
            return h0, compute_c5(h0)
        res = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': 5000, 'iprint': -1})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def pgd_optimize(h0, n, target_sum, deadline, tau=30.0, max_iter=2000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.05
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = soft_max_obj_and_grad(h, tau)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 50:
                lr *= 0.7
                tau *= 1.5
                stall = 0
                if lr < 0.001:
                    break
    return best_h, best_c5
