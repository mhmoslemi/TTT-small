    return float(np.max(c))

def project_feasible(h, target):

    hi = 1.0 - np.min(h) + 0.5
    for _ in range(200):
        mid = (lo + hi) * 0.5

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_max_grad(x, alpha):
    m = np.max(x)
    e = np.exp(alpha * (x - m))
    w = e / e.sum()
    return w  # gradient of smooth_max w.r.t. x is w

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    e = np.exp(alpha * (c - m))
    e_sum = e.sum()
    w = e / e_sum
    sm = m + np.log(e_sum) / alpha
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t] * dx
        if wt < 1e-15:
            continue
        k = t - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += wt * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wt * h[lo2 - k:hi2 - k]
    return sm, grad


def sa_binary(h_init, n, time_limit, rng, T_scale=5.0):
    dx = 2.0 / n
    target = n // 2
    h = h_init.copy()
    idx = np.argsort(-h)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    c_arr = np.correlate(h, 1.0 - h, mode='full') * dx
    current_cost = float(np.max(c_arr))
    best_cost = current_cost
    best_h = h.copy()
    T = T_scale * dx
    peak_T = T
    ones_arr = np.where(h == 1)[0].tolist()
    zeros_arr = np.where(h == 0)[0].tolist()
    start = time.time()
    stall = 0
    max_stall = 8000
    cool_rate = 0.999995
    min_T = dx * 0.05
    use_double = n >= 60
    while time.time() - start < time_limit:
        if use_double and rng.random() < 0.3 and len(ones_arr) >= 2 and len(zeros_arr) >= 2:
            i1 = ones_arr[rng.randint(len(ones_arr))]
            i2 = ones_arr[rng.randint(len(ones_arr)-1)]
            if i2 >= (i1 if i2 > i1 else i1):
                i2 -= 1
                if i2 < 0:
                    i2 = 0
            j1 = zeros_arr[rng.randint(len(zeros_arr))]
            j2 = zeros_arr[rng.randint(len(zeros_arr)-1)]
            if j2 >= (j1 if j2 > j1 else j1):
                j2 -= 1
                if j2 < 0:
                    j2 = 0
            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode='full') * dx
            new_cost = float(np.max(c_new))
            if new_cost <= current_cost or rng.random() < np.exp(-(new_cost - current_cost) / T):
                current_cost = new_cost
                if new_cost < best_cost:
                    best_cost = new_cost
                    best_h = h.copy()
                    stall = 0
                ones_arr.remove(i1)
                ones_arr.remove(i2)
                ones_arr.extend([j1, j2])
                zeros_arr.remove(j1)
                zeros_arr.remove(j2)
                zeros_arr.extend([i1, i2])
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        else:
            i = ones_arr[rng.randint(len(ones_arr))]
            j = zeros_arr[rng.randint(len(zeros_arr))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode='full') * dx
            new_cost = float(np.max(c_new))
            if new_cost <= current_cost or rng.random() < np.exp(-(new_cost - current_cost) / T):
                current_cost = new_cost
                if new_cost < best_cost:
                    best_cost = new_cost
                    best_h = h.copy()
                    stall = 0
                ones_arr.remove(i)
                ones_arr.append(j)
                zeros_arr.remove(j)
                zeros_arr.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
        stall += 1
        if stall > max_stall:
            T = peak_T * 1.5
            stall = 0
        else:
            T = max(T * cool_rate, min_T)
    return best_h, best_cost


def continuous_refine(h_init, n, time_limit, alpha_start=10.0):
    target = n / 2.0
    h = h_init.astype(float).copy()
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_start
    lr = 0.5
    start = time.time()
    patience = 0
    max_patience = 800
    while time.time() - start < time_limit:
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        sm, grad = smooth_max_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        improved = False
        for attempt in range(25):
            step = lr / (gnorm + 1e-15)
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_val + 1e-12:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-14:
                break
        if patience % 200 == 0 and patience > 0:
            alpha = min(alpha * 2.0, 500.0)
    return best_h, best_c5


def interpolate_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) + 0.5
    x_new = np.arange(n_new) + 0.5
    return np.interp(x_new, x_old, h)


def make_structured_init(n, rng):
    h = np.zeros(n)
    period = rng.randint(4, 12)
    fill = period // 3 + rng.randint(0, 2)
    fill = max(1, min(fill, period - 1))
    for i in range(n):
        phase = (i * period // n + rng.uniform(-0.3, 0.3)) % period
        if phase < fill:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)
