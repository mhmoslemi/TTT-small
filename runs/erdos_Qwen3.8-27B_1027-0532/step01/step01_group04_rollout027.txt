def subgradient_c5(h):
    """Compute subgradient of max correlation w.r.t. h"""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    k_star = np.argmax(corr)
    j = k_star - (n - 1)  # the lag
    
    grad = np.zeros(n)
    # First term: (1 - h[m-j]) for valid m
    # m-j valid means 0 <= m-j < n, i.e., j <= m < n+j
    m_lo = max(0, j)
    m_hi = min(n, n + j)
    if m_lo < m_hi:
        grad[m_lo:m_hi] += 1.0 - h[m_lo - j:m_hi - j]
    
    # Second term: -h[m+j] for valid m
    # m+j valid means 0 <= m+j < n, i.e., -j <= m < n-j
    m_lo2 = max(0, -j)
    m_hi2 = min(n, n - j)
    if m_lo2 < m_hi2:
        grad[m_lo2:m_hi2] -= h[m_lo2 + j:m_hi2 + j]
    
    return grad

    return g


def cons_eq_global(h):
    return np.sum(h) - _g_ts


def cons_eq_jac_global(h):
    return np.ones(_g_n)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 1.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_penalty, _g_ts, _g_n
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    ts_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def time_left():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: Polish initial with SLSQP
    _g_n = n_init
    _g_ts = ts_init
    bounds_init = [(0.0, 1.0)] * n_init

    for alpha in [30, 80, 150, 300, 500]:
        if time_left() < 25:
            break
        _g_alpha = alpha
        try:
            res = minimize(obj_lse_global, best_h, method='SLSQP',
                           bounds=bounds_init,
                           constraints=[{'type': 'eq', 'fun': cons_eq_global,
                                         'jac': cons_eq_jac_global}],
                           options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts_init)
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Explore higher resolutions with L-BFGS-B
    grid_sizes = [350, 400, 450, 500, 550, 600, 700, 800]

    for n_try in grid_sizes:
        if time_left() < 100:
            break
        ts_try = n_try / 2.0
        h_cur = project_feasible(interpolate_h(best_h, n_try), ts_try)

        # L-BFGS-B with penalty on sum constraint
        for alpha in [15, 50, 120]:
            if time_left() < 50:
                break
            _g_n = n_try
            _g_ts = ts_try
            _g_alpha = alpha
            _g_penalty = 200.0
            bounds_t = [(0.0, 1.0)] * n_try
            try:
                res = minimize(obj_pen_global, h_cur, jac=obj_pen_grad_global,
                               method='L-BFGS-B', bounds=bounds_t,
                               options={'maxiter': 800, 'ftol': 1e-16,
                                        'gtol': 1e-9, 'maxls': 50})
                h_new = project_feasible(res.x, ts_try)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
                    h_cur = h_new
            except Exception:
                pass

        # SLSQP polish on this grid if improved
        if time_left() < 50:
            continue
        _g_n = n_try
        _g_ts = ts_try
        bounds_t2 = [(0.0, 1.0)] * n_try
        if len(best_h) == n_try:
            h_pol = best_h.copy()
        else:
            h_pol = project_feasible(interpolate_h(best_h, n_try), ts_try)

        for alpha in [80, 200, 400]:
            if time_left() < 30:
                break
            _g_alpha = alpha
            try:
                res = minimize(obj_lse_global, h_pol, method='SLSQP',
                               bounds=bounds_t2,
                               constraints=[{'type': 'eq', 'fun': cons_eq_global,
                                             'jac': cons_eq_jac_global}],
                               options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts_try)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
                    h_pol = h_new
            except Exception:
                pass

    # Phase 3: Diverse restarts on best resolution
    n_best = len(best_h)
    ts_best = n_best / 2.0
    _g_n = n_best
    _g_ts = ts_best
    bounds_best = [(0.0, 1.0)] * n_best

    restart_count = 0
    while time_left() > 60 and restart_count < 80:
        restart_count += 1
        strategy = rng.randint(0, 4)

        if strategy == 0:
            noise = rng.choice([0.03, 0.06, 0.1, 0.15])
            h_pert = best_h + rng.randn(n_best) * noise
        elif strategy == 1:
            h_pert = best_h.copy()
            start = rng.randint(0, n_best - 20)
            length = rng.randint(10, 40)
            h_pert[start:start+length] += rng.uniform(0.2, 0.5) * (1 if rng.rand() > 0.5 else -1)
        elif strategy == 2:
            freq = rng.randint(2, 8)
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 1, n_best)
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * x + phase)
        else:
            h_pert = rng.rand(n_best) * 0.3 + 0.35
            h_pert = h_pert + (best_h - 0.5) * 0.5

        h_pert = project_feasible(h_pert, ts_best)
        alpha = int(rng.choice([20, 50, 100, 200]))
        _g_alpha = alpha
        _g_penalty = 200.0

        try:
            res = minimize(obj_pen_global, h_pert, jac=obj_pen_grad_global,
                           method='L-BFGS-B', bounds=bounds_best,
                           options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-9})
            h_new = project_feasible(res.x, ts_best)
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new
        except Exception:
            pass

    # Phase 4: Final polish with increasing alpha
    if time_left() > 20:
        _g_n = n_best
        _g_ts = ts_best
        for alpha in [200, 500, 1000, 2000]:
            if time_left() < 15:
                break
            _g_alpha = alpha
            try:
                res = minimize(obj_lse_global, best_h, method='SLSQP',
                               bounds=bounds_best,
                               constraints=[{'type': 'eq', 'fun': cons_eq_global,
                                             'jac': cons_eq_jac_global}],
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts_best)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 5: Try one more L-BFGS-B sweep at the very end
    if time_left() > 20:
        _g_n = n_best
        _g_ts = ts_best
        for alpha in [50, 150]:
            if time_left() < 15:
                break
            _g_alpha = alpha
            _g_penalty = 500.0
            try:
                res = minimize(obj_pen_global, best_h, jac=obj_pen_grad_global,
                               method='L-BFGS-B', bounds=bounds_best,
                               options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_feasible(res.x, ts_best)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
            except Exception:
                pass

    h_final = project_feasible(best_h, n_best / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
