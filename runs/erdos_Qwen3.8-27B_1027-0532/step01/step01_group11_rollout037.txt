            break

        # For each pair (i, j), optimize h[i] + h[j] = S
        # by finding the best split t vs S-t
        
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                S = h_curr[i] + h_curr[j]
                t_lo = max(0.0, S - 1.0)
                t_hi = min(1.0, S)
                if t_hi - t_lo < 1e-12:
                    continue
                
                # Compute alpha and beta for all shifts
                # alpha[s] = coefficient of t in corr[n-1+s]
                # beta[s] = constant part
                
                # For shift s (correlation index n-1+s):
                # The terms involving i:
                #   if 0 <= i+s <= n-1: h[i]*(1-h[i+s]) = t*(1-h[i+s])
                #   if 0 <= i-s <= n-1: h[i-s]*(1-h[i]) = h[i-s]*(1-t)
                # The terms involving j:

                #   if 0 <= j-s <= n-1: h[j-s]*(1-h[j]) = h[j-s]*(1-(S-t))
