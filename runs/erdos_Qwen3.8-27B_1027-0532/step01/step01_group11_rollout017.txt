            remaining = 0.0

            if cv < best_v:

                best_t = t
            else:
                if t <= 0.3:
                    break
        if best_t > 0:
            stall = 0
            h = (1-best_t)*h + best_t*v
            h = project_onto_constraint(h, ts)
            if best_v < bc5:
                bc5 = best_v
                bh = h.copy()
        else:
            stall += 1
            if stall > 500:
                break
    return bh, bc5

def slsqp_smooth(h_init, eps, maxiter, time_limit):
    n = len(h_init)
    ts = n / 2.0
    dx = 2.0 / n
    h0 = project_onto_constraint(h_init.copy(), ts)
    
    def obj(h):
        corr = compute_all_corr(h)
        return smooth_max_val(corr, eps)
    
    def grad(h):
        corr = compute_all_corr(h)
        w = smooth_max_grad_w(corr, eps)
        omh = 1.0 - h
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n-1)
            if s >= 0:
                g[s:] += wk * dx * omh[:n-s]
                g[:n-s] -= wk * dx * h[s:]
            else:
                r = -s
                g[:n-r] += wk * dx * omh[r:]
                g[r:] -= wk * dx * h[:n-r]
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - ts, 'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(res.x, ts)
    except:
        h_opt = h0
    return h_opt, compute_c5(h_opt)
