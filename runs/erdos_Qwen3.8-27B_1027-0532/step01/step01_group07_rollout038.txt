def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)  # periodic interpolation
    return h_new

def smooth_max_descent(h, n, alpha_start, alpha_end, max_iter, deadline, lr_init=0.1):
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = alpha_start
    lr = lr_init
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 500:
            break
        
        # Adapt alpha
        alpha = min(alpha * (1 + 2.0 / max(it, 1)), alpha_end)
        
        _, grad = grad_smooth_max(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        accepted = False
        for backtrack in range(20):
            step = lr / (2 ** backtrack)
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_val + 1e-12:
                h = h_new
                accepted = True
                break
        if not accepted:
            break
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = c - c_max
    exp_c = np.exp(alpha * shifted)
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum
    smooth_val = c_max + np.log(exp_sum) / alpha
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def smooth_max_descent(h, n, alpha_start, alpha_end, max_iter, deadline, lr_init=0.1, momentum=0.8):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_start
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        alpha = min(alpha * (1.0 + 3.0 / (it + 5.0)), alpha_end)
        _, grad = grad_smooth_max(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_dir = grad / gnorm
        v = momentum * v + (1.0 - momentum) * grad_dir
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-14:
            break
        v_dir = v / vnorm
        accepted = False
        for bt in range(20):
            step = lr / (2.0 ** bt)
            h_new = h - step * v_dir
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_val + 1e-12:
                h = h_new
                accepted = True
                break
        if not accepted:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c5

def round_to_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:int(target)]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_rounds=500, deadline=None, seed=99):
    target = n / 2.0
    rng_ls = np.random.RandomState(seed)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = min(len(ones) * len(zeros), 300)
        for _ in range(n_tries):
            i = ones[rng_ls.randint(len(ones))]

            h_try[i] = 0.0
            h_try[j] = 1.0
            c5_try = compute_c5(h_try)

                if len(ones) == 0 or len(zeros) == 0:

    return best_h, best_c5
