```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 1.0)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 1.0)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def slsqp_optimize(h, n, maxiter=300, ftol=1e-12):
    try:
        def obj(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full")
            return float(np.max(corr) * dx)
        
        def constraint_fun(x):
            return np.sum(x) - n / 2.0
        
        r = minimize(
            obj, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint_fun}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if r.success or np.all(r.x >= -1e-9) and np.all(r.x <= 1 + 1e-9):
            r.x = np.clip(r.x, 0.0, 1.0)
            r.x = repair_h(r.x, n)
            return r.x, evaluate_c5(r.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)
```

```python
def weighted_subgradient(h, n, weights, alpha):
    """One step of weighted subgradient descent."""
    # Compute weighted sum of correlations
    # C(d) = corr[n-1+d] for positive lags, corr[n-1-d] for negative lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Gradient of weighted sum
    grad = np.zeros(n)
    center = n - 1
    for k in range(len(corr)):
        d = k - center  # lag
        w = weights[k]
        if w < 1e-10:
            continue
        if d > 0:
            # C = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # grad[i] = w * (1-h[i+d]) for i < n-d
            #         + w * (-h[i-d]) for i >= d
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] += w * (-h[:n-d])
        elif d < 0:
            dd = -d
            # C = sum_{i=dd}^{n-1} h[i]*(1-h[i-dd])
            # grad[i] = w * (1-h[i-dd]) for i >= dd
            #         + w * (-h[i+dd]) for i < n-dd
            grad[dd:] += w * (1.0 - h[:n-dd])
            grad[:n-dd] += w * (-h[dd:])
        # d == 0: C = sum h[i]*(1-h[i]), grad[i] = 1 - 2*h[i]
        else:
            grad += w * (1.0 - 2.0 * h)
    
    return h - alpha * grad, corr
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 1.0)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 1.0)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def compute_c5_gradient(h, weights):
    """Compute gradient of sum_k weights[k] * corr[k] where corr = correlate(h, 1-h, 'full')"""
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - center
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
        else:
            e = -d
            grad[e:] += w * (1.0 - h[:-e])
            grad[:-e] -= w * h[e:]
    return grad

            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full")
            return float(np.max(corr) * dx)

    best_c5 = evaluate_c5(h_curr)
    best_h = h_curr.copy()
```

```python
def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 1.0)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 1.0)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def _c5_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - center
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
        else:
            e = -d
            grad[e:] += w * (1.0 - h[:-e])
            grad[:-e] -= w * h[e:]
    return grad
```
