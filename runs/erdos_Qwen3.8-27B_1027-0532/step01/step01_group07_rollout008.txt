            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _lsq_obj(val):
    return val[-1]

def _lsq_ineq(val, n, m_idx):
    h = val[:n]
    t = val[n]
    m = m_idx - (n - 1)
    dx = 2.0 / n
    c_val = 0.0
    for j in range(n):
        if 0 <= j + m < n:
            c_val += h[j + m] * (1.0 - h[j])
    return t - c_val * dx

def _lsq_eq(val, n, _):
    return np.sum(val[:n]) - n / 2.0

def active_set_optimize(h_init, n, M, maxiter, deadline):
    target = n / 2.0
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    cur_h = h_init.copy()

    for outer in range(12):
        if time.time() > deadline - 5:
            break

        c_all = compute_c5_all(cur_h)
        c_max = float(np.max(c_all))

        top_M = min(M, 2 * n - 1)
        top_idx = np.argsort(c_all)[-top_M:]

        x0 = np.concatenate([cur_h, [c_max + 0.02]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

        cons = []
        for k in top_idx:
            cons.append({'type': 'ineq', 'fun': _lsq_ineq, 'args': (n, int(k))})
        cons.append({'type': 'eq', 'fun': _lsq_eq, 'args': (n, 0)})

        try:
            res = minimize(_lsq_obj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        except Exception:
            break

        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            cur_h = h_new
            M = min(M + 2, n)
        else:
            M = min(M + 4, 2 * n - 1)
            if M > 2 * n - 2:
                break

    return best_h, best_c5

def smooth_gradient_descent(h, n, alpha, max_iter, deadline, lr_start=0.04):
    target = n / 2.0
    dx = 2.0 / n
    best_local = compute_c5(h)
    best_local_h = h.copy()
    lr = lr_start
    stagnant = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        c_all = compute_c5_all(h)
        c_max = float(np.max(c_all))
        if c_max < best_local - 1e-12:
            best_local = c_max
            best_local_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 100:
            break

        shift_c = c_all - c_max
        exp_c = np.exp(alpha * shift_c)
        w = exp_c / exp_c.sum()

        t = np.arange(n)
        rows, cols = np.meshgrid(t, t, indexing='ij')
        M1 = w[rows + n - 1 - cols]
        M2 = w[cols - rows + n - 1]
        Part1 = M1.dot(1.0 - h)
        Part2 = M2.dot(h)
        grad = dx * (Part1 - Part2)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = lr * np.sqrt(dx)
        h_new = h - step * grad / gnorm
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new <= c_max + 1e-12:
            h = h_new
            if stagnant % 50 == 0 and stagnant > 50:
                lr *= 0.6
                if lr < 1e-9:
                    break
        else:
            lr *= 0.65
            if lr < 1e-10:
                break

    return best_local_h, best_local

def upsample(h, factor=2):
    return np.repeat(h, factor)

def _make_sinusoidal_init(n, rng):

    if mx > 0:

    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)
