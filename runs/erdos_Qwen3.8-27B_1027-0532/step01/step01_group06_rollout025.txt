_CURRENT_T = 1.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / _CURRENT_T
    return dx * (cmax + _CURRENT_T * np.log(np.sum(np.exp(scaled))))

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    return result

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))
