def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = c - c_max
    max_s = np.max(shifted)
    exp_c = np.exp(alpha * (shifted - max_s))
    s = exp_c.sum()
    val = c_max + max_s + np.log(s) / alpha
    w = exp_c / s
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    grad = dx * (M1.dot(1.0 - h) - M2.dot(h))
    return val, grad
