    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)

    weights = exps / wsum
    obj = (cmax + np.log(wsum) / alpha) * dx

            m_vals = np.arange(m_cnt)

            grad[m_vals + s] += w * (-h[m_vals])

            grad[m_vals] += w * one_minus_h[m_vals + s]
            grad[m_vals + s] += w * (-h[m_vals])

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
