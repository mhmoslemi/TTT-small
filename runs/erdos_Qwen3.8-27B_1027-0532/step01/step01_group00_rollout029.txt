First sum: this is the convolution of p and (1-h) evaluated at position j+n-1:
(p * (1-h))[j+n-1] = Σ_i p[j+n-1-i] * (1-h)[i] ✓

Second sum: let i' = i, so we want Σ_i p[n-1-j+i] * h[i]
This is (flip(p) * h)[j+n-1] where flip(p)[m] = p[2n-2-m]:
(flip(p) * h)[j+n-1] = Σ_i flip(p)[j+n-1-i] * h[i] = Σ_i p[2n-2-(j+n-1-i)] * h[i] = Σ_i p[n-1-j+i] * h[i] ✓

So:
grad = dx * (convolve(p, 1-h, 'full')[n-1 : n-1+n] - convolve(p[::-1], h, 'full')[n-1 : n-1+n])

Let me verify with n=3, j=1:
First: conv(p, [1-a,1-b,1-c], 'full')[2 : 5]
conv[m] = Σ_i p[m-i] * (1-h)[i]
conv[3] = p[3]*(1-h[0]) + p[2]*(1-h[1]) + p[1]*(1-h[2]) + p[0]*(1-h[3])...
wait, p has length 5 (indices 0-4), (1-h) has length 3 (indices 0-2)
conv has length 5+3-1 = 7
conv[m] = Σ_i p[m-i] * (1-h)[i] for valid indices
conv[3] = p[3]*(1-h[0]) + p[2]*(1-h[1]) + p[1]*(1-h[2])
= p[3]*(1-a) + p[2]*(1-b) + p[1]*(1-c)

We want first sum for j=1: Σ_i p[1+2-i]*(1-h[i]) = p[3-i]*(1-h[i]) for i=0,1,2
= p[3]*(1-a) + p[2]*(1-b) + p[1]*(1-c) ✓

Second: conv(p[::-1], h, 'full')[2:5]
p[::-1] has indices 0-4 with values p[4], p[3], p[2], p[1], p[0]
conv2[m] = Σ_i p[::-1][m-i] * h[i]
conv2[3] = p[::-1][3]*h[0] + p[::-1][2]*h[1] + p[::-1][1]*h[2]
= p[1]*a + p[2]*b + p[3]*c

We want second sum for j=1: Σ_i p[2-1+i]*h[i] = p[1+i]*h[i] for i=0,1,2
= p[1]*a + p[2]*b + p[3]*c ✓

So the formula is:
grad = dx * (np.convolve(p, 1-h, mode='full')[n-1:n-1+n] - np.convolve(p[::-1], h, mode='full')[n-1:n-1+n])

This is O(n log n) with FFT-based convolution!

    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    val = m + np.log(np.sum(exp_shifted)) / alpha

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n

    return np.max(corr) * dx

    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    p = exp_shifted / s
    val = m + np.log(s) / alpha
    return val, p

def obj_and_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    val, p = smooth_max_and_weights(corr, alpha)
    first = np.convolve(p, 1.0 - h, mode='full')
    second = np.convolve(p[::-1], h, mode='full')
    grad = dx * (first[n-1:n-1+n] - second[n-1:n-1+n])
    diff = np.sum(h) - n / 2.0
    obj = val + penalty * diff * diff
    grad = grad + 2.0 * penalty * diff

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    p = exp_shifted / s
    val = m + np.log(s) / alpha
    return val, p


def compute_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    _, p = smooth_max_and_weights(corr, alpha)
    first = np.convolve(p, 1.0 - h, mode='full')
    second = np.convolve(p[::-1], h, mode='full')
    grad = dx * (first[n - 1:n - 1 + n] - second[n - 1:n - 1 + n])
    diff = np.sum(h) - n / 2.0
    grad = grad + 2.0 * penalty * diff
    return grad


def compute_obj(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    val, _ = smooth_max_and_weights(corr, alpha)
    diff = np.sum(h) - n / 2.0
    return val + penalty * diff * diff

            lo = mid
        else:

    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
