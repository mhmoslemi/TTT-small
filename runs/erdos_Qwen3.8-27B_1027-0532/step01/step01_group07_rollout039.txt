def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    s = exp_c.sum()
    w = exp_c / s
    smooth_val = c_max + np.log(s) / alpha
    
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    
    return smooth_val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    if hi <= lo:
        return np.full_like(h, target / len(h))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    s = exp_c.sum()
    w = exp_c / s
    smooth_val = c_max + np.log(s) / alpha
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')

    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad
