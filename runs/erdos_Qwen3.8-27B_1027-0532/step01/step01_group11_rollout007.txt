        for k in range(2 * n - 1):
            if not active_mask[k]:
                continue
            s = k - (n - 1)
            if s >= 0 and s < n:

            elif s < 0:
                r = -s
                if r < n:
                    grad[:n-r] += dx * (1.0 - h[r:])
                    grad[r:] -= dx * h[:n-r]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        lr = 0.5 / gnorm
        improved = False
        for ws in range(20):
            h_new = project_onto_constraint(h - lr * grad, target_sum)
            c5_new = compute_c5(h_new)

                improved = True
                break

            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 50:
                break
    
    return best_h, best_c5

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:

                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad
