from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            if diff > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h

            continue

        h[i] = 0.0
        h[j] = 0.0

        h[i] = 1.0

        h[j] = hj

        pj = cj - cb

        B = pi - pj

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
