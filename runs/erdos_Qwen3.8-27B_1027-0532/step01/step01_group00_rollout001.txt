    grad = np.zeros(n)
    if lag == 0:
        grad = (1 - h) - h  # = 1 - 2h
    elif lag > 0:

        grad[:n-lag] -= h[lag:]

        grad[:n-la] += (1 - h[la:])
        # -h[m-la] for m in [la, n-1]
        grad[la:] -= h[:n-la]

        top_k = min(10, len(corr))
        top_idx = np.argsort(-corr)[:top_k]
        top_vals = corr[top_idx]
        
        # Softmax weights
        weights = np.exp(alpha * (top_vals - top_vals[0]))
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = idx - (n - 1)
            grad += w * compute_gradient_for_lag(h, lag)
        
        # Step
        h_new = h - step * grad
        
        # Project
        h_new = project_to_feasible(h_new, target_sum)
        
        # Evaluate
        score = evaluate_score(h_new)
        
        # Adaptive step
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            step *= 1.01  # slight increase

            # Backtracking

                step *= 0.5
                h_new = h - step * grad
                h_new = project_to_feasible(h_new, target_sum)
                score = evaluate_score(h_new)
                if score < best_score or score < evaluate_score(h):
                    h = h_new
                    break
            else:
                step *= 0.5

def project_to_feasible(h, target_sum):

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def subgradient_run(h0, max_iters, alpha, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)

    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break

        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_gradient_for_lag(h, lag)

        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-12:
            break

        improved = False
        for trial in range(20):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.4

        if not improved:
            stall += 1
            if stall > 300:
                h = best_h + rng.randn(n) * (0.03 + 0.07 * stall / 500.0) / np.sqrt(n)
                h = project_to_feasible(h, target_sum)
                step = step_init * (0.5 + 0.5 * rng.rand())
                stall = 0

        if it > 0 and it % 1000 == 0:
            step = max(step_init * 0.95 ** (it // 1000), step_init * 0.01)

    return best_h, best_score


def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
