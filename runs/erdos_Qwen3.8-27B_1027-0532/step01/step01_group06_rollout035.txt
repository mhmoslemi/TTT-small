import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x):
    """Project x onto {h: 0 <= h_i <= 1, sum(h) = n/2}"""
    n = len(x)
    target = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    """Compute all correlation values"""
    return np.correlate(h, 1.0 - h, mode='full')

def score_c5(h):
    dx = 2.0 / len(h)
    return np.max(compute_corr(h)) * dx

def subgradient_step(h, corr, alpha, tol=0.005):
    """Compute subgradient step for min max corr"""
    n = len(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        # (1 - h[j+shift]) for valid j+shift
        idx1 = np.arange(n) + shift
        valid1 = (idx1 >= 0) & (idx1 < n)
        g[valid1] += (1.0 - h[idx1[valid1]])
        # -h[j-shift] for valid j-shift
        idx2 = np.arange(n) - shift
        valid2 = (idx2 >= 0) & (idx2 < n)
        g[valid2] -= h[idx2[valid2]]
    g /= len(active)
    
    new_h = project_feasible(h - alpha * g)
    return new_h

def exact_pair_swap(h, j, jp):
    """Find optimal δ for swapping mass from j to jp"""
    n = len(h)
    corr = compute_corr(h)
    
    # Compute gradients
    g = np.zeros(2*n - 1)
    gp = np.zeros(2*n - 1)
    max_corr = np.max(corr)
    
    for k in range(2*n - 1):
        shift = k - (n - 1)
        # Gradient at j
        s1 = j + shift
        s2 = j - shift
        g[k] = (1 - h[s1] if 0 <= s1 < n else 0) - (h[s2] if 0 <= s2 < n else 0)
        # Gradient at jp
        s1p = jp + shift
        s2p = jp - shift
        gp[k] = (1 - h[s1p] if 0 <= s1p < n else 0) - (h[s2p] if 0 <= s2p < n else 0)
    
    # delta = g'_k - g_k
    delta = gp - g
    base = corr
    
    # Find optimal δ: minimize max_k (base[k] + δ * delta[k])
    # subject to 0 <= h[j] - δ <= 1 and 0 <= h[jp] + δ <= 1

    # So new h[j] = h[j] - δ, new h[jp] = h[jp] + δ
    # Constraints: 0 <= h[j] - δ and h[jp] + δ <= 1
    # => δ <= h[j] and δ >= -(1 - h[jp])... wait
    # δ can be negative (meaning we swap the other way)

    # 0 <= h[jp] + δ => δ >= -h[jp]
    # h[j] - δ <= 1 => δ >= h[j] - 1
    # h[jp] + δ <= 1 => δ <= 1 - h[jp]
    
    delta_min = max(-h[jp], h[j] - 1.0)
    delta_max = min(h[j], 1.0 - h[jp])

        return h.copy(), 0.0

    # This is convex in δ, piecewise linear
    # Find minimum by checking all breakpoints
    
    # Breakpoints occur where base[k] + δ*delta[k] = base[k'] + δ*delta[k']

    candidates = [delta_min, delta_max]
    
    # Add breakpoints
    for i in range(len(base)):
        for j2 in range(i+1, len(base)):
            if abs(delta[i] - delta[j2]) > 1e-12:
                bp = (base[j2] - base[i]) / (delta[j2] - delta[i])
                if delta_min <= bp <= delta_max:
                    candidates.append(bp)
    
    # Evaluate at all candidates

        if f < best_f:

    new_h[j] -= best_delta
    new_h[jp] += best_delta

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_subgrad(h, corr, tol):
    n = len(h)
    max_c = np.max(corr)
    active = np.where(corr >= max_c - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    g = np.zeros(n)
    for k in active:

        idx1 = np.arange(n) + shift
        v1 = (idx1 >= 0) & (idx1 < n)
        g[v1] += (1.0 - h[idx1[v1]])
        idx2 = np.arange(n) - shift
        v2 = (idx2 >= 0) & (idx2 < n)
        g[v2] -= h[idx2[v2]]
    g /= len(active)
    return g, active

def lp_guided_step(h, damping=0.5):
    n = len(h)
    y = 1.0 - h
    
    # LP: minimize t subject to sum_i h_i * y_{i+shift_k} <= t, bounds, sum constraint
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        shift = k - (n-1)
        A_ub[k, -1] = -1.0
        for i in range(n):
            s = i + shift
            if 0 <= s < n:
                A_ub[k, i] = y[s]
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, 1)]*n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=[n/2.0], bounds=bounds, method='highs')
    
    if res.status == 0:
        h_lp = res.x[:n]
        # Damped step
        h_new = (1 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new)
        return h_new
    return h

from scipy.optimize import linprog
import time

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def subgrad_step(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    tol = max(0.001, 0.01 * max_c)

    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    g = np.zeros(n)

        shift = k - (n - 1)
        idx1 = np.arange(n) + shift

        g[v1] += (1.0 - h[idx1[v1]])
        idx2 = np.arange(n) - shift
        v2 = (idx2 >= 0) & (idx2 < n)
        g[v2] -= h[idx2[v2]]
    g /= len(active)
    return g, max_c
