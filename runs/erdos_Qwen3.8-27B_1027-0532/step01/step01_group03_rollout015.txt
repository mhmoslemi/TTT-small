from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha, top_m=None):
    if top_m is not None and top_m < len(x):
        # Use top-m sum as approximation
        idx = np.argpartition(x, -top_m)[-top_m:]
        x_sel = x[idx]
    else:
        x_sel = x
    m = np.max(x_sel)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x_sel - m)))) / alpha

def _smooth_obj(h, alpha, top_m=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha, top_m) * (2.0 / len(h))

def _smooth_obj_grad(h, alpha, top_m=None):
    # Numerical gradient (expensive but reliable)
    eps = 1e-7
    n = len(h)
    grad = np.zeros(n)
    f0 = _smooth_obj(h, alpha, top_m)
    for i in range(n):
        h[i] += eps
        f1 = _smooth_obj(h, alpha, top_m)
        h[i] -= eps
        grad[i] = (f1 - f0) / eps
    return grad

    corr = np.correlate(h, one_minus_h, mode='full')

    weights = exps / exps.sum()

    # For each shift k (from -(N-1) to N-1), compute dc(k)/dh

from scipy.optimize import minimize

    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    Z = exps.sum()
    return (m + np.log(Z) / alpha) * (2.0 / N)

def _corr_lse_grad(h, alpha):
    N = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    Z = exps.sum()
    w = exps / Z
    LSE = (m + np.log(Z) / alpha) * (2.0 / N)
    
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        w_i = w[i]
        if w_i < 1e-15:
            continue

        if s >= 0:
            if N - s > 0:
                grad[s:] += w_i * one_minus_h[:N - s]
                grad[:N - s] -= w_i * h[s:]
        else:
            sp = -s
            if N - sp > 0:
                grad[:N - sp] += w_i * one_minus_h[sp:]
                grad[sp:] -= w_i * h[:N - sp]
    grad *= (2.0 / N)

        return _corr_lse(h, alpha)
    def obj_grad(h):
        _, g = _corr_lse_grad(h, alpha)
        return g
    return obj, obj_grad

    res = minimize(obj, h0, jac=obj_grad, method='SLSQP',

             'hess': _cons_hess}]

        res = minimize(_obj, h0, args=(alpha,), jac=_grad, method='trust-constr',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'xtol': 1e-12})
        return res.x
    except Exception:

def _cons_hess(x, v):
    return np.ones(len(x)) * (v @ np.ones(len(x)))  # v^T * J^T * J * v where J = ones^T
    # Actually for a linear constraint g(x) = x.sum() - c, Hessian of the Lagrangian is 0
    # But trust-constr might need it. Let's just return zeros.
    return np.zeros(len(x))
