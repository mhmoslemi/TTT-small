    dx = 2.0 / n
    h = h0.copy()
    t = compute_c5(h)  # current upper bound
    bundle = []  # list of (grad, g_value, h_point)
    step = initial_step
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr) * dx
        
        if max_val < t:
            t = max_val
        
        # Find active constraints (within tolerance of max)
        active_tol = 1e-4 * n / dx  # tolerance in correlation units
        active = np.where(corr > corr.max() - active_tol / dx)[0]
        
        # Compute gradient of a weighted combination of active constraints
        weights = corr[active] - corr.max() + 1.0  # positive weights, larger for more active
        weights /= weights.sum()
        
        grad_combo = np.zeros(n)
        for k in active:
            shift = k - (n - 1)
            g = np.zeros(n)
            for i in range(n):
                j = i + shift
                if 0 <= j < n:
                    if i == j:
                        g[i] += 1.0 - 2.0 * h[i]
                    else:
                        g[i] += (1.0 - h[j])
                        g[j] += -h[i]
            grad_combo += weights[np.where(active == k)[0][0]] * g

            t = new_val
            h = h_new
            step *= 0.9
        else:
            step *= 0.5
            if step < 1e-8:
                break
    
    return h, t

        else:
            mask = h > 0.0
