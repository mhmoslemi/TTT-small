def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _logsumexp(corr, alpha):
    m = np.max(corr)
    if m < -500:
        return m, np.zeros_like(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    val = m + np.log(S) / alpha
    weights = exp_vals / S
    return val, weights

def _smooth_obj_and_weights(h, alpha):
    corr = _correlations(h)
    val, weights = _logsumexp(corr, alpha)
    return val * (2.0/len(h)), weights, corr

def _gradient(h, weights, corr):
    """Compute gradient of log-sum-exp of correlations with respect to h."""
    n = len(h)
    dx = 2.0 / n
    p = n - 1  # shift for np.correlate full mode
    grad = np.zeros(n)
    
    for j in range(n):
        # First term: h[j] appears as the first factor in C[k] for k in [j, j+p]
        # Contribution: (1-h[j-k+p]) for k in [j, min(j+p, 2*n-2)]
        k_start = j
        k_end = min(j + p, 2*n-2)
        if k_start <= k_end:
            # h[j-k+p] for k in [k_start, k_end]
            # j-k+p ranges from j-k_end+p to j-k_start+p
            # = j-(j+p)+p to j-j+p = 0 to p = n-1
            # So we're looking at h[0] through h[n-1], which is the full range
            idx = j - np.arange(k_start, k_end+1) + p
            grad[j] += np.sum(weights[k_start:k_end+1] * (1 - h[idx]))
        
        # Second term: h[j] appears as the second factor (1-h[j]) in C[k]
        # h[m-k+p] = j means m = j+k-p
        # Valid when m in [max(0, k-p), min(n-1, k)]
        # k in [p-j, 2n-2-j] = [n-1-j, 2n-2-j]
        k_start2 = p - j
        k_end2 = min(2*n-2-j, 2*n-2)
        if k_start2 <= k_end2:
            # h[j+k-p] for k in [k_start2, k_end2]
            # j+k-p ranges from j+p-j-p to j+k_end2-p = 0 to j+k_end2-p
            # = 0 to j + 2n-2-j-p = 2n-2-p = n-1
            idx2 = j + np.arange(k_start2, k_end2+1) - p
            grad[j] -= np.sum(weights[k_start2:k_end2+1] * h[idx2])
    
    return grad * dx
