```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_all(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * (2.0 / len(h))

def _project(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    h = h.copy()
    # Binary search for lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining error
    diff = target_sum - p.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - p)
        else:
            order = np.argsort(p)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - p[i])
                p[i] += add
                diff -= add
            else:
                sub = min(-diff, p[i])
                p[i] -= sub
                diff += sub
    return p

def _subgrad_step(h, n, target_sum, step_size):
    """One subgradient step on the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    m = k_star - (n - 1)  # actual lag
    
    # Compute subgradient
    g = np.zeros(n)
    if m == 0:
        g = 1.0 - 2.0 * h
    elif m > 0:
        # c_m = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
        for j in range(n - m):
            g[j] += (1.0 - h[j + m])
        for j in range(m, n):
            g[j] -= h[j - m]
    else:
        d = -m
        # c_{-d} = sum_{j=d}^{n-1} h[j]*(1-h[j-d])
        for j in range(d, n):
            g[j] += (1.0 - h[j - d])
        for j in range(0, n - d):
            g[j] -= h[j + d]
    
    dx = 2.0 / n
    h_new = h - step_size * dx * g
    h_new = _project(h_new, target_sum)
    return h_new, k_star, corr[k_star] * dx

def _subgradient_method(h_start, n, target_sum, max_iter, step_start=0.1, step_end=0.001):
    h = h_start.copy()
    best_h = h_start.copy()
    best_c5 = _c5_of(h_start)
    
    for it in range(max_iter):
        # Decay step size
        t = it / max(max_iter - 1, 1)
        step = step_start * (step_end / step_start) ** t
        
        h_new, _, _ = _subgrad_step(h, n, target_sum, step)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def _binary_local_search(h_start, n, target_sum, max_iters=5000):

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha * (2.0 / len(h))

def _do_slsqp(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        weights = np.exp(alpha * (corr - m))
        weights /= np.sum(weights)
        # d/dx of sum_k weights_k * corr_k * dx
        # This is complex; let SLSQP approximate it
        return None
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_start, args=(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def _fix_sum(h, target_sum):

    best_h = h_start.copy()
    best_c5 = _c5_of(h_start)
    
    total_iters = sum(p[0] for p in phases)
    global_it = 0

            t = (global_it) / max(total_iters - 1, 1)

            global_it += 1
```

```python
def _compute_grad_smoothed(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute weights
    m_max = np.max(corr)
    weights = np.exp(alpha * (corr - m_max))
    weights /= np.sum(weights)
    
    # Only keep significant weights
    threshold = np.max(weights) * 1e-6
    active = np.where(weights > threshold)[0]
    
    # Compute gradient
    grad = np.zeros(n)
    for k in active:
        p_k = weights[k]
        m = k - (n - 1)
        if m == 0:
            grad += p_k * (1.0 - 2.0 * h)
        elif m > 0:
            for j in range(n - m):
                grad[j] += p_k * (1.0 - h[j + m])
            for j in range(m, n):
                grad[j] -= p_k * h[j - m]
        else:
            d = -m
            for j in range(d, n):
                grad[j] += p_k * (1.0 - h[j - d])
            for j in range(n - d):
                grad[j] -= p_k * h[j + d]
    
    return grad * dx
```

```python
    # Then for h[j] = x: c_k(x) = c_k(h) + a_k * (x - h_j) + (correction for m=0)
    # For m=0: c_0 = h[j]*(1-h[j]) + (other terms) so the quadratic term matters
    # c_0(x) = x*(1-x) + (c_0(h) - h_j*(1-h_j))
    
    # For simplicity, let's ignore the quadratic correction for m=0
    # (it's usually not the binding constraint)
    
    # First compute current correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    # Compute a_k for each k
    nk = 2 * n - 1
    a = np.zeros(nk)
    for k in range(nk):
        m = k - (n - 1)
        if m == 0:
            a[k] = 1.0 - 2.0 * h_j
        elif m > 0:
            val = 0.0
            if j + m < n:
                val += (1.0 - h[j + m])
            if j - m >= 0:
                val -= h[j - m]
            a[k] = val
        else:
            d = -m
            val = 0.0
            if j - d >= 0:
                val += (1.0 - h[j - d])
            if j + d < n:
                val -= h[j + d]
            a[k] = val
    
    # Now c_k(x) ≈ corr[k] + a[k] * (x - h_j) for all k
    # (except k=n-1 which has the quadratic term, but we'll ignore it)
    # We want to minimize max_k (corr[k] + a[k] * (x - h_j)) * dx over x in [0,1]
    
    # This is minimizing max_k (a[k] * x + (corr[k] - a[k]*h_j)) * dx
    # = minimizing max_k (a[k] * x + b[k]) * dx where b[k] = corr[k] - a[k]*h_j
    
    b = corr - a * h_j
    
    # Find x in [0,1] minimizing max_k (a[k]*x + b[k])
    # Since this is convex in x, use binary search on the subgradient
    
    def max_val(x):
        return np.max(a * x + b)
    
    # Subgradient at x: any a[k] where k is argmax
    # At minimum, 0 should be in the subgradient
    # i.e., there should be active k's with a[k] > 0 and a[k] < 0 (or a[k]=0)
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        k_max = np.argmax(vals)
        # Right subgradient
        grad_right = a[k_max]
        if grad_right > 0:
            hi = mid  # We're to the left of minimum, increase x
        else:
            lo = mid  # We're to the right of minimum, decrease x
    
    x_opt = (lo + hi) / 2.0
    
    # Also check endpoints
    candidates = [0.0, 1.0, x_opt]
    best_x = min(candidates, key=lambda x: max_val(x))
    
    return best_x
```
