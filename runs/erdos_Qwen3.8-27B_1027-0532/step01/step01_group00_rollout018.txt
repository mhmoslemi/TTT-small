Let me try a different approach: **Frank-Wolfe (Conditional Gradient) Method**.

The Frank-Wolfe method for minimizing f(x) over a convex set P:
1. Start at x_0 ∈ P
2. At each iteration:
   a. Compute s = argmin_{y ∈ P} ∇f(x) · y (linear minimization over P)
   b. Update x ← x + γ(s - x) where γ = 2/(t+2) or via line search

For our problem, P = {h : 0 ≤ h ≤ 1, sum(h) = n/2}.

The linear minimization: minimize ∇f(x) · y over y ∈ P.
This is: minimize sum_i g_i * y_i s.t. 0 ≤ y_i ≤ 1, sum(y) = n/2.

Where g = ∇f(x). The solution is: set y_i = 1 for the n/2 indices with smallest g_i, and y_i = 0 for the rest. (If n/2 is not integer, one coordinate gets a fractional value.)

- Sort g in increasing order: g_σ(1) ≤ g_σ(2) ≤ ... ≤ g_σ(n)
- Set y_{σ(1)} = ... = y_{σ(212)} = 1
- Set y_{σ(213)} = 0.5
- Set y_{σ(214)} = ... = y_{σ(425)} = 0

This is O(n log n) per iteration (for the sort).

The Frank-Wolfe method is great because:

                B[i] = corr[k] - gi * h[j]
            candidates = [0.0, 1.0, h[j]]
            for i in range(len(top_idx)):
                for j2 in range(i + 1, len(top_idx)):
                    denom = A[i] - A[j2]
                    if abs(denom) > 1e-12:
                        x = (B[j2] - B[i]) / denom
                        if -0.01 <= x <= 1.01:
                            candidates.append(np.clip(x, 0.0, 1.0))
            best_x = h[j]
            best_mv = max_val
            for x in candidates:
                h[j] = x
                h_proj = project_feasible(h, target)
                mv = compute_corr(h_proj).max()
                if mv < best_mv - 1e-13:
                    best_mv = mv
                    best_x = x
                    h[:] = h_proj
            h[j] = best_x
            if best_x != h[j]:
                h = project_feasible(h, target)
    return h


def binary_flip_phase(h, target, n, t0, deadline, rng, max_flips):
    for _ in range(max_flips):
        if time.time() > deadline:
            break
        j = rng.randint(n)
        for val in [0.0, 1.0]:
            h_old = h[j]
            h[j] = val
            h_proj = project_feasible(h, target)
            s = evaluate_score(h_proj)
            c_curr = evaluate_score(h)
            if s < c_curr - 1e-12:
                h = h_proj
            else:
                h[j] = h_old
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    best_h = project_feasible(init, target)
    best_score = evaluate_score(best_h)

    deadline = t0 + budget_s - 30

    # Phase 1: PGD with momentum
    h = best_h.copy()
    h = pgd_phase(h, target, n, t0, deadline, rng, 30000)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 2: Coordinate descent
    h = best_h.copy()
    h = coordinate_phase(h, target, n, t0, deadline, 200)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 3: More PGD
    h = best_h.copy()
    h = pgd_phase(h, target, n, t0, deadline, rng, 20000)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 4: Binary flips
    h = best_h.copy()
    h = binary_flip_phase(h, target, n, t0, deadline, rng, 5000)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 5: Final coordinate refinement
    h = best_h.copy()
    h = coordinate_phase(h, target, n, t0, deadline, 500)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 6: Final PGD polish
    h = best_h.copy()
    h = pgd_phase(h, target, n, t0, deadline, rng, 5000)
    s = evaluate_score(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()

    # Phase 7: Try higher resolution
    if time.time() < deadline - 60:
        for n_try in [2 * n, 3 * n]:
            if n_try > 3000 or time.time() > deadline - 30:
                continue
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_feasible(h_init, n_try / 2.0)
            h_tmp = pgd_phase(h_init, n_try / 2.0, n_try, t0, deadline, rng, 10000)
            s_tmp = evaluate_score(h_tmp)
            if s_tmp < best_score:
                best_score = s_tmp
                best_h = h_tmp.copy()
            h_tmp2 = coordinate_phase(h_tmp, n_try / 2.0, n_try, t0, deadline, 100)
            s_tmp2 = evaluate_score(h_tmp2)
            if s_tmp2 < best_score:
                best_score = s_tmp2
                best_h = h_tmp2.copy()

    # Final projection to ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
