import time

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0

        projected = np.clip(h - mid, 0.0, 1.0)

        else:
            hi = mid

def grad_at_lag(h, n, lag):
    """Compute gradient of correlation at a specific lag."""
    grad = np.zeros(n)
    dx = 2.0 / n
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_at_lag(h, n, lag):
    grad = np.zeros(n)
    dx = 2.0 / n
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx


def softmax_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.maximum(shifted, -500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    grad *= dx
    return obj, grad


def slsqp_optimize(h0, n, t, max_iter=300):
    bounds = [(0.0, 1.0)] * n

    def obj_fun(h):
        return softmax_obj_and_grad(h, n, t)[0]

        return softmax_obj_and_grad(h, n, t)[1]

    def con_fun(h):
        return np.sum(h) - n / 2.0

    def con_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        result = minimize(obj_fun, h0.copy(), method='SLSQP', jac=grad_fun,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

    h = h0.copy().astype(float)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    step = step_scale
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad += grad_at_lag(h, n, lag)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.997
    return best_h, best_c5
