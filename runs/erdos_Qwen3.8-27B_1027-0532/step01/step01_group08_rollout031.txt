import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_lag_gradient(h, lag, n, dx):
    """Compute gradient of f_lag(h) = sum h[i]*h[i+lag]*dx (the quadratic part)"""
    g = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
    j2_lo = max(0, -lag)
    j2_hi = min(n - 1, n - 1 - lag)
    if j2_lo <= j2_hi:
        g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return g * dx

def soft_max_obj_and_grad(h, tau, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

    """Adam optimizer for the soft-max approximation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    tau = tau0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        tau = min(tau0 + (tau_max - tau0) * t / max_iter, tau_max)
        
        _, g = soft_max_obj_and_grad(h, tau, n, dx)
        
        m_hat = beta1 * m_hat + (1 - beta1) * g

def slsqp_optimize(h0, n, target_sum, deadline, tau=80.0):

        o, _ = soft_max_obj_and_grad(x, tau, n, 2.0/n)
        return o

        _, g = soft_max_obj_and_grad(x, tau, n, 2.0/n)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
        if hasattr(res, 'x'):

            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    shifted = tau_corr - max_tc
    log_sum_exp = max_tc + np.log(np.sum(np.exp(shifted)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def top_m_grad(h, n, dx, top_m=5):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[-top_m:]
    g = np.zeros(n)
    for m_idx in top_indices:
        lag = m_idx - (n - 1)
        j_lo = max(0, lag)

        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return g * dx / top_m

def nesterov_pgd(h0, n, target_sum, deadline, max_iter=5000, top_m=5, lr=0.04):

    best_h = h.copy()

    dx = 2.0 / n
    mu = 0.9
    lr_cur = lr

    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + mu * v
        g = top_m_grad(y, n, dx, top_m)
        v_new = mu * v - lr_cur * g
        h_new = y + v_new
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:

            best_h = h_new.copy()

        h = h_new

    return best_h, best_c5

def adam_optimize(h0, n, target_sum, deadline, max_iter=4000, tau0=50.0, tau_max=400.0):

        pass
