import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def lse_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    return dx * (cmax + T * np.log(np.sum(np.exp(scaled))))

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def try_slsqp(h, T, max_iter):
    n = len(h)
    try:
        res = minimize(lambda x: lse_obj(x, T), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)

def eval_move(h, i, j, delta, n):
    """Evaluate objective after moving delta from h[i] to h[j]"""
    trial = h.copy()
    trial[i] -= delta
    trial[j] += delta
    # Check bounds
    if trial[i] < -1e-12 or trial[j] > 1.0 + 1e-12:
        return 1e10
    trial[i] = max(0.0, trial[i])
    trial[j] = min(1.0, trial[j])
    # If clipping changed the sum, project
    if abs(np.sum(trial) - n/2.0) > 1e-10:
        trial = project_feasible(trial, n)
    return score_c5(trial)

    best = h.copy()
    best_score = score_c5(best)
    no_improve = 0
    max_no_improve = 5
    
    while time.time() - t_start < time_limit and no_improve < max_no_improve:
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            if best[i] < 1e-9:
                continue

                if i == j:

                if time.time() - t_start > time_limit:
                    break

                    continue

                max_d = min(best[i], 1.0 - best[j])
                if max_d < 1e-9:

                # Ternary search for convex function
                a = 0.0
                b = max_d
                for _ in range(20):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    v1 = eval_move(best, i, j, m1, n)
                    v2 = eval_move(best, i, j, m2, n)
                    if v1 < v2:
                        b = m2
                    else:
                        a = m1
                delta = (a + b) / 2.0
                if delta < 1e-9:

                v = eval_move(best, i, j, delta, n)
                if v < best_score - 1e-13:
                    best[i] -= delta
                    best[j] += delta
                    if best[i] < 0: best[i] = 0

                    best_score = v

                    break  # Restart the j loop
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best, best_score

corr_lag(δ) = sum_m [h[m] + δ*(1[m=j] - 1[m=i])] * [1 - h[m+lag] - δ*(1[m+lag=j] - 1[m+lag=i])]

= sum_m h[m]*(1-h[m+lag]) 
  + δ * sum_m (1[m=j]-1[m=i])*(1-h[m+lag])
  - δ * sum_m h[m]*(1[m+lag=j]-1[m+lag=i])
  - δ² * sum_m (1[m=j]-1[m=i])*(1[m+lag=j]-1[m+lag=i])

The last term is:
-δ² * [1[j+lag=j] - 1[j+lag=i] - 1[i+lag=j] + 1[i+lag=i]]

Since j+lag=j only if lag=0, and i+lag=i only if lag=0:
For lag=0: -δ² * [1 - 1 - 1 + 1] = 0 (since j≠i)
For lag≠0: -δ² * [0 - 1[j+lag=i] - 1[i+lag=j] + 0] = -δ² * [-1[j+lag=i] - 1[i+lag=j]]

Since j+lag=i means i-j=lag, and i+lag=j means j-i=lag, these can't both be true (since i≠j). At most one is true.

So the δ² term is: 0 if neither i-j=lag nor j-i=lag, and -δ² if one of them is true.

So corr_lag(δ) is:
- Linear in δ for most lags
- Quadratic (with negative δ² coefficient) for the specific lag where |i-j| = |lag|

Therefore, max_lag corr_lag(δ) is the max of functions that are mostly linear with at most one or two being quadratic (concave, since the δ² coefficient is negative).

The max of concave functions is not necessarily convex or concave. So ternary search might not work perfectly!

However, in practice, the quadratic term is usually small (it's -δ², and δ is small), so the function is "nearly" convex. Ternary search should still work well in most cases.

To be safe, I'll use a hybrid approach:
1. First, do a coarse grid search (e.g., 10 points) to find the approximate minimum
2. Then, refine with ternary search around the best point

This should be robust even if the function isn't perfectly unimodal.

import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def lse_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    return dx * (cmax + T * np.log(np.sum(np.exp(scaled))))

def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def try_slsqp(h, T, max_iter):
    n = len(h)
    def objective(x):
        return lse_obj(x, T)
    try:
        res = minimize(objective, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)
