```python
    """Given current h and corr, find optimal h[j] by 1D optimization."""
    n = len(h)
    t = h[j]

    # and the intercept (value at t=0)
    
    # corr[m] for lag k = m - (n-1):
    # If k >= 0: corr = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
    # If k < 0, let j_lag = -k: corr = sum_{i=0}^{n-1-j_lag} h[i] * (1-h[i+j_lag])
    
    # For coordinate j, the change in corr[m] when h[j] changes:
    # We need to find all m where h[j] appears.
    
    # Let me just compute the slope numerically:
    # slope_m = corr_m(t+eps) - corr_m(t-eps) / (2*eps)
    # But this is O(n) per m, and O(n) m's, so O(n²) total. For n=331, that's ~110k. Fine.
    
    # Actually, let me compute it analytically.
    # For m corresponding to lag k:
    # If k = 0: corr[m] = sum h[i](1-h[i]), d/dh[j] = 1 - 2*h[j] = 1 - 2t (but this is the derivative, and the function is h[j] - h[j]², so it's nonlinear)
    # If k > 0: 
    #   h[j] appears as h[i+k] when i = j-k (valid if 0 <= j-k <= n-1-k, i.e., k <= j <= n-1)
    #     slope contribution: (1 - h[j-k])
    #   h[j] appears as (1-h[i]) when i = j (valid if j <= n-1-k, i.e., k <= n-1-j)
    #     slope contribution: -h[j+k]
    # If k < 0, let l = -k > 0:
    #   corr = sum_{i=0}^{n-1-l} h[i] * (1-h[i+l])
    #   h[j] appears as h[i] when i = j (valid if j <= n-1-l, i.e., l <= n-1-j)
    #     slope contribution: (1 - h[j+l])
    #   h[j] appears as (1-h[i+l]) when i+l = j, i.e., i = j-l (valid if 0 <= j-l, i.e., l <= j)
    #     slope contribution: -h[j-l]
    
    # So the slopes are:
    slopes = np.zeros(2*n - 1)  # for each output index m
    for m in range(2*n - 1):
        k = m - (n - 1)
        if k == 0:
            # Nonlinear: h[j] - h[j]^2, slope = 1 - 2*t
            slopes[m] = 1.0 - 2.0 * t
        elif k > 0:
            s = 0.0
            if k <= j and j <= n - 1:  # h[j] appears as h[i+k]
                s += (1.0 - h[j - k])
            if k <= n - 1 - j:  # h[j] appears as (1-h[i])
                s += (-h[j + k])
            slopes[m] = s
        else:
            l = -k
            s = 0.0
            if l <= n - 1 - j:  # h[j] appears as h[i]
                s += (1.0 - h[j + l])
            if l <= j:  # h[j] appears as (1-h[i+l])
                s += (-h[j - l])
            slopes[m] = s
    
    # Now corr[m] as a function of t:
    # For k=0: corr[center](t) = corr[center]_current + (t - t_old) * (1 - 2*t) - (t-t_old)² ... no
    # Actually: corr[center] = sum h[i](1-h[i]) = (sum_{i≠j} h[i](1-h[i])) + t(1-t)
    # So corr[center](t) = C0 + t - t² where C0 = sum_{i≠j} h[i](1-h[i])
    # The other corr[m](t) = corr[m]_current + slopes[m] * (t - t_old) [linear]
    
    # For the 1D minimization of max_m corr[m](t) over t in [0,1]:
    # Ignore the k=0 term (handle separately)
    # For the linear terms: min of max of linear functions = convex PL minimization
    
    # Let's just do a fine grid search for simplicity (331 points is fast)
    # Actually, let's use the binary search approach for the convex part
    
    # Simpler: evaluate f(t) at a set of candidate points
    # Candidate points: 0, 1, and intersections of pairs of lines
    # But that's O(n²) intersections. Instead, use the fact that the minimum of
    # a convex PL function is where the slope changes sign.
    
    # For now, let's use a simple approach: ternary search (works for unimodal)
    # Since f(t) = max(linear terms) + max(0, quadratic term - linear terms) is not exactly
    # unimodal, let's just use a fine grid.
    
    # Fine grid: 1000 points
    t_grid = np.linspace(0, 1, 1000)
    best_t = t
    best_val = np.inf
    
    C0 = np.sum(h * (1 - h)) - t * (1 - t)  # sum of h[i](1-h[i]) for i ≠ j
    
    for t_val in t_grid[::10]:  # use 100 points for speed
        # Compute all corr[m](t_val)
        vals = corr.copy().astype(float)
        # For linear terms (m ≠ center):
        center = n - 1
        linear_mask = np.arange(2*n-1) != center
        vals[linear_mask] = corr[linear_mask] + slopes[linear_mask] * (t_val - t)
        # For center:
        vals[center] = C0 + t_val * (1 - t_val)
        
        max_val = np.max(vals)
        if max_val < best_val:
            best_val = max_val
            best_t = t_val
    
    return best_t
```

1. Sort the lines by slope

Algorithm:
- Start at t=0. Find the active line (the one with the max value). Its slope is a*.
- If a* ≥ 0, the minimum is at t=0 (the function is increasing from the start).
- If a* < 0, the function is decreasing at t=0. Find the next breakpoint (intersection with another line). Move to that breakpoint.
- Repeat until the active slope ≥ 0 or t > 1.

This is essentially walking along the upper envelope. In the worst case, it visits O(n) breakpoints, each taking O(n) to find the next one. Total: O(n²) per coordinate.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def obj_topk(h, k, alpha):
    """Penalize the top-k correlations using a soft threshold."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Soft top-k: weight = sigmoid(alpha * (corr - c_max + k/len(corr)))
    # This gives ~full weight to the top k and ~zero to the rest
    frac = k / len(corr)
    weights = 1.0 / (1.0 + np.exp(-alpha * (corr - c_max + frac)))
    return float(np.sum(weights * corr) / np.sum(weights))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n_iter=10, n_grid=500):
    """Coordinate descent for the Erdős overlap problem."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    
    for it in range(n_iter):
        for j in range(n):
            t = h[j]
            
            # Compute slopes for each output index
            slopes = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = m - center
                if k == 0:
                    slopes[m] = 1.0 - 2.0 * t
                elif k > 0:
                    s = 0.0
                    if k <= j:
                        s += (1.0 - h[j - k])
                    if k <= n - 1 - j:
                        s += (-h[j + k])
                    slopes[m] = s
                else:
                    l = -k
                    s = 0.0
                    if l <= n - 1 - j:
                        s += (1.0 - h[j + l])
                    if l <= j:
                        s += (-h[j - l])
                    slopes[m] = s
            
            # C0 = sum of h[i](1-h[i]) for i ≠ j
            C0 = float(np.sum(h * (1.0 - h)) - t * (1.0 - t))
            
            # Evaluate f(t) on a grid
            t_grid = np.linspace(0.0, 1.0, n_grid)
            best_t = t
            best_val = np.inf
            
            for t_val in t_grid:
                if t_val == t:
                    continue
                # Linear terms
                vals = corr + slopes * (t_val - t)
                # Fix the center (quadratic) term
                vals[center] = C0 + t_val * (1.0 - t_val)
                max_val = np.max(vals)
                if max_val < best_val:
                    best_val = max_val
                    best_t = t_val
            
            # Update
            if best_t != t:
                corr += slopes * (best_t - t)
                corr[center] = C0 + best_t * (1.0 - best_t)
                h[j] = best_t
    
    return h
```

```python
def coord_descent(h, n_passes=10, n_grid=300, random_order=False, rng=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    center = n - 1
    m_range = np.arange(2 * n - 1)
    k_vals = m_range - center
    
    # Precompute masks for efficiency
    pos_k = k_vals[k_vals > 0]
    pos_idx = np.where(k_vals > 0)[0]
    neg_k = -k_vals[k_vals < 0]
    neg_idx = np.where(k_vals < 0)[0]
    
    for it in range(n_passes):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for j in order:
            t = h[j]
            
            slopes = np.zeros(2 * n - 1)
            
            # Positive k contributions
            # h[j] as h[i+k] when i = j-k, valid if 0 <= j-k and j-k <= n-1-k, i.e., k <= j
            mask1 = pos_k <= j
            if np.any(mask1):
                idx = pos_idx[mask1]
                ks = pos_k[mask1]
                slopes[idx] += (1.0 - h[j - ks])
            
            # h[j] as (1-h[i]) when i = j, valid if j <= n-1-k, i.e., k <= n-1-j
            mask2 = pos_k <= (n - 1 - j)
            if np.any(mask2):
                idx = pos_idx[mask2]
                ks = pos_k[mask2]
                slopes[idx] += (-h[j + ks])
            
            # Negative k contributions (l = -k > 0)
            # h[j] as h[i] when i = j, valid if j <= n-1-l, i.e., l <= n-1-j
            mask3 = neg_k <= (n - 1 - j)
            if np.any(mask3):
                idx = neg_idx[mask3]
                ls = neg_k[mask3]
                slopes[idx] += (1.0 - h[j + ls])
            
            # h[j] as (1-h[i+l]) when i+l = j, i.e., i = j-l, valid if j-l >= 0, i.e., l <= j

                idx = neg_idx[mask4]
                ls = neg_k[mask4]
                slopes[idx] += (-h[j - ls])

            C0 = float(np.sum(h * (1.0 - h)) - t * (1.0 - t))

            t_grid = np.linspace(0.0, 1.0, n_grid)
            deltas = t_grid[:, None] - t
            vals = corr[None, :] + slopes[None, :] * deltas
            vals[:, center] = C0 + t_grid * (1.0 - t_grid)
            
            max_vals = np.max(vals, axis=1)
            best_idx = np.argmin(max_vals)
            best_t = t_grid[best_idx]
            
            if abs(best_t - t) > 1e-10:
                corr += slopes * (best_t - t)
                corr[center] = C0 + best_t * (1.0 - best_t)
                h[j] = best_t
    
    return h
```

```python
_g_n = 0
_g_target = 0.0

def _con_eq(h):
    return np.sum(h) - _g_target

def _con_eq_jac(h):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def proj(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h < 1.0) if diff > 0 else (h > 0.0)
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n_grid=300, rng=None):
    """One full pass of coordinate descent."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    center = n - 1
    m_range = np.arange(2 * n - 1)
    k_vals = m_range - center
    
    pos_k = k_vals[k_vals > 0]
    pos_idx = np.where(k_vals > 0)[0]
    neg_k = -k_vals[k_vals < 0]
    neg_idx = np.where(k_vals < 0)[0]
    
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = range(n)
    
    for j in order:
        t = h[j]
        slopes = np.zeros(2 * n - 1)
        
        m1 = pos_k <= j
        if np.any(m1):
            slopes[pos_idx[m1]] += (1.0 - h[j - pos_k[m1]])
        
        m2 = pos_k <= (n - 1 - j)
        if np.any(m2):
            slopes[pos_idx[m2]] += (-h[j + pos_k[m2]])
        
        m3 = neg_k <= (n - 1 - j)
        if np.any(m3):
            slopes[neg_idx[m3]] += (1.0 - h[j + neg_k[m3]])
        
        m4 = neg_k <= j
        if np.any(m4):
            slopes[neg_idx[m4]] += (-h[j - neg_k[m4]])
        
        slopes[center] = 1.0 - 2.0 * t
        
        C0 = float(np.dot(h, 1.0 - h)) - t * (1.0 - t)
        
        t_grid = np.linspace(0.0, 1.0, n_grid)
        deltas = (t_grid - t)[:, None]
        vals = corr[None, :] + slopes[None, :] * deltas
        vals[:, center] = C0 + t_grid * (1.0 - t_grid)
        
        max_vals = np.max(vals, axis=1)
        bi = np.argmin(max_vals)
        best_t = t_grid[bi]
        
        if abs(best_t - t) > 1e-12:
            corr += slopes * (best_t - t)
            corr[center] = C0 + best_t * (1.0 - best_t)
            h[j] = best_t
    
    return h, corr

def coord_descent(h, n_passes=10, n_grid=300, rng=None):
    for _ in range(n_passes):
        h, _ = cd_pass(h, n_grid=n_grid, rng=rng)
    return h

def upscale(h, n_new):
    """Upscale h from its current length to n_new using interpolation."""
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, n_new / 2.0)
```

```python
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def _con_eq(h):
    return np.sum(h) - _g_target


def _con_eq_jac(h):
    return np.ones(_g_n)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def obj_topk(h, k, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    frac = float(k) / len(corr)
    weights = 1.0 / (1.0 + np.exp(-alpha * (corr - c_max + frac)))
    return float(np.sum(weights * corr) / (np.sum(weights) + 1e-30))


def proj(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def cd_single_pass(h, n_grid, rng=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    center = n - 1
    m_range = np.arange(2 * n - 1)
    k_vals = m_range - center

    pos_k = k_vals[k_vals > 0]
    pos_idx = np.where(k_vals > 0)[0]
    neg_k = -k_vals[k_vals < 0]
    neg_idx = np.where(k_vals < 0)[0]

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for j in order:
        t = h[j]
        slopes = np.zeros(2 * n - 1)

        m1 = pos_k <= j
        if np.any(m1):
            slopes[pos_idx[m1]] += (1.0 - h[j - pos_k[m1]])

        m2 = pos_k <= (n - 1 - j)
        if np.any(m2):
            slopes[pos_idx[m2]] += (-h[j + pos_k[m2]])

        m3 = neg_k <= (n - 1 - j)
        if np.any(m3):
            slopes[neg_idx[m3]] += (1.0 - h[j + neg_k[m3]])

        m4 = neg_k <= j
        if np.any(m4):
            slopes[neg_idx[m4]] += (-h[j - neg_k[m4]])

        slopes[center] = 1.0 - 2.0 * t
        C0 = float(np.dot(h, 1.0 - h)) - t * (1.0 - t)

        t_grid = np.linspace(0.0, 1.0, n_grid)
        deltas = (t_grid - t)[:, None]
        vals = corr[None, :] + slopes[None, :] * deltas
        vals[:, center] = C0 + t_grid * (1.0 - t_grid)

        max_vals = np.max(vals, axis=1)
        bi = np.argmin(max_vals)
        best_t = t_grid[bi]

        if abs(best_t - t) > 1e-14:
            corr += slopes * (best_t - t)
            corr[center] = C0 + best_t * (1.0 - best_t)
            h[j] = best_t

    return h


def coord_descent(h, n_passes, n_grid, rng=None):
    result = h.copy()
    for _ in range(n_passes):
        result = cd_single_pass(result, n_grid, rng=rng)
    return result


def cd_adaptive(h, base_grid, refine_grid, n_passes_base, n_passes_refine, rng=None):
    result = h.copy()
    for _ in range(n_passes_base):
        result = cd_single_pass(result, base_grid, rng=rng)
    for _ in range(n_passes_refine):
        result = cd_single_pass(result, refine_grid, rng=None)
    return result


def upscale(h, n_new, target_sum):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return proj(h_new, target_sum)
```
