def project_feasible(h, target_sum):

def golden_min(f_params, lo, hi, xatol=1e-8):
    # Golden section search for minimum of a 1D function
    # f is evaluated as golden_step(*f_params, x)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = golden_step(*f_params, c)
    fd = golden_step(*f_params, d)
    while abs(b - a) > xatol:
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = golden_step(*f_params, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = golden_step(*f_params, d)
    return (a + b) / 2.0
