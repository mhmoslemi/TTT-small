def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0

        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_all_gradients(h):
    n = len(h)
    all_grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        l = k - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + l < n:
                g += (1.0 - h[j + l])
            if 0 <= j - l < n:
                g += (-h[j - l])
            all_grads[k, j] = g
    return all_grads

def solve_lp_cutting_plane(h_current, n, num_active=30):
    raw_corr = np.correlate(h_current, 1 - h_current, mode="full")
    sorted_idx = np.argsort(raw_corr)[::-1]
    active = sorted_idx[:num_active]
    
    all_grads = compute_all_gradients(h_current)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in active:
        g = all_grads[k]
        rhs = -raw_corr[k] + np.dot(g, h_current)
        constraints.append(g @ h_var - t <= rhs)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None
    
    if h_var.value is not None and np.all(np.isfinite(h_var.value)):

    return None

            h_pert = best_h + scale * np.random.randn(n)
        else:
            block_len = np.random.randint(3, 15)
            start_idx = np.random.randint(0, n - block_len)
            h_pert = best_h.copy()
            h_pert[start_idx:start_idx + block_len] += (np.random.rand(block_len) - 0.5) * 0.3
        h_pert = project_feasible(h_pert, n)
        h_pert = projected_gradient_descent(h_pert, 200.0, n, 0.01, 30)
        res = minimize(
            obj_wrapper, h_pert.copy(), jac=grad_wrapper,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_func}],
            options={'maxiter': 250, 'ftol': 1e-13}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Coordinate descent
    for sweep in range(150):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 30:
            break
        improved = False
        for idx in range(n):
            for delta in [-0.04, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.04]:
                h_trial = best_h.copy()
                h_trial[idx] = np.clip(h_trial[idx] + delta, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5 - 1e-13:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    improved = True
                    break
        if not improved and sweep > 5:
            break

    # Phase 5: Final LP polish with increasing active sets
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        for num_active in [10, 30, 60, 100, 2 * n - 1]:
            if time.time() - start_time > budget_s - 20:
                break
            h_lp = solve_lp_cutting_plane(best_h, n, num_active)
            if h_lp is not None:
                h_lp = project_feasible(h_lp, n)
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    h_trial = (1.0 - alpha) * best_h + alpha * h_lp
                    h_trial = project_feasible(h_trial, n)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        break

    # Phase 6: Fine coordinate adjustments
    for idx in range(n):
        if time.time() - start_time > budget_s - 5:
            break
        for delta in [-0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01]:
            h_trial = best_h.copy()
            h_trial[idx] = np.clip(h_trial[idx] + delta, 0.0, 1.0)
            h_trial = project_feasible(h_trial, n)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                break

    # Phase 7: One final SLSQP polish with high epsilon
    if time.time() - start_time < budget_s - 30:
        _global_eps = 1000.0
        res = minimize(
            obj_wrapper, best_h.copy(), jac=grad_wrapper,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_func}],
            options={'maxiter': 300, 'ftol': 1e-14}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
