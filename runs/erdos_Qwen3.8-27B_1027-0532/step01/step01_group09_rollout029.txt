    corr = np.correlate(h, 1.0 - h, 'full') * dx
    m = corr.max()
    w = np.exp(tau * (corr - m))
    w = w / w.sum()  # softmax weights

        if w[k_idx] < 1e-10:

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def eval_c5(h):

    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()


def compute_weighted_gradient(h, w, dx, n):
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k == 0:
            grad += wk * dx * (1.0 - 2.0 * h)

    return grad


def smooth_objective(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    m = corr.max()
    shifted = np.clip(tau * (corr - m), -500, 500)
    s = np.exp(shifted).sum()

    if r == 0:
        h = rng.random(n)
    elif r == 1:
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.3 * np.sin(np.pi * x + rng.uniform(0, np.pi))
    elif r == 2:
        h = (rng.random(n) > 0.5).astype(float)
    elif r == 3:
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * x + rng.uniform(0, np.pi))
    elif r == 4:
        block = max(1, n // 4)
        vals = rng.uniform(0.2, 0.8, size=4)
        h = np.tile(vals, n // 4 + 1)[:n]
    else:
        x = np.linspace(0.0, 2.0, n)
        h = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(np.pi * x)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left(frac=0.9):
        return max(0.0, budget_s * frac - (time.time() - t_start))

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    best = project_feasible(h_init, len(h_init) / 2.0)
    best_c5 = eval_c5(best)
    n = len(best)
    dx = 2.0 / n

    # Phase 1: Annealed smooth-max SLSQP on initial solution
    taus = [4.0, 10.0, 25.0, 50.0, 100.0, 200.0, 400.0]
    for tau in taus:
        if time_left(0.25) < 5:
            break
        max_it = max(50, min(300, int(time_left(0.25) / 0.5)))
        h_new = selpolish_smooth(best, n, dx, tau, max_it, time_left(0.25) * 0.9)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best, best_c5 = h_new, c5_new

    # Phase 2: Primal-dual minimax at original resolution
    if time_left(0.55) > 20:
        h_pd = primal_dual_method(best, n, dx, time_left(0.55) * 0.8,
                                  max_iter=80, eta_init=3.0)
        c5_pd = eval_c5(h_pd)
        if c5_pd < best_c5:
            best, best_c5 = h_pd, c5_pd

    # Phase 3: Subgradient refinement
    if time_left(0.65) > 10:
        h_sg = subgrad_refine(best, n, dx, min(30, time_left(0.65) * 0.5),
                              lr_init=0.05, beta=0.92)
        c5_sg = eval_c5(h_sg)
        if c5_sg < best_c5:
            best, best_c5 = h_sg, c5_sg

    # Phase 4: Multi-resolution exploration
    n_targets = [300, 400, 500, 250, 350]
    for i, n_t in enumerate(n_targets):
        if time_left(0.85) < 15 or n_t == n:
            continue
        dx_t = 2.0 / n_t
        h_res = resample_h(best, n_t)
        h_div = make_diverse_init(n_t, rng, i + seed % 6)

        for h_guess in [h_res, h_div]:
            if time_left(0.85) < 10:
                break
            h_opt = selpolish_smooth(h_guess, n_t, dx_t, 30.0,
                                     max(30, int(time_left(0.85) / 2)), time_left(0.85) * 0.4)
            c5_opt = eval_c5(h_opt)
            if c5_opt < best_c5:
                best, best_c5 = h_opt, c5_opt
                n = len(best)
                dx = 2.0 / n

            if time_left(0.85) > 10:
                h_pd2 = primal_dual_method(h_opt, n_t, dx_t,
                                           min(30, time_left(0.85) * 0.4),
                                           max_iter=40, eta_init=5.0)
                c5_pd2 = eval_c5(h_pd2)
                if c5_pd2 < best_c5:
                    best, best_c5 = h_pd2, c5_pd2
                    n = len(best)
                    dx = 2.0 / n

    # Phase 5: Final intensive polish
    n_final = len(best)
    dx_final = 2.0 / n_final

    if time_left(0.98) > 20:
        for tau in [50.0, 150.0, 300.0]:
            if time_left(0.98) < 10:
                break
            h_f = selpolish_smooth(best, n_final, dx_final, tau,
                                   min(200, int(time_left(0.98) / 0.5)), time_left(0.98) * 0.6)
            cf = eval_c5(h_f)
            if cf < best_c5:
                best, best_c5 = h_f, cf

    if time_left(0.99) > 10:
        h_f2 = subgrad_refine(best, n_final, dx_final,
                              min(20, time_left(0.99) * 0.5), lr_init=0.03, beta=0.93)
        cf2 = eval_c5(h_f2)
        if cf2 < best_c5:
            best, best_c5 = h_f2, cf2

    if time_left(0.995) > 10:
        h_f3 = primal_dual_method(best, n_final, dx_final,
                                  min(20, time_left(0.995) * 0.5),
                                  max_iter=30, eta_init=10.0)
        cf3 = eval_c5(h_f3)
        if cf3 < best_c5:
            best, best_c5 = h_f3, cf2

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
