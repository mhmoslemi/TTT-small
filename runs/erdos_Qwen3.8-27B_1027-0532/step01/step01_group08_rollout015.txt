import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo-lag:j_hi-lag+1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+lag:j2_hi+lag+1])
    return grad * dx

def worst_lag_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] += (1.0 - h[j_lo-lag:j_hi-lag+1])
    j2_lo = max(0, -lag)
    j2_hi = min(n - 1, n - 1 - lag)
    if j2_lo <= j2_hi:
        grad[j2_lo:j2_hi+1] += (-h[j2_lo+lag:j2_hi+lag+1])
    return grad * dx

def multi_lag_grad(h, top_k=5):
    """Gradient weighted by top-k worst lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    weights = np.zeros(2*n-1)
    w_max = corr[idx[0]]
    for i in idx:
        weights[i] = (corr[i] / w_max) ** 4  # heavy weighting of worst
    weights /= weights.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo-lag:j_hi-lag+1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+lag:j2_hi+lag+1])
    return grad * dx

def pgd_optimize(h0, target_sum, deadline, max_iter=10000, mode="softmax", tau=50.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if mode == "softmax":
            g = soft_max_grad(h, tau)
        elif mode == "worst":
            g = worst_lag_grad(h)
        else:
            g = multi_lag_grad(h, top_k=3)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.01, 0.05)
        else:
            h = h_new
            stall += 1
            if stall > 30:
                lr *= 0.7
                if mode == "softmax":
                    tau *= 1.5
                stall = 0
                if lr < 0.001:
                    # restart with perturbation
                    h = best_h + 0.01 * np.random.randn(n)
                    h = np.clip(h, 0, 1)
                    h = project_onto_constraint(h, target_sum)
                    lr = 0.02
                    tau = 50.0
    return best_h, best_c5

def binary_round(h, target_sum):
    """Round h to binary and repair sum constraint"""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    deficit = target_sum - h_bin.sum()
    if abs(deficit) < 1e-10:
        return h_bin
    # Adjust: if deficit > 0, add 1s; if deficit < 0, remove 1s
    if deficit > 0:
        # Need to add int(deficit) ones - change 0s with highest h value to 1
        zeros = np.where(h_bin == 0)[0]
        n_add = int(round(deficit))
        if n_add > 0 and n_add <= len(zeros):
            top_zeros = zeros[np.argsort(h[zeros])[::-1]][:n_add]
            h_bin[top_zeros] = 1.0
    else:
        # Need to remove ones
        ones = np.where(h_bin == 1)[0]
        n_remove = int(round(-deficit))
        if n_remove > 0 and n_remove <= len(ones):
            bottom_ones = ones[np.argsort(h[ones])[:n_remove]]
            h_bin[bottom_ones] = 0.0
    return h_bin

def upscale(h_old, n_new):
    """Upscale h to new resolution"""
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

**Alternative: Subgradient method with clever step sizes**

The objective f(h) = max_k R_k(h) where R_k(h) = ∑h[i](1-h[i+k])*dx

This is a max of bilinear functions. The subgradient at h is the gradient of the active (worst) function(s).

A subgradient method:
h_{t+1} = P_C(h_t - α_t * g_t)

where P_C is projection onto the constraint set {h: 0≤h≤1, ∑h=target}, g_t is the subgradient (gradient of worst R_k), and α_t is the step size.

For max of functions, a good step size is:
α_t = (T² - ||h_t - h*||²) / ||g_t||²

But we don't know h*. A practical alternative: α_t = c / sqrt(t) or α_t = c / t.

**Alternative: Cutting plane / ellipsoid method**

Since f(h) = max_k R_k(h), we can add linear constraints R_k(h) ≤ t for the active k's and solve:
min t
s.t. R_k(h) ≤ t for k in K_active
     0 ≤ h ≤ 1
     ∑h = target

But R_k(h) is bilinear, not linear. So this isn't a LP.

However, at a given point h*, we can linearize:
R_k(h) ≈ R_k(h*) + ∇R_k(h*)^T (h - h*)

And solve the LP:
min t
s.t. R_k(h*) + ∇R_k(h*)^T (h - h*) ≤ t for k in K_active
     0 ≤ h ≤ 1
     ∑h = target

This gives a lower bound on the optimal value (since we're relaxing the non-convex constraint). The solution h_new of this LP can then be used as the next iterate.

This is essentially the cutting plane method applied to a non-convex problem, used as a heuristic.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_for_lag(h, lag, n, dx):
    """Gradient of R_lag(h) = sum_i h[i](1-h[i+lag]) * dx"""
    grad = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] += (1.0 - h[j_lo-lag:j_hi-lag+1])
    j2_lo = max(0, -lag)
    j2_hi = min(n - 1, n - 1 - lag)
    if j2_lo <= j2_hi:
        grad[j2_lo:j2_hi+1] += (-h[j2_lo+lag:j2_hi+lag+1])

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    tc = tau * corr
    mx = np.max(tc)

    corr = compute_corr(h)
    idx = np.argsort(corr)[::-1][:top_k]
    w = np.zeros(2 * n - 1)

def slsqp_wrapper(h, tau, target, n):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        grad[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
    j2_lo = max(0, -lag)
    j2_hi = min(n - 1, n - 1 - lag)
    if j2_lo <= j2_hi:
        grad[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return grad * dx

def weighted_lag_grad(h, corr, top_k, power, n, dx):
    idx = np.argsort(corr)[::-1][:top_k]
    cmax = max(corr[idx[0]], 1e-10)
    grads = np.zeros(n)
    wsum = 0.0
    for i in idx:
        w = (corr[i] / cmax) ** power
        lag = i - (n - 1)
        grads += w * grad_for_lag(h, lag, n, dx)
        wsum += w
    if wsum < 1e-15:
        return np.zeros(n)
    return grads / wsum

def soft_max_grad_fast(h, corr, tau, n, dx):
    tc = tau * corr
    mx = np.max(tc)
    w = np.exp(tc - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        lag = m - (n - 1)
        grad += w[m] * grad_for_lag(h, lag, n, dx)
    return grad

        for mode, tk, pw, tau in configs:
            if time.time() > deadline - 3:
                break
            h_opt, c5_opt = pgd_optimize(best_h, target, deadline - 3, mode=mode, top_k=tk, power=pw, tau=tau)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final binary rounding check
    if best_h is not None:
        n = len(best_h)
        h_bin = binary_round(best_h, n / 2.0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    return best_h, best_c5, len(best_h)
