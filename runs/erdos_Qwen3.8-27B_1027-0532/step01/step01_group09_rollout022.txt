def soft_gradient(h, top_k=5, temperature=0.001):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    # Get top shifts
    idx = np.argsort(corr)[-top_k:]
    weights = np.exp((corr[idx] - corr.max()) / temperature)
    weights /= weights.sum()
    
    j = np.arange(n)
    grad = np.zeros(n)
    for w, ki in zip(weights, idx):
        k = ki - (n - 1)
        a = j - k
        b = j + k
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += w * (1.0 - h[a[m1]])
        grad[m2] -= w * h[b[m2]]
    grad *= dx
    return grad

    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_shift(h, k, dx):
    n = len(h)
    j = np.arange(n)
    grad = np.zeros(n)
    a = j - k
    b = j + k
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += (1.0 - h[a[m1]])
    grad[m2] -= h[b[m2]]
    grad *= dx
    return grad

def soft_subgrad(h, top_k=5, temp=0.003):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(corr)[-top_k:]
    cvals = corr[idx]
    weights = np.exp((cvals - cvals.max()) / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, ki in zip(weights, idx):
        k = int(ki) - (n - 1)
        grad += w * grad_shift(h, k, dx)
    return grad

def subgrad_opt(h0, tm=60, lr_init=0.4, beta=0.82, top_k=5, temp=0.003):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    prev_c5 = best_c5
    for it in range(200000):
        if time.time() - t0 > tm:
            break
        grad = soft_subgrad(h, top_k=top_k, temp=temp)
        mom = beta * mom + (1.0 - beta) * grad
        h_new = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            h = h_new
        else:
            lr *= 0.995
            beta = min(0.95, beta + 0.0005)
        if lr < 1e-9:
            break
    return best

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()

    def obj(x):
        if time.time() - t0 > tm * 0.88:
            raise Exception("timeout")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

    def cons_fun(x):
        return x.sum() - tgt

    def cons_jac(x):
        return np.ones(n)

    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                     options={'maxiter': 30000, 'ftol': 1e-16})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def constraint_focused(h0, tm=40, n_constraints=3):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()

    def cons_fun(x):
        return x.sum() - tgt

    def cons_jac(x):
        return np.ones(n)

    for round_idx in range(20):
        if time.time() - t0 > tm:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-n_constraints:]
        for ki in top_idx:
            if time.time() - t0 > tm * 0.9:
                break
            k = int(ki) - (n - 1)
            tm_single = min(5, time.time() - t0 if False else tm - (time.time() - t0))

            def obj_single(x):
                if time.time() - t0 > tm * 0.95:
                    raise Exception("timeout")
                return (np.correlate(x, 1.0 - x, 'full') * dx)[ki]

            try:
                r = minimize(obj_single, h, method='SLSQP',
                             bounds=[(0.0, 1.0)] * n,
                             constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                             options={'maxiter': 2000, 'ftol': 1e-14})
                h1 = project_feasible(r.x, tgt)
                c5 = eval_c5(h1)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h1.copy()
                    h = h1
            except Exception:
                pass
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def make_structured_init(n, rng):
    r = rng.randint(4)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if r == 0:
        freq = float(rng.randint(2, 7))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(np.pi * freq * x + phase)
    elif r == 1:
        m = rng.randint(3, 8)
        h = np.zeros(n)
        for i in range(m):
            start = int(i * n / m)
            end = int((i + 1) * n / m)
            h[start:end] = rng.uniform(0.1, 0.9)
    elif r == 2:
        h = np.zeros(n)
        n_seg = rng.randint(4, 10)
        for i in range(n_seg):
            start = int(i * n / n_seg)
            end = int((i + 1) * n / n_seg)
            h[start:end] = rng.uniform(0, 1)
    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x / 2.0) + 0.1 * np.sin(4 * np.pi * x / 2.0)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)

def nesterov_subgrad(h0, tm=60, lr_init=0.3):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    y = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    beta = 0.9
    t0 = time.time()
    lr = lr_init
    for it in range(100000):
        if time.time() - t0 > tm:
            break
        grad = soft_subgrad(y, top_k=5, temp=0.003)
        lr_new = 2.0 / (it + 2.0)
        h_new = project_feasible(y - lr_new * lr * grad, tgt)
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = h_new + beta_new * (h_new - h)
        y = np.clip(y, 0, 1)
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left(frac=0.95):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    best = h_init.astype(float).copy()
    best_c5 = eval_c5(best)
    n_orig = len(best)

    # Phase 1: Intensive optimization at original resolution
    if time_left(0.25) > 30:
        h1 = subgrad_opt(best, tm=min(90, time_left(0.25)), lr_init=0.5, beta=0.82, top_k=5, temp=0.002)
        c1 = eval_c5(h1)
        if c1 < best_c5:
            best, best_c5 = h1, c1

    if time_left(0.30) > 20:
        h2 = selpolish(best, tm=min(80, time_left(0.30)))
        c2 = eval_c5(h2)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    if time_left(0.35) > 20:
        h3 = constraint_focused(best, tm=min(40, time_left(0.35)), n_constraints=3)
        c3 = eval_c5(h3)
        if c3 < best_c5:
            best, best_c5 = h3, c3

    if time_left(0.38) > 15:
        h4 = selpolish(best, tm=min(30, time_left(0.38)))
        c4 = eval_c5(h4)
        if c4 < best_c5:
            best, best_c5 = h4, c4

    # Phase 2: Perturbation search at original resolution
    for attempt in range(10):
        if time_left(0.45) < 8:
            break
        scale = 0.03 + 0.02 * attempt
        h_p = project_feasible(best + rng.normal(0, scale, size=n_orig), n_orig / 2.0)
        h_opt = subgrad_opt(h_p, tm=min(25, time_left(0.45)), lr_init=0.4, beta=0.85)
        c5_opt = eval_c5(h_opt)
        if c5_opt < best_c5:
            best, best_c5 = h_opt, c5_opt
        h_pol = selpolish(h_opt, tm=min(20, time_left(0.45)))
        c5_pol = eval_c5(h_pol)
        if c5_pol < best_c5:
            best, best_c5 = h_pol, c5_pol

    # Phase 3: Higher resolution optimization
    n_targets = [400, 500, 600, 800]
    for n_t in n_targets:
        if time_left(0.80) < 30:
            break
        h_up = resample_h(best, n_t)
        c_up = eval_c5(h_up)
        if c_up < best_c5:
            best, best_c5 = h_up, c_up

        h_sg = subgrad_opt(h_up, tm=min(80, time_left(0.80) // 2), lr_init=0.3, beta=0.85, top_k=7, temp=0.002)
        c_sg = eval_c5(h_sg)
        if c_sg < best_c5:
            best, best_c5 = h_sg, c_sg

        if time_left(0.80) > 20:
            h_sp = selpolish(h_sg, tm=min(40, time_left(0.80)))
            c_sp = eval_c5(h_sp)
            if c_sp < best_c5:
                best, best_c5 = h_sp, c_sp

        if time_left(0.80) > 15:
            h_cf = constraint_focused(best, tm=min(20, time_left(0.80)), n_constraints=4)
            c_cf = eval_c5(h_cf)
            if c_cf < best_c5:
                best, best_c5 = h_cf, c_cf

        if time_left(0.80) > 10:
            h_fp = selpolish(best, tm=min(15, time_left(0.80)))
            c_fp = eval_c5(h_fp)
            if c_fp < best_c5:
                best, best_c5 = h_fp, c_fp

    # Phase 4: Try structured inits at high resolution
    for n_t in [400, 600]:
        if time_left(0.90) < 30:
            break
        for _ in range(3):
            if time_left(0.90) < 15:
                break
            h_struct = make_structured_init(n_t, rng)
            h_so = subgrad_opt(h_struct, tm=min(30, time_left(0.90)), lr_init=0.4, beta=0.85)
            c_so = eval_c5(h_so)
            if c_so < best_c5:
                best, best_c5 = h_so, c_so
            if time_left(0.90) > 10:
                h_sp2 = selpolish(h_so, tm=min(20, time_left(0.90)))
                c_sp2 = eval_c5(h_sp2)
                if c_sp2 < best_c5:
                    best, best_c5 = h_sp2, c_sp2

    # Phase 5: Final intensive polish
    if time_left(0.99) > 30:
        h_f = selpolish(best, tm=min(100, time_left(0.99) * 0.5))
        cf = eval_c5(h_f)
        if cf < best_c5:
            best, best_c5 = h_f, cf

    if time_left(0.99) > 20:
        h_f2 = subgrad_opt(best, tm=min(40, time_left(0.99)), lr_init=0.2, beta=0.9, top_k=7, temp=0.001)
        cf2 = eval_c5(h_f2)
        if cf2 < best_c5:
            best, best_c5 = h_f2, cf2

    if time_left(0.99) > 15:
        h_f3 = selpolish(best, tm=min(30, time_left(0.99)))
        cf3 = eval_c5(h_f3)
        if cf3 < best_c5:
            best, best_c5 = h_f3, cf3

    if time_left(0.99) > 10:
        h_f4 = constraint_focused(best, tm=min(20, time_left(0.99)), n_constraints=5)
        cf4 = eval_c5(h_f4)
        if cf4 < best_c5:
            best, best_c5 = h_f4, cf4

    if time_left(0.99) > 10:
        h_f5 = selpolish(best, tm=min(15, time_left(0.99)))
        cf5 = eval_c5(h_f5)
        if cf5 < best_c5:
            best, best_c5 = h_f5, cf5

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
