import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_grad_single_k(h, k, n):
    """Compute gradient of f_k with respect to h."""
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += 1.0 - h[a[m1]]
    grad[m2] -= h[b[m2]]
    return grad * dx

def subgrad_top_m(h0, tm=60, lr_init=0.3, beta=0.8, top_m=3, soft_temp=10.0):
    """Subgradient optimization using top-m active shifts with softmax weighting."""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)

    t0 = time.time()

        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        # Find top-m shifts
        top_idx = np.argsort(corr)[::-1][:top_m]
        val = corr[top_idx]
        # Softmax weights
        w = np.exp(soft_temp * (val - val[0]))
        w /= w.sum()
        
        grad = np.zeros(n)
        for mi in range(len(top_idx)):
            if w[mi] < 1e-8:

            k = top_idx[mi] - (n - 1)
            grad += w[mi] * compute_grad_single_k(h, k, n)

            best = h_new.copy()
            h = h_new.copy()
        else:
            h = h_new.copy()  # still update h but track best separately
        lr *= 0.9998

    s = 10.0
    for stage in range(6):

            break
        def fs(x, s_val=s):
            corr = np.correlate(x, 1.0 - x, 'full') * dx
            m = corr.max()
            return m + np.log(np.exp(s_val * (corr - m)).sum()) / s_val

            return x.sum() - tgt

        try:
            r = minimize(fs, best, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,

                         options={'maxiter': 2000, 'ftol': 1e-12})
            h1 = project_feasible(r.x, tgt)
            c1 = eval_c5(h1)
            if c1 < best_val:
                best, best_val = h1, c1
        except Exception:
            pass
        s *= 2.5
    return best

def subgrad_top_m(h0, tm=60, lr_init=0.3, beta=0.8, top_m=3):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    for it in range(30000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        top_idx = np.argsort(corr)[::-1][:top_m]
        val = corr[top_idx]
        w = np.exp(20.0 * (val - val[0]))
        w /= w.sum()
        grad = np.zeros(n)
        for mi in range(len(top_idx)):
            if w[mi] < 1e-8:
                continue
            k = top_idx[mi] - (n - 1)
            grad += w[mi] * compute_grad_k(h, k)
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9998
        if lr < 1e-8:
            break
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)
