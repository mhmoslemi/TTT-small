    lc = min(la, n - c)
    h_new[a:a+la], h_new[c:c+lc] = h_new[c:c+lc].copy(), h_new[a:a+la].copy()
    return h_new

def block_flip(h, n, rng):
    h_new = h.copy()
    a = rng.randint(0, n - 5)
    la = rng.randint(3, min(10, n - a))
    h_new[a:a+la] = 1.0 - h_new[a:a+la]
    return h_new

def shift_perturb(h, n, rng):
    s = rng.randint(1, 4)
    return np.roll(h, s)

def golden_init(n):
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * golden) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n)

def alternating_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % 2 == 0) else 0.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c5 = evaluate_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Epsilon annealing from initial
    eps_schedule = [3.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]
    for eps in eps_schedule:
        if time_left() < 90.0:
            break
        h_new, c5 = do_slsqp(best_h.copy(), n, eps, 600, 1e-14)
        if h_new is not None and c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: LBFGS refinement
    if time_left() > 60.0:
        for eps in [100.0, 300.0, 800.0]:
            h_new, c5 = do_lbfgs(best_h.copy(), n, eps, 500)
            if h_new is not None and c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 3: Simulated annealing with block operations
    sa_temp = 0.01
    sa_min_temp = 0.0005
    sa_cool = 0.97
    n_sa = 0
    while time_left() > 200.0:
        n_sa += 1
        strategy = n_sa % 4
        if strategy == 0:
            h_pert = block_swap(best_h, n, rng)
        elif strategy == 1:
            h_pert = block_flip(best_h, n, rng)
        elif strategy == 2:
            h_pert = shift_perturb(best_h, n, rng)
        else:
            scale = 0.01 + 0.03 * min(n_sa // 4, 10)
            h_pert = best_h + scale * rng.randn(n)

        h_pert = project_feasible(h_pert, n)
        c5_pert = evaluate_c5(h_pert)

        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            sa_temp = max(sa_min_temp, sa_temp * 1.2)
        else:
            delta = c5_pert - best_c5
            if delta < sa_temp:
                accept_prob = np.exp(-delta / sa_temp)
                if rng.random() < accept_prob:
                    best_h = h_pert.copy()
        sa_temp = max(sa_min_temp, sa_temp * sa_cool)

    # Phase 4: Re-optimize from current best with high epsilon
    if time_left() > 80.0:
        for eps in [200.0, 500.0, 1000.0, 2000.0]:
            if time_left() < 40.0:
                break
            h_new, c5 = do_slsqp(best_h.copy(), n, eps, 800, 1e-15)
            if h_new is not None and c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 5: Try alternative n values
    alt_ns = [84, 86, 83, 87]
    for alt_n in alt_ns:
        if time_left() < 120.0:
            break
        h_alt = golden_init(alt_n)
        h_alt2 = alternating_init(alt_n)
        for h_init in [h_alt, h_alt2]:
            if time_left() < 60.0:
                break
            for eps in [50.0, 200.0, 800.0]:
                if time_left() < 40.0:
                    break
                h_new, c5 = do_slsqp(h_init.copy(), alt_n, eps, 500, 1e-14)
                if h_new is not None and c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    n = alt_n

    # Phase 6: Coordinate descent refinement
    n_iter_coord = 0
    while time_left() > 40.0:
        n_iter_coord += 1
        improved = False
        for idx in range(n):
            if time_left() < 40.0:
                break
            for delta in [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]:
                h_trial = best_h.copy()
                h_trial[idx] += delta
                if h_trial[idx] < 0.0 or h_trial[idx] > 1.0:
                    continue
                h_trial = project_feasible(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    improved = True
                    break
        if n_iter_coord > 5 and not improved:
            break

    # Phase 7: Final SLSQP polish
    if time_left() > 30.0:
        for eps in [500.0, 2000.0, 5000.0]:
            if time_left() < 20.0:
                break
            h_new, c5 = do_slsqp(best_h.copy(), n, eps, 400, 1e-15)
            if h_new is not None and c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
