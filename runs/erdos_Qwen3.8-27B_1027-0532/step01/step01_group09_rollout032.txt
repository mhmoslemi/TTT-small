import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def lse_gradient(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    exps = np.exp(beta * (corr - m))
    weights = exps / exps.sum()
    grad = np.zeros(n)
    N = 2 * n - 1
    for idx in range(N):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += w * dx * (1.0 - h[:n - k])
            grad[:n - k] -= w * dx * h[k:]
        else:
            s = -k
            grad[:n - s] += w * dx * (1.0 - h[s:])
            grad[s:] -= w * dx * h[:s]
    return grad

def lse_adam(h0, beta_start, beta_end, tm, lr=0.3):

    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0.astype(float).copy(), target)
    best = h.copy()
    best_c5 = eval_c5(h)
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    t0 = time.time()
    it = 0
    while time.time() - t0 < tm:
        it += 1
        frac = min(1.0, it / 2000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        grad = lse_gradient(h, beta, dx)
        t_step += 1
        m_s = b1 * m_s + (1.0 - b1) * grad
        v_s = b2 * v_s + (1.0 - b2) * (grad * grad)
        m_hat = m_s / (1.0 - b1 ** t_step)
        v_hat = v_s / (1.0 - b2 ** t_step)
        lr_t = lr / (1.0 + 0.003 * it)

        c5 = eval_c5(h)

            best = h.copy()
    return best
