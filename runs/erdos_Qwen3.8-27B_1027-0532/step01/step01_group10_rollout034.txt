    hi = 1.0 - y.min() + 1.0

        s = sum_clipped(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

5. Project

    return grad


def projected_gradient(h, n, dx, max_iter, beta_low, beta_high, lr_init, use_momentum):
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    lr = lr_init
    vel = np.zeros(n) if use_momentum else None
    mom = 0.85

    for it in range(max_iter):
        frac = float(it) / float(max_iter)
        if frac < 0.4:
            beta = beta_low
        elif frac < 0.7:
            beta = (beta_low + beta_high) / 3.0
        else:
            beta = beta_high

        val = smooth_obj(h, dx, beta)
        grad = compute_grad(h, dx, beta)

        if use_momentum:
            vel = mom * vel - lr * grad
        else:
            vel = -lr * grad

        h_new = project_h(h + vel, n)
        val_new = smooth_obj(h_new, dx, beta)

        if val_new > val:
            for _ in range(10):
                lr *= 0.5
                if use_momentum:
                    vel = mom * vel - lr * grad
                else:
                    vel = -lr * grad
                h_new = project_h(h + vel, n)
                val_new = smooth_obj(h_new, dx, beta)
                if val_new < val:
                    break

        h = h_new
        c5 = c5_value(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5


def slsqp_fun(h, dx, beta):
    return smooth_obj(h, dx, beta)


def slsqp_jac(h, dx, beta):
    return compute_grad(h, dx, beta)


def sum_eq(h):
    return float(np.sum(h) - len(h) * 0.5)


def sum_eq_jac(h):
    return np.ones(len(h))


def run_slsqp(h0, n, dx, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(slsqp_fun, h0, args=(dx, beta), jac=slsqp_jac,
                   method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.isfinite(res.fun):
        return repair_h(res.x, n)
    return h0


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return repair_h(h_dst, n_dst)


def gen_sinusoidal(n, freq, amp, phase):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return repair_h(h, n)


def gen_piecewise(n, n_blocks):
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        val = ((i % 3) + 1) / 3.0
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        h[start:end] = val
    return repair_h(h, n)


def elapsed_time(t0):
    return time.time() - t0


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.asarray(initial_h_values, dtype=np.float64)
    n_parent = len(h_parent)
    dx_parent = 2.0 / n_parent
    best_h = h_parent.copy()
    best_c5 = c5_value(best_h, dx_parent)
    best_n = n_parent

    # Phase 1: SLSQP at parent resolution
    for beta in [80.0, 40.0, 20.0, 10.0]:
        if elapsed_time(t0) > budget_s * 0.12:
            break
        h_opt = run_slsqp(best_h, best_n, 2.0 / best_n, beta, maxiter=400)
        c = c5_value(h_opt, 2.0 / best_n)
        if c < best_c5:
            best_h = h_opt
            best_c5 = c

    # Phase 2: Projected gradient + SLSQP at n=512
    n2 = 512
    dx2 = 2.0 / n2
    h2 = interpolate_h(best_h, best_n, n2)
    c2 = c5_value(h2, dx2)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2, c2, n2

    if elapsed_time(t0) < budget_s * 0.4:
        h_pgd, c_pgd = projected_gradient(h2, n2, dx2, max_iter=6000,
                                           beta_low=10.0, beta_high=200.0,
                                           lr_init=0.01, use_momentum=True)
        if c_pgd < best_c5:
            best_h, best_c5, best_n = h_pgd, c_pgd, n2

    if elapsed_time(t0) < budget_s * 0.5:
        h_pgd, c_pgd = projected_gradient(best_h, n2, dx2, max_iter=4000,
                                           beta_low=50.0, beta_high=500.0,
                                           lr_init=0.005, use_momentum=False)
        if c_pgd < best_c5:
            best_h, best_c5, best_n = h_pgd, c_pgd, n2

    if elapsed_time(t0) < budget_s * 0.65:
        h_opt = run_slsqp(best_h, n2, dx2, 30.0, maxiter=300)
        c = c5_value(h_opt, dx2)
        if c < best_c5:
            best_h, best_c5, best_n = h_opt, c, n2

    # Phase 3: Try n=1024
    if elapsed_time(t0) < budget_s * 0.55:
        n3 = 1024
        dx3 = 2.0 / n3
        h3 = interpolate_h(best_h, best_n, n3)
        c3 = c5_value(h3, dx3)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3, c3, n3

        h_pgd3, c_pgd3 = projected_gradient(h3, n3, dx3, max_iter=5000,
                                             beta_low=10.0, beta_high=200.0,
                                             lr_init=0.005, use_momentum=True)
        if c_pgd3 < best_c5:
            best_h, best_c5, best_n = h_pgd3, c_pgd3, n3

        if elapsed_time(t0) < budget_s * 0.75:
            h_pgd3b, c_pgd3b = projected_gradient(best_h, n3, dx3, max_iter=3000,
                                                   beta_low=50.0, beta_high=500.0,
                                                   lr_init=0.003, use_momentum=False)
            if c_pgd3b < best_c5:
                best_h, best_c5, best_n = h_pgd3b, c_pgd3b, n3

        if elapsed_time(t0) < budget_s * 0.85:
            h_opt3 = run_slsqp(best_h, n3, dx3, 20.0, maxiter=200)
            c = c5_value(h_opt3, dx3)
            if c < best_c5:
                best_h, best_c5, best_n = h_opt3, c, n3

    # Phase 4: Diversification - try alternate starts at best resolution
    fn = best_n
    fdx = 2.0 / fn
    if elapsed_time(t0) < budget_s * 0.85:
        starts = []
        for freq in [3, 5, 7]:
            starts.append(gen_sinusoidal(fn, freq, 0.4, 0.0))
            starts.append(gen_sinusoidal(fn, freq, 0.4, np.pi / 3.0))
        starts.append(gen_piecewise(fn, 4))
        starts.append(gen_piecewise(fn, 6))

        for h_start in starts:
            if elapsed_time(t0) > budget_s * 0.92:
                break
            h_pgd_d, c_pgd_d = projected_gradient(h_start, fn, fdx, max_iter=2000,
                                                   beta_low=10.0, beta_high=100.0,
                                                   lr_init=0.01, use_momentum=True)
            if c_pgd_d < best_c5:
                best_h, best_c5, best_n = h_pgd_d, c_pgd_d, fn

    # Phase 5: Final polishing
    fn = best_n
    fdx = 2.0 / fn

    if elapsed_time(t0) < budget_s * 0.95:
        h_pgd_f, c_pgd_f = projected_gradient(best_h, fn, fdx, max_iter=3000,
                                               beta_low=100.0, beta_high=1000.0,
                                               lr_init=0.002, use_momentum=True)
        if c_pgd_f < best_c5:
            best_h, best_c5, best_n = h_pgd_f, c_pgd_f, fn

    if elapsed_time(t0) < budget_s * 0.98:
        h_opt_f = run_slsqp(best_h, fn, fdx, 10.0, maxiter=300)
        c = c5_value(h_opt_f, fdx)
        if c < best_c5:
            best_h, best_c5, best_n = h_opt_f, c, fn

    # Final repair and evaluation
    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
