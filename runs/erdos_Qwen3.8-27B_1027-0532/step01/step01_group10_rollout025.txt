    h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_at_lag(h, k, n):
    """Compute d(corr_k)/d(h) for the given lag k (can be negative)."""
    grad = np.zeros(n)
    # grad[j] = (1-h[j-k]) * [0 <= j-k < n] - h[j+k] * [0 <= j+k < n]
    # Term 1: (1-h[j-k]) where j-k is in [0, n-1]
    lo = max(0, k)
    hi = min(n, n + k)  # j ranges from max(0,k) to min(n-1, n-1+k) + 1
    j1 = np.arange(lo, hi)
    j1k = j1 - k
    grad[j1] += (1.0 - h[j1k])
    # Term 2: -h[j+k] where j+k is in [0, n-1]
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    j2 = np.arange(lo2, hi2)
    j2k = j2 + k
    grad[j2] -= h[j2k]
    return grad

def projected_subgradient(h, n, dx, iters, step_size, top_k_frac=0.0):
    """
    Projected subgradient method for min max_k corr_k * dx.
    top_k_frac: if > 0, use average gradient over the top (top_k_frac * (2n-1)) lags.
    """
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    
    for t in range(iters):
        corr = compute_correlation(h) * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find lags to use
        if top_k_frac > 0:
            n_lags = len(corr)
            threshold = c5 - top_k_frac * (c5 - np.min(corr))
            top_lags = np.where(corr > threshold)[0]
        else:
            top_lags = np.array([np.argmax(corr)])
        
        # Average gradient over selected lags
        grad = np.zeros(n)
        for lag_idx in top_lags:
            k = lag_idx - (n - 1)  # convert index to lag
            grad += gradient_at_lag(h, k, n)
        grad /= len(top_lags)
        
        # Step
        alpha = step_size / (1.0 + 0.5 * np.sqrt(t + 1))
        h_new = h - alpha * grad
        h = repair_h(h_new, n)
    
    return best_h, best_c5

    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)
