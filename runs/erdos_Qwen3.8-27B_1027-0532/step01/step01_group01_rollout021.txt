        for a in range(min(30, n)):
            for b in range(max(0, n - 30), n):
                i = idx_desc[a]
                j = idx_asc[b]
                if i == j:
                    continue
                hi_val = best_h[i]
                hj_val = best_h[j]

                # Move mass from i (high grad, decrease) to j (low grad, increase)
                for d in all_deltas:
                    if hi_val < d or hj_val >= 1.0:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] = hi_val - d
                    h_trial[j] = hj_val + d
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break

                if time_left() < 250:
                    break

            if time_left() < 250:
                break

        # Also try random pairs for diversity
        if time_left() > 250:
            for _ in range(200):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                hi_val = best_h[i]
                hj_val = best_h[j]
                for d in all_deltas[:4]:
                    if hj_val < d or hi_val >= 1.0:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break
                    if hi_val < d or hj_val >= 1.0:
                        continue
                    h_trial[i] -= d
                    h_trial[j] += d
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break

        if not improved and sweep > 50:
            break

    # Phase 3: Simulated annealing
    if time_left() > 150:
        T = 0.0003
        best_h_sa = best_h.copy()
        T_min = 0.00001
        for step in range(20000):
            if time_left() < 150:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue

            d = 0.02 * np.random.randn()
            max_d_up_i = 1.0 - best_h_sa[i]
            max_d_down_i = best_h_sa[i]
            max_d_up_j = 1.0 - best_h_sa[j]
            max_d_down_j = best_h_sa[j]
            d = max(min(d, max_d_up_i), -max_d_down_i)
            d = max(min(d, max_d_up_j), -max_d_down_j)

            if abs(d) < 1e-7:
                continue

            h_trial = best_h_sa.copy()
            h_trial[i] += d
            h_trial[j] -= d

            c5_trial = evaluate_c5(h_trial)
            delta = c5_trial - evaluate_c5(best_h_sa)

            if delta < 0 or np.random.random() < np.exp(-delta / T):
                best_h_sa = h_trial.copy()
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()

            T = max(T * 0.9995, T_min)

    # Phase 4: Fine pair moves with smaller deltas
    fine_deltas = [0.0002, 0.0005, 0.001, 0.002, 0.003]
    for sweep in range(100):
        if time_left() < 100:
            break
        improved = False
        idx_desc = np.argsort(-grad)
        idx_asc = np.argsort(grad)
        for a in range(min(40, n)):
            for b in range(max(0, n - 40), n):
                i = idx_desc[a]
                j = idx_asc[b]
                if i == j:
                    continue
                hi_val = best_h[i]
                hj_val = best_h[j]
                for d in fine_deltas:
                    if hi_val < d or hj_val >= 1.0:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] = hi_val - d
                    h_trial[j] = hj_val + d
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break
                if time_left() < 100:
                    break
            if time_left() < 100:
                break
        if not improved and sweep > 20:
            break

    # Phase 5: Final SLSQP polish
    if time_left() > 80:
        for eps in [2000.0, 5000.0, 10000.0]:
            if time_left() < 40:
                break
            _eps_global = eps
            res = minimize(smooth_max_obj, best_h.copy(), jac=smooth_max_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': eq_constraint}],
                           options={'maxiter': 5000, 'ftol': 1e-16})
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Phase 6: Ultra-fine single coordinate moves
    if time_left() > 30:
        ultra_deltas = [0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002]
        for round_num in range(50):
            if time_left() < 30:
                break
            improved = False
            for i in range(n):
                for d in ultra_deltas:
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    if h_trial[i] < 0 or h_trial[i] > 1:
                        continue
                    h_trial = project_feasible(h_trial, n)
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break
            if not improved:
                break

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
