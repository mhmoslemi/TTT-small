        if lag == 0:
            grad += w * (omh - h)
        elif lag > 0:
            grad[lag:] += w * omh[:n - lag]
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * omh[l:]
            grad[l:] -= w * h[:n - l]
    grad *= dx
    return obj, grad


def slsqp_solve(h0, n, temp, max_iter=400):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_max_eval(x, n, temp)[0]

    def _grad(x):
        return smooth_max_grad(x, n, temp)[1]

    def _cons_fun(x):
        return np.sum(x) - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        result = minimize(_obj, h0.copy(), method='SLSQP', jac=_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-13})
        h_out = np.clip(result.x, 0.0, 1.0)
        h_out = project_onto_constraint(h_out, n)
        return h_out
    except Exception:
        return h0.copy()


def projected_grad_descent(h0, n, temp, max_iter=2000, verbose=False):
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    alpha = 0.1
    min_alpha = 1e-6
    for it in range(max_iter):
        _, grad = smooth_max_grad(h, n, temp)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        g_centered = grad - np.mean(grad)
        g_norm_c = np.linalg.norm(g_centered)
        if g_norm_c < 1e-15:
            break
        step = alpha * 2.0 / g_norm_c
        for _ in range(40):
            h_trial = project_onto_constraint(h - step * g_centered, n)
            obj_trial, _, _ = smooth_max_eval(h_trial, n, temp)
            obj_cur, _, _ = smooth_max_eval(h, n, temp)
            if obj_trial <= obj_cur - 1e-4 * step * g_norm_c:
                break
            step *= 0.5
        if step < min_alpha * 1e-3:
            break
        h = h_trial
        c5_val = compute_c5(h, n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        alpha = max(alpha * 0.999, min_alpha)
    return best_h, best_c5


def targeted_reduction(h0, n, worst_lag, temp, max_iter=200):
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    alpha = 0.05
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        target_idx = worst_lag + (n - 1)
        target_val = corr[target_idx]
        max_val = np.max(corr)
        blend = 0.5 + 0.5 * (target_val / max_val) if max_val > 0 else 1.0
        effective_temp = temp * (1.0 + 2.0 * (1.0 - blend))
        _, grad = smooth_max_grad(h, n, effective_temp)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        g_centered = grad - np.mean(grad)
        step = alpha / max(g_norm, 1e-10)
        h_trial = project_onto_constraint(h - step * g_centered, n)
        c5_trial = compute_c5(h_trial, n)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
        else:
            alpha *= 0.9
        if alpha < 1e-8:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    parent = np.array(initial_h_values, dtype=float)
    n_base = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n_base)
    best_n = n_base

    # Phase 1: Optimize on base grid with SLSQP at increasing temperatures
    h_work = parent.copy()
    temps_phase1 = [50, 100, 200, 500, 1000, 2000]
    for t in temps_phase1:
        if remaining() < 180:
            break
        h_opt = slsqp_solve(h_work, n_base, t, max_iter=500)
        c5_opt = compute_c5(h_opt, n_base)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        c5_work = compute_c5(h_work, n_base)
        if c5_opt < c5_work:
            h_work = h_opt

    # Phase 1b: PGD polish on base grid
    if remaining() > 120:
        h_pg, c5_pg = projected_grad_descent(h_work, n_base, 1500, max_iter=3000)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n_base

    # Phase 2: Targeted reduction of worst lag on base grid
    if remaining() > 150:
        corr = np.correlate(best_h, 1.0 - best_n and best_h, mode="full")
        idx_worst = np.argmax(corr)
        worst_lag = idx_worst - (n_base - 1)
        h_tar, c5_tar = targeted_reduction(best_h, n_base, worst_lag, 500, max_iter=500)
        if c5_tar < best_c5:
            best_c5 = c5_tar
            best_h = h_tar.copy()

    # Phase 3: Grid refinement to 2x
    n_fine = n_base * 2
    if remaining() > 400 and n_fine <= 560:
        x_src = np.linspace(0, 2.0, best_n, endpoint=False)
        x_fine = np.linspace(0, 2.0, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_src, best_h)
        h_fine = project_onto_constraint(h_fine, n_fine)

        h_work_fine = h_fine.copy()
        temps_fine = [100, 300, 600, 1200, 2500]
        for t in temps_fine:
            if remaining() < 200:
                break
            h_opt = slsqp_solve(h_work_fine, n_fine, t, max_iter=300)
            c5_opt = compute_c5(h_opt, n_fine)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_fine
            c5_work = compute_c5(h_work_fine, n_fine)
            if c5_opt < c5_work:
                h_work_fine = h_opt

        # PGD on fine grid
        if remaining() > 150:
            h_pg, c5_pg = projected_grad_descent(h_work_fine, n_fine, 2000, max_iter=2000)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n_fine

        # Targeted reduction on fine grid
        if remaining() > 100:
            corr_f = np.correlate(best_h, 1.0 - best_h, mode="full")
            idx_w = np.argmax(corr_f)
            worst_lag_f = idx_w - (n_fine - 1)
            h_tar, c5_tar = targeted_reduction(best_h, n_fine, worst_lag_f, 800, max_iter=400)
            if c5_tar < best_c5:
                best_c5 = c5_tar
                best_h = h_tar.copy()
                best_n = n_fine

    # Phase 4: Random restarts around best solution
    if remaining() > 200:
        for i in range(15):
            if remaining() < 120:
                break
            scale = 0.01 + 0.015 * (i % 4)
            h_pert = best_h + rng.randn(best_n) * scale
            h_pert = project_onto_constraint(h_pert, best_n)
            t_use = [200, 500, 1000][i % 3]
            h_opt = slsqp_solve(h_pert, best_n, t_use, max_iter=300)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Final PGD polish
    if remaining() > 80:
        h_final, c5_final = projected_grad_descent(best_h, best_n, 3000, max_iter=5000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final verification and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
