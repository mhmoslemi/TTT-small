import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def pgd_with_annealing(h0, n, target_sum, deadline, 
                        tau_start=5.0, tau_end=200.0,
                        n_anneal=50, lr_base=0.05, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    total_anneal = n_anneal
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Anneal tau
        if it < total_anneal:
            frac = it / max(total_anneal - 1, 1)
            tau = tau_start * (tau_end / tau_start) ** frac
        else:
            tau = tau_end
        
        obj, g = soft_max_obj_and_grad(h, tau)
        
        # Backtracking line search
        lr = lr_base
        for bt in range(10):
            h_new = h - lr * g
            h_new = project_onto_constraint(h_new, target_sum)
            c5 = compute_c5(h_new)
            if c5 < obj + 1e-10:  # sufficient decrease
                break
            lr *= 0.5
        else:
            lr = 0.001
            h_new = h - lr * g
            h_new = project_onto_constraint(h_new, target_sum)
            c5 = compute_c5(h_new)
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def exact_subgradient_method(h0, n, target_sum, deadline, max_iter=3000):
    """Use exact max with subgradient, Polyak-like step size."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    n = len(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr_raw)
        lag = k_star - (n - 1)
        
        # Subgradient of the max correlation w.r.t. h
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        
        g *= (2.0 / n)  # dx factor
        
        # Polyak step size: step = (f(h) - f*) / ||g||^2
        # We don't know f*, use a heuristic
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-12:
            break
        lr = step / (1.0 + it * 0.01)
        
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.01, 0.2)
        else:
            step *= 0.9
            if step < 0.001:
                # Restart with perturbation
                h = best_h + 0.01 * np.random.randn(n)
                h = project_onto_constraint(h, target_sum)
                step = 0.03
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, deadline, tau=80.0, maxiter=200):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def multi_lag_cycling(h0, n, target_sum, deadline, max_iter=2000, lr=0.03):
    """Cycle through lags, minimizing each in turn."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Select lag: cycle or random
        m = it % n_lags
        lag = m - (n - 1)
        
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        
        if c5 <= best_c5 + 1e-10:  # allow slight increase for cycling
            h = h_new
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
        else:
            # Revert if too bad
            h = best_h
    
    return best_h, best_c5

def coordinate_block_descent(h0, n, target_sum, deadline, block_size=8, max_iter=5000):
    """Optimize small blocks of coordinates at a time."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a random block
        start = np.random.randint(0, n - block_size + 1)
        end = start + block_size
        
        # Compute gradient at current h for the worst lag
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr_raw)
        lag = k_star - (n - 1)
        
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        
        # Try to improve the block
        block_g = g[start:end]
        block_h = h[start:end].copy()
        
        # Simple 1D line search along -block_g
        for lr in [0.1, 0.05, 0.02, 0.01]:
            new_block = np.clip(block_h - lr * block_g, 0, 1)
            # Project to maintain sum constraint locally (approximate)
            h_trial = h.copy()
            h_trial[start:end] = new_block
            h_trial = project_onto_constraint(h_trial, target_sum)
            c5 = compute_c5(h_trial)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_trial.copy()
                h = h_trial
                break
    
    return best_h, best_c5

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)) + 1e-30)
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

        return o

        _, g = soft_max_obj_and_grad(x, tau)
        return g

            h_new = h - lr * g
            h_new = project_onto_constraint(h_new, target_sum)
            c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def exact_subgradient_method(h0, n, target_sum, deadline, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr_raw)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        lr = step / (1.0 + 0.005 * it)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.005, 0.15)
        else:
            step *= 0.92
            if step < 0.002:
                h = best_h + 0.008 * np.random.randn(n)
                h = project_onto_constraint(h, target_sum)
                step = 0.03
    return best_h, best_c5

def simulated_annealing(h0, n, target_sum, deadline, T_start=0.005, T_end=0.00005,
                        n_temps=30, local_frac=0.6):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    dx = 2.0 / n
    for t_idx in range(n_temps):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (t_idx / max(n_temps - 1, 1))
        n_steps = max(50, int(200000 * T / T_start))
        for s in range(n_steps):
            if time.time() > deadline:
                break
            if np.random.random() < local_frac:
                i = np.random.randint(n - 1)
                j = i + 1
            else:
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
            eps = np.random.uniform(0.01, 0.08)
            if h[i] < eps:
                if h[j] < eps:
                    continue
                i, j = j, i
            if h[i] < eps:
                continue
            if h[j] + eps > 1.0:
                eps = 1.0 - h[j]
                if eps <= 1e-6:
                    continue
            old_hi = h[i]
            old_hj = h[j]
            h[i] = old_hi - eps
            h[j] = old_hj + eps
            new_corr = np.correlate(h, 1.0 - h, mode="full") * dx
            new_c5 = np.max(new_corr)
            delta = new_c5 - current_c5
            if delta <= 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = old_hi
                h[j] = old_hj
    return best_h, best_c5

def multi_tau_slsqp(h0, n, target_sum, deadline, taus=None, maxiter_each=80):
    if taus is None:
        taus = [15.0, 30.0, 60.0, 120.0, 200.0]
    h = h0.copy()
    best_c5 = compute_c5(h)
    for tau in taus:
        if time.time() > deadline:
            break
        t_phase = min(deadline, time.time() + (deadline - time.time()) * 0.7)
        h_opt, c5_opt = slsqp_optimize(h, n, target_sum, t_phase, tau=tau, maxiter=maxiter_each)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            h = h_opt.copy()
    return h, best_c5

def block_perturbation(h0, n, target_sum, deadline, n_trials=200, block_sizes=None):
    if block_sizes is None:
        block_sizes = [4, 8, 16]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        h_trial = best_h.copy()
        bs = block_sizes[np.random.randint(len(block_sizes))]
        start = np.random.randint(0, n - bs + 1)
        sign = np.random.choice([-1.0, 1.0])
        amp = np.random.uniform(0.02, 0.1)
        h_trial[start:start+bs] += sign * amp
        h_trial = project_onto_constraint(h_trial, target_sum)
        c5 = compute_c5(h_trial)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_trial.copy()
            h = h_trial
        else:
            h = best_h + 0.01 * np.random.randn(n)
            h = project_onto_constraint(h, target_sum)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 10

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0
        n0 = len(h_init)
    else:
        n0 = 120
        best_h = np.full(n0, 0.5)
        best_h = project_onto_constraint(best_h, n0 / 2.0)
        best_c5 = compute_c5(best_h)

    n = n0

    # Phase 1: Multi-tau SLSQP (coarse to fine)
    if time.time() < deadline - 400:
        t_phase = deadline - 300
        h_opt, c5_opt = multi_tau_slsqp(best_h, n, n / 2.0, t_phase,
                                         taus=[20.0, 40.0, 80.0], maxiter_each=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Simulated annealing (global search)
    if time.time() < deadline - 300:
        t_phase = deadline - 200
        h_opt, c5_opt = simulated_annealing(best_h, n, n / 2.0, t_phase,
                                             T_start=0.008, T_end=0.0001,
                                             n_temps=25, local_frac=0.7)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: PGD with annealing
    if time.time() < deadline - 200:
        t_phase = deadline - 150
        h_opt, c5_opt = pgd_with_annealing(best_h, n, n / 2.0, t_phase,
                                            tau_start=5.0, tau_end=200.0,
                                            n_anneal=30, lr_base=0.06, max_iter=4000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Exact subgradient method
    if time.time() < deadline - 150:
        t_phase = deadline - 100
        h_opt, c5_opt = exact_subgradient_method(best_h, n, n / 2.0, t_phase, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: More SA from current best (finer temperature)
    if time.time() < deadline - 120:
        t_phase = deadline - 80
        h_opt, c5_opt = simulated_annealing(best_h, n, n / 2.0, t_phase,
                                             T_start=0.002, T_end=0.00002,
                                             n_temps=20, local_frac=0.8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: SLSQP refinement at high tau
    if time.time() < deadline - 60:
        t_phase = deadline - 30
        h_opt, c5_opt = slsqp_optimize(best_h, n, n / 2.0, t_phase, tau=150.0, maxiter=150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Block perturbation
    if time.time() < deadline - 40:
        t_phase = deadline - 20
        h_opt, c5_opt = block_perturbation(best_h, n, n / 2.0, t_phase, n_trials=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final SA + SLSQP
    if time.time() < deadline - 15:
        t_phase = deadline - 10
        h_opt, c5_opt = simulated_annealing(best_h, n, n / 2.0, t_phase,
                                             T_start=0.001, T_end=0.00001,
                                             n_temps=10, local_frac=0.85)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if best_h is not None and time.time() < deadline - 5:
        t_phase = deadline - 2
        h_opt, c5_opt = slsqp_optimize(best_h, n, n / 2.0, t_phase, tau=300.0, maxiter=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    if best_h is not None:
        n = len(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_constraint(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
