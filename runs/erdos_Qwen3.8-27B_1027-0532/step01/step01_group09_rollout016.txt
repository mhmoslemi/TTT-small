import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def block_optimize(h, start, sz, n_trials=50):
    """Optimize a block of sz consecutive coordinates."""
    n = len(h)
    dx = 2.0 / n
    end = start + sz
    S = h[start:end].sum()
    
    # For sz=1: just project (already in [0,1] and sum is fixed)
    if sz <= 1:
        return h[start:end].copy()
    
    # For sz=2: 1D search
    if sz == 2:
        lo = max(0.0, S - 1.0)
        hi = min(1.0, S)
        best_x = h[start:end].copy()
        best_val = eval_c5(h)
        for t in np.linspace(lo, hi, n_trials):
            h_work = h.copy()
            h_work[start] = t
            h_work[start+1] = S - t
            c = eval_c5(h_work)
            if c < best_val:
                best_val = c
                best_x = np.array([t, S-t])
        return best_x
    
    # For sz=3: 2D search
    if sz == 3:
        # x[0] = t1, x[1] = t2, x[2] = S - t1 - t2
        # Constraints: 0 ≤ ti ≤ 1
        best_x = h[start:end].copy()
        best_val = eval_c5(h)
        # Grid search
        t1_vals = np.linspace(0, 1, n_trials)

I want to increase h[i] where g[i] < 0 (to reduce f_{k*}) and decrease h[i] where g[i] > 0.

def bottleneck_move(h, rng, n_grid=20):
    """Find the bottleneck lag and make a targeted move."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(corr) - (n - 1)  # actual lag
    if k_star == 0:
        k_star = 1  # skip zero lag (it's always 0 for this formula? Let me check)

    # f_k = sum_i h[i] * (1 - h[i+k]) for valid i
    # df_k/dh[i] = (1-h[i+k]) - h[i-k] (approximately)
    g = np.zeros(n)
    for i in range(n):
        ip = i + k_star
        im = i - k_star
        term = 0.0
        if 0 <= ip < n:
            term += 1.0 - h[ip]
        if 0 <= im < n:
            term -= h[im]
        g[i] = term * dx

    # Use a random top-k selection for diversity

    low_idx = np.argsort(g)[:n_cand]    # positions where g is smallest

    j = low_idx[rng.randint(n_cand)]

        return False

    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def project_feasible(x, target):
    x = np.clip(x.astype(float), 0.0, 1.0)
