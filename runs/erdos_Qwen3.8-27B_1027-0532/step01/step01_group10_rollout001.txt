def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, 0.999)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = max(diff / cnt, -0.999)
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final exact repair
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, dx, max_sweeps=100, t_limit=None, t_start=None):
    best_h = h.copy()
    best_c5 = c5_value(best_h, dx)
    rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if t_limit and t_start and (time.time() - t_start) > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = best_h[i]
            # Try several candidate values
            candidates = [0.0, 1.0, old]
            # Add some structured candidates
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(frac)
            # Add nearby values
            candidates.append(max(0, old - 0.05))
            candidates.append(min(1, old + 0.05))
            
            best_val = old
            best_obj = best_c5
            
            for cv in candidates:
                if cv < 0 or cv > 1:
                    continue
                best_h[i] = cv

                c = float(np.max(corr)) * dx
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            
            best_h[i] = best_val
            if best_obj < best_c5 - 1e-15:
                best_c5 = best_obj
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def simulated_annealing(h, n, dx, t_start, t_limit, initial_T=0.01, cooling=0.999):
    best_h = h.copy()
    best_c5 = c5_value(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5
    T = initial_T
    rng = np.random.RandomState(77)
    
    while (time.time() - t_start) < t_limit:
        # Pick a random coordinate
        i = rng.randint(n)
        new_val = rng.uniform(0, 1)
        delta = new_val - current_h[i]
        
        # Find another coordinate to adjust for sum constraint
        if delta > 0:
            mask = current_h > 1e-6
        else:
            mask = current_h < 1.0 - 1e-6
        if not np.any(mask):
            T *= cooling
            continue
        j = rng.choice(np.where(mask)[0])
        
        # Adjust j to maintain sum
        old_j = current_h[j]
        new_j = old_j - delta
        if new_j < 0 or new_j > 1:
            T *= cooling
            continue
        
        current_h[i] = new_val
        current_h[j] = new_j
        new_c5 = c5_value(current_h, dx)
        
        if new_c5 < current_c5 or rng.uniform() < np.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            current_h[i] = new_val - delta  # revert
            current_h[j] = old_j
        
        T *= cooling
    
    return best_h, best_c5

def subgradient_method(h, n, dx, max_iter=5000, t_limit=None, t_start=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    
    lr = 0.01
    beta = 10.0
    
    for it in range(max_iter):
        if t_limit and t_start and (time.time() - t_start) > t_limit:
            break
        
        corr = compute_corr(h) * dx
        # Smooth max
        m = np.max(corr)
        weights = np.exp(beta * (corr - m))
        weights /= np.sum(weights)
        
        # Gradient of c_k with respect to h
        # c_k = sum_n h[n+k] * (1-h[n]) * dx
        # d(c_k)/d(h[i]) = dx * [ (1-h[i-k]) if 0<=i-k<n and k!=0 
        #                          - h[i+k] if 0<=i+k<n and k!=0
        #                          (1-2*h[i]) if k=0 ]
        
        grad = np.zeros(n)
        for idx in range(2*n - 1):
            k = idx - (n - 1)
            if weights[idx] < 1e-10:
                continue
            if k == 0:
                grad += weights[idx] * (1 - 2*h) * dx
            else:
                if k > 0:
                    # h[i] appears as h[n+k] when n = i-k, valid if 0 <= i-k < n
                    # contribution: (1-h[i-k])
                    mask1 = np.arange(n) >= k  # i >= k, so i-k >= 0
                    grad[mask1] += weights[idx] * (1 - h[:n-k]) * dx
                    # h[i] appears in (1-h[n]) when n = i, valid if i+k < n
                    # contribution: -h[i+k]
                    mask2 = np.arange(n) < n - k  # i < n-k, so i+k < n
                    grad[mask2] += weights[idx] * (-h[k:]) * dx
                else:  # k < 0
                    kk = -k
                    # h[i] appears as h[n+k] = h[i] when n = i-k = i+kk, valid if i+kk < n
                    # contribution: (1-h[i+kk])
                    mask1 = np.arange(n) < n - kk
                    grad[mask1] += weights[idx] * (1 - h[kk:]) * dx
                    # h[i] appears in (1-h[n]) = (1-h[i]) when n = i, valid if i+k = i-kk >= 0
                    # contribution: -h[i-kk]
                    mask2 = np.arange(n) >= kk
                    grad[mask2] += weights[idx] * (-h[:n-kk]) * dx
        
        # Gradient descent step
        h = h - lr * grad
        h = repair_h(h, n)
        
        c = c5_value(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        
        # Adaptive learning rate
        lr *= 0.999
    
    return best_h, best_c5

    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def subgradient_step(h, n, dx, beta, lr):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        if w < 1e-12:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            mask1 = np.arange(n) >= k
            grad[mask1] += w * (1.0 - h[:n - k])
            mask2 = np.arange(n) < n - k
            grad[mask2] += w * (-h[k:])
        else:
            kk = -k
            mask1 = np.arange(n) < n - kk
            grad[mask1] += w * (1.0 - h[kk:])
            mask2 = np.arange(n) >= kk
            grad[mask2] += w * (-h[:n - kk])
    
    return grad * dx
