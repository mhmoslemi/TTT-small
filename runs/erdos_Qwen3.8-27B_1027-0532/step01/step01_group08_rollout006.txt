import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_vec(h):
    """Return the full correlation vector (not just the max)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_single_lag(h, lag, n):
    """Gradient of corr[lag] with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    # corr[lag] = sum_i h[i] * (1 - h[i+lag]) for valid i
    # d/dh[j] = sum over terms where j appears
    # If j appears as h[i]: contributes (1-h[i+lag])
    # If j appears as h[i+lag]: contributes -h[i]
    
    # For lag >= 0: valid i are 0 to n-1-lag
    #   h[i] for i in [0, n-1-lag]: appears in h[i] term
    #   h[i+lag] for i in [0, n-1-lag]: i+lag in [lag, n-1]
    
    if lag >= 0:
        # h[j] appears as h[i] when j=i, i in [0, n-1-lag], so j in [0, n-1-lag]
        # contribution: (1-h[j+lag])
        j_max = n - 1 - lag
        g[0:j_max+1] += (1.0 - h[lag:n])
        # h[j] appears as h[i+lag] when j=i+lag, i in [0, n-1-lag], so j in [lag, n-1]
        # contribution: -h[j-lag]
        g[lag:n] += (-h[0:n-lag])
    else:
        # lag < 0, let lag' = -lag
        lag_pos = -lag
        # By symmetry, corr[lag] with h and (1-h) is related to corr[-lag]
        # Actually let me just handle it directly
        # corr at index (n-1+lag) where lag < 0
        # This is sum_i h[i] * (1-h[i+lag]) where i+lag >= 0, so i >= -lag = lag_pos
        # i in [lag_pos, n-1], and h[i+lag] = h[i-lag_pos]
        # h[j] appears as h[i] when j=i, i in [lag_pos, n-1]: contribution (1-h[j-lag_pos])
        g[lag_pos:n] += (1.0 - h[0:n-lag_pos])
        # h[j] appears as h[i+lag] = h[i-lag_pos] when j = i-lag_pos, i in [lag_pos, n-1]: j in [0, n-1-lag_pos]
        # contribution: -h[j+lag_pos]
        g[0:n-lag_pos] += (-h[lag_pos:n])
    
    return g * dx

def grad_worst_lag(h, n):
    """Subgradient using only the worst lag."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    return grad_single_lag(h, lag, n)

def grad_soft_max(h, tau, n):
    """Gradient of log-sum-exp approximation of max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    shifted = tau * (corr - max_corr)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights /= w_sum
    obj = (max_corr + np.log(w_sum) / tau)
    
    # Gradient
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        g += weights[m] * grad_single_lag(h, lag, n)
    
    return obj, g

def project_onto_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)

def nesterov_subgrad(h0, n, target_sum, deadline, max_iter=10000):
    """Averaged subgradient method targeting the worst lag."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    x = h0.copy()  # running average
    count = 0
    step_scale = np.linalg.norm(h0)  # initial step scale
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        
        # Compute subgradient at h
        g = grad_worst_lag(h, n)
        
        # Step size: eta_t = step_scale / sqrt(t)
        eta = step_scale / np.sqrt(t)
        
        # Update
        h_new = h - eta * g
        h_new = project_onto_box_sum(h_new, target_sum)
        
        # Update running average
        x = (count * x + h_new) / (count + 1)
        count += 1
        
        h = h_new

        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, deadline, max_sweeps=50):
    """Coordinate descent: optimize one variable at a time."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c5 = np.max(corr)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            # Try setting h[j] to different values
            # The optimal h[j] given others is determined by the active constraint
            # For simplicity, try a few candidate values
            old_val = h[j]
            # Candidate values: current, 0, 1, and a few in between
            candidates = [0.0, 1.0, old_val]
            # Add some random candidates
            candidates.extend(np.random.uniform(0, 1, 3).tolist())
            
            best_val = old_val
            best_c5_j = max_c5
            
            for cv in candidates:
                if abs(cv - old_val) < 1e-6:
                    continue
                h[j] = cv
                # Quick check: recompute only affected correlations
                # For efficiency, just check if this could improve things
                # Full recompute for correctness
                c = np.correlate(h, 1.0 - h, mode="full") * dx
                c5_j = np.max(c)
                if c5_j < best_c5_j - 1e-10:
                    best_c5_j = c5_j
                    best_val = cv
            
            if best_val != old_val:
                h[j] = best_val
                # Fix the sum constraint
                h = project_onto_box_sum(h, target_sum)
                improved = True
                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                max_c5 = np.max(corr)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target_sum, deadline, tau=80.0):
    """SLSQP optimization with log-sum-exp smoothing."""
    def obj_fun(x):
        o, _ = grad_soft_max(x, tau, n)
        return o
    
    def grad_fun(x):
        _, g = grad_soft_max(x, tau, n)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_box_sum(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_single_lag(h, lag, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] += (-h[lag:])
    else:
        p = -lag
        g[:n - p] += (1.0 - h[p:])
        g[p:] += (-h[:n - p])
    return g * dx

def grad_soft_max(h, tau, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    corr_scaled = corr * dx
    max_c = np.max(corr_scaled)
    shifted = tau * (corr_scaled - max_c)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w_sum = np.sum(exp_shifted)
    weights = exp_shifted / w_sum
    obj = max_c + (max_shifted + np.log(w_sum)) / tau
    g = np.zeros(n)
    active = weights > 1e-12
    for m in np.where(active)[0]:
        lag = m - (n - 1)
        g += weights[m] * grad_single_lag(h, lag, n)
    return obj, g

def project_box_sum(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def pgd_backtrack(h0, n, target_sum, deadline, max_iter=8000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = 300.0
    base_lr = 2.0 / n
    lr = base_lr
    stall = 0
    max_stall = 60
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, g = grad_soft_max(h, tau, n)
        f_cur = compute_c5(h)
        alpha = lr
        improved = False
        for _ in range(12):
            h_trial = project_box_sum(h - alpha * g, target_sum)
            f_trial = compute_c5(h_trial)
            if f_trial < f_cur - 1e-10:
                h = h_trial
                improved = True
                break
            alpha *= 0.5
        if improved:
            lr = min(alpha * 1.8, base_lr * 8.0)
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            lr = max(lr * 0.4, base_lr * 0.1)
            stall += 1
        if stall >= max_stall:
            tau = max(tau * 0.6, 10.0)
            lr = base_lr
            stall = 0
            if tau <= 10.0:
                tau = 300.0
                h = best_h + 0.01 * np.random.randn(n)
                h = project_box_sum(h, target_sum)
    return best_h, best_c5

def nesterov_worst_lag(h0, n, target_sum, deadline, max_iter=20000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    x_avg = h0.copy()
    cnt = 0
    h_norm = max(np.linalg.norm(h0), 1e-6)
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        g = grad_single_lag(h, lag, n)
        eta = h_norm / (np.sqrt(t) * 1.5)
        h_new = project_box_sum(h - eta * g, target_sum)
        cnt += 1
        x_avg = (cnt - 1) * x_avg / cnt + h_new / cnt
        h = h_new
        if t % 50 == 0:
            x_proj = project_box_sum(x_avg, target_sum)
            c5_x = compute_c5(x_proj)
            c5_h = compute_c5(h)
            if c5_x < best_c5:
                best_c5 = c5_x
                best_h = x_proj.copy()
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h.copy()
    x_proj = project_box_sum(x_avg, target_sum)
    c5_x = compute_c5(x_proj)
    if c5_x < best_c5:
        best_c5 = c5_x
        best_h = x_proj.copy()
    return best_h, best_c5

def topk_subgrad(h0, n, target_sum, deadline, k=3, max_iter=10000):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    x_avg = h0.copy()
    cnt = 0

            break

        for m in top_k:
            lag = m - (n - 1)
            g += grad_single_lag(h, lag, n)
        g /= k
        eta = h_norm / (np.sqrt(t) * 2.0)
        h_new = project_box_sum(h - eta * g, target_sum)
        cnt += 1
        x_avg = (cnt - 1) * x_avg / cnt + h_new / cnt
        h = h_new
        if t % 100 == 0:
            x_proj = project_box_sum(x_avg, target_sum)
            c5_x = compute_c5(x_proj)
            c5_h = compute_c5(h)
            if c5_x < best_c5:
                best_c5 = c5_x
                best_h = x_proj.copy()
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h.copy()
    x_proj = project_box_sum(x_avg, target_sum)

def coord_descent(h0, n, target_sum, deadline, max_sweeps=30):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            old_val = h[j]
            best_val = old_val
            best_local = best_c5
            candidates = np.array([0.0, 1.0, old_val * 0.5, (1.0 + old_val) * 0.5])
            candidates = np.unique(np.clip(candidates, 0.0, 1.0))
            for cv in candidates:
                if abs(cv - old_val) < 1e-8:
                    continue
                h[j] = cv
                c = compute_c5(h)
                if c < best_local - 1e-10:
                    best_local = c
                    best_val = cv
            h[j] = best_val
            if best_val != old_val:
                h = project_box_sum(h, target_sum)
                any_improved = True
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if not any_improved:
            break
    return best_h, best_c5

def slsqp_opt(h0, n, target_sum, deadline, tau=100.0):
    remaining = deadline - time.time()
    if remaining < 15:
        return h0, compute_c5(h0)
    def _obj(x):
        o, _ = grad_soft_max(x, tau, n)
        return o
    def _grad(x):
        _, g = grad_soft_max(x, tau, n)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_box_sum(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def interpolate_h(h_src, n_src, n_dst):
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
