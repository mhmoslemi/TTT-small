    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    lo, hi = -1.5, 1.5
    for _ in range(60):

        if s > target_sum:

Actually, for the continuous case, a single-coordinate move doesn't maintain the constraint. I need to move two coordinates simultaneously: increase h[i] by δ and decrease h[j] by δ (or vice versa).

Let me think about this more carefully. If I increase h[i] by δ, the correlation changes as:
- For each m where h[i] appears in raw[m], raw[m] changes by the derivative * δ
- The derivative d(raw[m])/d(h[i]) is what I computed above

So I can compute the effect of a small perturbation without recomputing the full correlation! This makes coordinate descent very efficient.

For a perturbation δ in direction v (where v has +δ at index i and -δ at index j):
d(C_change) ≈ δ * (grad[i] - grad[j])

where grad is the gradient of the soft-max objective. If grad[i] > grad[j], then decreasing h[i] and increasing h[j] would help (since we want to go in the -grad direction).

So the coordinate descent is essentially: find the pair (i, j) that maximizes grad[i] - grad[j] subject to feasibility (h[i] > 0, h[j] < 1), and move in that direction.

This is actually just the steepest descent direction in the ℓ₁ norm, projected onto the constraint set! And it's equivalent to what the projected gradient method does.

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    # Binary search for the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def soft_max_obj_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    raw = np.correlate(h, omh, mode='full')
    corr = raw * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    val = cmax + eps * np.log(np.sum(np.exp(x)))
    ex = np.exp(x)
    w = ex / np.sum(ex)
    
    # Gradient computation
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m] * dx
        if wk < 1e-14:
            continue
        if m < n:
            # Term 1: p in [0, m], d(raw[m])/d(h[p]) += (1-h[p+n-1-m])
            p1 = np.arange(0, m + 1)
            grad[p1] += wk * omh[p1 + n - 1 - m]
            # Term 2: p in [n-1-m, n-1], d(raw[m])/d(h[p]) -= h[p-(n-1-m)]
            p2 = np.arange(n - 1 - m, n)
            grad[p2] -= wk * h[p2 - (n - 1 - m)]
        else:
            j = m - n + 1
            # Term 1: p in [j, n-1], d(raw[m])/d(h[p]) += (1-h[p-j])
            p1 = np.arange(j, n)
            grad[p1] += wk * omh[p1 - j]
            # Term 2: p in [0, n-1-j], d(raw[m])/d(h[p]) -= h[p+j]
            p2 = np.arange(0, n - j)
            grad[p2] -= wk * h[p2 + j]
    
    return val, grad

def projected_gradient_descent(h, eps, n_steps, lr_init, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        val, grad = soft_max_obj_and_grad(h, eps)
        momentum = beta * momentum - lr * grad
        h_new = project_onto_constraint(h + momentum, target_sum)
        # Clip is already done in project
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.7, 0.001)
        h = h_new
    
    return best_h, best_c5

def coordinate_descent(h, n_passes, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n):
            if h[i] <= 0.01:
                continue
            # Find best j to transfer to
            # Try transferring small amounts
            for j in range(n):
                if j == i or h[j] >= 0.99:
                    continue
                # Try delta = min(h[i], 1-h[j]) * 0.1
                delta = min(h[i] - 0.001, 1.0 - h[j] - 0.001) * 0.1
                if delta <= 0:
                    continue
                h_try = h.copy()
                h_try[i] -= delta
                h_try[j] += delta
                c5 = compute_c5(h_try)
                if c5 < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    
    return h, best_c5

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        return soft_max_obj_and_grad(h, eps)[0]
    
    def jac(h):
        return soft_max_obj_and_grad(h, eps)[1]
    
    def cons_fun(h):
        return float(np.sum(h) - target_sum)
    
    def cons_jac(h):
        return np.ones(n)
    
    try:
        result = minimize(obj, h0, jac=jac, method='SLSQP',
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                         options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except:
        return h0, compute_c5(h0)

from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def soft_max_obj_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    raw = np.correlate(h, omh, mode='full')
    corr = raw * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    val = cmax + eps * np.log(np.sum(np.exp(x)))
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m] * dx
        if wk < 1e-14:
            continue
        if m < n:
            p1 = np.arange(0, m + 1)
            grad[p1] += wk * omh[p1 + n - 1 - m]
            p2 = np.arange(n - 1 - m, n)
            grad[p2] -= wk * h[p2 - (n - 1 - m)]
        else:
            j = m - n + 1
            p1 = np.arange(j, n)
            grad[p1] += wk * omh[p1 - j]
            p2 = np.arange(0, n - j)
            grad[p2] -= wk * h[p2 + j]
    return val, grad

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-15):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    def obj_func(h):
        return soft_max_obj_and_grad(h, eps)[0]
    def jac_func(h):
        return soft_max_obj_and_grad(h, eps)[1]
    def cons_fun(h):
        return float(np.sum(h) - target_sum)
    def cons_jac(h):
        return np.ones(n)
    try:
        result = minimize(obj_func, h0, jac=jac_func, method='SLSQP',
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                         options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient_with_momentum(h, eps, n_steps, lr_init, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    moment = np.zeros(n)
    beta = 0.85
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        val, grad = soft_max_obj_and_grad(h, eps)
        moment = beta * moment - lr * grad
        h_candidate = project_onto_constraint(h + moment, target_sum)
        c5 = compute_c5(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            h = h_candidate
            lr = min(lr * 1.05, 0.3)
        else:
            lr = max(lr * 0.6, 0.0005)
            h = h_candidate
    return best_h, best_c5

def coordinate_polish(h, max_passes, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()

                        improved = True
                        break
