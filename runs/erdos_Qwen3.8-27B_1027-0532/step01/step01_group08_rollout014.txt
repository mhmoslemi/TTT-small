def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

        effect[j, n-1-j : 2*n-1-j] += (1.0 - h)
        effect[j, j : j+n] -= h

    return best_h, best_c5
