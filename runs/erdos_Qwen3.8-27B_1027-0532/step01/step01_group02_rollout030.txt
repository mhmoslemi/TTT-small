```python
import time
import cvxpy as cp

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0

                break

        else:

                break
```

```python
    subgrad = np.zeros(n)
    for k in active_indices:
        s = k - (n - 1)
        if s >= 0:
            # forward: j from 0 to n-1-s
            subgrad[:n-s] += (1.0 - h[s:n]) * dx
            # backward: j from s to n-1
            subgrad[s:] -= h[:n-s] * dx
        else:
            sp = -s
            # forward: j from sp to n-1
            subgrad[sp:] += (1.0 - h[:n-sp]) * dx
            # backward: j from 0 to n-1-sp
            subgrad[:n-sp] -= h[sp:n] * dx
    subgrad /= len(active_indices)
    return subgrad
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.05)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.05)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def slsqp_optimize(h, n, maxiter=500, ftol=1e-13):
    def objective(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr) * dx)
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    try:
        r = minimize(
            objective, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint_fun}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if np.all(r.x >= -1e-8) and np.all(r.x <= 1 + 1e-8):
            r.x = np.clip(r.x, 0.0, 1.0)
            obj = evaluate_c5(r.x)
            if obj < 0.8:
                return r.x, obj
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

def compute_subgradient(h, n, active_indices):

    subgrad = np.zeros(n)
    for k in active_indices:
        s = k - (n - 1)
        if s >= 0:

            subgrad[s:] -= h[:n-s] * dx
        else:

            subgrad[sp:] += (1.0 - h[:n-sp]) * dx
            subgrad[:n-sp] -= h[sp:n] * dx
    subgrad /= float(len(active_indices))

        diff = target - np.sum(h)

            mask = h < 1.0

        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = -diff / mask.sum()
            h[mask] = np.maximum(0.0, h[mask] - step)
    return np.clip(h, 0.0, 1.0)

def subgradient_method(h_init, n, t_start, t_limit, rng):
    h = h_init.copy()
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    curr_c5 = best_c5
    alpha = 0.01
    momentum = 0.7
    v = np.zeros(n)
    stall_count = 0
    min_c5_seen = curr_c5
    
    it = 0

        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        curr_c5 = float(np.max(corr))

            best_h = h.copy()

def smoothing_optimize(h, n, betas=(20, 50, 120, 250, 500), maxiter=200):
    h_curr = h.copy()

            dx = 2.0 / n

            m = np.max(corr)
            val = (m + np.log(np.sum(np.exp(corr - m)))) / beta
            return float(val * dx)

                smoothing_obj, h_curr, method='SLSQP',

                constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}],
                options={'maxiter': maxiter, 'ftol': 1e-11}

                h_curr = np.clip(r.x, 0.0, 1.0)
        except Exception:
```
