   def c5_value(h, dx):

   def repair_h(h, n):
       h = np.asarray(h, dtype=np.float64).copy()
       h = np.clip(h, 0.0, 1.0)
       target = n * 0.5
       # Distribute the difference evenly
       diff = target - np.sum(h)
       if abs(diff) > 1e-14:
           # Add/subtract to non-boundary elements
           if diff > 0:
               mask = h < 1.0 - 1e-10
           else:
               mask = h > 1e-10
           if np.sum(mask) > 0:
               h[mask] += diff / np.sum(mask)
               h = np.clip(h, 0.0, 1.0)
           # Second pass for residuals
           diff = target - np.sum(h)
           if abs(diff) > 1e-14:
               h += diff / n
               h = np.clip(h, 0.0, 1.0)
       return h

   def smooth_max(h, dx, beta):

       m = np.max(corr)
       shifted = beta * (corr - m)
       # Clip for numerical stability
       shifted = np.clip(shifted, -500, 500)
       return float(m + np.log(np.sum(np.exp(shifted))) / beta)
