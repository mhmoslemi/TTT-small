def fista_optimize(h0, n, t, n_iter, dt=None):

    h = project_onto_constraint(h0.astype(float), n)
    y = h.copy()
    t_mom = 1.0
    best_h = h.copy()
    best_obj = compute_c5(h, n)

        dx = 2.0 / n
        corr = np.correlate(y, 1.0 - y, mode="full")
        max_corr = np.max(corr)
        shifted = t * (corr - max_corr)

        sum_exp = np.sum(exp_vals)

            t_mom = 1.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_grad_smoothed(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    mid = n - 1
    ncorr = len(corr)
    for idx in range(ncorr):

        lag = idx - mid

        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])

    return obj, grad

                best_c5 = c5_opt
                best_h = h_opt.copy()
                current_n = n_try

    # Phase 6: Final targeted polish
    if elapsed() < budget_s * 0.96:
        h_pol, c5_pol = subgradient_polish(best_h, current_n, n_iter=5000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

        if elapsed() < budget_s * 0.97:
            h_ml, c5_ml = multi_lag_subgradient(best_h, current_n, n_iter=3000, top_m=2)
            if c5_ml < best_c5:
                best_c5 = c5_ml
                best_h = h_ml.copy()

    # Final verification
    n_final = len(best_h)
    best_h = project_onto_constraint(best_h, n_final)
    best_c5 = compute_c5(best_h, n_final)

    return best_h, best_c5, n_final
