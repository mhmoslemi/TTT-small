                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            s = score_c5(x_new)
            if s < 1.0:
                return x_new, s
    except Exception:
        pass
    return h, score_c5(h)

def try_slsqp(h, n, T, max_iter=2000):
    global _CURRENT_T
    _CURRENT_T = T
    def obj(x):
        return lse_obj_pen(x)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            s = score_c5(x_new)
            if s < 1.0:
                return x_new, s
    except Exception:
        pass
    return h, score_c5(h)

def fast_local_search(h, n, time_limit, t_start):
    best = h.copy()
    best_score = score_c5(best)
    dx = 2.0 / n
    steps = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    step_idx = 0
    stall = 0
    max_stall = 2
    while time.time() - t_start < time_limit and step_idx < len(steps):
        step = steps[step_idx]
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        cmax_val = np.max(corr)
        active_thresh = cmax_val - 0.05 * dx
        active = np.where(corr * dx > active_thresh)[0]
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            old = best[i]
            if old < 1e-9 and 1.0 - old < 1e-9:
                continue
            for direction in [1, -1]:
                new_val = old + direction * step
                if new_val < -0.01 or new_val > 1.01:
                    continue
                trial = best.copy()
                trial[i] = np.clip(new_val, 0.0, 1.0)
                trial = project_feasible(trial, n)
                s = score_c5(trial)
                if s < best_score - 1e-13:
                    best = trial
                    best_score = s
                    improved = True
                    break
        if not improved:
            stall += 1
            if stall >= max_stall:
                step_idx += 1
                stall = 0
        else:
            stall = 0
    return best, best_score

def pair_swap_search(h, n, time_limit, t_start):
    best = h.copy()
    best_score = score_c5(best)
    step = 0.1
    stall = 0
    while time.time() - t_start < time_limit:
        improved = False
        idx_dec = np.where(best > step * 0.5)[0]
        idx_inc = np.where(best < 1.0 - step * 0.5)[0]
        if len(idx_dec) == 0 or len(idx_inc) == 0:
            break
        rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)
        n_try = min(len(idx_dec) * len(idx_inc), 200)
        for _ in range(n_try):
            if time.time() - t_start > time_limit:
                break
            i = idx_dec[rng_local.randint(len(idx_dec))]
            j = idx_inc[rng_local.randint(len(idx_inc))]
            if i == j:
                continue
            amt = min(step, best[i], 1.0 - best[j])
            if amt < 1e-6:
                continue
            trial = best.copy()
            trial[i] -= amt
            trial[j] += amt
            trial = project_feasible(trial, n)
            s = score_c5(trial)
            if s < best_score - 1e-13:
                best = trial
                best_score = s
                improved = True
        if not improved:
            stall += 1
            step *= 0.7
            if stall > 15 or step < 5e-5:
                break
        else:
            stall = 0
    return best, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _CURRENT_T
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = rng.random(85)

    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = score_c5(best_h)

    temps = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for T in temps:
        if time.time() - t0 > budget_s * 0.20:
            break
        x_new, s_new = try_lbfgs(best_h, n, T, 2500)
        if s_new < best_c5 - 1e-15:
            best_c5 = s_new
            best_h = x_new.copy()

    if time.time() - t0 < budget_s * 0.30:
        for T in [0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s * 0.30:
                break
            x_new, s_new = try_slsqp(best_h, n, T, 1500)
            if s_new < best_c5 - 1e-15:
                best_c5 = s_new
                best_h = x_new.copy()

    if time.time() - t0 < budget_s * 0.55:
        tl = budget_s * 0.55 - (time.time() - t0)
        best_h, best_c5 = fast_local_search(best_h, n, tl, t0)

    if time.time() - t0 < budget_s * 0.65:
        for T in [0.005, 0.002, 0.001]:
            if time.time() - t0 > budget_s * 0.65:
                break
            x_new, s_new = try_lbfgs(best_h, n, T, 3000)
            if s_new < best_c5 - 1e-15:
                best_c5 = s_new
                best_h = x_new.copy()

    if time.time() - t0 < budget_s * 0.80:
        tl = budget_s * 0.80 - (time.time() - t0)
        best_h, best_c5 = pair_swap_search(best_h, n, tl, t0)

    if time.time() - t0 < budget_s * 0.92:
        remaining = budget_s * 0.92 - (time.time() - t0)
        n_attempts = int(remaining / 4)
        for attempt in range(min(n_attempts, 80)):
            if time.time() - t0 > budget_s * 0.92:
                break
            h_try = best_h.copy()
            n_pert = rng.randint(1, max(2, n // 8 + 1))
            idx = rng.choice(n, n_pert, replace=False)
            h_try[idx] += rng.uniform(-0.2, 0.2, n_pert)
            h_try = project_feasible(h_try, n)
            x_new, s_new = try_lbfgs(h_try, n, 0.02, 1000)
            if s_new < best_c5 - 1e-14:
                best_c5 = s_new
                best_h = x_new.copy()
                x_new, s_new = try_lbfgs(best_h, n, 0.003, 1500)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = x_new.copy()

    if time.time() - t0 < budget_s * 0.96:
        for n_try in [75, 85, 95, 105, 120]:
            if time.time() - t0 > budget_s * 0.96:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new_pts = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new_pts, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            x_opt, s_opt = try_lbfgs(h_interp, n_try, 0.1, 1500)
            if time.time() - t0 < budget_s * 0.96:
                x_opt, s_opt = try_lbfgs(x_opt, n_try, 0.01, 2000)
            if s_opt < best_c5:
                best_c5 = s_opt
                best_h = x_opt.copy()

    if time.time() - t0 < budget_s - 5:
        x_new, s_new = try_lbfgs(best_h, n, 0.0005, 5000)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = x_new.copy()
        tl = budget_s - 5 - (time.time() - t0)
        if tl > 15:
            best_h, best_c5 = fast_local_search(best_h, n, tl, t0)

    best_h = project_feasible(best_h, n)
    best_c5 = score_c5(best_h)
    return best_h, best_c5, n
