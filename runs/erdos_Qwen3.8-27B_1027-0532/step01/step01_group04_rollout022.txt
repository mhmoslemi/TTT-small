from scipy.optimize import minimize
import time

_current_n = 0
_current_target_sum = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def subgradient_at_lag(h, L, n):
    grad = np.ones(n)
    if L == 0:
        grad -= 2.0 * h
    elif 0 < L < n:
        grad[:n-L] -= h[L:]
        if L < n:
            grad[L:] -= h[:n-L]
    elif -n < L < 0:
        L = -L
        grad[:n-L] -= h[L:]
        grad[L:] -= h[:n-L]
    return grad

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def constraint_eq(h):
    return np.sum(h) - _current_target_sum

def constraint_eq_jac(h):
    return np.ones(_current_n)

    n = len(h)

    v = np.zeros(n)

    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        L_star = k_star - (n - 1)

        h_new = h - lr / (1.0 + 0.001 * it) * v

    global _current_n, _current_target_sum
    _current_n = len(h)
    _current_target_sum = target_sum
    bounds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]
    
    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)

def active_set_solve(h_init, target_sum, max_iters=100):
    n = len(h_init)
    h = h_init.copy()
    
    for it in range(max_iters):
        # Compute active set
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-10)[0]
        active_lags = active - (n - 1)  # convert to actual lags
        
        # Equal weights for active set (simple choice)
        m = len(active)
        w = np.ones(m) / m
        
        # Determine free vs bound variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        at_zero = (h <= 1e-10)
        at_one = (h >= 1.0 - 1e-10)
        
        # Build linear system for free variables
        # sum_i w_i * (1 - h[j+L_i] - h[j-L_i]) + λ = 0 for free j
        # This gives: sum_i w_i * (h[j+L_i] + h[j-L_i]) = sum_i w_i * 1 + λ = 1 + λ
        # (since sum w_i = 1)
        
        # For each free j: sum_i w_i * (h[j+L_i] + h[j-L_i]) = 1 + λ
        # The LHS is linear in h (for the free variables) plus known constants (for bound variables)
        
        free_idx = np.where(free)[0]
        n_free = len(free_idx)
        
        if n_free == 0:
            break
        
        # Build A matrix: A[p, q] = coefficient of h[free_idx[q]] in equation p
        # Equation p (for free_idx[p]): sum_i w_i * (h[free_idx[p]+L_i] + h[free_idx[p]-L_i]) = 1 + λ
        
        A = np.zeros((n_free, n_free + 1))  # +1 for λ
        b = np.ones(n_free)
        
        for p, j in enumerate(free_idx):
            for i, L in enumerate(active_lags):
                # h[j+L]
                j_plus = j + L
                j_minus = j - L
                if 0 <= j_plus < n:
                    if free[j_plus]:
                        q = np.where(free_idx == j_plus)[0][0]
                        A[p, q] += w[i]
                    else:
                        b[p] += w[i] * h[j_plus]
                if 0 <= j_minus < n and j_minus != j_plus:  # avoid double counting when L=0
                    if free[j_minus]:
                        q = np.where(free_idx == j_minus)[0][0]
                        A[p, q] += w[i]
                    else:
                        b[p] += w[i] * h[j_minus]
                elif 0 <= j_minus < n and j_minus == j_plus and L == 0:
                    # Already counted above
                    pass
            # The equation is: A[p,:] * [h_free; λ] = b[p]
            # But we need: sum_i w_i * (h[j+L_i] + h[j-L_i]) - (1 + λ) = 0
            # i.e., sum of contributions - 1 - λ = 0
            # So: A[p,:] * h_free - λ = b[p] - 1
            # Rearranging: A[p,:] * h_free - λ = b[p] - 1
            # Or: A[p,:] * h_free + (-1) * λ = b[p] - 1
            A[p, n_free] = -1.0  # coefficient of λ
            b[p] = b[p] - 1.0
    
        # Add constraint: sum(h) = target_sum
        # sum(h_free) + sum(h_bound) = target_sum
        # sum(h_free) = target_sum - sum(h_bound)
        const_sum = target_sum - (np.sum(h[at_zero]) + np.sum(h[at_one]))
        
        A_ext = np.zeros((n_free + 1, n_free + 1))
        b_ext = b.copy()
        A_ext[:n_free, :] = A
        b_ext = b
        # Last row: sum(h_free) = const_sum (no λ)
        A_ext[n_free, :n_free] = 1.0
        A_ext[n_free, n_free] = 0.0
        b_ext[n_free] = const_sum
        
        # Solve
        try:
            sol = np.linalg.solve(A_ext, b_ext)
        except np.linalg.LinAlgError:
            break
        
        h_free_new = sol[:n_free]
        
        # Update h
        h_new = h.copy()
        h_new[free_idx] = h_free_new
        
        # Check if solution is valid
        if np.any(h_free_new < -1e-10) or np.any(h_free_new > 1 + 1e-10):
            # Solution is outside [0,1], need to update active set
            # Move variables that violated to their bounds
            for p, j in enumerate(free_idx):
                if h_free_new[p] < -1e-10:
                    h_new[j] = 0.0
                elif h_free_new[p] > 1 + 1e-10:
                    h_new[j] = 1.0
        
        # Project to ensure exact feasibility
        h_new = project_feasible(h_new, target_sum)
        
        # Check convergence
        if np.max(np.abs(h_new - h)) < 1e-12:
            h = h_new
            break
        
        h = h_new
    
    return h

The inner min is over exponentially many vertices. But if we restrict to a finite set of candidate vertices, we can solve it as an LP.
