                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def local_coord_search(h, n_inner, rng, target_sum):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_inner):
        improved = False
        order = rng.permutation(n)
        for i in order:
            for delta in [0.03, 0.07, 0.12, 0.2]:
                for sign in [1, -1]:
                    h_try = best_h.copy()
                    h_try[i] = np.clip(h_try[i] + sign * delta, 0.0, 1.0)
                    h_try = project_onto_constraint(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
    return best_h, best_c5


def binary_nudge_search(h, n_trials, rng, target_sum):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = best_h.copy()
        n_flip = rng.integers(1, max(2, n // 8))
        idx = rng.choice(n, size=n_flip, replace=False)
        for i in idx:
            h_try[i] = 1.0 - h_try[i]
        h_try = project_onto_constraint(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def simulated_annealing(h_init, n_iter, rng, target_sum):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    T_start = 0.008
    T_end = 0.00005
    if n_iter <= 0:
        return best_h, best_c5
    cooling = (T_end / T_start) ** (1.0 / n_iter)
    T = T_start
    for _ in range(n_iter):
        T *= cooling
        n_pert = rng.integers(1, max(2, n // 12))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        for i in idx:
            h_try[i] = np.clip(h_try[i] + rng.uniform(-0.4, 0.4), 0.0, 1.0)
        h_try = project_onto_constraint(h_try, target_sum)
        c5_try = compute_c5(h_try)
        delta = c5_try - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_try
            c5_cur = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _TARGET_SUM, _TARGET_ONES
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 90
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    _TARGET_SUM = target_sum
    _TARGET_ONES = np.ones(n0)
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    eps_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for eps in eps_schedule:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 45:
            break
        h_new, c5_new = run_slsqp(best_h, eps, 400, n0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        best_h, best_c5 = local_coord_search(best_h, 40, rng, target_sum)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        h_bn, c5_bn = binary_nudge_search(best_h, 300, rng, target_sum)
        if c5_bn < best_c5:
            best_c5 = c5_bn
            best_h = h_bn

    elapsed = time.time() - start_time
    if budget_s - elapsed > 150:
        h_sa, c5_sa = simulated_annealing(best_h, 3000, rng, target_sum)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        eps_post = [0.005, 0.002, 0.001, 0.0003]
        for eps in eps_post:
            if time.time() - start_time > budget_s - 30:
                break
            h_new, c5_new = run_slsqp(best_h, eps, 250, n0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    restart_count = 0
    while budget_s - (time.time() - start_time) > 70 and restart_count < 30:
        strategy = restart_count % 7
        if strategy == 0:
            h_pert = best_h + 0.04 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_h + 0.1 * (rng.random(n0) - 0.5)
        elif strategy == 2:
            block_start = rng.integers(0, n0 - 10)
            block_len = rng.integers(3, 16)
            h_pert = best_h.copy()
            h_pert[block_start:block_start + block_len] = 1.0 - h_pert[block_start:block_start + block_len]
        elif strategy == 3:
            freq = rng.uniform(1.5, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            x_grid = np.linspace(0, 2, n0)
            h_pert = 0.5 + 0.35 * np.sin(freq * np.pi * x_grid + phase)
        elif strategy == 4:
            h_bin = (rng.random(n0) < 0.5).astype(float)
            h_bin = project_onto_constraint(h_bin, target_sum)
            alpha = rng.uniform(0.6, 0.9)
            h_pert = alpha * best_h + (1.0 - alpha) * h_bin
        elif strategy == 5:
            h_pert = best_h.copy()
            nswap = rng.integers(2, max(3, n0 // 6))
            idx = rng.choice(n0, size=2 * nswap, replace=False)
            for j in range(0, 2 * nswap, 2):
                h_pert[idx[j]], h_pert[idx[j + 1]] = h_pert[idx[j + 1]], h_pert[idx[j]]
        else:
            h_pert = best_h + 0.02 * rng.standard_normal(n0)
            h_pert = np.clip(h_pert, 0.0, 1.0)

        h_pert = project_onto_constraint(h_pert, target_sum)
        eps_restart = [0.008, 0.003, 0.001]
        for eps in eps_restart:
            if time.time() - start_time > budget_s - 30:
                break
            h_new, c5_new = run_slsqp(h_pert, eps, 200, n0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

        if time.time() - start_time < budget_s - 20:
            h_ls, c5_ls = local_coord_search(best_h, 10, rng, target_sum)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

        restart_count += 1

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_sa2, c5_sa2 = simulated_annealing(best_h, 1500, rng, target_sum)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2

    elapsed = time.time() - start_time
    if budget_s - elapsed > 50:
        for eps in [0.0001, 0.00005, 0.00002]:
            if time.time() - start_time > budget_s - 15:
                break
            h_new, c5_new = run_slsqp(best_h, eps, 300, n0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    if time.time() - start_time < budget_s - 5:
        best_h, best_c5 = local_coord_search(best_h, 20, rng, target_sum)

    final_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
