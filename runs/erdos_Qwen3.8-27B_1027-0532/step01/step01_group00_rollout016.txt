import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_logsumexp(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_max_logsumexp_grad(x, alpha, corr):
    # Returns gradient of smooth_max with respect to h
    # x = corr * dx (the scaled correlations)
    # Actually let's work with unscaled corr
    m = np.max(x)
    weights = np.exp(alpha * (x - m))
    weights /= np.sum(weights)
    return weights  # These will be used to weight the individual gradients

def smooth_max_power(x, p):
    # (1/m * sum(x^p))^(1/p)
    m = len(x)
    if np.any(x < 0):
        # Handle negative values - clip
        x = np.maximum(x, 0)
    return (np.sum(x**p) / m) ** (1.0/p)

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def correlation_gradient(h, s):
    """Compute gradient of c_s = sum_i h[i]*(1-h[i+s]) with respect to h.
    s is the shift (k - (n-1))."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        valid_i = np.arange(0, n-s)
    else:
        valid_i = np.arange(-s, n)
    
    # First term: (1-h[i+s]) for i in valid range → contributes to grad[i]
    grad[valid_i] += (1.0 - h[valid_i + s])
    
    # Second term: -h[i] for i = j-s, j in [0,n-1], i in valid range
    # i = j-s must be in valid range, so j = i+s must be in [0, n-1]
    j = valid_i + s
    valid_j = (j >= 0) & (j < n)
    grad[j[valid_j]] -= h[valid_i[valid_j]]
    
    return grad

def fw_subproblem(grad, n):
    """Solve min <grad, h> s.t. 0<=h<=1, sum(h)=n/2.
    Solution: h[i]=1 for the n/2 smallest grad[i], h[i]=0 for rest."""
    n_ones = n // 2
    # Need to handle the case where n is odd (sum = n/2 not integer)
    # For now, assume n is even
    idx = np.argsort(grad)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1.0
    return h

def fw_optimize(h0, max_iter=1000, tol=1e-10):
    """Frank-Wolfe optimization for the Erdős problem."""
    n = len(h0)
    h = h0.copy()
    target = n / 2.0
    
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    for t in range(max_iter):
        corr = compute_correlations(h)
        dx = 2.0 / n
        c_scaled = corr * dx
        
        # Find max lag
        k_max = np.argmax(c_scaled)
        s = k_max - (n - 1)
        
        # Compute gradient of c_{k_max}
        g = correlation_gradient(h, s)
        
        # Solve FW subproblem
        h_v = fw_subproblem(g, n)
        
        # Step size
        gamma = 2.0 / (t + 2)
        
        h_new = (1 - gamma) * h + gamma * h_v
        h_new = project_sum(h_new, target)
        
        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h = h_new
        
        if t % 100 == 0 and score < best_score * (1 + 1e-12):
            # Stagnation check
            if t > 100:
                break
    
    return best_h, best_score

def pair_swap_optimize(h, i, j, corr):
    """Optimize the mass transfer t between coordinates i and j."""
    n = len(h)
    s = np.arange(2*n-1) - (n-1)
    
    # Compute B for coordinate i and j
    idx1_i = (i + s >= 0) & (i + s < n)
    idx2_i = (i - s >= 0) & (i - s < n)
    Bi = np.zeros(2*n-1)
    Bi[idx1_i] = (1.0 - h[i + s[idx1_i]])
    Bi[idx2_i] -= h[i - s[idx2_i]]
    
    idx1_j = (j + s >= 0) & (j + s < n)
    idx2_j = (j - s >= 0) & (j - s < n)
    Bj = np.zeros(2*n-1)
    Bj[idx1_j] = (1.0 - h[j + s[idx1_j]])
    Bj[idx2_j] -= h[j - s[idx2_j]]
    
    D = Bi - Bj  # c_k(t) = c_k + t * D[k]
    
    # Feasible range for t:
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    
    if t_lo >= t_hi:
        return 0.0  # No feasible movement
    
    # Minimize max_k(corr[k] + t*D[k]) over t ∈ [t_lo, t_hi]
    # Grid + refine
    t_grid = np.linspace(t_lo, t_hi, 21)
    f_vals = np.max((corr[None, :] + t_grid[:, None] * D[None, :]) * (2.0/n), axis=1)
    t_best = t_grid[np.argmin(f_vals)]
    
    # Golden section refinement
    a, b = t_lo, t_hi
    for _ in range(50):
        m1 = a + (b - a) / (1 + np.sqrt(5))
        m2 = b - (b - a) / (1 + np.sqrt(5))
        f1 = np.max(corr + m1 * D)

            a = m1
        else:

    return t_opt

    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_shifts(n):
    """Precompute the shift array."""
    return np.arange(2 * n - 1) - (n - 1)

def compute_B(h, i, shifts):
    """Compute the coefficient vector B for coordinate i."""
    n = len(h)
    idx1 = (i + shifts >= 0) & (i + shifts < n)
    idx2 = (i - shifts >= 0) & (i - shifts < n)
    B = np.zeros(2 * n - 1)
    if np.any(idx1):
        B[idx1] = (1.0 - h[i + shifts[idx1]])
    if np.any(idx2):
        B[idx2] -= h[i - shifts[idx2]]
    return B

def pair_swap_optimize(h, i, j, corr, shifts):
    """Optimize mass transfer t between i and j. Returns (t_opt, D)."""
    Bi = compute_B(h, i, shifts)
    Bj = compute_B(h, j, shifts)
    D = Bi - Bj
    
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    
    if t_lo >= t_hi - 1e-15:
        return 0.0, D
    
    # Grid + golden section
    t_grid = np.linspace(t_lo, t_hi, 21)
    f_vals = np.max(corr[None, :] + t_grid[:, None] * D[None, :], axis=1)
    idx_best = np.argmin(f_vals)
    
    # Narrow the interval
    if idx_best == 0:
        a, b = t_lo, t_grid[1]
    elif idx_best == len(t_grid) - 1:
        a, b = t_grid[-2], t_hi
    else:
        a = max(t_lo, t_grid[idx_best - 1])
        b = min(t_hi, t_grid[idx_best + 1])
    
    # Golden section
    for _ in range(60):
        m1 = a + (b - a) / (1 + np.sqrt(5))
        m2 = b - (b - a) / (1 + np.sqrt(5))
        f1 = np.max(corr + m1 * D)
        f2 = np.max(corr + m2 * D)
        if f1 > f2:
            a = m1
        else:
            b = m2
        if b - a < 1e-14:
            break
    
    t_opt = (a + b) / 2.0
    return t_opt, D

def coordinate_descent_pairs(h, max_sweeps=50, seed=42):
    """Pair-swap coordinate descent."""
    n = len(h)
    h = h.copy()
    shifts = compute_shifts(n)
    rng = np.random.RandomState(seed)
    
    best_local_score = evaluate_score(h)
    
    for sweep in range(max_sweeps):
        corr = compute_correlations(h)
        
        # Random permutation of indices for pair selection
        perm = rng.permutation(n)
        
        improved = 0
        for idx in range(n):
            i = perm[idx]
            j = perm[(idx + 1) % n]  # Pair with next in permutation
            if i == j:
                continue
            t_opt, D = pair_swap_optimize(h, i, j, corr, shifts)
            if abs(t_opt) > 1e-12:
                corr += t_opt * D
                h[i] += t_opt
                h[j] -= t_opt
                improved += 1
        
        score = evaluate_score(h)
        if score < best_local_score:
            best_local_score = score
        
        if improved == 0:
            break
    
    return h, best_local_score

            res = minimize(slsqp_objective, best_h, args=(alpha, n_cur),
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
            if res.x is not None:
                h_new = project_sum(np.clip(res.x, 0.0, 1.0), n_cur / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 4: Another coordinate descent round
    if time.time() - t0 < budget_s - 90:
        h_cd, score_cd = coordinate_descent_pairs(best_h, 300, seed + 100)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd.copy()

    # Phase 5: Try perturbation + local search to escape local min
    if time.time() - t0 < budget_s - 90:
        n_cur = len(best_h)
        for attempt in range(30):
            if time.time() - t0 > budget_s - 90:
                break
            scale = 0.05 + 0.15 * (attempt / 30.0)
            pert = best_h + rng.randn(n_cur) * scale
            pert = project_sum(pert, n_cur / 2.0)
            h_cd, score_cd = coordinate_descent_pairs(pert, 100, seed + 200 + attempt)
            if score_cd < best_score:
                best_score = score_cd
                best_h = h_cd.copy()

    # Phase 6: Try different n values
    if time.time() - t0 < budget_s - 60:
        for n_try in [424, 426, 450, 500]:
            if time.time() - t0 > budget_s - 60:
                break
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = project_sum(h_init, n_try / 2.0)
            h_cd, score_cd = coordinate_descent_pairs(h_init, 100, seed + 500)
            if score_cd < best_score:
                best_score = score_cd
                best_h = h_cd.copy()
                # Refine with SLSQP
                if time.time() - t0 < budget_s - 60:
                    bounds = [(0.0, 1.0)] * n_try
                    constraints = [{'type': 'eq', 'fun': slsqp_constraint_fun,
                                    'jac': slsqp_constraint_jac, 'args': (n_try,)}]
                    res = minimize(slsqp_objective, best_h, args=(5000, n_try),
                                   method='SLSQP', bounds=bounds, constraints=constraints,
                                   options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0})
                    if res.x is not None:
                        h_new = project_sum(np.clip(res.x, 0.0, 1.0), n_try / 2.0)
                        s = evaluate_score(h_new)
                        if s < best_score:
                            best_score = s
                            best_h = h_new.copy()

    # Phase 7: Final polish
    if time.time() - t0 < budget_s - 30:
        h_final, score_final = coordinate_descent_pairs(best_h, 200, seed + 999)
        if score_final < best_score:
            best_score = score_final
            best_h = h_final.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
