    best = h.copy()
    best_score = score_c5(best)
    step = 0.1
    alpha = 0.9  # momentum
    v = np.zeros(n)  # velocity
    patience = 0
    max_patience = 500
    
    while time.time() - t_start < time_limit:
        # Compute subgradient
        dx = 2.0 / n
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_val = np.max(corr)
        
        # Active lags
        tol = 1e-4 * max_val
        active = np.where(corr >= max_val - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            # ∂f_k/∂h[j] = (1-h[j-m]) * 1{j-m in [0,n-1]} - h[j+m] * 1{j+m in [0,n-1]}
            for j in range(n):
                gm = j - m
                pm = j + m
                term = 0.0
                if 0 <= gm < n:
                    term += (1.0 - best[gm])
                if 0 <= pm < n:
                    term -= best[pm]
                grad[j] += term
        
        grad *= dx / len(active)
        
        # Momentum step
        v = alpha * v - step * grad
        new_h = best + v
        new_h = project_feasible(new_h)
        new_score = score_c5(new_h)
        
        if new_score < best_score:
            best = new_h
            best_score = new_score
            patience = 0
            step = min(step * 1.05, 0.5)
        else:
            patience += 1
            step *= 0.95
            if patience > max_patience:
                break
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, n, active_indices):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_indices:
        m = k - (n - 1)
        if m == 0:
            grad += 1.0 - 2.0 * h
        elif m > 0:
            grad[m:] += (1.0 - h[:-m])
            grad[:n-m] -= h[m:]
        else:
            mm = -m
            grad[:n-mm] += (1.0 - h[mm:])
            grad[mm:] -= h[:n-mm]
    return dx * grad / len(active_indices)

def get_active_lags(h, n, tol=1e-4):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    return np.where(corr >= max_val - tol * max_val)[0], max_val

def subgrad_momentum(h, n, time_limit, t_start, initial_step=0.2):
    best = h.copy()
    best_score = score_c5(best)
    step = initial_step
    beta = 0.9  # momentum
    v = np.zeros(n)
    stall = 0
    max_stall = 200
    
    while time.time() - t_start < time_limit:
        active, _ = get_active_lags(best, n, tol=1e-3)
        grad = compute_subgradient(best, n, active)
        
        v = beta * v - step * grad
        trial = best + v
        trial = project_feasible(trial)
        trial_score = score_c5(trial)
        
        if trial_score < best_score - 1e-12:
            best = trial
            best_score = trial_score
            stall = 0
            step = min(step * 1.1, 1.0)
        else:
            stall += 1
            step *= 0.9
            v *= 0.5
            if stall > max_stall or step < 1e-8:
                break
    
    return best, best_score

def randomized_local_search(h, n, time_limit, t_start, rng):
    best = h.copy()
    best_score = score_c5(best)
    step_size = 0.1
    stall = 0
    
    while time.time() - t_start < time_limit and step_size > 1e-6:
        improved = False
        # Random pair swap
        for _ in range(min(200, n)):
            if time.time() - t_start > time_limit:
                break
            i, j = rng.choice(n, 2, replace=False)
            trial = best.copy()
            amt = step_size * rng.random()
            if best[i] >= amt and best[j] <= 1.0 - amt:
                trial[i] -= amt
                trial[j] += amt
            elif best[i] <= 1.0 - amt and best[j] >= amt:
                trial[i] += amt
                trial[j] -= amt
            else:
                continue
            trial = project_feasible(trial)
            s = score_c5(trial)
            if s < best_score - 1e-12:
                best = trial
                best_score = s
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            step_size *= 0.8
            if stall > 100:
                break
    
    return best, best_score

def lse_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Soft max approximation
    shifted = corr - cmax
    logsumexp = cmax + T * np.log(np.sum(np.exp(shifted / T)))
    return dx * logsumexp

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def sum_jacobian(h):
    return np.ones(len(h))

def slsqp_refine(h, n, T, max_iter=500):
    try:
        res = minimize(lse_objective, h, args=(T,), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian}],
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            x = project_feasible(res.x)
            return x, score_c5(x)
    except Exception:
        pass
    return h, score_c5(h)

    if mode == 0:

        h = np.zeros(n)
        block = n // 3
        h[:block] = 1.0

        h = 0.5 + 0.3 * np.sin(2 * np.pi * x + rng.random())
        return project_feasible(h)
    elif mode == 2:

        h = 0.5 + 0.2 * np.sin(2*np.pi*x) + 0.15*np.sin(4*np.pi*x + rng.random())
        return project_feasible(h)

        # Smooth it

        h = np.convolve(h, kernel, mode='same')
        return project_feasible(h)
    else:
        # Pure random
        return project_feasible(rng.random(n))

    temps = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

        if s < best_c5:

            best_h = x.copy()
    
    # Phase 2: Subgradient momentum (30% - 55% budget)
    if time.time() < t0 + budget_s * 0.55:
        tl = t0 + budget_s * 0.55 - time.time()
        best_h, best_c5 = subgrad_momentum(best_h, n, tl, t0, initial_step=0.3)
    
    # Phase 3: Try different resolutions and initializations (55% - 80% budget)

                # Quick SLSQP
                x, s = slsqp_refine(h_try, n_try, 1.0, 200)

                # Quick subgradient
                x, s = subgrad_momentum(x, n_try, min(15, budget_s*0.02), t0, 0.2)

                    best_h = x.copy()
                    n = n_try

        x, s = slsqp_refine(best_h, n, 0.005, 500)

                best_h = x.copy()
    
    best_h = project_feasible(best_h)
    best_c5 = score_c5(best_h)
    return best_h, best_c5, len(best_h)

        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid

    max_val = np.max(corr)
    threshold = max_val - tol_frac * max_val

def lse_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    if T < 1e-10:
        return dx * cmax
    shifted = (corr - cmax) / T
    # Numerically stable log-sum-exp
    max_shift = np.max(shifted)
    lse = cmax + T * (max_shift + np.log(np.sum(np.exp(shifted - max_shift))))
    return dx * lse

    return np.sum(h) - len(h) / 2.0

    return np.ones(len(h))

            x = project_feasible(res.x)
