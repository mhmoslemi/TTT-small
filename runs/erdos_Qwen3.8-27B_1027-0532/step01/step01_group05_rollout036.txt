def binary_coordinate_descent(h, n, max_iter=10000):
    """Minimize max overlap for binary h using coordinate descent."""
    # Convert to integer array for speed
    h = h.astype(np.int8).copy()
    # Compute initial overlap

        if s >= 0:

        else:

            overlap[idx] = np.sum(h[l:] * (1 - h[:n-l]))

            best_h = h_new.copy()
        h = h_new

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    
    # Fast gradient via FFT

    w_flipped = weights[::-1]  # reversed weights, length 2n-1
    conv = np.convolve(g, w_flipped, mode='full')  # length 3n-2
    grad = dx * conv[n-1:2*n-1][::-1]
    
    return obj, grad
