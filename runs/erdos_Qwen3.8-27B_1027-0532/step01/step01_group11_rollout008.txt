from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def objective_exact(h):
    return compute_c5(h)

def smooth_objective(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):

    sexp = np.sum(ex)
    w = ex / sexp

        else:
            r = -s

                grad[:n - r] += wk * dx * omh[r:]

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, eps):
    corr = compute_corr_values(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):

    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:

                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for step in range(n_steps):
        corr = compute_corr_values(h)
        cmax = np.max(corr)
        tol = 1e-8 + 2e-4 * cmax
        sorted_idx = np.argsort(-corr)
        active = sorted_idx[:top_m]
        active = active[corr[active] >= cmax - tol]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)

            s = k_idx - (n - 1)
            if s >= 0:
                if s < n:
                    grad[s:] += dx * (1.0 - h[:n - s])
                    grad[:n - s] -= dx * h[s:]
            else:
                r = -s
