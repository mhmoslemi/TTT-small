    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_gradient(h, beta):
    """Compute gradient of the smooth max objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    corr_scaled = beta * corr
    m = np.max(corr_scaled)
    w = np.exp(corr_scaled - m)
    w /= np.sum(w)
    # Compute C[j] = sum_m w[j+m] * h[m]
    corr_w = np.correlate(w[::-1], h, mode='full')
    C_array = corr_w[n-1:2*n-1][::-1]
    g = dx * (1.0 - 2.0 * C_array)
    return g

def project_feasible(h, n):

    target = n / 2.0

    n = len(h)

    return float(np.max(corr) * dx)

def compute_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p_sum = np.sum(p)
    if p_sum == 0:
        return np.zeros(n)
    p = p / p_sum
    # S[j] = sum_{k=j}^{j+n-1} p[k]
    S = np.convolve(p, np.ones(n), mode='valid')
    # C[j] = sum_{m=0}^{n-1} p[j+m] * h[m]
    C = np.correlate(p[::-1], h, mode='full')[n-1:2*n-1][::-1]
    return dx * (S - 2.0 * C)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-12
        else:
            mask = h_proj > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def pgd_optimize(h, n, beta, max_iter=500, tol=1e-12):

    f_curr = evaluate_c5(h_curr)

        g = compute_smooth_gradient(h_curr, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < tol:
            break

        accepted = False
        step_trial = step
        for _ in range(60):
            h_new = project_feasible(h_curr - step_trial * g, n)
            f_new = evaluate_c5(h_new)

                f_curr = f_new
                step = min(step_trial * 1.5, 1.0 / (beta * dx + 1e-10))
                accepted = True
                break
            step_trial *= 0.5

from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    S = np.convolve(p, np.ones(n), mode='valid')
    C = np.correlate(p[::-1], h, mode='full')[n-1:2*n-1][::-1]
    return dx * (S - 2.0 * C)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-12
        else:
            mask = h_proj > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
    return np.clip(h_proj, 0.0, 1.0)
