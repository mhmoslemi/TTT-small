    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-14
    else:
        mask = h > 1e-14
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def lse_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    active = np.where(weights > 1e-5)[0]
    for k in active:
