def project_sum(h, target):
    """
    h: ndarray, current vector of function values
    target: scalar, target sum

    n = len(h)
    h = h.copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
