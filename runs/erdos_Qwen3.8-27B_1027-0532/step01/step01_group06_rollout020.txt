Let me simplify: I'll do **pairwise coordinate descent**:
- Pick a pair (j, m)
- Transfer δ from h[j] to h[m] (or vice versa)
- Find optimal δ by minimizing the resulting max

For a transfer of δ from j to m:
h[j] → h[j] - δ, h[m] → h[m] + δ

c_k changes by:
Δc_k = -δ * [(1-h[j-k])*v1(j,k) - h[j+k]*v2(j,k)] + δ * [(1-h[m-k])*v1(m,k) - h[m+k]*v2(m,k)]
= δ * [s(m,k) - s(j,k)]

where s(j,k) = (1-h[j-k])*v1(j,k) - h[j+k]*v2(j,k)

So c_k(δ) = c_k(h) + δ * (s(m,k) - s(j,k))

And we want to minimize max_k c_k(δ) over δ.

This is a convex piecewise-linear function of δ, and we can find its minimum efficiently.

The minimum of max_k (a_k + b_k * δ) occurs at a point where at least two of the (a_k + b_k * δ) are equal and maximal, or at a boundary (δ such that h[j]=0 or h[m]=1, etc.).

To find it: sort the breakpoints where the active k changes, and evaluate at each.

But with 2n-1 values of k, this could be expensive. However, for practical purposes, we can:
1. Compute the current max and which k's are near-max
2. Only consider those k's for the optimization
3. Find the optimal δ by considering intersections of the top few lines

This is essentially a 1D linear programming problem: minimize t subject to a_k + b_k * δ ≤ t for all k.

```python
def optimal_transfer(h, j, m, n):
    """Find optimal δ to transfer from h[j] to h[m]."""
    corr = np.correlate(h, 1-h, mode='full')
    # For the top k's (say top 50), compute the slope
    # and find the optimal δ
    
    top_k = np.argsort(corr)[-50:]  # indices of top 50 correlations
    a = corr[top_k].copy()
    b = np.zeros(50)
    
    for idx, k_idx in enumerate(top_k):
        k = k_idx - (n - 1)
        # s(j, k) and s(m, k)
        s_j = 0.0
        s_m = 0.0
        # v1(j,k): j-k valid (0 <= j-k < n) AND j valid (always)
        if 0 <= j - k < n:
            s_j += (1.0 - h[j - k])
        # v2(j,k): j+k valid (0 <= j+k < n) AND j valid (always)
        if 0 <= j + k < n:
            s_j -= h[j + k]
        if 0 <= m - k < n:
            s_m += (1.0 - h[m - k])
        if 0 <= m + k < n:
            s_m -= h[m + k]
        b[idx] = s_m - s_j
        a[idx] = corr[k_idx]  # base value
    
    # Minimize max_k (a_k + b_k * δ)
    # Subject to: 0 <= h[j] - δ and h[m] + δ <= 1 and h[j] - δ >= 0 and h[m] + δ <= 1
    # i.e., max(-h[m], -h[j])... wait:
    # h[j] - δ >= 0 → δ <= h[j]
    # h[j] - δ <= 1 → δ >= h[j] - 1 (always true since h[j] <= 1)
    # h[m] + δ >= 0 → δ >= -h[m] (always true since h[m] >= 0)
    # h[m] + δ <= 1 → δ <= 1 - h[m]
    
    delta_lo = max(0.0, -h[m])  # = 0
    delta_hi = min(h[j], 1.0 - h[m])
    
    # Binary search on δ to find where the max is minimized
    # The function f(δ) = max_k (a_k + b_k * δ) is convex in δ
    # Use ternary search or binary search on the derivative
    
    # Simple approach: evaluate at a grid and refine
    # Or: the minimum of a convex PL function occurs where the slope changes sign
    # Slope of f at δ: f'(δ) = max{b_k : a_k + b_k*δ = f(δ)}
    # (subgradient is the max b_k among active lines)
    
    # Let's just do a fine grid search in [delta_lo, delta_hi]
    # Since delta_hi is at most 1, a grid of 200 points should work
    grid = np.linspace(delta_lo, delta_hi, 200)
    vals = a + np.outer(b, grid)  # shape (50, 200)
    f_vals = np.max(vals, axis=0)
    best_idx = np.argmin(f_vals)
    best_delta = grid[best_idx]
    
    # Refine with a finer grid around the best point
    if 0 < best_idx < len(grid) - 1:
        fine_grid = np.linspace(grid[best_idx - 1], grid[best_idx + 1], 100)
        vals = a + np.outer(b, fine_grid)
        f_vals = np.max(vals, axis=0)
        best_delta = fine_grid[np.argmin(f_vals)]
    
    return best_delta
```

For a given pair (j, m), the problem is:

To find it efficiently:
- The function f(δ) = max_k (a_k + b_k * δ) is convex PL
- Its minimum is where the slope changes from negative to positive
- The slope at δ is: max{b_k : k is active at δ} - min{b_k : k is active at δ} (subgradient)
- Actually, the right-derivative is max{b_k : a_k + b_k*δ = f(δ)} and left-derivative is min{b_k : a_k + b_k*δ = f(δ)}
- The minimum is where the right-derivative ≥ 0 and left-derivative ≤ 0

A simple approach: binary search on δ using the slope.
- At the current δ, find which k's are active (within tolerance)
- Compute the slope (max b_k among active)
- If slope < 0, increase δ; if slope > 0, decrease δ
- The minimum is where slope changes sign

```python
def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_slopes(h, j, m, n):
    """Compute b_k = s(m,k) - s(j,k) for all k."""
    corr_len = 2 * n - 1
    b = np.zeros(corr_len)
    for k_idx in range(corr_len):
        k = k_idx - (n - 1)
        s_j = 0.0
        s_m = 0.0
        if 0 <= j - k < n:
            s_j += (1.0 - h[j - k])
        if 0 <= j + k < n:
            s_j -= h[j + k]
        if 0 <= m - k < n:
            s_m += (1.0 - h[m - k])
        if 0 <= m + k < n:
            s_m -= h[m + k]
        b[k_idx] = s_m - s_j
    return b

    """Find δ ∈ [0, delta_max] minimizing max_k (a_k + b_k * δ)."""
    if delta_max <= 0:
        return 0.0
    
    # Ternary search
    lo = 0.0
    hi = delta_max
    for _ in range(60):

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a + b * m1)
        f2 = np.max(a + b * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
```

```python
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def get_b(h, j, m, n, k_vals):
    jm = j - k_vals
    jp = j + k_vals
    mm = m - k_vals
    mp = m + k_vals
    v1j = (jm >= 0) & (jm < n)
    v2j = (jp >= 0) & (jp < n)
    v1m = (mm >= 0) & (mm < n)
    v2m = (mp >= 0) & (mp < n)
    s_j = np.where(v1j, 1.0 - h[np.clip(jm, 0, n-1)], 0.0) - np.where(v2j, h[np.clip(jp, 0, n-1)], 0.0)
    s_m = np.where(v1m, 1.0 - h[np.clip(mm, 0, n-1)], 0.0) - np.where(v2m, h[np.clip(mp, 0, n-1)], 0.0)
    return s_m - s_j

def find_optimal_delta(corr, b, delta_max, n_iter=40):
    lo = 0.0
    hi = delta_max
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + b * m1) <= np.max(corr + b * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h_in, n, rng, deadline):
    h = h_in.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    best_max = current_max
    k_vals = np.arange(-(n-1), n)

                    h[m] += delta
                    corr += b * delta
```
