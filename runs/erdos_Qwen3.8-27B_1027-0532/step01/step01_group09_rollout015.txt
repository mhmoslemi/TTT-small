```python
def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    max_c = corr.max()
    exp_shift = np.exp(beta * (corr - max_c))
    exp_sum = exp_shift.sum()
    val = max_c + np.log(exp_sum) / beta
    weights = exp_shift / exp_sum
    
    s = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)
    js = j_idx[:, None] + s[None, :]
    js2 = j_idx[:, None] - s[None, :]
    mask1 = (js >= 0) & (js < n)
    mask2 = (js2 >= 0) & (js2 < n)
    h_js = np.where(mask1, h[np.clip(js, 0, n - 1)], 0.0)
    h_js2 = np.where(mask2, h[np.clip(js2, 0, n - 1)], 0.0)
    grad = dx * (np.sum(weights[None, :] * (1.0 - h_js) * mask1, axis=1) -
                 np.sum(weights[None, :] * h_js2 * mask2, axis=1))
    return val, grad

    n = len(h0)

    y = h.copy()  # Nesterov
    grad_prev = np.zeros(n)

    t0 = time.time()
    lr = 0.5  # initial step size

        if time.time() - t0 > tm:
            break
        beta = beta_start + (beta_end - beta_start) * (it / max(1, n_iter - 1))
        
        # Nesterov lookahead
        val, grad = smooth_obj_grad(y, beta)
        
        # Update Nesterov momentum

        # Actually standard Nesterov: y_{k+1} = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
        # Simplified: use momentum beta = (t_k - 1) / (t_{k+1})

        step = lr
        val_y, _ = smooth_obj_grad(h - step * grad, beta)
        while val_y > val + 0.1 * step * np.dot(grad, -grad) and step > 1e-10:
            step *= 0.5
            val_y, _ = smooth_obj_grad(h - step * grad, beta)

        y = x_new + momentum * (x_new - h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    max_c = corr.max()
    exp_shift = np.exp(beta * (corr - max_c))
    exp_sum = exp_shift.sum()
    val = max_c + np.log(exp_sum) / beta
    weights = exp_shift / exp_sum
    s = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)
    js = j_idx[:, None] + s[None, :]
    js2 = j_idx[:, None] - s[None, :]
    mask1 = (js >= 0) & (js < n)
    mask2 = (js2 >= 0) & (js2 < n)
    h_js = np.where(mask1, h[np.clip(js, 0, n - 1)], 0.0)
    h_js2 = np.where(mask2, h[np.clip(js2, 0, n - 1)], 0.0)
    grad = dx * (np.sum(weights[None, :] * (1.0 - h_js) * mask1, axis=1) -
                 np.sum(weights[None, :] * h_js2 * mask2, axis=1))
    return val, grad

def smoothed_grad_descent(h0, beta_start, beta_end, n_iter, tm):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    y = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    lr = 0.3
    for it in range(n_iter):
        if time.time() - t0 > tm:
            break
        frac = it / max(1, n_iter - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        val, grad = smooth_obj_grad(y, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        step = lr / (it + 1) ** 0.3
        x_trial = project_feasible(h - step * grad, tgt)
        val_trial = smooth_obj_grad(x_trial, beta)[0]
        cap = 10
        while val_trial > val - 1e-14 * step * grad_norm ** 2 and step > 1e-12 and cap > 0:
            step *= 0.5
            x_trial = project_feasible(h - step * grad, tgt)
            val_trial = smooth_obj_grad(x_trial, beta)[0]
            cap -= 1
        x_new = x_trial
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(it + 1) ** 2)) / 2.0
        t_k1 = (1.0 + np.sqrt(1.0 + 4.0 * float(it + 2) ** 2)) / 2.0
        momentum = (t_k - 1.0) / t_k1
        y = x_new + momentum * (x_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = x_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        if val_trial >= val - 1e-15:
            lr *= 0.95
    return best

def subgrad_nesterov(h0, tm=30, lr_init=0.2):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    y = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    j = np.arange(n)
    for it in range(50000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(y, 1.0 - y, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        a = j - k
        b = j + k
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += 1.0 - y[a[m1]]
        grad[m2] -= y[b[m2]]
        grad *= dx
        lr = lr_init / (np.sqrt(it + 1) + 1.0)
        x_new = project_feasible(h - lr * grad, tgt)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(it + 1) ** 2)) / 2.0
        t_k1 = (1.0 + np.sqrt(1.0 + 4.0 * float(it + 2) ** 2)) / 2.0
        momentum = (t_k - 1.0) / t_k1
        y = x_new + momentum * (x_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = x_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()

    def f(x):
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

    def gfun(x):
        return x.sum() - tgt

    def gjac(x):
        return np.ones(n)

    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 5000, 'ftol': 1e-14})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def make_structured_init(n, rng):

        phase = rng.uniform(0.0, 2.0 * np.pi)

    elif r == 3:
        n_blocks = rng.randint(2, 8)
        block_sizes = np.ones(n_blocks) * (n // n_blocks)
        block_sizes[-1] = n - block_sizes[:-1].sum()
        vals = rng.uniform(0, 1, n_blocks)
        h = np.concatenate([np.ones(int(b)) * v for b, v in zip(block_sizes, vals)])
    elif r == 4:
        period = rng.randint(3, 15)
        duty = rng.uniform(0.3, 0.7)
        h = np.zeros(n)
        for i in range(n):
            phase_i = (i % period) / period
            h[i] = 1.0 if phase_i < duty else 0.0
    elif r == 5:
        x = np.linspace(0, 2, n)
        center = rng.uniform(0.2, 1.8)
        width = rng.uniform(0.1, 0.5)
        h = 1.0 / (1.0 + np.exp((x - center) / width))
    elif r == 6:
        x = np.linspace(0, 2, n)
        c1, c2 = sorted(rng.uniform(0.1, 1.9, 2))
        w1, w2 = rng.uniform(0.1, 0.4, 2)
        h = np.exp(-0.5 * ((x - c1) / w1) ** 2) + np.exp(-0.5 * ((x - c2) / w2) ** 2)
    elif r == 7:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.cos(np.pi * (x - 0.5))
    elif r == 8:
        x = np.linspace(0, 2, n)
        freq = rng.randint(2, 6)
        h = (freq * x % 1.0)
    elif r == 9:
        x = np.linspace(0, 2, n)
        freq = rng.randint(2, 6)
        h = 1.0 - 2.0 * np.abs(freq * x % 1.0 - 0.5)
    elif r == 10:
        h = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            center = rng.uniform(0, n)
            width = rng.uniform(n / 10, n / 3)

        h = 0.5 + 0.5 * np.tanh(h)
    else:
        half = n // 2
        h = np.concatenate([rng.random(half), 1.0 - rng.random(n - half)])
    return project_feasible(h, n / 2.0)

def perturbed(h0, rng, scale=0.03):
    n = len(h0)
    h = h0 + rng.normal(0, scale, size=n)
    return project_feasible(h, n / 2.0)

def block_optimize(h0, n_blocks, tm=30):

    block_size = n // n_blocks
    if block_size < 1:
        return h0
    blocks = h0[:n_blocks * block_size].reshape(n_blocks, block_size).mean(axis=1).copy()
    t0 = time.time()

    def f(b):
        if time.time() - t0 > tm:
            raise Exception("tl")
        h_temp = np.repeat(b, block_size)
        if len(h_temp) < n:
            h_temp = np.concatenate([h_temp, np.zeros(n - len(h_temp))])
        h_temp = project_feasible(h_temp, n / 2.0)
        return (np.correlate(h_temp, 1.0 - h_temp, 'full') * (2.0 / n)).max()

    try:
        r = minimize(f, blocks, method='Nelder-Mead',

            h_block = np.concatenate([h_block, np.zeros(n - len(h_block))])
```
