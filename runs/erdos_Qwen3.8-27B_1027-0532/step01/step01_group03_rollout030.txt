    pop_c5 = [_c5_of(h) for h in pop]
    best_h = pop[int(np.argmin(pop_c5))].copy()
    best_c5 = min(pop_c5)
    F = 0.75
    CR = 0.85
    stagnation = 0
    while time.time() < t_end:
        gen_improved = False
        for i in range(act_size):
            if time.time() > t_end:
                break
            others = list(range(act_size))
            others.remove(i)
            a, b, c = rng.choice(others, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            cr_mask = rng.rand(n) < CR
            if not cr_mask.any():
                cr_mask[rng.randint(n)] = True
            trial = np.where(cr_mask, mutant, pop[i])
            trial = np.clip(trial, 0, 1)
            trial = _fix_sum(trial, target_sum)
            tc5 = _c5_of(trial)
            if tc5 < pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = tc5
                if tc5 < best_c5:
                    best_c5 = tc5
                    best_h = trial.copy()
                    gen_improved = True
        if gen_improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation % 20 == 0:
                F = max(0.3, F * 0.95)
                CR = min(0.95, CR + 0.02)
                n_replace = max(1, act_size // 8)
                for reps in range(n_replace):
                    pert = best_h + 0.1 * rng.randn(n)
                    pert = np.clip(pert, 0, 1)
                    pert = _fix_sum(pert, target_sum)
                    idx = int(rng.randint(act_size))
                    pop[idx] = pert
                    pop_c5[idx] = _c5_of(pert)
                    if pop_c5[idx] < best_c5:
                        best_c5 = pop_c5[idx]
                        best_h = pert.copy()
    return best_h, best_c5

def _run_projected_grad(h_start, n, target_sum, t_start, t_end):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    lr = 1.0 / (n ** 0.7)
    alpha = 30.0
    no_improve = 0
    max_no_improve = 80
    while time.time() < t_end:
        alpha = min(2000.0, alpha * 1.05)
        val, grad = _lse_value_and_grad(h, alpha)
        step = h - lr * grad
        step = np.clip(step, 0, 1)
        step = _fix_sum(step, target_sum)
        c5_new = _c5_of(step)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = step.copy()
            h = step.copy()
            no_improve = 0
            lr = min(lr * 1.1, 5.0)
        else:
            h = step.copy()
            no_improve += 1
            if no_improve >= max_no_improve:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-5:
                    break
    return best_h, best_c5

def _run_slsqp_polish(h_start, n, target_sum, alpha, maxiter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0, np.zeros(n)
        scaled = alpha * (corr - m)
        exp_sum = np.sum(np.exp(scaled))
        val = (m + np.log(exp_sum) / alpha) * (2.0 / n)
        weights = np.exp(scaled) / exp_sum
        grad = _grad_corr_weighted(x, n, weights) * (2.0 / n)
        return val, grad
    def cons_fun(x):
        return x.sum() - target_sum
    def cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj_grad, h_start, jac=True, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20
    rng = np.random.RandomState(seed)
    best_overall_c5 = 1.0
    best_overall_h = None
    best_overall_n = 86

    n_primary = 86
    n_others = [84, 88, 92, 96, 100, 104]

    t_phase1_end = t_start + (deadline - t_start) * 0.45
    t_phase2_end = t_start + (deadline - t_start) * 0.80

    if n_primary == 86 and 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(n_primary, 0.5)

    n = n_primary
    target_sum = n / 2.0
    h_init = _fix_sum(h_init, target_sum)

    best_h, best_c5 = _run_de(h_init, n, target_sum, rng, t_start, t_phase1_end, 35)
    best_h, best_c5 = _run_projected_grad(best_h, n, target_sum, t_start, t_phase2_end)

    if time.time() < deadline - 120:
        for alpha in [200, 500, 1000, 2000, 4000]:
            if time.time() > deadline - 100:
                break
            h_new = _run_slsqp_polish(best_h, n, target_sum, alpha, 150)
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if best_c5 < best_overall_c5:
        best_overall_c5 = best_c5
        best_overall_h = best_h.copy()
        best_overall_n = n

    time_per_other = max(60, (deadline - time.time() - 50) / max(1, len(n_others)))
    for n_try in n_others:
        if time.time() > deadline - 80:
            break
        if n_try % 2 != 0:
            continue
        t_sub_end = time.time() + time_per_other
        if t_sub_end > deadline - 30:
            t_sub_end = deadline - 30
        target_sum_o = n_try / 2.0
        h_o = np.full(n_try, 0.5)
        h_o[:int(n_try * 0.4)] = 0.8
        h_o = np.clip(h_o, 0, 1)
        h_o = _fix_sum(h_o, target_sum_o)
        t_de_end = time.time() + (t_sub_end - time.time()) * 0.5
        t_pg_end = time.time() + (t_sub_end - time.time()) * 0.85
        h_o, c5_o = _run_de(h_o, n_try, target_sum_o, rng, time.time(), t_de_end, 20)
        h_o, c5_o = _run_projected_grad(h_o, n_try, target_sum_o, time.time(), t_pg_end)
        if c5_o < best_overall_c5:
            best_overall_c5 = c5_o
            best_overall_h = h_o.copy()
            best_overall_n = n_try

    best_overall_h = _fix_sum(best_overall_h, best_overall_n / 2.0)
    best_overall_c5 = _c5_of(best_overall_h)

    return (best_overall_h, best_overall_c5, best_overall_n)
