def smooth_max_grad(h, tau):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        if lag >= 0:
            j_lo = 0
            j_hi = n - 1 - lag
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo+lag:j_hi+1+lag])
            j2_lo = lag
            j2_hi = n - 1
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo-lag:j2_hi+1-lag])

            k = -lag
            j_lo = k
            j_hi = n - 1
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo-k:j_hi+1-k])
            j2_lo = 0
            j2_hi = n - 1 - k
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+k:j2_hi+1+k])
    return obj, grad * dx

def multi_lag_pgd(h0, n, target_sum, deadline, K=5, momentum=0.9):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum
    lr = 0.05
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Find top K lags
        top_k = np.argsort(-corr)[:K]
        
        # Compute gradient as average over top K lags
        g = np.zeros(n)
        dx = 2.0 / n
        for m in top_k:
            lag = m - (n - 1)
            if lag >= 0:
                for j in range(n - lag):
                    g[j] += (1.0 - h[j + lag])
                for j in range(lag, n):
                    g[j] -= h[j - lag]
            else:
                k = -lag
                for j in range(k, n):
                    g[j] += (1.0 - h[j - k])
                for j in range(n - k):
                    g[j] -= h[j + k]
        g *= dx / K
        
        # Nesterov step
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        
        # Decay learning rate
        if it % 100 == 0:
            lr *= 0.99
    
    return best_h, best_c5

def compute_c5(h):

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0

        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:

def grad_single_lag(h, lag, n):
    """Gradient of correlation at a specific lag."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])

One more idea: **coordinate descent**. Instead of updating all coordinates simultaneously, update one coordinate at a time (or a small block). For each coordinate, compute the optimal value given the others fixed. This is exact for the quadratic objective (if we fix the max lag).

For a fixed lag k, the correlation is:
corr(k) = ∑_i h[i](1-h[i+k])

This is linear in each h[i] separately (quadratic jointly). So for a fixed k, the optimal h[i] (given all others fixed) is:
- If the coefficient of h[i] is positive, set h[i] = 0
- If the coefficient of h[i] is negative, set h[i] = 1
- (Subject to the sum constraint)

The coefficient of h[i] in corr(k):
- From the term h[i](1-h[i+k]): coefficient is (1-h[i+k])
- From the term h[i-k](1-h[i]): coefficient is -h[i-k]
- Total coefficient: (1-h[i+k]) - h[i-k] (when both terms are valid)

So for a fixed lag k, the optimal h[i] is:
- Set h[i] = 1 if (1-h[i+k]) - h[i-k] < 0, i.e., h[i+k] + h[i-k] > 1
- Set h[i] = 0 if (1-h[i+k]) - h[i-k] > 0, i.e., h[i+k] + h[i-k] < 1

This gives a binary solution that minimizes corr(k) for the fixed lag k!

    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 30

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0
    else:
        n0 = 120
        h_init = np.full(n0, 0.5)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)

    n_init = len(h_init)

    # Phase 1: PGD at original resolution with small K (focused)
    if time.time() < deadline - 300:
        dl = min(deadline - 250, t_start + 120)
        h_opt, c5_opt = pgd_momentum(h_init, n_init, n_init/2.0, dl, K=2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: PGD at original resolution with larger K (balanced)
    if time.time() < deadline - 250:
        dl = min(deadline - 200, t_start + 200)
        h_opt, c5_opt = pgd_momentum(best_h, n_init, n_init/2.0, dl, K=8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Interpolate to n=200 and optimize
    if time.time() < deadline - 180:
        n2 = 200
        h2 = interp_to_n(best_h, n2)
        h2 = project_onto_constraint(h2, n2/2.0)
        dl = min(deadline - 130, time.time() + 180)
        h_opt, c5_opt = pgd_momentum(h2, n2, n2/2.0, dl, K=4)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Interpolate to n=300 and optimize
    if time.time() < deadline - 120:
        n3 = 300
        h3 = interp_to_n(best_h, n3)
        h3 = project_onto_constraint(h3, n3/2.0)
        dl = min(deadline - 80, time.time() + 180)
        h_opt, c5_opt = pgd_momentum(h3, n3, n3/2.0, dl, K=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: FW on best solution for structural improvement
    if time.time() < deadline - 60:
        n = len(best_h)
        dl = min(deadline - 40, time.time() + 80)
        h_opt, c5_opt = fw_topk(best_h, n, n/2.0, dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: SLSQP polish with high tau (close to exact max)
    if time.time() < deadline - 30:
        n = len(best_h)
        dl = min(deadline - 15, time.time() + 60)
        h_opt, c5_opt = smooth_slsqp(best_h, n, n/2.0, dl, tau=400.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Final PGD polish
    if time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c5_opt = pgd_momentum(best_h, n, n/2.0, deadline, K=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: One more SLSQP if time allows
    if time.time() < deadline - 2:
        n = len(best_h)
        h_opt, c5_opt = smooth_slsqp(best_h, n, n/2.0, deadline, tau=500.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
