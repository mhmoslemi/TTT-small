- There exist m1, m2 active at t* with b_{m1} ≤ 0 ≤ b_{m2}
- t* = (a_{m2} - a_{m1}) / (b_{m1} - b_{m2}) (if b_{m1} ≠ b_{m2})

1. For each pair (m1, m2) with b_{m1} < 0 < b_{m2}, compute t = (a_{m2} - a_{m1}) / (b_{m1} - b_{m2})
2. Check if t ∈ [0,1] and if m1 and m2 are both active at t (i.e., a_m + b_m*t = f(t) for both, and no other line exceeds this value)
3. Among all valid t values, pick the one with smallest f(t)

Two lines l1 = a1 + b1*t and l2 = a2 + b2*t (with b1 < b2) intersect at t* = (a1-a2)/(b2-b1). Line l2 is above l1 for t > t* and below for t < t*.

The upper envelope algorithm:
- Start with the line with smallest slope
- For each subsequent line (in order of increasing slope):
  - Find the intersection with the current last line on the envelope
  - If this intersection is to the left of the previous intersection, pop the last line
  - Add the new line to the envelope

This gives the upper envelope in O(K log K) time (O(K) after sorting).

Then the minimum of the upper envelope on [0,1] is at:
- The intersection point of two consecutive lines on the envelope where the slope changes from negative to positive
- Or at t=0 or t=1 if all slopes are positive (or all negative)

    """Solve min_{t in [0,1]} max_m (intercept[m] + coeff[m] * t)"""
    K = len(coeff)
    
    # Create array of (slope, intercept) pairs
    # Sort by slope
    order = np.argsort(coeff)
    slopes = coeff[order]
    ints = intercept[order]
    
    # Remove lines that are never on the upper envelope
    # Use the standard algorithm
    # Keep (slope, intercept) in a list
    envelope = []  # list of (slope, intercept)
    
    for i in range(K):
        b, a = slopes[i], ints[i]
        # Check if this line is redundant
        while envelope:
            b_prev, a_prev = envelope[-1]
            if b == b_prev:
                # Same slope: keep the one with higher intercept
                if a <= a_prev:
                    break  # new line is dominated
                else:
                    envelope.pop()
                    continue
            # Intersection of new line with last envelope line
            # a + b*t = a_prev + b_prev*t => t = (a_prev - a) / (b - b_prev)
            t_int = (a_prev - a) / (b - b_prev)
            if len(envelope) >= 2:

                t_int_prev = (a_prev2 - a_prev) / (b_prev - b_prev2)
                if t_int <= t_int_prev:
                    # The last envelope line is never optimal, remove it
                    envelope.pop()
                    continue
            envelope.append((b, a))
    
    # Now find the minimum of the upper envelope on [0,1]
    # The envelope is a piecewise linear convex function
    # Find where the slope changes from negative to positive
    
    if not envelope:
        return 0.5  # fallback

    f0 = max(ints)  # max of intercepts
    f1 = max(ints + coeff)

    # At t=0, the active line is the one with max intercept

            f_int = env_a[i] + env_b[i] * t_int
            if f_int < best_val:
                best_val = f_int
                best_t = t_int
    
    return float(best_t)

def get_coeff_vector(h, i, n):
    """Compute the coefficient vector for coordinate i."""
    K = 2 * n - 1
    coeff = np.zeros(K)
    # First term: (1 - h[i-lag]) for valid lag
    # lag = m - (n-1), valid when 0 <= i - lag <= n-1
    # i.e., i - (n-1) <= lag <= i, i.e., i <= m <= i + n - 1
    m_lo = max(0, i)
    m_hi = min(K - 1, i + n - 1)
    if m_lo <= m_hi:
        m_range = np.arange(m_lo, m_hi + 1)
        lags = m_range - (n - 1)
        idx = i - lags
        coeff[m_lo:m_hi + 1] += (1.0 - h[idx])
    
    # Second term: -h[i+lag] for valid lag
    # valid when 0 <= i + lag <= n-1
    # i - (n-1) <= -lag <= ... wait
    # 0 <= i + lag <= n-1 => -i <= lag <= n-1-i
    # m - (n-1) is lag, so m - (n-1) >= -i => m >= n-1-i
    # m - (n-1) <= n-1-i => m <= 2(n-1)-i
    m_lo2 = max(0, n - 1 - i)
    m_hi2 = min(K - 1, 2 * (n - 1) - i)
    if m_lo2 <= m_hi2:
        m_range = np.arange(m_lo2, m_hi2 + 1)
        lags = m_range - (n - 1)
        idx = i + lags
        coeff[m_lo2:m_hi2 + 1] += (-h[idx])
    
    return coeff

def coordinate_descent(h_init, n, target_sum, deadline, max_sweeps=200, verbose=False):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    best_stall_thresh = 20
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        for i in range(n):
            if time.time() > deadline:
                break
            coeff = get_coeff_vector(h, i, n)
            intercept = corr - h[i] * coeff
            new_val = solve_1d_subproblem(intercept, coeff)
            h[i] = new_val
        
        h = project_onto_constraint(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > best_stall_thresh:
                break
    
    return best_h, best_c5

def subgradient_refine(h_init, n, target_sum, deadline, max_iter=50000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.1
    t_star_est = 0.38  # estimate of optimal value
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        dx = 2.0 / n
        
        # Subgradient of C_{k_star} w.r.t. h
        g = np.zeros(n)
        # C_k = sum_j h[j+lag] * (1-h[j]) for valid j
        # dC_k/dh[i] = (1-h[i-lag]) if 0<=i-lag<=n-1  minus  h[i+lag] if 0<=i+lag<=n-1
        m_lo = max(0, 0)  # for dC/dh[i], i ranges 0 to n-1
        for i in range(n):
            val = 0.0
            if 0 <= i - lag <= n - 1:
                val += (1.0 - h[i - lag])
            if 0 <= i + lag <= n - 1:
                val += (-h[i + lag])
            g[i] = val * dx
        
        # Polyak step size
        f_current = np.max(corr) * dx
        num = f_current - t_star_est
        den = np.dot(g, g)
        if den > 1e-15:
            lr_polyak = min(num / den, 10.0)
        else:
            lr_polyak = lr
        
        h_new = h - lr_polyak * g
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def solve_1d_subproblem(h, i, n, corr, coeff_all, intercept_all):
    """Solve min_{t in [0,1]} max_k corr_k(t) where h[i] is replaced by t."""
    # coeff_all and intercept_all are for all m, but m=n-1 is special (quadratic)
    # For m = n-1: the dependence is t - t^2, not linear
    # intercept_q = corr[n-1] - h[i] + h[i]^2
    # For m ≠ n-1: corr[m](t) = intercept_all[m] + t * coeff_all[m]
    
    i_n1 = n - 1  # index for lag=0
    intercept_q = corr[i_n1] - h[i] + h[i] * h[i]
    
    # Build the PL part (excluding m = n-1)
    mask = np.ones(len(coeff_all), dtype=bool)
    mask[i_n1] = False
    coeff_pl = coeff_all[mask]
    intercept_pl = intercept_all[mask]
    
    # Evaluate on grid
    grid = np.linspace(0, 1, 500)
    # PL part: max_m (intercept_pl[m] + grid[j] * coeff_pl[m])
    pl_vals = (grid[:, None] * coeff_pl[None, :] + intercept_pl[None, :]).max(axis=1)
    # Quadratic part
    quad_vals = intercept_q + grid - grid * grid
    # Combined
    f_vals = np.maximum(pl_vals, quad_vals)
    
    # Find minimum
    idx = np.argmin(f_vals)
    if idx == 0:
        return 0.0
    if idx == len(grid) - 1:
        return 1.0
    
    # Refine: narrow the grid around the minimum
    lo = grid[max(0, idx-5)]
    hi = grid[min(len(grid)-1, idx+5)]
    fine_grid = np.linspace(lo, hi, 200)
    pl_vals_f = (fine_grid[:, None] * coeff_pl[None, :] + intercept_pl[None, :]).max(axis=1)
    quad_vals_f = intercept_q + fine_grid - fine_grid * fine_grid
    f_vals_f = np.maximum(pl_vals_f, quad_vals_f)
    idx_f = np.argmin(f_vals_f)
    
    return float(fine_grid[idx_f])

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_coeff_for_coord(h, i, n):
    K = 2 * n - 1
    coeff = np.zeros(K)
    m_lo = max(0, i)
    m_hi = min(K - 1, i + n - 1)
    if m_lo <= m_hi:
        m_range = np.arange(m_lo, m_hi + 1)
        lags = m_range - (n - 1)
        idx = i - lags
        coeff[m_lo:m_hi + 1] += (1.0 - h[idx])
    m_lo2 = max(0, n - 1 - i)
    m_hi2 = min(K - 1, 2 * (n - 1) - i)
    if m_lo2 <= m_hi2:
        m_range = np.arange(m_lo2, m_hi2 + 1)
        lags = m_range - (n - 1)
        idx = i + lags
        coeff[m_lo2:m_hi2 + 1] += (-h[idx])
    return coeff

def solve_1d_subproblem(h, i, n, corr):
    K = 2 * n - 1
    coeff = get_coeff_for_coord(h, i, n)
    intercept = corr - h[i] * coeff
    i_n1 = n - 1
    intercept_q = corr[i_n1] - h[i] + h[i] * h[i]
    mask = np.ones(K, dtype=bool)
    mask[i_n1] = False
    coeff_pl = coeff[mask]
    intercept_pl = intercept[mask]
    grid = np.linspace(0.0, 1.0, 400)
    pl_vals = (grid[:, None] * coeff_pl[None, :] + intercept_pl[None, :]).max(axis=1)
    quad_vals = intercept_q + grid - grid * grid
    f_vals = np.maximum(pl_vals, quad_vals)
    idx = np.argmin(f_vals)
    if idx == 0:
        return 0.0
    if idx == len(grid) - 1:
        return 1.0
    lo = grid[max(0, idx - 3)]
    hi = grid[min(len(grid) - 1, idx + 3)]
    fine_grid = np.linspace(lo, hi, 300)
    pl_vals_f = (fine_grid[:, None] * coeff_pl[None, :] + intercept_pl[None, :]).max(axis=1)
    quad_vals_f = intercept_q + fine_grid - fine_grid * fine_grid
    f_vals_f = np.maximum(pl_vals_f, quad_vals_f)
    idx_f = np.argmin(f_vals_f)
    return float(fine_grid[idx_f])

def coordinate_descent(h_init, n, target_sum, deadline, max_sweeps=300):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        for i in range(n):
            if time.time() > deadline:
                break
            new_val = solve_1d_subproblem(h, i, n, corr)
            h[i] = new_val
            corr = np.correlate(h, 1.0 - h, mode="full")
        h = project_onto_constraint(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5

def subgradient_refine(h_init, n, target_sum, deadline, max_iter=50000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    avg_h = np.zeros(n)
    count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo <= i_hi:
            g[i_lo:i_hi + 1] += (1.0 - h[i_lo + lag:i_hi + lag + 1])
        i_lo2 = max(0, lag)
        i_hi2 = min(n - 1, n - 1 + lag)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2 + 1] += (-h[i_lo2 - lag:i_hi2 - lag + 1])
        g *= dx
        f_current = np.max(corr) * dx
        t_star_est = best_c5
        num = f_current - t_star_est
        den = float(np.dot(g, g))
        if den > 1e-15:
            lr = max(0.0, min(num / den, 5.0))
        else:
            lr = 0.01
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h_new.copy()
        avg_h += h_new
        count += 1
        h = h_new
        if count % 1000 == 0:
            avg_h /= count
            avg_h = project_onto_constraint(avg_h, target_sum)
            c5_avg = compute_c5(avg_h)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = avg_h.copy()
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def smooth_binary_nudge(h, strength):
    h_nudged = h + strength * (h - 0.5) * 2.0 * (1.0 - strength)
    return np.clip(h_nudged, 0.0, 1.0)

    if h_init is None:

    best_c5 = compute_c5(h_init)

    deadline2 = t_start + 250

    if c5_opt < best_c5:

    deadline3 = t_start + 380
    h_opt, c5_opt = coordinate_descent(best_h, n0, n0 / 2.0, deadline3, max_sweeps=200)
