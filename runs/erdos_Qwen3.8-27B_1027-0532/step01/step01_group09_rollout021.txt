    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_subgrad(h, ki, n):
    dx = 2.0 / n
    k = ki - (n - 1)
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += 1.0 - h[a[m1]]
    grad[m2] -= h[b[m2]]
    grad *= dx
    return grad

def nesterov_subgrad(h0, tm=60, lr_init=0.1):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    y = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    for it in range(300000):
        if time.time() - t0 > tm:
            break
        t = it + 1
        lr = lr_init / np.sqrt(t)
        corr = np.correlate(y, 1.0 - y, 'full') * (2.0 / n)
        ki = np.argmax(corr)
        grad = compute_subgrad(y, ki, n)
        h_new = project_feasible(y - lr * grad, tgt)
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((beta_t - 1.0) / beta_t) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def active_set_slsqp(h0, tm=50, max_outer=10):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    t0 = time.time()
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    for outer in range(max_outer):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        j = np.arange(n)
        a = j - k
        b = j + k
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)

        def obj_k(x):
            if time.time() - t0 > tm * 0.95:
                raise Exception("timeout")
            xc = np.clip(x, 0.0, 1.0)
            return (np.correlate(xc, 1.0 - xc, 'full') * dx)[ki]

        def grad_k(x):
            xc = np.clip(x, 0.0, 1.0)
            g = np.zeros(n)
            g[m1] += 1.0 - xc[a[m1]]
            g[m2] -= xc[b[m2]]
            g *= dx
            return g

        def gfun(x):
            return x.sum() - tgt

        def gjac(x):
            return np.ones(n)

        try:
            r = minimize(obj_k, h, jac=grad_k, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                         options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(r.x, tgt)
            c5 = eval_c5(h_new)
            if c5 < best_c5:
                best, best_c5 = h_new, c5
            h = h_new
        except Exception:
            break
    return best, best_c5

def multi_k_slsqp(h0, tm=50, top_k=5):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    t0 = time.time()

    def obj(x):
        if time.time() - t0 > tm * 0.92:
            raise Exception("timeout")
        xc = np.clip(x, 0.0, 1.0)
        c = np.correlate(xc, 1.0 - xc, 'full') * dx
        return np.sort(c)[-top_k:].mean()

    def gfun(x):
        return x.sum() - tgt

    def gjac(x):
        return np.ones(n)

    h = project_feasible(h0, tgt)
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 8000, 'ftol': 1e-15})
        h_new = project_feasible(r.x, tgt)
        return h_new, eval_c5(h_new)
    except Exception:
        return h0, eval_c5(h0)

def averaged_subgrad(h0, tm=60, lr_init=0.15, beta=0.9, avg_window=500):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    window = []
    lr = lr_init
    t0 = time.time()
    for it in range(400000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
        ki = np.argmax(corr)
        grad = compute_subgrad(h, ki, n)
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        window.append(h.copy())
        if len(window) > avg_window:
            window.pop(0)
        if len(window) == avg_window and it % avg_window == 0:
            avg_h = project_feasible(np.mean(window, axis=0), tgt)
            c5_avg = eval_c5(avg_h)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best = avg_h.copy()
        lr *= 0.99995
        if lr < 1e-10:
            break
    return best, best_c5

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    t_start = time.time()

    def tl(frac=0.95):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    n = len(h_init)
    best = project_feasible(h_init, n / 2.0)
    best_c5 = eval_c5(best)

    # Phase 1: Nesterov subgradient (global progress)
    if tl(0.18) > 15:
        h1, c1 = nesterov_subgrad(best, tm=min(55, tl(0.16)), lr_init=0.12)
        if c1 < best_c5:
            best, best_c5 = h1, c1

    # Phase 2: Active-set SLSQP (exploit exact gradient for worst k)
    if tl(0.35) > 15:
        h2, c2 = active_set_slsqp(best, tm=min(50, tl(0.33)), max_outer=8)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    # Phase 3: Averaged subgradient (different dynamics)
    if tl(0.50) > 15:
        h3, c3 = averaged_subgrad(best, tm=min(45, tl(0.48)), lr_init=0.08, beta=0.93, avg_window=300)
        if c3 < best_c5:
            best, best_c5 = h3, c3

    # Phase 4: Active-set SLSQP again (refine after subgradient)
    if tl(0.62) > 15:
        h4, c4 = active_set_slsqp(best, tm=min(40, tl(0.60)), max_outer=10)
        if c4 < best_c5:
            best, best_c5 = h4, c4

    # Phase 5: Multi-k smoothed SLSQP
    if tl(0.74) > 15:
        h5, c5 = multi_k_slsqp(best, tm=min(40, tl(0.72)), top_k=5)
        if c5 < best_c5:
            best, best_c5 = h5, c5

    # Phase 6: Fine Nesterov (small steps for convergence)
    if tl(0.84) > 15:
        h6, c6 = nesterov_subgrad(best, tm=min(35, tl(0.82)), lr_init=0.03)
        if c6 < best_c5:
            best, best_c5 = h6, c6

    # Phase 7: Try slightly different resolution
    for n_new in [256, 320]:
        if tl(0.92) < 12:
            break
        h_res = resample_h(best, n_new)
        h7, c7 = active_set_slsqp(h_res, tm=min(20, tl(0.92)), max_outer=5)
        if c7 < best_c5:
            best, best_c5 = h7, c7

    # Phase 8: Final active-set polish
    if tl(0.97) > 10:
        h8, c8 = active_set_slsqp(best, tm=min(15, tl(0.97)), max_outer=5)
        if c8 < best_c5:
            best, best_c5 = h8, c8

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
