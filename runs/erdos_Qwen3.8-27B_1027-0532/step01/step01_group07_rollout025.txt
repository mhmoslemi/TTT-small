    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    # Smooth max
    shifted = alpha * (c - c_max)
    exp_c = np.exp(shifted - shifted.max())  # for numerical stability
    w = exp_c / exp_c.sum()

    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return smooth_val, grad

        if best_h is not None and n_alt != n_primary:
            h_interp = np.interp(np.linspace(0, 1, n_alt), np.linspace(0, 1, n_primary), best_h)
            h_interp = project_feasible(h_interp, n_alt / 2.0)
        else:
            h_interp = np.full(n_alt, 0.5)
        h_opt_alt, c5_alt = gd_momentum(h_interp, n_alt, 60.0, 3000, deadline - 20, lr_start=0.05, momentum=0.9)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_opt_alt.copy()
            n_primary = n_alt
        if _time_left() > 60:
            h_pol, c5_pol = gd_momentum(h_opt_alt, n_alt, 200.0, 3000, deadline - 10, lr_start=0.02, momentum=0.93)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
