```python
def smooth_max_grad(h, dx, beta, n):
    """Compute gradient of smooth_max_corr with respect to h.
    
    f(h) = max(corr*dx) + log(sum(exp(corr*dx/beta)))/beta
    """
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx  # length 2n-1
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / beta)
    p = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for m_idx in range(2*n - 1):
        w = p[m_idx]
        if w < 1e-18:
            continue
        k = m_idx - (n - 1)  # actual lag
        if k >= 0:
            # d(c_k)/dh[j] = (1-h[j-k]) * I[j in [k, n-1]] - h[j+k] * I[j in [0, n-1-k]]
            if n - 1 - k >= 0:
                # Second term: j in [0, n-1-k]
                grad[0:n-k] -= w * h[k:n]
                # First term: j in [k, n-1]
                grad[k:n] += w * (1.0 - h[0:n-k])
        else:
            kp = -k  # positive
            # d(c_k)/dh[j] = (1-h[j+kp]) * I[j in [0, n-1-kp]] - h[j-kp] * I[j in [kp, n-1]]
            if n - 1 - kp >= 0:
                # First term: j in [0, n-1-kp]
                grad[0:n-kp] += w * (1.0 - h[kp:n])
                # Second term: j in [kp, n-1]
                grad[kp:n] -= w * h[0:n-kp]
    
    return grad
```

```python
def project_h(y, n):
    """Project y onto {0<=h<=1, sum(h)=n/2}."""
    target_sum = n * 0.5
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    # Bisection
    lo = y.min() - 1.0  # ensures sum > target
    hi = y.max() + 1.0  # ensures sum < target
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    return h
```

For coordinate descent:

- The optimal value minimizes the smooth max (or even the true max) over h[j] in [0, 1], subject to the sum constraint

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0: break
            step = min(diff / cnt, 1e-10)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0: break
            step = max(diff / cnt, -1e-10)
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_constrained(y, n):
    """Project y onto {0<=h<=1, sum(h)=n/2}."""
    target_sum = n * 0.5
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def smooth_max_and_grad(h, dx, beta, n):
    """Compute smooth max value and gradient."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / beta)
    Z = np.sum(exp_vals)
    p = exp_vals / Z
    f = m + np.log(Z) / beta
    
    grad = np.zeros(n)
    half = n - 1
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-18:
            continue
        k = m_idx - half
        if k >= 0:
            length = n - k
            if length > 0:
                grad[0:length] -= w * h[k:k+length]
                grad[k:k+length] += w * (1.0 - h[0:length])
        else:
            kp = -k
            length = n - kp
            if length > 0:
                grad[0:length] += w * (1.0 - h[kp:kp+length])
                grad[kp:kp+length] -= w * h[0:length]
    return f, grad

def projected_gradient_desc(h, n, dx, beta, max_iter, lr_init=0.1, tol=1e-14):
    """Projected gradient descent on the constrained set."""
    h = h.copy()
    lr = lr_init
    f_prev, _ = smooth_max_and_grad(h, dx, beta, n)
    
    for it in range(max_iter):
        f, g = smooth_max_and_grad(h, dx, beta, n)
        
        # Backtracking line search
        for _ in range(30):
            y = h - lr * g
            h_new = project_constrained(y, n)
            f_new, _ = smooth_max_and_grad(h_new, dx, beta, n)
            if f_new <= f - 0.1 * lr * np.sum(g * (h_new - h)):
                h = h_new
                break
            lr *= 0.5
        else:
            h = project_constrained(h - lr * g, n)
            f_new, _ = smooth_max_and_grad(h, dx, beta, n)
            f_new = min(f_new, f)
        
        if abs(f_prev - f) < tol:
            break
        f_prev = f
        lr *= 1.05  # gradually increase
    
    return h

def coordinate_descent(h, n, dx, max_rounds=10):
    """Coordinate descent: for each coordinate, find optimal value."""
    h = h.copy()
    target_sum = n * 0.5
    
    for round_idx in range(max_rounds):
        improved = False
        for j in range(n):
            # Try to optimize h[j] while maintaining sum constraint
            # If we change h[j] by delta, we need to subtract delta/(n-1) from all other coordinates
            # Or simpler: try a few candidate values for h[j] and pick the best
            old_val = h[j]
            # Try setting h[j] to various values in [0,1]
            # The adjustment to maintain sum: spread (target_sum - (sum - old_val + new_val)) 
            # over the other coordinates proportionally
            
            best_val = old_val
            best_c5 = c5_value(h, 2.0/n)
            
            # Try a few candidate values
            candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
            # Also try values near the current
            candidates.extend([max(0, old_val - 0.1), max(0, old_val - 0.01),
                              min(1, old_val + 0.01), min(1, old_val + 0.1)])
            
            for cv in candidates:
                if abs(cv - old_val) < 1e-15:
                    continue
                h[j] = cv
                # Repair to maintain sum
                h = repair_h(h, n)
                c5_try = c5_value(h, 2.0/n)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_val = cv
                    # Keep this change
                else:
                    h[j] = old_val
                    h = repair_h(h, n)  # restore
            
            if best_val != old_val:
                improved = True
        
        if not improved:
            break
    
    return h
```

```python
    return float(np.max(corr) * dx)

def repair_h(h, n):
```

```python
    target = n * 0.5
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(y, n):
    target = n * 0.5
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, dx, beta, n):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    m = corr.max()
    ev = np.exp((corr - m) / beta)
    Z = ev.sum()
    p = ev / Z
    f = m + np.log(Z) / beta
    g = np.zeros(n)
    half = n - 1
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-20:
            continue
        k = idx - half
        if k >= 0:
            ln = n - k
            if ln > 0:
                g[0:ln] -= w * h[k:k+ln]
                g[k:k+ln] += w * (1.0 - h[0:ln])
        else:
            kp = -k
            ln = n - kp
            if ln > 0:
                g[0:ln] += w * (1.0 - h[kp:kp+ln])
                g[kp:kp+ln] -= w * h[0:ln]
    return f, g

def pgd_optimize(h0, n, dx, beta, max_iter=5000, lr=0.05):

    momentum = 0.9
    v = np.zeros(n)

        h = h_new

        if lr < 1e-8:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(y, n):
    target = n * 0.5
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, dx, beta, n):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    m = corr.max()
    ev = np.exp((corr - m) / beta)
    Z = ev.sum()
    p = ev / Z
    f = m + np.log(Z) / beta
    g = np.zeros(n)
    half = n - 1
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-20:
            continue
        k = idx - half
        if k >= 0:
            ln = n - k
            if ln > 0:
                g[0:ln] -= w * h[k:k + ln]
                g[k:k + ln] += w * (1.0 - h[0:ln])
        else:
            kp = -k
            ln = n - kp
            if ln > 0:
                g[0:ln] += w * (1.0 - h[kp:kp + ln])
                g[kp:kp + ln] -= w * h[0:ln]
    return f, g

def pgd_optimize(h0, n, dx, beta, max_iter=8000, lr_init=0.03, time_limit=60.0):
    h = h0.copy()
    t_start = time.time()
    momentum = 0.85
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_f, _ = smooth_max_grad(h, dx, beta, n)
    stall = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        f, g = smooth_max_grad(h, dx, beta, n)
        v = momentum * v + g
        y = h - lr * v
        h_new = project_feasible(y, n)
        f_new, _ = smooth_max_grad(h_new, dx, beta, n)
        if f_new <= f + 1e-15:
            h = h_new
            if f_new < best_f:
                best_f = f_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.5
            v = g
            if lr < 1e-10:
                break
        if stall > 200:
            break
    return best_h

def binary_coord_descent(h, n, dx, max_rounds=30, time_limit=30.0):
    h = h.copy()
    t_start = time.time()
    best_c5 = c5_value(h, dx)
    order = list(range(n))
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        np.random.shuffle(order)
        for j in order:
            if time.time() - t_start > time_limit:
                break
            h_save = h.copy()
            for val in (0.0, 1.0):
                if abs(h[j] - val) < 1e-14:
                    continue
                h[j] = val
                h = project_feasible(h, n)
                c5_try = c5_value(h, dx)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved:
                h = h_save
    return h

def continuous_coord_descent(h, n, dx, max_rounds=5, time_limit=30.0):
    h = h.copy()
    t_start = time.time()
    best_c5 = c5_value(h, dx)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t_start > time_limit:
                break
            h_save = h.copy()
            old = h[j]
            cands = [0.0, 1.0, max(0.0, old - 0.15), max(0.0, old - 0.05),
                     min(1.0, old + 0.05), min(1.0, old + 0.15)]
            best_local = best_c5
            best_h_local = None
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                h[j] = cv
                h = project_feasible(h, n)
                c5_try = c5_value(h, dx)
                if c5_try < best_local:
                    best_local = c5_try
                    best_h_local = h.copy()
            if best_h_local is not None:
                h = best_h_local
                best_c5 = best_local
                improved = True
            else:
                h = h_save
        if not improved:
            break
    return h

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    return np.interp(x_dst, x_src, h_src)

def gen_sinusoidal(n, freq, amp, phase=0.0):
    x = np.arange(n, dtype=np.float64) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return repair_h(h, n)

def gen_piecewise(n, n_blocks, rng):
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        h[start:end] = rng.uniform(0.0, 1.0)
    return repair_h(h, n)

def random_perturb(h, n, rng, n_perturb, strength):
    h_new = h.copy()
    indices = rng.choice(n, size=min(n_perturb, n), replace=False)
    h_new[indices] += rng.uniform(-strength, strength, size=len(indices))
    return project_feasible(h_new, n)
```
