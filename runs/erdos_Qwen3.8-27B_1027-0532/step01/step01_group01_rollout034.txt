from scipy.optimize import minimize

_DX = 2.0 / _N

def eval_c5(h):

    return float(np.max(corr) * _DX)

def proj(x, n=_N):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:

    corr = np.correlate(x, 1 - x, mode="full")

    sh = eps * (corr - m)

The "worst" lag k* is the one where corr[k*] is maximum. To reduce corr[k*], we need to reduce ∑ h[i](1-h[i+k*]) for that specific k*. This means we want h[i] and h[i+k*] to be more similar (both 0 or both 1) for the lag k*.

So a targeted perturbation would be:
1. Find the worst lag k*
2. For each i where h[i]=1 and h[i+k*]=0 (or vice versa), consider swapping h[i] and h[i+k*] (or flipping one of them)
3. This reduces the correlation at lag k* but might increase it at other lags

This is a "repair" strategy: fix the worst lag, then re-optimize.

def eval_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * _DX)

def proj(x):
    n = _N
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def sm_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    sh = _EPS * (corr - m)
    return (m + np.log(np.sum(np.exp(sh))) / _EPS) * _DX

def sm_grad(x):
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    sh = _EPS * (corr - m)
    w = np.exp(sh) / np.sum(np.exp(sh))
    n = _N
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        for j in range(n):
            if 0 <= j + l < n:
                grad[j] += wk * (1.0 - x[j + l])
            if 0 <= j - l < n:
                grad[j] += wk * (-x[j - l])
    return grad * _DX

def eq_con(x):
    return np.sum(x) - _N / 2.0

def do_slsqp(h, eps, iters=300):
    global _EPS
    _EPS = eps
    res = minimize(
        sm_obj, h, jac=sm_grad,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * _N,
        constraints=[{'type': 'eq', 'fun': eq_con}],
        options={'maxiter': iters, 'ftol': 1e-14}
    )
    if np.all(np.isfinite(res.x)):
        return proj(res.x)
    return h

    n = _N
    h = h.copy()
    best = eval_c5(h)
    for _ in range(30):
        corr = np.correlate(h, 1 - h, mode="full")
        worst_k = np.argmax(corr)

            indices = list(range(n - l))
        else:
            indices = list(range(-l, n))
        np.random.shuffle(indices)
        tried = 0

            if tried >= max_tries:

            j = i + l

                h_trial = proj(h_trial)
                score = eval_c5(h_trial)
                tried += 1
                if score < best:
                    best = score
                    h = h_trial
    return h, best
