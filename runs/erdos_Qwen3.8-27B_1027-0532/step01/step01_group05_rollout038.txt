def coordinate_descent(h, n, n_sweeps=10, tolerance=1e-8):
    h = h.copy().astype(float)
    corr = correlate_full(h, n)
    for sweep in range(n_sweps):
        max_change = 0.0
        for i in range(n):
            v_new = coord_descent_step(h, i, corr, n)
            change = v_new - h[i]
            if abs(change) > 1e-12:
                h[i] = v_new
                max_change = max(max_change, abs(change))
                # Update correlations efficiently
                # delta_c[s] = change * a_s for s≠0, and change*(1-v_new-h[i]) for s=0 (approximately)
                # For simplicity, recompute correlations every few steps
                if i % 10 == 0:
                    corr = correlate_full(h, n)
            elif i % 10 == 0:
                corr = correlate_full(h, n)
        # Project
        h = project_onto_constraint(h, n)
        corr = correlate_full(h, n)
        if max_change < tolerance:
            break
    return h

def thue_morse(n):
    """Generate the Thue-Morse sequence of length n."""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def create_initialization(n, method="parent", parent=None, seed=42):
    rng = np.random.RandomState(seed)
    if method == "parent" and parent is not None:
        x_src = np.linspace(0, 2, len(parent), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_src, parent)
        return project_onto_constraint(h, n)
    elif method == "thue_morse":
        h = thue_morse(n).astype(float)
        h = project_onto_constraint(h, n)
        return h
    elif method == "random_01":
        h = (rng.rand(n) < 0.5).astype(float)
        h = project_onto_constraint(h, n)
        return h
    elif method == "sawtooth":
        M = max(2, n // 70)  # number of teeth
        h = np.zeros(n)
        period = 2 * n // M
        for start in range(0, n - period + 1, period):
            h[start:start + period//2] = 1.0
        if n - period > 0:
            remaining = n - (n // period) * period
            if remaining > 0:
                h[-(remaining//2 + 1):] = 1.0  # partial tooth
        h = project_onto_constraint(h, n)
        return h
    elif method == "smooth":
        x = np.linspace(0, 2, n)
        # Multi-frequency smooth function
        h = 0.5 + 0.3*np.cos(np.pi*x) + 0.15*np.cos(3*np.pi*x) + 0.1*np.cos(5*np.pi*x)
        h = np.clip(h, 0, 1)
        h = project_onto_constraint(h, n)
        return h
    else:
        h = np.full(n, 0.5)
        return h

The first term (max over s≠0 of linear functions) is convex in v.
The second term (quadratic) is concave in v (downward parabola).
The max of a convex and a concave function is not necessarily convex or concave.

However, the maximum of n-1 linear functions is a convex piecewise-linear function with at most n-1 pieces. The quadratic is smooth. The max of these two is a piecewise function with at most O(n) pieces (each piece being either the quadratic, a linear function, or the point where they cross).

To find the minimum:
1. The minimum of a convex piecewise-linear function over [0,1] is at a "kink" point or at the boundary.
2. The minimum of the max with a concave quadratic could be at:
   - v = 0 or v = 1 (boundary)
   - A point where the quadratic = one of the linears (and the quadratic is below all other linears)
   - A kink point of the max-of-linears (where two linears cross, and both are above the quadratic)

Finding all these points exactly is O(n²) in the worst case. For our purposes, a grid-based approach is simpler and fast enough.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def coord_step(h, i, corr, n, v_grid):
    hi_val = h[i]
    shifts = np.arange(-(n-1), n)  # s values: -(n-1), ..., 0, ..., n-1
    idx_plus = i + shifts
    idx_minus = i - shifts
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    h_plus = np.zeros(n*2-1)
    h_minus = np.zeros(n*2-1)
    h_plus[valid_plus] = h[idx_plus[valid_plus]]
    h_minus[valid_minus] = h[idx_minus[valid_minus]]
    a = 1.0 - h_plus - h_minus
    a[n-1] = 0.0  # s=0 handled separately
    
    # corr indices: for shift s, the index in corr is n-1-s
    # So corr[n-1-s] for s in -(n-1)..n-1 gives corr[n-1-(-(n-1))..n-1-(n-1)] = corr[2n-2..0]

    corr_reversed = corr[::-1]  # corr_reversed[n-1+s] = corr[n-1-s]
    
    # For v grid: compute max_s [corr[n-1-s] + (v-hi)*a_s] and quadratic
    # Linear part: for each s, value = corr_reversed[n-1+s] + (v-hi)*a[s+(n-1)]
    # Let's index by j = s+(n-1), ranging from 0 to 2n-2

    # For shift s, index in original corr is: n-1-s
    # j = n-1-s, so s = n-1-j
    # corr index j corresponds to shift s = n-1-j
    # a value for this j: a at position n-1-j in the shifts array
    
    # Actually, let me just compute it directly without the shift indexing.
    # For each correlation index j (0 to 2n-2), the shift is s = n-1-j.
    # a_s = 1 - h[i+s] - h[i-s] = 1 - h[i+n-1-j] - h[i-(n-1-j)] = 1 - h[i+n-1-j] - h[i+1-n+j]
    
    # This is getting confusing. Let me just use the shifts array directly.
    
    # shifts[j] = -(n-1) + j for j = 0, ..., 2n-2
    # For shift s = shifts[j], corr index is n-1-s = n-1-(j-(n-1)) = 2n-2-j
    # So corr[2n-2-j] is the correlation value corresponding to shifts[j]
    # Equivalently: corr_reversed[j] = corr[2n-2-j]
    
    # The linear functions: L_j(v) = corr_reversed[j] + (v - hi_val) * a[j] for j ≠ n-1 (s ≠ 0)
    # The quadratic: Q(v) = corr[n-1] + (v-hi_val)*(1-v-hi_val)
    
    # f(v) = max(max_j≠(n-1) L_j(v), Q(v))

    vals = corr_reversed[np.newaxis, :] + np.outer(v_grid - hi_val, a)  # shape: (len(v_grid), 2n-1)
    vals[:, n-1] = -np.inf  # exclude s=0 from the linear max
    linear_max = np.max(vals, axis=1)
    quad = corr[n-1] + (v_grid - hi_val) * (1 - v_grid - hi_val)
    f = np.maximum(linear_max, quad)
    idx_best = np.argmin(f)
    return v_grid[idx_best]

def coordinate_descent(h, n, n_sweeps=100, v_grid=None):
    if v_grid is None:
        v_grid = np.linspace(0, 1, 80)
    h = h.copy().astype(float)
    for sweep in range(n_sweeps):
        corr = correlate_full(h)
        max_change = 0.0
        for i in range(n):
            v_new = coord_step(h, i, corr, n, v_grid)
            diff = v_new - h[i]
            if abs(diff) > 1e-10:
                h[i] = v_new
                max_change = max(max_change, abs(diff))
        h = project_onto_constraint(h, n)
        if max_change < 1e-7:
            break
    return h

def thue_morse_seq(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def init_thue_morse(n):
    h = thue_morse_seq(n).astype(float)
    return project_onto_constraint(h, n)

def init_smooth(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    n_freq = 5 + rng.randint(3)
    h = np.full(n, 0.5)
    for k in range(1, n_freq+1):
        freq = rng.randint(1, 8) * k
        amp = rng.uniform(0.05, 0.2) / np.sqrt(k)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_onto_constraint(h, n)

def init_sawtooth(n, n_teeth=7):
    h = np.zeros(n)
    period = 2 * n // (2 * n_teeth)
    half = period // 2
    for start in range(0, n - period, period):
        h[start:start+half] = 1.0
    # Fill remaining
    remaining = n - (n // period) * period
    for i in range(min(remaining, half)):
        h[-(remaining) + i] = 1.0
    return project_onto_constraint(h, n)
