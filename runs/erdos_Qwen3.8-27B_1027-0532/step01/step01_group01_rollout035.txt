import numpy as np

_G_n = 85
_G_eps = 100.0

_G_lam = 100.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode="full")

def coord_update(i, h, corr, n):
    dx = 2.0 / n
    b = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        val = 0.0
        if 0 <= i + l < n:
            val += (1.0 - h[i + l])
        if 0 <= i - l < n:
            val += (-h[i - l])
        b[k] = val
    a = corr - b * h[i]
    best_v = 0.0
    best_f = np.max(a) * dx
    for check_v in [0.0, 1.0]:
        f = np.max(a + b * check_v) * dx
        if f < best_f:

            best_v = check_v

    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = _G_eps * (corr - m)
    return (m + (1.0 / _G_eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x):
    n = _G_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = _G_eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        for j in range(n):
            term = 0.0
            if 0 <= j + l < n:
                term += (1.0 - x[j + l])
            if 0 <= j - l < n:
                term += (-x[j - l])
            grad[j] += w * term
    grad *= dx
    return grad

def eq_constraint(x):
    return np.sum(x) - _G_n / 2.0

def penalty_obj(x):
    n = _G_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    viol = np.maximum(0, corr * dx - _G_t)
    return np.sum(viol * viol) + _G_lam * (np.sum(x) - n / 2.0) ** 2
