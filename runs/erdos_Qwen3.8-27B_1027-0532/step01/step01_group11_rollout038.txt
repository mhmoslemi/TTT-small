    c_obj[n] = 1.0
    
    n_constraints = (2*n - 1) + 2 * n  # correlation + box upper + box (lower handled by bounds)
    # Actually, let's use bounds for 0<=h<=1 and only have the correlation constraints
    
    A_ub = np.zeros((2*n - 1, n + 1))
    b_ub = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        A_ub[k, :n] = grad_all[k]
        A_ub[k, n] = -1.0
        b_ub[k] = grad_all[k] @ h_m - corr_m[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return h_m.copy(), np.inf

def slp_optimize(h_init, max_iters=100, damp=1.0, time_limit=None):
    h = h_init.copy()
    n = len(h)
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break
        
        h_new, t_lp = slp_solve_lp(h)
        h_new = project_onto_constraint(h_new, n / 2.0)
        
        # Damped update
        h_update = damp * h_new + (1 - damp) * h
        h_update = project_onto_constraint(h_update, n / 2.0)
        
        c5_update = compute_c5(h_update)
        if c5_update < best_c5:
            best_c5 = c5_update
            best_h = h_update.copy()
            h = h_update
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                # Try smaller step
                damp = max(0.1, damp * 0.5)
    
    return best_h, best_c5

def smooth_objective_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    val = cmax + eps * np.log(np.sum(np.exp(x)))
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * dx * (1.0 - h[s:n])
            grad[s:n] -= wk * dx * h[:n-s]
        else:
            r = -s
            grad[r:n] += wk * dx * (1.0 - h[:n-r])
            grad[:n-r] -= wk * dx * h[r:n]
    return val, grad

def gradient_polish(h_init, n_steps=500, time_limit=None):
    h = h_init.copy()
    n = len(h)
    target_sum = n / 2.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    eps = 0.001
    lr = 0.05
    for step in range(n_steps):
        if time_limit is not None and time.time() - start > time_limit:
            break
        val, grad = smooth_objective_and_grad(h, eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Adaptive step
        step_size = lr / max(gnorm, 0.1)
        h_trial = project_onto_constraint(h - step_size * grad, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < compute_c5(h):
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        else:
            lr *= 0.7
        eps = max(0.0001, eps * 0.98)
    
    return best_h, best_c5

Wait, I'm concerned about the SLP convergence. The issue is that the linearization at h_m gives a lower bound on each corr_k (since we're linearizing a concave... wait, is the function concave or convex in h?).

f_k(h) = ∑_i h[i]*(1-h[i+s]) = ∑_i h[i] - ∑_i h[i]*h[i+s]

The Hessian of f_k with respect to h is:
∂²f_k/∂h[i]∂h[j] is non-zero only when j = i+s or j = i-s:
- ∂²f_k/∂h[i]∂h[i+s] = -1
- ∂²f_k/∂h[i+s]∂h[i] = -1

So the Hessian has -1 on the (i, i+s) and (i+s, i) entries. This is not positive or negative semidefinite in general, so f_k is neither convex nor concave. The SLP may not converge monotonically.

However, in practice, if the solution is near a local minimum, the linearization should be reasonable, and the SLP should make progress. The damped update helps stabilize.

Let me also think about whether the LP might be infeasible. The constraints are:
- grad_k · h - t ≤ grad_k · h_m - corr_m[k] for all k
- sum(h) = n/2
- 0 ≤ h ≤ 1

The LP is infeasible if no h satisfies all linearized constraints simultaneously. This can happen if the linearization is too aggressive (the true feasible set is smaller than the linearized one). In that case, `linprog` will return status 2 (infeasible).

To handle this, I should:
1. If infeasible, try a damped step (blend with current h)
2. Or use a trust region to limit the step

Let me add a trust region:
||h - h_m||_∞ ≤ Δ

This means: h_m[i] - Δ ≤ h[i] ≤ h_m[i] + Δ

In the LP, this adds:
-h[i] ≤ -h_m[i] + Δ  (i.e., h[i] ≥ h_m[i] - Δ)
h[i] ≤ h_m[i] + Δ

These are simple bound adjustments.

Actually, the simplest approach: just check if the LP solution is close to h_m (within some tolerance), and if not, blend.

    best_h = h_init.copy()

    global n2scale
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(90), 0.0, 1.0)

    n0 = len(h0)
    n2scale = float(n0)
    h0 = project_onto_constraint(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def remaining():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLP from initial
    if remaining() > 200:
        tl = min(remaining() - 120, 150)
        h_slp, c5_slp = slp_optimize(best_h, max_iters=80, time_limit=tl)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp
        if remaining() > 80:
            h_pol, c5_pol = gradient_polish(best_h, n_steps=300, time_limit=min(remaining() - 40, 80))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 2: Gradient + active set refinement
    if remaining() > 100:
        h_act = best_h.copy()
        c5_act = best_c5
        for i in range(200):
            if remaining() < 60:
                break
            h_act, c5_cur = active_set_gradient_step(h_act, 0.01, 0.3 / (i + 1))
            if c5_cur < c5_act:
                c5_act = c5_cur
        c5_act = compute_c5(h_act)
        if c5_act < best_c5:
            best_c5 = c5_act
            best_h = h_act
        if remaining() > 50:
            h_pol, c5_pol = gradient_polish(best_h, n_steps=200, time_limit=min(remaining() - 20, 50))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 3: Multi-restart with SLP
    n_restarts = 0
    while remaining() > 120 and n_restarts < 20:
        strategy = n_restarts % 6
        if strategy == 0:
            h_restart = best_h + 0.08 * rng.standard_normal(n0)
        elif strategy == 1:
            h_restart = np.clip(best_h + 0.15 * (rng.random(n0) - 0.5), 0, 1)
        elif strategy == 2:
            h_restart = best_h.copy()
            k_flip = max(3, n0 // 8)
            idx = rng.choice(n0, size=k_flip, replace=False)
            h_restart[idx] = 1.0 - h_restart[idx]
        elif strategy == 3:
            h_rand = np.clip(0.5 + 0.35 * rng.standard_normal(n0), 0, 1)
            alpha = rng.uniform(0.5, 0.85)
            h_restart = alpha * best_h + (1.0 - alpha) * h_rand
        elif strategy == 4:
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            h_restart = best_h + 0.06 * np.sin(2.0 * freq * np.pi * np.arange(n0) / n0 + phase)
        else:
            perm = rng.permutation(n0)
            h_restart = best_h[perm].copy()

        h_restart = project_onto_constraint(h_restart, n0 / 2.0)
        tl_r = min(remaining() - 60, 50)
        if tl_r > 10:
            h_opt, c5_opt = slp_optimize(h_restart, max_iters=40, time_limit=tl_r)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        n_restarts += 1

    # Phase 4: Simulated annealing for diversity
    if remaining() > 150:
        tl_sa = min(remaining() - 80, 200)
        h_sa, c5_sa = simulated_annealing(best_h, tl_sa, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        if remaining() > 60:
            h_pol, c5_pol = gradient_polish(best_h, n_steps=200, time_limit=min(remaining() - 20, 80))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 5: Upscale to 2x resolution
    if remaining() > 250:
        n1 = 2 * n0
        n2scale = float(n1)
        h_up = np.repeat(best_h, 2).astype(float)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        tl_up = min(remaining() - 100, 300)
        h_up_opt, c5_up = slp_optimize(h_up, max_iters=50, time_limit=tl_up * 0.5)
        if remaining() > 60:
            h_up_opt, c5_up2 = gradient_polish(h_up_opt, n_steps=300, time_limit=remaining() - 30)
            if c5_up2 < c5_up:
                c5_up = c5_up2
            else:
                h_up_opt = h_up_opt
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_opt

        if remaining() > 100:
            h_rand_up = np.clip(0.5 + 0.3 * rng.standard_normal(n1), 0, 1)
            h_rand_up = project_onto_constraint(h_rand_up, n1 / 2.0)
            tl_ru = min(remaining() - 40, 120)
            h_ru, c5_ru = slp_optimize(h_rand_up, max_iters=40, time_limit=tl_ru)
            if c5_ru < best_c5:
                best_c5 = c5_ru
                best_h = h_ru

    # Phase 6: Final SLP + gradient polish
    if remaining() > 30:
        nf = len(best_h)
        n2scale = float(nf)
        h_fin, c5_fin = slp_optimize(best_h, max_iters=20, time_limit=min(remaining() - 10, 40))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        if remaining() > 15:
            h_pol, c5_pol = gradient_polish(best_h, n_steps=100, time_limit=remaining() - 5)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
