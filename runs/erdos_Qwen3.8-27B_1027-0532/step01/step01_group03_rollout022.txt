    return corr

def _swap_change(overlap, h, i, j, n):
    """Compute new overlap after swapping h[i] (1→0) and h[j] (0→1)."""
    new_overlap = overlap.copy()
    # i goes from 1 to 0
    # Effect on overlap: for shift s, m=i gives -(1-h[i+s]), m=i-s gives -h[i-s]
    for s in range(-(n-1), n):

        # m=i-s, m+s=i must be in [0,n): valid if 0<=i-s<n

        if 0 <= i+s < n:

import time
import numpy as np

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _compute_overlap(h):
    """Compute the overlap vector for all shifts."""
    return np.correlate(h, 1.0 - h, mode='full')

def _binary_local_search(h, n, target_sum, max_time=30, seed=42):
    """Greedy binary local search: always accept the best improving swap."""
    rng = np.random.RandomState(seed)
    h = np.where(h >= 0.5, 1.0, 0.0)
    # Fix sum to be exactly target_sum
    while h.sum() > target_sum:
        idx = rng.choice(np.where(h == 1.0)[0])
        h[idx] = 0.0
    while h.sum() < target_sum:
        idx = rng.choice(np.where(h == 0.0)[0])
        h[idx] = 1.0
    
    best_h = h.copy()
    best_c5 = _c5_of(h)
    overlap = _compute_overlap(h)
    best_overlap_max = float(np.max(overlap))
    
    t_start = time.time()
    no_improve_count = 0
    max_no_improve = 200
    
    while time.time() - t_start < max_time:
        if no_improve_count > max_no_improve:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap = None
        best_new_max = best_overlap_max
        
        # Try a random subset of swaps
        n_tries = min(2000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Compute change in overlap
            # Δ(s) for shift s (stored at index n-1+s):
            # = -(1-h[i+s]) - h[i-s] + (1-h[j+s]) - h[j-s]
            # with validity checks
            
            new_overlap = overlap.copy()
            delta_max = 0.0
            for s in range(-(n-1), n):
                d = 0.0
                if 0 <= i + s < n:
                    d -= (1.0 - h[i+s])
                if 0 <= i - s < n:
                    d -= h[i-s]
                if 0 <= j + s < n:
                    d += (1.0 - h[j+s])
                if 0 <= j - s < n:
                    d -= h[j-s]
                if d != 0.0:
                    new_overlap[n-1+s] += d
            
            new_max = float(np.max(new_overlap))
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j, new_overlap)
        
        if best_swap is not None:
            i, j, new_overlap = best_swap
            h[i] = 0.0
            h[j] = 1.0
            overlap = new_overlap
            best_overlap_max = best_new_max
            best_h = h.copy()
            best_c5 = best_overlap_max * (2.0 / n)
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return best_h, best_c5

def _simulated_annealing(h, n, target_sum, max_time=60, seed=42):
    """Simulated annealing for binary sequences."""
    rng = np.random.RandomState(seed)
    h = np.where(h >= 0.5, 1.0, 0.0)
    while h.sum() > target_sum:
        idx = rng.choice(np.where(h == 1.0)[0])
        h[idx] = 0.0
    while h.sum() < target_sum:
        idx = rng.choice(np.where(h == 0.0)[0])
        h[idx] = 1.0
    
    best_h = h.copy()
    best_c5 = _c5_of(h)
    current_c5 = best_c5
    overlap = _compute_overlap(h)
    current_max = float(np.max(overlap))
    
    T = 0.5
    T_min = 0.001
    cooling = 0.99
    
    t_start = time.time()
    while time.time() - t_start < max_time:
        if T < T_min:
            T = T_min
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_overlap = _compute_overlap(h)
        new_max = float(np.max(new_overlap))
        new_c5 = new_max * (2.0 / n)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            current_max = new_max
            overlap = new_overlap
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_c5

def _pattern_init(n, target_sum, pattern='block', seed=42):
    """Generate a pattern-based initial binary sequence."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    
    if pattern == 'block':
        h[:int(target_sum)] = 1.0
    elif pattern == 'alternating':
        h[::2] = 1.0
        if h.sum() > target_sum:
            h[-1] = 0.0  # adjust
        # Fix sum
        while h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1.0)[0])
            h[idx] = 0.0
        while h.sum() < target_sum:
            idx = rng.choice(np.where(h == 0.0)[0])
            h[idx] = 1.0
    elif pattern == 'golden':
        alpha = (1 + 5**0.5) / 2.0
        for i in range(n):
            h[i] = 1.0 if (i * (1.0/alpha)) % 1.0 < 0.5 else 0.0
        while h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1.0)[0])
            h[idx] = 0.0
        while h.sum() < target_sum:
            idx = rng.choice(np.where(h == 0.0)[0])
            h[idx] = 1.0
    elif pattern == 'random':
        h = (rng.rand(n) < 0.5).astype(float)
        while h.sum() > target_sum:
            idx = rng.choice(np.where(h == 1.0)[0])
            h[idx] = 0.0
        while h.sum() < target_sum:
            idx = rng.choice(np.where(h == 0.0)[0])
            h[idx] = 1.0
    
    return h

Hmm, wait. I realize the binary local search might be too slow because of the inner loop computing the overlap change for each candidate swap. Let me optimize it.

The key insight: for the overlap change due to a swap (i: 1→0, j: 0→1), I can vectorize the computation.

```python
def _overlap_change(h, i, j, n):
    """Compute the change in the overlap vector when h[i] goes 1→0 and h[j] goes 0→1."""
    s = np.arange(-(n-1), n)  # all shifts
    delta = np.zeros(2*n-1)
    
    # From i going 1→0:
    mask1 = (i + s >= 0) & (i + s < n)
    mask2 = (i - s >= 0) & (i - s < n)
    delta[mask1] -= (1.0 - h[i + s[mask1]])
    delta[mask2] -= h[i - s[mask2]]
    
    # From j going 0→1:
    mask3 = (j + s >= 0) & (j + s < n)
    mask4 = (j - s >= 0) & (j - s < n)
    delta[mask3] += (1.0 - h[j + s[mask3]])
    delta[mask4] -= h[j - s[mask4]]
    
    return delta
```

This is O(n) per swap with numpy vectorization, which should be fast.

But even with this, trying 2000 swaps per iteration with O(n) work each is 2000*86 ≈ 172K operations per iteration, which in numpy should take ~1ms. So we could do ~10000 iterations in 10 seconds. That's good.

In 60 seconds, that's 6000 iterations. Each iteration makes one swap, so we make 6000 improvements. That should be enough to converge for n=86.
