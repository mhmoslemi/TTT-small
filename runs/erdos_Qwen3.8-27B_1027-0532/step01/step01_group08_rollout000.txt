import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def top_k_pgd(h0, n, target_sum, deadline, top_k=5, max_iter=8000):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.04
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_indices = np.argsort(corr)[-top_k:]
        cmax = np.max(corr)
        weights = np.exp((corr[top_indices] - cmax) / 0.008)
        sw = np.sum(weights)
        if sw < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights /= sw
        g = np.zeros(n)
        for idx, w in zip(top_indices, weights):
            lag = int(idx) - (n - 1)
            j_lo = max(0, lag)
            j_hi = min(n - 1, n - 1 + lag)
            if j_lo <= j_hi:

            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                g[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
        g *= dx
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:

            h = h_new

def slsqp_optimize(h0, n, target_sum, deadline, tau=100.0):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': 400, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def lbfgsb_optimize(h0, n, target_sum, deadline, tau=150.0, penalty=500.0):
    def obj_p(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o + penalty * (x.sum() - target_sum) ** 2
    def grad_p(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g + 2.0 * penalty * (x.sum() - target_sum)

    if remaining < 5:

                                               'gtol': 1e-10, 'maxfun': 5000})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)

        pass

def interpolate_h(h, target_n):
    old_n = len(h)
    if old_n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def trig_eval(params, x, c):

        freq = (j + 1) * np.pi
        z += a * np.cos(freq * x) + b * np.sin(freq * x)

def structured_init(n, target_sum, pattern, seed=0):
    rng = np.random.RandomState(seed)
    if pattern == 'alt':
        h = np.zeros(n)
        h[::2] = 1.0
        excess = h.sum() - target_sum
        if excess > 0:
            ones_idx = np.where(h == 1.0)[0]
            rng.shuffle(ones_idx)
            h[ones_idx[:int(excess)]] = 0.0
        deficit = target_sum - h.sum()
        if deficit > 0:
            zero_idx = np.where(h == 0.0)[0]
            rng.shuffle(zero_idx)
            h[zero_idx[:int(deficit)]] = 1.0
    elif pattern == 'blocks':
        nblocks = 4
        block_size = n // (2 * nblocks)
        h = np.zeros(n)
        for i in range(nblocks):
            start = i * 2 * block_size
            end = min(start + block_size, n)
            h[start:end] = 1.0
    elif pattern == 'sine':
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif pattern == 'sine2':
        x = np.linspace(0, 2, n, endpoint=False)
        f1 = rng.uniform(1, 3)
        f2 = rng.uniform(3, 6)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x + p1) + 0.2 * np.sin(2 * np.pi * f2 * x + p2)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.uniform(0.2, 0.8, n)
    return project_onto_constraint(h, target_sum)

                h0 = interpolate_h(best_h, n)
                h0 = project_onto_constraint(h0, target)
            h_opt, c5_opt = lbfgsb_optimize(h0, n, target, deadline - 100, tau=150, penalty=500)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: SLSQP refinement at best resolution
    if best_h is not None and time.time() < deadline - 150:
        n = len(best_h)
        target = n / 2.0
        for tau in [80, 150, 300]:
            if time.time() > deadline - 80:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, n, target, deadline - 80, tau=tau)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Top-k PGD
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        h_opt, c5_opt = top_k_pgd(best_h, n, n / 2.0, deadline - 100, top_k=5, max_iter=8000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Multi-resolution PGD
    if best_h is not None and time.time() < deadline - 60:
        for n in [100, 120, 140, 160]:
            if time.time() > deadline - 30:
                break
            target = n / 2.0
            h_try = interpolate_h(best_h, n)
            h_try = project_onto_constraint(h_try, target)
            h_opt, c5_opt = top_k_pgd(h_try, n, target, deadline - 30, top_k=3, max_iter=4000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Worst-case PGD polish
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_opt, c5_opt = worst_case_pgd(best_h, n, n / 2.0, deadline - 30, max_iter=8000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: L-BFGS-B final with higher penalty
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        h_opt, c5_opt = lbfgsb_optimize(best_h, n, n / 2.0, deadline - 20, tau=200, penalty=1000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final worst-case PGD
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c5_opt = worst_case_pgd(best_h, n, n / 2.0, deadline - 5, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Top-k PGD final
    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        h_opt, c5_opt = top_k_pgd(best_h, n, n / 2.0, deadline - 2, top_k=3, max_iter=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure valid output
    if best_h is None:
        n_fb = 120
        best_h = structured_init(n_fb, n_fb / 2.0, 'sine', seed)
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
