def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    
    # Project to satisfy constraints
    target_sum = n / 2.0
    def project(h):
        h = np.clip(h, 0.0, 1.0)
        # Bisection for the sum constraint
        for _ in range(80):
            s = np.sum(h)
            diff = target_sum - s
            if abs(diff) < 1e-15:
                break
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
        return h
    
    best_h = project(init.copy())
    best_score = evaluate_score(best_h)
    
    # Projected gradient method with smooth max
    alphas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    
    for alpha in alphas:
        if time.time() - t0 > budget_s - 120:
            break
        
        h = best_h.copy()
        dx = 2.0 / n
        
        # L-BFGS-B with the smooth objective
        # Actually, let me use projected gradient with backtracking line search
        
        eta = 1.0 / (alpha * n * 0.01)  # initial step size
        for it in range(2000):
            if time.time() - t0 > budget_s - 120:
                break
            
            corr = compute_corr(h)
            max_c = np.max(corr)
            exp_c = np.exp(alpha * (corr - max_c))
            weights = exp_c / np.sum(exp_c)
            
            # Compute gradient
            first_sum = np.convolve(1.0 - h, weights, mode='full')[n-1:2*n-1]
            second_sum = np.convolve(h, weights[::-1], mode='full')[n-1:2*n-1]
            grad = first_sum - second_sum
            
            # Normalize gradient (the objective includes dx factor)
            # Actually, the objective is max(corr) * dx, so the gradient should also be scaled by dx
            # But for the optimization, the scaling doesn't matter (it just affects step size)
            # Let's not scale and adjust eta accordingly
            
            # Projected gradient step with backtracking
            for _ in range(50):
                h_new = project(h - eta * grad)
                # Check if objective decreased (respecting the smooth max)
                corr_new = compute_corr(h_new)
                max_new = np.max(corr_new)
                exp_new = np.exp(alpha * (corr_new - max_new))
                smooth_new = max_new + np.log(np.sum(exp_new)) / alpha
                
                corr_curr = compute_corr(h)
                max_curr = np.max(corr_curr)
                exp_curr = np.exp(alpha * (corr_curr - max_curr))
                smooth_curr = max_curr + np.log(np.sum(exp_curr)) / alpha
                
                if smooth_new < smooth_curr or eta < 1e-10:
                    break
                eta *= 0.5
            
            h = h_new
            eta *= 1.01  # slightly increase step size
            
        h = project(h)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    
    # Now try SLSQP refinement with high alpha
    for alpha in [1000, 5000, 10000, 20000]:
        if time.time() - t0 > budget_s - 60:
            break
        
        n_curr = len(best_h)
        target = n_curr / 2.0
        
        def obj(hv):
            corr = compute_corr(hv)
            m = np.max(corr)
            return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        
        def obj_grad(hv):
            corr = compute_corr(hv)
            m = np.max(corr)
            exp_c = np.exp(alpha * (corr - m))
            w = exp_c / np.sum(exp_c)
            first_sum = np.convolve(1.0 - hv, w, mode='full')[n_curr-1:2*n_curr-1]
            second_sum = np.convolve(hv, w[::-1], mode='full')[n_curr-1:2*n_curr-1]
            return first_sum - second_sum
        
        bounds = [(0.0, 1.0)] * n_curr
        constraints = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target, 
                       'jac': lambda hv: np.ones(n_curr)}]
        
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                      constraints=constraints, jac=obj_grad,
                      options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
        
        if res.x is not None:
            h_new = project(np.clip(res.x, 0.0, 1.0))
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
    
    # Try different resolutions
    for n_try in [2*n, 3*n, 4*n]:
        if n_try > 3000:
            continue
        if time.time() - t0 > budget_s - 90:
            break
        
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        
        target_try = n_try / 2.0
        def project_try(h):
            h = np.clip(h, 0.0, 1.0)
            for _ in range(80):
                s = np.sum(h)
                diff = target_try - s
                if abs(diff) < 1e-15:
                    break
                h = h + diff / n_try
                h = np.clip(h, 0.0, 1.0)
            return h
        
        h_init = project_try(h_init)
        
        for alpha in [100, 500, 2000]:
            if time.time() - t0 > budget_s - 60:
                break
            
            # Projected gradient
            eta = 1.0 / (alpha * n_try * 0.01)
            h = h_init.copy()
            for it in range(500):
                if time.time() - t0 > budget_s - 60:
                    break
                corr = compute_corr(h)
                max_c = np.max(corr)
                exp_c = np.exp(alpha * (corr - max_c))
                weights = exp_c / np.sum(exp_c)
                
                first_sum = np.convolve(1.0 - h, weights, mode='full')[n_try-1:2*n_try-1]
                second_sum = np.convolve(h, weights[::-1], mode='full')[n_try-1:2*n_try-1]
                grad = first_sum - second_sum
                
                for _ in range(30):
                    h_new = project_try(h - eta * grad)
                    corr_new = compute_corr(h_new)
                    max_new = np.max(corr_new)
                    exp_new = np.exp(alpha * (corr_new - max_new))
                    smooth_new = max_new + np.log(np.sum(exp_new)) / alpha
                    
                    if smooth_new < smooth_c or eta < 1e-10:
                        break
                    eta *= 0.5
                
                h = h_new
                eta *= 1.05
                
                # Track best
                s_try = evaluate_score(h)
                if s_try < best_score:
                    best_score = s_try
                    best_h = h.copy()
        
        h_init = project_try(h)
    
    # Final SLSQP polish
    if time.time() - t0 < budget_s - 30:
        n_fin = len(best_h)
        target_fin = n_fin / 2.0
        alpha_fin = 10000
