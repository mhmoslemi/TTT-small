def ihs_new_candidate(pop, pop_c5, n, mixing, mut_rate, rng):
    best_idx = np.argmin(pop_c5)
    n_pop = len(pop)
    if rng.random() < 0.7:
        other_idx = rng.integers(n_pop)
        if other_idx == best_idx:
            other_idx = (other_idx + 1) % n_pop
        m = mixing * (0.4 + 0.6 * rng.random())
        cand = pop[best_idx] + m * (pop[other_idx] - pop[best_idx])
    else:
        cand = pop[best_idx] + mixing * 0.5 * rng.standard_normal(n)
    if rng.random() < mut_rate:
        mu_scale = 0.005 + 0.02 * (1.0 - mixing)
        cand += mu_scale * rng.standard_normal(n)
    return project_feasible(cand, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_eps
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    _g_n = n
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: IHS Global Exploration
    pop_size = 50
    ihs_time_limit = min(180.0, max(60.0, budget_s * 0.15))
    max_ihs_iters = 8000

    pop = []
    pop_c5 = []
    for i in range(pop_size):
        ind = generate_diverse_individual(i, best_h, n, rng)
        pop.append(ind)
        pop_c5.append(evaluate_c5(ind))

    pop_c5 = np.array(pop_c5)

    for it in range(max_ihs_iters):
        elapsed = time.time() - start_time
        if elapsed > ihs_time_limit:
            break

        progress = min(1.0, it / (max_ihs_iters * 0.8))
        mixing = 0.5 * (1.0 - progress) + 0.15 * progress
        mut_rate = 0.4 * (1.0 - progress) + 0.05 * progress

        cand = ihs_new_candidate(pop, pop_c5, n, mixing, mut_rate, rng)
        c5 = evaluate_c5(cand)

        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()

        worst_idx = np.argmax(pop_c5)
        if c5 < pop_c5[worst_idx]:
            pop[worst_idx] = cand
            pop_c5[worst_idx] = c5
        elif rng.random() < 0.03 and it > 500:
            pop[worst_idx] = generate_diverse_individual(rng.integers(1, 5), best_h, n, rng)
            pop_c5[worst_idx] = evaluate_c5(pop[worst_idx])

    # Phase 2: SLSQP Local Refinement on top candidates
    sorted_idx = np.argsort(pop_c5)
    n_top = 6
    top_candidates = [pop[i].copy() for i in sorted_idx[:n_top]]
    if not np.allclose(best_h, top_candidates[0]):
        top_candidates.append(best_h.copy())
    top_candidates = list(set([tuple(c.tolist()) for c in top_candidates]))
    top_candidates = [np.array(c) for c in top_candidates]

    remaining = budget_s - (time.time() - start_time)
    slsqp_budget = remaining * 0.45
    eps_schedule = [30.0, 80.0, 150.0, 300.0, 600.0, 1200.0, 2500.0]

    for ci in range(len(top_candidates)):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 90.0:
            break

        cur = top_candidates[ci].copy()
        cur_c5 = evaluate_c5(cur)

        for eps in eps_schedule:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 40.0:
                break
            _g_eps = eps
            res = minimize(
                smooth_max_obj,
                cur.copy(),
                jac=smooth_max_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': eq_constraint}],
                options={'maxiter': 250, 'ftol': 1e-13}
            )
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < cur_c5:
                    cur_c5 = c5_new
                    cur = h_new.copy()
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()

    # Phase 3: Multi-start perturbation + SLSQP
    elapsed = time.time() - start_time
    n_pert = 0
    max_pert = 50
    while budget_s - (time.time() - start_time) > 80.0 and n_pert < max_pert:
        n_pert += 1
        scale = 0.003 + 0.006 * min(n_pert, 12) * rng.random()
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)

        _g_eps = 300.0
        res = minimize(
            smooth_max_obj,
            h_pert.copy(),
            jac=smooth_max_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_constraint}],
            options={'maxiter': 200, 'ftol': 1e-13}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Coordinate descent with adaptive step
    elapsed = time.time() - start_time
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]
    n_rounds = 0
    while budget_s - (time.time() - start_time) > 30.0 and n_rounds < 10:
        improved = False
        for step in step_sizes:
            for idx in range(n):
                for delta in [-step, step]:
                    h_trial = best_h.copy()
                    h_trial[idx] += delta
                    if h_trial[idx] < 0 or h_trial[idx] > 1:
                        continue
                    h_trial = project_feasible(h_trial, n)
                    c5 = evaluate_c5(h_trial)
                    if c5 < best_c5 - 1e-9:
                        best_c5 = c5
                        best_h = h_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
            elapsed = time.time() - start_time
            if budget_s - elapsed < 30.0:
                break
        n_rounds += 1

    # Phase 5: Final high-precision SLSQP
    elapsed = time.time() - start_time
    if budget_s - elapsed > 50.0:
        for eps in [5000.0, 10000.0]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 30.0:
                break
            _g_eps = eps
            res = minimize(
                smooth_max_obj,
                best_h.copy(),
                jac=smooth_max_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': eq_constraint}],
                options={'maxiter': 300, 'ftol': 1e-14}
            )
            if np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
