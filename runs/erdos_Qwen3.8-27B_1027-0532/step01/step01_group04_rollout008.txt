from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to entries that are < 1
        idx = np.where(h < 1.0)[0]
    else:
        # Need to decrease sum: subtract from entries that are > 0
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    # Distribute evenly among available entries, respecting bounds
    step = diff / len(idx)
    h[idx] += step
    h = np.clip(h, 0.0, 1.0)
    # If still not exact, do a second pass
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += diff2 / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def log_sum_exp_obj(h, alpha, target_sum, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_violation = np.sum(h) - target_sum
    lse += penalty * sum_violation ** 2
    return float(lse)

        # Sort lines by slope, use a convex hull / upper envelope algorithm
        # But that's O(n log n) per coordinate, still fine

        # for each pair of lines with different slopes, check their intersection
        # But limit to the most relevant pairs
        
        # Even simpler: since we have O(n) lines, just do a fine grid search
        # on [0,1] and refine
        x_candidates = np.array([0.0, 1.0])
        # Add the point where the "active" line has B closest to 0
        # The active line at x=0 is argmax(A), at x=1 is argmax(A+B)
        active_0 = np.argmax(A)
        active_1 = np.argmax(A + B)
        
        # The new "active" lines will have slopes between B[active_0] and B[active_1]
        # The minimum is where the slope of the upper envelope crosses 0
        
        # Quick approach: bisection on the derivative
        # f(x) = max_m (A[m] + B[m]*x)
        # f'(x) = B[argmax_m (A[m] + B[m]*x)] (at non-breakpoints)
        
        # We want f'(x) = 0, i.e., B[argmax] = 0
        # Since f is convex, f' is non-decreasing, so we can bisect
        
        lo, hi = 0.0, 1.0
        for _ in range(30):  # 30 iterations gives precision ~1e-9
            mid = (lo + hi) / 2
            vals = A + B * mid
            active = np.argmax(vals)
            if B[active] < 0:
                lo = mid
            elif B[active] > 0:
                hi = mid
            else:
                lo = hi = mid
                break
        
        x_star = (lo + hi) / 2
        x_candidates = np.append(x_candidates, [lo, hi, x_star])
        
        # Evaluate at all candidates
        best_val = 0.0
        best_obj = np.inf
        for xc in x_candidates:
            xc = np.clip(xc, 0.0, 1.0)
            obj_c = np.max(A + B * xc)
            if obj_c < best_obj:
                best_obj = obj_c
                best_val = xc
        
        h[i] = best_val
    
    # After the sweep, project to feasible
    h = project_feasible(h, target_sum)
    return h

The bisection will converge to the breakpoint in this case (since for x slightly less than the breakpoint, the active line has B < 0, and for x slightly more, B > 0). So actually, the bisection correctly handles this case!

1. For each coordinate i:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h[idx] += step
    h = np.clip(h, 0.0, 1.0)
    # Second pass if needed
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

        idx_minus = i - t
        valid_minus = (idx_minus >= 0) & (idx_minus < n)

        neg_idx_plus = np.clip(idx_plus, 0, n-1)
        B = B - np.where(valid_plus, h[neg_idx_plus], 0.0)
        # Subtract h[i-t] where valid
