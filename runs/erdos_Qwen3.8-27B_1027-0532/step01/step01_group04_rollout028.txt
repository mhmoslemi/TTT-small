def fista_smooth(h, alpha, target_sum, max_iter=5000, lr=None):
    n = len(h)
    # Lipschitz constant estimation
    # The gradient involves correlations which are at most n in magnitude
    # L ~ alpha * n (rough bound)
    L = alpha * n * 2  # conservative Lipschitz constant
    
    y = h.copy()
    t = 1.0
    h_prev = h.copy()
    
    for it in range(max_iter):
        # Compute gradient at y
        g = grad_smooth(y, alpha)
        
        # Gradient step
        z = y - g / L
        
        # Project
        h_new = project_box_hyperplane(z, target_sum)
        
        # Update y for next iteration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t - 1.0) / t_new * (h_new - h_prev)
        y = np.clip(y, 0.0, 1.0)  # keep in bounds
        # Actually for FISTA, y doesn't need to be in the feasible set
        # but it's safer to clip
        
        t = t_new
        h_prev = h_new
    
    return h_new
