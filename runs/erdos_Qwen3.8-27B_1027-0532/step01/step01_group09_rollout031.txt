import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def smooth_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp((corr - m) * T))) / T
    w = np.exp((corr - m) * T)
    w /= w.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-12:
            continue
        k = ki - (n - 1)
        if k >= 0:
            end = n - k
            grad[:end] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:end]
        else:
            kp = -k
            grad[kp:] += wk * (1.0 - h[:-kp])
            grad[:n - kp] -= wk * h[kp:]
    grad *= dx
    return val, grad

def optimize_smooth(h0, T, tm, maxiter=500):

    target = n / 2.0
    h0 = project_feasible(h0, target)
    c5_before = eval_c5(h0)

        return smooth_val_grad(x, T)

                c5_best = c1
                h_best = h1
        except Exception:

            grad[kp:] += (1.0 - h[:-kp])
