np.correlate(a, v, mode='full'):
The output has length len(a) + len(v) - 1 = 2n - 1.
output[k] = sum_{t} a[t] * v[t + k - (len(v) - 1)]
         = sum_{t: 0≤t<len(a), 0≤t+k-(len(v)-1)<len(v)} a[t] * v[t+k-(len(v)-1)]

For a=h, v=1-h, both of length n:
output[k] = sum_{t: 0≤t<n, 0≤t+k-(n-1)<n} h[t] * (1-h[t+k-(n-1)])

Let me substitute k' = k - (n-1):
output[k' + (n-1)] = sum_{t: 0≤t<n, 0≤t+k'<n} h[t] * (1-h[t+k'])

output[p] = sum_{t: 0≤t<n, 0≤t+(p-(n-1))<n} h[t] * (1-h[t+(p-(n-1))])

Let m = p - (n-1) (the lag):
output[p] = sum_{t: max(0,-m)≤t≤min(n-1, n-1-m)} h[t] * (1-h[t+m])

This is the "full" correlation. The score is max(output) * dx.

            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    init = project_sum(init, n / 2.0)
    best_h = init.copy()
    best_score = evaluate_score(init)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent from initial
    if time_left() > 120:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(80, time_left() - 40))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 2: SLSQP polish
    if time_left() > 80:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        for alpha in [200, 1000, 5000, 20000]:
            if time_left() < 30:
                break
            maxiter = max(100, 400000 // n_cur)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 3: CD on SLSQP result
    if time_left() > 100:
        h_cd = coordinate_descent(best_h, n_sweeps=500, time_budget=min(70, time_left() - 30))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 4: Random perturbations + CD
    n_cur = len(best_h)
    n_perturb = 120
    for i in range(n_perturb):
        if time_left() < 40:
            break
        frac = i / max(n_perturb - 1, 1)
        scale = 0.03 + 0.20 * frac
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_sum(h_pert, n_cur / 2.0)

        tb = min(4.0, (time_left() - 30) / max(n_perturb - i, 1))
        h_cd = coordinate_descent(h_pert, n_sweeps=200, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    # Phase 5: Try different resolutions
    base_n = len(best_h)
    res_list = [2 * base_n, int(base_n * 1.5), 3 * base_n]
    res_list = [r for r in res_list if 50 <= r <= 3000]
    for n_try in res_list:
        if time_left() < 80:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)

        tb_cd = min(45, time_left() - 50)
        h_cd = coordinate_descent(h_interp, n_sweeps=300, time_budget=tb_cd)

        if time_left() > 40:
            bounds_try = [(0.0, 1.0)] * n_try
            for alpha in [500, 5000]:
                if time_left() < 20:
                    break
                maxiter = max(50, 250000 // n_try)
                res_x = slsqp_optimize(h_cd, alpha, maxiter, bounds_try)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s = evaluate_score(h_new)
                        if s < best_score:
                            best_score = s
                            best_h = h_new
                            h_cd = h_new

        if time_left() > 30:
            h_cd2 = coordinate_descent(h_cd, n_sweeps=200, time_budget=min(20, time_left() - 10))
            s = evaluate_score(h_cd2)
            if s < best_score:
                best_score = s
                best_h = h_cd2

    # Phase 6: Binary initializations
    for bi in range(5):
        if time_left() < 50:
            break
        h_bin = np.zeros(base_n)
        ones_pos = rng.choice(base_n, size=base_n // 2, replace=False)
        h_bin[ones_pos] = 1.0
        h_bin = project_sum(h_bin, base_n / 2.0)
        tb = min(10.0, (time_left() - 20) / max(5 - bi, 1))
        h_cd = coordinate_descent(h_bin, n_sweeps=200, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 7: Final CD on best
    if time_left() > 40:
        h_cd = coordinate_descent(best_h, n_sweeps=800, time_budget=min(60, time_left() - 10))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Final SLSQP
    if time_left() > 20:
        n_final = len(best_h)
        bounds = [(0.0, 1.0)] * n_final
        for alpha in [10000, 50000, 200000]:
            if time_left() < 10:
                break
            res_x = slsqp_optimize(best_h, alpha, 600, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
