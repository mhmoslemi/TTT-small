    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_softmax(h, tau):
    """Gradient of soft-max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr

    w = np.exp(tc - mc)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_softmax(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    mc = np.max(tc)
    w = np.exp(tc - mc)
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = abs(m - (n - 1))
        wm = w[m]
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        else:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] += wm * (-h[k:])
    return grad * dx

def project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def adam_opt(h0, target, deadline, tau=50.0, lr=0.005):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    stall = 0
    best_lr = lr
    
    while time.time() < deadline:
        t += 1
        g = grad_softmax(h, tau)
        m_v = b1 * m_v + (1 - b1) * g
