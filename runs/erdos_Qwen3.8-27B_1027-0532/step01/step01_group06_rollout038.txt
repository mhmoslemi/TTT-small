def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_jacobian(h):
    n = len(h)
    J = np.zeros((n, 2 * n - 1))
    omh = 1.0 - h
    for i in range(n):
        for shift in range(-(n - 1), n):
            k = shift + (n - 1)
            q = i + shift
            if 0 <= q < n:
                J[i, k] += omh[q]
            p = i - shift
            if 0 <= p < n:
                J[i, k] -= h[p]
    return J

def lse_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    max_scaled = np.max(scaled)
    return dx * (cmax + T * (max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))))

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def try_slsqp(h, n, T, max_iter, ftol):
    try:
        def obj(x):
            return lse_obj(x, T, n)
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': ftol})

def try_lbfgsb(h, n, T, max_iter):
    try:
        def obj(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            scaled = (corr - cmax) / T
            max_s = np.max(scaled)
            return dx * (cmax + T * (max_s + np.log(np.sum(np.exp(scaled - max_s)))))
        def obj_grad(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            scaled = (corr - cmax) / T
            max_s = np.max(scaled)

            weights /= np.sum(weights)
            J = compute_jacobian(x)

            return grad
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',

def basin_hopping_step(h, rng, magnitude):
    n = len(h)
    perturbation = rng.randn(n) * magnitude
    h_new = h + perturbation
    h_new = project_feasible(h_new, n)
    return h_new

def cos_param_init(n, rng):
    a = rng.uniform(0.3, 1.5)
    p = rng.randint(1, n // 4 + 1)
    phase = rng.uniform(0, 2 * np.pi)
    b = rng.uniform(-0.5, 0.5)
    x = np.arange(n)
    h = 0.5 + a * np.cos(2 * np.pi * p * x / n + phase) + b
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h

def multiharmonic_init(n, rng, n_harm=3):
    h = np.full(n, 0.5)
    for k in range(1, n_harm + 1):

        h += amp * np.cos(2 * np.pi * p * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h
