def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))

    M2 = w[cols - rows + n - 1]

    Part2 = M2.dot(h)
