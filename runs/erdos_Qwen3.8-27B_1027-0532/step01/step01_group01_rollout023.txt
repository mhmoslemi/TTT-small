- Compute gradient of smooth_max
- Project the gradient step to satisfy sum(h) = n/2
- Project to [0,1]

This is essentially a projected gradient method (or FISTA for accelerated version).

Let me implement a **projected gradient / FISTA** approach:

```python
def projected_gradient_step(h, grad, step_size, n):
    # h - step_size * grad, then project to {x: 0<=x<=1, sum(x)=n/2}
    y = h - step_size * grad
    return project_feasible(y, n)
```

The projection onto {x: 0<=x<=1, sum(x)=n/2} is exactly what `project_feasible` does (it finds the right shift to make the sum correct, after clipping to [0,1]).

For FISTA, I need to handle the momentum step correctly with the constraint. The standard approach is:
- y_k = x_k + (t_{k-1}-1)/t_k * (x_k - x_{k-1})
- x_{k+1} = P(y_k - step_size * grad(y_k))
- t_{k+1} = (1 + sqrt(1+4t_k^2))/2

But the projection P is onto the constrained set, so this should work.

Let me also think about the step size. For the smooth max with epsilon, the Lipschitz constant of the gradient is roughly O(eps * n * dx) = O(eps * 2). So a safe step size would be 1/(eps * 2) or so. But we can also use backtracking.

Actually, for the log-sum-exp smooth max, the Hessian is bounded by eps * I (times the correlation structure), so the Lipschitz constant of the gradient is at most eps * (something). Let me just use a fixed step size and adjust.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0/eps) * np.log(np.sum(np.exp(eps * (corr - m))))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m_lag = k - (n - 1)
        if m_lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m_lag > 0:
            term = np.zeros(n)
            term[m_lag:] = 1.0 - h[:n - m_lag]
            term[:n - m_lag] += -h[m_lag:]
            grad += w * term
        else:
            p = -m_lag
            term = np.zeros(n)
            term[:n - p] = 1.0 - h[p:]
            term[p:] += -h[:n - p]
            grad += w * term
    
    grad *= dx
    return grad

def fista_optimize(h0, n, eps, max_iter=5000, tol=1e-10):
    """FISTA (accelerated projected gradient) for the smooth max objective."""
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    # Step size: use 1/(L) where L is Lipschitz constant of gradient
    # For smooth max with parameter eps, L ≈ eps * 2 (rough bound)
    # Be conservative
    step = 1.0 / (eps * 4.0)
    
    fx_prev = smooth_max_obj(z, eps)
    
    for it in range(max_iter):
        gz = smooth_max_grad(z, eps)
        # New x
        x_new = z - step * gz
        x_new = project_feasible(x_new, n)
        fx_new = smooth_max_obj(x_new, eps)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1.0) / t_new) * (x_new - x)
        z = project_feasible(z, n)
        t = t_new
        x = x_new
        
        if it % 100 == 0:
            if abs(fx_new - fx_prev) < tol:
                break
            fx_prev = fx_new
    
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(eps * (corr - m))))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            term = np.zeros(n)
            term[lag:] = 1.0 - h[:n - lag]
            term[:n - lag] += -h[lag:]
            grad += w * term
        else:
            p = -lag
            term = np.zeros(n)
            term[:n - p] = 1.0 - h[p:]
            term[p:] += -h[:n - p]
            grad += w * term
    grad *= dx
    return grad

def fista_optimize(h0, n, eps, max_iter=5000, step_scale=0.25):
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    step = step_scale / eps

    best_f = smooth_max_obj(x, eps)
    
    for it in range(max_iter):
        gz = smooth_max_grad(z, eps)
        x_new = z - step * gz
        x_new = project_feasible(x_new, n)
        f_new = smooth_max_obj(x_new, eps)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1.0) / t_new) * (x_new - x)
        z = project_feasible(z, n)
        t = t_new
        x = x_new
    
    return best_x

def coordinate_descent(h, n, max_rounds=200):

    c5 = evaluate_c5(h)
    candidates = [0.0, 1.0]
    for _ in range(max_rounds):
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            for new_val in candidates:
                if abs(new_val - h[i]) < 1e-8:

                h_trial = h.copy()
                h_trial[i] = new_val

                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5 - 1e-12:
                    h = h_trial
                    c5 = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return h, c5

    # Fix sum
    # Count ones
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n % 2 == 0:

    else:

    if n_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        for i in range(n_ones - target_ones):
            h[ones_idx[i]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        for i in range(target_ones - n_ones):
            h[zeros_idx[i]] = 1
    
    # Add the 0.5 if n is odd
    if n % 2 == 1:
        # Set one of the zeros to 0.5
        zeros_idx = np.where(h == 0)[0]
        if len(zeros_idx) > 0:
            h[zeros_idx[0]] = 0.5
    
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(max_iterations):
        improved = False
        # Try all single-bit flips (pairs: 1->0 and 0->1 simultaneously)
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # Try swapping: change h[i] from 1 to 0 and h[j] from 0 to 1
        for i in ones_idx:
            for j in zeros_idx:
                h_trial = h.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                # Handle the 0.5 case
                if n % 2 == 1:

                    c5 = c5_trial
                    improved = True

                        best_c5 = c5
                        best_h = h.copy()
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(eps * (corr - m))))) * dx

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < n:
                term = np.zeros(n)
                term[lag:] = 1.0 - h[:n - lag]
                term[:n - lag] += -h[lag:]
                grad += w * term
        else:
            p = -lag
            if p < n:
                term = np.zeros(n)
                term[:n - p] = 1.0 - h[p:]
                term[p:] += -h[:n - p]
                grad += w * term
    grad *= dx
    return grad

def slsqp_optimize(h0, n, eps, maxiter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: smooth_max_obj(x, eps),
        h0.copy(),
        jac=lambda x: smooth_max_grad(x, eps),
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    if np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h0.copy()

def fista_optimize(h0, n, eps, max_iter=8000, step_scale=0.5):
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    step = step_scale / (eps * n)
    best_x = x.copy()
    best_f = smooth_max_obj(x, eps)
    for it in range(max_iter):
        gz = smooth_max_grad(z, eps)
        x_new = z - step * gz
        x_new = project_feasible(x_new, n)
        f_new = smooth_max_obj(x_new, eps)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = x_new + ((t - 1.0) / t_new) * (x_new - x)
        z = project_feasible(z, n)
        t = t_new
        x = x_new
    return best_x

def coordinate_descent(h, n, max_rounds=300):
    h = h.copy()
    c5 = evaluate_c5(h)
    for _ in range(max_rounds):
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            for new_val in [0.0, 1.0]:
                if abs(new_val - h[i]) < 1e-10:
                    continue
                h_trial = h.copy()
                h_trial[i] = new_val
                h_trial = project_feasible(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5 - 1e-13:
                    h = h_trial
                    c5 = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def pairwise_swap_search(h, n, max_iters=2000, n_trials=100):
    h = h.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for _ in range(max_iters):
        improved = False
        ones_mask = h > 0.5
        zeros_mask = h < 0.5
        n_ones = np.sum(ones_mask)
        n_zeros = np.sum(zeros_mask)
        if n_ones == 0 or n_zeros == 0:
            break
        for _ in range(n_trials):
            i = np.random.choice(np.where(ones_mask)[0])
            j = np.random.choice(np.where(zeros_mask)[0])
            h_trial = h.copy()
            h_trial[i] = 0.0
            h_trial[j] = 1.0
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5 - 1e-12:
                h = h_trial
                c5 = c5_trial
                improved = True
                ones_mask[i] = False
                zeros_mask[i] = True
                ones_mask[j] = True
                zeros_mask[j] = False
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5
