        new_pop[0] = pop[best_prev_idx]
        new_fit[0] = fitness[best_prev_idx]

        for i in range(1, pop_size):
            p1 = tournament_select(pop_size, fitness, 3, rng)
            p2 = tournament_select(pop_size, fitness, 3, rng)

            if rng.random() < p_c:
                mask = rng.random(n) < 0.5
                child = np.where(mask, pop[p1], pop[p2])
            else:
                child = pop[p1].copy()

            if rng.random() < p_m:
                child = child + sigma_mu * rng.standard_normal(n)

            if rng.random() < 0.3:
                child = block_swap_mutation(child, rng, max(1, n // 30))

            if rng.random() < p_bin:
                child = binarize_pressure(child, rng, 0.5)

            child = project_onto_constraint(child, n / 2.0)
            new_pop[i] = child
            new_fit[i] = compute_c5(child)

        pop = new_pop
        fitness = new_fit

        gen_best_idx = int(np.argmin(fitness))
        if fitness[gen_best_idx] < best_ever - 1e-10:
            best_ever = fitness[gen_best_idx]
            best_h = pop[gen_best_idx].copy()
            best_c5 = best_ever
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 25:
            for j in range(5):
                worst_idx = int(np.argmax(fitness))
                pop[worst_idx] = project_onto_constraint(0.5 + 0.4 * rng.standard_normal(n), n / 2.0)
                fitness[worst_idx] = compute_c5(pop[worst_idx])
            no_improve = 10

        if (gen + 1) % refine_every == 0 and elapsed < budget_s - 100:
            order = np.argsort(fitness)
            for rank in range(min(2, pop_size)):
                el = time.time() - start_time
                if el > budget_s - 80:
                    break
                rem = min(budget_s - 80 - el, 80)
                cand = pop[order[rank]].copy()
                h_ref, c5_ref = local_refine(cand, [0.005, 0.001, 0.0003], 150, rem, start_time, budget_s)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
                    pop[order[rank]] = h_ref
                    fitness[order[rank]] = c5_ref

    elapsed = time.time() - start_time
    if elapsed < budget_s - 50:
        order = np.argsort(fitness)
        candidates = [pop[order[0]].copy(), pop[order[1]].copy(), best_h.copy()]
        seen = [None]
        unique_cands = []
        for c in candidates:
            is_dup = False
            for s in seen:
                if np.allclose(c, s, atol=1e-10):
                    is_dup = True
                    break
            if not is_dup:
                unique_cands.append(c)
                seen.append(c)

        for cand in unique_cands:
            el = time.time() - start_time
            if el > budget_s - 40:
                break
            rem = min(budget_s - 40 - el, 120)
            h_ref, c5_ref = local_refine(cand, [0.01, 0.003, 0.001, 0.0003, 0.0001], 200, rem, start_time, budget_s)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref

    elapsed = time.time() - start_time
    if elapsed < budget_s - 30:
        rem = budget_s - 30 - elapsed
        h_ref, c5_ref = local_refine(best_h, [0.0001, 0.00005, 0.00002], 300, rem, start_time, budget_s)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    elapsed = time.time() - start_time
    if elapsed < budget_s - 200 and len(best_h) < 180:
        n2 = 2 * len(best_h)
        h_up = np.repeat(best_h, 2).astype(float)
        h_up = project_onto_constraint(h_up, n2 / 2.0)
        rem = min(budget_s - 80 - elapsed, 500)
        h_ref, c5_ref = local_refine(h_up, [0.005, 0.001, 0.0003, 0.0001], 150, rem, start_time, budget_s)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    n_final = len(best_h)
    best_h = project_onto_constraint(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
