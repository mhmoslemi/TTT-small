def smooth_max_with_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    weights = np.exp(corr - m)
    total_w = np.sum(weights)
    weights /= total_w
    val = m + np.log(total_w) / beta

    # For each lag m, d = m - (n-1)
    # d/dh[j] corr[m] = dx * [(1-h[j+d]) if 0<=j+d<n else 0] - dx * [h[j-d] if 0<=j-d<n else 0]
    # Sum over m with weights[m]
    
    # Let's compute this efficiently
    # For a given j, the contribution to grad[j] is:
    # dx * sum_m weights[m] * (1-h[j+d_m]) * 1[0<=j+d_m<n] - dx * sum_m weights[m] * h[j-d_m] * 1[0<=j-d_m<n]
    # where d_m = m - (n-1)
    
    # Let's reindex: let d = d_m range from -(n-1) to (n-1)
    # w[d] = weights[d + n - 1]

    # For term 1: sum over d where 0 <= j+d < n, i.e., -j <= d < n-j
    #   w[d + n - 1] * (1 - h[j+d])
    # For term 2: sum over d where 0 <= j-d < n, i.e., j-n+1 < d <= j
    #   w[d + n - 1] * h[j-d]
    
    for j in range(n):
        # Term 1: d ranges from max(-(n-1), -j) to min(n-1, n-1-j)
        d_lo = max(-(n-1), -j)
        d_hi = min(n-1, n-1-j)
        if d_lo <= d_hi:
            d_range = np.arange(d_lo, d_hi+1)
            idx_range = d_range + n - 1
            grad[j] += dx * np.sum(w[idx_range] * (1.0 - h[j + d_range]))
        
        # Term 2: d ranges from max(-(n-1), j-n+1) to min(n-1, j)
        d_lo2 = max(-(n-1), j-n+1)
        d_hi2 = min(n-1, j)
        if d_lo2 <= d_hi2:
            d_range2 = np.arange(d_lo2, d_hi2+1)
            idx_range2 = d_range2 + n - 1
            grad[j] -= dx * np.sum(w[idx_range2] * h[j - d_range2])
    
    return val, grad

import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask].max()) if cnt > 0 else 0
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)

    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

        d_lo = max(-(n-1), -j)

            d_vals2 = np.arange(d_lo2, d_hi2+1)
            grad[j] -= np.dot(weights[d_vals2 + n - 1], h[j - d_vals2])

                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
    if res.x is not None and res.success or (res.x is not None and c5_value(res.x, dx) < c5_value(h0, dx)):

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))

                                        max_iter=20000, t0=0.3, t_end=0.005,
                                        time_limit=min(120, remaining() - 50))
        if c_sa < best_c5:
            best_h = h_sa
            best_c5 = c_sa

    # Phase 6: Final L-BFGS-B refinement with very low beta
    n = best_n
    dx = 2.0 / n
    if remaining() > 40:
        for beta in [10.0, 5.0, 3.0, 2.0]:
            if remaining() < 20:
                break
            h_opt = lbfgsb_opt(best_h, n, dx, beta, 5000.0, maxiter=500)
            c = c5_value(h_opt, dx)
            if c < best_c5:
                best_h = h_opt
                best_c5 = c

    # Phase 7: More discrete swaps on the refined solution
    if remaining() > 50:
        h_disc, c_disc = discrete_swap_search(best_h, n, dx,
                                              max_iter=2000, time_limit=min(60, remaining() - 20))
        if c_disc < best_c5:
            best_h = h_disc
            best_c5 = c_disc

    # Phase 8: Final continuous polish
    if remaining() > 20:
        for beta in [5.0, 3.0]:
            if remaining() < 15:
                break
            h_opt = lbfgsb_opt(best_h, n, dx, beta, 5000.0, maxiter=300)
            c = c5_value(h_opt, dx)
            if c < best_c5:
                best_h = h_opt
                best_c5 = c

    # Final repair and evaluation
    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
