import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fft_correlate(a, b):
    n, m = len(a), len(b)
    N = 2 ** (n + m - 2).bit_length()
    if N < n + m - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b[::-1], N)
    result = np.fft.irfft(fa * fb)
    return result[:n + m - 1]

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fft_correlate(h, one_minus_h)
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            free = h < 1.0
            if free.sum() == 0:
                break
            h[free] = np.minimum(1.0, h[free] + diff / free.sum())
        else:
            free = h > 0.0
            if free.sum() == 0:
                break
            h[free] = np.maximum(0.0, h[free] + diff / free.sum())
    return h

def compute_soft_gradient(h, overlaps, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    scaled = beta * overlaps
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    
    idx = np.argsort(overlaps)[-top_k:]
    w_idx = w[idx]
    w_sum = w_idx.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w_idx /= w_sum
    
    grad = np.zeros(n)
    for i, s_idx in enumerate(idx):
        s = s_idx - (n - 1)
        wi = w_idx[i]
        if s >= 0:
            nsmooth = n - s
            grad[:nsmooth] += wi * dx * (1.0 - h[s:s + nsmooth])
            grad[s:s + nsmooth] -= wi * dx * h[:nsmooth]
        else:
            m_age = -s
            nsmooth = n - m_age
            grad[:nsmooth] -= wi * dx * h[m_age:m_age + nsmooth]
            grad[m_age:m_age + nsmooth] += wi * dx * (1.0 - h[:nsmooth])
    
    return grad

def softmax_pg(h, n, max_steps, beta_start, beta_end, lr_start, top_k, 
               eval_interval=50, patience=200, rng=None):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_curr = h.copy()
    no_improve = 0
    
    for step in range(max_steps):
        frac = step / max(max_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start / (1.0 + 0.002 * step)
        
        overlaps = compute_overlaps(h_curr)
        grad = compute_soft_gradient(h_curr, overlaps, beta, top_k)
        
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        
        if step % eval_interval == 0 or step == max_steps - 1:
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += eval_interval

            else:

            h_curr = project_feasible(h_curr, n)
            no_improve = 0
        else:
            h_curr = h_new
    
    return best_h, best_c5

def slsqp_refine(h, n, maxiter=300):
    try:
        def obj(x):
            return evaluate_c5(x)
        def con_fun(x):
            return x.sum() - n / 2.0
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con_fun}],
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < np.inf:
            res.x = np.clip(res.x, 0.0, 1.0)
            return res.x, evaluate_c5(res.x)
        return h, evaluate_c5(h)
    except Exception:
        return h, evaluate_c5(h)

    t0 = time.time()

    if w_sum < 1e-15:

    for i, j in enumerate(idx):

        wi = w_idx[i]

            # For m in [k, n-1]: grad[m] += wi * dx * (1 - h[m-k])
            # For m in [0, n-1-k]: grad[m] -= wi * dx * h[m+k]
            grad[k:] += wi * dx * (1.0 - h[:n-k])

        else:

            # For m in [0, n-1-p]: grad[m] += wi * dx * (1 - h[m+p])

            grad[p:] -= wi * dx * h[:n-p]

import numpy as np

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fft_correlate(a, b):
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b[::-1], N)
    result = np.fft.irfft(fa * fb)
    return result[:n + m - 1]

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fft_correlate(h, one_minus_h)
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            if free.sum() == 0:
                break
            h[free] = np.minimum(1.0, h[free] + diff / free.sum())
        else:
            free = h > 0.0
            if free.sum() == 0:
                break
            h[free] = np.maximum(0.0, h[free] + diff / free.sum())
    return h

def compute_soft_gradient(h, overlaps, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    scaled = beta * overlaps
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    idx = np.argsort(overlaps)[-top_k:]
    w_idx = w[idx]
    w_sum = w_idx.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w_idx /= w_sum
    grad = np.zeros(n)
    for i, j in enumerate(idx):
        k = j - (n - 1)
        wi = w_idx[i]
        if k >= 0:
            grad[k:] += wi * dx * (1.0 - h[:n - k])
            grad[:n - k] -= wi * dx * h[k:]
        else:
            p = -k
            grad[:n - p] += wi * dx * (1.0 - h[p:])
            grad[p:] -= wi * dx * h[:n - p]
    return grad

def c5_objective_and_grad(x, n):
    dx = 2.0 / n
    one_minus_x = 1.0 - x
    corr = fft_correlate(x, one_minus_x)
    overlaps = corr * dx
    max_val = overlaps.max()
    active = np.where(overlaps >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for j in active:
        k = j - (n - 1)
        if k >= 0:
            grad[k:] += dx * (1.0 - x[:n - k])
            grad[:n - k] -= dx * x[k:]
        else:
            p = -k
            grad[:n - p] += dx * (1.0 - x[p:])
            grad[p:] -= dx * x[:n - p]
    grad /= len(active)
    return float(max_val), grad

def slsqp_refine(h, n, maxiter=300):
    def obj(x):
        val, _ = c5_objective_and_grad(x, n)
        return val
    def obj_grad(x):
        _, g = c5_objective_and_grad(x, n)
        return g
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < np.inf:
            res.x = np.clip(res.x, 0.0, 1.0)
            val, _ = c5_objective_and_grad(res.x, n)
            return res.x, val
        return h, evaluate_c5(h)
    except Exception:

def softmax_pg(h, n, max_steps, beta_start, beta_end, lr_start, top_k,
               eval_interval=30, patience=300, rng=None):

    no_improve = 0
    t_local = time.time()
    for step in range(max_steps):
        if time.time() - t_local > 500:
            break
        frac = step / max(max_steps - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start / (1.0 + 0.001 * step)
        overlaps = compute_overlaps(h_curr)
        grad = compute_soft_gradient(h_curr, overlaps, beta, top_k)
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, n)
        if step % eval_interval == 0 or step == max_steps - 1:

                best_h = h_new.copy()
                no_improve = 0
                current_c5 = c5_new
            else:
                no_improve += eval_interval
                current_c5 = c5_new

            no_improve = 0
        else:
            h_curr = h_new
    return best_h, best_c5
