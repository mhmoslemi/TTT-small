import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_and_worst_k(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    return np.max(corr) * dx, k_star

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def worst_case_gradient(h, k_star):
    """Compute gradient of the overlap for a specific shift k_star."""
    n = len(h)
    dx = 2.0 / n
    lag = k_star - (n - 1)
    g = np.zeros(n)
    j_lo = max(0, lag)
    j_hi = min(n - 1, n - 1 + lag)
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
    j2_lo = max(0, -lag)
    j2_hi = min(n - 1, n - 1 - lag)
    if j2_lo <= j2_hi:
        g[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return g * dx

def slsqp_optimize(h0, n, target_sum, deadline, tau=100.0, max_iter=1000):
    def obj_fun(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        tc = tau * corr
        mtc = np.max(tc)
        lse = mtc + np.log(np.sum(np.exp(tc - mtc)))
        return (lse / tau) * dx
    
    def grad_fun(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        tc = tau * corr
        mtc = np.max(tc)
        weights = np.exp(tc - mtc)
        weights /= np.sum(weights)
        g = np.zeros(n)
        for m in range(2 * n - 1):
            if weights[m] < 1e-14:
                continue
            lag = m - (n - 1)
            w = weights[m]
            j_lo = max(0, lag)
            j_hi = min(n - 1, n - 1 + lag)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += w * (1.0 - x[j_lo - lag:j_hi - lag + 1])
            j2_lo = max(0, -lag)
            j2_hi = min(n - 1, n - 1 - lag)
            if j2_lo <= j2_hi:
                g[j2_lo:j2_hi + 1] += w * (-x[j2_lo + lag:j2_hi + lag + 1])
        return g * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        x_opt = np.clip(res.x, 0.0, 1.0)
        x_opt = project_onto_constraint(x_opt, target_sum)
        return x_opt, compute_c5(x_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def pgd_worst_case(h0, n, target_sum, deadline, lr=0.05, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5, k_star = compute_c5_and_worst_k(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        g = worst_case_gradient(h, k_star)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        # Adaptive learning rate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            h = h_new
        else:
            h = h_new
            lr *= 0.9
            if lr < 0.001:
                lr = 0.05
                # Restart from best
                h = best_h.copy()
    return best_h, best_c5
