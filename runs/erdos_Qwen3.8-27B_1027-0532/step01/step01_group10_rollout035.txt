```python
def project_onto_constraint(x, target_sum):
    """Project x onto {x : sum(x) = target_sum, 0 <= x_i <= 1}"""
    # Use bisection on threshold
    lo = x.min() - target_sum - 1  # rough lower bound
    hi = x.max() + 1 - target_sum  # rough upper bound
    
    def do_project(theta):
        return np.clip(x - theta, 0, 1)
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = do_project(mid)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(x - (lo+hi)/2, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, target_sum):
    lo = h.min() - target_sum - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        y = np.clip(h - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_and_grad_fast(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    raw = np.exp(beta * (corr - m))
    Z = raw.sum()
    value = m + np.log(Z) / beta
    W = raw / Z
    u = 1.0 - h
    conv1 = np.convolve(W, u, mode="full")
    conv2 = np.convolve(W[::-1], h, mode="full")
    grad = dx * (conv1[n - 1:2 * n - 1] - conv2[n - 1:2 * n - 1])
    return value, grad


def projected_grad_descent(h, n, dx, beta, max_iter, lr_init=0.01):
    h = h.copy()
    target = n * 0.5
    h = project_onto_constraint(h, target)
    lr = lr_init
    prev_val, _ = smooth_max_and_grad_fast(h, n, dx, beta)
    for it in range(max_iter):
        val, grad = smooth_max_and_grad_fast(h, n, dx, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step_size = lr / grad_norm
        h_new = h - step_size * grad
        h_new = project_onto_constraint(h_new, target)
        new_val, _ = smooth_max_and_grad_fast(h_new, n, dx, beta)
        if new_val < prev_val:
            h = h_new
            prev_val = new_val
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.6
            if lr < 1e-10:
                break
    return h


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.arange(n_src, dtype=np.float64)
    x_dst = np.arange(n_dst, dtype=np.float64) * (n_src / n_dst)
    return np.interp(x_dst, x_src, h_src)


def perturb_h(h, sigma, rng):
    h_new = h + rng.normal(0, sigma, len(h))
    target = len(h) * 0.5
    return project_onto_constraint(h_new, target)


def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return project_onto_constraint(h, target)


def slsqp_objective(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + np.log(np.sum(np.exp(corr - m))) / beta


def slsqp_objective_grad(h, n, dx, beta):
    val, grad = smooth_max_and_grad_fast(h, n, dx, beta)
    return val, grad


def constraint_sum_fun(h, target):
    return h.sum() - target


def constraint_sum_jac(h, target):
    return np.ones(len(h))


def run_slsqp_local(h0, n, dx, beta, maxiter):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum_fun, 'jac': constraint_sum_jac, 'args': (target,)}]
    try:
        res = minimize(slsqp_objective_grad, h0, args=(n, dx, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return repair_h(res.x, n)
    except Exception:
        pass
    return repair_h(h0, n)


def run_nelder_mead_local(h0, n, dx, penalty, maxiter):
    def penalty_obj(h):
        base = c5_value(h, dx)
        sum_viol = abs(h.sum() - n * 0.5)
        bound_viol = np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h))
        return base + penalty * (sum_viol + bound_viol)

    try:
        res = minimize(penalty_obj, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
        if res.x is not None:
            return repair_h(res.x, n)
    except Exception:
        pass
    return repair_h(h0, n)
```
