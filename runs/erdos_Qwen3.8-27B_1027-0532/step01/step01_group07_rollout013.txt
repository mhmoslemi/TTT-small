def coord_polish(h, n, deadline, max_rounds=100):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        c5_val = compute_c5(h)
        c_all = compute_c5_all(h)
        
        # Find top-3 lags
        top_idx = np.argsort(c_all)[-3:]
        
        improved = False
        for lag in top_idx:
            shift = lag - (n - 1)
            # For this lag, c = sum_i h[i]*(1-h[i-shift])*dx (for valid i)
            # Try to reduce by adjusting h at positions that contribute most
            
            for j in range(n):
                if time.time() > deadline - 3:
                    break
                # Compute contribution of h[j] to this lag
                # h[j] appears when i=j (first factor) or i=j+shift (second factor)
                contrib = 0.0
                if 0 <= j - shift < n:
                    contrib += (1.0 - h[j - shift])
                if 0 <= j + shift < n:
                    contrib -= h[j + shift]
                
                if abs(contrib) < 1e-6:
                    continue
                
                # Try to reduce: if contrib > 0, decrease h[j]; if contrib < 0, increase h[j]
                direction = -1 if contrib > 0 else 1
                for delta in [0.02 * direction, 0.01 * direction, 0.05 * direction]:
                    h_new = h.copy()
                    h_new[j] += delta
                    h_new = project_feasible(h_new, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_val - 1e-14:
                        h = h_new
                        c5_val = c5_new
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    if compute_c5(h) < best_c5:
        best_h = h.copy()
        best_c5 = compute_c5(h)
    return best_h, best_c5

For a given lag m (0 to 2n-2), the shift is s = m - (n-1).
c[m] = ∑_i h[i] * (1-h[i-s]) * dx (for valid i, i.e., 0≤i<n and 0≤i-s<n)

The contribution of h[j] to c[m]:
- h[j] appears as the first factor when i=j: contributes (1-h[j-s]) * dx (if 0≤j-s<n)
- h[j] appears in the second factor (inside 1-()) when i-s=j, i.e., i=j+s: contributes -h[j+s] * dx (if 0≤j+s<n)

dx * [(1-h[j-s]) * I(0≤j-s<n) - h[j+s] * I(0≤j+s<n)]

To reduce c[m], we want to decrease h[j] if the contribution is positive, or increase h[j] if negative.

def smooth_grad(h, alpha):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
