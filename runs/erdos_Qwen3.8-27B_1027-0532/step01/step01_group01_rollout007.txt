from scipy.optimize import minimize
import time

_g_n = 85
_g_dx = 2.0 / 85
_g_eps = 100.0
_g_lam = 0.0
_g_bounds = None


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)


def _corr(h):
    return np.correlate(h, 1 - h, mode="full")


def _obj(h):
    n = _g_n
    dx = _g_dx
    eps = _g_eps
    corr = _corr(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    log_sum = np.log(np.sum(exp_s)) + max_s
    val = (m + (1.0 / eps) * log_sum) * dx
    val += _g_lam * (np.sum(h) - n / 2.0)
    return val


def _grad(h):
    n = _g_n
    dx = _g_dx
    eps = _g_eps
    corr = _corr(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        j1 = max(0, l)
        j2 = min(n, n + l)
        if j1 < j2:
            grad[j1:j2] += w * (1.0 - h[j1 - l:j2 - l])
        j3 = max(0, -l)
        j4 = min(n, n - l)
        if j3 < j4:
            grad[j3:j4] += w * (-h[j3 + l:j4 + l])
    grad *= dx
    grad += _g_lam
    return grad


def _set_params(n, eps, lam):

    _g_n = n
    _g_dx = 2.0 / n
    _g_eps = eps
    _g_lam = lam
    _g_bounds = [(0.0, 1.0)] * n


def _lbfgsb(h0, maxiter=200, maxls=50):
    res = minimize(
        _obj, h0.copy(), jac=_grad,
        method='L-BFGS-B', bounds=_g_bounds,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                 'maxls': maxls, 'maxfun': maxiter * 3, 'iprint': -1}
    )
    if np.all(np.isfinite(res.x)):
        return res.x
    return h0


def _dual_lbfgsb(h0, eps, n, max_bisect=20, max_iter=150):
    _set_params(n, eps, 0.0)
    h_curr = h0.copy()
    lo, hi = -10.0, 10.0
    target = n / 2.0
    for _ in range(max_bisect):
        lam = (lo + hi) / 2.0
        _g_lam = lam
        h_new = _lbfgsb(h_curr, maxiter=max_iter)

            lo = lam
            h_curr = h_new
        elif s < target - 0.05:
            hi = lam
            h_curr = h_new
        else:
            break
    h_final = project_feasible(h_curr, n)
    return h_final


def _block_perturb(h, n, rng):
    h_p = h.copy()

        bstart = rng.randint(0, n - blen)
        h_p[bstart:bstart + blen] += 0.02 * rng.randn(blen)
    return h_p
