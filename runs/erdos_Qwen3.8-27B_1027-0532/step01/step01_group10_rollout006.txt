from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def compute_grad_smooth_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = corr - np.max(corr)
    p = np.exp(beta * shifted)
    p /= np.sum(p)
    # p[k] for k in 0..2n-2, lag d = k - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2*n-1):
            d = k - (n-1)
            t = 0.0
            j1 = i - d
            if 0 <= j1 < n:
                t += (1.0 - h[j1])
            j2 = i + d
            if 0 <= j2 < n:
                t += (-h[j2])
            grad[i] += p[k] * dx * t
    return grad

def project_feasible(v, n):
    # Project v onto {h: sum(h) = n/2, 0 <= h <= 1}
    # Using bisection on the Lagrange multiplier
    target = n * 0.5
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_clip = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h_clip)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_clip = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(corr_vals, beta):
    m = np.max(corr_vals)
    return m + np.log(np.sum(np.exp(corr_vals - m))) / beta

def projected_gradient(h, n, dx, beta, max_iter, time_limit):
    t_start = time.time()
    h = h.copy()
    step = 0.1 / beta  # initial step size
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        grad = compute_grad(h, dx, beta, corr)
        # Backtracking line search
        best_t = step
        best_f = smooth_max_value(corr, beta)
        for ls in range(30):
            t = step * (0.5 ** ls)
            h_trial = h - t * grad
            h_trial = project_feasible(h_trial, n)
            corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode="full") * dx
            f_trial = smooth_max_value(corr_trial, beta)
            if f_trial < best_f:
                best_t = t
                best_f = f_trial
            else:
                break
        h = project_feasible(h - best_t * grad, n)
    return h

def compute_grad(h, dx, beta, corr_dx):
    n = len(h)
    shifted = corr_dx - np.max(corr_dx)
    p = np.exp(beta * shifted)
    p /= np.sum(p)
    r = p[n-1:].copy()  # length n: p[n-1], p[n], ..., p[2n-2]
    ones_minus_h = 1.0 - h
    # A[i] = sum_{j=0}^{i} r[i-j] * (1-h[j])
    A = np.convolve(r, ones_minus_h, mode='full')[:n]
    # B[i] = sum_{j=0}^{n-1-i} r[j] * h[j+i]
    B = np.zeros(n)
    for i in range(n):
        B[i] = np.dot(r[:n-i], h[i:])
    return dx * (A - B)

def project_feasible(v, n):
    target = n * 0.5
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if
