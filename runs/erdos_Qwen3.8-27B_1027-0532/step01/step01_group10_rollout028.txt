def coordinate_descent_pass(h, n, dx, max_rounds=5, tol=1e-10):
    """One full pass of coordinate descent."""
    for round in range(max_rounds):
        improved = False
        for j in range(n):
            # Compute alpha_k and beta_k for each lag k
            # corr[k] = sum_i h[i](1-h[i+s_k]) where s_k = n-1-k
            # As function of h[j]:
            # h[j] appears as h[i] when i=j: contributes h[j](1-h[j+s_k]) if 0<=j+s_k<n
            # h[j] appears as h[i+s_k] when i=j-s_k: contributes h[j-s_k](1-h[j]) if 0<=j-s_k<n
            #
            # corr[k] = (sum over i≠j-related) + h[j](1-h[j+s_k]) + h[j-s_k](1-h[j])
            # Let me compute the base (without h[j] terms) and the coefficient
            
            s_arr = n - 1 - np.arange(2*n - 1)  # s_k for k=0,...,2n-2
            # For each k, the coefficient of h[j] is:
            # (1 - h[j+s_k]) if 0<=j+s_k<n, else 0
            # minus h[j-s_k] if 0<=j-s_k<n, else 0
            
            js = j + s_arr  # j + s_k for each k
            jm = j - s_arr  # j - s_k for each k
            
            valid_plus = (js >= 0) & (js < n)
            valid_minus = (jm >= 0) & (jm < n)
            
            # Coefficient of h[j]:
            beta = np.zeros(2*n - 1)
            beta[valid_plus] = 1.0 - h[js[valid_plus]]
            beta[valid_minus] -= h[jm[valid_minus]]

The kinks of the linear functions occur where alpha_k + beta_k * t = alpha_l + beta_l * t, i.e., t = (alpha_l - alpha_k) / (beta_k - beta_l) for pairs with beta_k ≠ beta_l.

There are O(n²) such kinks (for 2n-1 functions), but most will be outside [0,1]. I can just evaluate at all kinks that fall in [0,1] plus the boundaries.

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def obj_val(h, dx, beta):
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full") * dx
    return smooth_max_val(corr, beta)

def obj_grad(h, dx, beta):
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full") * dx
    m = np.max(corr)
    w = np.exp(corr - m)
    w_sum = np.sum(w)
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = n - 1 - k
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        count = i_hi - i_lo + 1
        if count <= 0:
            continue
        wk = w[k]
        grad[i_lo:i_hi+1] += wk * v[i_lo+s:i_hi+s+1]
        grad[i_lo+s:i_hi+s+1] += -wk * h[i_lo:i_hi+1]
    grad *= dx / beta
    return grad

def constraint_sum_fun(h, n):
    return float(np.sum(h) - n * 0.5)

def constraint_sum_jac(h, n):
    return np.ones(len(h))

def run_slsqp_analytical(h0, n, dx, beta, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum_fun, 'jac': constraint_sum_jac, 'args': (n,)}]
    res = minimize(obj_val, h0, jac=lambda x: obj_grad(x, dx, beta),
                   args=(dx, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None:
        return repair_h(res.x, n)
    return h0

def coordinate_descent(h, n, dx, grid_size=80, max_rounds=3):
    h = h.copy()
    t_grid = np.linspace(0.0, 1.0, grid_size)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old_val = h[j]
            best_t = old_val
            best_c5 = c5_value(h, dx)
            curr = h[j]
            for t in t_grid:
                if abs(t - curr) < 1e-12:
                    continue
                h[j] = t
                repair_h_inplace(h, j, n)
                c = c5_value(h, dx)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_t = t
            h[j] = best_t
            repair_h_inplace(h, j, n)
            if abs(best_t - old_val) > 1e-10:
                improved = True
        if not improved:
            break
    return h

def repair_h_inplace(h, j, n):
    target = n * 0.5
    curr_sum = np.sum(h)
    diff = target - curr_sum
    if abs(diff) > 1e-15:
        h[j] = h[j] + diff
        h[j] = np.clip(h[j], 0.0, 1.0)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def generate_block_h(n, n_blocks, active_frac):
    h = np.zeros(n)
    block_size = n // n_blocks
    active_count = max(1, int(block_size * active_frac))
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + active_count, n)
        h[start:end] = 1.0
    return repair_h(h, n)

def generate_sinusoidal_h(n, freq, amp, phase=0.0):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / float(n) + phase)
    return repair_h(h, n)
