    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0

        alpha = alpha0 / (1.0 + 0.001 * it)

import numpy as np
from scipy.optimize import minimize
import time

_N = None

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            per = min(diff / mask.sum(), 1.0)
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            per = min(-diff / mask.sum(), 1.0)
            h[mask] = np.maximum(0.0, h[mask] - per)
    return h

def obj_func(x):
    return evaluate_c5(x)

def con_func(x):
    return np.sum(x) - _N / 2.0

def slsqp(h, n, maxiter=300, ftol=1e-12):
    global _N
    _N = n
    try:
        r = minimize(obj_func, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_func}],
                     options={'maxiter': maxiter, 'ftol': ftol})
        x = np.clip(r.x, 0.0, 1.0)
        x = repair_h(x, n)
        return x, evaluate_c5(x)
    except Exception:
        return h, evaluate_c5(h)

def subgradient_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    j1s = max(0, -k)
    j1e = min(n, n - k)
    if j1s < j1e:
        grad[j1s:j1e] += (1.0 - h[j1s + k:j1e + k])
    j2s = max(0, k)
    j2e = min(n, n + k)
    if j2s < j2e:
        grad[j2s:j2e] -= h[j2s - k:j2e - k]
    return grad, corr[m_star]

def subgradient_method(h, n, t_deadline, n_iters=200000):

            h_new[idx] = 1.0 - curr_h[idx]
        else:
            idx = rng.choice(n, size=4, replace=False)
            h_new = curr_h.copy()
            h_new[idx[0]], h_new[idx[1]] = h_new[idx[1]], h_new[idx[0]]
            h_new[idx[2]], h_new[idx[3]] = h_new[idx[3]], h_new[idx[2]]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_h(h_new, n)
        new_c5 = evaluate_c5(h_new)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / (0.003 * T + 1e-12)):
            curr_h = h_new
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s - 15

    h = initial_h_values.copy().astype(float)
    n = len(h)
    h = repair_h(h, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Phase 1: SLSQP on initial
    h_opt, c5_opt = slsqp(h, n, maxiter=400)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    if time.time() > t_deadline - 80:
        return best_h, best_c5, n

    # Phase 2: Subgradient polishing
    h_sg, c5_sg = subgradient_method(best_h, n, t_deadline - 300, n_iters=50000)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 3: SLSQP after subgradient
    h_opt, c5_opt = slsqp(best_h, n, maxiter=300)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    if time.time() > t_deadline - 150:
        if time.time() < t_deadline - 25:
            h_f, c5_f = slsqp(best_h, n, maxiter=800)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f
        return best_h, best_c5, n

    # Phase 4: Stochastic tunneling
    best_h, best_c5 = sto_tunneling(best_h, n, t_deadline, rng, n_restarts=40)

    # Phase 5: Refine best from tunneling
    if time.time() < t_deadline - 40:
        h_opt, c5_opt = slsqp(best_h, n, maxiter=500)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() > t_deadline - 80:
        return best_h, best_c5, n

    # Phase 6: Annealing from best
    best_h2, best_c5_2 = anneal(best_h, n, t_deadline, rng, n_iter=40000)
    if best_c5_2 < best_c5:
        best_h, best_c5 = best_h2, best_c5_2
        if time.time() < t_deadline - 30:
            h_f, c5_f = slsqp(best_h, n, maxiter=500)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    # Phase 7: Second tunneling round if time allows
    if time.time() < t_deadline - 120:
        best_h, best_c5 = sto_tunneling(best_h, n, t_deadline, rng, n_restarts=25)
        if time.time() < t_deadline - 25:
            h_f, c5_f = slsqp(best_h, n, maxiter=800)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    # Final: subgradient + SLSQP polish
    if time.time() < t_deadline - 40:
        h_sg2, c5_sg2 = subgradient_method(best_h, n, t_deadline - 15, n_iters=30000)
        if c5_sg2 < best_c5:
            best_h, best_c5 = h_sg2, c5_sg2
        if time.time() < t_deadline - 15:
            h_f, c5_f = slsqp(best_h, n, maxiter=600)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    return best_h, best_c5, n
