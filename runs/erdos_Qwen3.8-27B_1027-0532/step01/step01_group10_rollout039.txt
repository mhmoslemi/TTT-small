    # This is an approximation that works well when beta is large

import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_h(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_value(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)

def compute_gradient(h, beta, dx):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    
    first = np.correlate(p, g, mode="full")[:n]
    second = np.convolve(p, h, mode="full")[n-1:2*n-1]
    
    return dx * (first - second)

def projected_grad_descent(h0, n, dx, beta, max_iter, lr0=0.01, verbose=False):
    h = h0.copy()
    target = n * 0.5
    lr = lr0
    prev_obj = smooth_max_value(h, beta, dx)
    best_obj = prev_obj
    best_h = h.copy()
    stall_count = 0
    
    for it in range(max_iter):
        obj = smooth_max_value(h, beta, dx)
        grad = compute_gradient(h, beta, dx)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Backtracking line search (Armijo condition)
        step = lr
        improved = False
        c = 1e-4
        for _ in range(40):
            h_new = project_h(h - step * grad, target)
            obj_new = smooth_max_value(h_new, beta, dx)
            if obj_new < obj - c * step * grad_norm * grad_norm:
                h = h_new
                improved = True
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_h = h_new.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                break
            step *= 0.5
        
        if not improved:
            # Try a very small step
            h_new = project_h(h - step * grad, target)
            obj_new = smooth_max_value(h_new, beta, dx)
            if obj_new <= obj + 1e-13:
                h = h_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_h = h_new.copy()
            break
        
        # Adaptive learning rate (Barzilai-Borwein-like)
        # Simple: if we improved a lot, increase lr; if barely, decrease
        if obj_new < best_obj:
            lr = min(step * 1.5, lr0 * 20.0)
        else:
            lr = max(step, lr0 * 0.01)
    
    # Return the best found, not the final
    return best_h if best_obj < prev_obj else repair_h(h, n)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

We have:
C_k = dx * sum_i h[i] * (1-h[i+shift_k]) where shift_k = k-(n-1)

This is the overlap at lag k.

The smooth max: f(h) = (1/beta) * log(sum_k exp(beta * C_k))

Now dC_k/dh[j]:
C_k/dx = sum_i h[i] * (1-h[i+shift]) where shift = shift_k

d/dh[j] of this:

import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def project_h(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_value(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)


def compute_gradient(h, beta, dx):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    first = np.correlate(p, g, mode="full")[:n]
    second = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    return dx * (first - second)


def projected_grad_momentum(h0, n, dx, beta, max_iter, lr0=0.02, momentum=0.9, time_limit=None, t_start=None):
    h = h0.copy()
    target = n * 0.5
    lr = lr0
    v = np.zeros(n)
    best_obj = smooth_max_value(h, beta, dx)
    best_h = h.copy()
    stall = 0

    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        obj = smooth_max_value(h, beta, dx)
        grad = compute_gradient(h, beta, dx)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        v = momentum * v - lr * grad
        step = 1.0
        improved = False
        for _ in range(40):
            h_new = project_h(h + step * v, target)
            obj_new = smooth_max_value(h_new, beta, dx)
            if obj_new < obj - 1e-14:
                h = h_new
                improved = True
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                break
            step *= 0.5

        if not improved:
            if stall > 20:
                break
            h = project_h(h + step * v, target)
            obj_new = smooth_max_value(h, beta, dx)
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
            if step < 1e-12:
                break
            stall += 1

        lr = min(lr * 1.05, lr0 * 50.0) if improved else max(lr * 0.9, lr0 * 0.01)

    return best_h


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)


def generate_structured_starts(n, rng):
    starts = []
    x = np.arange(n, dtype=np.float64)
    h = np.full(n, 0.5)
    starts.append(repair_h(h, n))
    for freq in [1, 2, 3, 4, 5, 7, 8, 12, 16]:
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n)
        starts.append(repair_h(h, n))
    for freq in [2, 3, 5]:
        h = 0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x / n) + 0.15 * np.sin(4.0 * np.pi * x / n)
        starts.append(repair_h(h, n))
    block_size = max(1, n // 8)
    for phase in range(3):
        h = np.zeros(n)
        for i in range(n):
            cycle = (i + phase) // block_size
            h[i] = 1.0 if (cycle % 2 == 0) else 0.0
        starts.append(repair_h(h, n))
    for _ in range(4):
        h = rng.uniform(0.0, 1.0, n)
        starts.append(repair_h(h, n))
    tm = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = 1.0 if c % 2 == 0 else 0.0
    starts.append(repair_h(tm, n))
    return starts
