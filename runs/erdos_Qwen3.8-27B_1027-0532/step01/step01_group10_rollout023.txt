The previous approach used SLSQP (which uses a QP subproblem for constraints) and Nelder-Mead (a simplex method). Both are local methods that can get stuck.

Coordinate descent with exact per-coordinate minimization is also a local method, but:
1. It's much faster per iteration (can do many more sweeps)
2. The per-coordinate step is "exact" (finds the best value for that coordinate given others fixed)
3. It can escape some types of local minima that gradient-based methods get stuck in (because it does large jumps in individual coordinates)

To avoid getting stuck, I can:
1. Use different coordinate orderings (random, forward, backward)
2. Do multiple restarts with perturbed initial points

- f_k(t) = A_k * t + (corr[k] - A_k * old) [linear, for m ≠ 0]

```python
import numpy as np
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-16
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def coord_best_t(correlation, h, j, n):
    """Find best value for h[j] given current correlation and h."""
    old = h[j]
    if old == 0.0 or old == 1.0:
        # Still evaluate, but the quadratic term is simpler
        pass
    
    # Determine affected shifts
    # m range: max(-j, j-n+1) to min(n-1-j, j)
    lo = max(-j, j - n + 1)
    hi = min(n - 1 - j, j)
    
    # For each affected shift m, compute coefficient A_m of (t - old)
    shifts = np.arange(lo, hi + 1)
    A = np.zeros(len(shifts))
    
    for idx, m in enumerate(shifts):
        a = 0.0
        if m == 0:
            # Quadratic case, handle separately
            A[idx] = np.nan  # marker
            continue
        elif m > 0:
            # j in first factor: m < n-j, i.e., j+m < n
            if j + m < n:
                a += (1.0 - h[j + m])
            # j in second factor: m <= j, i.e., j-m >= 0
            if m <= j:
                a -= h[j - m]
        else:
            p = -m
            # j in first factor: p <= j
            if p <= j:
                a += (1.0 - h[j - p])
            # j in second factor: p <= n-1-j
            if p <= n - 1 - j:
                a -= h[j + p]
        A[idx] = a
    
    # For m=0, the function is: corr[n-1] + (t-old)(1-t-old)
    # = corr[n-1] + t - old - t^2 + old*t + old - old^2 ... 
    # Let me just expand: (t-old)(1-t-old) = t(1-t-old) - old(1-t-old)
    # = t - t^2 - old*t - old + old*t + old^2 = t - t^2 + old^2 - old
    
    # So f_0(t) = corr[n-1] + t - t^2 + old^2 - old
    # For linear terms: f_m(t) = corr[n-1+m] + A_m * (t - old)
    
    # Build the function to evaluate
    # Grid search
    num_grid = 200
    t_grid = np.linspace(0.0, 1.0, num_grid)
    
    # Evaluate g(t) = max of all affected f_k(t)
    # But we also need to consider UN affected lags!
    # The max over ALL lags includes lags not affected by h[j]
    # Those lags have value corr[k] (unchanged)
    
    # So g(t) = max(max_{unaffected} corr[k], max_{affected} f_k(t))
    # The max over unaffected lags is just a constant: max_corr_unaffected
    
    # Find max over unaffected lags
    affected_indices = np.arange(n - 1 + lo, n - 1 + hi + 1)
    unaffected_mask = np.ones(2 * n - 1, dtype=bool)
    unaffected_mask[affected_indices] = False
    max_unaffected = np.max(correlation[unaffected_mask]) if np.any(unaffected_mask) else -np.inf
    
    # Now evaluate g(t) on the grid
    best_t = old
    best_g = np.inf
    
    for t in t_grid:
        # Quadratic term (m=0)
        g_val = correlation[n-1] + t - t*t + old*old - old
        
        # Linear terms
        for idx, m in enumerate(shifts):
            if m == 0:
                continue
            val = correlation[n-1+m] + A[idx] * (t - old)
            if val > g_val:
                g_val = val
        
        # Also consider unaffected
        if max_unaffected > g_val:
            g_val = max_unaffected
        
        if g_val < best_g:
            best_g = g_val
            best_t = t
    
    return best_t, best_g
```

```python
    for j in order:
        old = h[j]
        lo = max(-j, j - n + 1)
        hi = min(n - 1 - j, j)
        m_arr = np.arange(lo, hi + 1)
        n_shifts = len(m_arr)
        
        # Affected correlation indices
        aff_k = n - 1 + m_arr
        
        # Unaffected max
        mask = np.ones(2*n-1, dtype=bool)
        mask[aff_k] = False
        max_unaff = np.max(corr[mask]) if np.any(mask) else -np.inf
        
        # Compute A
        jm = j + m_arr
        jm_v = (jm >= 0) & (jm < n)
        jm_sub = j - m_arr
        jm_sub_v = (jm_sub >= 0) & (jm_sub < n)
        
        A = np.zeros(n_shifts)
        A[jm_v] += (1.0 - h[jm[jm_v]])
        A[jm_sub_v] -= h[jm_sub[jm_sub_v]]
        A[m_arr == 0] = 0.0
        
        # Current correlation values at affected lags
        corr_aff = corr[aff_k]
        
        # Grid search
        num_grid = 300
        t_grid = np.linspace(0.0, 1.0, num_grid)
        dt = t_grid - old
        
        # Build f matrix
        f_lin = np.outer(dt, A) + corr_aff[None, :]  # (num_grid, n_shifts)
        
        # Find the quadratic index
        quad_idx = np.where(m_arr == 0)[0]
        if len(quad_idx) > 0:
            qi = quad_idx[0]
            f_quad = corr_aff[qi] + t_grid - t_grid**2 + old**2 - old
            f_lin[:, qi] = f_quad
        
        # Add unaffected
        f_all = np.column_stack([f_lin, np.full(num_grid, max_unaff)])
        
        g = np.max(f_all, axis=1)
        best_t = t_grid[np.argmin(g)]
        
        # Update
        delta = best_t - old
        if abs(delta) > 1e-15:
            h[j] = best_t
            corr[aff_k] += A * delta
    
    return h, corr
```
