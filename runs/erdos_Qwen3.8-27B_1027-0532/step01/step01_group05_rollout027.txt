    best_h = h.copy()

    best_h = h.copy()

def lbfgs_optimize(h0, n, t, max_iter=200, top_k=40):
    # Augmented Lagrangian approach
    h = h0.copy()
    lam = 0.0
    mu = 1.0
    
    def aug_obj(hx):
        val, _ = topk_gradient(hx, n, t, top_k)
        s = np.sum(hx) - n / 2.0
        return val + 0.5 * mu * s * s + lam * s
    
    def aug_grad(hx):
        _, grad = topk_gradient(hx, n, t, top_k)
        s = np.sum(hx) - n / 2.0
        grad = grad + (mu * s + lam)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(aug_obj, h.copy(), method='L-BFGS-B', jac=aug_grad,
                      bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-6})
    h = result.x
    
    # A few more outer iterations to tighten the constraint
    for _ in range(3):

        if abs(s) < 1e-6:
            break
        lam += mu * s
        if abs(s) > 0.05:
            mu *= 5.0
        
        result = minimize(aug_obj, h.copy(), method='L-BFGS-B', jac=aug_grad,
                         bounds=bounds, options={'maxiter': 50, 'ftol': 1e-12})
        h = result.x

    weights = exps / sum_e

    obj = (max_v + np.log(sum_e) / t) * dx

    g = np.zeros(n)
    center = n - 1
    for i in range(k):
        w = weights[i]
        if w < 1e-10:
            continue
        lag = top_idx[i] - center
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:n - lag]
        else:
            l = -lag
            g[l:] += w * (1.0 - h[:n - l])
            g[:n - l] -= w * h[l:]
    g *= dx
    return obj, g


def lbfgs_augmented_optimize(h0, n, t, top_k, max_inner=200, n_outer=5):
    h = h0.copy()
    lam = 0.0
    mu = 1.0
    bounds = [(0.0, 1.0)] * n

    for outer in range(n_outer):
        def obj_fn(hx):
            val, _ = topk_softmax_gradient(hx, n, t, top_k)
            s = np.sum(hx) - n / 2.0
            return val + 0.5 * mu * s * s + lam * s

        def grad_fn(hx):
            _, g = topk_softmax_gradient(hx, n, t, top_k)
            s = np.sum(hx) - n / 2.0
            return g + (mu * s + lam)

        result = minimize(obj_fn, h.copy(), method='L-BFGS-B', jac=grad_fn,
                          bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-13, 'gtol': 1e-7})
        h = result.x
        h = np.clip(h, 0.0, 1.0)

        s = np.sum(h) - n / 2.0
        lam += mu * s
        if abs(s) > 0.1:
            mu *= 5.0
        if abs(s) < 1e-5:
            break

    h = project_onto_constraint(h, n)
    return h


def momentum_subgrad_polish(h, n, iterations, top_k):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.02
    mom = 0.8

    for it in range(iterations):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = min(top_k, len(corr))
        part = np.argpartition(-corr, k)[:k]
        top_idx = part[np.argsort(-corr[part])]

        max_v = corr[top_idx[0]]
        vals = corr[top_idx] - max_v
        vals = np.clip(vals, -500.0, 0.0)
        exps = np.exp(30.0 * vals)
        weights = exps / np.sum(exps)

        g = np.zeros(n)
        center = n - 1
        for i in range(k):
            w = weights[i]
            if w < 1e-10:
                continue
            lag = top_idx[i] - center
            if lag == 0:
                g += w * (1.0 - 2.0 * h)
            elif lag > 0:
                g[:n - lag] += w * (1.0 - h[lag:])
                g[lag:] -= w * h[:n - lag]
            else:
                l = -lag
                g[l:] += w * (1.0 - h[:n - l])
                g[:n - l] -= w * h[l:]
        g *= (2.0 / n)

        v = mom * v + g
        hn = h - step * v
        hn = project_onto_constraint(hn, n)
        c5 = compute_c5(hn, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn.copy()
            h = hn
        else:
            h = hn
            step *= 0.98
        step = max(step, 0.001)
        mom = min(mom + 0.00005, 0.95)

    return best_h, best_c5


def active_set_alternating(h, n, max_rounds=50, top_k=5):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    n_center = n - 1

    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_part = np.argpartition(-corr, top_k)[:top_k]
        top_idx = top_part[np.argsort(-corr[top_part])]
        lags = top_idx - n_center

        active_grad = np.zeros(n)
        for lag in lags:
            active_grad += gradient_single_lag(h, n, lag)
        active_grad /= len(lags)

        # Minimize sum of active lags using a few gradient steps
        step = 0.01
        for _ in range(30):
            h = h - step * active_grad
            h = project_onto_constraint(h, n)
            step *= 0.95

        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    parent = np.array(initial_h_values, dtype=float)
    n0 = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(best_h, n0)

    def remaining():
        return budget_s - (time.time() - start)

    # Phase 1: Optimize on parent grid with increasing temperature
    temps_parent = [15, 30, 60, 120, 250, 500]
    h_cur = parent.copy()
    for t in temps_parent:
        if remaining() < 60:
            break
        h_opt = lbfgs_augmented_optimize(h_cur, n0, t, top_k=50, max_inner=200, n_outer=4)
        c5_opt = compute_c5(h_opt, n0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if c5_opt < compute_c5(h_cur, n0):
            h_cur = h_opt

    # Phase 2: Active set alternating on parent grid
    if remaining() > 100:
        h_as, c5_as = active_set_alternating(h_cur, n0, max_rounds=30, top_k=5)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as

    # Phase 3: Grid refinement (2x)
    n2 = n0 * 2
    if remaining() > 200 and n2 <= 600:
        x_c = np.linspace(0, 2, len(best_h), endpoint=False)
        x_f = np.linspace(0, 2, n2, endpoint=False)
        h_fine = np.interp(x_f, x_c, best_h)
        h_fine = project_onto_constraint(h_fine, n2)
        c5_fine = compute_c5(h_fine, n2)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

        for t in [30, 80, 150, 300, 500]:
            if remaining() < 40:
                break
            h_opt = lbfgs_augmented_optimize(h_fine, n2, t, top_k=50, max_inner=150, n_outer=4)
            c5_opt = compute_c5(h_opt, n2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_fine = h_opt.copy()
            elif c5_opt < compute_c5(h_fine, n2):
                h_fine = h_opt

        if remaining() > 60:
            h_as, c5_as = active_set_alternating(h_fine, n2, max_rounds=20, top_k=5)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()

    # Phase 4: Grid refinement (3x)
    n3 = n0 * 3
    if remaining() > 150 and n3 <= 900:
        x_c = np.linspace(0, 2, len(best_h), endpoint=False)
        x_f = np.linspace(0, 2, n3, endpoint=False)
        h_fine3 = np.interp(x_f, x_c, best_h)
        h_fine3 = project_onto_constraint(h_fine3, n3)
        c5_fine3 = compute_c5(h_fine3, n3)
        if c5_fine3 < best_c5:
            best_c5 = c5_fine3
            best_h = h_fine3.copy()

        for t in [50, 150, 300]:
            if remaining() < 30:
                break
            h_opt = lbfgs_augmented_optimize(h_fine3, n3, t, top_k=40, max_inner=100, n_outer=3)
            c5_opt = compute_c5(h_opt, n3)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_fine3 = h_opt.copy()

    # Phase 5: Momentum subgradient polish on best grid
    if remaining() > 50:
        n_pol = len(best_h)
        n_iters = min(int(remaining() * 5), 20000)
        h_pol, c5_pol = momentum_subgrad_polish(best_h, n_pol, iterations=n_iters, top_k=25)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 6: Random perturbations and quick re-optimization
    if remaining() > 80:
        n_try = len(best_h)
        for i in range(20):
            if remaining() < 20:
                break
            noise_scale = 0.02 + 0.03 * (i % 4)
            h_pert = best_h + rng.randn(n_try) * noise_scale
            h_pert = project_onto_constraint(h_pert, n_try)
            t_use = [50, 100, 200, 400][i % 4]
            h_opt = lbfgs_augmented_optimize(h_pert, n_try, t_use, top_k=40, max_inner=100, n_outer=3)
            c5_opt = compute_c5(h_opt, n_try)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final subgradient polish
    if remaining() > 30:
        n_pol = len(best_h)
        n_iters = min(int(remaining() * 8), 15000)
        h_pol, c5_pol = momentum_subgrad_polish(best_h, n_pol, iterations=n_iters, top_k=20)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final validation
    best_h = project_onto_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
