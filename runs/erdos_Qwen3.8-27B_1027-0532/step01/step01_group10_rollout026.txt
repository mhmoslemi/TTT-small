    h_nm = run_nelder_mead(best_h, n, dx, 200.0, maxiter=30000)
    c = c5_value(h_nm, dx)
    if c < best_c5:
        best_c5 = c
        best_h = h_nm.copy()

    for beta in [200.0, 100.0, 50.0]:
        if budget_s - (time.time() - t0) < 60:
            break
        h_opt = run_slsqp(best_h, n, dx, beta, maxiter=1500)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    order = np.arange(n)
    for restart in range(5):
        if budget_s - (time.time() - t0) < 30:
            break
        h_restarted = best_h.copy()
        if restart > 0:
            perturb = rng.uniform(-0.02, 0.02, n)
            h_restarted = np.clip(h_restarted + perturb, 0.0, 1.0)
            h_restarted = repair_h(h_restarted, n)
        for sweep in range(300):
            if budget_s - (time.time() - t0) < 15:
                break
            rng.shuffle(order)
            old_c5_r = c5_value(h_restarted, dx)
            new_c5_r = coord_desc_sweep(h_restarted, n, dx, order, n_iters=40)
            if abs(old_c5_r - new_c5_r) < 1e-13:
                break
        c_restarted = c5_value(h_restarted, dx)
        if c_restarted < best_c5:
            best_c5 = c_restarted
            best_h = h_restarted.copy()

    h_nm_final = run_nelder_mead(best_h, n, dx, 500.0, maxiter=20000)
    c = c5_value(h_nm_final, dx)
    if c < best_c5:
        best_c5 = c
        best_h = h_nm_final.copy()

    for sweep in range(200):
        if budget_s - (time.time() - t0) < 10:
            break
        rng.shuffle(order)
        old_c5 = c5_value(best_h, dx)
        new_c5 = coord_desc_sweep(best_h, n, dx, order, n_iters=45)
        if new_c5 < best_c5:
            best_c5 = new_c5
        if abs(old_c5 - new_c5) < 1e-14:
            break

    best_h = repair_h(best_h, n)
    best_c5 = c5_value(best_h, 2.0 / n)

    return best_h, best_c5, n
