    B = np.zeros(n)  # B[j] = sum_k w[k] * h[j-k] * 1[valid]
    
    # w has length 2n-1, w[idx] corresponds to lag k = idx - (n-1)
    for idx in range(2*n - 1):
        k = idx - (n-1)
        wi = w[idx]
        if abs(wi) < 1e-15:
            continue
        if k >= 0:
            # j+k valid: 0 <= j+k < n => 0 <= j < n-k
            A[:n-k] += wi * (1.0 - h[k:])
            B[k:] += wi * h[:n-k]
        else:
            # j+k valid: 0 <= j+k < n => -k <= j < n
            A[-k:] += wi * (1.0 - h[:n+k])
            B[:n+k] += wi * h[-k:]
    
    grad = dx * (A - B)
    return grad
