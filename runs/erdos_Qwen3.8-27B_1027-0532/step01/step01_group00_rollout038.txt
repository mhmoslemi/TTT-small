def project_to_simplex(x, target):
    """Project x onto {y : 0 <= y_i <= 1, sum(y) = target}"""
    n = len(x)
    lo, hi = x.min() - target - 1, x.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    y = np.clip(x - mid, 0, 1)
    # Fix numerical issues
    diff = target - y.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound variables
        free = (y > 0) & (y < 1)
        if free.sum() > 0:
            y[free] += diff / free.sum()
            np.clip(y, 0, 1, out=y)
    return y

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_constraint(x, target_sum):
    n = len(x)
    lo = x.min() - 2.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - mid, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-13:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.sum() > 0:
            y[free] += diff / free.sum()
            np.clip(y, 0.0, 1.0, out=y)
    return y


def compute_gradient_analytic(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    c_max = np.max(corr)
    logits = alpha * (corr - c_max)
    logits -= np.max(logits)
    weights = np.exp(logits)
    weights /= weights.sum()

    W_plus = weights[n - 1:]
    W_minus = weights[n - 1::-1]
    h_rev = h[::-1]
    W0 = weights[n - 1]

    c1 = np.convolve(W_plus, h, mode='full')[:n]
    c2 = np.convolve(W_minus, h, mode='full')[:n]

    def smooth_obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        logits = alpha * (corr - m)
        logits -= np.max(logits)
        s = np.sum(np.exp(logits))
        return m + np.log(s) / alpha

    def constraint_eq(h):
        return np.sum(h) - target_sum

    def constraint_jac(h):
        return np.ones(len(h))

    try:
        res = minimize(
            smooth_obj, h0, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
        )
        return res.x if res.x is not None else None
    except Exception:
        return None
