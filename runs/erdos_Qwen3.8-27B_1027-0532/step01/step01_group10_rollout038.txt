import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_box_sum(h, n):
    target = n * 0.5
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, n)
    return h

def circular_c(h, n):
    dx = 2.0 / n
    H = np.fft.fft(h)
    R = np.fft.ifft(np.abs(H) ** 2).real
    return dx * (n / 2.0 - R)

def grad_weighted_circ(h, w, n):
    dx = 2.0 / n
    w_sym = 0.5 * (w + np.concatenate([w[:1], w[:0:-1]]))
    H = np.fft.fft(h)
    W = np.fft.fft(w_sym)
    return -2.0 * dx * np.fft.ifft(W * H).real
