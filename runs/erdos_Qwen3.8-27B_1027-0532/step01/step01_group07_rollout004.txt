def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = (c - c_max) * alpha
    shifted = np.clip(shifted, -60, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / s
    idx = np.arange(n)
    rows, cols = np.meshgrid(idx, idx, indexing='ij')
    M1 = w[rows + n - 1 - cols]
    M2 = w[cols - rows + n - 1]
    Part1 = M1.dot(1.0 - h)
    Part2 = M2.dot(h)
    grad = dx * (Part1 - Part2)
    return c_max, grad

def _slsqp_obj(val):
    return val[-1]

def _slsqp_ineq(val, n, m_idx):
    h = val[:n]
    t = val[n]
    m = m_idx - (n - 1)
    dx = 2.0 / n
    c_val = 0.0
    for j in range(n):
        jm = j + m
        if 0 <= jm < n:
            c_val += h[jm] * (1.0 - h[j])
    return t - c_val * dx

def _slsqp_eq(val, n, _):
    return np.sum(val[:n]) - n / 2.0

def _run_slsqp_active(h_init, n, active_indices, maxiter=150):
    target = n / 2.0
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = []
    for k in active_indices:
        cons.append({'type': 'ineq', 'fun': _slsqp_ineq, 'args': (n, k)})
    cons.append({'type': 'eq', 'fun': _slsqp_eq, 'args': (n, 0)})
    res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    h_opt = np.clip(res.x[:n], 0.0, 1.0)
    h_opt = project_feasible(h_opt, target)
    return h_opt
