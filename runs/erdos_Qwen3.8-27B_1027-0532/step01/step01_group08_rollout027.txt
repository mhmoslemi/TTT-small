from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx, np.arange(len(corr)) - (n - 1)

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, lag, n):
    """Exact gradient of c(lag) w.r.t. h, where lag is the actual shift."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:

        g[k:] += (1.0 - h[:n-k]) * dx
        g[:n-k] += (-h[k:n]) * dx

One more idea: **use the "Frank-Wolfe" (conditional gradient) algorithm**. At each step:
1. Compute the current C₅
2. Find the worst shift k*
3. Solve a linear program: minimize ∇g_{k*}(h) · h subject to 0≤h≤1, ∑h=n/2
4. Take a step towards the LP solution

The LP in step 3 can be solved analytically:
- The objective is linear: ∑_j a_j * h_j
- Subject to: 0 ≤ h_j ≤ 1, ∑h_j = n/2
- The optimal solution is: set h_j = 1 for the n/2 indices with the smallest a_j, and h_j = 0 for the rest.

Wait, that's only if a_j are all distinct. In general, it's a "water-filling" solution.

Actually, the Frank-Wolfe step is:
z = argmin_{h ∈ P} ∇f(h) · h where P = {h : 0≤h≤1, ∑h=n/2}

The solution is: sort the components of ∇f(h) in ascending order, set the first n/2 to 1 and the rest to 0.

This is O(n log n) per step! Very fast.

Then: h_new = (1-γ)*h + γ*z where γ is the step size.

This is the "conditional gradient" or "Frank-Wolfe" method. It's known to converge in O(1/T) iterations for smooth objectives, and O(1/√T) for non-smooth ones.

Let me combine Frank-Wolfe with the top-k idea:
- At each step, compute the average gradient over the top-k shifts
- Use Frank-Wolfe step with that gradient

def frank_wolfe_step(h, g, target):
    """Frank-Wolfe (conditional gradient) step.
    z = argmin_{h in P} g · h where P = {h: 0<=h<=1, sum(h)=target}
    """
    n = len(h)
    # The minimizer of sum(g[j]*h[j]) subject to 0<=h[j]<=1, sum(h[j])=target
    # is: h[j] = 1 for the 'target' indices with smallest g[j], 0 otherwise
    # (assuming target is integer, which it is: n/2)
    k = int(target)
    idx = np.argsort(g)[:k]
    z = np.zeros(n)
    z[idx] = 1.0
    return z

def fw_optimize(h0, n, target, deadline, k_top=10, max_iter=10000):

        max_val = np.max(shifts)
        top_idx = np.argsort(shifts)[-k_top:]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            g += grad_shift(h, lag, n)
        g /= k_top
        
        z = frank_wolfe_step(h, g, target)
        # Step size: standard FW uses gamma = 2/(it+2)
        gamma = 2.0 / (it + 2.0)
        h_new = (1 - gamma) * h + gamma * z
        h_new = np.clip(h_new, 0, 1)

        h = h_new
    
    return best_h, best_c5

                h_opt2, c5_opt2 = pgd_topk(best_h, n_cur, target_cur, min(deadline - 30, time.time() + 40), k_top=8)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()

    # Phase 6: Final PGD polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n_final = len(best_h)
        h_opt, c5_opt = pgd_topk(best_h, n_final, n_final / 2.0, deadline, k_top=3, max_iter=30000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Final FW polish
    if best_h is not None and time.time() < deadline - 3:
        n_final = len(best_h)
        h_opt, c5_opt = fw_optimize(best_h, n_final, n_final / 2.0, deadline, k_top=15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
