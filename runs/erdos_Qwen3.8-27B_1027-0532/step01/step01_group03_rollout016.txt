def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add

        for i in order:

            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

            # corr[j] = sum_m h[m+k]*(1-h[m]) where k = j-(n-1)
            # h[i] contributes:
            #   as h[m+k] when m = i-k: term = h[i]*(1-h[i-k]) if valid
            #   as (1-h[m]) when m = i: term = h[i+k]*(1-h[i]) if valid
            # So slope for corr[j] w.r.t. h[i]:
            #   (1-h[i-k]) if 0<=i-k<n, plus -h[i+k] if 0<=i+k<n
            
            jj = np.arange(2*n-1)
            kk = jj - (n-1)  # lag values
            
            s1 = (kk == -kk)  # placeholder
            # i-k valid: 0 <= i-k < n → k <= i and k > i-n → k in (i-n, i]
            valid1 = (i - kk >= 0) & (i - kk < n)  # 0 <= i-k < n
            # i+k valid: 0 <= i+k < n → k >= -i and k < n-i → k in [-i, n-i)
            valid2 = (i + kk >= 0) & (i + kk < n)  # 0 <= i+k < n
            
            slopes = np.zeros(2*n-1)
            # For valid1: slope += (1 - h[i-k])
            idx1 = i - kk[valid1]  # these are the indices of h that pair with h[i]
            slopes[valid1] += (1.0 - h[idx1])
            # For valid2: slope += -h[i+k]
            idx2 = i + kk[valid2]
            slopes[valid2] += (-h[idx2])
            
            # Current correlation with h
            curr_corr = np.correlate(h, 1.0-h, mode='full')
            base = curr_corr - h[i] * slopes
            
            adj_slopes = slopes + lam
            
            # Minimize f(v) = max_j (base[j] + adj_slopes[j]*v) over v in [0,1]
            # Find breakpoints
            candidates = [0.0, 1.0]
            # For efficiency, only consider the top few correlations
            # Actually, let's just do all pairs for n=86 (171 choose 2 ≈ 14.5K pairs)
            for a in range(2*n-1):
                for b in range(a+1, 2*n-1):
                    ds = adj_slopes[a] - adj_slopes[b]
                    if abs(ds) > 1e-14:
                        v = (base[b] - base[a]) / ds
                        if -1e-12 < v < 1.0 + 1e-12:
                            candidates.append(np.clip(v, 0.0, 1.0))
            
            best_v = 0.0
            best_f = np.inf
            for v in candidates:
                f = np.max(base + adj_slopes * v)
                if f < best_f:
                    best_f = f
                    best_v = v
            
            delta = abs(best_v - h[i])
            if delta > max_delta:
                max_delta = delta
            h[i] = best_v
        
        if max_delta < tol:
            break
    
    return h

    """Minimize f(v) = max(base + slopes*v) over v in [lo, hi] using ternary search."""

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(base + slopes * m1)
        f2 = np.max(base + slopes * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    v = (lo + hi) / 2.0
    v = np.clip(v, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _ternary_min(base, slopes, lo=0.0, hi=1.0, iters=50):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(base + slopes * m1)
        f2 = np.max(base + slopes * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return np.clip((lo + hi) / 2.0, 0.0, 1.0)

def _coord_descent(h_init, n, lam, max_sweeps=100, tol=1e-12):
    """Coordinate descent minimizing max_j corr[j] + lam*(sum(h)-target)."""
    target = n / 2.0
    h = h_init.copy()
    m = 2 * n - 1  # length of full correlation
    
    for sweep in range(max_sweeps):
        max_delta = 0.0
        for i in range(n):
            # Compute slopes for each correlation index
            kk = np.arange(m) - (n - 1)  # lag values
            # valid1: 0 <= i - k < n  (h[i] appears as h[m+k] when m = i-k)
            valid1 = (i - kk >= 0) & (i - kk < n)
            # valid2: 0 <= i + k < n  (h[i] appears in (1-h[m]) when m = i, so h[i+k] in h[m+k])
            valid2 = (i + kk >= 0) & (i + kk < n)
            
            slopes = np.zeros(m)
            idx1 = i - kk[valid1]
            slopes[valid1] += (1.0 - h[idx1])
            idx2 = i + kk[valid2]
            slopes[valid2] += (-h[idx2])
            
            # Current correlation
            curr = np.correlate(h, 1.0 - h, mode='full')
            base = curr - h[i] * slopes
            
            adj_slopes = slopes + lam
            
            # Ternary search for optimal v
            v = _ternary_min(base, adj_slopes)
            
            delta = abs(v - h[i])
            if delta > max_delta:
                max_delta = delta
            h[i] = v
        
        if max_delta < tol:
            break
    
    return h

def _refine_slsqp(h, n, target_sum, alphas, maxiter_per_alpha, ftol):
    """Refine using SLSQP with log-sum-exp smoothing."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    best = h.copy()
    best_obj = _c5_of(h)
    
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, best, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = _fix_sum(res.x, target_sum)
            c5 = _c5_of(h_new)
            if c5 < best_obj - 1e-12:
                best = h_new.copy()
                best_obj = c5
        except Exception:
            pass
    
    return best, best_obj

def _upsample(h, n_from, n_to):
    """Upsample h from n_from to n_to using linear interpolation."""
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_to, x_from, h)

            v = _ternary_min(base, adj_slopes)
            delta = abs(v - h[i])
            if delta > max_delta:
                max_delta = delta
            h[i] = v
        
        if max_delta < tol:
            break
    
    return h
