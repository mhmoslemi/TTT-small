import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.01)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.01)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    g = np.correlate(h, one_minus_h, mode='full') * dx
    
    max_g = np.max(g)
    scaled = beta * (g - max_g)
    exp_g = np.exp(scaled)
    Z = np.sum(exp_g)
    F = max_g + np.log(Z) / beta
    
    p = exp_g / Z
    
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n)[None, :]
    P = p[(j_idx - k_idx) + (n-1)]
    
    grad = (dx / beta) * (P @ one_minus_h - P.T @ h)
    
    return F, grad

def penalized_obj_and_grad(x, beta, mu):
    n = len(x)
    F, grad = smooth_max_obj_and_grad(x, beta)
    cv = np.sum(x) - n / 2.0
    F += mu * cv * cv
    grad = grad + 2.0 * mu * cv
    return F, grad

def lbfgsb_optimize(h_init, beta, mu=1e6, maxiter=2000):
    n = len(h_init)
    def obj(x):
        F, _ = penalized_obj_and_grad(x, beta, mu)
        return F
    def grad(x):
        _, g = penalized_obj_and_grad(x, beta, mu)
        return g
    try:
        result = minimize(obj, h_init, method='L-BFGS-B', jac=grad,
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                                  'maxls': 50, 'maxfun': maxiter * 50})
        if result.x is not None and np.all(result.x >= -1e-9) and np.all(result.x <= 1 + 1e-9):
            h_opt = repair_h(np.clip(result.x, 0, 1), n)
            return h_opt, evaluate_c5(h_opt)
    except Exception:
        pass
    return h_init, evaluate_c5(h_init)

def slsqp_smooth_max(h_init, beta, maxiter=500):
    n = len(h_init)
    def obj(x):
        F, _ = smooth_max_obj_and_grad(x, beta)
        return F
    def grad(x):
        _, g = smooth_max_obj_and_grad(x, beta)
        return g
    try:
        result = minimize(obj, h_init, method='SLSQP', jac=grad,
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                                       'jac': lambda x: np.ones(n)}],
                         options={'maxiter': maxiter, 'ftol': 1e-12})

            h_opt = repair_h(np.clip(result.x, 0, 1), n)
            return h_opt, evaluate_c5(h_opt)
    except Exception:
        pass
    return h_init, evaluate_c5(h_init)

    t0 = time.time()

    # Start from initial

    betas = [5, 10, 20, 50, 100, 200, 500]

        if c5_curr < c5_best:

    # Phase 2: Try different n values

        # Interpolate best solution to new n

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break

            h[mask] = np.minimum(1.0, h[mask] + 0.0)

def smooth_max_components(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    g = np.correlate(h, one_minus_h, mode='full') * dx
    max_g = np.max(g)
    scaled = beta * (g - max_g)
    max_scaled = np.max(scaled)
    exp_g = np.exp(scaled - max_scaled)
    Z = np.sum(exp_g)
    F = max_g + (max_scaled + np.log(Z)) / beta
    p = exp_g / Z
    return F, p, g

def build_P(p, n):
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n)[None, :]
    P = p[(j_idx - k_idx) + (n - 1)]
    return P

def _sm_objective(x, params):
    beta, mu = params
    n = len(x)
    F, p, g = smooth_max_components(x, beta)
    cv = np.sum(x) - n / 2.0
    return F + 0.5 * mu * cv * cv

def _sm_gradient(x, params):
    beta, mu = params
    n = len(x)
    dx = 2.0 / n
    F, p, g = smooth_max_components(x, beta)
    P = build_P(p, n)
    one_minus_h = 1.0 - x
    grad = (dx / beta) * (P @ one_minus_h - P.T @ x)
    cv = np.sum(x) - n / 2.0
    grad = grad + mu * cv
    return grad

def _sm_objective_only(x, beta):
    F, p, g = smooth_max_components(x, beta)

def _sm_gradient_only(x, beta):

    P = build_P(p, n)

    grad = (dx / beta) * (P @ one_minus_h - P.T @ x)

def subgradient_polish(h, n, num_steps=200, step_size=0.01):

        grad = np.zeros(n)
        for lag_idx in active:
            lag = lag_idx - (n - 1)
            for j in range(n):
                j_minus_lag = j - lag
                j_plus_lag = j + lag
                val = 0.0
                if 0 <= j_minus_lag < n:
                    val += (1.0 - h_curr[j_minus_lag])
                if 0 <= j_plus_lag < n:
                    val -= h_curr[j_plus_lag]
                grad[j] += val
        grad *= dx / len(active)
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        h_new = h_curr - step_size * grad
        h_new = repair_h(h_new, n)
        c5_new = evaluate_c5(h_new)

            best_c5 = c5_new
            h_curr = h_new
        else:
            step_size *= 0.8
            if step_size < 1e-6:

    for n_try in [300, 350, 400, 450, 500]:
        if time_left() < 80:
            break
        starts = generate_structured_starts(n_try, rng, h_best)
        for h_start in starts[:4]:
            if time_left() < 30:
                break
            h_curr, c5_curr = smooth_max_lbfgs(h_start, 50, mu=1e5, maxiter=1000)
            if c5_curr < c5_best:
                h_best, c5_best = h_curr, c5_curr
                if time_left() > 30:
                    h_curr2, c5_curr2 = smooth_max_lbfgs(h_best, 200, mu=1e6, maxiter=1500)
                    if c5_curr2 < c5_best:
                        h_best, c5_best = h_curr2, c5_curr2

    # Phase 7: Random restarts
    n = len(h_best)
    for trial in range(40):
        if time_left() < 25:
            break
        strat = trial % 6
        if strat == 0:
            h_pert = h_best + rng.normal(0, 0.03, n)
        elif strat == 1:
            h_pert = h_best + rng.normal(0, 0.08, n)
        elif strat == 2:
            h_pert = h_best.copy()
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif strat == 3:
            shift = rng.randint(1, max(2, n // 10))
            h_pert = np.roll(h_best, shift)
        elif strat == 4:
            h_pert = 0.5 + 0.35 * np.sin(2 * np.pi * rng.uniform(2, 5) *
                         np.linspace(0, 2, n) + rng.uniform(0, 6.28))
        else:
            h_pert = h_best.copy()
            idx1 = rng.choice(n, size=max(1, n // 8), replace=False)
            idx2 = rng.choice(n, size=len(idx1), replace=False)
            h_pert[idx1] = h_best[idx2]
            h_pert[idx2] = h_best[idx1]
        h_pert = repair_h(h_pert, n)
        h_curr, c5_curr = smooth_max_lbfgs(h_pert, 80, mu=1e5, maxiter=800)
        if c5_curr < c5_best:
            h_best, c5_best = h_curr, c5_curr
            if time_left() > 20:
                h_curr2, c5_curr2 = smooth_max_lbfgs(h_best, 300, mu=1e6, maxiter=1500)
                if c5_curr2 < c5_best:
                    h_best, c5_best = h_curr2, c5_curr2

    # Phase 8: Final high-precision polish
    if time_left() > 40:
        n = len(h_best)
        for beta in [500, 1000]:
            if time_left() < 20:
                break
            h_curr, c5_curr = smooth_max_lbfgs(h_best, beta, mu=1e7, maxiter=5000)
            if c5_curr < c5_best:
                h_best, c5_best = h_curr, c5_curr

    if time_left() > 30:
        h_curr, c5_curr = subgradient_polish(h_best, len(h_best), num_steps=300, step_size=0.005)
        if c5_curr < c5_best:
            h_best, c5_best = h_curr, c5_curr

    if time_left() > 20:
        h_curr, c5_curr = smooth_max_slsqp(h_best, 500, maxiter=500)
        if c5_curr < c5_best:
            h_best, c5_best = h_curr, c5_curr

    return h_best, c5_best, len(h_best)
