```python
def solve_1d_max_lines(c_vals, a_vals):
    """Solve min_x max_l (c_l + a_l * x) for x in [0,1]
    Returns the optimal x."""
    # The optimal x is at a crossing point of two lines or at a boundary
    # Try x = 0 and x = 1 first
    best_x = 0.0
    best_val = np.max(c_vals)
    val_at_1 = np.max(c_vals + a_vals)
    if val_at_1 < best_val:
        best_val = val_at_1
        best_x = 1.0
    
    # Try all crossing points
    n_lines = len(c_vals)
    for i in range(n_lines):
        for j in range(i+1, n_lines):
            if abs(a_vals[i] - a_vals[j]) < 1e-15:
                continue
            x_cross = (c_vals[j] - c_vals[i]) / (a_vals[i] - a_vals[j])
            if 0.0 <= x_cross <= 1.0:
                val = np.max(c_vals + a_vals * x_cross)
                if val < best_val:
                    best_val = val
                    best_x = x_cross
    return best_x
```

But wait, I need to compute a_vals and c_vals for each coordinate. Let me think about how to do this efficiently.

For coordinate i, the correlation at shift k (with shift = k - (n-1) = s) is:
f_k = sum_j h[j] * (1 - h[j+s])

The terms involving h[i]:
- j = i: h[i] * (1 - h[i+s]) if 0 <= i+s < n
- j = i-s: h[i-s] * (1 - h[i]) if 0 <= i-s < n

So f_k = [h[i] * (1 - h[i+s]) if valid] + [h[i-s] * (1 - h[i]) if valid] + rest_k

= h[i] * (1 - h[i+s] - h[i-s]) + h[i-s] + rest_k (when both are valid)

where rest_k doesn't involve h[i].

So a_k = (1 - h[i+s] - h[i-s]) for the shift s = k-(n-1), when both i+s and i-s are valid.
And c_k = h[i-s] + rest_k.

But computing rest_k for all k is expensive. However, I can compute it as:

So I can precompute f_k(h) for all k, then for each coordinate i, compute a_k and c_k using the formula.

```python
    valid = np.abs(denom) > 1e-14
    if not np.any(valid):
        # All parallel, just check endpoints
        v0 = np.max(c_vals)
        v1 = np.max(c_vals + a_vals)
        return 0.0 if v0 <= v1 else 1.0
    
    xc = np.full(len(i_idx), -1.0)
    xc[valid] = (c_vals[j_idx[valid]] - c_vals[i_idx[valid]]) / denom[valid]
    xc_valid = xc[(xc >= 0.0) & (xc <= 1.0)]
    
    # Evaluate at endpoints and valid crossing points
    candidates = np.concatenate([[0.0, 1.0], xc_valid])
    best_x = candidates[0]
    best_val = np.inf
    for x in candidates:
        val = np.max(c_vals + a_vals * x)
        if val < best_val:
            best_val = val
            best_x = x
    return best_x
```

So I need to find x* where some active line has slope ≤ 0 and another has slope ≥ 0 (or a single line has slope 0).

Algorithm:
1. Sort lines by slope: a_(1) ≤ a_(2) ≤ ... ≤ a_(n)
2. The upper envelope changes from lines with smaller slope to lines with larger slope as x increases
3. The minimum of g is where the slope of the upper envelope crosses 0

To find this:
- At x=0, the active line is argmax c_k
- As x increases, lines with larger slope overtake
- The minimum is where the upper envelope's slope changes from negative to positive

This can be found by:
- Computing the upper envelope of the lines (O(n log n) or O(n^2))
- Finding where the slope changes sign

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(300):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def solve_1d_max(c_vals, a_vals):
    n_lines = len(c_vals)
    best_x = 0.0
    best_val = float(np.max(c_vals))
    val_1 = float(np.max(c_vals + a_vals))
    if val_1 < best_val:
        best_val = val_1
        best_x = 1.0
    for i in range(n_lines):
        ai = a_vals[i]
        ci = c_vals[i]
        for j in range(i + 1, n_lines):
            aj = a_vals[j]
            diff = ai - aj
            if abs(diff) < 1e-14:
                continue
            xc = (c_vals[j] - ci) / diff
            if xc < -1e-12 or xc > 1.0 + 1e-12:
                continue
            xc = max(0.0, min(1.0, xc))
            val = float(np.max(c_vals + a_vals * xc))
            if val < best_val - 1e-14:
                best_val = val
                best_x = xc
    return best_x

def coordinate_descent_lagrangian(h, lam, max_sweeps, active_tol):
    n = len(h)
    n_shifts = 2 * n - 1
    s_vals = np.arange(n_shifts) - (n - 1)
    best_h = h.copy()
    best_c5 = np.inf
    for sweep in range(max_sweeps):
        corr = compute_correlations(h)
        C_max = np.max(corr)
        active_mask = corr >= C_max - active_tol
        active_indices = np.where(active_mask)[0]
        if len(active_indices) < 2:
            active_indices = np.argsort(corr)[-5:]
        changed = False
        for i in range(n):
            j1_vals = i + s_vals
            j2_vals = i - s_vals
            valid_j1 = (j1_vals >= 0) & (j1_vals < n)
            valid_j2 = (j2_vals >= 0) & (j2_vals < n)
            a_vals = np.zeros(len(active_indices))
            c_vals = np.zeros(len(active_indices))
            for idx, k in enumerate(active_indices):
                s = s_vals[k]
                a = lam
                c = corr[k]
                jj1 = i + s
                jj2 = i - s
                if 0 <= jj1 < n:
                    a += (1.0 - h[jj1])
                    c -= h[i] * (1.0 - h[jj1])
                if 0 <= jj2 < n:
                    a += (-h[jj2])
                    c -= h[jj2] * (1.0 - h[i])
                a_vals[idx] = a
                c_vals[idx] = c
            x_opt = solve_1d_max(c_vals, a_vals)
            if abs(x_opt - h[i]) > 1e-12:
                delta = x_opt - h[i]
                h[i] = x_opt
                for k in active_indices:
                    s = s_vals[k]
                    dc = 0.0
                    jj1 = i + s
                    jj2 = i - s
                    if 0 <= jj1 < n:
                        dc += delta * (1.0 - h[jj1])
                    if 0 <= jj2 < n:
                        dc += (-h[jj2]) * delta
                    corr[k] += dc
                if delta != 0.0:
                    changed = True
        h = project_feasible(h, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not changed:
            break
    return best_h, best_c5

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    s_vals = np.arange(n_corr) - (n - 1)
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-12:
            continue
        s = s_vals[k]
        j1 = np.arange(n) + s
        j2 = np.arange(n) - s
        mask1 = (j1 >= 0) & (j1 < n)
        mask2 = (j2 >= 0) & (j2 < n)
        g = np.zeros(n)
        g[mask1] += (1.0 - x[j1[mask1]])
        g[mask2] += (-x[j2[mask2]])
        grad += w * g
    grad *= dx
    return grad

def equality_constraint(x, n):
    return np.sum(x) - n / 2.0
```
