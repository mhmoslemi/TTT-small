For a specific lag k, corr[k] = ∑_i h[i+k] * (1-h[i]). To reduce this:
- Find i where h[i+k]=1 and h[i]=0 (these contribute to corr[k])
- Swap: change h[i] to 1 and h[i+k] to 0 (or swap h[i] with some other 0, and h[i+k] with some other 1)

This gives a more directed search.

Let me implement this:

```python
def directed_binary_search(h, n, time_limit, t_start, rng):
    best = h.copy()
    best_score = score_c5(best)
    corr = np.correlate(best, 1-best, mode='full')
    
    while time.time() - t_start < time_limit:
        # Find the lag with maximum correlation
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Find contributing terms at this lag
        # corr[max_k] = sum_i best[i + max_k - n + 1] * (1-best[i]) for valid i
        lag = max_k - n + 1

        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                if best[j] > 0.5 and best[i] < 0.5:
                    contributors.append((i, j))
        
        if not contributors:
            break
        
        # Try to fix: for each (i, j) in contributors, swap h[i] with some 0 and h[j] with some 1
        # Best case: if there's a 0 at position j' and a 1 at position i' such that the swap reduces multiple max correlations
        # Simplification: swap h[i] (0→1) with h[j] (1→0) directly
        improved = False
        for i, j in contributors:
            trial = best.copy()
            trial[i], trial[j] = 1.0, 0.0
            s = score_c5(trial)
            if s < best_score - 1e-12:
                best = trial
                best_score = s
                corr = np.correlate(best, 1-best, mode='full')
                improved = True
                break

            ones = np.where(best > 0.5)[0]
            zeros = np.where(best < 0.5)[0]

                break
```

Let me implement the delta formula properly.

Given:
- corr[m] for m = 0, ..., 2n-2 (full correlation of h and 1-h)
- We swap: h[i]: 1→0, h[j]: 0→1 (so i is a 1-position, j is a 0-position)

The new correlation corr_new[m] = corr[m] + Δ[m] where:

Δ[m] = contribution from h[i] changing + contribution from h[j] changing

For h[i]: 1→0:
- i is a position where h was 1, now 0
- In the correlation corr[m] = ∑_q h[q + m - n + 1] * (1-h[q]):
  - As first factor (h[p] where p = i): m - n + 1 + q = i → q = i - m + n - 1
    Valid when 0 ≤ q < n, i.e., n-1-i ≤ m ≤ n-1+i... wait let me redo.
    p = q + m - n + 1 = i → q = i - m + n - 1
    Need 0 ≤ q ≤ n-1: 0 ≤ i - m + n - 1 ≤ n-1 → m ≤ i + n - 1 and m ≥ i
    So m ranges from i to i+n-1 (but also 0 ≤ m ≤ 2n-2)
    Change: -(1-h[q]) where q = i - m + n - 1
    = -(1-h[i-m+n-1])
    
  - As second factor (1-h[q] where q = i): q = i, so p = i + m - n + 1
    Valid when 0 ≤ p < n: 0 ≤ i + m - n + 1 < n → n-1-i ≤ m ≤ n-1-i... 
    Wait: 0 ≤ i + m - n + 1 ≤ n-1 → 1-n+i ≤ m ≤ n-1-i
    So m ranges from max(0, 1-n+i) to min(2n-2, n-1-i)
    Change: +h[p] where p = i + m - n + 1
    = +h[i+m-n+1]

For h[j]: 0→1:
- As first factor: +(1-h[q]) where q = j - m + n - 1
  Valid: m ranges from j to j+n-1 (clamped to [0, 2n-2])
- As second factor: -h[p] where p = j + m - n + 1
  Valid: m ranges from max(0, 1-n+j) to min(2n-2, n-1-j)

So:
Δ[m] = -(1-h[i-m+n-1]) for m ∈ [i, min(i+n-1, 2n-2)]  (first factor, h[i] 1→0)
     + h[i+m-n+1] for m ∈ [max(0, 1-n+i), min(n-1-i, 2n-2)]  (second factor, h[i] 1→0)
     + (1-h[j-m+n-1]) for m ∈ [j, min(j+n-1, 2n-2)]  (first factor, h[j] 0→1)
     - h[j+m-n+1] for m ∈ [max(0, 1-n+j), min(n-1-j, 2n-2)]  (second factor, h[j] 0→1)

```python
def swap_delta_vec(h, i, j, n):
    """Compute delta corr vector when h[i]:1->0, h[j]:0->1"""
    delta = np.zeros(2*n-1)
    
    # h[i]: 1 -> 0
    # As first factor: for m where i = q + m - n + 1, q valid
    # q = i - m + n - 1, valid when 0 <= q < n
    # m = i + n - 1 - q, q in [0, n-1]
    # m in [i, i+n-1], clamped to [0, 2n-2]
    m_lo = max(0, i)
    m_hi = min(2*n-2, i+n-1)
    for m in range(m_lo, m_hi+1):
        q = i - m + n - 1
        delta[m] -= (1 - h[q])
    
    # As second factor: for m where q = i, p = i + m - n + 1, p valid
    # p in [0, n-1]: 0 <= i+m-n+1 < n → n-1-i <= m < n+1-i
    m_lo = max(0, n-1-i)
    m_hi = min(2*n-2, n-i)  # m < n+1-i means m <= n-i
    for m in range(m_lo, m_hi+1):
        p = i + m - n + 1
        delta[m] += h[p]
    
    # h[j]: 0 -> 1
    # As first factor: +(1-h[q]) where q = j - m + n - 1
    m_lo = max(0, j)
    m_hi = min(2*n-2, j+n-1)
    for m in range(m_lo, m_hi+1):
        q = j - m + n - 1
        delta[m] += (1 - h[q])
    
    # As second factor: -h[p] where p = j + m - n + 1
    m_lo = max(0, n-1-j)
    m_hi = min(2*n-2, n-j)
    for m in range(m_lo, m_hi+1):
        p = j + m - n + 1
        delta[m] -= h[p]
    
    return delta
```

```python
def eval_swap_fast(h, corr, i, j, n):
    """Evaluate new max corr after swapping h[i]:1->0, h[j]:0->1. O(n) per call."""
    # Compute delta
    delta = np.zeros(2*n-1)
    
    # h[i] 1->0, as first factor: delta[m] -= (1-h[q]) for m=i..i+n-1, q=i-m+n-1
    m_start = max(0, i)
    m_end = min(2*n-2, i+n-1)
    q_indices = i - np.arange(m_start, m_end+1) + n - 1
    delta[m_start:m_end+1] -= (1 - h[q_indices])
    
    # h[i] 1->0, as second factor: delta[m] += h[p] for m=n-1-i..n-i, p=i+m-n+1
    m_start = max(0, n-1-i)
    m_end = min(2*n-2, n-i)
    p_indices = i + np.arange(m_start, m_end+1) - n + 1
    delta[m_start:m_end+1] += h[p_indices]
    
    # h[j] 0->1, as first factor: delta[m] += (1-h[q]) for m=j..j+n-1, q=j-m+n-1
    m_start = max(0, j)
    m_end = min(2*n-2, j+n-1)
    q_indices = j - np.arange(m_start, m_end+1) + n - 1
    delta[m_start:m_end+1] += (1 - h[q_indices])
    
    # h[j] 0->1, as second factor: delta[m] -= h[p] for m=n-1-j..n-j, p=j+m-n+1
    m_start = max(0, n-1-j)
    m_end = min(2*n-2, n-j)
    p_indices = j + np.arange(m_start, m_end+1) - n + 1
    delta[m_start:m_end+1] -= h[p_indices]
    
    new_corr = corr + delta
    return np.max(new_corr), new_corr
```

```python
def sa_binary(h, n, time_limit, t_start, rng):
    """Simulated annealing for binary swap problem"""
    cur = h.copy()
    dx = 2.0 / n
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = np.max(cur_corr)
    
    best = cur.copy()
    best_corr = cur_corr.copy()
    best_max = cur_max
    best_score = best_max * dx
    
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur < 0.5)[0])
    
    T = 0.05
    T_min = 0.001
    cooling = 0.999
    
    step = 0
    while time.time() - t_start < time_limit:
        step += 1
        if step % 1000 == 0:
            T = max(T_min, T * cooling)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        d = swap_delta(cur, i, j, n)
        new_corr = cur_corr + d
        new_max = np.max(new_corr)
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            # Accept
            cur[i], cur[j] = 0.0, 1.0
            cur_corr = new_corr
            cur_max = new_max
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
        
        if new_max < best_max - 1e-12:
            best = cur.copy()
            best_corr = new_corr.copy()
            best_max = new_max
            best_score = best_max * dx
    
    return best, best_score
```

For the continuous case, the gradient of the max correlation is the correlation gradient of the active lag(s). If there's a unique active lag k*, then:
∇corr[k*] = ∂/∂h[corr[k*]]

And the projected gradient step would be:
h_new = project(h - α * ∇corr[k*])

This is straightforward to implement.

Let me also think about whether the optimal solution is 0-1 or continuous. 

For the 0-1 case with n even, the problem is well-defined. The question is: can a continuous solution do better?

I suspect that for this problem, the optimal (or near-optimal) solution is either 0-1 or very close to 0-1. The reason is that the objective is the max of bilinear terms, and the feasible set is a polytope (box ∩ hyperplane). def score_c5(h):
```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def swap_delta(h, i, j, n):
    """O(n) computation of correlation change when h[i]:1->0, h[j]:0->1"""
    delta = np.zeros(2*n-1)
    # h[i] 1->0, first factor
    ms, me = max(0, i), min(2*n-2, i+n-1)
    if ms <= me:
        delta[ms:me+1] -= (1.0 - h[i - np.arange(ms, me+1) + n - 1])
    # h[i] 1->0, second factor
    ms, me = max(0, n-1-i), min(2*n-2, n-i)
    if ms <= me:
        delta[ms:me+1] += h[i + np.arange(ms, me+1) - n + 1]
    # h[j] 0->1, first factor
    ms, me = max(0, j), min(2*n-2, j+n-1)
    if ms <= me:
        delta[ms:me+1] += (1.0 - h[j - np.arange(ms, me+1) + n - 1])
    # h[j] 0->1, second factor
    ms, me = max(0, n-1-j), min(2*n-2, n-j)
    if ms <= me:
        delta[ms:me+1] -= h[j + np.arange(ms, me+1) - n + 1]
    return delta

    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur < 0.5)[0])
    
    T = 0.02
    stall = 0

    while time.time() - t_start < time_limit and stall < max_stall:
        # Greedy phase: try all pairs, find best improvement
        best_new_max = cur_max
        best_i, best_j = -1, -1
        best_d = None
        
        for i in ones:
            for j in zeros:

                nm = np.max(cur_corr + d)

                    best_d = d

            cur_corr = cur_corr + best_d
            cur_max = np.max(cur_corr)
            ones[ones.index(best_i)] = best_j
            zeros[zeros.index(best_j)] = best_i
            if cur_max < best_max - 1e-14:
                best = cur.copy()
                best_max = cur_max
                best_score = best_max * dx
            stall = 0
        else:

def continuous_refine(h, n, time_limit, t_start, rng):
    """Projected gradient descent + SLSQP for continuous refinement"""
    x = h.astype(float).copy()
    dx = 2.0 / n
    best = x.copy()
    best_score = score_c5(best)
    
    alpha = 0.1
    for it in range(10000):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_idx = np.argmax(corr)
        # Gradient of corr[max_idx] w.r.t. x
        lag = max_idx - n + 1
        grad = np.zeros(n)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                # corr[max_idx] includes term x[j] * (1-x[i])
                # d/dx[j] = (1-x[i])
                # d/dx[i] = -x[j]
                grad[j] += (1.0 - x[i])
                grad[i] -= x[j]
        
        # Projected gradient step
        x_new = x - alpha * grad
        x_new = project_feasible(x_new)
        s_new = score_c5(x_new)
        if s_new < best_score - 1e-14:
            best = x_new.copy()
            best_score = s_new
            x = x_new
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    
    # SLSQP polish
    if time.time() - t_start < time_limit:
        def lse_obj(xx):
            T = 0.01
            c = np.correlate(xx, 1.0 - xx, mode='full')
            cm = np.max(c)
            scaled = (c - cm) / T
            return dx * (cm + T * np.log(np.sum(np.exp(scaled))))
        
        def sum_con(xx):
            return np.sum(xx) - n / 2.0
        
        def sum_con_jac(xx):
            return np.ones(n)
        
        try:
            res = minimize(lse_obj, x, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                           options={'maxiter': 5000, 'ftol': 1e-14})
            if res.x is not None:

                s_new = score_c5(x_new)
                if s_new < best_score:
                    best = x_new
                    best_score = s_new
        except:
            pass
```
