    # Phase 2: SLSQP refinement with various alphas
    alphas = [50, 100, 200, 400, 800]
    for alpha in alphas:
        if time_left() < 100:
            break
        h_new = slsqp_optimize(h, ts, bounds, cons, lse_objective, (alpha,), 3000, 1e-14)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
            h = h_new

    # Phase 3: Top-k SLSQP refinement
    for k in [3, 5, 10, 20]:
        if time_left() < 80:
            break
        h_new = slsqp_optimize(h, ts, bounds, cons, topk_objective, (k,), 2000, 1e-13)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()
            h = h_new

    # Phase 4: Multi-grid exploration
    n_candidates = []
    for m in range(150, 550, 20):
        if m != n_init:
            n_candidates.append(m)
    # Add some specific values
    for m in [200, 300, 400, 500]:
        if m not in n_candidates and m != n_init:
            n_candidates.append(m)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if time_left() < 120:
            break
        ts_t = n_try / 2.0
        if n_try == len(best_h):
            h_t = best_h.copy()
        else:
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, len(best_h))
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)

        bounds_t = [(0.0, 1.0)] * n_try
        cons_t = make_slqp_constraints(n_try, ts_t)

        # Subgradient on new grid
        v_t = np.zeros(n_try)
        eta_t = 0.008
        h_t_prev = h_t.copy()
        g_t_prev = None
        for it2 in range(3000):
            if time_left() < 80:
                break
            alpha2 = 100.0 + 300.0 * (it2 / 3000.0)
            dx_t = 2.0 / n_try
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full') * dx_t
            weights_t = get_weights_from_corr(corr_t, alpha2)
            g_t = weighted_subgradient(h_t, weights_t)
            if g_t_prev is not None:
                s_vec = h_t - h_t_prev
                y_vec = g_t - g_t_prev
                sy = np.dot(s_vec, y_vec)
                if sy > 1e-16:
                    yy = np.dot(y_vec, y_vec)
                    eta_t = 0.5 * eta_t + 0.5 * (yy / sy)
                    eta_t = max(1e-5, min(eta_t, 0.05))
            h_t_prev = h_t.copy()
            g_t_prev = g_t.copy()
            v_t = 0.85 * v_t + 0.15 * g_t
            h_t = project_feasible(h_t - eta_t * v_t, ts_t)

        s_t = compute_c5(h_t)
        if s_t < best_score:
            best_score = s_t
            best_h = h_t.copy()

        # SLSQP polish
        if time_left() > 60:
            h_t_new = slsqp_optimize(h_t, ts_t, bounds_t, cons_t, lse_objective, (200,), 2000, 1e-13)
            s_t = compute_c5(h_t_new)
            if s_t < best_score:
                best_score = s_t
                best_h = h_t_new.copy()
                h = project_feasible(np.interp(np.linspace(0, 1, len(best_h)),
                                               np.linspace(0, 1, len(h_t_new)), h_t_new),
                                     len(best_h) / 2.0)

    # Phase 5: Random restarts with SLSQP
    n_use = len(best_h)
    ts_use = n_use / 2.0
    bounds_use = [(0.0, 1.0)] * n_use
    cons_use = make_slqp_constraints(n_use, ts_use)

    while time_left() > 40:
        noise = rng.choice([0.02, 0.04, 0.06, 0.10, 0.15])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, ts_use)
        alpha_r = float(rng.randint(50, 500))
        h_new = slsqp_optimize(h_pert, ts_use, bounds_use, cons_use, lse_objective,
                               (alpha_r,), 800, 1e-13)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Final: try power mean objectives
    for p in [20, 50, 100, 200]:
        if time_left() < 20:
            break
        def power_obj(hh, pp):
            n_h = len(hh)
            dx_h = 2.0 / n_h
            corr_h = np.correlate(hh, 1.0 - hh, mode='full') * dx_h
            return float(np.sum(corr_h ** pp) ** (1.0 / pp))
        h_new = slsqp_optimize(best_h, ts_use, bounds_use, cons_use, power_obj, (float(p),), 500, 1e-13)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    h_final = project_feasible(best_h, ts_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
