def lse_gradient(h, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*f_k)))"""
    n = len(h)
    dx = 2.0 / n
    fk = compute_fk(h)
    # Softmax weights
    exp_fk = np.exp(beta * (fk - fk.max()))
    p = exp_fk / exp_fk.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: sum_k p_k * (1-h[j-k]) for 0<=j-k<n
        i_lo = max(0, j - (n-1))
        i_hi = min(n - 1, j)
        if i_lo <= i_hi:
            k_vals = j - np.arange(i_lo, i_hi + 1)
            grad[j] += np.dot(1.0 - h[i_lo:i_hi+1], p[k_vals + (n-1)])
        # Term 2: -sum_k p_k * h[j+k] for 0<=j+k<n

    # Term 2: -h[j+s] for valid j+s
    # j+s in [0, n-1] means j in [-s, n-1-s], intersected with [0, n-1]
    j_lo2 = max(0, -s)
    j_hi2 = min(n-1, n-1-s)
    if j_lo2 <= j_hi2:
        grad[j_lo2:j_hi2+1] -= ps * h[j_lo2+s:j_hi2+s+1]

grad *= dx

This loops over s (2n-1 iterations) and each iteration does O(n) vectorized numpy operations. Total: O(n²) but with good constants due to vectorization.

For n=200: 399 iterations, each doing 2 slice operations of length up to 200. Should be very fast (~1ms).

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, 'full') * dx).max())

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    fk = np.correlate(h, 1.0 - h, 'full') * dx
    exp_fk = np.exp(beta * (fk - fk.max()))
    p = exp_fk / exp_fk.sum()
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ps = p[s + (n - 1)]
        if ps < 1e-15:
            continue
        j_lo = max(0, s)
        j_hi = min(n - 1, n - 1 + s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += ps * (1.0 - h[j_lo-s:j_hi-s+1])
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= ps * h[j_lo2+s:j_hi2+s+1]
    grad *= dx
    return grad, fk

def lse_opt(h0, tm=60, beta_start=5.0, beta_end=100.0, n_stages=20, step_frac=0.05):
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    
    t0 = time.time()
    betas = np.geomspace(beta_start, beta_end, n_stages)
    
    for stage in range(n_stages):
        if time.time() - t0 > tm:
            break
        beta = betas[stage]
        
        while time.time() - t0 < tm:
            grad, fk = lse_gradient(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = step_frac / gnorm
            h = project_feasible(h - step * grad, target)
        
        h = project_feasible(h, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best
