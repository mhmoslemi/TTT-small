from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}."""
    h = h.copy()
    if abs(np.sum(h) - target_sum) < 1e-12:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def objective(h, alpha, dx):
    corr = compute_correlations(h)
    return smooth_max(corr, alpha) * dx

def objective_grad(h, alpha, dx):
    """Gradient of the smooth max objective."""
    corr = compute_correlations(h)
    # softmax weights
    a = alpha * (corr - np.max(corr))
    weights = np.exp(a)
    weights /= np.sum(weights)
    # gradient of corr[l] w.r.t. h[j]
    # corr[l] corresponds to lag = l - (n-1)
    n = len(h)
    grad = np.zeros(n)
    for l in range(len(corr)):
        if weights[l] < 1e-10:
            continue
        lag = l - (n - 1)
        if lag >= 0:
            # corr[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j < n-lag, minus h[j-lag] for j >= lag
            c = np.zeros(n)
            c[:n-lag] = 1.0 - h[lag:]
            c[lag:] -= h[:n-lag]
            grad += weights[l] * c
        else:
            l_abs = -lag
            # corr[-l_abs] = sum_{i=l_abs}^{n-1} h[i]*(1-h[i-l_abs])
            c = np.zeros(n)
            c[l_abs:] = 1.0 - h[:n-l_abs]
            c[:n-l_abs] -= h[l_abs:]
            grad += weights[l] * c
    return grad * dx

def slsqp_optimize(h0, alpha, maxiter, bounds, dx):
    def obj(h):
        return objective(h, alpha, dx)
    def obj_grad(h):
        return objective_grad(h, alpha, dx)
    def cons(h):
        return np.sum(h) - len(h) / 2.0
    def cons_jac(h):
        return np.ones(len(h))
    res = minimize(
        obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    return res.x if res.x is not None else None

def binary_local_search(h, n_rounds, rng, corr_func=None):
    """Swap-based local search on binary solution."""
    n = len(h)
    dx = 2.0 / n
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:n_ones - target]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    
    for round_idx in range(n_rounds):
        corr = compute_correlations(h)
        current_max = np.max(corr) * dx
        
        if current_max < best_score:
            best_score = current_max
            best_h = h.copy()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_gain = 0
        best_j = None
        best_k = None
        
        # Try random subset of swaps for efficiency
        n_trials = min(20000, len(ones) * len(zeros))
        for _ in range(n_trials):
            j = ones[rng.randint(len(ones))]
            k = zeros[rng.randint(len(zeros))]
            h_test = h.copy()
            h_test[j] = 0
            h_test[k] = 1
            new_max = np.max(compute_correlations(h_test)) * dx
            gain = current_max - new_max
            if gain > best_gain:
                best_gain = gain
                best_j = j
                best_k = k
        
        if best_j is not None:
            h[best_j] = 0
            h[best_k] = 1
        else:
            break  # Local minimum
    
    return best_h, best_score, h

    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(compute_correlations(h)) * dx

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum} using binary search."""
    h = np.asarray(h, dtype=float)
    if abs(np.sum(h) - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(200):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(x, alpha):
    m = np.max(x)
    if alpha < 1e6:
        return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha
    return m

def compute_active_gradient(h, top_k=5):
    """Compute gradient of the average of the top-k correlations."""
    n = len(h)
    corr = compute_correlations(h)
    # Get top k indices
    k = min(top_k, len(corr))
    active_idx = np.argsort(corr)[-k:]
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for m in active_idx:
        lag = m - (n - 1)

            # corr(lag) = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])

            c[lag:] -= h[:n-lag]

            c = np.zeros(n)
