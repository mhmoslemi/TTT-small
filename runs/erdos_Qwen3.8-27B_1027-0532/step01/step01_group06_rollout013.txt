    if np.any(I2):

    """Solve min_{t in [0,1]} max_k(A[k] + B[k]*t) * dx"""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = A + B * mid
        max_val = vals.max()
        active = np.where(vals >= max_val - 1e-14)[0]
        sg_min = dx * B[active].min()
        sg_max = dx * B[active].max()
        if sg_min <= 0.0 <= sg_max:
            return mid
        elif sg_min > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

    """One pass of coordinate descent. Returns updated h and corr."""
    for i in order:
        D = compute_slope(h, i, n)
        h_old = h[i]
        A = corr - h_old * D
        t_opt = solve_1d_max(A, D, dx)
        delta = t_opt - h_old
        if abs(delta) > 1e-14:
            corr += delta * D
            h[i] = t_opt
    return h, corr
