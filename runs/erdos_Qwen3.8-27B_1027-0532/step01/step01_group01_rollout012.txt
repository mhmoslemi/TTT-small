```python
def pairwise_update(h, i, j, dx):
    n = len(h)
    # Compute d_k = a_ik - a_jk for all k
    # a_ik = (1-h[i+k-n+1]) - h[i-k+n-1] (with zero for OOB)
    # a_jk = (1-h[j+k-n+1]) - h[j-k+n-1] (with zero for OOB)
    
    d = np.zeros(2*n-1)
    for k in range(2*n-1):
        l = k - (n-1)
        a_ik = 0.0
        if 0 <= i+l < n:
            a_ik += (1.0 - h[i+l])
        if 0 <= i-l < n:
            a_ik -= h[i-l]
        a_jk = 0.0
        if 0 <= j+l < n:
            a_jk += (1.0 - h[j+l])
        if 0 <= j-l < n:
            a_jk -= h[j-l]
        d[k] = a_ik - a_jk
    
    c = np.correlate(h, 1-h, mode="full")
    
    # f(delta) = max_k (c[k] + d[k]*delta) * dx
    # Minimize over delta in [lo, hi]
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], h[j])
    
    # Ternary search
    for _ in range(100):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        f1 = np.max(c + d*m1)
        f2 = np.max(c + d*m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    delta = (lo+hi)/2
    h[i] += delta
    h[j] -= delta
    return h
```

```python
    # h[i-lk]: valid when 0 <= i-lk < n, i.e., i-n < lk <= i
```
