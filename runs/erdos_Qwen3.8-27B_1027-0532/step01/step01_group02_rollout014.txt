    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def compute_sensitivity(h, i):
    """Compute b[p] = d(corr[p])/d(h[i]) for all p."""
    n = len(h)
    dx = 2.0 / n
    s = np.arange(-(n-1), n)  # lags
    b = np.zeros(2*n - 1)
    
    # Term 1: (1 - h[i+s]) where 0 <= i+s < n
    idx1 = i + s
    valid1 = (idx1 >= 0) & (idx1 < n)
    b[valid1] += (1.0 - h[idx1[valid1]])
    
    # Term 2: -h[i-s] where 0 <= i-s < n
    idx2 = i - s
    valid2 = (idx2 >= 0) & (idx2 < n)
    b[valid2] -= h[idx2[valid2]]
    
    b *= dx
    return b

def coord_descent_sweep(h, corr, n_sweeps=10, rng=None, random_order=False):
    """Perform coordinate descent sweeps with incremental corr updates."""
    n = len(h)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            b = compute_sensitivity(h, i)
            
            # 1D subproblem: minimize max(corr + b*delta) for delta in [-h[i], 1-h[i]]
            lo = -h[i]
            hi = 1.0 - h[i]
            
            if hi - lo < 1e-14:
                continue  # h[i] is at a boundary and can't move
            
            # Generate candidates
            candidates = np.array([lo, hi])
            mask_b = np.abs(b) > 1e-12
            if mask_b.any():
                bp = -corr / b
                bp = bp[mask_b]
                in_range = (bp >= lo) & (bp <= hi)
                if in_range.any():
                    candidates = np.concatenate([candidates, bp[in_range]])
            
            # Evaluate objective at each candidate
            # scores[k] = max(corr + b * candidates[k])
            # Vectorized
            outer = np.outer(candidates, b)  # (M, 2n-1)
            scores = np.max(corr[None, :] + outer, axis=1)
            best_idx = np.argmin(scores)
            best_delta = candidates[best_idx]
            
            if abs(best_delta) > 1e-14:
                h[i] += best_delta
                corr += best_delta * b
        
        # Repair constraint
        h = repair_h(h, n)
        # Recompute corr after repair (repair might have changed multiple values)
        corr = np.correlate(h, 1.0 - h, mode="full")
    
    return h, corr

One more optimization: for the 1D subproblem, instead of generating all O(n) quadratic root candidates, I can note that:
- The concave function f₀(δ) = corr[p0] + b[p0]*δ - δ² is a downward parabola
- It intersects each linear function f_p(δ) = corr[p] + b[p]*δ at most twice
- The minimum of max(f₀, max_{p≠p0} f_p) occurs where:
  a) f₀ = f_p for the "active" p (the one with max linear value)
  b) At a breakpoint of the max of linear functions
  c) At a boundary

In practice, the "active" p for the linear part is the one with the largest value at the current δ. I can first find the minimizer of just the linear part (ignoring the concave), then check if f₀ is above it at that point. If so, the true minimum is where f₀ = max_linear.

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt

        else:

            cnt = mask.sum()
            if cnt == 0:

def compute_sens(h, i, n):
    s = np.arange(-(n-1), n)
    b = np.zeros(2*n - 1)
    idx1 = i + s
    v1 = (idx1 >= 0) & (idx1 < n)
    b[v1] = (1.0 - h[idx1[v1]])
    idx2 = i - s
    v2 = (idx2 >= 0) & (idx2 < n)
    b[v2] -= h[idx2[v2]]
    return b

def solve_1d(corr, b, h_i, n):
    """Solve the 1D subproblem: min max_p (corr[p] + b[p]*d - d^2*I(p==n-1)) for d in [-h_i, 1-h_i]."""
    p0 = n - 1
    lo = -h_i
    hi = 1.0 - h_i
    span = hi - lo
    if span < 1e-14:
        return 0.0
    
    # Build candidate list
    cands = np.array([lo, hi, 0.0])
    
    # Linear breakpoints
    mask_b = np.abs(b) > 1e-12
    if mask_b.any():
        bp = -corr / b
        bp = bp[mask_b]

        if len(bp) > 0:
            cands = np.concatenate([cands, bp])
    
    # Quadratic intersections (concave part with linear parts)
    b_minus = b - b[p0]  # (2n-1,)

        d1 = np.full(2*n-1, np.nan)

    obj = np.max(values, axis=1)
    
    best_idx = np.argmin(obj)
