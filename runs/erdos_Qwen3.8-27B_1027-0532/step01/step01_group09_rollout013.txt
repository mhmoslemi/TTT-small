    corr = np.correlate(h, 1.0-h, 'full') * dx

    m = corr.max()

def soft_max_grad(h, s):

    corr = np.correlate(h, 1.0-h, 'full') * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def soft_max_value(h, s):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    return m + s * np.log(np.sum(np.exp((corr - m) / s)))

def soft_max_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    p = np.exp((corr - m) / s)
    p /= p.sum()
    j = np.arange(n)
    grad = np.zeros(n)
    for idx in range(len(p)):
        pk = p[idx]
        if pk < 1e-14:
            continue
        k = idx - (n - 1)
        a = j - k
        b = j + k
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += pk * (1.0 - h[a[m1]])
        grad[m2] -= pk * h[b[m2]]
    grad *= dx
    return grad

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def soft_max_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    exp_c = np.exp((corr - m) / s)
    p = exp_c / exp_c.sum()

    prefix = np.zeros(2 * n)
    prefix[1:] = np.cumsum(p)
    i = np.arange(n)
    term_A = prefix[2*n - 1 - i] - prefix[n - 1 - i]

        pk = p[idx]
        if pk < 1e-14:
            continue
        k = idx - (n - 1)
        # term_B: subtract pk * h[i+k] for valid i

        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= pk * h[i_start2-k:i_end2-k]

    return grad
