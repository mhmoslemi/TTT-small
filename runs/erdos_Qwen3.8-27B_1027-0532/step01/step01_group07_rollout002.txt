import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    lo = -float(np.max(np.abs(h))) - 1.0
    hi = 1.0 - float(np.min(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(alpha * shifted)
    log_sum = np.log(exp_c.sum())
    val = c_max + log_sum / alpha
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        i_lo1 = max(0, m - n + 1)
        i_hi1 = min(n - 1, m - 1)
        for i in range(i_lo1, i_hi1 + 1):
            j = i - m + n - 1
            grad[i] += w * dx * (1.0 - h[j])
        i_lo2 = max(0, n - 1 - m)
        i_hi2 = min(n - 1, 2 * n - 2 - m)
        for i in range(i_lo2, i_hi2 + 1):
            k = i + m - n + 1
            grad[i] -= w * dx * h[k]
    return val, grad

def _lbfgs_fun(x, alpha, mu, target):
    n = len(x)
    val, grad = smooth_c5_grad(x, alpha)
    s = x.sum() - target
    val = val + mu * s * s
    grad = grad + 2.0 * mu * s
    return val, grad

def _slsqp_obj(val):
    return val[-1]

def _slsqp_ineq(val, n, m, dx):
    h = val[:n]
    t = val[n]
    c_val = 0.0
    for j in range(n):
        j2 = j + m - n + 1
        if 0 <= j2 < n:
            c_val += h[j2] * (1.0 - h[j])
    return t - c_val * dx

def _slsqp_eq(val, n, target):
    return val[:n].sum() - target
