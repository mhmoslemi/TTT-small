import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    max_tc = np.max(tc)
    exp_vals = np.exp(tc - max_tc)
    sum_exp = np.sum(exp_vals)
    obj = (max_tc + np.log(sum_exp)) / tau * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        # Term 1: h[i] appears with (1-h[i-lag]) for i in [max(0,lag), min(n-1, n-1+lag)]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        # Term 2: h[i] appears with -h[i+lag] for i in [max(0,-lag), min(n-1, n-1-lag)]
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_optimize(h0, n, target_sum, deadline, tau=100.0, max_iter=500):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        if hasattr(res, 'x'):
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return h0, compute_c5(h0)

def lbfgsb_optimize(h0, n, target_sum, deadline, tau=100.0, penalty=100.0):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        penalty_term = penalty * (x.sum() - target_sum) ** 2
        return o + penalty_term
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        g = g + 2 * penalty * (x.sum() - target_sum) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 5:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-14, 'disp': False,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    max_tc = np.max(tc)
    exp_vals = np.exp(tc - max_tc)
    sum_exp = np.sum(exp_vals)
    obj = (max_tc + np.log(sum_exp)) / tau * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return obj, grad * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)

        pass
