        n_corr = 2 * n - 1
        for k in range(n_corr):
            l = k - (n - 1)
            w = weights[k]
            if w < 1e-10:
                continue
            for j in range(n):
                term = 0.0
                jml = j - l
                if 0 <= jml < n:
                    term += (1.0 - h[jml])
                jpl = j + l
                if 0 <= jpl < n:
                    term += (-h[jpl])
                g[j] += w * term
        g *= dx
        return g
    return obj, grad

def slsqp_refine(h_init, n, eps, bounds, maxiter=300, ftol=1e-10):
    target = n / 2.0
    obj, grad = make_smooth_max(eps, n)
    def eq_con(x):
        return np.sum(x) - target
    res = minimize(
        obj, h_init.copy(), jac=grad, method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': eq_con}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    if np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h_init.copy()

def subgradient_step(h, n, step_size, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    subgrad = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            subgrad[j] += term
    subgrad /= len(active)
    subgrad *= dx
    h_new = h - step_size * subgrad
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 99999)
    start_time = time.time()

    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - start_time

    # Phase 1: Differential Evolution
    pop_size = 100
    pop = np.zeros((pop_size, n))
    fitness = np.zeros(pop_size)

    pop[0] = project_feasible(h_parent.copy(), n)
    pop[1] = project_feasible(1.0 - h_parent, n)
    pop[2] = project_feasible(h_parent[::-1], n)

    for idx in range(3, 15):
        h_s = np.zeros(n)
        num_ones = int(target)
        offset = (idx * 7) % n
        positions = np.unique((np.arange(num_ones).astype(float) * n // num_ones + offset).astype(int) % n)
        h_s[positions] = 1.0
        if positions.size < num_ones:
            remaining = np.setdiff1d(np.arange(n), positions)
            extra = remaining[:num_ones - positions.size]
            h_s[extra] = 1.0
        pop[idx] = project_feasible(h_s, n)

    for idx in range(15, pop_size):
        pop[idx] = project_feasible(rng.random(n), n)

    for i in range(pop_size):
        fitness[i] = evaluate_c5(pop[i])

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    de_time_limit = budget_s * 0.35
    F_start = 0.9
    F_end = 0.4
    CR_start = 0.9
    CR_end = 0.6
    gen = 0
    max_gen = 20000

    while elapsed() < de_time_limit and gen < max_gen:
        gen += 1
        progress = min(1.0, gen / max_gen)
        F = F_start + (F_end - F_start) * progress + 0.1 * (rng.random() - 0.5)
        CR = CR_start + (CR_end - CR_start) * progress + 0.1 * (rng.random() - 0.5)
        F = np.clip(F, 0.3, 1.0)
        CR = np.clip(CR, 0.3, 0.95)

        improved_this_gen = 0
        for i in range(pop_size):
            ri = list(range(pop_size))
            ri.remove(i)
            a, b, c = rng.choice(ri, size=3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(n) < CR
            mask[rng.integers(n)] = True
            u = np.where(mask, v, pop[i])
            u = project_feasible(u, n)
            f_u = evaluate_c5(u)
            if f_u <= fitness[i]:
                pop[i] = u
                fitness[i] = f_u
                if f_u < best_c5:
                    best_c5 = f_u
                    best_h = u.copy()
                    improved_this_gen += 1

        if gen % 200 == 0 and improved_this_gen == 0:
            worst_15 = np.argsort(fitness)[-15:]
            for wi in worst_15:
                pop[wi] = project_feasible(pop[wi] + 0.4 * rng.standard_normal(n), n)
                fitness[wi] = evaluate_c5(pop[wi])

    # Phase 2: SLSQP refinement of top DE solutions
    time_limit_2 = start_time + budget_s * 0.60
    order = np.argsort(fitness)
    top_k = min(10, pop_size)
    candidates = [pop[order[i]].copy() for i in range(top_k)]
    candidates.append(best_h.copy())

    for h_init in candidates:
        if elapsed() > budget_s * 0.60:
            break
        for eps in [200.0, 1000.0, 5000.0, 20000.0]:
            if elapsed() > budget_s * 0.60:
                break
            h_new = slsqp_refine(h_init, n, eps, bounds, maxiter=400)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 3: Multi-start with subgradient + SLSQP
    time_limit_3 = start_time + budget_s * 0.82
    n_starts = 25
    for s in range(n_starts):
        if elapsed() > budget_s * 0.82:
            break
        scale = 0.03 + 0.12 * s / max(1, n_starts - 1)
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)

        # Subgradient steps
        for sg_iter in range(200):
            if elapsed() > budget_s * 0.82:
                break
            alpha = 0.01 / (1.0 + sg_iter * 0.05)
            h_sg = subgradient_step(h_pert, n, alpha, rng)
            c5_sg = evaluate_c5(h_sg)
            if c5_sg < evaluate_c5(h_pert):
                h_pert = h_sg
            else:
                break

        for eps in [1000.0, 10000.0]:
            if elapsed() > budget_s * 0.82:
                break
            h_new = slsqp_refine(h_pert, n, eps, bounds, maxiter=250)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Coordinate descent + paired moves
    time_limit_4 = start_time + budget_s * 0.95
    cd_iter = 0
    max_cd = 10000
    while cd_iter < max_cd and elapsed() < budget_s * 0.95:
        cd_iter += 1
        if cd_iter % 3 == 0:
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 == i2:
                i2 = (i2 + 1) % n
            for d in [-0.03, -0.015, -0.005, 0.005, 0.015, 0.03]:
                h_trial = best_h.copy()
                h_trial[i1] = np.clip(h_trial[i1] + d, 0.0, 1.0)
                h_trial[i2] = np.clip(h_trial[i2] - d, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5 = evaluate_c5(h_trial)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_trial.copy()
                    break
        else:
            idx = rng.integers(n)
            for delta in [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]:
                h_trial = best_h.copy()
                h_trial[idx] = np.clip(h_trial[idx] + delta, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5 = evaluate_c5(h_trial)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_trial.copy()
                    break

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s - 10:
        for eps in [5000.0, 50000.0]:
            if elapsed() < budget_s - 5:
                h_new = slsqp_refine(best_h, n, eps, bounds, maxiter=500, ftol=1e-12)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Final subgradient polish
    if elapsed() < budget_s - 3:
        for sg_iter in range(500):
            if elapsed() > budget_s - 2:
                break
            alpha = 0.005 / (1.0 + sg_iter * 0.1)
            h_sg = subgradient_step(best_h, n, alpha, rng)
            c5_sg = evaluate_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
            else:
                break

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
