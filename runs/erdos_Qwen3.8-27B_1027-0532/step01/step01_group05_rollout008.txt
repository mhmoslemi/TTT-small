def fast_gradient(h, n, weights):
    """Compute gradient using FFT for the correlation-like terms."""
    dx = 2.0 / n
    # weights has length 2n-1, indexed as lags -(n-1) to n-1
    # grad[j] = dx * (sum_w - A[j] - B[j])
    # where A[j] = sum_l w_l * h[j+l] (with h[k]=0 outside [0,n-1])
    #      B[j] = sum_l w_l * h[j-l]
    
    n_corr = 2 * n - 1
    # Compute A[j] = sum_l w_l * h[j+l]
    # This is the cross-correlation of h and w, at specific lags
    # correlate(h, w, 'full')[k] = sum_i h[i] * w[i + k - (n-1)]
    
    # Actually, let me just use the direct approach but vectorized
    # For each j, A[j] = sum_{l: j+l in [0,n-1]} w_{l+(n-1)} * h[j+l]
    
    # This is equivalent to: A = correlate(h_padded, w, 'same') or similar
    # Let me just use the loop approach which is O(n * n_corr) but with numpy vectorization
    
    grad = np.zeros(n)
    for idx in range(n_corr):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += -w * h[:n - lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - h[:n - l])
            grad[:n - l] += -w * h[l:]
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smoothed_objective(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (max_corr + np.log(sum_exp) / t) * dx
    weights = exp_vals / sum_exp
    return obj, weights, corr


def compute_gradient(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for idx in range(n_corr):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += -w * h[:n - lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - h[:n - l])
            grad[:n - l] += -w * h[l:]
    grad *= dx
    return grad


def nesterov_projected(h0, n, t, n_iter, step_init=None, momentum_init=0.5):
    dx = 2.0 / n
    h = h0.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    mu = momentum_init

    if step_init is None:
        step_init = dx * 0.5

    for it in range(n_iter):
        if it > 0:
            h_y = h + mu * (h - h_prev)
            h_y = project_onto_constraint(h_y, n)
            h_prev = h
            h = h_y
        else:
            h_y = h.copy()

        obj, weights, corr = smoothed_objective(h_y, n, t)
        grad = compute_gradient(h_y, n, weights)

        alpha = step_init
        for _ in range(8):
            h_trial = h_y - alpha * grad
            h_trial = project_onto_constraint(h_trial, n)
            obj_trial, _, _ = smoothed_objective(h_trial, n, t)
            if obj_trial <= obj - 1e-4 * alpha * np.dot(grad, h_trial - h_y) + 1e-14:
                break
            alpha *= 0.5

        h_new = h_y - alpha * grad
        h_new = project_onto_constraint(h_new, n)

        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        if it > 0:
            mu = (1.0 + 0.5 / (it + 1.0)) / (1.0 + mu)

    return best_h, best_c5


def subgradient_focus(h, n, n_iter=200, step_init=0.05):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step_init

    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] += -h[:n - lag]
        else:
            l = -lag
            grad[l:] += (1.0 - h[:n - l])
            grad[:n - l] += -h[l:]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.998

    return best_h, best_c5


def slsqp_polish(h0, n, t, max_iter=150):
    h0 = project_onto_constraint(np.clip(h0.astype(float), 0.0, 1.0), n)

    def obj_func(x):
        val, _, _ = smoothed_objective(x, n, t)
        return val

    def grad_func(x):
        _, weights, _ = smoothed_objective(x, n, t)
        return compute_gradient(x, n, weights)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        result = minimize(
            obj_func, h0.copy(), method='SLSQP',
            jac=grad_func, bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-10}
        )
        h_opt = project_onto_constraint(np.clip(result.x, 0.0, 1.0), n)
        return h_opt
    except Exception:
        return h0.copy()
