        grad[idx] += (1.0 - h[idx - jj])
        grad[idx - jj] -= h[idx]

        dx_try = 2.0 / n_try
        h_start = interpolate_h(best_h, best_n, n_try)

        for beta in [80.0, 40.0, 20.0, 12.0, 8.0, 5.0, 4.0]:
            if remaining() < 50:
                break
            h_opt = run_lbfgs(h_start, n_try, dx_try, beta, maxiter=250, penalty=500.0)
            c = c5_value(h_opt, dx_try)
            if c < best_c5:
                best_h = h_opt.copy()
                best_c5 = c
                best_n = n_try
            h_start = h_opt

        if remaining() > 50 and best_n == n_try:
            h_sg, c_sg = subgradient_refine(best_h, n_try, dx_try, max_iter=1000, lr_init=0.025)
            if c_sg < best_c5:
                best_h = h_sg.copy()
                best_c5 = c_sg
                best_n = n_try

    n = best_n
    dx = 2.0 / n

    for attempt in range(25):
        if remaining() < 35:
            break
        scale = 0.04 * (1.0 - attempt / float(max(1, 25))) + 0.008
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = repair_h(h_pert, n)
        h_opt = run_lbfgs(h_pert, n, dx, 10.0, maxiter=200, penalty=500.0)
        c = c5_value(h_opt, dx)
        if c < best_c5:
            best_h = h_opt.copy()
            best_c5 = c

    if remaining() > 40:
        h_bin, h_blend = binary_push(best_h, n)
        for h_try in [h_bin, h_blend]:
            if remaining() < 25:
                break
            h_opt = run_lbfgs(h_try, n, dx, 10.0, maxiter=300, penalty=500.0)
            c = c5_value(h_opt, dx)
            if c < best_c5:
                best_h = h_opt.copy()
                best_c5 = c

    if remaining() > 30:
        h_sg, c_sg = multi_active_subgrad(best_h, n, dx, max_iter=3000, lr_init=0.02, num_active=5)
        if c_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c_sg

    if remaining() > 25:
        for beta in [4.0, 3.0, 2.5, 2.0]:
            if remaining() < 15:
                break
            h_opt = run_lbfgs(best_h, n, dx, beta, maxiter=600, penalty=1000.0)
            c = c5_value(h_opt, dx)
            if c < best_c5:
                best_h = h_opt.copy()
                best_c5 = c

    if remaining() > 20:
        h_sg, c_sg = subgradient_refine(best_h, n, dx, max_iter=5000, lr_init=0.015)
        if c_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c_sg

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    assert np.all(np.isfinite(best_h))
    assert np.all(best_h >= -1e-14)
    assert np.all(best_h <= 1.0 + 1e-14)

    return best_h, best_c5, best_n
