import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def fast_full_corr(h):
    """Compute correlate(h, 1-h, 'full') via FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    one_minus_h = 1.0 - h
    b_flip = one_minus_h[::-1]
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(b_flip, N)
    return np.fft.irfft(fa * fb, N)[:2 * n - 1]

def project_onto_constraint(h, target_sum):
    lo = -0.5
    hi = 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_lag(h, lag, n):
    """Gradient of C(h, lag) w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    j_lo = max(0, -lag)
    j_hi = min(n - 1, n - 1 - lag)
    if j_lo <= j_hi:
        g[j_lo:j_hi+1] += (1.0 - h[j_lo + lag:j_hi + lag + 1])
    j2_lo = max(0, lag)
    j2_hi = min(n - 1, n - 1 + lag)
    if j2_lo <= j2_hi:
        g[j2_lo:j2_hi+1] += (-h[j2_lo - lag:j2_hi - lag + 1])
    return g * dx

def soft_max_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_corr(h)
    tc = tau * corr
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    sum_exp = np.sum(exp_tc)
    obj = (tc_max + np.log(sum_exp)) / tau * dx
    w = exp_tc / sum_exp
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        lag = m - (n - 1)
        grad += w[m] * compute_grad_lag(h, lag, n)
    return obj, grad

    """L-BFGS-B with penalty for sum constraint."""
    
    def obj(x):
        o, _ = soft_max_obj_grad(x, tau)
        return o + mu * (x.sum() - target_sum) ** 2
    
    def grad(x):

        h_opt = np.clip(res.x, 0.0, 1.0)

        return h_opt, compute_c5(h_opt)

def multi_lag_subgrad(h0, n, target_sum, deadline, top_k=5, max_iter=10000):
    """Subgradient method using top-k lags."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(corr)
        
        # Get top-k lags
        top_indices = np.argsort(corr)[::-1][:top_k]
        
        # Weight by value (softmax)
        top_vals = corr[top_indices]
        w = np.exp(top_vals / 0.05)
        w /= w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for idx, m in enumerate(top_indices):
            lag = m - (n - 1)
            grad += w[idx] * compute_grad_lag(h, lag, n)
        
        # Step with diminishing learning rate
        lr = 0.1 / (1 + 0.005 * it)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Accept or reject
        if c5_new < c5_val + 1e-10:
            h = h_new
        else:
            h = h_new  # still move, but don't update best
    
    return best_h, best_c5

def soft_max_obj_grad(h, tau):

            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Interpolate to n=240, coordinate descent
    n1 = 240
    if time.time() < deadline - 250:
        h1 = interpolate_h(best_h, n1)
        h1 = project_onto_constraint(h1, n1 / 2.0)
        h1, c5_1 = coord_descent(h1, n1, n1 / 2.0, deadline - 150, max_sweeps=8000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 3: L-BFGS-B at n=240
    if time.time() < deadline - 120:
        for tau in [25, 60, 150]:
            if time.time() > deadline - 40:
                break
            h_opt, c5_opt = lbfgsb_optimize(best_h, 240, 120.0, tau, max_iter=80, deadline=deadline, mu=800.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Coordinate descent again at n=240
    if time.time() < deadline - 80:
        h4 = interpolate_h(best_h, 240)
        h4 = project_onto_constraint(h4, 120.0)
        h4, c5_4 = coord_descent(h4, 240, 120.0, deadline - 30, max_sweeps=5000)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Try n=360
    n2 = 360
    if time.time() < deadline - 80:
        h5 = interpolate_h(best_h, n2)
        h5 = project_onto_constraint(h5, n2 / 2.0)
        h5, c5_5 = coord_descent(h5, n2, n2 / 2.0, deadline - 30, max_sweeps=4000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Subgradient polish on best
    if time.time() < deadline - 20:
        n_best = len(best_h)
        h6 = project_onto_constraint(best_h, n_best / 2.0)
        h6, c5_6 = subgrad_polish(h6, n_best, n_best / 2.0, deadline - 5, top_k=5, max_iter=3000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final coordinate descent polish
    if time.time() < deadline - 2:
        n_final = len(best_h)
        h_final = project_onto_constraint(best_h, n_final / 2.0)
        h_final, c5_final = coord_descent(h_final, n_final, n_final / 2.0, deadline, max_sweeps=5000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    return best_h, best_c5, len(best_h)
