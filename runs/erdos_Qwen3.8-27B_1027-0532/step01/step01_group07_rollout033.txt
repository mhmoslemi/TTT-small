```python
    # The first term: sum_m w[m] * I(0≤j+lags[m]<n)
    # For a fixed j, lags[m] ∈ [-j, n-1-j]
    # m ∈ [n-1-j, 2n-2-j] intersect [0, 2n-2]
    
    # Let me use a different approach: create a (2n-1) x n matrix W where
    # W[m, j] = w[m] if 0 ≤ j + lags[m] < n, else 0
    # Then Part1_sum = W.T @ np.ones(n) ... no, that's just column sums of W.
    
    # Actually: Part1[j] = sum_m W[m,j] * (1 - h[j+lags[m]])
    # This is not simply W.T @ something because h[j+lags[m]] depends on j and m differently.
    
    # Let me think of it as: Part1[j] = S1[j] - P1[j]
    # where S1[j] = sum_m w[m] * I(0≤j+lags[m]<n) and P1[j] = sum_m w[m]*h[j+lags[m]]*I(0≤j+lags[m]<n)
    
    # S1[j] can be precomputed: it's the sum of w[m] over m where lags[m] ∈ [-j, n-1-j]
    # = sum of w[m] for m ∈ [n-1-j, 2n-2-j] ∩ [0, 2n-2]
    
    # For the vectorized computation, let me use a different approach:
    # Note that P1[j] = sum_{valid m} w[m] * h[j + lags[m]]
    # Let k = j + lags[m], so j = k - lags[m] = k - m + n - 1
    # For fixed k, m = k - j + n - 1... this is getting circular.
    
    # Alternative: P1 is a "weighted correlation" of h with w.
    # P1[j] = sum_{l} w[j+l-(n-1)] * h[j+l] * I(0≤j+l<n) * I(0≤j+l-(n-1)≤2n-2)
    # Hmm, let l = lags[m] = m - (n-1), so m = l + n - 1
    # P1[j] = sum_{l: 0≤j+l<n, 0≤l+n-1≤2n-2} w[l+n-1] * h[j+l]
    # = sum_{l: -j≤l≤n-1-j, -(n-1)≤l≤n-1} w[l+n-1] * h[j+l]
    # = sum_{l: max(-j, -(n-1))≤l≤min(n-1-j, n-1)} w[l+n-1] * h[j+l]
    
    # This is like a cross-correlation of h with a windowed version of w!
    # P1[j] = cross_corr(h, w_windowed)[j] where w_windowed is w shifted and windowed.
    
    # Specifically, if we define w' as a length (2n-1) array where w'[l+(n-1)] = w[l+(n-1)] = w[l+n-1],
    # then P1[j] = sum_l h[j+l] * w'[l+n-1] * I(valid)
    
    # This is the cross-correlation of h and w'!
    # np.correlate(h, w', mode='full') at the right offset gives us P1.
    
    # Similarly for Part2.
    
    # Let me just use the direct approach with numpy for n=44 (it's small enough).
    
    # Create index arrays
    j_idx = np.arange(n)[:, None]  # shape (n, 1)
    l_idx = lags[None, :]  # shape (1, 2n-1)
    
    ip = j_idx + l_idx  # shape (n, 2n-1): j + lags[m]
    im = j_idx - l_idx  # shape (n, 2n-1): j - lags[m]
    
    valid_p = (ip >= 0) & (ip < n)  # shape (n, 2n-1)
    valid_m = (im >= 0) & (im < n)  # shape (n, 2n-1)
    
    # For Part 1: sum over m of w[m] * (1-h[ip]) * valid_p
    # We need h[ip] for valid entries
    ip_clipped = np.clip(ip, 0, n-1)
    im_clipped = np.clip(im, 0, n-1)
    
    h_ip = h[ip_clipped]  # shape (n, 2n-1)
    h_im = h[im_clipped]  # shape (n, 2n-1)
    
    part1 = np.sum(w[None, :] * (1.0 - h_ip) * valid_p, axis=1)  # shape (n,)
    part2 = np.sum(w[None, :] * h_im * valid_m, axis=1)  # shape (n,)
    
    grad = dx * (part1 - part2)
    
    return sm_val, grad
```

This is O(n * 2n) = O(n²) which for n=44 is 44*87 = 3828 operations per gradient eval. Very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _project(h, target):
    """Project h to {0≤h≤1, sum(h)=target} using bisection on shift."""
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_and_grad(h, alpha):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()  # numerical stability
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    sm_val = c_max + (shifted.max() + np.log(exp_c.sum())) / alpha
    
    lags = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    l_idx = lags[None, :]
    ip = j_idx + l_idx
    im = j_idx - l_idx
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    ip_c = np.clip(ip, 0, n - 1)
    im_c = np.clip(im, 0, n - 1)
    part1 = np.sum(w[None, :] * (1.0 - h[ip_c]) * valid_p, axis=1)
    part2 = np.sum(w[None, :] * h[im_c] * valid_m, axis=1)
    grad = dx * (part1 - part2)
    return sm_val, grad

def _lbfgs(h_init, n, max_time=30.0, alpha=50.0):
    """Optimize using L-BFGS-B with projection."""
    start = time.time()
    target = n / 2.0
    
    def obj_and_grad(x):
        h = _project(x, target)
        val, grad = _smooth_obj_and_grad(h, alpha)
        # Need to propagate gradient through projection... 
        # For simplicity, just return the gradient at the projected point
        return val, grad
    
    # Actually, L-BFGS-B doesn't work well with projection in the objective.
    # Let me use a different approach: optimize in the unconstrained space
    # and project after each iteration.
    
    # Better: use the penalty method
    penalty = 1000.0
    
    def penalized_obj(x):
        h = np.clip(x, 0.0, 1.0)
        val, grad = _smooth_obj_and_grad(h, alpha)
        pen = penalty * (h.sum() - target) ** 2
        pg = 2 * penalty * (h.sum() - target) * np.ones(n)
        return val + pen, grad + pg
    
    res = minimize(penalized_obj, h_init, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 500, 'maxfun': 2000, 
                            'ftol': 1e-12, 'gtol': 1e-8,
                            'maxls': 50})
    
    h_opt = _project(res.x, target)
    return h_opt

def _gd(h_init, n, alpha=50.0, max_time=30.0, lr=None):
    """Gradient descent with adaptive parameters."""
    start = time.time()
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if lr is None:
        lr = 0.1 / max(1, n)
    
    alpha_current = alpha
    for it in range(20000):
        if time.time() - start > max_time:
            break
        val, grad = _smooth_obj_and_grad(h, alpha_current)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Normalize gradient and scale
        step = lr * grad / gnorm
        h_new = h - step
        h_new = _project(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new <= c5 + 1e-10:
            h = h_new
            # Adapt alpha
            if it % 500 == 0 and it > 0:
                alpha_current = min(alpha_current * 1.5, 500.0)
            # Adapt lr
            if it % 1000 == 0 and it > 0:
                lr *= 0.8
        else:
            lr *= 0.7
            if lr < 1e-15:
                break
    
    return best_h, best_c5

def _simulated_annealing_binary(n, max_time=10.0, rng=None):
    """Find good binary configuration via simulated annealing."""
    if rng is None:
        rng = np.random.RandomState(0)
    start = time.time()
    
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    temp = 0.1
    for it in range(50000):
        if time.time() - start > max_time:
            break
        
        # Pick a random position to flip
        i = rng.randint(n)
        if h[i] > 0.5:
            # Try to flip to 0
            j = rng.randint(n)
            if h[j] < 0.5:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / temp):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i] = 1
                    h[j] = 0
        else:
            j = rng.randint(n)
            if h[j] > 0.5:
                h[i] = 1
                h[j] = 0
                new_c5 = compute_c5(h)
                if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / temp):
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[i] = 0
                    h[j] = 1
        
        temp *= 0.9999
    
    return best_h
```

```python
def _lbfgs_reduced(h_init, n, max_time=30.0, alpha=50.0):
    """L-BFGS-B with one variable eliminated for the equality constraint."""
    start = time.time()
    target = n / 2.0
    
    # Work with first n-1 variables; last is determined
    x0 = h_init[:n-1].copy()
    
    def obj_and_grad(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target - x.sum()
        
        # Check feasibility
        if h[n-1] < 0 or h[n-1] > 1:
            # Infeasible - return large value
            return 1e6, np.zeros(n-1)
        
        val, grad = _smooth_obj_and_grad(h, alpha)
        # Gradient w.r.t. x (first n-1 variables)
        # h[n-1] = target - sum(x), so dh[n-1]/dx[i] = -1
        # d(val)/dx[i] = grad[i] - grad[n-1]  (chain rule)
        gx = grad[:n-1] - grad[n-1]
        return val, gx
    
    bounds = [(0.0, 1.0)] * (n-1)
    
    res = minimize(obj_and_grad, x0, method='L-BFGS-B',
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def _project(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    shifted -= shifted.max()
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    sm_val = c_max + (shifted.max() + np.log(exp_c.sum())) / alpha
    lags = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    l_idx = lags[None, :]
    ip = j_idx + l_idx
    im = j_idx - l_idx
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    ip_c = np.clip(ip, 0, n - 1)
    im_c = np.clip(im, 0, n - 1)
    part1 = np.sum(w[None, :] * (1.0 - h[ip_c]) * valid_p, axis=1)
    part2 = np.sum(w[None, :] * h[im_c] * valid_m, axis=1)
    grad = dx * (part1 - part2)
    return sm_val, grad

def _gd_refine(h_init, n, alpha_start, max_iters, deadline):

def _lbfgs_refine(h_init, n, max_time, alpha):
    start = time.time()
    target = n / 2.0
    penalty = 5000.0

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        val, grad = _smooth_obj_grad(h, alpha)
        s = h.sum() - target
        pen_val = penalty * s * s
        pen_grad = 2.0 * penalty * s
        in_box = (x >= 0.0) & (x <= 1.0)
        total_grad = (grad + pen_grad) * in_box
        return val + pen_val, total_grad

    try:
        res = minimize(obj_grad, h_init, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,

                                'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50,
                                'max_time': max(max_time - (time.time() - start), 1.0)})
        h_opt = _project(res.x, target)
        return h_opt
    except Exception:

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        temp *= 0.9998
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def time_left():
        return budget_s - (time.time() - start_time)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    primary_n = 44
    if h0 is not None:
        primary_n = len(h0)

    n_list = [primary_n]
    for cand in [44, 42, 46, 48, 40, 50, 52]:
        if cand != primary_n and cand not in n_list:
            n_list.append(cand)

    deadline = start_time + budget_s - 10

    # Phase 1: SA for binary starts (primary n only)
    if time_left() > 15:
        h_sa, c5_sa = _sa_binary(primary_n, min(8.0, time_left() - 5), rng)
        h_cont = _project(h_sa * 0.85 + 0.15, primary_n / 2.0)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 2: Multi-start local optimization
    all_inits = []
    for n in n_list:
        if h0 is not None and n == len(h0):
            all_inits.append((n, h0.copy()))
            all_inits.append((n, 1.0 - h0.copy()))
            for pert_seed in range(4):
                r2 = np.random.RandomState(seed + 100 + pert_seed)
                hp = h0 + r2.normal(0, 0.03, n)
                all_inits.append((n, _project(hp, n / 2.0)))
        for k in range(4):
            all_inits.append((n, _make_init(n, rng, k)))

    if best_h is not None:
        for n in n_list:
            if len(best_h) == n:
                all_inits.append((n, best_h.copy()))
            else:
                x_old = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n)
                hi = np.interp(x_new, x_old, best_h)
                all_inits.append((n, _project(hi, n / 2.0)))

    for (n, h_init) in all_inits:
        if time.time() > deadline or time_left() < 8:
            break
        h_init = _project(h_init.copy(), n / 2.0)
        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()

        # GD pass 1 (moderate alpha)
        t_limit = min(12.0, time_left() - 3)
        if t_limit > 2:
            h1, c5_1 = _gd_refine(h_init, n, 25.0, 8000, start_time + time.time() + t_limit)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

        # GD pass 2 (high alpha)
        t_limit = min(12.0, time_left() - 3)
        if t_limit > 2:
            h2, c5_2 = _gd_refine(h1, n, 150.0, 6000, start_time + time.time() + t_limit)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

        # L-BFGS-B
        t_limit = min(12.0, time_left() - 3)
        if t_limit > 3:
            h3 = _lbfgs_refine(h2, n, t_limit, 80.0)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

        # Coordinate descent polish
        t_limit = min(8.0, time_left() - 2)
        if t_limit > 2:
            h4, c5_4 = _coord_descent(h3, n, t_limit)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 3: Intensive refinement on best
    if best_h is not None and time_left() > 10:
        n = len(best_h)
        # Very high alpha GD
        t_limit = min(30.0, time_left() - 5)
        if t_limit > 5:
            h_f1, c5_f1 = _gd_refine(best_h, n, 500.0, 20000, start_time + time.time() + t_limit)
            if c5_f1 < best_c5:
                best_c5 = c5_f1
                best_h = h_f1.copy()

        # L-BFGS-B high alpha
        t_limit = min(20.0, time_left() - 5)
        if t_limit > 5:
            h_f2 = _lbfgs_refine(h_f1, n, t_limit, 300.0)
            c5_f2 = compute_c5(h_f2)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

        # Final coordinate descent
        t_limit = min(20.0, time_left() - 3)
        if t_limit > 3:
            h_f3, c5_f3 = _coord_descent(h_f2, n, t_limit)
            if c5_f3 < best_c5:
                best_c5 = c5_f3
                best_h = h_f3.copy()

        # One more GD pass
        t_limit = min(10.0, time_left() - 2)
        if t_limit > 2:
            h_f4, c5_f4 = _gd_refine(h_f3, n, 1000.0, 10000, start_time + time.time() + t_limit)
            if c5_f4 < best_c5:
                best_c5 = c5_f4
                best_h = h_f4.copy()

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_h = _project(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
