    corr = np.correlate(x, 1 - x, mode="full")
    return np.sum(corr[_global_active]) * dx

def active_grad(x):
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in _global_active:
        l = k - (n - 1)
        for j in range(n):
            jml = j - l
            if 0 <= jml < n:
                grad[j] += (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                grad[j] += (-x[jpl])
    return grad * dx

def equality_constraint(x):
    return np.sum(x) - len(x) / 2.0

def run(seed=42, budget_s=1300, **kwargs):
    global _global_eps, _global_active
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    pop_size = 100
    pop = np.zeros((pop_size, n))
    pop[0] = h0
    for i in range(1, pop_size):
        strategy = i % 6
        if strategy == 0:
            pop[i] = np.random.random(n)
        elif strategy == 1:
            num_blocks = 3 + (i % 5)
            block_size = n // num_blocks
            for b in range(num_blocks):
                val = 0.0 + 1.0 * (b % 2)
                s_b = b * block_size
                e_b = min(s_b + block_size, n)
                pop[i][s_b:e_b] = val + 0.15 * np.random.randn(e_b - s_b)
        elif strategy == 2:
            x_arr = np.linspace(0, 2, n)
            freq = 1.0 + (i % 6) * 0.5
            amp = 0.2 + 0.3 * ((i * 7) % 10) / 10.0
            pop[i] = 0.3 + amp * np.sin(freq * np.pi * x_arr + (i % 3) * 0.5)
        elif strategy == 3:
            pop[i] = h0 + 0.25 * np.random.randn(n)
        elif strategy == 4:
            pop[i] = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.linspace(0, 2, n) * (1 + (i % 4) * 0.5))
        else:
            pop[i] = np.random.random(n) ** 2
        pop[i] = project_feasible(pop[i], n)

    pop_fit = np.array([evaluate_c5(pop[i]) for i in range(pop_size)])
    pop_best_idx = np.argmin(pop_fit)
    if pop_fit[pop_best_idx] < best_c5:
        best_c5 = pop_fit[pop_best_idx]
        best_h = pop[pop_best_idx].copy()

    max_gens = 400
    stagnant = 0
    for gen in range(max_gens):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 500:
            break

        progress = gen / max_gens
        F_adapt = 0.9 - 0.5 * progress
        CR_adapt = 0.95 - 0.3 * progress

        gen_best = best_c5
        for i in range(pop_size):
            if i % 25 == 0:
                elapsed = time.time() - start_time
                if budget_s - elapsed < 500:
                    break
            candidates = np.array([x for x in range(pop_size) if x != i])
            b, c, d = np.random.choice(candidates, size=3, replace=False)
            mutant = pop[i] + F_adapt * (pop[b] - pop[c])
            j_rand = np.random.randint(n)
            mask = np.random.random(n) < CR_adapt
            mask[j_rand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_feasible(trial, n)
            trial_fit = evaluate_c5(trial)
            if trial_fit < pop_fit[i]:
                pop[i] = trial
                pop_fit[i] = trial_fit
                if trial_fit < gen_best:
                    gen_best = trial_fit
                    if trial_fit < best_c5:
                        best_c5 = trial_fit
                        best_h = trial.copy()

        if gen_best < best_c5 + 1e-8:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= 40:
            worst_count = min(5, pop_size // 10)
            worst_idx = np.argsort(pop_fit)[-worst_count:]
            for wi in worst_idx:
                pop[wi] = best_h + 0.5 * np.random.randn(n)
                pop[wi] = project_feasible(pop[wi], n)
                pop_fit[wi] = evaluate_c5(pop[wi])
            stagnant = 0

    for outer in range(30):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 200:
            break
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-4)[0]
        _global_active = active
        res = minimize(
            active_obj, best_h.copy(), jac=active_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': equality_constraint}],
            options={'maxiter': 300, 'ftol': 1e-15}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    for eps_val in [50.0, 100.0, 300.0, 1000.0, 5000.0]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 80:
            break
        _global_eps = eps_val
        res = minimize(
            smooth_max_obj, best_h.copy(), jac=smooth_max_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': equality_constraint}],
            options={'maxiter': 500, 'ftol': 1e-15}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    n_restarts = 50
    for r in range(n_restarts):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 40:
            break
        if r % 4 == 0:
            scale = 0.002 + 0.006 * (r / float(n_restarts))
            h_pert = best_h + scale * np.random.randn(n)
        elif r % 4 == 1:
            idx1, idx2 = np.random.choice(n, size=2, replace=False)
            h_pert = best_h.copy()
            h_pert[idx1], h_pert[idx2] = h_pert[idx2], h_pert[idx1]
            h_pert[idx1] += 0.008 * np.random.choice([-1, 1])
            h_pert[idx2] -= 0.008 * np.random.choice([-1, 1])
        elif r % 4 == 2:
            block_start = np.random.randint(0, max(1, n - 12))
            block_len = np.random.randint(4, min(12, n - block_start))
            h_pert = best_h.copy()
            h_pert[block_start:block_start + block_len] += 0.012 * np.random.choice([-1, 1])
        else:
            h_pert = best_h + 0.003 * np.random.randn(n)
            h_pert[np.random.randint(n)] += 0.01 * np.random.choice([-1, 1])
        h_pert = project_feasible(h_pert, n)
        corr = np.correlate(h_pert, 1 - h_pert, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-4)[0]
        _global_active = active
        res = minimize(
            active_obj, h_pert.copy(), jac=active_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': equality_constraint}],
            options={'maxiter': 150, 'ftol': 1e-14}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    for cd_round in range(15):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 20:
            break
        improved = False
        for idx in range(n):
            if idx % 20 == 0:
                elapsed = time.time() - start_time
                if budget_s - elapsed < 20:
                    break
            original = best_h[idx]
            for delta in [-0.02, -0.008, -0.002, 0.002, 0.008, 0.02]:
                h_trial = best_h.copy()
                h_trial[idx] = original + delta
                if h_trial[idx] < 0.0 or h_trial[idx] > 1.0:
                    continue
                h_trial = project_feasible(h_trial, n)
                c5 = evaluate_c5(h_trial)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_trial.copy()
                    improved = True
                    break
        if not improved:
            break

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        _global_eps = 20000.0
        res = minimize(
            smooth_max_obj, best_h.copy(), jac=smooth_max_grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': equality_constraint}],
            options={'maxiter': 500, 'ftol': 1e-16}
        )
        if np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n
