def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

    scaled = _corr_scaled(h)
    m = np.max(scaled)
    if m < -500:
        return 0.0
    exp_s = np.exp(alpha * (scaled - m))
    return m + np.log(np.sum(exp_s)) / alpha

    """Upsample from n_low to n_high points using nearest neighbor."""

def _subgradient_step(h, n, target_sum, step_size=0.01):
    """Take a subgradient descent step on the max of C(s)."""
    scaled = _corr_scaled(h)
    k_max = np.argmax(scaled)
    s_max = k_max - (n - 1)

import time
import numpy as np
from scipy.optimize import minimize


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _smooth_obj(h, alpha):
    scaled = _corr_scaled(h)
    m = np.max(scaled)
    if m < -500:
        return 0.0
    exp_s = np.exp(alpha * (scaled - m))
    return m + np.log(np.sum(exp_s)) / alpha


def _smooth_obj_grad(h, alpha):
    n = len(h)
    scaled = _corr_scaled(h)
    m = np.max(scaled)
    if m < -500:
        return 0.0, np.zeros(n)
    exp_s = np.exp(alpha * (scaled - m))
    w = exp_s / np.sum(exp_s)
    obj = m + np.log(np.sum(exp_s)) / alpha
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    idx1 = j_idx + k_idx - (n - 1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n - 1)], 0.0)
    term1 = h1 @ w
    idx2 = j_idx - k_idx + (n - 1)
    mask2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n - 1)], 0.0)
    term2 = h2 @ w
    dx = 2.0 / n
    grad = dx * (1.0 - term1 - term2)
    return obj, grad


def _optimize(h0, n, target_sum, alpha, maxiter=300, method='SLSQP'):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_obj(x, alpha)

    def obj_grad(x):
        return _smooth_obj_grad(x, alpha)

    def con_fun(x):
        return x.sum() - target_sum

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        if method == 'SLSQP':
            res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
        elif method == 'trust-constr':
            res = minimize(obj, h0, jac=obj_grad, method='trust-constr',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'gtol': 1e-10,
                                    'xtol': 1e-12})
        else:
            return h0
        return res.x
    except Exception:
        return h0


def _upsample(h_low, n_high):
    n_low = len(h_low)
    indices = np.floor(np.arange(n_high) * n_low / n_high).astype(int)
    indices = np.clip(indices, 0, n_low - 1)
    return h_low[indices].copy()


def _subgradient_step(h, n, target_sum, step_size=0.01):
    scaled = _corr_scaled(h)
    k_max = np.argmax(scaled)
    s_max = k_max - (n - 1)
    j = np.arange(n)
    idx_p = j + s_max
    idx_m = j - s_max
    hp = np.where((idx_p >= 0) & (idx_p < n), h[np.clip(idx_p, 0, n - 1)], 0.0)
    hm = np.where((idx_m >= 0) & (idx_m < n), h[np.clip(idx_m, 0, n - 1)], 0.0)
    g = 1.0 - hp - hm
    g_proj = g - np.mean(g)
    h_new = h - step_size * g_proj
    h_new = _fix_sum(h_new, target_sum)
    return h_new

                if not time_ok(25):
                    break
                h_new = _optimize(h_try, n_try, ts_try, alpha, maxiter=200, method='SLSQP')
                h_new = _fix_sum(h_new, ts_try)
                c5_new = _c5_of(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    n = n_try
                    target_sum = ts_try
                    break

    if time_ok(80):
        for alpha in [500, 1000, 2000, 5000]:
            if not time_ok(30):
                break
            h_new = _optimize(best_h, n, target_sum, alpha, maxiter=400, method='trust-constr')
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    if time_ok(40):
        h_ref, c5_ref = _local_refine(best_h, n, target_sum, max_iters=2000)
        if c5_ref < best_c5 - 1e-12:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if time_ok(30):
        h_ref, c5_ref = _pair_swap_refine(best_h, n, target_sum, max_iters=5000)
        if c5_ref < best_c5 - 1e-12:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    best_h = _fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, len(best_h))
