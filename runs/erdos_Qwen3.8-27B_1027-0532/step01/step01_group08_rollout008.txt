    # Let's compute using the direct formula but vectorized:
    # For each lag ell in [-(n-1), n-1]:
    #   weight = weights[ell + n - 1]
    #   contribution to g from this lag:
    #     for i where 0 <= i+ell < n: g[i] += weight * (1 - h[i+ell])
    #     for i where 0 <= i-ell < n: g[i] -= weight * h[i-ell]
    
    # Using FFT approach:
    # g[i] = sum_ell w[ell] * (1-h[i+ell]) * 1[0<=i+ell<n] - sum_ell w[ell] * h[i-ell] * 1[0<=i-ell<n]
    
    # Let's define W as array indexed from -(n-1) to (n-1)
    # W[ell] = weights[ell + n - 1]
    
    # First term: sum_ell W[ell] * (1-h[i+ell]) for valid i
    # = sum_ell W[ell] - sum_ell W[ell]*h[i+ell] for valid i
    # = sum_ell W[ell] - (W circ conv h)[i]  (with appropriate truncation)
    
    # Second term: sum_ell W[ell] * h[i-ell] for valid i
    # = (W rev circ conv h)[i]
    
    # Actually, let me just use the direct O(n^2) approach for now since n is at most ~500
    # and it's simple and correct.
    
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        # First part: (1 - h[i+lag]) for valid i
        if lag >= 0:
            i_range = slice(0, n - lag)
            g[i_range] += w * (1.0 - h[lag:n])
        else:
            i_range = slice(-lag, n)
            g[i_range] += w * (1.0 - h[0:n+lag])
        # Second part: -h[i-lag] for valid i
        if lag >= 0:
            i_range2 = slice(lag, n)
            g[i_range2] += -w * h[0:n-lag]
        else:
            i_range2 = slice(0, n+lag)
            g[i_range2] += -w * h[-lag:n]
    
    return g * dx

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_tc = tau * np.max(corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau * corr - max_tc)))
    return (log_sum_exp / tau) * dx

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    m = np.max(tc)
    return (m + np.log(np.sum(np.exp(tc - m))) / tau) * dx


def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    m = np.max(tc)
    exps = np.exp(tc - m)
    w = exps / exps.sum()
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:])
            g[lag:] += -wi * h[:n - lag]
        else:
            g[-lag:] += wi * (1.0 - h[:n + lag])
            g[:-lag] += -wi * h[-lag:]
    return g * dx
