            dx = 2.0 / n

def subgradient_optimize(h, n, n_iter=5000, lr_init=0.05):

        h_curr = h_curr - lr * grad

        h_curr = np.clip(h_curr, 0.0, 1.0)
        h_curr = repair_h(h_curr, n)

But I realize that recomputing `corr = np.correlate(h, 1-h, mode="full")` at each step is expensive (O(n²)). Can I avoid it?

When I change h[j] to v, the new correlation is:
new_corr[k] = corr[k] + (v - h[j]) * a_k

where a_k = (1-h[j+lag_k]) - h[j-lag_k] (computed BEFORE the change).

So I can update corr incrementally! After changing h[j]:
- Update corr: corr += (v - old_h[j]) * a
- But then for the next coordinate j', I need to recompute a' based on the new h.

The issue is that a depends on h, so I can't precompute it. But I can compute a on the fly for each coordinate, which is O(n) per coordinate (just indexing into h).

So the algorithm is:
1. Compute corr initially: O(n²)
2. For each coordinate j:
   a. Compute a_k = (1-h[j+lag_k]) - h[j-lag_k]: O(n)
   b. Compute b_k = corr[k] - h[j]*a_k: O(n)
   c. Find best_v: O(n) [vectorized numpy]
   d. Update corr: corr += (best_v - h[j]) * a: O(n)
   e. Set h[j] = best_v

Total per sweep: O(n²) for initial corr + O(n) per coordinate × n coordinates = O(n²). 

For n=350, each sweep is about 122,500 operations (dominated by the initial correlate). Multiple sweeps: maybe 10-50 sweeps in a few seconds.

This is very efficient! Let me implement it properly.

Actually wait, I need to be careful. When I compute a_k for coordinate j, I use the CURRENT values of h (which have been updated for coordinates 0, ..., j-1 in this sweep). So the a_k is always up-to-date. Good.

But there's a subtlety: when I update corr by (v - h[j]) * a, this is correct because:
new_corr[k] = Σ_i h'[i]·(1-h'[i+lag_k])
= Σ_i h[i]·(1-h[i+lag_k]) + (v-h[j])·(1-h[j+lag_k]) - (v-h[j])·h[j-lag_k]
= corr[k] + (v-h[j])·((1-h[j+lag_k]) - h[j-lag_k])
= corr[k] + (v-h[j])·a_k

Yes, this is correct.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return h

def coord_descent(h, n, n_sweeps=50, time_limit=60, adaptive_grid=True):
    """Coordinate descent with incremental correlation update."""
    t0 = time.time()
    h_curr = h.copy()
    dx = 2.0 / n
    N = 2 * n - 1
    lags = np.arange(N) - (n - 1)
    
    corr = np.correlate(h_curr, 1.0 - h_curr, mode="full").astype(float)
    current_max = np.max(corr)
    
    v_grid_coarse = np.linspace(0, 1, 100)
    v_grid_fine = np.linspace(0, 1, 500)
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            # Compute a_k = (1-h[j+lag]) - h[j-lag]
            j_plus = j + lags
            j_minus = j - lags
            
            h_jp = np.zeros(N)
            mp = (j_plus >= 0) & (j_plus < n)
            h_jp[mp] = h_curr[j_plus[mp]]
            
            h_jm = np.zeros(N)
            mm = (j_minus >= 0) & (j_minus < n)
            h_jm[mm] = h_curr[j_minus[mm]]
            
            a = (1.0 - h_jp) - h_jm
            b = corr - h_curr[j] * a
            
            # Minimize max_k (b[k] + v * a[k]) over v in [0,1]
            # Coarse grid
            vals = b[None, :] + v_grid_coarse[:, None] * a[None, :]
            max_vals = np.max(vals, axis=1)
            idx = np.argmin(max_vals)
            best_v = v_grid_coarse[idx]
            best_val = max_vals[idx]
            
            # Fine grid around best
            if 0.01 < best_v < 0.99:
                lo = max(0.0, best_v - 0.02)
                hi = min(1.0, best_v + 0.02)
                v_fine = np.linspace(lo, hi, 200)
                vals_f = b[None, :] + v_fine[:, None] * a[None, :]
                max_f = np.max(vals_f, axis=1)
                idx_f = np.argmin(max_f)
                best_v = v_fine[idx_f]
                best_val = max_f[idx_f]
            
            if best_val < current_max - 1e-10:
                # Update
                delta = best_v - h_curr[j]
                corr += delta * a
                h_curr[j] = best_v
                current_max = np.max(corr)
                improved = True
        if not improved:
            break
    
    return h_curr, current_max * dx

def binary_search_local(h_bin, n, time_limit=50):
    """Local search on binary patterns."""
    t0 = time.time()

    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    current_max = np.max(corr)
    N = 2*n - 1
    lags = np.arange(N) - (n-1)
    
    dx = 2.0 / n
    
    def safe_arr(arr, idx):
        out = np.zeros(len(idx))
        m = (idx >= 0) & (idx < n)
        out[m] = arr[idx[m]]
        return out
    
    while time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        # Try random swaps
        for _ in range(2000):
            a = zeros[np.random.randint(len(zeros))]
            b = ones[np.random.randint(len(ones))]
            
            delta = (safe_arr(h, b + lags) - safe_arr(h, a + lags) - 
                   safe_arr(h, a - lags) + safe_arr(h, b - lags))
            new_max = np.max(corr + delta)
            
            if new_max < current_max - 1e-12:
                h[a] = 1.0
                h[b] = 0.0

                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
        
        if not improved:
            break
    
    return h, current_max * dx
