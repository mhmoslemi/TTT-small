    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def _do_slsqp(h_start, n, target_sum, alpha, maxiter, ftol):

             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x

        return h_start

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add > 1e-15:
                    h[i] += add
                    diff -= add
            else:
                sub = min(-diff, h[i])
                if sub > 1e-15:
                    h[i] -= sub
                    diff += sub
    return h

def _wgrad(h, weights):
    n = len(h)
    G = np.zeros(n)
    w0 = weights[n-1]
    if w0 > 1e-15:
        G += w0 * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = weights[n-1+m]
        wn = weights[n-1-m]
        if wp > 1e-15:
            G[:n-m] += wp * (1.0 - h[m:])
            G[m:] -= wp * h[:n-m]
        if wn > 1e-15:
            G[m:] += wn * (1.0 - h[:n-m])
            G[:n-m] -= wn * h[m:]
    return G

def _sm_weights(corr, alpha):
    m = corr.max()
    e = np.exp(alpha * (corr - m))
    s = e.sum()
    return e / s if s > 0 else np.ones_like(corr) / len(corr)

def _tk_weights(corr, k):
    n = len(corr)
    idx = np.argsort(corr)[-k:]
    w = np.zeros(n)
    w[idx] = 1.0 / k
    return w

def _smooth_obj(h, alpha):
    corr = _corr(h)
    m = corr.max()
    if m < -500:
        return 0.0
    return float((m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h)))

def _slsqp(h0, n, ts, alpha, mi, ft):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_smooth_obj, h0, args=(alpha,), method='SLSQP',
                     bounds=b, constraints=c, options={'maxiter': mi, 'ftol': ft})
        return r.x
    except Exception:
        return h0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, ts):
    # Project onto {0<=x<=1, sum(x)=ts}
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > ts:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    d = ts - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a > 0:
                    h[i] += a; d -= a
            else:
                s = min(-d, h[i])
                if s > 0:
                    h[i] -= s; d += s
    return h

def _grad(h, w):
    n = len(h)
    G = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-15:
        G += w0 * (1.0 - 2.0*h)

        wp = w[n-1+m]
        wn = w[n-1-m]
