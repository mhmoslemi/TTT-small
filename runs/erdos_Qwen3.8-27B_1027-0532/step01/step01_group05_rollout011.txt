def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _objective(h, n, t, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(t * (corr - mc), -400.0, 400.0)
    ev = np.exp(shifted)
    se = np.sum(ev)
    obj = (mc + np.log(se) / t) * dx
    v = np.sum(h) - n * 0.5
    obj += lam * v * v
    return obj


def _gradient(h, n, t, lam):

                                'maxls': 50})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()


def _subgrad(h, n, n_iter, step0):
    dx = 2.0 / n
    h = h.copy()
    bh = h.copy()
    bc = compute_c5(h, n)
    step = step0
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            r = n - lag
            g[lag:] += 1.0 - h[:r]
            g[:r] -= h[lag:]
        else:
            l = -lag
            r = n - l
            g[:r] += 1.0 - h[l:]
            g[l:] -= h[:r]
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        h = project_onto_constraint(h, n)
        c = compute_c5(h, n)
        if c < bc:
            bc = c
            bh = h.copy()
        step *= 0.998
    return bh, bc


def _upsample(h, n_old, n_new):
    x = np.linspace(0, 2, n_old, endpoint=False)
    y = np.linspace(0, 2, n_new, endpoint=False)
    out = np.interp(y, x, h)
    return out

        if c < bc:
            bc = c
            bh = h_new.copy()
        if c < compute_c5(h, n) + 1e-9:
            h = h_new
    return bh, bc
