        T = T_start * (T_end / T_start) ** (phase / max(1, n_phases - 1))
        def f(x, T=T):
            return logsumexp_obj(x, T)
        def gfun(x):
            return x.sum() - tgt
        def gjac(x):
            return np.ones(n)
        try:
            r = minimize(f, h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                         options={'maxiter': 4000, 'ftol': 1e-14})
            h = project_feasible(r.x, tgt)
        except Exception:
            pass
    return h

def subgrad_multi_active(h0, tm=30, lr_init=0.35, beta=0.87):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)
    for it in range(80000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        max_val = corr.max()
        threshold = max_val - 0.015
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for ki in active:
            k = ki - (n - 1)
            w = max(0.0, (corr[ki] - threshold) / (max_val - threshold + 1e-10))
            a = j - k
            b = j + k
            m1 = (a >= 0) & (a < n)
            m2 = (b >= 0) & (b < n)
            grad[m1] += w * (1.0 - h[a[m1]])
            grad[m2] -= w * h[b[m2]]
        grad *= dx / len(active)
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9999
        if lr < 1e-8:
            break
    return best

def active_set_balance(h0, tm=30):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    for _ in range(15):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        max_val = corr.max()
        active = np.where(corr >= max_val - 0.008)[0]
        if len(active) < 2:
            break
        def f(x):
            c = np.correlate(x, 1.0 - x, 'full') * dx
            act = c[active]
            return np.max(act) + 0.2 * (np.max(act) - np.min(act))
        def gfun(x):
            return x.sum() - tgt
        def gjac(x):
            return np.ones(n)
        try:
            r = minimize(f, h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                         options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(r.x, tgt)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new.copy()
        except Exception:
            break
    return best

def cutting_plane_opt(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    cut_k_list = []
    for _ in range(10):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        max_val = corr.max()
        active = np.where(corr >= max_val - 0.01)[0]
        for ki in active:
            if ki not in cut_k_list:
                cut_k_list.append(ki)
        def f(x):
            c = np.correlate(x, 1.0 - x, 'full') * dx
            vals = c[cut_k_list]
            return np.max(vals) + 0.05 * (np.max(vals) - np.min(vals))
        def gfun(x):
            return x.sum() - tgt
        def gjac(x):
            return np.ones(n)
        try:
            r = minimize(f, h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                         options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(r.x, tgt)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new.copy()
        except Exception:
            break
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def make_structured_init(n, rng):
    r = rng.randint(8)
    if r == 0:
        n_bands = rng.randint(2, 6)
        edges = np.sort(rng.uniform(0, n, n_bands))
        h = np.zeros(n)
        h[:int(edges[0])] = rng.uniform(0.3, 1.0)
        for i in range(len(edges) - 1):
            h[int(edges[i]):int(edges[i + 1])] = rng.uniform(0.0, 0.7)
        h[int(edges[-1]):] = rng.uniform(0.3, 1.0)
    elif r == 1:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif r == 2:
        gap_size = rng.randint(max(1, n // 10), max(2, n // 3))
        gap_pos = rng.randint(0, max(1, n - gap_size))
        h = np.ones(n)
        h[gap_pos:gap_pos + gap_size] = 0.0
    elif r == 3:
        h = rng.random(n)
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        h = np.convolve(h, kernel, mode='same')
    elif r == 4:
        period = rng.randint(max(2, n // 4), max(3, n // 2))
        x = np.arange(n) % period
        h = 1.0 - 2.0 * np.abs(x - period / 2) / (period / 2)
        h = 0.5 + 0.3 * h
    elif r == 5:
        freqs = rng.randint(1, max(2, n // 4), size=rng.randint(2, 5))
        h = np.zeros(n)
        for f in freqs:
            h += np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        h = 0.5 + 0.3 * h / len(freqs)
    elif r == 6:
        bs = max(2, n // rng.randint(8, 20))
        n_blocks = n // bs + 1
        block_vals = rng.random(n_blocks)
        h = np.tile(block_vals, bs)[:n]
    else:
        h = (rng.random(n) > 0.5).astype(float)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def tl(frac=0.9):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    best = project_feasible(h_init, len(h_init) / 2.0)
    best_c5 = eval_c5(best)

    # Phase 0: Smooth max SLSQP with temperature annealing
    if tl(0.12) > 15:
        h1 = selpolish_smooth(best, tm=min(60, tl(0.10)), T_start=5.0, T_end=200.0, n_phases=4)
        c1 = eval_c5(h1)
        if c1 < best_c5:
            best, best_c5 = h1, c1

    # Phase 1: Multi-active subgradient
    if tl(0.20) > 10:
        h2 = subgrad_multi_active(best, tm=min(40, tl(0.18)), lr_init=0.4, beta=0.88)
        c2 = eval_c5(h2)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    # Phase 2: Cutting plane optimization
    if tl(0.28) > 15:
        h3 = cutting_plane_opt(best, tm=min(50, tl(0.26)))
        c3 = eval_c5(h3)
        if c3 < best_c5:
            best, best_c5 = h3, c3

    # Phase 3: Active set balancing
    if tl(0.33) > 10:
        h4 = active_set_balance(best, tm=min(30, tl(0.31)))
        c4 = eval_c5(h4)
        if c4 < best_c5:
            best, best_c5 = h4, c4

    # Phase 4: Local perturbation with smooth objective
    n_orig = len(best)
    for attempt in range(30):
        if tl(0.50) < 5:
            break
        scale = 0.015 + 0.005 * min(attempt, 8)
        h_p = best + rng.normal(0, scale, size=n_orig)
        h_p = project_feasible(h_p, n_orig / 2.0)
        h_opt = selpolish_smooth(h_p, tm=min(20, tl(0.50)), T_start=10.0, T_end=200.0, n_phases=3)
        c5_opt = eval_c5(h_opt)
        if c5_opt < best_c5:
            best, best_c5 = h_opt, c5_opt

    # Phase 5: Multi-resolution search
    n_targets = [100, 140, 168, 200, 240, 280, 320, 400, 500]
    for n_t in n_targets:
        if tl(0.80) < 15:
            break
        if abs(n_t - n_orig) < 15:
            continue
        h_res = resample_h(best, n_t)
        h_opt = selpolish_smooth(h_res, tm=min(40, tl(0.80)), T_start=5.0, T_end=300.0, n_phases=4)
        c5_opt = eval_c5(h_opt)
        if c5_opt < best_c5:
            best, best_c5 = h_opt, c5_opt
            if tl(0.80) > 10:
                h_opt2 = subgrad_multi_active(h_opt, tm=min(15, tl(0.80)), lr_init=0.3, beta=0.87)
                c5_opt2 = eval_c5(h_opt2)
                if c5_opt2 < best_c5:
                    best, best_c5 = h_opt2, c5_opt2
            if tl(0.80) > 10:
                h_opt3 = active_set_balance(h_opt2 if c5_opt2 < c5_opt else h_opt, tm=min(15, tl(0.80)))
                c5_opt3 = eval_c5(h_opt3)
                if c5_opt3 < best_c5:
                    best, best_c5 = h_opt3, c5_opt3
        if tl(0.80) > 20:
            h_struct = make_structured_init(n_t, rng)
            h_opt_s = selpolish_smooth(h_struct, tm=min(25, tl(0.80)), T_start=5.0, T_end=150.0, n_phases=3)
            c5_s = eval_c5(h_opt_s)
            if c5_s < best_c5:
                best, best_c5 = h_opt_s, c5_s

    # Phase 6: Final intensive polishing
    if tl(0.97) > 30:
        h_f = selpolish_smooth(best, tm=min(90, tl(0.95)), T_start=20.0, T_end=1500.0, n_phases=5)
        cf = eval_c5(h_f)
        if cf < best_c5:
            best, best_c5 = h_f, cf

    if tl(0.97) > 15:
        h_f2 = cutting_plane_opt(best, tm=min(30, tl(0.96)))
        cf2 = eval_c5(h_f2)
        if cf2 < best_c5:
            best, best_c5 = h_f2, cf2

    if tl(0.98) > 10:
        h_f3 = active_set_balance(best, tm=min(20, tl(0.975)))
        cf3 = eval_c5(h_f3)
        if cf3 < best_c5:
            best, best_c5 = h_f3, cf3

    if tl(0.99) > 15:
        h_f4 = selpolish_smooth(best, tm=min(20, tl(0.985)), T_start=200.0, T_end=5000.0, n_phases=3)
        cf4 = eval_c5(h_f4)
        if cf4 < best_c5:
            best, best_c5 = h_f4, cf4

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
