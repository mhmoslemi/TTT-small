import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            grad[:n - r] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient_descent(h_init, eps, lr, n_steps, target_sum):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        grad = smooth_gradient(h, eps)
        h_new = project_onto_constraint(h - lr * grad, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def binary_sa(h_start, n_swaps, t_start, t_end, rng):
    """Simulated annealing on binary solutions."""
    n = len(h_start)
    h = np.round(h_start)
    h = np.clip(h, 0, 1)
    # Ensure exactly n/2 ones
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    # Take the n/2 closest to 1, rest closest to 0
    h = np.zeros(n)
    h[order[n//2:]] = 1.0
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    t = t_start
    for i in range(n_swaps):
        t = t_start * (t_end / t_start) ** (i / max(n_swaps - 1, 1))
        # Pick random one and zero to swap
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        if oi == zi:
            continue
        # Apply swap
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            # Accept
            current_c5 = new_c5
            ones[ones == oi] = zi
            zeros[zeros == zi] = oi
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reject
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c5

        h_opt, c5_opt = slsqp_optimize(h_polish, 0.00001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
