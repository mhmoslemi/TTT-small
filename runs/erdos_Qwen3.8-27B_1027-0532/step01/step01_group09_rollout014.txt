def project_feasible(x, target):
    """Project x onto {0<=v<=1, sum(v)=target}"""
    n = len(x)

        if s > target:

        else:

    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def smooth_max_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    tc = t * corr
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    p = exp_tc / exp_tc.sum()
    obj = (tc_max + np.log(exp_tc.sum())) / t
    k_vals = np.arange(2 * n - 1) - (n - 1)
    g = np.zeros(n)
    idx1 = np.arange(2 * n - 1)[:, None]  # will compute per i
    for i in range(n):
        a_vals = i - k_vals
        b_vals = i + k_vals
        m1 = (a_vals >= 0) & (a_vals < n)
        m2 = (b_vals >= 0) & (b_vals < n)
        if m1.any():
            g[i] += np.dot(p[m1], 1.0 - h[a_vals[m1]])
        if m2.any():
            g[i] -= np.dot(p[m2], h[b_vals[m2]])
    g *= dx
    return obj, g

def pgd_smooth(h0, t_lo, t_hi, max_iters, t_deadline, lr_init=0.04, momentum=0.0):
    n = len(h0)
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    lr = lr_init
    v = np.zeros(n)
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        t_curr = t_lo * (t_hi / t_lo) ** frac
        obj, g = smooth_max_obj_grad(h, t_curr)
        v = momentum * v + (1.0 - momentum) * g
        alpha = lr * (1.0 - 0.5 * frac)
        h_trial = project_feasible(h - alpha * v, target)
        obj_trial, _ = smooth_max_obj_grad(h_trial, t_curr)
        if obj_trial > obj:
            h_trial = project_feasible(h - alpha * 0.25 * v, target)
            obj_trial, _ = smooth_max_obj_grad(h_trial, t_curr)
        h = h_trial
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def coord_descent(h0, max_sweeps, t_deadline):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        for i in range(n):
            a_vals = i - k_vals
            b_vals = i + k_vals
            m1 = (a_vals >= 0) & (a_vals < n)
            m2 = (b_vals >= 0) & (b_vals < n)
            coeff = np.zeros(2 * n - 1)
            if m1.any():
                coeff[m1] = dx * (1.0 - h[a_vals[m1]])
            if m2.any():
                coeff[m2] -= dx * h[b_vals[m2]]
            h_save = h[i]
            const = corr - coeff * h_save
            lo_s, hi_s = 0.0, 1.0
            for _ in range(40):
                mid_s = 0.5 * (lo_s + hi_s)
                vals = coeff * mid_s + const
                max_val = vals.max()
                active = np.abs(vals - max_val) < 1e-12
                if not active.any():
                    active = (vals == max_val)
                slope = coeff[active].mean()
                if slope > 0:
                    hi_s = mid_s
                else:
                    lo_s = mid_s
            h[i] = 0.5 * (lo_s + hi_s)
            h = project_feasible(h, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    t0 = time.time()
    def f(x):
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()
    def gfun(x):
        return x.sum() - target
    def gjac(x):
        return np.ones(n)
    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 8000, 'ftol': 1e-15})
        h1 = project_feasible(r.x, target)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(h_new, n_dst / 2.0)

def diverse_init(n, rng):
    r = rng.randint(14)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if r == 0:
        freq = float(rng.randint(1, 10))
        h = 0.5 + 0.45 * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    elif r == 1:
        h = (rng.random(n) > 0.5).astype(float)
    elif r == 2:
        nb = rng.randint(3, 10)
        h = np.concatenate([np.full(max(1, int(n / nb)), rng.uniform(0, 1)) for _ in range(nb)])[:n]
    elif r == 3:
        freq = float(rng.randint(1, 6))
        h = 0.5 - 0.4 * np.abs(2 * ((freq * x) % 1.0) - 1.0)
    elif r == 4:
        h = rng.choice([0.0, 1.0], size=n)
    elif r == 5:
        noise = rng.random(n)
        k = max(1, n // 15)
        h = np.convolve(noise, np.ones(k) / k, mode='same')
    elif r == 6:
        h = 0.5
        for _ in range(rng.randint(2, 5)):
            f = float(rng.randint(1, 8))
            h += rng.uniform(-0.12, 0.12) * np.sin(np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif r == 7:
        bl = rng.randint(3, 12)
        h = np.tile(rng.uniform(0, 1, bl), n // bl + 1)[:n]
    elif r == 8:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n * rng.randint(2, 8))
    elif r == 9:
        h = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            c = rng.randint(0, n)
            w = rng.randint(5, max(6, n // 3))
            h[max(0, c - w):min(n, c + w)] += rng.uniform(0.3, 0.7)
        mx = h.max()
        if mx > 0:
            h /= mx
    elif r == 10:
        freq = float(rng.randint(1, 6))
        h = 0.5 - 0.4 * (2 * np.abs(((freq * x) % 1.0) - 0.5) - 0.5)
    elif r == 11:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * x * rng.uniform(2, 5)) * np.cos(2 * np.pi * x * rng.uniform(1, 3))
    elif r == 12:
        h = np.zeros(n)
        period = rng.randint(4, 12)
        pattern = rng.uniform(0, 1, period)
        h = np.tile(pattern, n // period + 1)[:n]
    else:
        h = rng.random(n)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def tl(frac=0.95):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    n_orig = len(h_init)
    best = h_init.astype(float).copy()
    best_c5 = eval_c5(best)

    # Phase 1: Smooth max PGD with low-to-high temperature
    if tl(0.95) > 30:
        h1 = pgd_smooth(best, 20, 300, 3000, time.time() + min(50, tl(0.2)), lr_init=0.04, momentum=0.3)
        c1 = eval_c5(h1)
        if c1 < best_c5:
            best, best_c5 = h1, c1

    # Phase 2: Coordinate descent
    if tl(0.95) > 25:
        h2 = coord_descent(best, 80, time.time() + min(40, tl(0.25)))
        c2 = eval_c5(h2)
        if c2 < best_c5:
            best, best_c5 = h2, c2

    # Phase 3: SLSQP polish
    if tl(0.95) > 20:
        h3 = selpolish(best, tm=min(50, tl(0.25)))
        c3 = eval_c5(h3)
        if c3 < best_c5:
            best, best_c5 = h3, c3

    # Phase 4: High temperature PGD
    if tl(0.95) > 20:
        h4 = pgd_smooth(best, 100, 2000, 2000, time.time() + min(40, tl(0.25)), lr_init=0.02, momentum=0.5)
        c4 = eval_c5(h4)
        if c4 < best_c5:
            best, best_c5 = h4, c4

    # Phase 5: Resolution adaptation
    n_targets = []
    for mult in [0.7, 0.85, 1.0, 1.2, 1.4, 1.7, 2.0, 2.5, 3.0]:
        nt = int(n_orig * mult)
        if 50 <= nt <= 700 and nt not in n_targets:
            n_targets.append(nt)

    for nt in n_targets:
        if tl(0.85) < 20:
            break
        if nt == n_orig:
            continue
        h_res = resample_h(best, nt)
        h_new = pgd_smooth(h_res, 30, 400, 2000, time.time() + min(30, tl(0.5)), lr_init=0.03, momentum=0.3)
        c5n = eval_c5(h_new)
        if c5n < best_c5:
            best, best_c5 = h_new, c5n
        if tl(0.85) > 15:
            h_pol = selpolish(h_new, tm=min(25, tl(0.4)))
            c5p = eval_c5(h_pol)
            if c5p < best_c5:
                best, best_c5 = h_pol, c5p

    # Phase 6: Diverse restarts
    n_restart = min(50, max(8, int(tl(0.5) // 12)))
    for _ in range(n_restart):
        if tl(0.65) < 10:
            break
        h_r = diverse_init(n_orig, rng)
        if rng.random() < 0.3:
            h_r = project_feasible(1.0 - best + rng.normal(0, 0.05, n_orig), n_orig / 2.0)
        elif rng.random() < 0.5:
            h_r = project_feasible(best + rng.normal(0, 0.06, n_orig), n_orig / 2.0)
        h_opt = pgd_smooth(h_r, 30, 300, 1500, time.time() + min(12, tl(0.5)), lr_init=0.04, momentum=0.3)
        c5o = eval_c5(h_opt)
        if c5o < best_c5:
            best, best_c5 = h_opt, c5o
            if tl(0.65) > 15:
                h_pol = coord_descent(best, 40, time.time() + min(15, tl(0.5)))
                c5p = eval_c5(h_pol)
                if c5p < best_c5:
                    best, best_c5 = h_pol, c5p

    # Phase 7: Final intensive polish
    if tl(0.95) > 20:
        h_f = pgd_smooth(best, 200, 5000, 3000, time.time() + min(70, tl(0.55)), lr_init=0.01, momentum=0.6)
        cf = eval_c5(h_f)
        if cf < best_c5:
            best, best_c5 = h_f, cf

    if tl(0.95) > 15:
        h_f2 = coord_descent(best, 60, time.time() + min(30, tl(0.65)))
        cf2 = eval_c5(h_f2)
        if cf2 < best_c5:
            best, best_c5 = h_f2, cf2

    if tl(0.95) > 15:
        h_f3 = selpolish(best, tm=min(40, tl(0.75)))
        cf3 = eval_c5(h_f3)
        if cf3 < best_c5:
            best, best_c5 = h_f3, cf3

    if tl(0.95) > 10:
        h_f4 = pgd_smooth(best, 500, 10000, 2000, time.time() + min(25, tl(0.85)), lr_init=0.005, momentum=0.7)
        cf4 = eval_c5(h_f4)
        if cf4 < best_c5:
            best, best_c5 = h_f4, cf4

    if tl(0.95) > 8:
        h_f5 = selpolish(best, tm=min(15, tl(0.95)))
        cf5 = eval_c5(h_f5)
        if cf5 < best_c5:
            best, best_c5 = h_f5, cf5

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
