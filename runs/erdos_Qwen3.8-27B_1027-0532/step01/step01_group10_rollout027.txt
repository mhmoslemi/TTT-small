    n_work = max(best_n, 512)
    dx_work = 2.0 / n_work

    if check_time(250):
        starts = [interpolate_h(best_h, best_n, n_work)]
        for freq in [3, 5, 7, 11]:
            starts.append(generate_sinusoidal(n_work, freq, 0.4))
        starts.append(generate_block_pattern(n_work, 8, rng))
        starts.append(generate_xor_pattern(n_work, 16))
        starts.append(generate_xor_pattern(n_work, 24))

        for h_start in starts:
            if not check_time(120):
                break
            h_curr = h_start.copy()
            for p in [16, 32]:
                if not check_time(80):
                    break
                h_opt = run_slsqp_power(h_curr, n_work, dx_work, p, maxiter=250)
                c = c5_value(h_opt, dx_work)
                h_curr = h_opt
                c_check = c5_value(interpolate_h(h_opt, n_work, best_n), 2.0 / best_n)
                if c_check < best_c5:
                    best_h = interpolate_h(h_opt, n_work, best_n)
                    best_c5 = c_check
                if c < best_c5:
                    best_h = h_opt
                    best_c5 = c
                    best_n = n_work

            if not check_time(40):
                break
            h_opt = run_nm(h_curr, n_work, dx_work, 50.0, maxiter=3000)
            c = c5_value(h_opt, dx_work)
            c_check = c5_value(interpolate_h(h_opt, n_work, best_n), 2.0 / best_n)
            if c_check < best_c5:
                best_h = interpolate_h(h_opt, n_work, best_n)
                best_c5 = c_check
            if c < best_c5:
                best_h = h_opt
                best_c5 = c
                best_n = n_work

    # Phase 5: Block coordinate descent on best
    if check_time(100):
        n_bcd = best_n
        dx_bcd = 2.0 / n_bcd
        n_blocks = 16
        max_rounds = 3
        h_bcd, c_bcd = block_coordinate_descent(best_h, n_bcd, dx_bcd, n_blocks, max_rounds, 20.0)
        if c_bcd < best_c5:
            best_h = h_bcd
            best_c5 = c_bcd

    # Phase 6: Final aggressive refinement
    n_final = best_n
    dx_final = 2.0 / n_final

    if check_time(60):
        for p in [64, 128, 256]:
            if not check_time(30):
                break
            h_opt = run_slsqp_power(best_h, n_final, dx_final, p, maxiter=500)
            c = c5_value(h_opt, dx_final)
            if c < best_c5:
                best_h = h_opt
                best_c5 = c

    if check_time(40):
        for beta in [5.0, 3.0, 2.0]:
            if not check_time(25):
                break
            h_opt = run_slsqp_log(best_h, n_final, dx_final, beta, maxiter=500)
            c = c5_value(h_opt, dx_final)
            if c < best_c5:
                best_h = h_opt
                best_c5 = c

    if check_time(30):
        for k in [1, 2]:
            if not check_time(20):
                break
            h_opt = run_slsqp_topk(best_h, n_final, dx_final, k, maxiter=500)
            c = c5_value(h_opt, dx_final)
            if c < best_c5:
                best_h = h_opt
                best_c5 = c

    if check_time(80):
        h_opt = run_nm(best_h, n_final, dx_final, 200.0, maxiter=15000)
        c = c5_value(h_opt, dx_final)
        if c < best_c5:
            best_h = h_opt
            best_c5 = c

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
