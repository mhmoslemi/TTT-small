from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_set_gradient(h, K=5):
    """Compute gradient using only the top-K active lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top K lags
    top_k = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for m in top_k:
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return grad * dx / K

Let me also think about whether there's a smarter approach to the non-convexity.

**Idea: Bilinear decomposition**

f_k(h) = sum_i h[i] - sum_i h[i]*h[i+k]

Let's write this as: f_k(h) = sum_i h[i](1-h[i+k])

For a fixed lag k, this is bilinear in (h[0:n-k], h[k:n]). If we fix h[0:n-k], then f_k is linear in h[k:n]. If we fix h[k:n], it's linear in h[0:n-k].

This suggests an alternating minimization:
- Fix "even" indices, optimize "odd" indices (linear subproblem)
- Fix "odd" indices, optimize "even" indices (linear subproblem)
- Repeat

This is like the "bilinear alternating minimization" (ALM) approach.

But we're minimizing the MAX over k, so it's not straightforward.

**Alternative: Treat it as a game**

min_h max_k f_k(h)

This is a zero-sum game where "nature" chooses k and we choose h. By von Neumann's minimax theorem (if we relax to a continuous version), we have:
min_h max_k f_k(h) = max_{p ∈ simplex} min_h sum_k p_k f_k(h)

The dual is:
max_{p} min_{h: 0≤h≤1, sum(h)=n/2} sum_k p_k f_k(h)

For fixed p, the inner problem is:
min_h sum_k p_k [sum_i h[i] - sum_i h[i]*h[i+k]]
= min_h sum_i h[i] * (sum_k p_k) - sum_{i,k} p_k h[i]*h[i+k]
= (sum_k p_k) * sum_i h[i] - sum_{i,k} p_k h[i]*h[i+k]
= 1 * (n/2) - sum_{i,k} p_k h[i]*h[i+k]  [since sum(h) = n/2 is fixed]
= n/2 - sum_{i,k} p_k h[i]*h[i+k]

So we want to maximize sum_{i,k} p_k h[i]*h[i+k] subject to 0≤h≤1, sum(h)=n/2.

This is a quadratic maximization! And it's a quadratic form: h' Q h where Q depends on p.

If Q is positive semi-definite, this is a convex maximization (maximize convex = hard, at extreme points → binary).
If Q is negative semi-definite, this is a concave maximization (easy, has unique solution).

The matrix Q has Q[i,j] = sum_k p_k * (indicator that j-i = k or i-j = k) / (2n-1) or something like that.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_set_gradient(h, K=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for m in top_k:
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += (-h[j2_lo + lag:j2_hi + lag + 1])
    return grad * dx / K

def soft_max_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    weights = np.exp(tau_corr - max_tc)
    s = np.sum(weights)
    if s < 1e-300:
        return np.zeros(n)
    weights /= s
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
