    if remaining_frac() > 0.25:
        temps2 = [3, 2, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05]
        for t in temps2:
            if remaining_frac() < 0.15:
                break
            time_left = (remaining_frac() - 0.15) * budget_s / max(1, len(temps2))
            n_iter = max(200, int(min(1000, time_left * 3.5)))
            h_opt, c5_opt, h_last = optimize_temperature(best_h, n, t, n_iter, 0.02, top_k=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            if compute_c5(h_last, n) < compute_c5(best_h, n) + 0.001:
                best_h = h_last.copy()

    # Phase 4: Structured perturbations + local refinement
    if remaining_frac() > 0.15:
        for attempt in range(8):
            if remaining_frac() < 0.05:
                break
            if attempt % 4 == 0:
                freqs = np.fft.rfftfreq(n)
                noise = np.fft.irfft(rng.randn(n // 2 + 1) * np.exp(-3.0 * freqs * 2) * 0.25)
                h_pert = best_h + noise
            elif attempt % 4 == 1:
                block_size = max(1, n // 14)
                blocks = n // block_size
                noise = np.repeat(rng.randn(blocks) * 0.08, block_size)[:n]
                h_pert = best_h + noise
            elif attempt % 4 == 2:
                h_pert = best_h + rng.randn(n) * 0.04
            else:
                corr = np.correlate(best_h, 1.0 - best_h, mode="full")
                top_lags = np.argsort(corr)[-6:]
                h_pert = best_h.copy()
                for lag_idx in top_lags:
                    lag = lag_idx - (n - 1)
                    if lag >= 0:
                        h_pert[lag:] += rng.randn(n - lag) * 0.04
                    else:
                        l = -lag
                        h_pert[:n - l] += rng.randn(n - l) * 0.04

            h_pert = project_onto_constraint(h_pert, n)

            for t in [5, 2, 1.0, 0.5, 0.2]:
                if remaining_frac() < 0.05:
                    break
                h_opt, c5_opt, h_last = optimize_temperature(h_pert, n, t, 400, 0.02, top_k=200)
                h_pert = h_last
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 5: Final subgradient polish
    if remaining_frac() > 0.03:
        n_iter_final = int(min(1500, remaining_frac() * budget_s * 4))
        h_pol, c5_pol = subgradient_polish(best_h, n, n_iter=n_iter_final, step_init=0.015)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 6: Try even finer grid if time permits
    if remaining_frac() > 0.10:
        n_fine = 840
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_new, best_h)
        h_fine = project_onto_constraint(h_fine, n_fine)
        c5_fine = compute_c5(h_fine, n_fine)

        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

        for t in [5, 2, 1, 0.5, 0.2]:
            if remaining_frac() < 0.03:
                break
            n_iter = max(100, int(min(500, remaining_frac() * budget_s * 1.5)))
            h_opt, c5_opt, _ = optimize_temperature(best_h, n_fine, t, n_iter, 0.02, top_k=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n = n_fine
                x_new = x_fine

        if remaining_frac() > 0.03:
            h_pol, c5_pol = subgradient_polish(best_h, n, n_iter=500, step_init=0.01)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final verification
    best_h = project_onto_constraint(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
