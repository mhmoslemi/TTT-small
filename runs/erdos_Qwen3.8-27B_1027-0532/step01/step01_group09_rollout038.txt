import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_full_corr(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def weighted_grad(h, weights):
    """Compute weighted gradient of correlation."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = len(weights)  # 2n-1
    for m in range(n_corr):
        if weights[m] < 1e-12:
            continue
        d = m - (n - 1)
        # First term: (1-h[i-d]) for valid i-d
        if d >= 0:
            # i-d >= 0 means i >= d, i-d < n means i < n+d (always true since i<n)
            valid_i = np.arange(d, n)
            grad[valid_i] += weights[m] * (1.0 - h[valid_i - d])
            # Second term: -h[i+d] for valid i+d
            valid_i2 = np.arange(0, n - d)
            grad[valid_i2] -= weights[m] * h[valid_i2 + d]
        else:
            ad = -d
            # i-d >= 0 means i >= ad, i-d < n means i < n+ad (always true)
            valid_i = np.arange(ad, n)
            grad[valid_i] += weights[m] * (1.0 - h[valid_i - d])
            # i+d >= 0 means i >= ad, i+d < n means i < n-ad
            valid_i2 = np.arange(0, n - ad)
            grad[valid_i2] -= weights[m] * h[valid_i2 + d]
    grad *= dx
    return grad

    """Fast weighted gradient using only significant lags."""
    n = len(h)
    grad = np.zeros(n)
    thresh = 1e-10
    for m in range(len(p)):
        if p[m] < thresh:
            continue
        d = m - (n - 1)
        pm = p[m]
        if d >= 0:

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def weighted_grad_fast(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    thresh = 1e-10
    for m in range(len(p)):

def logsumexp_opt(h0, tm, t_start, T_init=0.5, T_final=0.005, max_steps=10000):

    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    
    v = np.zeros(n)  # Nesterov momentum
    lr = 0.3
    beta = 0.9
    n_active = max(5, len(h) // 5)  # only consider top lags for gradient
    
    for it in range(max_steps):
        if time.time() - t_start > tm:
            break
        
        T = T_init * (T_final / T_init) ** (it / max_steps)
        T = max(T, T_final)
        
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        corr_max = corr.max()
        
        # Softmax weights (numerically stable)
        shifted = (corr - corr_max) / T
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(corr)
            p[np.argmax(corr)] = 1.0
        else:
            p /= p_sum
        
        # Weighted gradient
        grad = weighted_grad_fast(h, p, dx)
        
        # Nesterov update
        v = beta * v + (1 - beta) * grad
        h_new = project_feasible(h - lr * v, target)

            best_c5 = c5
            best = h_new.copy()

        lr *= 0.9998

        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

        return x.sum() - tgt

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def weighted_grad_fast(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(p)):
        if p[m] < 1e-10:
            continue
        d = m - (n - 1)
        pm = p[m]
        if d >= 0:
            if d < n:
                grad[d:] += pm * (1.0 - h[:n - d])
                grad[:n - d] -= pm * h[d:]
        else:
            ad = -d
            if ad < n:
                grad[:n - ad] += pm * (1.0 - h[ad:])
                grad[ad:] -= pm * h[:n - ad]
    return grad * dx

def logsumexp_opt(h0, tm, T_init=0.5, T_final=0.003, max_steps=10000, lr_init=0.3):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    v = np.zeros(n)
    lr = lr_init
    beta = 0.9
    t0 = time.time()
    stall = 0
    stall_thresh = 300
    reheat_count = 0
    for it in range(max_steps):
        if time.time() - t0 > tm:
            break
        frac = it / max_steps
        T = T_init * (T_final / T_init) ** frac
        if reheat_count > 0:
            T = max(T, 0.1)
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        corr_max = corr.max()
        shifted = (corr - corr_max) / T
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(corr)
            p[int(np.argmax(corr))] = 1.0
        else:
            p /= p_sum
        grad = weighted_grad_fast(h, p, dx)
        v = beta * v + (1.0 - beta) * grad
        h_new = project_feasible(h - lr * v, target)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_thresh and reheat_count < 3:
            lr = lr_init * 0.8
            v *= 0.1
            stall = 0
            reheat_count += 1
        h = h_new
        lr *= 0.99985
        if lr < 0.003:
            break
    return best

def selpolish(h0, tm, maxiter=5000):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()

    def f(x):
        if time.time() - t0 > tm * 0.9:
            raise Exception("tl")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

    def gfun(x):
        return x.sum() - tgt

    def gjac(x):
        return np.ones(n)

    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def resample_h(h_src, n_dst):
    n_src = len(h_src)

    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)
