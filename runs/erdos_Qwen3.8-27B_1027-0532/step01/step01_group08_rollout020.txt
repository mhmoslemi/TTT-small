```python
def adam_pgd_optimize(h0, n, target_sum, deadline, tau_start=20, tau_end=500, max_iter=5000):
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    t = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        # Anneal tau
        frac = min(t / max_iter, 1.0)
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Compute gradient
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        tau_corr = tau * corr
        max_tc = np.max(tau_corr)
        weights = np.exp(tau_corr - max_tc)
        weights /= np.sum(weights)
        
        # Compute gradient
        g = np.zeros(n)
        for m_idx in range(len(weights)):
            if weights[m_idx] < 1e-14:
                continue
            lag = m_idx - (n - 1)
            w = weights[m_idx]
            if lag >= 0:
                g[:n-lag] += w * (1.0 - h[lag:])
                g[lag:] += -w * h[:n-lag]
            else:
                g[:n+lag] += w * (1.0 - h[-lag:])
                g[-lag:] += -w * h[:n+lag]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_new = project_onto_constraint(h_new, target_sum)
        
        # Check improvement
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        
        # Adaptive lr
        if it % 500 == 0 and it > 0:
            lr *= 0.85
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, weights, n):
    """Compute gradient given weights for each lag."""
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        if lag >= 0:
            g[:n-lag] += w * (1.0 - h[lag:])
            g[lag:] += -w * h[:n-lag]
        else:
            g[:n+lag] += w * (1.0 - h[-lag:])
            g[-lag:] += -w * h[:n+lag]
    return g * dx

def adam_topk_optimize(h0, n, deadline, max_iter=10000):
    h = h0.copy()
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    base_lr = 0.02
    t_adam = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_adam += 1
        frac = t_adam / max_iter
        
        # Anneal parameters
        tau = 10.0 + 200.0 * frac  # 10 -> 210
        top_k = max(5, int(2 * (n - 1) - (2 * (n - 1) - 10) * frac))  # all -> 10
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        
        # Compute weights (top-k or softmax)
        sorted_idx = np.argsort(corr)[::-1]
        weights = np.zeros(2*n-1)
        for i in range(top_k):
            weights[sorted_idx[i]] = 1.0 / top_k
        
        # Optionally blend with softmax
        tau_corr = tau * corr
        max_tc = np.max(tau_corr)
        sw = np.exp(tau_corr - max_tc)
        sw /= np.sum(sw)

        # Adam update

        m_hat = m_buf / (1 - beta1**t_adam)

        lr = base_lr / (1 + 0.1 * t_adam / 100)

        h_new = project_onto_constraint(h_new, target)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            # Step back if too much degradation
            h = 0.5 * h + 0.5 * h_new
            h = project_onto_constraint(h, target)
            # Reduce lr
            base_lr *= 0.98
    
    return best_h, best_c5

    lo = -2.0

    dx = 2.0 / n

    wp = w[n-1:2*n-1].copy()  # length n

    wn = np.zeros(n)
