            for delta in [0.001, 0.005, 0.01, 0.02, 0.05]:

                excess = delta

    best_c5 = compute_c5(h)

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        y = h + momentum * (h - h_prev)
        y_proj = project_onto_constraint(y, target_sum)
        g = smooth_gradient(y_proj, eps)
        lr = lr0 / (1.0 + step / 5000.0)
        h_new = project_onto_constraint(y_proj - lr * g, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h_prev = h
        h = h_new

    return best_h, best_c5

def binary_round(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 90
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    h0 = project_onto_constraint(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def remaining_time():
        return budget_s - (time.time() - start_time)

    phase1_limit = start_time + min(250, budget_s * 0.25)
    for eps in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() > phase1_limit:
            break
        h, c5 = slsqp_run(best_h, eps, 400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    phase2_limit = start_time + min(500, budget_s * 0.5)
    for eps in [0.001, 0.0005, 0.0002]:
        if time.time() > phase2_limit:
            break
        h, c5 = nesterov_phase(best_h, eps, 300000, 0.9, 0.05, start_time, phase2_limit)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    phase3_limit = start_time + min(700, budget_s * 0.65)
    for eps in [0.001, 0.0005, 0.0001]:
        if time.time() > phase3_limit:
            break
        h, c5 = slsqp_run(best_h, eps, 800)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    phase4_limit = start_time + min(900, budget_s * 0.8)
    h, c5 = nesterov_phase(best_h, 0.0001, 500000, 0.95, 0.01, start_time, phase4_limit)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    if remaining_time() > 80:
        hb = binary_round(best_h)
        hb = project_onto_constraint(hb, n0 / 2.0)
        for eps in [0.01, 0.001, 0.0001]:
            if remaining_time() < 30:
                break
            h, c5 = slsqp_run(hb, eps, 500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                hb = h.copy()

    if remaining_time() > 150:
        n1 = 2 * n0
        h_up = np.repeat(best_h, 2).astype(float)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        up_limit = start_time + budget_s - 60
        for eps in [0.01, 0.005, 0.001, 0.0005]:
            if time.time() > up_limit:
                break
            h, c5 = slsqp_run(h_up, eps, 300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                h_up = h.copy()
        if remaining_time() > 60:
            h, c5 = nesterov_phase(best_h, 0.0002, 200000, 0.9, 0.03, start_time, start_time + remaining_time() - 20)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    if remaining_time() > 60:
        restart_limit = start_time + budget_s - 15
        for r in range(8):
            if time.time() > restart_limit:
                break
            if r == 0:
                h_pert = best_h + 0.005 * rng.standard_normal(n0)
            elif r == 1:
                h_pert = best_h + 0.01 * (rng.random(n0) - 0.5)
            elif r == 2:
                h_pert = best_h.copy()
                nflip = max(2, n0 // 10)
                idx = rng.choice(n0, size=nflip, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]
            elif r == 3:
                h_pert = binary_round(best_h + 0.1 * rng.standard_normal(n0))
            elif r == 4:
                h_pert = 0.8 * best_h + 0.2 * binary_round(best_h + 0.2 * rng.standard_normal(n0))
            else:
                h_pert = best_h + 0.02 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, len(h_pert) / 2.0)
            nn = len(h_pert)
            h, c5 = slsqp_run(h_pert, 0.001, 500)
            if c5 < best_c5:
                h, c5 = slsqp_run(h, 0.0001, 500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    if remaining_time() > 30:
        h, c5 = slsqp_run(best_h, 0.00005, 2000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        h, c5 = nesterov_phase(best_h, 0.00005, 200000, 0.98, 0.005, start_time, start_time + remaining_time() - 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
