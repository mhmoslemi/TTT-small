import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def subgradient_h(h, k, n):
    """Subgradient of corr_k with respect to h."""
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += 1.0 - h[a[m1]]
    grad[m2] -= h[b[m2]]
    return grad * dx

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()
    
    def obj(x):
        if time.time() - t0 > tm * 0.85:
            raise Exception("tl")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()
    
    def cons_fun(x):
        return x.sum() - tgt
    
    def cons_jac(x):
        return np.ones(n)
    
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                     options={'maxiter': 20000, 'ftol': 1e-16, 'maxls': 50})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def block_cd(h0, tm=60, block_size=10, max_passes=500):
    """Block coordinate descent with SLSQP on each block."""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt).copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    
    n_blocks = (n + block_size - 1) // block_size
    
    for pass_i in range(max_passes):
        if time.time() - t0 > tm:
            break
        for bi in range(n_blocks):
            if time.time() - t0 > tm:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            B = end - start
            # Target sum for this block
            other_sum = h.sum() - h[start:end].sum()
            target_block = tgt - other_sum
            
            if target_block < 0 or target_block > B:
                continue
            
            h_block = h[start:end].copy()
            
            def block_obj(xb):
                h_test = h.copy()
                h_test[start:end] = xb
                return (np.correlate(h_test, 1.0 - h_test, 'full') * dx).max()
            
            def block_con_fun(xb):
                return xb.sum() - target_block
            
            def block_con_jac(xb):
                return np.ones(B)
            
            try:
                r = minimize(block_obj, h_block, method='SLSQP',
                           bounds=[(0.0, 1.0)] * B,
                           constraints=[{'type': 'eq', 'fun': block_con_fun, 'jac': block_con_jac}],
                           options={'maxiter': 200, 'ftol': 1e-12})
                if r.success or r.fun < block_obj(h_block):
                    h[start:end] = np.clip(r.x, 0.0, 1.0)
            except Exception:
                pass
        
        # Project to ensure feasibility after full pass
        h = project_feasible(h, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def nest_subgrad(h0, tm=30):
    """Nesterov-accelerated subgradient method."""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt).copy()
    y = h.copy()  # Nesterov's auxiliary variable
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    j = np.arange(n)
    
    T_max = 20000
    for t in range(1, T_max + 1):
        if time.time() - t0 > tm:
            break
        # Nesterov momentum
        alpha = 2.0 / (t + 1.0)
        y_new = h + (alpha - 1.0) * (h - y)
        y = y_new
        
        # Evaluate at y
        corr = np.correlate(y, 1.0 - y, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        
        # Subgradient
        a = j - k
        b = j + k
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += 1.0 - y[a[m1]]
        grad[m2] -= y[b[m2]]
        grad *= dx
        
        # Learning rate
        lr = 1.0 / np.sqrt(t + 1.0) * 0.5
        
        h = project_feasible(y - lr * grad, tgt)
        
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def softmax_opt(h0, tm=60, tau_start=0.1, tau_end=0.01):
    """Optimize using softmax-smoothed objective with L-BFGS-B."""
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0, tgt).copy()
    t0 = time.time()
    
    # Use penalty method: minimize soft_c5 + penalty*(sum(h)-tgt)^2
    # with L-BFGS-B (handles bounds natively)
    
    def soft_obj(x, tau, lam):
        corr = np.correlate(x, 1.0 - x, 'full') * dx
        m = corr.max()
        soft = m + (1.0 / tau) * np.log(np.sum(np.exp(tau * (corr - m))))
        pen = lam * (x.sum() - tgt) ** 2
        return soft + pen
    
    def soft_grad(x, tau, lam):
        corr = np.correlate(x, 1.0 - x, 'full') * dx
        m = corr.max()
        exp_vals = np.exp(tau * (corr - m))
        weights = exp_vals / exp_vals.sum()
        j = np.arange(n)
        grad = np.zeros(n)
        # Only use top-30 weights for efficiency
        nw = min(30, len(weights))
        top_idx = np.argsort(-weights)[:nw]
        for ki in top_idx:
            w = weights[ki]
            if w < 1e-8:
                continue
            k = ki - (n - 1)
            a = j - k
            b = j + k
            m1 = (a >= 0) & (a < n)
            m2 = (b >= 0) & (b < n)
            grad[m1] += w * (1.0 - x[a[m1]]) * dx
            grad[m2] -= w * x[b[m2]] * dx
        grad += 2.0 * lam * (x.sum() - tgt)
        return grad
    
    # Anneal temperature
    n_anneal = 5
    tau = tau_start
    for ai in range(n_anneal):
        if time.time() - t0 > tm * 0.9:
            break
        tau = tau_start * (tau_end / tau_start) ** (ai / max(1, n_anneal - 1))
        lam = 50.0 + 100.0 * ai

            h = project_feasible(r.x, tgt)
        except Exception:

    j = np.arange(n)
    for t in range(1, 50001):
        if time.time() > deadline:
            break
        alpha = 2.0 / (t + 1.0)
        y = h + (alpha - 1.0) * (h - y)
        corr = np.correlate(y, 1.0 - y, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        a = j - k
        b = j + k
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += 1.0 - y[a[m1]]
        grad[m2] -= y[b[m2]]
        grad *= dx
        lr = 0.8 / np.sqrt(t + 1.0)
        h = project_feasible(y - lr * grad, tgt)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def make_perturbation(h0, rng, mode):
    n = len(h0)
    tgt = n / 2.0
    if mode == 0:
        h = h0 + rng.normal(0, 0.04, size=n)
    elif mode == 1:
        h = h0 + rng.normal(0, 0.1, size=n)
    elif mode == 2:
        h = h0.copy()
        seg = max(3, n // 10)
        s = rng.randint(0, n - seg)
        h[s:s+seg] += rng.uniform(-0.15, 0.15)
    elif mode == 3:
        h = h0.copy()
        s1 = rng.randint(0, n // 2)
        s2 = n // 2 + rng.randint(0, n // 2)
        seg = max(3, n // 12)
        tmp = h[s1:s1+seg].copy()
        h[s1:s1+seg] = h[s2:s2+seg].copy()
        h[s2:s2+seg] = tmp
    elif mode == 4:
        h = h0.copy()
        seg = max(2, n // 15)
        s = rng.randint(0, n - seg)
        h[s:s+seg] = rng.uniform(0.0, 1.0, seg)
    else:
        h = h0 + rng.normal(0, 0.02, size=n)
    return project_feasible(h, tgt)
