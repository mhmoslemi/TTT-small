```python
def grad_smooth_lse(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    scaled = (corr - cmax) / T
    scaled = np.clip(scaled, -50, 0)
    exp_scaled = np.exp(scaled)
    weights = exp_scaled / np.sum(exp_scaled)  # these are the softmax weights
    
    # Now compute gradient
    # ∂F/∂h[p] = (1/T) * Σ_k w[k] * [(1-h[p-k+n-1]) * [valid] - h[p+k-n+1] * [valid]]
    
    grad = np.zeros(n)
    for p in range(n):
        s = 0.0
        for k in range(2*n - 1):
            # First term: (1-h[p-k+n-1]) if 0 <= p-k+n-1 < n
            idx1 = p - k + n - 1
            if 0 <= idx1 < n:
                s += weights[k] * (1.0 - x[idx1])
            # Second term: -h[p+k-n+1] if 0 <= p+k-n+1 < n
            idx2 = p + k - n + 1
            if 0 <= idx2 < n:
                s -= weights[k] * x[idx2]
        grad[p] = s / T
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_objective(x, T):
    corr = get_corr(x)
    cmax = np.max(corr)
    scaled = np.clip((corr - cmax) / T, -50, 0)
    return cmax + T * np.log(np.mean(np.exp(scaled)))

def lse_gradient(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    scaled = np.clip((corr - cmax) / T, -50, 0)
    exp_s = np.exp(scaled)
    w = exp_s / np.sum(exp_s)
    
    # grad[p] = (1/T) * sum_k w[k] * [(1-h[p-k+n-1])*[valid1] - h[p+k-n+1]*[valid2]]
    # Use correlation for efficiency
    # First part: sum_k w[k] * (1-x[p-k+n-1]) for valid indices
    # This is correlate(1-x, w) at specific positions
    # (1-x) has length n, w has length 2n-1
    # full correlate gives length n + 2n-1 - 1 = 3n-2
    
    # Actually, let me just do it directly for clarity (n is small)
    grad = np.zeros(n)
    one_minus_x = 1.0 - x
    for p in range(n):
        s1 = 0.0
        s2 = 0.0
        for k in range(2*n - 1):
            idx1 = p - k + n - 1
            if 0 <= idx1 < n:
                s1 += w[k] * one_minus_x[idx1]
            idx2 = p + k - n + 1
            if 0 <= idx2 < n:
                s2 += w[k] * x[idx2]
        grad[p] = (s1 - s2) / T
    return grad

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, max_iter=3000):
    def obj(x):
        return lse_objective(x, T)
    
    def grad(x):
        return lse_gradient(x, T)
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h, score_c5(h)

def targeted_local_search(h, n, t_start, time_limit):
    best = h.copy()
    best_score = score_c5(best)
    best_corr = get_corr(best)
    
    step = 0.05
    stall = 0
    max_stall = 40
    
    rng = np.random.RandomState(777)
    
    while time.time() - t_start < time_limit:
        improved = False
        corr = best_corr
        n_corr = len(corr)
        
        # Get top 8 worst shifts
        top_indices = np.argsort(corr)[-8:]
        
        for k in top_indices:
            if time.time() - t_start > time_limit:
                break
            offset = k - (n - 1)  # the actual displacement
            if offset == 0:
                continue
            
            # For this shift, try moves that reduce corr[k]
            # Move from i to i+offset: reduces h[i], increases h[i+offset]
            # Effect on corr[k]: -h[i+offset offset stuff] + ...
            # Simpler: just try it and check
            
            # Try a subset of positions for speed
            # Focus on positions where h is "intermediate" (not 0 or 1)
            candidates = np.where((best > 0.05) & (best < 0.95))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Shuffle and try a subset
            rng.shuffle(candidates)
            for i in candidates[:30]:
                if time.time() - t_start > time_limit:
                    break
                
                # Try moving from i to i+offset
                j = i + offset
                if 0 <= j < n and j != i and best[j] < 1.0 - 1e-6:
                    amt = min(step, best[i], 1.0 - best[j])
                    if amt > 1e-8:
                        trial = best.copy()
                        trial[i] -= amt
                        trial[j] += amt
                        trial = project_feasible(trial, n)
                        s = score_c5(trial)
                        if s < best_score - 1e-12:
                            best = trial
                            best_score = s
                            best_corr = get_corr(best)
                            improved = True
                            break
                
                # Try moving from i-offset to i (i.e., from i to i-offset)
                j = i - offset
                if 0 <= j < n and j != i and best[j] < 1.0 - 1e-6:
                    amt = min(step, best[i], 1.0 - best[j])
                    if amt > 1e-8:
                        trial = best.copy()
                        trial[i] -= amt
                        trial[j] += amt
                        trial = project_feasible(trial, n)
                        s = score_c5(trial)
                        if s < best_score - 1e-12:
                            best = trial
                            best_score = s
                            best_corr = get_corr(best)
                            improved = True
                            break
        
        if not improved:
            # Try random moves
            for _ in range(n):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j or best[i] < 1e-6 or best[j] > 1.0 - 1e-6:
                    continue
                amt = min(step, best[i], 1.0 - best[j])
                if amt < 1e-8:
                    continue
                trial = best.copy()
                trial[i] -= amt
                trial[j] += amt
                trial = project_feasible(trial, n)
                s = score_c5(trial)
                if s < best_score - 1e-12:
                    best = trial
                    best_score = s
                    best_corr = get_corr(best)
                    improved = True
                    break
            
            if not improved:
                stall += 1
                step *= 0.75
                if stall > max_stall or step < 1e-8:
                    break
        else:
            stall = 0
    
    return best, best_score
```

```python
def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_objective(x, T):
    corr = get_corr(x)
    cmax = np.max(corr)
    scaled = np.clip((corr - cmax) / T, -50, 0)
    return cmax + T * np.log(np.mean(np.exp(scaled)))

def lse_gradient(x, T):
    n = len(x)
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx_val
    cmax = np.max(corr)
    scaled = np.clip((corr - cmax) / T, -50, 0)
    exp_s = np.exp(scaled)
    w = exp_s / np.sum(exp_s)
    one_minus_x = 1.0 - x
    grad = np.zeros(n)
    nc = 2 * n - 1
    for p in range(n):
        s1 = 0.0
        s2 = 0.0
        for k in range(nc):
            idx1 = p - k + n - 1
            if 0 <= idx1 < n:
                s1 += w[k] * one_minus_x[idx1]
            idx2 = p + k - n + 1
            if 0 <= idx2 < n:
                s2 += w[k] * x[idx2]
        grad[p] = (s1 - s2) / T
    return grad
```
