import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj_and_grad(h, tau, penalty=0.0, target_sum=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        # Term 1: (1-h[j-lag]) for j in [max(0,lag), min(n-1, n-1+lag)]
        j1_lo = max(0, lag)
        j1_hi = min(n-1, n-1+lag)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += w * (1.0 - h[j1_lo-lag:j1_hi-lag+1])
        # Term 2: -h[j+lag] for j in [max(0,-lag), min(n-1, n-1-lag)]
        j2_lo = max(0, -lag)
        j2_hi = min(n-1, n-1-lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] += w * (-h[j2_lo+lag:j2_hi+lag+1])
    
    grad *= dx
    
    if penalty > 0 and target_sum is not None:
        obj += penalty * (h.sum() - target_sum)**2
        grad += 2 * penalty * (h.sum() - target_sum)
    
    return obj, grad

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgsb_optimize(h0, n, target_sum, deadline, tau=100.0, penalty=100.0, max_iter=500):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau, penalty, target_sum)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau, penalty, target_sum)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def pgd_backtracking(h0, n, target_sum, deadline, tau=100.0, max_iter=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = project_onto_constraint(h, target_sum)
    
    step = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = soft_max_obj_and_grad(h, tau)
        # Backtracking line search
        alpha = step
        obj0 = soft_max_obj_and_grad(project_onto_constraint(h, target_sum), tau)[0]
        for _ in range(20):
            h_trial = project_onto_constraint(h - alpha * g, target_sum)
            obj_trial = soft_max_obj_and_grad(h_trial, tau)[0]
            if obj_trial < obj0 - 1e-4 * alpha * np.dot(g, g):
                break
            alpha *= 0.5
        h_new = project_onto_constraint(h - alpha * g, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        step = alpha * 1.5  # scale back up
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    j1_lo = max(0, lag)

    if j1_lo <= j1_hi:

    j2_lo = max(0, -lag)
    j2_hi = min(n-1, n-1-lag)
    if j2_lo <= j2_hi:
        grad[j2_lo:j2_hi+1] += (-h[j2_lo+lag:j2_hi+lag+1])
    grad *= dx
    return corr[k_star] * dx, grad, lag

The key observation: R(k) = ∫h(x)(1-h(x+k))dx = 1 - ∫h(x)h(x+k)dx (when the domains overlap completely, i.e., for k=0).

For k≠0, the domains don't overlap completely:

So R(k) = (2-|k|)*mean(h) - ∫h(x)h(x+k)dx... no that's not right either.

Let me just compute it directly:
R(k) = ∫_{max(0,-k)}^{min(2,2-k)} h(x)(1-h(x+k))dx

For k ≥ 0: R(k) = ∫_0^{2-k} h(x)(1-h(x+k))dx
For k < 0: R(k) = ∫_{-k}^2 h(x)(1-h(x+k))dx = ∫_0^{2+k} h(x-k)(1-h(x))dx (substituting)

In the discrete case with n points and dx = 2/n:
For lag ℓ (where ℓ = k/dx, an integer from -(n-1) to n-1):
R(ℓ) = dx * ∑_{i: 0≤i<n, 0≤i+ℓ<n} h[i](1-h[i+ℓ])

The number of terms is n-|ℓ|.

Now, the sum ∑h[i](1-h[i+ℓ]) = ∑h[i] - ∑h[i]h[i+ℓ] (over the valid range).

∑h[i] over the valid range is NOT n/2 in general (it's the sum over a subset of indices).

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lbfgsb_pass(h0, n, target_sum, tau, penalty, max_iter, t_limit):
    t0 = time.time()
    killed = [False]

    def obj_fun(x):
        if time.time() - t0 > t_limit:
            killed[0] = True
            return float('inf')
        o, _ = soft_max_obj_and_grad(x, tau)
        o += penalty * (x.sum() - target_sum) ** 2
        return o

    def grad_fun(x):
        if time.time() - t0 > t_limit:
            killed[0] = True
            return np.zeros(n)
        _, g = soft_max_obj_and_grad(x, tau)
        g = g + 2.0 * penalty * (x.sum() - target_sum)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxls': 30, 'maxfun': max_iter * 3})
        if killed[0]:
            return h0, compute_c5(h0)
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient(h0, n, target_sum, deadline, mode='multi', n_lags=5, max_iter=50000):
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    step = 0.08
    no_improve = 0
    patience = 400
    tau_w = 30.0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")

        if mode == 'exact':
            k_star = np.argmax(corr)
            lag = k_star - (n - 1)
            grad = gradient_of_R_at_lag(h, lag)
        elif mode == 'multi':
            top_idx = np.argsort(corr)[-n_lags:][::-1]
            top_vals = corr[top_idx]
            sv = tau_w * top_vals
            w = np.exp(sv - np.max(sv))
            w /= w.sum()
            grad = np.zeros(n)
            for wi, mi in zip(w, top_idx):
                if wi < 1e-10:
                    continue
                lag = mi - (n - 1)
                grad += wi * gradient_of_R_at_lag(h, lag)
        else:
            _, grad = soft_max_obj_and_grad(h, 150.0)

        h_new = h - step * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                step *= 0.75
                no_improve = 0
                if step < 1e-6:
                    break
        h = h_new

    return best_h, best_c5

def binary_perturb(h, n, target_sum, n_flips):
    h_new = h.copy()
    indices = np.random.choice(n, size=min(n_flips, n), replace=False)
    for i in indices:
        h_new[i] = 0.0 if h_new[i] > 0.5 else 1.0
    return project_onto_constraint(h_new, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 10

    best_h = None
    best_c5 = float('inf')

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        c5_0 = compute_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_0
    else:
        n0 = 120
        best_h = np.full(n0, 0.5)
        best_c5 = compute_c5(best_h)

    n = len(best_h)
    target = n / 2.0

    # Phase 1: L-BFGS-B tau annealing
    taus = [20, 50, 100, 200, 400, 700]
    for tau in taus:
        if time.time() > deadline - 250:
            break
        tlim = min(45.0, deadline - time.time() - 200)
        if tlim < 5:
            break
        h_opt, c5_opt = lbfgsb_pass(best_h, n, target, tau, 800.0, 300, tlim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Multi-lag projected gradient (long run)
    if time.time() < deadline - 120:
        tlim = min(100.0, deadline - time.time() - 100)
        h_opt, c5_opt = projected_gradient(best_h, n, target, time.time() + tlim,
                                            mode='multi', n_lags=5, max_iter=30000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Exact max PGD
    if time.time() < deadline - 80:
        tlim = min(60.0, deadline - time.time() - 60)
        h_opt, c5_opt = projected_gradient(best_h, n, target, time.time() + tlim,
                                            mode='exact', max_iter=20000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Another L-BFGS-B with high tau after PGD
    if time.time() < deadline - 80:
        tlim = min(30.0, deadline - time.time() - 60)
        h_opt, c5_opt = lbfgsb_pass(best_h, n, target, 500, 1000.0, 200, tlim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Binary perturbation + re-optimize
    for attempt in range(12):
        if time.time() > deadline - 50:
            break
        n_flips = np.random.randint(2, 14)
        h_pert = binary_perturb(best_h, n, target, n_flips)
        tlim = min(15.0, deadline - time.time() - 30)
        if tlim < 4:
            break
        h_opt, c5_opt = lbfgsb_pass(h_pert, n, target, 200, 800.0, 150, tlim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() < deadline - 30:
            h_opt2, c5_opt2 = projected_gradient(h_opt, n, target, time.time() + 12,
                                                  mode='multi', n_lags=4, max_iter=10000)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    # Phase 6: Try other n values
    for n_try in [80, 100, 140, 160, 200]:
        if time.time() > deadline - 35:
            break
        if n_try == n:
            continue
        target_try = n_try / 2.0
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_onto_constraint(h0, target_try)
        tlim = min(18.0, deadline - time.time() - 20)
        if tlim < 4:
            continue
        h_opt, c5_opt = lbfgsb_pass(h0, n_try, target_try, 150, 800.0, 200, tlim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Final multi-lag PGD polish
    if time.time() < deadline - 5:
        n = len(best_h)
        target = n / 2.0
        tlim = deadline - time.time() - 2
        h_opt, c5_opt = projected_gradient(best_h, n, target, time.time() + tlim,
                                            mode='multi', n_lags=5, max_iter=50000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final exact max quick pass
    if time.time() < deadline - 1:
        n = len(best_h)
        target = n / 2.0
        h_opt, c5_opt = projected_gradient(best_h, n, target, time.time() + (deadline - time.time() - 1),
                                            mode='exact', max_iter=10000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
