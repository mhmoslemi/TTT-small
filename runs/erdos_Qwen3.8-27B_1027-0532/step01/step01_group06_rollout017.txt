from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def lse_objective(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    if T < 1e-12:
        return dx * cmax
    shifted = (corr - cmax) / T
    shifted = np.minimum(shifted, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return dx * (cmax + T * logsum)

def try_slsqp(h0, T, max_iter, ftol):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        return lse_objective(x, T)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            return x_new, score_c5(x_new)
    except Exception:
        pass
    return h0, score_c5(h0)

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / T
    shifted = np.minimum(shifted, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to (n-1)
    
    for k in range(2*n-1):
        w = dx * weights[k]
        lag = lags[k]
        if w < 1e-8:
            continue
        # d(corr[k])/d(h[p]) = (1-h[p-lag]) - h[p+lag] (with bounds)
        if lag >= 0:
            # p-lag valid: p >= lag
            # p+lag valid: p <= n-1-lag
            for p in range(lag, n - lag + 1):
                grad[p] += w * ((1.0 - h[p-lag]) - h[p+lag])
        else:
            # lag < 0, let ll = -lag > 0
            ll = -lag
            # p-(-ll) = p+ll valid: p <= n-1-ll
            # p+(-ll) = p-ll valid: p >= ll
            for p in range(ll, n - ll + 1):
                grad[p] += w * ((1.0 - h[p+ll]) - h[p-ll])
    
    return grad
