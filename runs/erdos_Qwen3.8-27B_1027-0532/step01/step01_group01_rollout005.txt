    n_active = np.sum(active)

    for k in range(len(corr)):

from scipy.optimize import minimize, differential_evolution, minimize_scalar
import time

_g_n = 85
_g_eps = 100.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam_val = (lo + hi) / 2.0
    return np.clip(x - lam_val, 0.0, 1.0)

def smooth_max_obj(x):
    n = _g_n
    dx = 2.0 / n
    eps = _g_eps
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x):
    n = _g_n
    dx = 2.0 / n
    eps = _g_eps
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        for j in range(n):
            jml = j - l
            if 0 <= jml < n:
                grad[j] += w * (1.0 - x[jml])
            jpl = j + l
            if 0 <= jpl < n:
                grad[j] += w * (-x[jpl])
    grad *= dx
    return grad

def eq_constraint(x):
    return np.sum(x) - _g_n / 2.0

def coord_objective(t, old_val, corr, coeffs, dx, lam):
    vals = corr + (t - old_val) * coeffs
    return np.max(vals) * dx + lam * t

def slpsqp_refine(h_current, n, bounds, eps, maxiter):

    _g_n = n
    _g_eps = eps
    res = minimize(
        smooth_max_obj, h_current.copy(),
        jac=smooth_max_grad,
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': eq_constraint}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return res

def coordinate_descent_pass(h, n, lam):
    dx = 2.0 / n
    improved = False
    for i in range(n):
        old_val = h[i]
        corr = np.correlate(h, 1 - h, mode="full")
        coeffs = np.zeros(2 * n - 1)
        for l in range(-(n - 1), n):
            c = 0.0
            if 0 <= i + l < n:
                c += (1.0 - h[i + l])
            if 0 <= i - l < n:
                c -= h[i - l]
            coeffs[l + n - 1] = c
        res = minimize_scalar(
            coord_objective,
            args=(old_val, corr, coeffs, dx, lam),
            bounds=(0.0, 1.0), method='brent'
        )
        new_val = np.clip(res.x, 0.0, 1.0)
        if abs(new_val - old_val) > 1e-10:
            h[i] = new_val
            improved = True
    return h, improved

def subgradient_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-8
    active = corr >= max_val - tol
    n_active = float(np.sum(active))
    if n_active == 0:
        n_active = 1.0
    grad = np.zeros(n)
    for k in range(len(corr)):
        if not active[k]:
            continue
        l = k - (n - 1)
        w = 1.0 / n_active
        for j in range(n):
            jml = j - l
            if 0 <= jml < n:
                grad[j] += w * (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                grad[j] += w * (-h[jpl])
    grad *= dx
    h_new = h - alpha * grad
    return project_feasible(h_new, n)
