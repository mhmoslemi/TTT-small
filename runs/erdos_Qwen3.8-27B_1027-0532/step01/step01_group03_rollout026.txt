```python
        for j in range(i+1, n_lines):

            if abs(da) < 1e-15:

            best_t = t
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _solve_1d(a, b, lo, hi):
    """Minimize max(a*t + b) for t in [lo, hi]. Returns optimal t."""
    if lo > hi:
        return lo
    # Candidates: lo, hi, and pairwise intersections + intersection with... 
    # Actually, just do lo, hi, and all pairwise intersections
    na = len(a)
    candidates = [lo, hi]
    for i in range(na):
        for j in range(i+1, na):
            da = a[i] - a[j]
            if abs(da) < 1e-14:
                continue
            t = (b[j] - b[i]) / da
            if lo - 1e-12 <= t <= hi + 1e-12:
                candidates.append(max(lo, min(hi, t)))
    
    best_t = lo
    best_v = np.inf
    for t in candidates:
        v = np.max(a * t + b)
        if v < best_v:
            best_v = v
            best_t = t
    return best_t

def _coord_descent(h, n, max_sweeps=100, verbose=False):
    target_sum = n / 2.0
    best = h.copy()
    best_c5 = _c5_of(best)
    
    for sweep in range(max_sweeps):
        h_work = best.copy()
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        
        for i in range(n):
            # Compute c_k for coordinate i
            # c_k = (1-h)[i+k-(n-1)] - h[i-k+(n-1)], out of bounds = 0
            # i+k-(n-1) in [0,n-1] => k in [n-1-i, 2(n-1)-i]
            # i-k+(n-1) in [0,n-1] => k in [i, i+n-1]
            
            c = np.zeros(2*n - 1)
            # First term: (1-h)[i+k-(n-1)]
            k1 = n - 1 - i
            k2 = 2*(n-1) - i
            k1 = max(0, k1)
            k2 = min(2*n-2, k2)
            if k1 <= k2:
                idx = k1 + np.arange(k2 - k1 + 1)
                h_idx = i + idx - (n-1)  # should be in [0, n-1]
                c[idx] += 1.0 - h_work[h_idx]
            
            # Second term: -h[i-k+(n-1)]
            k3 = i
            k4 = i + n - 1
            k3 = max(0, k3)
            k4 = min(2*n-2, k4)
            if k3 <= k4:
                idx = k3 + np.arange(k4 - k3 + 1)
                h_idx = i - idx + (n-1)  # should be in [0, n-1]
                c[idx] -= h_work[h_idx]
            
            # Now: corr_new[k] = corr[k] + delta * c[k]
            # where delta = new_h[i] - old_h[i]
            # old_h[i] = h_work[i]
            # new_h[i] = h_work[i] + delta, so delta in [-h_work[i], 1-h_work[i]]
            
            lo = -h_work[i]
            hi = 1.0 - h_work[i]
            
            if abs(hi - lo) < 1e-15:
                continue
            
            # Minimize max(corr + delta * c) over delta in [lo, hi]
            # This is min max_k (c[k] * delta + corr[k]) over delta in [lo, hi]
            best_delta = _solve_1d(c, corr, lo, hi)
            
            h_work[i] += best_delta
        
        h_fixed = _fix_sum(h_work, target_sum)
        c5 = _c5_of(h_fixed)
        if c5 < best_c5 - 1e-14:
            best = h_fixed
            best_c5 = c5
            if verbose:
                print(f"Sweep {sweep}: C5 = {best_c5:.8f}")
        else:
            # No improvement, try a few more sweeps
            if sweep > 5:
                break
    
    return best, best_c5
```

Wait, I realize the `_solve_1d` function with pairwise intersections is O(n^2) where n is the number of correlation lags (2*n_points - 1). For n_points=100, that's 199 lags, so O(199^2) ≈ 40000 per coordinate, times 100 coordinates = 4 million per sweep. With 200 sweeps, that's 800 million operations. This might be too slow in Python.

Let me optimize. Instead of checking all pairs, I can use a simpler approach:
- Since the function is convex and piecewise linear, the minimum is where the slope changes from negative to positive
- I can use a "galloping" approach or just evaluate at a few smart points

Actually, a better approach: the minimum of max of lines in 1D is at a vertex of the upper envelope. The upper envelope of n lines has at most n vertices. We can find which vertex is the minimum by:
1. Computing the upper envelope (O(n log n) with a divide-and-conquer or monotone chain approach)
2. Finding the minimum vertex

But implementing the upper envelope is non-trivial. Let me use a simpler approach:

Since the function is convex, I can use ternary search. But ternary search doesn't work perfectly on piecewise linear functions (it converges to an interval, not a point). However, I can do a few iterations of ternary search to narrow down, then check the remaining candidates.

Actually, the simplest efficient approach for 1D convex minimization of a piecewise linear function:
1. Evaluate at lo and hi
2. The subgradient at the minimum contains 0
3. Binary search: evaluate the "slope" at the midpoint. The slope of max(f_k) at t is:
   - If there's a unique maximizer k*, the slope is a_{k*}
   - If there are multiple maximizers, the subdifferential is [min a_k, max a_k] over the maximizers
   
   If all maximizers have a_k > 0, the function is increasing, so the minimum is to the left.
   If all maximizers have a_k < 0, the function is decreasing, so the minimum is to the right.
   Otherwise, we're at the minimum.

This gives O(n log n) per coordinate (log n for binary search iterations, n for evaluating the max and checking the slope at each step). Total: O(n * log n * n) per sweep = O(n^2 log n). For n=100, that's about 100 * 7 * 100 = 70000 per sweep. Very fast!

Let me implement this:

```python
def _solve_1d_convex(a, b, lo, hi, tol=1e-12, max_iter=50):
    """Minimize max(a*t + b) for t in [lo, hi], where the function is convex."""
    n_lines = len(a)
    
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        max_val = np.max(vals)
        # Find all lines achieving (approximately) the max
        active = np.where(vals >= max_val - 1e-10)[0]
        
        a_min = np.min(a[active])
        a_max = np.max(a[active])
        
        if a_min > 1e-14:
            # Function is increasing at mid, minimum is to the left
            hi = mid
        elif a_max < -1e-14:
            # Function is decreasing at mid, minimum is to the right
            lo = mid
        else:
            # 0 is in the subdifferential, we're at the minimum
            return mid
    
    return 0.5 * (lo + hi)
```

```python
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

                break

        for i in range(n):
            p = i + k_all - (n - 1)
            q = i - k_all + (n - 1)
            c = np.zeros(2 * n - 1)
            mask_p = (p >= 0) & (p < n)
            mask_q = (q >= 0) & (q < n)
            c[mask_p] += 1.0 - h_work[p[mask_p]]
            c[mask_q] -= h_work[q[mask_q]]
```

Hmm, wait. There's an issue with the coordinate descent. Each time I change h[i], the correlation changes. But I'm computing the correlation from scratch at each coordinate (via `np.correlate(h_work, 1.0 - h_work, mode='full')`). This is O(n^2) per coordinate, making the total O(n^3) per sweep. For n=100, that's 10^6 per sweep, which is fine.

But actually, I can avoid recomputing the full correlation each time. Let me use the incremental update:
When h[i] changes by δ, corr changes by δ * c where c is the coefficient vector I computed.

Let me restructure:

```python
def _coord_descent(h, n, max_sweeps=200, time_limit=None, t_start=None):
    target_sum = n / 2.0
    best = h.copy()
    best_c5 = _c5_of(best)
    k_all = np.arange(2 * n - 1)
    
    for sweep in range(max_sweeps):
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        h_work = best.copy()
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        
        for i in range(n):
            p = i + k_all - (n - 1)
            q = i - k_all + (n - 1)
            c = np.zeros(2 * n - 1)
            mask_p = (p >= 0) & (p < n)
            mask_q = (q >= 0) & (q < n)
            c[mask_p] += 1.0 - h_work[p[mask_p]]
            c[mask_q] -= h_work[q[mask_q]]
            
            lo = -h_work[i]
            hi = 1.0 - h_work[i]
            
            if hi - lo < 1e-14:
                continue
            
            best_delta = _solve_1d_convex(c, corr, lo, hi)
            
            if abs(best_delta) > 1e-14:
                h_work[i] += best_delta
                corr += best_delta * c
        
        h_fixed = _fix_sum(h_work, target_sum)
        c5 = _c5_of(h_fixed)
        if c5 < best_c5 - 1e-14:
            best = h_fixed
            best_c5 = c5
        else:
            if sweep > 3:
                break
    
    return best, best_c5
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _solve_1d_convex(a, b, lo, hi):
    vals_lo = a * lo + b
    max_lo = np.max(vals_lo)
    active_lo = np.where(vals_lo >= max_lo - 1e-12)[0]
    if len(active_lo) > 0 and np.min(a[active_lo]) >= -1e-12:
        return lo
    vals_hi = a * hi + b
    max_hi = np.max(vals_hi)
    active_hi = np.where(vals_hi >= max_hi - 1e-12)[0]
    if len(active_hi) > 0 and np.max(a[active_hi]) <= 1e-12:
        return hi
    lo_b = lo
    hi_b = hi
    for _ in range(60):
        if hi_b - lo_b < 1e-14:
            break
        mid = 0.5 * (lo_b + hi_b)
        vals = a * mid + b
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-12)[0]
        a_min = np.min(a[active])
        a_max = np.max(a[active])
        if a_min > 1e-12:
            hi_b = mid
        elif a_max < -1e-12:
            lo_b = mid
        else:
            return mid
    return 0.5 * (lo_b + hi_b)

def _coord_descent(h, n, max_sweeps=200, time_limit=60, t_start=None):
    target_sum = n / 2.0
    best = h.copy()
    best_c5 = _c5_of(best)
    k_all = np.arange(2 * n - 1)
    if t_start is None:
        t_start = time.time()
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        h_work = best.copy()
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        for i in range(n):
            p = i + k_all - (n - 1)
            q = i - k_all + (n - 1)
            c = np.zeros(2 * n - 1)
            mask_p = (p >= 0) & (p < n)
            mask_q = (q >= 0) & (q < n)
            c[mask_p] += 1.0 - h_work[p[mask_p]]
            c[mask_q] -= h_work[q[mask_q]]
            lo = -h_work[i]
            hi = 1.0 - h_work[i]
            if hi - lo < 1e-14:
                continue
            best_delta = _solve_1d_convex(c, corr, lo, hi)
            if abs(best_delta) > 1e-14:
                h_work[i] += best_delta
                corr += best_delta * c
        h_fixed = _fix_sum(h_work, target_sum)
        c5 = _c5_of(h_fixed)
        if c5 < best_c5 - 1e-14:
            best = h_fixed
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break
    return best, best_c5

def _slsqp_refine(h, n, target_sum, alpha, maxiter=150, ftol=1e-14):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = _fix_sum(res.x, target_sum)
        return h_out, _c5_of(h_out)
    except Exception:
        return h, _c5_of(h)
```
