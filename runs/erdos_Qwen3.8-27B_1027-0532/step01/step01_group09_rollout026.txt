   - This gives a linear constraint on h
   - Solve the resulting convex problem (minimize t subject to linear constraints)
   - Iterate

import numpy as np
from scipy.optimize import linprog, minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr_k(h, k):
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += 1.0 - h[a[m1]]
    grad[m2] -= h[b[m2]]
    return grad * dx

    n = len(h_init)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h_init.copy(), tgt)
    t0 = time.time()
    
    # Store linearized constraints: (grad, const) pairs
    # where grad @ h + const is a linear approximation of some corr_k
    active_grads = []
    active_consts = []
    
    for it in range(max_iter):
        if time.time() - t0 > tm:
            break
        
        corr = compute_corr(h)
        # Find top worst k values
        k_indices = np.argsort(corr)[::-1][:top_k]
        
        for ki in k_indices:
            k = int(ki) - (n - 1)
            c_val = corr[ki]
            g = grad_corr_k(h, k)
            const = c_val - g @ h
            # Only add if this k is not already in active set (check by k value)
            # Actually, we should add each unique k only once per "round"
            # For now, just add them all
            active_grads.append(g)
            active_consts.append(const)
        
        # Cap the number of constraints to avoid LP slowdown
        if len(active_grads) > 200:
            # Keep only the most recent 200
            active_grads = active_grads[-200:]
            active_consts = active_consts[-200:]
        
        # Solve LP

        c_vec[-1] = 1.0

            A_ub = np.zeros((len(active_grads), n + 1))
            b_ub = np.zeros(len(active_grads))

                A_ub[idx, :n] = gm
                A_ub[idx, n] = -1.0
                b_ub[idx] = -cm
        else:
            A_ub = None
            b_ub = None
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([tgt])
        
        bounds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]

                h = project_feasible(h_new, tgt)

            break
        
        # Check convergence

            corr_new = compute_corr(h)
            if corr_new.max() < 0.37:  # Good enough
