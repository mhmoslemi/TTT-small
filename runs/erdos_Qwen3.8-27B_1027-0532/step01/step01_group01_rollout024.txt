    rng = np.random.RandomState(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Optimize at original n with progressive eps
    for eps in [30.0, 60.0, 120.0, 250.0]:
        if time.time() - start_time > budget_s - 300:
            break
        h_new = run_alm(best_h, n0, eps)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 2: Try larger n values
    for n_try in [100, 130, 170, 220, 280]:
        if time.time() - start_time > budget_s - 180:
            break
        h_interp = interpolate_to_n(best_h, len(best_h), n_try)
        h_interp = project_feasible(h_interp, n_try)
        for eps in [50.0, 120.0, 250.0]:
            if time.time() - start_time > budget_s - 60:
                break
            h_new = run_alm(h_interp, n_try, eps)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    # Phase 3: Coordinate descent refinement
    if time.time() - start_time < budget_s - 40:
        best_h, best_c5 = coord_descent(best_h, len(best_h), max_passes=50, seed=seed)

    # Phase 4: Random restarts with moderate perturbation
    n_use = len(best_h)
    for r in range(25):
        if time.time() - start_time > budget_s - 40:
            break
        scale = 0.008 + 0.004 * r
        h_pert = best_h + scale * rng.randn(n_use)
        h_pert = project_feasible(h_pert, n_use)
        h_new = run_alm(h_pert, n_use, eps=100.0, max_outer=10, max_inner=250)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 5: Block local search
    if time.time() - start_time < budget_s - 30:
        best_h, best_c5 = block_local_search(best_h, n_use, block_size=5, max_rounds=40, seed=seed + 7)

    # Phase 6: Final high-eps refinement
    if time.time() - start_time < budget_s - 20:
        h_new = run_alm(best_h, n_use, eps=500.0, max_outer=8, max_inner=300)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 7: Final coordinate descent
    if time.time() - start_time < budget_s - 5:
        best_h, best_c5 = coord_descent(best_h, len(best_h), max_passes=30, seed=seed + 200)

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
