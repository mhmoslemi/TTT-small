import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _smooth_obj_grad(h, n, t, penalty):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_c = np.max(corr)
    nc = len(corr)

    if t <= 1e-6:
        idx_max = np.argmax(corr)
        obj = max_c * dx
        grad = np.zeros(n)
        d = idx_max - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += omh[d:]
            grad[d:] += -h[:m]
        else:
            e = -d
            m = n - e
            grad[e:] += omh[:m]
            grad[:m] += -h[e:]
        grad *= dx
    else:
        shifted = t * (corr - max_c)
        shifted = np.clip(shifted, -500.0, 500.0)
        ev = np.exp(shifted)
        se = np.sum(ev)
        weights = ev / se
        obj = (max_c + np.log(se) / t) * dx
        grad = np.zeros(n)
        sig = np.where(weights > 1e-10)[0]
        for idx in sig:
            w = weights[idx]
            d = idx - (n - 1)
            if d >= 0:
                m = n - d
                grad[:m] += w * omh[d:]
                grad[d:] += -w * h[:m]
            else:
                e = -d
                m = n - e
                grad[e:] += w * omh[:m]
                grad[:m] += -w * h[e:]
        grad *= dx

    if penalty > 0:
        s = np.sum(h) - n / 2.0
        obj += 0.5 * penalty * s * s
        grad += penalty * s

    return obj, grad


def _obj(x, n, t, penalty):
    return _smooth_obj_grad(x, n, t, penalty)[0]


def _grad(x, n, t, penalty):
    return _smooth_obj_grad(x, n, t, penalty)[1]

    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step0
    for it in range(n_iter):
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode="full")
        idx = np.argmax(corr)
        d = idx - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            grad[:m] += omh[d:]
            grad[d:] += -h[:m]
        else:
            e = -d
            m = n - e
            grad[e:] += omh[:m]
            grad[:m] += -h[e:]
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h = h - step * grad / gn
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.997
    return best_h, best_c5

        h_interp = project_feasible(h_interp, n1)
        h_sa3, c5_sa3 = _sa_binary(h_interp, n1, 100000, seed + 3, 0.05, 0.003)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()
            best_n = n1

        if tleft() > 100:
            h_sa4, c5_sa4 = _sa_binary(best_h, best_n, 60000, seed + 4, 0.02, 0.002)
            if c5_sa4 < best_c5:
                best_c5 = c5_sa4
                best_h = h_sa4.copy()
                best_n = best_n

    # Phase 5: Random restarts with continuous optimization
    if tleft() > 100:
        n_try = best_n
        for i in range(12):
            if tleft() < 50:
                break
            noise = 0.02 + 0.01 * (i % 3)
            h_pert = best_h + rng.randn(n_try) * noise
            h_pert = project_feasible(h_pert, n_try)
            t_use = [50, 150, 400][i % 3]
            bounds = [(0.0, 1.0)] * n_try
            try:
                res = minimize(_obj, h_pert, jac=_grad, method='L-BFGS-B',
                              bounds=bounds, args=(n_try, t_use, 500.0),
                              options={'maxiter': 250, 'ftol': 1e-16, 'gtol': 1e-10})
                h_opt = np.clip(res.x, 0.0, 1.0)
                h_opt = project_feasible(h_opt, n_try)
            except Exception:
                continue
            c5 = compute_c5(h_opt, n_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 6: Final polish
    if tleft() > 20:
        h_pol, c5_pol = _subgradient_polish(best_h, best_n, 500, 0.03)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = best_n

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
