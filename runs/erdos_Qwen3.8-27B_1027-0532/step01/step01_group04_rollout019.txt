    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    excess = np.maximum(0, corr - threshold / dx)
    active = excess > 0
    indices = np.where(active)[0]
    k_vals = indices - n + 1  # shape (num_active,)
    
    grad = np.zeros(n)
    
    for i, k in enumerate(k_vals):
        w = 2.0 * excess[indices[i]] * dx
        # Term 1: (1-h[j-k]) for valid j

        hi2 = min(n, n - k)
        if lo2 < hi2:

    return grad

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

    else:

    if count == 0:

    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    S = np.sum(exp_shifted)
    p = exp_shifted / S  # softmax weights
    
    obj = corr_max + np.log(S) / alpha
    
    # Compute gradient: grad[j] = sum_k p[k] * dc_k/dh[j]
    # dc_k/dh[j] = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)

    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - n + 1  # -n+1, ..., n-1
    
    for idx in range(2*n-1):
        wk = p[idx]
        if wk < 1e-15:
            continue
        k = k_vals[idx]
        # Term 1: (1-h[j-k]) for j where 0<=j-k<n
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:

        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2+k:hi2+k]
    
    return obj, grad

        return budget_s - 15.0 - (time.time() - start_time)

    phase1_alphas = [30, 80, 200, 500, 1000, 2000, 5000, 10000]
    k_tops = [3, 5, 8]
    for alpha in phase1_alphas:
        if time_left() < 40:
            break
        for kt in k_tops:
            if time_left() < 20:
                break
            h_new = run_lbfgsb(best_h, topk_smooth_combined,
                               (alpha, kt, ts, pen), bounds, 400, 1e-13)
            h_new = project_feasible(h_new, ts)
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new

    threshold = best_score
    num_hinge_steps = 40
    for step in range(num_hinge_steps):
        if time_left() < 30:
            break
        target_t = max(0.3807, best_score - 0.0004 * (step + 1))
        h_new = run_lbfgsb(best_h, hinge_combined,
                           (target_t, ts, pen), bounds, 300, 1e-13)
        h_new = project_feasible(h_new, ts)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    for alpha in [3000, 8000, 20000]:
        if time_left() < 25:
            break
        h_new = run_lbfgsb(best_h, full_smooth_combined,
                           (alpha, ts, pen), bounds, 300, 1e-14)
        h_new = project_feasible(h_new, ts)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    for p in [10, 30, 100]:
        if time_left() < 25:
            break
        h_new = run_lbfgsb(best_h, power_combined,
                           (p, ts, pen), bounds, 200, 1e-13)
        h_new = project_feasible(h_new, ts)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    grid_sizes = [139, 199, 251, 281, 331, 397, 449, 521]
    for n_try in grid_sizes:
        if time_left() < 80:
            break
        if n_try == n:
            h_cur = best_h.copy()
            ts_t = ts
            bounds_t = bounds
        else:
            ts_t = n_try / 2.0
            bounds_t = [(0.0, 1.0)] * n_try
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, len(best_h))
            h_cur = np.interp(x_new, x_old, best_h)
            h_cur = project_feasible(h_cur, ts_t)

        for alpha in [100, 500, 2000]:
            if time_left() < 20:
                break
            for kt in [3, 5]:
                if time_left() < 15:
                    break
                h_new = run_lbfgsb(h_cur, topk_smooth_combined,
                                   (alpha, kt, ts_t, pen), bounds_t, 300, 1e-13)
                h_new = project_feasible(h_new, ts_t)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
                    h_cur = h_new

    restart_configs = [(0.02, 500), (0.03, 1000), (0.05, 500),
                       (0.05, 2000), (0.08, 500), (0.1, 1000),
                       (0.01, 5000), (0.04, 3000)]
    while time_left() > 30:
        noise, alpha = restart_configs[rng.randint(len(restart_configs))]
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, ts)
        kt = [3, 5][rng.randint(2)]
        h_new = run_lbfgsb(h_pert, topk_smooth_combined,
                           (alpha, kt, ts, pen), bounds, 200, 1e-13)
        h_new = project_feasible(h_new, ts)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    for alpha in [5000, 10000, 30000]:
        if time_left() < 15:
            break
        h_new = run_lbfgsb(best_h, topk_smooth_combined,
                           (alpha, 3, ts, pen), bounds, 200, 1e-14)
        h_new = project_feasible(h_new, ts)
        s_new = compute_c5(h_new)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    h_final = project_feasible(best_h, ts)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
