def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:

            step = min(diff / mask.sum(), 0.5)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            step = min(-diff / mask.sum(), 0.5)
            h[mask] = np.maximum(0.0, h[mask] - step)
    # Final fix

    if abs(residual) > 1e-15:

    n = len(h0)
    try:
        r = minimize(_obj, h0, jac=_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        h = np.clip(r.x, 0.0, 1.0)
        h = repair_h(h, n)
        return h, evaluate_c5(h)
    except Exception:
        return h0, evaluate_c5(h0)

def optimize_annealed(h0, betas, maxiter_per=300):
    h = h0.copy()
    for beta in betas:
        h, c5 = optimize_at_beta(h, beta, maxiter=maxiter_per)
    return h, evaluate_c5(h)

def subspace_topology(h0, n, rng):
    h = h0.copy()
    for step in range(50):
        c5_curr = evaluate_c5(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        active_idx = np.where(corr * dx > c5_curr - 1e-4)[0]
        n_active = len(active_idx)
        if n_active < 1:
            active_idx = np.array([np.argmax(corr)])
            n_active = 1
        coords = rng.choice(n, size=min(n_active * 3, 40), replace=False)
        best_local = h.copy()
        best_local_c5 = c5_curr
        for _ in range(200):
            trial = h.copy()
            idx = rng.randint(n)
            delta = rng.uniform(-0.05, 0.05)
            trial[idx] = np.clip(trial[idx] + delta, 0.0, 1.0)
            trial = repair_h(trial, n)
            c5_trial = evaluate_c5(trial)
            if c5_trial < best_local_c5:
                best_local = trial.copy()
                best_local_c5 = c5_trial
                h = trial.copy()
        return h, best_local_c5
    return h, evaluate_c5(h)

def generate_varied_starts(n, rng, best_h):
    starts = []
    x = np.linspace(0, 2, n)
    x_best = np.linspace(0, 2, len(best_h))
    h_interp = np.interp(x, x_best, best_h)
    starts.append(repair_h(h_interp, n))
    for freq in [2, 3, 4, 5, 6, 7]:
        for amp in [0.35, 0.45, 0.55]:
            h = 0.5 + amp * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            starts.append(repair_h(np.clip(h, 0, 1), n))
    for block in [4, 6, 8, 12]:
        pattern = np.zeros(block)
        nh = rng.randint(1, block)
        pattern[:nh] = 1.0
        h = np.tile(pattern, n // block + 1)[:n]
        h = 0.3 + 0.4 * h + rng.normal(0, 0.05, n)
        starts.append(repair_h(np.clip(h, 0, 1), n))
    h_rand = rng.uniform(0, 1, n)
    starts.append(repair_h(h_rand, n))
    h_pert = h_interp + rng.normal(0, 0.1, n)
    starts.append(repair_h(np.clip(h_pert, 0, 1), n))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    h_init = initial_h_values.copy().astype(float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    if time_left() > 90:
        h_opt, c5_opt = optimize_annealed(best_h, betas=(40, 80, 150, 300, 600), maxiter_per=250)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    n_candidates = []
    for cand in [n_init + 50, n_init + 100, n_init + 200, n_init + 350, n_init + 500]:
        n_candidates.append(cand)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time_left() < 120:
            break
        x_best = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = repair_h(h_interp, n)
        h_opt, c5_opt = optimize_annealed(h_interp, betas=(50, 100, 200, 400), maxiter_per=250)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            if time_left() > 60:
                h_opt2, c5_opt2 = optimize_at_beta(best_h, beta=800, maxiter=300)
                if c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2

    n = len(best_h)
    starts = generate_varied_starts(n, rng, best_h)
    max_restarts = 8
    for i, h_start in enumerate(starts[:max_restarts]):
        if time_left() < 60:
            break
        if len(h_start) != n:
            x_s = np.linspace(0, 2, len(h_start))
            h_start = repair_h(np.interp(np.linspace(0, 2, n), x_s, h_start), n)
        h_opt, c5_opt = optimize_annealed(h_start, betas=(50, 100, 200), maxiter_per=200)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    for _ in range(6):
        if time_left() < 45:
            break
        n = len(best_h)
        scale = 0.03 + 0.04 * rng.random()
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = repair_h(np.clip(h_pert, 0, 1), n)
        h_opt, c5_opt = optimize_at_beta(h_pert, beta=300, maxiter=300)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    for _ in range(4):
        if time_left() < 30:
            break
        n = len(best_h)
        jump_size = 0.15 + 0.1 * rng.random()
        h_jump = best_h + rng.normal(0, jump_size, n)
        h_jump = repair_h(np.clip(h_jump, 0, 1), n)
        h_opt, c5_opt = optimize_annealed(h_jump, betas=(30, 80, 150, 300), maxiter_per=200)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time_left() > 20:
        n = len(best_h)
        h_opt, c5_opt = optimize_at_beta(best_h, beta=1500, maxiter=500)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if time_left() > 15:
            h_opt, c5_opt = optimize_at_beta(best_h, beta=3000, maxiter=400)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
