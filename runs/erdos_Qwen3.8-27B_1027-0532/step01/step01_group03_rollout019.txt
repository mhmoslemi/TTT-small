from scipy.optimize import minimize, differential_evolution

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return 0.0
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def _slsqp(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

        return res.x
    except Exception:

    """Try individual coordinate swaps to reduce C5."""
    h = h.copy()
    best_c5 = _c5_of(h)

                        improved = True
                    else:
                        h[i] -= eps
                        h[j] += eps
        if not improved:

    return h, best_c5

    h = np.zeros(n)
    seg = n // levels
    for k in range(levels):
        val = 1.0 - k / (levels - 1) if levels > 1 else 0.5
        start = k * seg
        end = start + seg if k < levels - 1 else n
        h[start:end] = val
    return _fix_sum(h, n / 2.0)

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for idx in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
    else:
        order = np.argsort(h)
        for idx in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

def _make_sum_constraint(target_sum):
    return {'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
            'jac': lambda x: np.ones(len(x))}

def _soft_max(corr, alpha):
    m = np.max(corr)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _soft_max(corr, alpha) * (2.0 / len(h))

def _grad_smooth_obj(h, alpha):
    # Numerical gradient
    eps = 1e-7
    n = len(h)
    g = np.zeros(n)
    f0 = _smooth_obj(h, alpha)
    for i in range(n):
        h[i] += eps
        g[i] = (_smooth_obj(h, alpha) - f0) / eps
        h[i] -= eps
    return g

def _optimize_slsqp(h_start, n, target_sum, alpha, maxiter=300, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [_make_sum_constraint(target_sum)]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < _smooth_obj(h_start, alpha):
            return np.clip(res.x, 0, 1)
        return h_start
    except Exception:
        return h_start

    bounds = [(0.0, 1.0)] * n
    cons = [_make_sum_constraint(target_sum)]

        res = minimize(_smooth_obj, h_start, args=(alpha,), method='trust-constr',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'xtol': 1e-12})
        if res.fun < _smooth_obj(h_start, alpha):
            return np.clip(res.x, 0, 1)
        return h_start
    except Exception:
        return h_start

        order = np.argsort(1.0 - h)
        for idx in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
    else:
        order = np.argsort(h)
        for idx in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

def _soft_max(corr, alpha):
    m = np.max(corr)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _soft_max(corr, alpha) * (2.0 / len(h))

def _opt_slsqp(h_start, n, alpha, maxiter=300, ftol=1e-14):
    global _sum_target_val
    _sum_target_val = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_obj, h_start, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_res = np.clip(res.x, 0, 1)
        if _c5_of(h_res) <= _c5_of(h_start) + 1e-10:
            return h_res
        return h_start
    except Exception:
        return h_start

def _lagr_obj(x, alpha, lam, n):
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        sm = 0.0
    else:
        sm = (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    return sm + lam * (x.sum() - target)

def _opt_lagrangian(h_start, n, alpha, max_bisect=35):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    lo, hi = -30.0, 30.0
    best_h = h_start.copy()
    best_c5 = _c5_of(h_start)
    h_work = h_start.copy()
    for _ in range(max_bisect):
        mid = (lo + hi) / 2.0
        try:
            res = minimize(_lagr_obj, h_work, args=(alpha, mid, n),
                          method='L-BFGS-B', bounds=bounds, maxiter=60)
            h_new = np.clip(res.x, 0, 1)
        except Exception:
            break
        s = h_new.sum()
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 0.005:
            break
        h_work = h_new
    return _fix_sum(best_h, target)

def _continuation(h_start, n, a0=20, a1=3000, steps=8, use_lagr=False):
    h = h_start.copy()
    alphas = np.geomspace(a0, a1, steps)
    for alpha in alphas:
        if use_lagr:
            h_new = _opt_lagrangian(h, n, int(alpha), max_bisect=25)
        else:
            h_new = _opt_slsqp(h, n, int(alpha), maxiter=200, ftol=1e-13)
        h = _fix_sum(h_new, n / 2.0)
    return h

def _sa(h, n, n_iter=1000, T0=0.05, T1=0.0001, seed=42):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    for i in range(n_iter):
        T = T0 * (T1 / T0) ** (float(i) / max(n_iter, 1))
        i1 = int(rng.randint(n))
        i2 = int(rng.randint(n))
        if i1 == i2:
            continue
        delta = rng.uniform(-0.1, 0.1)
        h1 = h[i1] + delta
        h2 = h[i2] - delta
        if 0.0 <= h1 <= 1.0 and 0.0 <= h2 <= 1.0:
            h[i1] = h1
            h[i2] = h2
            c5_new = _c5_of(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            elif rng.rand() > np.exp(-(c5_new - best_c5) / max(T, 1e-12)):
                h[i1] = h1 - delta
                h[i2] = h2 + delta
        if rng.rand() < 0.05:
            mask = rng.rand(n) < 0.1
            cnt = int(mask.sum())
            if cnt > 0:
                h[mask] = np.clip(h[mask] + rng.uniform(-0.15, 0.15, cnt), 0, 1)
                h = _fix_sum(h, target)
    return best_h, best_c5

def _make_h(n, kind, seed=42):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    if kind == 0:
        levels = 3 + int(rng.randint(0, 5))
        h = np.zeros(n)
        seg = max(n // levels, 1)
        for k in range(levels):
            val = 1.0 - k / max(levels - 1, 1)
            s = k * seg
            e = min((k + 1) * seg, n) if k < levels - 1 else n
            h[s:e] = val
    elif kind == 1:
        x = np.linspace(0, 1, n)
        p = 0.5 + rng.rand() * 2.5
        h = 1.0 - np.abs(2.0 * x - 1.0) ** p
    elif kind == 2:
        x = np.linspace(0, 1, n)
        f = 1 + int(rng.randint(0, 4))
        h = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * x + rng.rand() * 2.0 * np.pi)
    elif kind == 3:
        h = rng.rand(n)
    elif kind == 4:
        h = np.linspace(1.0, 0.0, n)
    elif kind == 5:
        x = np.linspace(0, 1, n)
        h = 1.0 - (2.0 * x - 1.0) ** 2
    else:
        h = np.full(n, 0.5)
    return _fix_sum(h, target)

def _perturb_optimize(h, n, scale, alpha, seed, maxiter=200):
    rng = np.random.RandomState(seed)
    h_pert = h + scale * rng.randn(n)
    h_pert = _fix_sum(h_pert, n / 2.0)
    h_new = _opt_slsqp(h_pert, n, alpha, maxiter=maxiter, ftol=1e-13)
    return _fix_sum(h_new, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_par = np.array(initial_h_values, dtype=float).copy()
    else:
        h_par = np.full(86, 0.5)
    n_par = len(h_par)
    h_par = _fix_sum(h_par, n_par / 2.0)

    best_h = h_par.copy()
    best_n = n_par
    best_c5 = _c5_of(h_par)

    # Phase 1: Continuation on parent with SLSQP
    if deadline - time.time() > 120:
        h_new = _continuation(h_par, n_par, 20, 5000, 10, use_lagr=False)
        h_new = _fix_sum(h_new, n_par / 2.0)
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 2: Lagrangian method
    if deadline - time.time() > 100:
        h_new = _opt_lagrangian(best_h, best_n, 200, max_bisect=30)
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 3: Try different n_points
    n_candidates = sorted(set([best_n] + list(range(max(55, best_n - 20), best_n + 30, 3))))
    for n_t in n_candidates[:14]:
        if deadline - time.time() < 45:
            break
        if n_t == n_par:
            h_init = h_par.copy()
        else:
            h_init = np.interp(np.linspace(0, 2, n_t, endpoint=False),
                              np.linspace(0, 2, n_par, endpoint=False), h_par)
            h_init = _fix_sum(h_init, n_t / 2.0)
        h_new = _continuation(h_init, n_t, 30, 1500, 6, use_lagr=False)
        h_new = _fix_sum(h_new, n_t / 2.0)
        c5 = _c5_of(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5, h_new.copy(), n_t

    # Phase 4: Multi-start with diverse initializations
    if deadline - time.time() > 150:
        for kind in range(7):
            for s in [seed, seed + 7, seed + 13]:
                if deadline - time.time() < 25:
                    break
                h_init = _make_h(best_n, kind, seed=s)
                h_new = _continuation(h_init, best_n, 30, 600, 5, use_lagr=False)
                h_new = _fix_sum(h_new, best_n / 2.0)
                c5 = _c5_of(h_new)
                if c5 < best_c5 - 1e-12:
                    best_c5, best_h = c5, h_new.copy()

    # Phase 5: Adaptive perturbation + reoptimize
    if deadline - time.time() > 120:
        rng = np.random.RandomState(seed + 100)
        for p_round in range(8):
            if deadline - time.time() < 25:
                break
            scale = 0.02 + 0.03 * rng.rand()
            a_try = int(100 + 200 * rng.rand())
            s_try = seed + int(rng.randint(0, 1000))
            h_new = _perturb_optimize(best_h, best_n, scale, a_try, s_try, maxiter=150)
            c5 = _c5_of(h_new)
            if c5 < best_c5 - 1e-12:
                best_c5, best_h = c5, h_new.copy()

    # Phase 6: Simulated annealing
    if deadline - time.time() > 80:
        iters = min(3000, max(500, int((deadline - time.time()) * 3)))
        h_sa, c5_sa = _sa(best_h, best_n, n_iter=iters, T0=0.05, T1=0.0001, seed=seed)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
        if deadline - time.time() > 35:
            h_new = _opt_slsqp(best_h, best_n, 500, maxiter=300, ftol=1e-14)
            h_new = _fix_sum(h_new, best_n / 2.0)
            c5 = _c5_of(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 7: Lagrangian continuation
    if deadline - time.time() > 60:
        h_new = _continuation(best_h, best_n, 50, 2000, 6, use_lagr=True)
        h_new = _fix_sum(h_new, best_n / 2.0)
        c5 = _c5_of(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 8: Final polish with high alpha
    if deadline - time.time() > 30:
        for a in [1000, 2000, 4000]:
            if deadline - time.time() < 12:
                break
            h_new = _opt_slsqp(best_h, best_n, a, maxiter=500, ftol=1e-15)
            h_new = _fix_sum(h_new, best_n / 2.0)
            c5 = _c5_of(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 9: Second SA + re-optimize
    if deadline - time.time() > 50:
        h_sa, c5_sa = _sa(best_h, best_n, n_iter=min(1500, int((deadline - time.time()) * 2)),
                          T0=0.03, T1=0.00005, seed=seed + 500)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
        if deadline - time.time() > 20:
            h_new = _opt_slsqp(best_h, best_n, 2000, maxiter=400, ftol=1e-15)
            h_new = _fix_sum(h_new, best_n / 2.0)
            c5 = _c5_of(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
