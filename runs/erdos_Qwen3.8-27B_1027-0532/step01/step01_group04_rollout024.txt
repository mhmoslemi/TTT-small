```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
```

```python
def simulated_annealing_binary(b, time_limit, rng, T_init=1.0, T_min=0.01, cooling=0.999):
    n = len(b)
    start = time.time()
    T = T_init
    corr = compute_corr(b)
    current_energy = float(np.max(corr))
    best_b = b.copy()
    best_energy = current_energy
    
    while time.time() - start < time_limit and T > T_min:
        # Pick random swap
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        b[i] = 0.0
        b[j] = 1.0
        corr_new = compute_corr(b)
        new_energy = float(np.max(corr_new))
        delta = new_energy - current_energy
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_energy = new_energy
            corr = corr_new
            if new_energy < best_energy:
                best_energy = new_energy
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        
        T *= cooling
    
    return best_b, best_energy
```

- But increases corr[0] by approximately ε * (2 - 2ε) * (number of changed positions)

The tradeoff depends on the specific structure.

Key optimization: Instead of recomputing the full correlation from scratch each time, I can compute the change in correlation due to a single swap.

When we swap b[i] from 1→0 and b[j] from 0→1:

corr[k] = sum_t b[t] * (1-b[t-k])

Term involving b[i]: b[i] * (1-b[i-k]) → changes from (1-b[i-k]) to 0. Change: -(1-b[i-k])
Term involving b[j]: b[j] * (1-b[j-k]) → changes from 0 to (1-b[j-k]). Change: +(1-b[j-k])

But also, b[i] and b[j] appear in the (1-b[...]) part:
Term where t-k = i, i.e., t = i+k: b[i+k] * (1-b[i]) → changes from 0 to b[i+k]. Change: +b[i+k]
Term where t-k = j, i.e., t = j+k: b[j+k] * (1-b[j]) → changes from b[j+k] to 0. Change: -b[j+k]

So total change in corr[k]:
Δcorr[k] = -(1-b[i-k]) + b[i+k] + (1-b[j-k]) - b[j+k]
          (all with appropriate boundary checks: index must be in [0, n-1])

This is O(1) per lag k! So the total change computation is O(n) (over all lags).

But we don't need to compute the full new correlation - we just need to check if the max decreases. We can:
1. Compute the current max
2. Compute Δcorr[k] for all k
3. new_corr[k] = corr[k] + Δcorr[k]
4. new_max = max(new_corr)

This is O(n) per swap attempt, which is much faster than O(n²) for full correlation.

```python
def compute_delta_corr(b, i, j, n):
    """Compute the change in correlation when swapping b[i]:1→0 and b[j]:0→1"""
    # corr[k] = sum_t b[t] * (1-b[t-k]) for k = -(n-1) to n-1
    # In the 'full' output, index = k + (n-1)
    
    delta = np.zeros(2*n - 1)
    
    # From b[i] changing 1→0:
    # -(1-b[i-k]) for valid i-k
    # +b[i+k] for valid i+k
    
    # From b[j] changing 0→1:
    # +(1-b[j-k]) for valid j-k
    # -b[j+k] for valid j+k
    
    # For each k from -(n-1) to n-1:
    # The index in the full correlation is k + (n-1)
    
    for k in range(-(n-1), n):
        idx = k + (n - 1)
        d = 0.0
        # Term: -(1 - b[i-k]) if 0 <= i-k < n
        if 0 <= i - k < n:
            d -= (1.0 - b[i - k])
        # Term: +b[i+k] if 0 <= i+k < n
        if 0 <= i + k < n:
            d += b[i + k]
        # Term: +(1 - b[j-k]) if 0 <= j-k < n
        if 0 <= j - k < n:
            d += (1.0 - b[j - k])
        # Term: -b[j+k] if 0 <= j+k < n
        if 0 <= j + k < n:
            d -= b[j + k]
        delta[idx] = d
    
    return delta
```

```python
    start = time.time()

    corr = compute_corr(b)
    current_energy = float(np.max(corr))
    best_b = b.copy()
    best_energy = current_energy
    
    T = T_init
    no_improve = 0
```
```python
    max_no_improve = 200000
```

```python
    zeros_list = [int(x) for x in np.where(b == 0.0)[0]]

    pos_in_zeros = [-1] * n
    for idx, val in enumerate(ones_list):
        pos_in_ones[val] = idx
    for idx, val in enumerate(zeros_list):
        pos_in_zeros[val] = idx
    
    def swap_remove(arr, pos_map, elem, n_elem):
        """Remove elem from arr, maintaining pos_map. Returns new n_elem"""
        pos = pos_map[elem]
        last = arr[n_elem - 1]
        arr[pos] = last
        pos_map[last] = pos
        pos_map[elem] = -1
        return n_elem - 1
    
    def add(arr, pos_map, elem, n_elem):
        """Add elem to arr at position n_elem"""
        arr[n_elem] = elem
        pos_map[elem] = n_elem
        return n_elem + 1
    
    ones_arr_ext = ones_list + [0] * n  # extra space for additions
    zeros_arr_ext = zeros_list + [0] * n
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    while time.time() - start < budget_s and T > T_min:
        if no_improve > max_no_improve:
            break
        
        i = int(ones_arr_ext[rng.randint(n_ones)])
        j = int(zeros_arr_ext[rng.randint(n_zeros)])
```

```python
    n_zeros = len(zeros_arr)

    pos_in_ones = np.full(n, -1, dtype=np.int32)
    pos_in_zeros = np.full(n, -1, dtype=np.int32)
    for idx in range(n_ones):
        pos_in_ones[ones_arr[idx]] = idx
    for idx in range(n_zeros):
        pos_in_zeros[zeros_arr[idx]] = idx

    ones_buf = np.zeros(n, dtype=np.int32)
    zeros_buf = np.zeros(n, dtype=np.int32)
    ones_buf[:n_ones] = ones_arr
    zeros_buf[:n_zeros] = zeros_arr

    exp_cache = {}

    while time.time() - start < budget_s and T > T_min:
        if no_improve > max_no_improve:
            T = T_init * 0.5
            no_improve = 0
            if T < T_min:
                break

        i = int(ones_buf[rng.randint(n_ones)])
        j = int(zeros_buf[rng.randint(n_zeros)])

        b[i] = 0.0
        b[j] = 1.0
        corr = compute_corr(b)
        new_energy = energy_topk(corr, k_top)
        delta = new_energy - current_energy

        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            exp_val = np.exp(-delta / T)
            if rng.random() < exp_val:
                accept = True

        if accept:
            current_energy = new_energy
            pi = int(pos_in_ones[i])
            pj = int(pos_in_zeros[j])
            last_o = int(ones_buf[n_ones - 1])
            last_z = int(zeros_buf[n_zeros - 1])
            ones_buf[pi] = last_o
            pos_in_ones[last_o] = pi
            pos_in_ones[i] = -1
            n_ones -= 1
            zeros_buf[pj] = last_z
            pos_in_zeros[last_z] = pj
            pos_in_zeros[j] = -1
            n_zeros -= 1
            ones_buf[n_ones] = j
            pos_in_ones[j] = n_ones
            n_ones += 1
            zeros_buf[n_zeros] = i
            pos_in_zeros[i] = n_zeros
            n_zeros += 1

            new_max = float(np.max(corr))
            if new_energy < best_energy:
                best_energy = new_energy
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            b[i] = 1.0
            b[j] = 0.0

        T *= 0.99999

    final_corr = compute_corr(best_b)
    final_max = float(np.max(final_corr))
    return best_b, final_max

def greedy_local_search(b, budget_s, rng):
    n = len(b)
    start = time.time()
    corr = compute_corr(b)
    max_val = float(np.max(corr))
    best_b = b.copy()
    best_max = max_val
    no_improve = 0

    while time.time() - start < budget_s and no_improve < 8000:
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)

        # Find contributing positions
        if lag >= 0:
            contrib = [i for i in range(lag, n) if b[i] == 1.0 and b[i - lag] == 0.0]
        else:
            contrib = [i for i in range(0, n + lag) if b[i] == 1.0 and b[i - lag] == 0.0]

        if not contrib:
            no_improve += 1
            continue

        best_delta = 0
        best_pair = None

        n_tries = min(len(contrib), 80)
        zeros = np.where(b == 0.0)[0]
        zeros = zeros[zeros != contrib[0]]  # avoid trivial
        if len(zeros) == 0:
            no_improve += 1
            continue

        if len(contrib) > n_tries:
            trial = [contrib[rng.randint(len(contrib))] for _ in range(n_tries)]
        else:
            trial = contrib

        for pos in trial:
            n_tries_z = min(len(zeros), 40)
            if len(zeros) > n_tries_z:
                trial_z = [zeros[rng.randint(len(zeros))] for _ in range(n_tries_z)]
            else:
                trial_z = list(zeros)

            for pos2 in trial_z:
                if pos2 == pos:
                    continue
                b[pos] = 0.0
                b[pos2] = 1.0
                corr_new = compute_corr(b)
                max_new = float(np.max(corr_new))
                d = max_new - max_val
                if d < best_delta:
                    best_delta = d
                    best_pair = (pos, pos2)
                b[pos] = 1.0
                b[pos2] = 0.0

        if best_pair is not None and best_delta < -1e-10:
            pos, pos2 = best_pair
            b[pos] = 0.0
            b[pos2] = 1.0
            corr = compute_corr(b)
            max_val = float(np.max(corr))
            no_improve = 0
            if max_val < best_max:
                best_max = max_val
                best_b = b.copy()
        else:
            no_improve += 1

    return best_b, best_max

def fractional_refine(h, budget_s, rng):
    n = len(h)
    ts = n / 2.0
    _g_n = n
    _g_target_sum = ts
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]

    best_h = h.copy()
    best_score = compute_c5(best_h)
    start = time.time()

    alphas = [20, 50, 100, 200, 500]
    for alpha in alphas:
        if time.time() - start > budget_s:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new
        except Exception:
            pass

    # Try perturbed starts
    for _ in range(5):
        if time.time() - start > budget_s:
            break
        noise = rng.randn(n) * 0.05
        h_pert = project_feasible(best_h + noise, ts)
        for alpha in [50, 200]:
            if time.time() - start > budget_s:
                break
            try:
                res = minimize(obj_smooth, h_pert, args=(alpha,),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
            except Exception:
                pass

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target_sum
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 10.0 - (time.time() - start_time)

    # Phase 1: SA on binary from initial (even n)
    n1 = n_init if n_init % 2 == 0 else n_init - 1
    x_old = np.linspace(0.0, 1.0, n_init)
    x_new = np.linspace(0.0, 1.0, n1)
    h_interp = np.interp(x_new, x_old, h_init)
    b1 = binary_ize(h_interp, n1)

    t_alloc1 = min(350, time_left() * 0.35)
    b1, max1 = sa_binary(b1, t_alloc1, rng, k_top=10)
    score1 = max1 * 2.0 / n1
    if score1 < best_score:
        best_score = score1
        best_h = b1.copy()

    # Phase 2: Greedy local search on best binary
    if time_left() > 100:
        n_cur = len(best_h)
        if n_cur % 2 == 0 and int(np.sum(best_h)) == n_cur // 2:
            t_alloc2 = min(200, time_left() * 0.2)
            b2, max2 = greedy_local_search(best_h.copy(), t_alloc2, rng)
            score2 = max2 * 2.0 / n_cur
            if score2 < best_score:
                best_score = score2
                best_h = b2.copy()

    # Phase 3: Try multiple even n values with SA
    n_candidates = [330, 332, 334, 336, 254, 256, 510, 512]
    for n_try in n_candidates:
        if time_left() < 80:
            break
        t_alloc = min(80, time_left() * 0.15)
        x_best = np.linspace(0.0, 1.0, len(best_h))
        x_try = np.linspace(0.0, 1.0, n_try)
        h_try = np.interp(x_try, x_best, best_h)
        b_try = binary_ize(h_try, n_try)
        b_try, max_try = sa_binary(b_try, t_alloc, rng, k_top=8)
        score_try = max_try * 2.0 / n_try
        if score_try < best_score:
            best_score = score_try
            best_h = b_try.copy()

    # Phase 4: Fractional refinement
    if time_left() > 80:
        n_best = len(best_h)
        t_alloc4 = min(250, time_left() * 0.3)
        best_h, best_score = fractional_refine(best_h, t_alloc4, rng)

    # Phase 5: More SA rounds on best (reheat)
    if time_left() > 60:
        n_best = len(best_h)
        if n_best % 2 == 0:
            b_cur = binary_ize(best_h, n_best)
            t_alloc5 = min(100, time_left() * 0.25)
            b_cur, max_cur = sa_binary(b_cur, t_alloc5, rng, k_top=5)
            score_cur = max_cur * 2.0 / n_best
            if score_cur < best_score:
                best_score = score_cur
                best_h = b_cur.copy()

    # Phase 6: Random restarts
    while time_left() > 40:
        n_try = int(rng.choice([330, 332, 334]))
        b_rand = np.zeros(n_try)
        idx = rng.choice(n_try, size=n_try // 2, replace=False)
        b_rand[idx] = 1.0
        t_alloc = min(25, time_left() * 0.3)
        b_rand, max_rand = sa_binary(b_rand, t_alloc, rng, k_top=10)
        score_rand = max_rand * 2.0 / n_try
        if score_rand < best_score:
            best_score = score_rand
            best_h = b_rand.copy()

    # Final fractional polish
    if time_left() > 30:
        n_final = len(best_h)
        best_h, best_score = fractional_refine(best_h, min(60, time_left() * 0.5), rng)

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
```
