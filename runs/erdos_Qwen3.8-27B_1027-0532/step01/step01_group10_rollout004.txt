import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def project_sum(h, target):
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_max(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(corr - m))) / beta)


def grad_smooth_max(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-16:
            continue
        lo1 = max(0, n - 1 - m)
        hi1 = min(n - 1, 2 * (n - 1) - m)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            a = idx + m - (n - 1)
            grad[idx] += w * (1.0 - h[a])
        lo2 = max(0, m - (n - 1))
        hi2 = min(n - 1, m)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            b = idx - m + (n - 1)
            grad[idx] -= w * h[b]
    return dx * grad


def fista_optimize(h0, n, dx, beta, max_iter, L_init=4.0):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    L = L_init
    f_prev = smooth_max(h, dx, beta)
    stale = 0
    for it in range(max_iter):
        g = grad_smooth_max(y, dx, beta)
        z = y - g / L
        h_new = project_sum(z, n * 0.5)
        f_new = smooth_max(h_new, dx, beta)
        if f_new > f_prev + 0.5 / L * np.dot(h_new - h, h_new - h) + 1e-14:
            L = L * 1.5
            z = y - g / L
            h_new = project_sum(z, n * 0.5)
            f_new = smooth_max(h_new, dx, beta)
        else:
            L = max(L * 0.85, L_init * 0.3)
        if f_new < f_prev - 1e-13:
            stale = 0
        else:
            stale += 1
        if stale > 200 and it > 100:
            break
        f_prev = f_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        mom = (t - 1.0) / t_new
        y = h_new + mom * (h_new - h)
        t = t_new
        h = h_new
    return h


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)


def objective_nm(h, dx, n, beta, penalty):
    base = smooth_max(h, dx, beta)
    sv = abs(np.sum(h) - n * 0.5)
    bv = np.sum(np.maximum(0.0, h - 1.0)) + np.sum(np.maximum(0.0, -h))
    return base + penalty * (sv + bv)


def run_slsqp(h0, n, dx, beta, maxiter):
    bounds = [(0.0, 1.0)] * n

    def obj_fn(h):
        return smooth_max(h, dx, beta)

    def con_fun(h):
        return np.sum(h) - n * 0.5

    res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None:
        return repair_h(res.x, n)
    return h0
