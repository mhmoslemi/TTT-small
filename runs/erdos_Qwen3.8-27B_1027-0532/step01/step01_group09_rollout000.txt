def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()


def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full') * dx
    corr_max = corr.max()
    shifted = T * (corr - corr_max)
    shifted = np.minimum(shifted, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    obj = corr_max + np.log(sum_exp) / T
    weights = exp_s / sum_exp
    j = np.arange(n)
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-7:
            continue
        k = ki - (n - 1)
        a = j - k
        b = j + k
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += w * dx * (1.0 - h[a[m1]])
        grad[m2] -= w * dx * h[b[m2]]
    return obj, grad
