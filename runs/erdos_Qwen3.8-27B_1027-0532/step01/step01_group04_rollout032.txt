    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)

        k = k_idx - (n - 1)

        # d/dh_j: if j is in first position: (1-h[j+k]) for valid j+k
        #          if j is in second position: -h[j-k] for valid j-k
        # Using np.correlate convention:
        # result[k_idx] = sum_i a[i] * b[i + k_idx - n + 1]
        # where a = h, b = 1-h

        h_new = h - lr * grad

        # = sum_{i=0}^{n-1} h[i]*(1-h[i+lag]) where i+lag = i-m >= 0 means i >= m
        # d/dh[j]:
        #   j in [m, n-1]: contributes (1-h[j-m]) (first factor)
        #   j in [0, n-1-m]: contributes -h[j+m] (second factor, since h[i-m] with i-m = j means i = j+m)
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

def get_value_lag(h, lag, n):
    """Compute corr value at given lag."""
    k_idx = lag + n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[k_idx]

def sca_solve(h, t_current, n_active=30, max_iter_sca=50):
    """Solve the SCA (successive convex approximation) problem."""
    n = len(h)
    best_h = h.copy()
    best_t = t_current
    
    for sca_it in range(max_iter_sca):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Select active lags (top n_active by correlation value)
        idx = np.argsort(corr)[::-1][:n_active]
        
        # Setup LP:
        # Variables: h[0], ..., h[n-1], t
        # Minimize: t
        # Subject to:
        #   g_k^T h - t <= f_k(h^t) - g_k^T h^t for each active lag k
        #   0 <= h_i <= 1
        #   sum(h) = n/2
        
        # LP formulation for scipy.optimize.linprog:
        # c vector: minimize c^T x where x = [h_0, ..., h_{n-1}, t]
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize t
        
        # Inequality constraints: A_ub @ x <= b_ub
        A_ub_rows = []
        b_ub_vals = []
        
        for k_idx in idx:
            lag = k_idx - (n - 1)
            g = get_gradient_lag(best_h, lag, n)
            f_val = best_h  # need actual value
            # f_val = corr[k_idx] (we already have it)
            f_val = corr[k_idx]
            g_dot_h = np.dot(g, best_h)
            rhs = f_val - g_dot_h
            
            # g^T h - t <= rhs
            # [g, -1] @ [h, t] <= rhs
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(rhs)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # Equality constraints: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h_i <= 1, t unbounded below (but we know t >= 0)
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                t_new = res.x[-1]
                # Check actual objective
                actual_t = compute_c5(h_new) / (2.0/n)  # unscaled
                # Actually, let me just check compute_c5
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(best_h):
                    best_h = h_new
                    best_t = c5_new
                else:
                    # Step back: take a fraction of the way
                    h_blend = 0.5 * best_h + 0.5 * h_new
                    h_blend = np.clip(h_blend, 0, 1)
                    # Project to feasible
                    diff = n/2.0 - np.sum(h_blend)
                    if abs(diff) > 1e-10:
                        if diff > 0:
                            mask = h_blend < 1.0
                        else:
                            mask = h_blend > 0.0
                        h_blend[mask] += diff / np.sum(mask)
                        h_blend = np.clip(h_blend, 0, 1)
                    c5_blend = compute_c5(h_blend)
                    if c5_blend < compute_c5(best_h):
                        best_h = h_blend
                        best_t = c5_blend
                    else:
                        # Stuck, break
                        break
            else:
                break
        except Exception:
            break
    
    return best_h

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = int(np.sum(mask))
    if count == 0:
        return h
    h = h.copy()
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

    """Gradient of corr[k_idx] w.r.t. h."""

    grad = np.zeros(n)

        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-lag]

        m = -lag
