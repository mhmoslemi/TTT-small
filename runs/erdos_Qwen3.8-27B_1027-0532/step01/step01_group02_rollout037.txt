```python
def project_to_polytope(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on tau
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_polytope(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on offset."""
    n = len(h)
    lo = -1.0
    hi = 1.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_polytope(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def lse_value(h, beta):
    ov = compute_overlaps(h)
    m = np.max(ov)
    return float(m + np.log(np.sum(np.exp(beta * (ov - m)))) / beta)

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    ov = compute_overlaps(h)
    ov_max = np.max(ov)
    exp_ov = np.exp(beta * (ov - ov_max))
    p = exp_ov / np.sum(exp_ov)
    
    G = np.zeros(n)
    # Shift 0
    G += p[n-1] * (1.0 - 2.0 * h) * dx
    
    # Right shifts (k=1 to n-1)
    # p_right[k] = p[n-1+k] for k=1,...,n-1
    for k in range(1, n):
        pk = p[n-1+k]
        if pk < 1e-14:
            continue
        G[k:] += pk * (1.0 - h[:n-k]) * dx
        if n-k > 0:
            G[:n-k] -= pk * h[k:] * dx
    
    # Left shifts (k=1 to n-1)
    # p_left[k] = p[n-1-k] for k=1,...,n-1
    for k in range(1, n):
        pk = p[n-1-k]
        if pk < 1e-14:
            continue
        if n-k > 0:
            G[:n-k] += pk * (1.0 - h[k:]) * dx
        G[k:] -= pk * h[:n-k] * dx
    
    return G, ov

def lse_projected_gradient(h_init, beta, max_iter=10000, lr=0.5, momentum=0.9):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)
    h_prev = h.copy()
    
    best_h = h.copy()
    best_val = lse_value(h, beta)
    
    for it in range(max_iter):
        G, ov = lse_gradient(h, beta)
        
        # Momentum update
        v = momentum * v - lr * G
        h_new = project_to_polytope(h + v, target)
        
        # Check improvement (every 10 steps to save time)
        if it % 10 == 0:
            val = lse_value(h_new, beta)
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        
        h = h_new
    
    # Final evaluation
    val = lse_value(h, beta)
    if val < best_val:
        best_h = h.copy()
    
    return best_h

def frank_wolfe_subgrad(h_init, max_iter=10000):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    
    for t in range(1, max_iter + 1):
        ov = compute_overlaps(h)
        max_val = np.max(ov)
        # Active set
        active = np.where(np.abs(ov - max_val) < 1e-8)[0]
        
        # Compute subgradient as average of active gradients
        G = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)  # shift
            G += _subgradient_single_shift(h, k)
        G /= len(active)
        
        # Frank-Wolfe step
        gamma = 2.0 / (t + 2.0)
        # s = vertex of polytope in direction -G
        # s[i] = 1 for the n/2 smallest G[i], 0 otherwise
        sorted_idx = np.argsort(G)
        s = np.zeros(n)
        s[sorted_idx[:n//2]] = 1.0
        
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = project_to_polytope(h_new, target)
        h = h_new
    
    return h

def _subgradient_single_shift(h, k):
    """Compute gradient of overlap at shift k."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros(n)
    
    if k >= 0:
        # ov(k) = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        # d/dh[j] = (1-h[j-k]) * [j>=k] + (-h[j+k]) * [j+k<n]
        if k == 0:
            G[:] = (1.0 - 2.0 * h) * dx
        else:
            G[k:] += (1.0 - h[:n-k]) * dx
            G[:n-k] -= h[k:] * dx
    else:
        # k < 0, let k' = -k
        kp = -k
        # ov(-kp) = sum_{i=0}^{n-1-kp} h[i] * (1-h[i+kp])
        # d/dh[j] = (1-h[j+kp]) * [j<n-kp] + (-h[j-kp]) * [j>=kp]
        G[:n-kp] += (1.0 - h[kp:]) * dx
        G[kp:] -= h[:n-kp] * dx
    
    return G
```
