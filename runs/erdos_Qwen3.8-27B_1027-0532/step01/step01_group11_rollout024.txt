def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(corr ** p) ** (1.0 / p))

            mask = rng.random(n) < CR
            mask[r] = True
            trial = np.where(mask, trial, pop[i])
            trial = project_onto_constraint(trial, target_sum)

                if fit_t < best_c5:
                    best_c5 = fit_t
                    best_h = trial.copy()
                    stagnation = 0
                else:
                    stagnation += 1

def _smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def _smooth_grad(h, eps):
    return compute_gradient(h, eps)

def _con_fun(h):
    return h.sum() - len(h) / 2.0

def _con_jac(h):
    return np.ones(len(h))

def compute_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    g = raw * dx
    cmax = np.max(g)
    x = (g - cmax) / eps
    x -= np.max(x)
    w = np.exp(x)
    w /= w.sum()

        time_for_de2 = min(remaining - 150, 250)
        deadline2 = start_time + time_for_de2
        bh2, bc52, pop2, fit2 = de_search(pop2, fit2, n2, target2, rng, deadline2)

        remaining2 = budget_s - (time.time() - start_time)
        if remaining2 > 60:
            for eps in [0.005, 0.002, 0.001, 0.0005, 0.0001]:
                if time.time() - start_time > budget_s - 20:
                    break
                h_new, c5_new = grad_refine(bh2, target2, 100, eps)
                if c5_new < bc52:
                    bc52 = c5_new
                    bh2 = h_new
            if bc52 < best_c5:
                best_c5 = bc52
                best_h = bh2

    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        h_final, c5_final = grad_refine(best_h, len(best_h) / 2.0, 80, 0.00005)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
