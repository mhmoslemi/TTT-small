def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

    m = corr.max()
    shifted = np.clip(tau * (corr - m), -500, 500)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    val = m + np.log(total) / tau
    return val, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for ki in range(len(weights)):
        w = weights[ki]
        if w < 1e-14:
            continue
        k = ki - (n - 1)
        if k >= 0:
            idx = np.arange(0, n - k)
            grad[idx] += w * dx * (1.0 - h[idx + k])
            grad[idx + k] -= w * dx * h[idx]
        else:
            idx = np.arange(-k, n)
            grad[idx] += w * dx * (1.0 - h[idx + k])
            grad[idx + k] -= w * dx * h[idx]
    return grad
