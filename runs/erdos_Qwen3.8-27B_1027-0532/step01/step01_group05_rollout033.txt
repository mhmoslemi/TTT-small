def subgradient_polish(h, n, n_iter=500, initial_step=0.3):

    h = h.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()

    step = initial_step

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] += -h[lag:]
        else:
            l = -lag
            g[:n - l] += (1.0 - h[l:])
            g[l:] += -h[:n - l]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            v = 0.5 * v
            step *= 0.95
        
        step *= 0.9995
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def get_smoothed_obj_and_weights(corr, t):
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    smooth_max = max_corr + np.log(sum_exp) / t
    weights = exp_vals / sum_exp
    return smooth_max, weights


def compute_grad_smoothed(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    active = weights > 1e-10
    active_indices = np.where(active)[0]
    for k in active_indices:
        w = weights[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag]) * w
            grad[:n - lag] += -h[lag:] * w
        else:
            l = -lag
            grad[:n - l] += (1.0 - h[l:]) * w
            grad[l:] += -h[:n - l] * w
    grad *= dx
    return grad


def lbfgsb_optimize(h0, n, t, mu, max_iter=500):
    dx = 2.0 / n
    target = n / 2.0

    def obj_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        smooth_max, _ = get_smoothed_obj_and_weights(corr, t)
        penalty = np.sum(h) - target
        return smooth_max * dx + 0.5 * mu * penalty * penalty

    def grad_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        smooth_max, weights = get_smoothed_obj_and_weights(corr, t)
        grad = compute_grad_smoothed(h, n, weights)
        penalty = np.sum(h) - target
        grad += mu * penalty * np.ones(n)
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_func, h0.copy(), method='L-BFGS-B',
                          jac=grad_func, bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
        return result.x
    except Exception:
        return h0.copy()


def slsqp_optimize(h0, n, t, max_iter=300):
    dx = 2.0 / n
    target = n / 2.0

    def obj_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        smooth_max, _ = get_smoothed_obj_and_weights(corr, t)
        return smooth_max * dx

    def grad_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        smooth_max, weights = get_smoothed_obj_and_weights(corr, t)
        return compute_grad_smoothed(h, n, weights)

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        result = minimize(obj_func, h0.copy(), method='SLSQP',
                          jac=grad_func, bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-13})
        return np.clip(result.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

            h = h_new

            step *= 0.95

            h_sls = project_onto_constraint(h_sls, best_n)
            c5_sls = compute_c5(h_sls, best_n)
            if c5_sls < best_c5:
                best_c5 = c5_sls
                best_h = h_sls.copy()

    # Subgradient polish
    if time_left() > budget_s * 0.25:
        h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=600, initial_step=0.25)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        best_h = project_onto_constraint(best_h, best_n)

    # Phase 2: Grid refinement
    target_ns = []
    for mult in [2, 3, 4]:
        nn = n_parent * mult
        if nn <= 1200:
            target_ns.append(nn)

    for idx, n_new in enumerate(target_ns):
        if time_left() < budget_s * 0.15:
            break

        h_fine = interpolate_h(best_h, best_n, n_new)
        c5_fine = compute_c5(h_fine, n_new)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_new

        fine_configs = [
            (150, 500, 200),
            (80, 1000, 300),
            (40, 2000, 400),
            (20, 5000, 500),
            (10, 10000, 500),
        ]

        for t, mu, max_iter in fine_configs:
            if time_left() < budget_s * (0.08 + 0.04 * idx):
                break
            h_opt = lbfgsb_optimize(h_fine, n_new, t, mu, max_iter=max_iter)
            h_opt = project_onto_constraint(h_opt, n_new)
            c5_opt = compute_c5(h_opt, n_new)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_new
            if c5_opt <= compute_c5(h_fine, n_new) + 1e-8:
                h_fine = h_opt

        if time_left() > budget_s * 0.08:
            h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=400, initial_step=0.15)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        best_h = project_onto_constraint(best_h, best_n)

    # Phase 3: Random restarts and block perturbations
    if time_left() > budget_s * 0.10:
        n_try = best_n
        max_restarts = 40
        for i in range(max_restarts):
            if time_left() < budget_s * 0.02:
                break

            if i % 3 == 0:
                noise_scale = 0.01 + 0.015 * (i % 5)
                h_pert = best_h + rng.randn(n_try) * noise_scale
                h_pert = project_onto_constraint(h_pert, n_try)
            elif i % 3 == 1:
                block_size = max(5, n_try // 20)
                h_pert = block_perturb(best_h, n_try, rng, block_size)
            else:
                noise_scale = 0.005 + 0.01 * (i % 4)
                h_pert = best_h + rng.randn(n_try) * noise_scale
                h_pert = project_onto_constraint(h_pert, n_try)

            t_use = [40, 80, 150, 250][i % 4]
            mu_use = 5000
            h_opt = lbfgsb_optimize(h_pert, n_try, t_use, mu_use, max_iter=250)
            h_opt = project_onto_constraint(h_opt, n_try)
            c5_opt = compute_c5(h_opt, n_try)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

            if i % 5 == 4 and time_left() > budget_s * 0.05:
                h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=200, initial_step=0.1)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 4: Final intensive polish
    if time_left() > 30:
        n_iter = min(800, int(time_left() * 25))
        h_pol, c5_pol = subgradient_polish(best_h, best_n, n_iter=n_iter, initial_step=0.08)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 15:
        h_sls = slsqp_optimize(best_h, best_n, 5, max_iter=200)
        h_sls = project_onto_constraint(h_sls, best_n)
        c5_sls = compute_c5(h_sls, best_n)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()

    # Final verification
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
