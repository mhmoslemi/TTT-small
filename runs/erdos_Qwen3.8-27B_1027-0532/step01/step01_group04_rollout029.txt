def spectral_iteration(h, n_iter=100, warm_start=True):
    """FFT-based spectral flattening"""
    n = len(h)
    target_sum = n / 2.0
    g = h - 0.5
    
    for it in range(n_iter):
        G = np.fft.fft(g)
        power = np.abs(G)**2
        # Exclude DC component (index 0)
        mean_power = np.mean(power[1:])
        # Scale to flatten
        scale = np.sqrt(mean_power / np.maximum(power, 1e-12))
        scale[0] = 1.0  # Don't touch DC
        G_new = G * scale
        g_new = np.fft.ifft(G_new).real
        h_new = g_new + 0.5
        h_new = project_feasible(h_new, target_sum)
        if np.allclose(h_new, h, atol=1e-6):
            break
        h = h_new
    return h

def top_k_smooth(corr_values, k, alpha):
    """Smooth approximation of the top-k mean"""
    sorted_c = np.sort(corr_values)[::-1]
    top_k_mean = np.mean(sorted_c[:k])
    # Add a penalty for values outside top-k
    below = sorted_c[k:]
    penalty = np.sum(np.log1p(np.exp(alpha * (below - top_k_mean)))) / alpha
    return top_k_mean + penalty / len(corr_values)

def coordinate_descent_step(h, n_inner=10, rng=None):
    """One pass of coordinate descent: for each i, optimize h[i]"""
    n = len(h)
    ts = n / 2.0
    
    for i in range(n):
        # Compute a_k and b_k for all k
        # c_k = a_k * h[i] + b_k
        # c_k = (2/n) * [sum_j h[j]*(1-h[j-k]) ] where the sum includes j=i and j+k=i terms
        # Term with j=i: h[i]*(1-h[i-k])
        # Term with j+k=i, i.e., j = i-k: h[i-k]*(1-h[i])
        # All other terms don't depend on h[i]
        
        # b_k = (2/n) * [sum_{j≠i, j+k≠i} h[j]*(1-h[j-k])]
        # a_k = (2/n) * [(1-h[i-k]) - h[i+k]] (with boundary handling)
        
        # For efficiency, compute the "base" correlation (with h[i] set to 0)
        h_base = h.copy()
        h_base[i] = 0.0
        corr_base = np.correlate(h_base, 1.0 - h_base, mode='full') * (2.0/n)
        
        # Now, the contribution of h[i] to c_k:
        # When h[i] changes by δ:
        # Δc_k = (2/n) * [δ*(1-h[i-k]) - δ*h[i+k]] (with boundary handling for the two shifted terms)
        # where "(1-h[i-k])" is 0 if i-k is out of bounds, and "h[i+k]" is 0 if i+k is out of bounds
        
        # So c_k(h[i]=v) = corr_base[k] + v * (2/n) * [(1-h[i-k])_valid - h[i+k]_valid]
        
        # Compute the "slope" for each k
        shift = None  # we need to figure out the shift for each k index
        # For full correlation index idx, the shift is k = idx - (n-1)
        # The slope = (2/n) * [(1-h[i-k])_valid - h[i+k]_valid]
        
        slopes = np.zeros(2*n-1)
        for idx in range(2*n-1):
            k = idx - (n-1)  # the shift
            # (1-h[i-k]) if 0 ≤ i-k < n else 0
            if 0 <= i-k < n:
                term1 = 1.0 - h[i-k]
            else:
                term1 = 0.0
            # -h[i+k] if 0 ≤ i+k < n else 0
            if 0 ≤ i+k < n:
                term2 = -h[i+k]
            else:
                term2 = 0.0
            slopes[idx] = (2.0/n) * (term1 + term2)
        
        # Now c_k(v) = corr_base[idx] + v * slopes[idx] for each idx
        # Minimize max_idx (corr_base[idx] + v * slopes[idx]) over v ∈ [0,1]
        
        # Evaluate at a grid of v values
        v_candidates = np.linspace(0, 1, 20)
        best_v = 0
        best_obj = np.inf
        for v in v_candidates:
            values = corr_base + v * slopes
            obj = np.max(values)
            if obj < best_obj:
                best_obj = obj
                best_v = v
        
        h[i] = best_v
    
    # Re-project to ensure sum constraint
    h = project_feasible(h, ts)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)


def grad_for_shift(h, shift):
    n = len(h)
    if shift >= 0:
        term1 = np.zeros(n)
        if shift < n:
            term1[shift:] = h[:n - shift]
    else:
        term1 = np.zeros(n)
        term1[:n + shift] = h[-shift:]
    if shift >= 0:
        term2 = np.zeros(n)
        if shift < n:
            term2[:n - shift] = h[shift:]
    else:
        term2 = np.zeros(n)
        term2[-shift:] = h[:n + shift]
    return (2.0 / n) * (1.0 - term1 - term2)


def obj_smooth(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _current_target_sum

    return np.ones(_current_n)


def spectral_flatten(h, n_iter=100, strength=1.0):

    g = h.astype(float).copy() - 0.5

        power = np.abs(G) ** 2
        if n > 1:

        else:
            mean_power = power[0] + 1e-12
        scale = np.ones(n)
        nonzero = power > 1e-12

    while time_left() > 80 and n_starts < 8:
        n_start = len(best_h)
        init_type = rng.randint(3)
        if init_type == 0:
            h_try = make_near_binary(n_start, rng)
        elif init_type == 1:
            h_try = make_pseudorandom(n_start, rng)
        else:
            h_try = best_h + rng.randn(n_start) * 0.1
            h_try = project_feasible(h_try, n_start / 2.0)

        # Quick coordinate descent from this start
        h_try = coordinate_descent(h_try, n_passes=50, budget_check=time_left)
        s_try = compute_c5(h_try)

        # SLSQP refinement
        if time_left() > 40:
            h_try, s_try = slsqp_refine(h_try, [100], 300, time_left)

        if s_try < best_score:
            best_score = s_try
            best_h = h_try
        n_starts += 1

    # Phase 6: Try different grid sizes
    n_use = len(best_h)
    for n_try in [421, 503, 661, 211]:
        if time_left() < 60:
            break
        if n_try == n_use:
            continue
        ts_try = n_try / 2.0
        x_new = np.linspace(0.0, 2.0, n_try)
        x_old = np.linspace(0.0, 2.0, n_use)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        # Coordinate descent on new grid
        h_interp = coordinate_descent(h_interp, n_passes=100, budget_check=time_left)
        # SLSQP
        if time_left() > 30:
            h_interp, s_interp = slsqp_refine(h_interp, [50, 150], 500, time_left)
        else:
            s_interp = compute_c5(h_interp)

        if s_interp < best_score:
            best_score = s_interp
            best_h = h_interp
            n_use = n_try

    # Phase 7: Final refinement on best grid
    n_use = len(best_h)
    if time_left() > 30:
        best_h, best_score = slsqp_refine(best_h, [50, 100, 200, 500], 1000, time_left)

    # Phase 8: Final coordinate descent polish
    if time_left() > 20:
        h_polish = coordinate_descent(best_h, n_passes=100, budget_check=time_left)
        s_polish = compute_c5(h_polish)
        if s_polish < best_score:
            best_score = s_polish
            best_h = h_polish

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
