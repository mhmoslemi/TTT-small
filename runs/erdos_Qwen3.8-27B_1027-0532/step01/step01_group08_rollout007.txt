def lbfgs_optimize(h0, n, deadline, p=20, M=1000):
    target = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        one_minus_x = 1.0 - x
        corr = np.correlate(x, one_minus_x, mode="full") * dx
        # p-norm
        fp = np.power(np.maximum(corr, 0), p)
        F = np.power(np.sum(fp), 1.0/p)
        # Penalty
        penalty = M * (x.sum() - target) ** 2
        return F + penalty
    
    def gradient(x):
        one_minus_x = 1.0 - x
        corr = np.correlate(x, one_minus_x, mode="full") * dx
        # p-norm gradient
        fp = np.maximum(corr, 0)
        sum_fp = np.sum(np.power(fp, p))
        coeff = np.power(sum_fp, 1.0/p - 1.0)
        # sum_k f_k^(p-1) * df_k/dh[j]
        grad_F = np.zeros(n)
        for idx in range(2*n-1):
            if fp[idx] < 1e-15:
                continue
            w = fp[idx] ** (p-1) * coeff
            lag = idx - (n-1)
            j_lo = max(0, lag)
            j_hi = min(n-1, n-1+lag)
            if j_lo <= j_hi:
                grad_F[j_lo:j_hi+1] += w * (1.0 - x[j_lo-lag:j_hi-lag+1])
            j2_lo = max(0, -lag)
            j2_hi = min(n-1, n-1-lag)
            if j2_lo <= j2_hi:
                grad_F[j2_lo:j2_hi+1] += w * (-x[j2_lo+lag:j2_hi+lag+1])
        # Penalty gradient
        grad_penalty = 2.0 * M * (x.sum() - target) * np.ones(n)
        return grad_F * dx + grad_penalty
    
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    
    try:
        res = minimize(objective, h0, jac=gradient, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 200, 'ftol': 1e-15,
                                             'gtol': 1e-12, 'disp': False})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, target)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
