        return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha
    return m

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _golden_section_1d_args(f, lo, hi, tol, maxiter, args):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = float(lo)
    b = float(hi)
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = f(c_pt, *args)
    fd = f(d_pt, *args)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc <= fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f(c_pt, *args)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f(d_pt, *args)
    return (a + b) / 2.0

def _eval_coord_obj(t, v, coeffs, base_affected, has_special, idx_special, unaff_max):
    delta = t - v
    aff = base_affected + delta * coeffs
    if has_special:
        aff = aff.copy()
        aff[idx_special] = base_affected[idx_special] + delta * (1.0 - 2.0 * v) - delta * delta
    local_max = np.max(aff)
    return max(local_max, unaff_max)
