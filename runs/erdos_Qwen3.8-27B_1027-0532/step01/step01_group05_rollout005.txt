h_{k+1} = prox_{α*g}(y_k) = proj_C(y_k - α * grad(f)(y_k))

where prox is the proximal operator of the indicator of C, which is just the projection onto C.

But this requires that the gradient is evaluated at y_k, not h_k. And the line search should be at y_k.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_objective(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    return (max_corr + np.log(np.sum(np.exp(shifted))) / t) * dx

def compute_gradient(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    n_corr = len(corr)
    grad = np.zeros(n)
    for idx in range(n_corr):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] += -w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] += -w * h[:n - l]
    
    grad *= dx
    return grad

def fista_optimize(h0, n, t, max_iter=3000, tol=1e-10):
    h = project_onto_constraint(h0.astype(float).copy(), n)
    h_prev = h.copy()
    mu = 1.0  # FISTA momentum parameter
    alpha = 2.0  # step size
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    
    for it in range(max_iter):
        # FISTA momentum
        if it > 0:
            mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            y = h + ((mu - 1.0) / mu_new) * (h - h_prev)
            y = project_onto_constraint(y, n)
            mu = mu_new
        else:
            y = h.copy()
        
        g = compute_gradient(y, n, t)
        
        # Backtracking
        step = alpha
        while step > 1e-8:
            h_new = project_onto_constraint(y - step * g, n)
            obj_new = smooth_objective(h_new, n, t)
            obj_y = smooth_objective(y, n, t)
            if obj_new <= obj_y - 0.4 * step * np.dot(g, h_new - y):
                break
            step *= 0.5
        
        h_prev = h.copy()
        h = h_new
        
        obj = compute_c5(h, n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        
        if it > 100 and np.max(np.abs(h - h_prev)) < tol:
            break
    
    return best_h, best_obj

def combinatorial_search(h, n, max_rounds=20):
    """Local search on step function."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(n_ones - target):

    elif n_ones < target:

        order = np.argsort(-h[zero_idx])
        for i in range(target - n_ones):
            h_bin[zero_idx[order[i]]] = 1
    
    best_c5 = compute_c5(h_bin, n)
    best_h = h_bin.copy()

        improved = False
        corr = compute_corr(h_bin, n)
        max_c5 = np.max(corr) * (2.0 / n)

            new_corr = compute_corr(h_bin, n)
