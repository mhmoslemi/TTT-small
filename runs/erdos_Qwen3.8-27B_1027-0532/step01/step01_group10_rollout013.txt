import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask].max())
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = max(diff / cnt, -h[mask].min())
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def proj_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    return repair_h(h, n)

def subgradient_step(h, dx, alpha, top_m=10):
    """One step of subgradient method for minimizing max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Find top-m correlations
    idx = np.argsort(corr)[-top_m:]
    weights = np.zeros(len(corr))
    max_c = corr[idx[0]]
    for i in idx:
        # Soft weighting based on how close to max
        w = np.exp(-(max_c - corr[i]) / (0.01 * max_c + 1e-10))
        weights[i] = w
    weights /= np.sum(weights)
    
    # Compute gradient of weighted sum of correlations
    grad = np.zeros(n)
    for k in idx:
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)  # actual shift
        # c[k] = sum_i h[i] * (1-h[i+s]) for valid i
        i_start = max(0, -s)
        i_end = min(n - 1, n - 1 - s)
        for i in range(i_start, i_end + 1):
            j = i + s
            grad[i] += weights[k] * (1.0 - h[j]) * dx
            grad[j] += weights[k] * (-h[i]) * dx
    
    h_new = h - alpha * grad
    return proj_feasible(h_new, n)

def subgradient_method(h0, n, dx, max_iter, alpha0, top_m=10):
    """Run subgradient method."""
    h = h0.copy()
    best_c = c5_value(h, dx)
    best_h = h.copy()
    
    for it in range(max_iter):
        # Adaptive step size
        alpha = alpha0 / np.sqrt(1 + it * 0.01)
        
        h_new = subgradient_step(h, dx, alpha, top_m)
        c = c5_value(h_new, dx)
        
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h = h_new
        else:
            # Stay at best
            pass
    
    return best_h, best_c

def anneal_step(h, dx, rng, n_perturb=3):
    """Propose a perturbation for simulated annealing."""
    n = len(h)
    h_new = h.copy()
    idx = rng.choice(n, size=n_perturb, replace=False)
    delta = rng.uniform(-0.1, 0.1, n_perturb)
    h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
    h_new = proj_feasible(h_new, n)
    return h_new

def simulated_annealing(h0, n, dx, max_iter, T0, T_min, rng):
    """Run simulated annealing."""
    h = h0.copy()
    c = c5_value(h, dx)
    best_c = c
    best_h = h.copy()
    
    for it in range(max_iter):
        T = T0 * (T_min / T0) ** (it / max_iter)
        h_new = anneal_step(h, dx, rng)
        c_new = c5_value(h_new, dx)
        
        if c_new < c or rng.random() < np.exp(-(c_new - c) / (T + 1e-15)):
            h = h_new
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

def fourier_perturb(h, n, rng, n_modes=3, amp=0.1):
    """Perturb h in Fourier space."""
    h_new = h.copy()
    F = np.fft.rfft(h_new)
    for _ in range(n_modes):
        k = rng.randint(1, len(F))
        F[k] += amp * (rng.randn() + 1j * rng.randn())
        if k > 0 and k < len(F) - 1:
            F[-k if -k >= 0 else len(F)-1-k] = np.conj(F[k])
    h_new = np.real(np.fft.irfft(F, n))
    return proj_feasible(h_new, n)

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

        else:
            h[b*block_size:(b+1)*block_size] = 0.0

    v = np.zeros(n)  # momentum
    best_c = c5_value(h, dx)
    best_h = h.copy()

        alpha = alpha0 / np.sqrt(1.0 + t * 0.05)
        
        # Compute subgradient at current h (or at y = h + momentum)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        # Find top-m
        sorted_idx = np.argsort(corr)[-top_m:]
        max_c = corr[sorted_idx[0]]

        temps = np.exp(-(max_c - corr[sorted_idx]) / (0.005 + 1e-10))
        weights = temps / np.sum(temps)
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for idx, w in zip(sorted_idx, weights):
            if w < 1e-10:
                continue

            i_start = max(0, -s)
            i_end = min(n - 1, n - 1 - s)

            grad[j_vals] += w * (-h[i_vals]) * dx

            best_h = h_new.copy()
            h = h_new
        else:
            # Don't update h, reduce momentum
            v *= 0.5
    
    return best_h, best_c

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(5000):

        if abs(diff) < 1e-15:

            mask = h < 1.0 - 1e-12

def compute_weighted_subgrad(h, dx, k_idx, weights):
    n = len(h)

            if c_ls < c_sg:
                h_cand, c_cand = h_ls, c_ls
            else:
                h_cand, c_cand = h_sg, c_sg
            if c_cand < best_c5:
                best_h = h_cand.copy()
                best_c5 = c_cand
                best_n = n_try

    # Phase 2: Simulated annealing at best resolution
    if time_left() > 50:
        n = best_n
        dx = 2.0 / n
        h_sa, c_sa = simulated_anneal(best_h, n, dx, rng, 4000, 0.02, 0.0005, 0.08)
        if c_sa < best_c5:
            best_h = h_sa
            best_c5 = c_sa
        h_ls, c_ls = local_search(best_h, n, dx, rng, 3000, 2, 0.08)
        if c_ls < best_c5:
            best_h = h_ls
            best_c5 = c_ls

    # Phase 3: SLSQP polish with multiple betas
    n = best_n
    dx = 2.0 / n
    for beta in [200.0, 100.0, 50.0, 25.0, 10.0, 5.0]:
        if time_left() < 15:
            break
        h_pol = run_slsqp(best_h, n, dx, beta, 800)
        c_pol = c5_value(h_pol, dx)
        if c_pol < best_c5:
            best_h = h_pol
            best_c5 = c_pol

    # Phase 4: Fourier perturbations + re-optimize
    if time_left() > 40:
        n = best_n
        dx = 2.0 / n
        for _ in range(4):
            if time_left() < 20:
                break
            h_pert = fourier_perturb(best_h, n, rng, 3, 0.04)
            h_fg, c_fg = nesterov_subgrad(h_pert, n, dx, 1000, 0.03, 6, 0.8)
            if c_fg < best_c5:
                best_h = h_fg
                best_c5 = c_fg
            h_pol = run_slsqp(best_h, n, dx, 50.0, 500)
            c_pol = c5_value(h_pol, dx)
            if c_pol < best_c5:
                best_h = h_pol
                best_c5 = c_pol

    # Phase 5: Try other resolutions from best solution
    for n_try in [64, 96, 192, 256, 320, 384]:
        if time_left() < 25:
            break
        if n_try == best_n:
            continue
        dx = 2.0 / n_try
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_sg, c_sg = nesterov_subgrad(h_interp, n_try, dx, 1500, 0.05, 8, 0.85)
        h_ls, c_ls = local_search(h_sg, n_try, dx, rng, 1000, 3, 0.1)
        c_cand = min(c_sg, c_ls)
        h_cand = h_sg if c_sg <= c_ls else h_ls
        if time_left() > 15:
            h_pol = run_slsqp(h_cand, n_try, dx, 50.0, 500)
            c_pol = c5_value(h_pol, dx)
            if c_pol < c_cand:
                c_cand = c_pol
                h_cand = h_pol
        if c_cand < best_c5:
            best_h = h_cand
            best_c5 = c_cand
            best_n = n_try

    # Final polish
    n = best_n
    dx = 2.0 / n
    if time_left() > 10:
        h_pol = run_slsqp(best_h, n, dx, 30.0, 1000)
        c_pol = c5_value(h_pol, dx)
        if c_pol < best_c5:
            best_h = h_pol
            best_c5 = c_pol

    best_h = repair_h(best_h, best_n)
    best_c5 = c5_value(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
