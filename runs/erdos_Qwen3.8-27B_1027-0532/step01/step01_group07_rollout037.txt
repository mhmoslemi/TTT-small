For a given offset d:
- If d ≥ 0: valid i = 0 to n-1-d
  grad[i+d] += (1-h[i]) * dx for i in 0..n-1-d
  grad[i] += -h[i+d] * dx for i in 0..n-1-d
  
- If d < 0 (let d' = -d): valid i = d' to n-1, equivalently j = 0 to n-1-d'
  grad[j] += (1-h[j+d']) * dx for j in 0..n-1-d'
  grad[j+d'] += -h[j] * dx for j in 0..n-1-d'

This can be vectorized with numpy slicing.

```python
def compute_gradient_for_lag(h, d, dx):
    """Compute gradient of c(d) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        L = n - d
        # h[i+d] * (1-h[i]) for i = 0..L-1
        a = h[d:d+L]  # h[i+d]
        b = h[0:L]     # h[i]
        grad[d:d+L] += (1.0 - b) * dx
        grad[0:L] += (-a) * dx
    else:
        d2 = -d
        L = n - d2
        # h[i] * (1-h[i+d2]) for i = 0..L-1
        a = h[0:L]     # h[i]
        b = h[d2:d2+L] # h[i+d2]
        grad[0:L] += (1.0 - b) * dx
        grad[d2:d2+L] += (-a) * dx
    return grad
```

This is O(n) per lag in vectorized numpy, which is fast.

- A move is: pick position i with h[i]=1 and position j with h[j]=0, swap them
- This changes h by: h[i] -= 1, h[j] += 1

The change in c(d) = dx * [contribution of new h at positions i, j - contribution of old h]

4. Repeat until no improvement

import numpy as np
import time

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_correlations(h)))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_c_for_lag(h, d, dx):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        L = n - d
        if L <= 0:
            return grad
        a = h[d:d+L]
        b = h[0:L]
        grad[d:d+L] += (1.0 - b) * dx
        grad[0:L] += (-a) * dx
    else:
        d2 = -d
        L = n - d2
        if L <= 0:
            return grad
        a = h[0:L]
        b = h[d2:d2+L]
        grad[0:L] += (1.0 - b) * dx
        grad[d2:d2+L] += (-a) * dx
    return grad

def active_gradient(h, c, alpha):
    n = len(h)
    dx = 2.0 / n
    m = len(c)  # 2n-1
    c_max = np.max(c)
    # Softmax weights
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for lag in range(m):
        if w[lag] < 1e-10:
            continue
        d = lag - (n - 1)
        grad += w[lag] * grad_c_for_lag(h, d, dx)
    return grad

def nesterov_descent(h, n, alpha, max_iter, deadline, lr=0.1, beta=0.9):
    target = n / 2.0
    c = compute_correlations(h)
    best_c5 = float(np.max(c))
    best_h = h.copy()
    y = h.copy()
    prev_h = h.copy()
    stale = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Nesterov extrapolation
        if it > 0:
            y = h + beta * (h - prev_h)
            y = project_feasible(y, target)
        else:
            y = h.copy()
        
        c_y = compute_correlations(y)
        c5_y = float(np.max(c_y))
        
        if c5_y < best_c5 - 1e-12:
            best_c5 = c5_y
            best_h = y.copy()
            stale = 0
        else:
            stale += 1
        
        if stale > 200:
            # Reduce lr and continue
            lr *= 0.5
            stale = 0
            if lr < 1e-12:
                break
        
        # Compute gradient at y
        grad = active_gradient(y, c_y, alpha)
        
        # Step
        prev_h = h
        h = project_feasible(y - lr * grad, target)
        
        # Check if step was good
        c_h = compute_correlations(h)
        if float(np.max(c_h)) > c5_y + 0.01 * lr:
            # Step was too big, revert
            h = y.copy()
            lr *= 0.5
    
    return best_h, best_c5

def binary_local_search(h, n, deadline, max_rounds=100):
    """Local search on binary solutions via swaps."""
    target = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones_count = int(h_bin.sum())
    if ones_count > n // 2:
        # Remove excess ones (set smallest h values to 0)
        one_positions = np.where(h_bin == 1)[0]
        extra = ones_count - n // 2
        remove = one_positions[np.argsort(h[one_positions])[:extra]]
        h_bin[remove] = 0
    elif ones_count < n // 2:
        zero_positions = np.where(h_bin == 0)[0]
        need = n // 2 - ones_count
        add = zero_positions[np.argsort(-h[zero_positions])[:need]]
        h_bin[add] = 1
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        best_swap = None
        best_val = best_c5
        
        one_pos = np.where(h_bin == 1)[0]
        zero_pos = np.where(h_bin == 0)[0]
        
        for i in one_pos:
            for j in zero_pos:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                val = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                if val < best_val - 1e-12:
                    best_val = val
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        best_c5 = best_val
        best_h = h_bin.copy()
    
    return best_h, best_c5

import numpy as np
import time

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_correlations(h)))

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, d, dx):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        L = n - d
        if L <= 0:
            return grad
        grad[d:d+L] += (1.0 - h[0:L]) * dx
        grad[0:L] += (-h[d:d+L]) * dx
    else:
        d2 = -d
        L = n - d2
        if L <= 0:
            return grad
        grad[0:L] += (1.0 - h[d2:d2+L]) * dx
        grad[d2:d2+L] += (-h[0:L]) * dx
    return grad

def smooth_gradient(h, c, alpha):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for lag in range(m):
        if w[lag] < 1e-10:
            continue
        d = lag - (n - 1)
        grad += w[lag] * grad_lag(h, d, dx)
    return grad

def active_gradient(h, c, top_k):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    top_idx = np.argsort(c)[-top_k:]
    w = np.zeros(m)
    w[top_idx] = 1.0 / top_k
    grad = np.zeros(n)
    for lag in top_idx:
        d = int(lag) - (n - 1)
        grad += w[lag] * grad_lag(h, d, dx)
    return grad

def nesterov_refine(h, n, deadline, alpha=100.0, lr=0.05, beta=0.8, max_iter=50000):
    target = n / 2.0
    c = compute_correlations(h)
    best_c5 = float(np.max(c))
    best_h = h.copy()
    y = h.copy()
    prev = h.copy()
    it = 0
    stagnant = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        if it > 1:
            y = h + beta * (h - prev)
            y = project_feasible(y, target)

        c_y = compute_correlations(y)
        c5_y = float(np.max(c_y))

        if c5_y < best_c5 - 1e-11:
            best_c5 = c5_y
            best_h = y.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 300:
            lr *= 0.6
            stagnant = 0
            if lr < 1e-12:
                break

        if it % 100 == 0:
            alpha = min(alpha * 1.2, 2000.0)

        grad = smooth_gradient(y, c_y, alpha)
        prev = h
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)

        c_new = compute_correlations(h_new)
        if float(np.max(c_new)) > c5_y + 1e-9:
            lr *= 0.5
            h_new = project_feasible(y - lr * grad, target)
            c_new = compute_correlations(h_new)
            if float(np.max(c_new)) > c5_y + 1e-9:
                h = y.copy()
                if lr < 1e-12:
                    break
        h = h_new

    return best_h, best_c5

def binary_local_search(h_init, n, deadline, max_rounds=2000):
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    while len(ones) > target:
        idx = ones[np.argmin(h_init[ones])]
        h[idx] = 0
        ones = np.setdiff1d(ones, idx)
        zeros = np.append(zeros, idx)
    while len(ones) < target:
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h_init[zeros])]
        h[idx] = 1
        zeros = np.setdiff1d(zeros, idx)
        ones = np.append(ones, idx)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    round_no_improve = 0

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 1e-12
        best_pair = None

        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                val = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = best_c5 - val
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)

        if best_pair is None:
            round_no_improve += 1
            if round_no_improve > 5:
                break
            continue

        round_no_improve = 0
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_c5 -= best_delta
        best_h = h.copy()

    return best_h, best_c5

    T = 0.02

    steps = 0

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / T):

        best_c5 = c5_out
        best_h = h_out

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0

def block_init(n, rng, n_blocks):
    h = np.zeros(n)
    for b in range(n_blocks):

        length = half + rng.randint(-1, 2)
        end = min(n, start + length)
        h[start:end] = 1.0
