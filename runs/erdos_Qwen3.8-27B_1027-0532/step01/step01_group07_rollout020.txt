```python
def compute_grad_correct(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    
    n_corr = 2 * n - 1
    # For each i (coordinate of h) and k (correlation index):
    # A(k,i) = i + k - (n-1)  [index in (1-h) term]
    # B(k,i) = i - k + (n-1)  [index in h term]
    # dc[k]/dh[i] = (1-h[A]) * [A valid] - h[B] * [B valid]
    
    i_idx = np.arange(n)[:, None]  # (n, 1)
    k_idx = np.arange(n_corr)[None, :]  # (1, 2n-1)
    
    A = i_idx + k_idx - (n - 1)  # (n, 2n-1)
    B = i_idx - k_idx + (n - 1)  # (n, 2n-1)
    
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    
    # (1-h[A]) where valid, 0 otherwise
    term1 = np.where(A_valid, 1.0 - h[np.clip(A, 0, n-1)], 0.0)
    # h[B] where valid, 0 otherwise
    term2 = np.where(B_valid, h[np.clip(B, 0, n-1)], 0.0)
    
    # grad[i] = dx * sum_k w[k] * (term1[i,k] - term2[i,k])
    grad = dx * ((term1 - term2) @ w)
    
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    return smooth_val, grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    
    n_corr = 2 * n - 1
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    
    A = i_idx + k_idx - (n - 1)
    B = i_idx - k_idx + (n - 1)
    
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    
    term1 = np.where(A_valid, 1.0 - h[np.clip(A, 0, n-1)].astype(float), 0.0)
    term2 = np.where(B_valid, h[np.clip(B, 0, n-1)].astype(float), 0.0)
    
    grad = dx * ((term1 - term2) @ w)
    return c_max + np.log(exp_c.sum()) / alpha, grad

def subgradient(h):
    """Exact subgradient: gradient of a random active constraint."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    # Find near-active constraints (within 1% of max)
    threshold = c_max - 0.01 * c_max
    active = np.where(c >= threshold)[0]
    k = active[np.random.randint(len(active))]
    
    # Gradient of c[k] w.r.t. h
    n_corr = 2 * n - 1
    A = np.arange(n) + k - (n - 1)
    B = np.arange(n) - k + (n - 1)
    
    grad = np.zeros(n)
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    grad[A_valid] += (1.0 - h[A[A_valid]])
    grad[B_valid] -= h[B[B_valid]]
    grad *= dx
    return grad, c_max

def grad_descent(h, n, target, deadline, alpha_start=20, alpha_end=5000, 
                 max_iter=100000, use_subgrad=False):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        if use_subgrad and it % 100 == 0:
            grad, _ = subgradient(h_cur)
        else:
            alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iter)
            _, grad = smooth_gradient(h_cur, alpha)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.1 / gnorm
        h_best = h_cur
        f_best = compute_c5(h_cur)
        for _ in range(15):
            h_try = h_cur - step * grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = compute_all_c(h)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    
    n_corr = 2 * n - 1
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    
    A = i_idx + k_idx - (n - 1)
    B = i_idx - k_idx + (n - 1)
    
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    
    term1 = np.where(A_valid, 1.0 - h[np.clip(A, 0, n-1)], 0.0)
    term2 = np.where(B_valid, h[np.clip(B, 0, n-1)], 0.0)
    
    grad = dx * ((term1 - term2) @ w)
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    return smooth_val, grad

def active_subgradient(h, rng):
    n = len(h)
    dx = 2.0 / n
    c = compute_all_c(h)
    c_max = np.max(c)
    threshold = c_max - 0.02 * max(c_max, 1e-10)
    active = np.where(c >= threshold)[0]
    k = int(active[rng.randint(len(active))])
    
    grad = np.zeros(n)
    A = np.arange(n) + k - (n - 1)
    B = np.arange(n) - k + (n - 1)
    
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    
    grad[A_valid] += (1.0 - h[A[A_valid]])
    grad[B_valid] -= h[B[B_valid]]
    grad *= dx
    return grad, c_max

def optimize_smooth(h, n, target, deadline, alpha_start, alpha_end, max_iter, use_subgrad_interval=0):
    """Gradient descent with smooth or subgradient method."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    rng = np.random.RandomState(int(time.time()) % 10000)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        if use_subgrad_interval > 0 and it % use_subgrad_interval == 0 and it > 0:
            grad, _ = active_subgradient(h_cur, rng)
        else:
            frac = min(1.0, it / max(1, max_iter))
            alpha = alpha_start * (alpha_end / alpha_start) ** frac
            _, grad = smooth_gradient(h_cur, alpha)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.5 / max(gnorm, 1e-10)
        h_cur_c5 = compute_c5(h_cur)
        h_best = h_cur
        f_best = h_cur_c5
        
        for _ in range(12):
            h_try = project_feasible(h_cur - step * grad, target)
            f_try = compute_c5(h_try)
            if f_try < f_best - 1e-14:
                h_best = h_try
                f_best = f_try
            step *= 0.5
        
        h_cur = h_best
        if f_best < best_c5:
            best_c5 = f_best
            best_h = h_cur.copy()
    
    return best_h, best_c5
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target):
    if abs(h.sum() - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = compute_all_c(h)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    n_corr = 2 * n - 1
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    A = i_idx + k_idx - (n - 1)
    B = i_idx - k_idx + (n - 1)
    A_valid = (A >= 0) & (A < n)
    B_valid = (B >= 0) & (B < n)
    term1 = np.where(A_valid, 1.0 - h[np.clip(A, 0, n - 1)], 0.0)
    term2 = np.where(B_valid, h[np.clip(B, 0, n - 1)], 0.0)
    grad = dx * ((term1 - term2) @ w)
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    return smooth_val, grad

    B_valid = (B >= 0) & (B < n)
    grad[A_valid] += (1.0 - h[A[A_valid]])
    grad[B_valid] -= h[B[B_valid]]
    grad *= dx
    return grad, c_max
```
