def best_t_for_coord(h, n, i):
    # Save old value
    old = h[i]
    
    # Compute S at t=0
    h[i] = 0.0
    S0 = np.correlate(h, 1.0-h, mode="full")
    
    # Compute S at t=1
    h[i] = 1.0
    S1 = np.correlate(h, 1.0-h, mode="full")
    
    # Restore
    h[i] = old
    
    # f(t) = max(R_0 + t(1-t), max_k(S0[k] + t*(S1[k]-S0[k])))
    R0 = S0[n-1]  # center index
    
    # Coarse search
    ts = np.linspace(0, 1, 50)
    linears = S0 + ts[:,None] * (S1-S0)[None,:]  # shape (50, 2n-1)
    linears[:, n-1] = R0 + ts*(1-ts)  # fix the center
    f_vals = np.max(linears, axis=1)
    
    best_m = np.argmin(f_vals)
    
    # Fine search around best
    if best_m > 0 and best_m < len(ts)-1:
        ts_fine = np.linspace(ts[best_m-1], ts[best_m+1], 200)
    else:
        ts_fine = np.linspace(max(0, ts[best_m]-0.02), min(1, ts[best_m]+0.02), 200)
    
    linears = S0 + ts_fine[:,None] * (S1-S0)[None,:]
    linears[:, n-1] = R0 + ts_fine*(1-ts_fine)
    f_vals = np.max(linears, axis=1)
    
    return ts_fine[np.argmin(f_vals)]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    return (max_corr + np.log(sum_exp) / t) * dx

def smoothed_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of S_k w.r.t. h for each lag k
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2*n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - center
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # S_lag = sum_{j=0}^{n-1-lag} h[j](1-h[j+lag])
            # dS/dh[i] = (1-h[i+lag]) if i <= n-1-lag, minus h[i-lag] if i >= lag
            if n - 1 - lag >= 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
            if lag <= n - 1:
                grad[lag:] += -w * h[:n-lag]
        else:
            l = -lag
            # S_{-l} = sum_{j=l}^{n-1} h[j](1-h[j-l])

    if w0 > 1e-12:

            grad[:n-k] += wk_pos * (1.0 - h[k:])
            # -h[i-k] for i = k..n-1
            grad[k:] += -wk_pos * h[:n-k]

        wk_neg = weights[center - k]
        if wk_neg > 1e-12:

            grad[k:] += wk_neg * (1.0 - h[:n-k])
            # -h[i+k] for i = 0..n-1-k
            grad[:n-k] += -wk_neg * h[k:]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_obj(h, n, t, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    base = (max_corr + np.log(sum_exp) / t) * dx
    sum_constraint = np.sum(h) - n / 2.0
    return base + penalty * sum_constraint * sum_constraint

def smoothed_grad(h, n, t, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    center = n - 1
    
    w0 = weights[center]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        wk_pos = weights[center + k]
        if wk_pos > 1e-12:
            grad[:n-k] += wk_pos * (1.0 - h[k:])
            grad[k:] += -wk_pos * h[:n-k]
        
        wk_neg = weights[center - k]
        if wk_neg > 1e-12:
            grad[k:] += wk_neg * (1.0 - h[:n-k])
            grad[:n-k] += -wk_neg * h[k:]
    
    grad *= dx
    
    # Penalty gradient
    sum_constraint = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * sum_constraint * np.ones(n)
    
    return grad

def l_bfgs_optimize(h0, n, t, penalty=100.0, max_iter=500):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smoothed_obj, h0.copy(),
            jac=smoothed_grad,
            args=(n, t, penalty),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()

def best_t_for_coord(h, n, i):

    h[i] = 0.0
    S0 = np.correlate(h, 1.0 - h, mode="full")
    h[i] = 1.0

def coordinate_descent(h, n, n_passes, time_limit):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()

def subgradient_polish(h, n, n_iter=300):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.05
    center = n - 1
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = int(np.argmax(corr))
        lag = idx_star - center
        grad = np.zeros(n)
        
        if lag == 0:
            grad = 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += -h[:n-lag]
        else:
            l = -lag
            grad[l:] += (1.0 - h[:n-l])
            grad[:n-l] += -h[l:]
        
        grad *= dx

        if gnorm < 1e-14:

        h = h - step * grad / gnorm
        h = project_onto_constraint(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.999
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smoothed_obj(h, n, t, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    base = (max_corr + np.log(sum_exp) / t) * dx
    sc = np.sum(h) - n / 2.0
    return base + penalty * sc * sc


def smoothed_grad(h, n, t, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = t * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    center = n - 1

    w0 = weights[center]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)

    for k in range(1, n):
        wk_pos = weights[center + k]
        if wk_pos > 1e-14:
            grad[:n - k] += wk_pos * (1.0 - h[k:])
            grad[k:] += -wk_pos * h[:n - k]

        wk_neg = weights[center - k]
        if wk_neg > 1e-14:
            grad[k:] += wk_neg * (1.0 - h[:n - k])
            grad[:n - k] += -wk_neg * h[k:]

    grad *= dx
    sc = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * sc
    return grad


def l_bfgs_optimize(h0, n, t, penalty=100.0, max_iter=400):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smoothed_obj, h0.copy(),
            jac=smoothed_grad,
            args=(n, t, penalty),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def best_t_for_coord(h, n, i):
    old = h[i]
    center = n - 1
    S = np.correlate(h, 1.0 - h, mode="full")

    dS = np.zeros(2 * n - 1)
    dS[center] = 1.0 - 2.0 * old

    for k in range(1, n):
        c = 0.0
        if i + k < n:
            c += 1.0 - h[i + k]
        if i - k >= 0:
            c -= h[i - k]
        dS[center + k] = c

        c = 0.0
        if i >= k:
            c += 1.0 - h[i - k]
        if i + k < n:
            c -= h[i + k]
        dS[center - k] = c

    ts = np.linspace(0.0, 1.0, 50)
    deltas = (ts - old)[:, None] * dS[None, :]
    vals = S[None, :] + deltas
    vals[:, center] = S[center] + ts * (1.0 - ts) - old * (1.0 - old)
    f_vals = np.max(vals, axis=1)
    best_m = int(np.argmin(f_vals))

    if 0 < best_m < len(ts) - 1:
        lo_t = ts[best_m - 1]
        hi_t = ts[best_m + 1]
    else:
        lo_t = max(0.0, ts[best_m] - 0.02)
        hi_t = min(1.0, ts[best_m] + 0.02)

    ts_fine = np.linspace(lo_t, hi_t, 300)
    deltas = (ts_fine - old)[:, None] * dS[None, :]
    vals = S[None, :] + deltas
    vals[:, center] = S[center] + ts_fine * (1.0 - ts_fine) - old * (1.0 - old)
    f_vals = np.max(vals, axis=1)

    return ts_fine[int(np.argmin(f_vals))]


def coordinate_descent(h, n, n_passes, time_limit):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    stagnant = 0

    for p in range(n_passes):
        if time.time() > time_limit:
            break
        for i in range(n):
            h[i] = best_t_for_coord(h, n, i)
        h = project_onto_constraint(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                break

    return best_h, best_c5


def subgradient_polish(h, n, n_iter=500):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.05
    center = n - 1

    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = int(np.argmax(corr))
        lag = idx_star - center
        grad = np.zeros(n)

        if lag == 0:
            grad = 1.0 - 2.0 * h
        elif lag > 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] += -h[:n - lag]
        else:
            l = -lag
            grad[l:] += (1.0 - h[:n - l])
            grad[:n - l] += -h[l:]

        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h = h - step * grad / gnorm
        h = project_onto_constraint(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= 0.999

    return best_h, best_c5


def interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_onto_constraint(h_dst, n_dst)
