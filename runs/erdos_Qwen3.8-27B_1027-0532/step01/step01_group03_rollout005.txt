def _optimize_single_coord(h, j):
    """Optimize h[j] given all other coordinates fixed."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # For each lag, compute the linear function of h[j]
    # g_k = corr_k + delta_k * (h[j] - h[j]) where delta_k is the change per unit change in h[j]
    # Actually: if we change h[j] by delta, how does corr_k change?
    # corr_k = sum_i h[i]*(1-h[i+k])
    # Changing h[j] by delta:
    #   - In the "h[i]" part: contributes delta*(1-h[j+k]) if j+k is valid
    #   - In the "(1-h[i+k])" part: contributes -delta*h[j-k] if j-k is valid
    # So delta_corr_k = delta * ((1-h[j+k])*I(valid) - h[j-k]*I(valid))
    
    dx = 2.0 / n
    old_j = h[j]
    
    # For each corr index m (0 to 2n-2), the lag is L = m - (n-1)
    # delta_corr[m] = (1-h[j+L])*I(0<=j+L<n) - h[j-L]*I(0<=j-L<n)
    n = len(h)
    offset = n - 1
    deltas = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        L = m - offset
        d = 0.0
        jp = j + L
        jm = j - L
        if 0 <= jp < n:
            d += (1.0 - h[jp])
        if 0 <= jm < n:
            d -= h[jm]
        deltas[m] = d
    
    # New corr when h[j] changes to t: corr_new[m] = corr[m] + deltas[m] * (t - old_j)
    # We want to minimize max_m (corr[m] + deltas[m]*(t - old_j)) * dx
    # Equivalently minimize max_m (corr[m] + deltas[m]*(t - old_j))
    # This is minimizing a convex piecewise-linear function of t on [0, 1]
    
    # The breakpoints are where corr[m] + deltas[m]*(t-old_j) = corr[m'] + deltas[m']*(t-old_j)
    # i.e., (deltas[m] - deltas[m'])*t = corr[m'] - corr[m] + (deltas[m'] - deltas[m])*old_j
    # i.e., t = (corr[m'] - corr[m]) / (deltas[m] - deltas[m'])
    
    # Collect all breakpoints in [0, 1]
    breakpoints = [0.0, 1.0]
    for m1 in range(2*n - 1):
        for m2 in range(m1+1, 2*n - 1):
            denom = deltas[m1] - deltas[m2]
            if abs(denom) > 1e-12:
                t = (corr[m2] - corr[m1]) / denom
                if 0.0 <= t <= 1.0:
                    breakpoints.append(t)
    
    breakpoints = sorted(set(breakpoints))

        val = np.max(corr_new)
        if val < best_val:
            best_val = val
            best_t = t
    
    h[j] = best_t
    return h

def _optimize_single_coord(h, j):
    n = len(h)
    dx = 2.0 / n
    old_hj = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    offset = n - 1
    
    deltas = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        L = m - offset
        d = 0.0
        jp = j + L
        jm = j - L
        if 0 <= jp < n:
            d += (1.0 - h[jp])
        if 0 <= jm < n:
            d -= h[jm]
        deltas[m] = d
    
    # f(t) = max_m (corr[m] + deltas[m] * (t - old_hj))
    # Find t in [0,1] minimizing f(t)
    
    # Coarse grid
    ts = np.linspace(0, 1, 200)
    vals = np.max(corr + np.outer(ts, deltas) - deltas*old_hj, axis=1)
    # Wait: corr[m] + deltas[m]*(t - old_hj) for each m, then max over m
    # = max_m (corr[m] - deltas[m]*old_hj + deltas[m]*t)
    # = max_m (intercept[m] + slope[m]*t) where intercept = corr - deltas*old_hj, slope = deltas
    
    intercept = corr - deltas * old_hj
    # f(t) = max_m (intercept[m] + deltas[m] * t)
    
    ts = np.linspace(0, 1, 500)
    vals = np.max(intercept + np.outer(ts, deltas), axis=1)
    i = np.argmin(vals)
    
    # Refine around the best point
    t_lo = ts[max(0, i-1)]
    t_hi = ts[min(len(ts)-1, i+1)]
    ts_fine = np.linspace(t_lo, t_hi, 100)
    vals_fine = np.max(intercept + np.outer(ts_fine, deltas), axis=1)
    best_t = ts_fine[np.argmin(vals_fine)]
    
    h[j] = best_t
    return h

Actually, I realize there might be an issue with coordinate descent and the sum constraint. Let me think about this more carefully.

When I do coordinate descent on h[j] without respecting the sum constraint, the sum will drift. The _fix_sum function then corrects it, but this might undo some of the improvement.

A better approach: incorporate the sum constraint into the coordinate update.

When optimizing h[j], I want to minimize max_k (corr_k + deltas_k * (t - old_hj)) subject to:
- 0 ≤ t ≤ 1
- sum(h) + (t - old_hj) = n/2, i.e., t = n/2 - sum(h_{-j}) where h_{-j} is h with the j-th entry removed.

Wait, that just fixes t! If I enforce the sum constraint exactly, then h[j] is determined by the other coordinates: h[j] = n/2 - sum_{i≠j} h[i]. So there's no freedom to optimize h[j] individually.

This means I need to do pairs of updates, or use a different approach.

Alternative: use the "split" approach where I update two coordinates at a time. Or use the projected gradient approach where I:
1. Take a gradient step on the smooth objective
2. Project onto {0≤h≤1, sum(h)=n/2}

The projection onto {0≤h≤1, sum(h)=n/2} is:
- First clip to [0,1]
- Then adjust to hit the sum: add/subtract a constant from all entries (clipping again if needed)
- This is the Euclidean projection onto the intersection of a box and a hyperplane, which can be found by bisection on the Lagrange multiplier

Let me implement this properly:

```python
def _project_box_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    # Bisection on lambda: x_i = clip(h_i - lambda, 0, 1), find lambda s.t. sum(x) = target_sum
    lo = h.min() - 1.0  # lambda such that all x_i = 1
    hi = h.max() + 1.0  # lambda such that all x_i = 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Small correction for floating point
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for i in order:
            if abs(diff) < 1e-12:
                break
            adj = np.sign(diff) * min(abs(diff), abs(1.0 - x[i]) if diff > 0 else x[i])
            x[i] += adj
            diff -= adj
    return x
```

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    return (m + np.log(exps.sum()) / alpha) * dx

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / exps.sum()
    
    g = 1.0 - h
    grad = np.zeros(n)
    offset = n - 1
    for j in range(n):
        s = 0.0
        for m_idx in range(2 * n - 1):
            L = m_idx - offset
            w = weights[m_idx]
            if w < 1e-15:
                continue
            jp = j + L
            jm = j - L
            if 0 <= jp < n:
                s += w * g[jp]
            if 0 <= jm < n:
                s -= w * h[jm]
        grad[j] = s * dx
    return grad

            h[j] = _coord_optimize(h, j)
            if abs(h[j] - old_val) > 1e-10:
                improved = True
        h = _fix_sum(h, target_sum)
        if not improved:
            break
    return h

    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    return (m + np.log(exps.sum()) / alpha) * dx

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / exps.sum()
    g = 1.0 - h
    grad = np.zeros(n)
    offset = n - 1
    for j in range(n):
        s = 0.0
        for m_idx in range(2 * n - 1):
            L = m_idx - offset
            w = weights[m_idx]
            if w < 1e-15:
                continue
            jp = j + L
            jm = j - L
            if 0 <= jp < n:
                s += w * g[jp]
            if 0 <= jm < n:
                s -= w * h[jm]
        grad[j] = s * dx
    return grad

def _coord_descent(h, target_sum, max_sweeps):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    offset = n - 1
    for sweep in range(max_sweeps):
        changed = False
        for j in range(n):
            old_hj = h[j]
            deltas = np.zeros(2 * n - 1)
            for m_idx in range(2 * n - 1):
                L = m_idx - offset
                d = 0.0
                jp = j + L
                jm = j - L
                if 0 <= jp < n:
                    d += (1.0 - h[jp])
                if 0 <= jm < n:
                    d -= h[jm]
                deltas[m_idx] = d
            intercept = corr - deltas * old_hj
            ts = np.linspace(0.0, 1.0, 200)
            vals = np.max(intercept + np.outer(ts, deltas), axis=1)
            i = int(np.argmin(vals))
            t_lo = ts[max(0, i - 1)]
            t_hi = ts[min(199, i + 1)]
            if t_hi <= t_lo:
                t_hi = min(1.0, t_lo + 0.01)
                if t_hi <= t_lo:
                    t_lo = max(0.0, t_hi - 0.01)
            ts_fine = np.linspace(t_lo, t_hi, 200)
            vals_fine = np.max(intercept + np.outer(ts_fine, deltas), axis=1)
            best_t = float(ts_fine[int(np.argmin(vals_fine))])
            if abs(best_t - old_hj) > 1e-10:
                h[j] = best_t
                corr = corr + deltas * (best_t - old_hj)
                changed = True
        h = _fix_sum(h, target_sum)
        if not changed:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
    return h

def _lbfgs_opt(h, target_sum, alpha, pgtol, maxiter):
    try:
        res = fmin_l_bfgs_b(_smooth_obj, h, fprime=_smooth_grad,
                            args=(alpha,), pgtol=pgtol, maxiter=maxiter,
                            iprint=0)
        return _fix_sum(res, target_sum)
    except Exception:
        return h.copy()
