    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l
            jpl = j + l
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            if 0 <= jpl < n:
                term += (-x[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

        return np.sum(x) - n / 2.0

        return smooth_max_obj(x, eps, n)

        return smooth_max_grad(x, eps, n)

                   options={'maxiter': maxiter, 'ftol': 1e-12})

    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def coord_update(h, j, n):
    lags = np.arange(-(n-1), n)
    jm = j + lags
    jmm = j - lags
    mask_hm = (jm >= 0) & (jm < n)
    mask_hmm = (jmm >= 0) & (jmm < n)
    b = np.zeros(2*n - 1)
    b[mask_hm] = 1.0 - h[jm[mask_hm]]
    b[mask_hmm] -= h[jmm[mask_hmm]]
    corr = np.correlate(h, 1 - h, mode="full")
    a = corr - h[j] * b
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        min_slope = np.min(b[active])
        max_slope = np.max(b[active])
        if min_slope <= 1e-10 and max_slope >= -1e-10:
            break
        elif max_slope < -1e-10:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def coord_descent(h, n, max_passes=200, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    for p in range(max_passes):
        max_change = 0.0
        order = rng.permutation(n)
        for j in order:
            new_val = coord_update(h, j, n)
            change = abs(new_val - h[j])
            if change > max_change:
                max_change = change
            h[j] = new_val
        h = project_feasible(h, n)
        if max_change < 1e-12:
            break
    return h

def smooth_max_obj(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, eps, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2*n - 1):
        l_val = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        for j in range(n):
            term = 0.0
            jml = j - l_val
            jpl = j + l_val
            if 0 <= jml < n:
                term += (1.0 - x[jml])
            if 0 <= jpl < n:
                term += (-x[jpl])
            grad[j] += w * term
    grad *= dx
    return grad

def _g_obj(x):
    return smooth_max_obj(x, _g_eps, _g_n)

def _g_grad(x):
    return smooth_max_grad(x, _g_eps, _g_n)

def _g_eq_con(x):
    return np.sum(x) - _g_n / 2.0

def slsqp_refine(h, n, eps=200.0, maxiter=300):
    global _g_eps, _g_n
    _g_eps = eps
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    res = minimize(_g_obj, h.copy(), jac=_g_grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _g_eq_con}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return h.copy()

def _g_periodic_obj(u):
    n = _g_n
    p = _g_p
    v = project_feasible(np.clip(u, 0, 1), p)
    h = np.tile(v, n // p)
    return evaluate_c5(h)

def periodic_optimize(n, p, seed=42, max_evals=3000):
    global _g_n, _g_p
    _g_n = n
    _g_p = p
    bounds = [(0.0, 1.0)] * p
    result = differential_evolution(_g_periodic_obj, bounds, seed=seed,
                                     maxiter=1000, tol=1e-10, polish=False,
                                     mutation=(0.3, 0.8), recombination=0.7)
    v = project_feasible(np.clip(result.x, 0, 1), p)
    h = np.tile(v, n // p)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(85, 0.5)
    n = len(h0)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Coordinate descent from initial
    if time_left() > 120:
        h_cd = coord_descent(h0, n, max_passes=300, seed=seed)
        c5 = evaluate_c5(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd
    
    # Phase 2: SLSQP refinement
    if time_left() > 120:
        for eps in [50.0, 200.0, 1000.0]:
            h_ref = slsqp_refine(best_h, n, eps=eps, maxiter=200)
            c5 = evaluate_c5(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref
            # Follow with coordinate descent
            h_cd = coord_descent(h_ref, n, max_passes=100, seed=seed+1)
            c5 = evaluate_c5(h_cd)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd
    
    # Phase 3: Periodic structure exploration
    if time_left() > 300:
        for p in [5, 17]:
            if time_left() < 120:
                break
            h_per = periodic_optimize(n, p, seed=seed+p, max_evals=3000)
            h_cd = coord_descent(h_per, n, max_passes=200, seed=seed+p+1)
            c5 = evaluate_c5(h_cd)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd
            # SLSQP refinement
            h_ref = slsqp_refine(h_cd, n, eps=200.0, maxiter=200)
            c5 = evaluate_c5(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref
            h_cd = coord_descent(h_ref, n, max_passes=100, seed=seed+p+2)
            c5 = evaluate_c5(h_cd)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd
    
    # Phase 4: Random restarts with coordinate descent
    if time_left() > 100:
        rng = np.random.RandomState(seed + 1000)
        n_restarts = 300
        for r in range(n_restarts):
            if time_left() < 60:
                break
            # Different types of perturbations
            rtype = r % 5
            if rtype == 0:
                scale = 0.01 + 0.03 * (r // 5)
                h_pert = best_h + scale * rng.randn(n)
