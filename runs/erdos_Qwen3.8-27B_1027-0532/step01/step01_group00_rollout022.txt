```python
from scipy.optimize import minimize
import time

def project_onto_simplex_box(y, target_sum):
    """Project y onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(y)
    # Bisection on t: find t such that sum(clip(y-t, 0, 1)) = target_sum
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def subgradient_step(h, lr, n_active=1):
    """One subgradient step (minimize max_k c_k)"""
    n = len(h)
    corr = compute_correlations(h)
    # Find top n_active lags
    top_idx = np.argsort(-corr)[:n_active]
    # Average gradient of top lags
    g = np.zeros(n)
    for k_abs in top_idx:
        # k_abs is the index in the 'full' correlation output
        # The actual lag is k_abs - (n-1)
        k = k_abs - (n - 1)
        # Gradient of c_k with respect to h[j]:
        # dc_k/dh[j] = (1-h[j+k])*1[j+k<n] - h[j-k]*1[j-k>=0]
        # But for k=0: dc_0/dh[j] = 1 - 2*h[j]
        if k == 0:
            g += (1 - 2*h)
        else:
            # j+k valid: j+k in [0, n-1], i.e., j in [0, n-1-k] if k>0, or j in [-k, n-1] if k<0
            # j-k valid: j-k in [0, n-1], i.e., j in [k, n-1] if k>0, or j in [0, n-1+k] if k<0
            if k > 0:
                # (1-h[j+k]) for j in [0, n-1-k]
                g[:n-k] += (1 - h[k:])  # wait, h[j+k] for j=0..n-1-k is h[k]..h[n-1]
                # -h[j-k] for j in [k, n-1]
                g[k:] -= h[:n-k]  # h[j-k] for j=k..n-1 is h[0]..h[n-1-k]
            else:  # k < 0
                kk = -k
                # (1-h[j+k]) = (1-h[j-kk]) for j in [kk, n-1]
                g[kk:] += (1 - h[:n-kk])
                # -h[j-k] = -h[j+kk] for j in [0, n-1-kk]
                g[:n-kk] -= h[kk:]
    g /= n_active
    return g, corr.max()

def subgradient_method(h0, budget_s, t0):
    """Run subgradient method"""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    
    # Diminishing step size
    C = 1.0  # will be tuned
    it = 0
    while time.time() - t0 < budget_s:
        it += 1
        lr = C / np.sqrt(it)
        g, _ = subgradient_step(h, lr)
        h_new = h - lr * g
        h_new = project_onto_simplex_box(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        # Re-project to fix sum exactly
        diff = target_sum - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        
        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

                if j2 - k >= 0:
                    b2 -= best_h[j2-k]

                    b2 += (1 - best_h[j2-kk])
                if j2 + kk < n:
                    b2 -= best_h[j2+kk]

                B2[k_abs] = b2

        #        = corr[k_abs] + t*(B1[k_abs] - B2[k_abs]) + best_h[j1]*(B2[k_abs] - B1[k_abs])

        D = B1 - B2
        C = corr + best_h[j1] * (B2 - B1)
        
        # Minimize max_k (C_k + t*D_k) over t in [t_lo, t_hi]
        # Grid search
        n_grid = 200
        t_grid = np.linspace(t_lo, t_hi, n_grid)
        vals = C + t_grid[:, None] * D[None, :]
        max_vals = vals.max(axis=1)
        best_t = t_grid[np.argmin(max_vals)]
        
        # Refine with local grid
        idx = np.argmin(max_vals)
        if 0 < idx < n_grid - 1:
            t_fine = np.linspace(t_grid[idx-1], t_grid[idx+1], 50)

            best_t = t_fine[np.argmin(max_vals_fine)]

        best_h[j1] = best_t
        best_h[j2] = s - best_t
        best_h = np.clip(best_h, 0, 1)
        
        score = evaluate_score(best_h)
        if score < best_score:
            best_score = score
    
    return best_h, best_score
```

```python
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(60):

    dx = 2.0 / n

    """Compute gradient of c_k with respect to all h, for a specific lag"""
    k = k_abs - (n - 1)
    g = np.zeros(n)
    if k == 0:

    elif k > 0:
```

```python
from scipy.optimize import linprog

def block_lp_update(h, block_start, block_size, n):
    B = block_size
    j = np.arange(block_start, block_start + B)
    s = h[j].sum()
    
    # Compute the contribution of each lag to the block
    # c_k = sum_i h[i](1-h[i+k])
    # Terms involving block variables h[j_1], ..., h[j_B]
    # For each lag k, c_k = A_k (fixed) + sum_m h[j_m] * B_{k,m}
    # where B_{k,m} = (1-h[j_m+k]) - h[j_m-k] (with appropriate valid checks)
    
    # Actually, for k≠0, c_k is affine in the block variables.
    # For k=0, c_0 has quadratic terms h[j_m]^2.
    
    # Let's handle k=0 separately: c_0 = (c_0 - sum_m h[j_m](1-h[j_m])) + sum_m h[j_m](1-h[j_m])
    # The second part is quadratic. For the LP, we can handle it by adding it to the objective
    # or by noting that for small blocks, the quadratic term is a perturbation.
    
    # For now, let's just do k≠0 (affine) in the LP, and handle k=0 in a post-check.
    
    corr = compute_correlations(h)
    ks_all = np.arange(2*n-1) - (n-1)
    
    # For each lag k and each block variable m, compute the coefficient
    # coeff[k, m] = (1-h[j_m+k]) * 1[valid] - h[j_m-k] * 1[valid]
    # and the constant term: A[k] = c_k - sum_m h[j_m] * coeff[k, m]
    
    # Vectorized computation:
    # For block index m (0..B-1), actual index is j[m] = block_start + m
    # For lag k: coeff = (1-h[j[m]+k])*1[0≤j[m]+k<n] - h[j[m]-k]*1[0≤j[m]-k<n]
    
    # Build coefficient matrix: shape (2n-1, B)
    indices = j[:, None] + ks_all[None, :]  # shape (B, 2n-1)
    indices2 = j[:, None] - ks_all[None, :]  # shape (B, 2n-1)
    
    valid1 = (indices >= 0) & (indices < n)  # (B, 2n-1)
    valid2 = (indices2 >= 0) & (indices2 < n)  # (B, 2n-1)
    
    # coeff[k, m] = (1-h[j[m]+k]) * valid1[m,k] - h[j[m]-k] * valid2[m,k]
    # We need h at indices indexed by (m, k)
    h_vals1 = np.where(valid1, h[np.clip(indices, 0, n-1)], 0)  # (B, 2n-1)
    h_vals2 = np.where(valid2, h[np.clip(indices2, 0, n-1)], 0)  # (B, 2n-1)
    
    coeff = (1 - h_vals1) * valid1 - h_vals2 * valid2  # (B, 2n-1)
    coeff = coeff.T  # (2n-1, B)
    
    # Constant: A[k] = corr[k] - sum_m h[j[m]] * coeff[k, m]
    A = corr - np.sum(h[j] * coeff, axis=1)
    
    # LP: min t
    # s.t. coeff[k] @ x + A[k] <= t  for all k
    #      0 <= x[i] <= 1
    #      sum(x) = s
    
    # In standard form for linprog:
    # min c @ vars where vars = [x_1, ..., x_B, t]
    # c = [0, 0, ..., 0, 1]
    
    n_vars = B + 1
    c_lp = np.zeros(n_vars)
    c_lp[-1] = 1.0  # minimize t
    
    # Inequality constraints: coeff[k] @ x - t <= -A[k]
    # i.e., [coeff[k], -1] @ vars <= -A[k]
    A_ub = np.hstack([coeff, -np.ones((2*n-1, 1))])
    b_ub = -A
    
    # Equality constraint: sum(x) = s
    A_eq = np.hstack([np.ones(B), 0.0]).reshape(1, -1)
    b_eq = np.array([s])
    
    bounds = [(0.0, 1.0)] * B + [(None, None)]  # t is unbounded
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method=' highs')
    
    if res.status == 0:
        x_new = res.x[:B]
        h[j] = x_new
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_onto_simplex_box(y, target_sum):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def thue_morse_seq(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def smooth_obj(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    if res.x is not None:
        return res.x
    return None

def block_lp_update(h, block_start, B, n, corr):
    """Update a block of B consecutive coordinates using LP"""
    j = np.arange(block_start, block_start + B)
    s = h[j].sum()
    
    ks_all = np.arange(2*n-1) - (n-1)
    
    # Compute coefficients for each (lag, block_var) pair
    # coeff[k, m] = dc_k/dh[j[m]] (for k != 0, this is the exact coefficient)
    indices_plus = j[:, None] + ks_all[None, :]   # (B, 2n-1)
    indices_minus = j[:, None] - ks_all[None, :]  # (B, 2n-1)
    
    valid_plus = (indices_plus >= 0) & (indices_plus < n)
    valid_minus = (indices_minus >= 0) & (indices_minus < n)
    
    # For valid indices, get the h values
    clip_plus = np.clip(indices_plus, 0, n-1)
    clip_minus = np.clip(indices_minus, 0, n-1)
    
    h_plus = h[clip_plus] * valid_plus  # (B, 2n-1)
    h_minus = h[clip_minus] * valid_minus  # (B, 2n-1)
    
    # coeff[k, m] = (1 - h[j[m]+k]) * valid_plus[m,k] - h[j[m]-k] * valid_minus[m,k]
    coeff = (1.0 - h_plus) * valid_plus - h_minus * valid_minus  # (B, 2n-1)
    coeff = coeff.T  # (2n-1, B)
    
    # Constant: A[k] = corr[k] - sum_m h[j[m]] * coeff[k, m]
    A_const = corr - h[j] @ coeff  # (2n-1,)
    
    # LP: min t
    # s.t. coeff[k] @ x - t <= -A_const[k]  for all k
    #      sum(x) = s
    #      0 <= x[i] <= 1
    
    n_vars = B + 1
    c_lp = np.zeros(n_vars)
    c_lp[-1] = 1.0
    
    A_ub = np.hstack([coeff, -np.ones((2*n-1, 1))])
    b_ub = -A_const
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :B] = 1.0
    b_eq = np.array([s])
    
    bounds_lp = [(0.0, 1.0)] * B + [(None, None)]
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    
    if res.status == 0:
        h[j] = np.clip(res.x[:B], 0.0, 1.0)
    
    return h
```

```python
from scipy.optimize import minimize, linprog
import time


def project_onto_simplex_box(y, target_sum):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx


def thue_morse_seq(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t


def smooth_obj(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
    )
    if res.x is not None:
        return res.x
    return None


def subgradient_method(h0, t_end, n_active, C_lr, seed=0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    it = 0
    rng = np.random.RandomState(seed)
    while time.time() < t_end:
        it += 1
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_active]
        g = np.zeros(n)
        for k_abs in top_idx:
            k = k_abs - (n - 1)
            if k == 0:
                g += (1.0 - 2.0 * h)
            elif k > 0:
                g[:n-k] += (1.0 - h[k:n])
                g[k:n] -= h[0:n-k]
            else:
                kk = -k
                g[kk:n] += (1.0 - h[0:n-kk])
                g[0:n-kk] -= h[kk:n]
        g /= n_active
        lr = C_lr / np.sqrt(it + 1.0)
        h_new = h - lr * g
        h_new = project_onto_simplex_box(h_new, target)
        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def block_lp_step(h, start, B, n, corr):
    j = np.arange(start, start + B)
    s = h[j].sum()
    if s < 1e-12 or s > B - 1e-12:
        pass
    ks_all = np.arange(2 * n - 1) - (n - 1)
    indices_plus = j[:, None] + ks_all[None, :]
    indices_minus = j[:, None] - ks_all[None, :]
    valid_plus = (indices_plus >= 0) & (indices_plus < n)
    valid_minus = (indices_minus >= 0) & (indices_minus < n)
    clip_plus = np.clip(indices_plus, 0, n - 1)
    clip_minus = np.clip(indices_minus, 0, n - 1)
    h_plus = np.where(valid_plus, h[clip_plus], 0.0)
    h_minus = np.where(valid_minus, h[clip_minus], 0.0)
    coeff = ((1.0 - h_plus) * valid_plus - h_minus * valid_minus).T
    A_const = corr - h[j] @ coeff
    n_vars = B + 1
    c_lp = np.zeros(n_vars)
    c_lp[-1] = 1.0
    A_ub = np.hstack([coeff, -np.ones((2 * n - 1, 1))])
    b_ub = -A_const
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :B] = 1.0
    b_eq = np.array([s])
    bounds_lp = [(0.0, 1.0)] * B + [(None, None)]
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.status == 0:
            h[j] = np.clip(res.x[:B], 0.0, 1.0)
    except Exception:
        pass
    return h


def block_cd(h0, t_end, seed=42, block_sizes=(3, 4, 5)):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(seed)
    sweeps = 0
    while time.time() < t_end:
        B = block_sizes[sweeps % len(block_sizes)]
        corr = compute_correlations(h)
        for start in range(0, max(1, n - B + 1), B):
            if time.time() > t_end:
                break
            block_lp_step(h, start, B, n, corr)
            corr = compute_correlations(h)
        if time.time() > t_end:
            break
        for _ in range(max(1, n // 4)):
            if time.time() > t_end:
                break
            start = rng.randint(0, max(1, n - B + 1))
            block_lp_step(h, start, B, n, corr)
            corr = compute_correlations(h)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        sweeps += 1
    return best_h, best_score


def slsqp_polish(h0, t_end):
    n = len(h0)
    best_h = h0.copy()
    best_score = evaluate_score(h0)
    alphas = [100, 300, 800, 2000, 5000]
    for alpha in alphas:
        if time.time() > t_end:
            break
        bounds = [(0.0, 1.0)] * n
        maxiter = max(50, min(500, 200000 // n))
        res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
        if res_x is not None:
            h_new = project_onto_simplex_box(np.clip(res_x, 0.0, 1.0), n / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
    return best_h, best_score
```
