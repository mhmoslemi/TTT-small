def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted - np.max(shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = N - 1 - index
        
        if lag == 0:
            # valid range: all j in [0, N-1]
            # dF/dh[j] = 1 - h[j] - h[j] = 1 - 2*h[j]
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:

            # dF/dh[j] = 1[j <= N-1-lag] - h[j+lag]*1[j+lag < N] - h[j-lag]*1[j-lag >= 0]
            # For j in [0, N-1-lag]: first term = 1, second term = h[j+lag] (valid), third term = h[j-lag]*1[j>=lag]

            # j in [lag, N-1-lag]: 1 - h[j+lag] - h[j-lag]  (need lag <= N-1-lag, i.e., 2*lag <= N-1)
            # j in [max(lag, 0), N-1-lag]: ... this is getting complicated
            
            # Let me just do it term by term:
            # Term 1: 1 for j in [0, N-1-lag]
            if lag < N:

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)

    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def local_search_pair(h, n_pts, n_iters, rng, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    for _ in range(n_iters):
        i = rng.randint(n_pts)
        j = rng.randint(n_pts)
        if i == j:
            continue
        for step in step_sizes:
            h_try = best.copy()
            h_try[i] = best[i] + step
            h_try[j] = best[j] - step
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            corr_try = compute_correlations(h_try)
            max_try = np.max(corr_try)
            if max_try < best_max - 1e-12:
                best = h_try
                best_corr = corr_try
                best_max = max_try
                break
    return best, best_max
