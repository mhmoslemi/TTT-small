import numpy as np
from scipy.optimize import minimize
import time

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

def lse_objective(x, T):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_corr = np.max(corr)
    shifted = (corr - max_corr) / T
    return dx * (max_corr + T * np.log(np.sum(np.exp(shifted))))

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def sum_jacobian(x):
    return np.ones(len(x))

def slsqp_optimize(h, T, max_iter, ftol=1e-12):
    n = len(h)
    
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return dx * (m + T * np.log(np.sum(np.exp((corr - m) / T))))
    
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian}],
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None:
            x_new = project_feasible(res.x)
            return x_new
    except:
        pass
    return h

def local_search(h, time_limit, t_start, step=0.1):
    n = len(h)
    best = h.copy()
    best_score = score_c5(best)
    no_improve = 0
    
    while time.time() - t_start < time_limit:
        improved = False
        # Try random pair modifications
        for _ in range(200):
            if time.time() - t_start > time_limit:
                break
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            trial = best.copy()
            amt = step * np.random.random()
            if best[i] > 0.01:
                trial[i] = max(0, best[i] - amt)
            if best[j] < 0.99:
                trial[j] = min(1, best[j] + amt)
            trial = project_feasible(trial)
            s = score_c5(trial)
            if s < best_score - 1e-10:
                best = trial
                best_score = s
                improved = True
        if not improved:
            step *= 0.8
            no_improve += 1
            if no_improve > 30 or step < 1e-6:
                break
        else:
            no_improve = 0
    
    return best, best_score

def sinusoidal_init(n, freq, phase, amp, rng):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h)

def multi_freq_init(n, n_freqs, rng):
    h = np.full(n, 0.5)
    for f in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, 0.25) / np.sqrt(f)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    return project_feasible(h)

def run(seed=42, budget_s=1300, **kwargs):

            sa_alloc = min(25, alloc * 0.25)
            if sa_alloc > 5:
                h_bin, s_bin = binary_sa(n, sa_alloc, t0, rng)
                inits.append(h_bin)
                inits.append(smooth_binary(h_bin, 1.5))
                inits.append(smooth_binary(h_bin, 3.0))

        for h_init in inits:
            if time_left_frac() < 15:
                break
            h_cur = h_init.copy()
            s_cur = score_c5(h_cur)

            for T in temps[:4]:
                if time_left_frac() < 10:
                    break
                h_new = slsqp_step(h_cur, T, max_iter=400)
                s_new = score_c5(h_new)
                if s_new < s_cur - 1e-12:
                    h_cur = h_new
                    s_cur = s_new

            if time_left_frac() > 10:
                tl = min(15, time_left_frac() * 0.15)
                if tl > 3:
                    h_cur, s_cur = local_search_continuous(h_cur, tl, t0, rng, step=0.1)

            if s_cur < best_c5:
                best_c5 = s_cur
                best_h = h_cur.copy()
                best_n = n

    if best_h is not None and time.time() - t0 < budget_s - 25:
        for T in [0.05, 0.02, 0.01, 0.005]:
            if time.time() - t0 > budget_s - 15:
                break
            h_new = slsqp_step(best_h, T, max_iter=800)
            s_new = score_c5(h_new)
            if s_new < best_c5 - 1e-12:
                best_c5 = s_new
                best_h = h_new

        if time.time() - t0 < budget_s - 10:
            tl = budget_s - 10 - (time.time() - t0)
            if tl > 5:
                best_h, best_c5 = local_search_continuous(best_h, tl, t0, rng, step=0.05)

        if time.time() - t0 < budget_s - 5:
            h_new = slsqp_step(best_h, 0.005, max_iter=500)
            s_new = score_c5(h_new)
            if s_new < best_c5 - 1e-12:
                best_c5 = s_new
                best_h = h_new

    if best_h is None:
        best_h = project_feasible(np.full(85, 0.5))
        best_c5 = score_c5(best_h)
        best_n = 85

    best_h = project_feasible(best_h)
    best_c5 = score_c5(best_h)
    return best_h, best_c5, len(best_h)
