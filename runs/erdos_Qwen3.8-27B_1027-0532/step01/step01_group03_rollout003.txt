import numpy as np
from scipy.optimize import minimize

def _c5_of(h):

    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / len(h))

def _subgrad_step(h, lr, top_k=1):
    n = len(h)
    corr = _corr_of(h)
    # Find top k lags
    top_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_indices:
        lag = k - (n - 1)
        # d(corr_lag)/d(h[j]) = (1-h[j+lag]) * [0<=j+lag<n] - h[j-lag] * [0<=j-lag<n]
        for j in range(n):
            g = 0.0
            j2 = j + lag
            if 0 <= j2 < n:
                g += (1.0 - h[j2])
            j3 = j - lag
            if 0 <= j3 < n:
                g -= h[j3]
            grad[j] += g
    # Normalize
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy()
    h_new = h - lr * grad / norm
    return _fix_sum(h_new, n / 2.0)

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _corr_grad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    elif lag < 0:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def _subgrad_descent(h_init, n, n_iter, lr0, lr_decay, top_k, momentum, check_every, deadline):
    target_sum = n / 2.0
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 20

    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            grad += _corr_grad(h, lag)
        grad /= len(top_idx)
        v = momentum * v - lr * grad
        h = _fix_sum(h + v, target_sum)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_refine(h_init, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()
