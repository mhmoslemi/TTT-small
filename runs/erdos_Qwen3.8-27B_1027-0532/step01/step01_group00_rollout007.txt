    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)

    s = np.sum(h)

        interior = (h > 1e-10) & (h < 1.0 - 1e-10)

            np.clip(h, 0.0, 1.0, out=h)

    n = len(h)
    num_lags = 2 * n - 1
    m = np.arange(num_lags) - (n - 1)
    
    ip = i + m
    im = i - m
    
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)

    h_im = np.zeros(num_lags)
    h_ip[valid_p] = h[ip[valid_p]]
    h_im[valid_m] = h[im[valid_m]]
    
    beta = valid_p.astype(np.float64) * (1.0 - h_ip) - valid_m.astype(np.float64) * h_im
    alpha_arr = corr - h[i] * beta
    
    # Find t in [0,1] minimizing max_k (alpha_k + beta_k * t)
    f0 = np.max(alpha_arr)
    f1 = np.max(alpha_arr + beta)
    
    lo = min(f0, f1)
    hi = max(f0, f1)
    
    if hi - lo < 1e-14:
        return h[i]
    
    pos = beta > 1e-15
    neg = beta < -1e-15
    
    for _ in range(45):
        mid = (lo + hi) / 2.0
        # Check if feasible: exists t in [0,1] with alpha_k + beta_k*t <= mid for all k
        lower = 0.0
        upper = 1.0
        feasible = True
        
        # Check zero-beta constraints
        zero = ~(pos | neg)
        if np.any(zero):
            if np.max(alpha_arr[zero]) > mid + 1e-15:
                feasible = False
        
        if feasible:
            if np.any(pos):
                upper = min(upper, np.min((mid - alpha_arr[pos]) / beta[pos]))
            if np.any(neg):
                lower = max(lower, np.max((mid - alpha_arr[neg]) / beta[neg]))
            if lower > upper + 1e-14:
                feasible = False
        
        if feasible:
            hi = mid
        else:
            lo = mid
    
    # Find t at optimal value
    v = hi + 1e-13
    lower = 0.0
    upper = 1.0
    if np.any(pos):
        upper = min(upper, np.min((v - alpha_arr[pos]) / beta[pos]))
    if np.any(neg):
        lower = max(lower, np.max((v - alpha_arr[neg]) / beta[neg]))
    
    lower = max(0.0, lower)
    upper = min(1.0, upper)
    if lower > upper:
        t_opt = 0.0 if f0 <= f1 else 1.0
    else:
        t_opt = (lower + upper) / 2.0
    
    return np.clip(t_opt, 0.0, 1.0)

def coordinate_descent(h, max_sweeps=30, time_limit=None, t0=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    best_score = np.max(corr)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None and time.time() - t0 > time_limit:
            break
        
        any_change = False
        for i in range(n):
            new_val = cd_step_exact(h, corr, i)
            if abs(new_val - h[i]) > 1e-11:
                h[i] = new_val
                corr = compute_correlations(h)
                any_change = True
        
        cur_max = np.max(corr)
        if cur_max < best_score:
            best_score = cur_max
        if not any_change:
            break
    
    return h, best_score
