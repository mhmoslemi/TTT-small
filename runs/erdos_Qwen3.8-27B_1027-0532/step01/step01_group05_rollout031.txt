def compute_impulse(h, n, p):
    """Compute the impulse response vector for position p.
    imp[k] = (1-h[p+k]) * 1[0<=p+k<n] - h[p-k] * 1[0<=p-k<n]
    Returns array of length 2n-1 indexed by lag offset (lag + n - 1).
    """
    imp = np.zeros(2*n - 1)
    # First term: (1-h[p+k]) for k from -p to n-1-p
    k_min = -p
    k_max = n - 1 - p
    if k_max >= k_min:
        lags = np.arange(k_min, k_max + 1)
        idx = lags + n - 1
        imp[idx] += 1.0 - h[p + lags]
    # Second term: -h[p-k] for k from p-(n-1) to p
    k_min2 = p - (n - 1)
    k_max2 = p
    if k_max2 >= k_min2:
        lags = np.arange(k_min2, k_max2 + 1)
        idx = lags + n - 1
        imp[idx] -= h[p - lags]
    return imp
