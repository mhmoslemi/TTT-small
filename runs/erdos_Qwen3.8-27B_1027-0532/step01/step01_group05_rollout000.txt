    _, weights, _, dx = smooth_max_vals(h, n, t)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-13:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            n_lag = n - lag
            grad[lag:] += w * (1.0 - h[:n_lag])
            grad[:n_lag] += (-w) * h[lag:]
        else:
            l = -lag
            n_l = n - l
            grad[:n_l] += w * (1.0 - h[l:])
            grad[l:] += (-w) * h[:n_l]
    grad *= dx
    penalty_grad = 2.0 * mu * (np.sum(h) - n / 2.0)
    grad += penalty_grad
    return grad

def lbgfsb_optimize(h0, n, t, mu, max_iter=200):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smooth_max_obj_fn, h0.copy(), method='L-BFGS-B',
            jac=smooth_max_grad_fn, args=(n, t, mu),
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'iprint': -1}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def slsqp_run(h0, n, t, max_iter=300):
    h0 = np.clip(h0.astype(float), 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n

    def obj_cb(x):
        return slsqp_obj(x, n, t)

    def jac_cb(x):
        return slsqp_grad(x, n, t)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        result = minimize(
            obj_cb, h0.copy(), method='SLSQP',
            jac=jac_cb,
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h_opt = np.clip(result.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def subgradient_single(h, n, n_iter=200):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.04
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_star = np.argmax(corr)
        lag = idx_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            n_lag = n - lag
            grad[lag:] += (1.0 - h[:n_lag])
            grad[:n_lag] += -h[lag:]
        else:
            l = -lag
            n_l = n - l
            grad[:n_l] += (1.0 - h[l:])
            grad[l:] += -h[:n_l]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h = h - step * grad / gnorm
        h = project_onto_constraint(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 25 == 24:
            step *= 0.93
    return best_h, best_c5


def subgradient_multi(h, n, n_iter=200, k_top=5, alpha=0.5):
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = 0.04
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_indices = np.argsort(corr)[-k_top:]
        max_corr = corr[top_indices[-1]]
        weights = np.exp(-alpha * (max_corr - corr[top_indices]))
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for w, idx in zip(weights, top_indices):
            lag = idx - (n - 1)
            if lag >= 0:
                n_lag = n - lag
                grad[lag:] += w * (1.0 - h[:n_lag])
                grad[:n_lag] += (-w) * h[lag:]
            else:
                l = -lag
                n_l = n - l
                grad[:n_l] += w * (1.0 - h[l:])
                grad[l:] += (-w) * h[:n_l]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h = h - step * grad / gnorm
        h = project_onto_constraint(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 25 == 24:
            step *= 0.93
    return best_h, best_c5


def binary_push(h, n, strength=0.05):
    h_push = h + strength * (2.0 * h - 1.0) * h * (1.0 - h)
    h_push = np.clip(h_push, 0.0, 1.0)
    h_push = project_onto_constraint(h_push, n)
    return h_push


def check_budget(start_time, budget_s, frac=1.0):
    return (time.time() - start_time) < budget_s * frac


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    parent = np.array(initial_h_values, dtype=float)
    n_parent = len(parent)

    best_h = parent.copy()
    best_n = n_parent
    best_c5 = compute_c5(best_h, best_n)

    # Phase 1: L-BFGS-B with penalty, fine temperature schedule
    h_curr = project_onto_constraint(parent.copy(), n_parent)
    c5_curr = compute_c5(h_curr, n_parent)
    mu = 50.0

    temps_phase1 = [400, 250, 150, 100, 80, 60, 50, 45, 40, 35, 30, 27, 25, 22, 20, 18, 16, 14, 12, 10, 9, 8, 7, 6, 5, 4.5, 4, 3.5, 3, 2.5, 2, 1.5, 1.2, 1.0]

    for t in temps_phase1:
        if not check_budget(start_time, budget_s, 0.35):
            break
        h_opt = lbgfsb_optimize(h_curr, n_parent, t, mu, max_iter=150)
        c5_opt = compute_c5(h_opt, n_parent)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent
        if c5_opt < c5_curr:
            h_curr = h_opt
            c5_curr = c5_opt
        else:
            h_curr = best_h.copy()
            c5_curr = best_c5

    # Phase 2: SLSQP refinement at low temperatures
    if check_budget(start_time, budget_s, 0.42):
        for t in [15, 12, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5]:
            if not check_budget(start_time, budget_s, 0.50):
                break
            h_opt = slsqp_run(best_h, best_n, t, max_iter=250)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Multi-subgradient polish
    if check_budget(start_time, budget_s, 0.52):
        h_pol, c5_pol = subgradient_multi(best_h, best_n, n_iter=400, k_top=5, alpha=0.4)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: L-BFGS-B again on the polished solution
    if check_budget(start_time, budget_s, 0.58):
        for t in [20, 15, 10, 8, 6, 4, 3, 2, 1.5, 1.0]:
            if not check_budget(start_time, budget_s, 0.65):
                break
            h_opt = lbgfsb_optimize(best_h, best_n, t, mu, max_iter=200)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Finer grid
    if check_budget(start_time, budget_s, 0.66):
        n_fine = n_parent * 2
        if n_fine <= 560:
            x_src = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_fine, endpoint=False)
            h_interp = np.interp(x_new, x_src, best_h)
            h_interp = project_onto_constraint(h_interp, n_fine)
            c5_interp = compute_c5(h_interp, n_fine)

            for t in [60, 40, 30, 25, 20, 15, 12, 10, 8, 6, 5, 4, 3, 2, 1.5, 1.0]:
                if not check_budget(start_time, budget_s, 0.78):
                    break
                h_opt = lbgfsb_optimize(h_interp, n_fine, t, mu, max_iter=120)
                c5_opt = compute_c5(h_opt, n_fine)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_fine
                    h_interp = h_opt

    # Phase 6: Random restarts
    if check_budget(start_time, budget_s, 0.80):
        for i in range(15):
            if not check_budget(start_time, budget_s, 0.90):
                break
            noise_scale = 0.015 + 0.02 * (i % 4)
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_onto_constraint(h_pert, best_n)
            t_use = [30, 20, 15, 10][i % 4]
            h_opt = lbgfsb_optimize(h_pert, best_n, t_use, mu, max_iter=150)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final subgradient polish
    if check_budget(start_time, budget_s, 0.92):
        h_pol, c5_pol = subgradient_multi(best_h, best_n, n_iter=300, k_top=7, alpha=0.3)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Very fine subgradient with single gradient
    if check_budget(start_time, budget_s, 0.96):
        h_pol2, c5_pol2 = subgradient_single(best_h, best_n, n_iter=300)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
