def compute_smooth_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = c - c_max
    exp_shifted = np.exp(alpha * shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    smooth_val = c_max + np.log(w_sum) / alpha
    
    idx = np.arange(n)
    m_arr = np.arange(2 * n - 1)
    # idx_A[i,m] = i - m + n - 1 (for the (1-h[...]) term)
    # idx_B[i,m] = i + m - n + 1 (for the h[...] term with negative sign)

    A = np.where(valid_A, 1.0 - h[idx_A], 0.0)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_smooth_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = c - c_max
    exp_shifted = np.exp(alpha * shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    smooth_val = c_max + np.log(w_sum) / alpha
    
    idx = np.arange(n)
    m_arr = np.arange(2 * n - 1)
    idx_A = idx[:, None] - m_arr[None, :] + n - 1
    idx_B = idx[:, None] + m_arr[None, :] - n + 1
    valid_A = (idx_A >= 0) & (idx_A < n)
    valid_B = (idx_B >= 0) & (idx_B < n)
    
    h_clipped = np.clip(h, 0.0, 1.0)
    A = np.where(valid_A, 1.0 - h_clipped[idx_A], 0.0)
    B = np.where(valid_B, h_clipped[idx_B], 0.0)
    
    grad = dx * (A @ w - B @ w)
    return smooth_val, grad

def project_feasible(h, target):
    h_c = np.clip(h, -10.0, 10.0)
    lo = -np.max(h_c) - 0.5
    hi = 1.0 - np.min(h_c) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        proj = np.clip(h_c + mid, 0.0, 1.0)
        if proj.sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_c + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad_descent(h_init, n, alpha_start, alpha_end, max_iter, deadline):
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = alpha_start
    lr_base = 0.3
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        f_val, grad = compute_smooth_and_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        step = lr_base
        
        improved = False
        for _ls in range(30):
            h_new = h + step * direction
            h_new = project_feasible(h_new, target)
            f_new, _ = compute_smooth_and_grad(h_new, alpha)
            if f_new <= f_val - 1e-5 * step:
                h = h_new
                improved = True
                break
            step *= 0.5
            if step < 1e-10:
                break
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if it % 50 == 0 and it > 0:
            alpha *= 1.15
            if alpha > alpha_end:
                alpha = alpha_end
        
        if not improved:
            lr_base *= 0.8
            if lr_base < 1e-8:
                break
    
    return best_h, best_c5

def penalty_method(h_init, n, deadline):
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    
    stages = [(1.0, 10), (10.0, 20), (100.0, 50), (1000.0, 100), 
              (10000.0, 200), (100000.0, 500)]
    
    for mu, alpha in stages:
        if time.time() > deadline:
            break
            
        def obj(v):
            h_c = np.clip(v, 0.0, 1.0)
            sv, _ = compute_smooth_and_grad(h_c, alpha)
            pen = mu * (h_c.sum() - target) ** 2
            return sv + pen
        
        def grad(v):
            h_c = np.clip(v, 0.0, 1.0)
            _, g = compute_smooth_and_grad(h_c, alpha)
            gp = 2.0 * mu * (h_c.sum() - target)
            return g + gp
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
            h = project_feasible(res.x, target)
        except Exception:
            break
    
    return h

def make_fourier_init(n, rng):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    n_terms = rng.randint(2, 8)
    for _ in range(n_terms):
        freq = rng.randint(1, 12)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        h += amp * np.cos(2.0 * np.pi * freq * t / 2.0 + phase)
    h = np.clip(h + 0.5, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def make_periodic_init(n, rng):
    period = rng.choice([3, 4, 5, 6, 7, 8, 9, 10, 12])
    pattern = rng.uniform(0.0, 1.0, size=period)
    # Adjust to have mean 0.5
    pattern = pattern - pattern.mean() + 0.5
    pattern = np.clip(pattern, 0.0, 1.0)
    h = np.tile(pattern, n // period + 1)[:n]
    return project_feasible(h, n / 2.0)

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 12)

        start = int(n * b / n_blocks)

        end = min(n, start + length)
        h[start:end] = 1.0

def make_binary_init(n, rng):
