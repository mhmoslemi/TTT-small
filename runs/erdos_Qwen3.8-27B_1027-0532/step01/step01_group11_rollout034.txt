```python
def smooth_gradient_v2(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    
    # For each m, compute: sum_k w[k] * ((1-h[m+s_k]) * 1[valid] - h[m-s_k] * 1[valid])
    # where s_k = k - (n-1)
    
    # Vectorized: create matrix M where M[m,k] = (1-h[m+s_k])*1[valid] - h[m-s_k]*1[valid]
    # This is n x (2n-1), which for n=90 is 90 x 179 = 16110 elements. Fine.
    
    m_idx = np.arange(n)
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # m + s: shape (n, 2n-1)
    ms = m_idx[:, None] + s[None, :]  # m + s_k
    ms_valid = (ms >= 0) & (ms < n)
    
    ms_neg = m_idx[:, None] - s[None, :]  # m - s_k
    ms_neg_valid = (ms_neg >= 0) & (ms_neg < n)
    
    # (1-h[m+s_k]) * 1[valid]
    term1 = np.zeros((n, 2*n-1))
    term1[ms_valid] = 1.0 - h[ms[ms_valid]]
    
    # h[m-s_k] * 1[valid]
    term2 = np.zeros((n, 2*n-1))
    term2[ms_neg_valid] = h[ms_neg[ms_neg_valid]]
    
    grad = dx * ((term1 - term2) @ w)
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    m_idx = np.arange(n)
    s = np.arange(2 * n - 1) - (n - 1)
    ms = m_idx[:, None] + s[None, :]
    ms_valid = (ms >= 0) & (ms < n)
    mt = m_idx[:, None] - s[None, :]
    mt_valid = (mt >= 0) & (mt < n)
    term1 = np.zeros((n, 2 * n - 1))
    term1[ms_valid] = 1.0 - h[ms[ms_valid]]
    term2 = np.zeros((n, 2 * n - 1))
    term2[mt_valid] = h[mt[mt_valid]]
    return dx * ((term1 - term2) @ w)

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def optimize_slsqp(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def coordinate_descent(h, n_sweeps=5):
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-10:
                    continue
                h[i] = val
                h_proj = project_onto_constraint(h, target_sum)
                c5 = compute_c5(h_proj)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    h = h_proj
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def block_swap_perturb(h, rng, n_blocks_range=(3, 8)):
    n = len(h)
    h_pert = h.copy()
    nb_min, nb_max = n_blocks_range
    nb = rng.integers(nb_min, min(nb_max, n // 3) + 1)
    block_size = n // nb
    blocks = [slice(i * block_size, (i + 1) * block_size) for i in range(nb)]
    for _ in range(rng.integers(1, 4)):
        i, j = rng.choice(nb, size=2, replace=False)
        tmp = h_pert[blocks[i]].copy()
        h_pert[blocks[i]] = h_pert[blocks[j]]
        h_pert[blocks[j]] = tmp
    return project_onto_constraint(h_pert, n / 2.0)

def region_flip_perturb(h, rng):
    n = len(h)
    h_pert = h.copy()
    n_regions = rng.integers(1, 4)
    for _ in range(n_regions):
        start = rng.integers(0, n - 5)
        length = rng.integers(5, max(6, n // 3))
        end = min(start + length, n)
        h_pert[start:end] = 1.0 - h_pert[start:end]
    return project_onto_constraint(h_pert, n / 2.0)

def smooth_random_construction(rng, n):
    n_freqs = rng.integers(2, 6)
    freqs = rng.uniform(0.5, 4.0, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.05, 0.25, n_freqs)
    x = np.linspace(0, 2, n)
    h = 0.5
    for a, f, p in zip(amps, freqs, phases):
        h = h + a * np.sin(2 * np.pi * f * x + p)
    h = np.clip(h, 0.0, 1.0)
    return project_onto_constraint(h, n / 2.0)

def binary_with_structure(rng, n, pattern='alternating'):
    if pattern == 'alternating':
        block_len = rng.integers(2, max(3, n // 20))
        h = np.zeros(n)
        for i in range(0, n, block_len * 2):
            h[i:min(i + block_len, n)] = 1.0
        h = project_onto_constraint(h, n / 2.0)
    elif pattern == 'halves':
        h = np.zeros(n)
        split = n // 2 + rng.integers(-n // 5, n // 5 + 1)
        h[:split] = 1.0
        h = project_onto_constraint(h, n / 2.0)
    else:
        h = (rng.random(n) < 0.5).astype(float)
        h = project_onto_constraint(h, n / 2.0)
    return h

        maxiter = max(30, int(maxiter_base * remaining_frac))

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best = h_opt
    return best, best_c5
