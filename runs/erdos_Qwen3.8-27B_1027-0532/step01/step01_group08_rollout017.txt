def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    if max_tc > 500:
        max_tc = 500
    log_sum_exp = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
    obj = (log_sum_exp / tau) * dx
    weights = np.exp(tau_corr - max_tc)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights /= wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi + 1)
            grad[idx2] += w * (-h[idx2 + lag])
    return obj, grad * dx

        o, _ = soft_max_obj_and_grad(x, tau)
        return o

        _, g = soft_max_obj_and_grad(x, tau)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=constraints,

        if hasattr(res, 'x') and res.x is not None:

            return h_opt, compute_c5(h_opt)

        pass

            # Gradient of penalty
            g = np.zeros(n)

                    continue
                lag = m - (n - 1)
                scale = 2 * violations[m] * (2.0 / n)
                j_lo = max(0, lag)
                j_hi = min(n - 1, n - 1 + lag)
                if j_lo <= j_hi:
                    idx = np.arange(j_lo, j_hi + 1)

            h_new = project_onto_constraint(h_new, target_sum)
            h_pen = h_new
        else:

    return best_h, best_c5

This is a convex piecewise linear function. Its minimum is at a point where the slope changes from negative to positive. The slope at t (where differentiable) is the a_m of the active constraint.

Algorithm to find the minimum:
1. Start at t = 0. Find the active constraint m* = argmax_m (a_m * 0 + b_m) = argmax_m b_m.
2. If a_{m*} ≥ 0, the minimum is at t = 0.
3. Otherwise, find the next breakpoint: the smallest t > 0 where another constraint m' becomes active (i.e., a_{m'} * t + b_{m'} = a_{m*} * t + b_{m*}).
   This is t_break = (b_{m*} - b_{m'}) / (a_{m'} - a_{m*}) for all m' with a_{m'} > a_{m*}.
   Take the smallest positive t_break.
4. Move to t_break, update the active set, and repeat.

This is O(K²) in the worst case but typically much faster.

However, for simplicity and robustness, I'll stick with the grid search + refinement approach. It's simpler to implement correctly and still very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    if max_tc > 500:
        tau_corr = tau_corr - max_tc + 500
        max_tc = 500
    exp_vals = np.exp(tau_corr - max_tc)
    log_sum_exp = max_tc + np.log(np.sum(exp_vals))
    obj = (log_sum_exp / tau) * dx
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi + 1)
            grad[idx2] += w * (-h[idx2 + lag])
    return obj, grad * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_optimize(h0, n, target_sum, deadline, tau=80.0, max_iter=200):
    def obj_fun(x):
        o, _ = soft_max_obj_and_grad(x, tau)
        return o
    def grad_fun(x):
        _, g = soft_max_obj_and_grad(x, tau)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    remaining = deadline - time.time()
    if remaining < 10:
        return h0, compute_c5(h0)
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            h_opt = np.clip(res.x, 0.0, 1.0)
            h_opt = project_onto_constraint(h_opt, target_sum)
            c5 = compute_c5(h_opt)
            if c5 < compute_c5(h0):
                return h_opt, c5
    except Exception:
        pass
    return h0, compute_c5(h0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tau_corr = tau * corr
    max_tc = np.max(tau_corr)
    if max_tc > 500:
        tau_corr = tau_corr - max_tc + 500
        max_tc = 500
    exp_vals = np.exp(tau_corr - max_tc)
    log_sum_exp = max_tc + np.log(np.sum(exp_vals))
    obj = (log_sum_exp / tau) * dx
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            idx2 = np.arange(j2_lo, j2_hi + 1)
            grad[idx2] += w * (-h[idx2 + lag])
    return obj, grad * dx
