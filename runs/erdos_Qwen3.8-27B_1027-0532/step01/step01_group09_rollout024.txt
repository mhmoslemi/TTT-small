import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def coord_descent(h0, max_sweeps=50, tol=1e-10, tm=30):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h = project_feasible(h0.copy(), tgt)
    t0 = time.time()
    offsets = np.arange(2 * n - 1) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() - t0 > tm:
            break
        total_change = 0.0
        for i in range(n):
            old_t = h[i]
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full').astype(float)
            h[i] = old_t
            j1 = i + offsets
            j2 = i - offsets
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            D = np.zeros(2 * n - 1)
            mask = v1 & v2 & (offsets != 0)
            D[mask] = 1.0 - h[j1[mask]] - h[j2[mask]]
            mask1 = v1 & ~v2 & (offsets != 0)
            D[mask1] = -h[j1[mask1]]
            mask2 = ~v1 & v2 & (offsets != 0)
            D[mask2] = 1.0 - h[j2[mask2]]
            idxs = np.array([k for k in range(2 * n - 1) if k != n - 1])
            b_q = D[idxs] - 1.0
            c_q = corr0[idxs] - corr0[n - 1]
            disc = b_q * b_q - 4.0 * c_q
            valid = disc >= 0
            sqrt_disc = np.sqrt(np.maximum(disc, 0.0))
            t1 = (-b_q + sqrt_disc) * 0.5
            t2 = (-b_q - sqrt_disc) * 0.5
            t1 = t1[valid]
            t2 = t2[valid]
            t1 = t1[(t1 >= -1e-9) & (t1 <= 1.0 + 1e-9)]
            t2 = t2[(t2 >= -1e-9) & (t2 <= 1.0 + 1e-9)]
            t1 = np.clip(t1, 0.0, 1.0)
            t2 = np.clip(t2, 0.0, 1.0)
            cands_arr = np.concatenate(([0.0, 1.0], t1, t2))
            vals_mat = corr0[None, :] + D[None, :] * cands_arr[:, None]
            vals_mat[:, n - 1] += cands_arr - cands_arr * cands_arr
            max_vals = vals_mat.max(axis=1)
            best_idx = np.argmin(max_vals)
            best_t = cands_arr[best_idx]
            h[i] = best_t
            total_change += abs(best_t - old_t)
        h = project_feasible(h, tgt)
        if total_change < tol:
            break
    return h

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = T * (corr - corr.max())
    return np.log(np.exp(shifted).sum()) / T + corr.max()

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = T * (corr - corr.max())
    exp_shifted = np.exp(shifted - shifted.max())
    p = exp_shifted / exp_shifted.sum()
    offsets = np.arange(2 * n - 1) - (n - 1)
    grad = p[n - 1] * (1.0 - 2.0 * h)
    for idx in range(2 * n - 1):
        if idx == n - 1:
            continue
        off = offsets[idx]
        w = p[idx]
        if abs(w) < 1e-14:
            continue
        a = abs(off)
        if a >= n:
            continue
        vi = np.arange(a, n - a)
        if len(vi) == 0:
            continue
        d = 1.0 - h[vi + off] - h[vi - off]
        grad[vi] += w * d
    grad *= dx
    return grad

                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 200, 'gtol': 1e-10})
            h1 = project_feasible(r.x, tgt)
            c5 = eval_c5(h1)
            if c5 < best_c5:
                best_c5 = c5
                best = h1.copy()
                h = h1.copy()
        except Exception:
            pass
    return best

def selpolish(h0, tm=60):
    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = project_feasible(h0, tgt)
    t0 = time.time()

    def f(x):
        if time.time() - t0 > tm * 0.85:
            raise Exception("tl")
        return (np.correlate(x, 1.0 - x, 'full') * dx).max()

    def gfun(x):
        return x.sum() - tgt

    def gjac(x):
        return np.ones(n)

    try:
        r = minimize(f, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 20000, 'ftol': 1e-16})
        h1 = project_feasible(r.x, tgt)
        if eval_c5(h1) < eval_c5(h0):
            return h1
    except Exception:
        pass
    return h0

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def thue_morse_h(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    return project_feasible(h, n / 2.0)

def vander_corput_h(n):
    vc = np.zeros(n)
    for i in range(n):
        v = 0.0
        f = 1.0
        k = i + 1
        while k > 0:
            v += (k % 2) * f
            k //= 2
            f /= 2.0
        vc[i] = v
    h = (vc - vc.min()) / (vc.max() - vc.min() + 1e-12)
    return project_feasible(h, n / 2.0)

def block_h(n, nb):
    block_size = n // nb
    rem = n - block_size * nb
    h = np.zeros(n)
    for b in range(nb):
        start = b * block_size
        end = start + block_size + (1 if b < rem else 0)
        if b % 2 == 0:
            h[start:end] = 1.0
    return project_feasible(h, n / 2.0)

def sinusoidal_h(n, freq, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n / 2.0)

def random_binary_h(n, rng):
    h = (rng.random(n) > 0.5).astype(float)
    return project_feasible(h, n / 2.0)

def random_continuous_h(n, rng):
    h = rng.random(n)
    return project_feasible(h, n / 2.0)

def make_diverse_init(n, rng):
    r = rng.randint(8)
    if r == 0:
        return thue_morse_h(n)
    elif r == 1:
        return vander_corput_h(n)
    elif r == 2:
        nb = rng.randint(2, 8)
        return block_h(n, nb)
    elif r == 3:
        freq = float(rng.choice([1, 2, 3, 4, 5, 6]))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        return sinusoidal_h(n, freq, phase)
    elif r == 4:
        return random_binary_h(n, rng)
    elif r == 5:
        return random_continuous_h(n, rng)
    elif r == 6:
        h1 = thue_morse_h(n)
        h2 = random_binary_h(n, rng)
        return project_feasible(0.5 * h1 + 0.5 * h2, n / 2.0)
    else:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        f1 = float(rng.choice([2, 3, 4]))
        f2 = float(rng.choice([3, 5, 7]))
        h = 0.5 + 0.3 * np.sin(np.pi * f1 * x) + 0.2 * np.sin(np.pi * f2 * x)
        return project_feasible(h, n / 2.0)

def pair_swap_search(h0, n_pairs=200, tm=20):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    for _ in range(n_pairs):
        if time.time() - t0 > tm:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        best_t = hi
        best_val = best_c5
        for frac in np.linspace(0.0, 1.0, 11):
            new_hi = hi * (1.0 - frac) + hj * frac
            new_hj = hj * (1.0 - frac) + hi * frac
            if new_hi < -1e-9 or new_hi > 1.0 + 1e-9:
                continue
            if new_hj < -1e-9 or new_hj > 1.0 + 1e-9:
                continue
            h[i] = new_hi
            h[j] = new_hj
            c5 = eval_c5(h)
            if c5 < best_val:
                best_val = c5
                best_t = frac
        h[i] = hi * (1.0 - best_t) + hj * best_t
        h[j] = hj * (1.0 - best_t) + hi * best_t
        if best_val < best_c5:
            best_c5 = best_val
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def tl(frac=0.95):
        return budget_s * frac - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(200, 0.5)

    best = h_init.astype(float).copy()
    best_c5 = eval_c5(best)

    if tl(0.15) > 10:
        h_cd = coord_descent(best, max_sweeps=30, tol=1e-12, tm=min(45, tl(0.12)))
        c5_cd = eval_c5(h_cd)

    if tl(0.25) > 20:

        c5_sm = eval_c5(h_sm)
        if c5_sm < best_c5:
            best, best_c5 = h_sm, c5_sm
        if tl(0.25) > 10:
            h_cd2 = coord_descent(best, max_sweeps=20, tol=1e-12, tm=min(30, tl(0.22)))
            c5_cd2 = eval_c5(h_cd2)
            if c5_cd2 < best_c5:
                best, best_c5 = h_cd2, c5_cd2

    if tl(0.35) > 10:
        h_sw = pair_swap_search(best, n_pairs=300, tm=min(20, tl(0.32)))
        c5_sw = eval_c5(h_sw)
        if c5_sw < best_c5:
            best, best_c5 = h_sw, c5_sw
            h_cd3 = coord_descent(best, max_sweeps=20, tol=1e-12, tm=min(20, tl(0.33)))
            c5_cd3 = eval_c5(h_cd3)
            if c5_cd3 < best_c5:
                best, best_c5 = h_cd3, c5_cd2

    n_orig = len(best)
    for attempt in range(15):
        if tl(0.65) < 20:
            break
        h_try = make_diverse_init(n_orig, rng)
        h_try = coord_descent(h_try, max_sweeps=20, tol=1e-12, tm=min(25, tl(0.65) // 3))
        c5_try = eval_c5(h_try)
        if c5_try < best_c5:
            best, best_c5 = h_try, c5_try
            h_cd4 = coord_descent(best, max_sweeps=15, tol=1e-12, tm=min(15, tl(0.65) // 3))
            c5_cd4 = eval_c5(h_cd4)
            if c5_cd4 < best_c5:
                best, best_c5 = h_cd4, c5_cd4

    n_targets = [100, 140, 168, 200, 250, 300, 350, 400, 480, 560]
    for n_t in n_targets:
        if tl(0.85) < 20:
            break
        if n_t == n_orig:
            continue
        h_res = resample_h(best, n_t)
        h_res = coord_descent(h_res, max_sweeps=20, tol=1e-12, tm=min(35, tl(0.85) // 2))
        c5_res = eval_c5(h_res)
        if c5_res < best_c5:
            best, best_c5 = h_res, c5_res
            if tl(0.85) > 10:
                h_sm2 = softmax_optimize(best, tm=min(20, tl(0.85) // 2))
                c5_sm2 = eval_c5(h_sm2)

                    best, best_c5 = h_sm2, c5_sm2
                h_cd5 = coord_descent(best, max_sweeps=15, tol=1e-12, tm=min(15, tl(0.85) // 2))
                c5_cd5 = eval_c5(h_cd5)
                if c5_cd5 < best_c5:
                    best, best_c5 = h_cd5, c5_cd5

    if tl(0.92) > 15:
        h_f = coord_descent(best, max_sweeps=40, tol=1e-13, tm=min(60, tl(0.90)))

            best, best_c5 = h_final, cfinal

    final_c5 = eval_c5(best)
    return best, final_c5, len(best)
