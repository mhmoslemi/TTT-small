import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return compute_corr(h).max()

def corr_subgrad(h, k):
    """Subgradient of corr_k w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += (1.0 - h[a[m1]])
    grad[m2] -= h[b[m2]]
    grad *= dx
    return grad

def lse_obj_grad(h, beta):
    """LSE smooth objective and its gradient."""
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    if m <= 0:
        return m, np.zeros(n)
    shifted = beta * (corr - m)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    obj = m + np.log(Z) / beta
    weights = exp_s / Z
    # Gradient: weighted sum of subgradients
    grad = np.zeros(n)
    n_active = np.sum(weights > 1e-6)
    active_idx = np.where(weights > 1e-6)[0]
    for k_idx in active_idx:
        k = k_idx - (n - 1)
        grad += weights[k_idx] * corr_subgrad(h, k)
    return obj, grad

def lse_slsqp(h0, betas, target, time_limit=60):
    """Optimize LSE with SLSQP, sweeping beta."""
    n = len(h0)
    h = project_feasible(h0, target)
    t0 = time.time()
    
    for beta in betas:
        if time.time() - t0 > time_limit * 0.9:
            break
        
        def f(x):
            o, g = lse_obj_grad(project_feasible(x, target), beta)
            return o, g
        
        def constraint_fun(x):
            return x.sum() - target
        
        def constraint_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(f, h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                          options={'maxiter': 2000, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            if eval_c5(h_new) <= eval_c5(h) + 1e-10:
                h = h_new
        except Exception:
            pass
    
    return h

def subgrad_momentum(h0, target, time_limit=30, lr_init=0.3, momentum=0.9):
    """Subgradient method with momentum."""
    n = len(h0)
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    vel = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)

        k_star = np.argmax(corr) - (n - 1)

        h = project_feasible(h - lr * vel / (np.linalg.norm(vel) + 1e-10), target)

def golden_ratio_init(n):
    """Create bang-bang initialization using golden ratio."""
    phi = (1 + np.sqrt(5)) / 2
    ones = np.zeros(n, dtype=bool)
    for i in range(n // 2):
        ones[int(i * phi) % n] = True
    h = ones.astype(float)
    # Ensure exactly n//2 ones
    while h.sum() > n // 2:
        idx = np.where(ones)[0]
        ones[idx[np.argmax(idx)]] = False
        h = ones.astype(float)
    while h.sum() < n // 2:
        idx = np.where(~ones)[0]
        ones[idx[np.argmin(idx)]] = True
        h = ones.astype(float)
    return h

def thue_morse_init(n):
    """Create bang-bang initialization using Thue-Morse sequence."""
    def tm_seq(k):
        # Thue-Morse: t(n) = n AND (n>>1) AND ... (parity of set bits)
        return np.binary_repr(k).count('1') % 2
    # Generate 2*ceil(log2(n)) bits, take first n
    m = 1
    while m < n:
        m *= 2
    t = np.zeros(m, dtype=int)
    for i in range(1, m):
        t[i] = t[i >> 1] ^ (i & 1)
    # Use first n values, map 0->0, 1->1, ensure n//2 ones
    h = t[:n].astype(float)
    n_ones = int(h.sum())
    if n_ones > n // 2:
        # Flip some ones to zeros
        one_idx = np.where(h == 1)[0]
        for idx in one_idx[n_ones - n//2:]:
            h[idx] = 0
    elif n_ones < n // 2:
        zero_idx = np.where(h == 0)[0]
        for idx in zero_idx[:n//2 - n_ones]:
            h[idx] = 1
    return h

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return compute_corr(h).max()

def corr_subgrad(h, k):
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    a = j - k
    b = j + k
    grad = np.zeros(n)
    m1 = (a >= 0) & (a < n)
    m2 = (b >= 0) & (b < n)
    grad[m1] += (1.0 - h[a[m1]])
    grad[m2] -= h[b[m2]]
    grad *= dx
    return grad

def lse_obj_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    if beta < 1e-6:
        return m, corr_subgrad(h, np.argmax(corr) - (n - 1))
    shifted = beta * (corr - m)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    obj = m + np.log(Z) / beta
    weights = exp_s / Z
    grad = np.zeros(n)
    active_idx = np.where(weights > 1e-7)[0]
    for k_idx in active_idx:
        k = k_idx - (n - 1)
        grad += weights[k_idx] * corr_subgrad(h, k)
    return obj, grad

def lse_slsqp(h0, betas, target, time_limit=60):
    n = len(h0)
    h = project_feasible(h0, target)
    t0 = time.time()
    for beta in betas:
        if time.time() - t0 > time_limit * 0.85:
            break
        def fobj(x):
            xp = project_feasible(x, target)
            return lse_obj_grad(xp, beta)
        def cfun(x):
            return x.sum() - target
        def cjac(x):
            return np.ones(n)
        try:
            res = minimize(fobj, h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if eval_c5(h_new) <= eval_c5(h) + 1e-10:
                h = h_new
        except Exception:
            pass
    return h

def subgrad_momentum(h0, target, time_limit=30, lr_init=0.4, momentum=0.85):
    n = len(h0)
    h = project_feasible(h0, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    vel = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    while time.time() - t0 < time_limit:
        corr = compute_corr(h)
        k_star = int(np.argmax(corr)) - (n - 1)
        g = corr_subgrad(h, k_star)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        vel = momentum * vel + (1.0 - momentum) * g / gnorm
        h = project_feasible(h - lr * vel, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9997
        if lr < 1e-8:
            break
    return best

def nesterov_subgrad(h0, target, time_limit=30, lr_init=1.0, momentum=0.9):
    n = len(h0)
    h = project_feasible(h0, target)
    y = h.copy()
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    t = 1
    while time.time() - t0 < time_limit:
        corr = compute_corr(y)
        k_star = int(np.argmax(corr)) - (n - 1)
        g = corr_subgrad(y, k_star)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        eta = lr_init / (np.sqrt(t) * gnorm)
        h_new = project_feasible(y - eta * g, target)
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t / beta_t - 1.0) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        t += 1
    return best

def golden_ratio_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n // 2):
        h[int(i * phi) % n] = 1.0
    return h

def thue_morse_init(n):
    m = 1
    while m < n:
        m *= 2
    t = np.zeros(m, dtype=np.int32)
    for i in range(1, m):
        t[i] = t[i >> 1] ^ (i & 1)
    h = t[:n].astype(float)
    n_ones = int(h.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        one_idx = np.where(h == 1)[0]
        perm = np.random.default_rng(1).permutation(len(one_idx))
        for idx in one_idx[perm[:n_ones - target_ones]]:
            h[idx] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(h == 0)[0]
        perm = np.random.default_rng(1).permutation(len(zero_idx))
        for idx in zero_idx[perm[:target_ones - n_ones]]:
            h[idx] = 1.0
    return h

    h = 0.5 + 0.3 * h / (np.abs(h).max() + 1e-10)
    return np.clip(h, 0.0, 1.0)
