                               method='SLSQP', bounds=bounds_t, constraints=cons_t,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
                    h_cur = h_new
            except Exception:
                pass

    # Phase 6: Final high-precision polish
    n_final = len(best_h)
    _g_n = n_final
    _g_target_sum = n_final / 2.0
    bounds_f = [(0.0, 1.0)] * n_final
    cons_f = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]

    for alpha in [5000, 10000, 20000]:
        if _time_left() < 20:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(alpha,),
                           method='SLSQP', bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, _g_target_sum)
            s_new = compute_c5(h_new)
            if s_new < best_score:
                best_score = s_new
                best_h = h_new
        except Exception:
            pass

    # Phase 7: Very fine jitter polish
    if _time_left() > 30:
        for _ in range(10):
            if _time_left() < 15:
                break
            h_pert = best_h + rng.randn(n_final) * 0.0001
            h_pert = project_feasible(h_pert, _g_target_sum)
            try:
                res = minimize(obj_smooth, h_pert, args=(10000,),
                               method='SLSQP', bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, _g_target_sum)
                s_new = compute_c5(h_new)
                if s_new < best_score:
                    best_score = s_new
                    best_h = h_new
            except Exception:
                pass

    h_final = project_feasible(best_h, _g_target_sum)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
