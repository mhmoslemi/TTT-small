    lo = h.min() - 1.0

        mid = (lo + hi) / 2

        if s > target_sum:

            hi = mid

    if abs(diff) > 1e-10:

        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if mask.sum() > 0:

        if time.time() - t0 > budget_s - 300:
            break
        dl = time.time() + min(35, (budget_s - 300 - (time.time() - t0)) / max(1, len(alphas_pgd) - alphas_pgd.index(alpha)))
        h = pgd_optimize(best_h, alpha, target, 8000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    for alpha in [50, 150, 500, 1000, 2000]:
        if time.time() - t0 > budget_s - 240:
            break
        dl = time.time() + 30
        h = fista_optimize(best_h, alpha, target, 6000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    for i in range(400):
        if time.time() - t0 > budget_s - 120:
            break
        if i % 25 == 0:
            sorted_idx = np.argsort(-best_h)
            binary = np.zeros(n)
            binary[sorted_idx[:n // 2]] = 1.0
            pert = binary + rng.randn(n) * 0.15
        elif i % 25 == 12:
            mask = (best_h > 0.05) & (best_h < 0.95)
            pert = best_h.copy()
            if mask.sum() > 0:
                pert[mask] = np.clip(pert[mask] + rng.randn(mask.sum()) * 0.1, 0, 1)
            else:
                pert = best_h + rng.randn(n) * 0.02
        else:
            scale = 0.005 + 0.05 * (i / 400.0)
            pert = best_h + rng.randn(n) * scale
        pert = project_onto_constraint(pert, target)
        alpha = 200 + 800 * rng.randint(4)
        dl = time.time() + 12
        h = pgd_optimize(pert, alpha, target, 3000, dl)
        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    if time.time() - t0 < budget_s - 60:
        for alpha in [5000, 10000]:
            if time.time() - t0 > budget_s - 40:
                break
            try:
                h = lbfgsb_refine(best_h, alpha, target, time.time() + 20)
                h = project_onto_constraint(np.clip(h, 0, 1), target)
                s = evaluate_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
            except Exception:
                pass

    for n_try in [n + 75, n + 150, 2 * n, 3 * n]:
        if n_try > 1800 or time.time() - t0 > budget_s - 80:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_onto_constraint(h_init, n_try / 2.0)
        for alpha in [50, 200, 500, 1500]:
            if time.time() - t0 > budget_s - 40:
                break
            dl = time.time() + 18
            h = pgd_optimize(h_init, alpha, n_try / 2.0, 5000, dl)
            h = project_onto_constraint(h, n_try / 2.0)
            s = evaluate_score(h)
            if s < best_score:
                best_score = s
                best_h = h.copy()
                h_init = h.copy()

    for alpha in [8000, 15000, 30000]:
        if time.time() - t0 > budget_s - 25:
            break
        nf = len(best_h)
        dl = time.time() + 8
        h = pgd_optimize(best_h, alpha, nf / 2.0, 5000, dl)
        h = project_onto_constraint(h, nf / 2.0)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
