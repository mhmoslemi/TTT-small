```python
def subgradient_line_search(h, n, max_iters=5000):

    best_c5 = evaluate_c5(h)

    n_corr = 2 * n - 1

        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)

        active = np.where(corr >= max_val - 1e-8)[0]
```

```python
def gradient_local_search(h, n, eps=200, max_iters=1000):
    """Fast gradient descent with projection"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    
    lr = 0.01
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        weights = np.exp(shifted) / np.exp(shifted).sum()
        
        # Gradient of smooth max
        grad = np.zeros(n)
        for k in range(2*n-1):
            w = weights[k]
            if w < 1e-10:
                continue
            l = k - (n-1)
            # Compute gradient of c[k] at h
            # c[k] = sum_{i in V} h[i]*(1-h[i-l])
            # d/dh[j] = I(j in V) - h[j-l]*I(j in V, valid) - h[j+l]*I(j+l in V)
            i_low = max(0, l)
            i_high = min(n-1, n-1+l)
            
            # I(j in V) for j in [i_low, i_high]
            grad[i_low:i_high+1] += w
            
            # -h[j-l]*I(j in V, 0<=j-l<n)
            for j in range(i_low, i_high+1):
                if 0 <= j-l < n:
                    grad[j] -= w * h[j-l]
            
            # -h[j+l]*I(j+l in V)
            jl_low = i_low - l
            jl_high = i_high - l
            for j in range(max(0, jl_low), min(n, jl_high+1)):
                grad[j] -= w * h[j+l]
        
        grad *= dx
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            # Decay learning rate
            if it % 100 == 0 and it > 0:
                lr *= 0.8
        else:
            # Try smaller step
            lr *= 0.7
    
    return best_h, best_c5
```

```python
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - (n - 1)
        # grad[j] += w * [(1-x[j-l]) if 0<=j-l<n] + w * [-x[j+l] if 0<=j+l<n]
        # Vectorize:
        # Term 1: for j where 0 <= j-l < n, i.e., max(0,l) <= j < n+l (clipped to [0,n))
        j1_start = max(0, l)
        j1_end = min(n, n + l)
        if j1_start < j1_end:
            grad[j1_start:j1_end] += w * (1.0 - x[j1_start - l:j1_end - l])
        # Term 2: for j where 0 <= j+l < n, i.e., max(0,-l) <= j < n-l (clipped to [0,n))
        j2_start = max(0, -l)
        j2_end = min(n, n - l)
        if j2_start < j2_end:
            grad[j2_start:j2_end] += w * (-x[j2_start + l:j2_end + l])
    grad *= dx
    return grad

def make_constraint(n):
    def constraint(x):
        return np.sum(x) - n / 2.0
    return constraint

def slsqp_optimize(h, n, eps, maxiter=300):
    def obj(x):
        return smooth_max_obj(x, n, eps)
    def grad(x):
        return smooth_max_grad(x, n, eps)
    def cons(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj, h.copy(), jac=grad, method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h.copy(), evaluate_c5(h)

def upsample_linear(h, factor=2):
    n = len(h)
    new_n = n * factor
    h_new = np.zeros(new_n)
    h_new[0::factor] = h
    if factor == 2:
        h_new[1::2] = (h[:-1] + h[1:]) / 2.0
    else:
        for i in range(1, factor):
            t = i / factor
            h_new[i::factor] = (1-t) * h[:-1] + t * h[1:]
    return h_new

def diverse_perturb(h, n, r, rng):
    strategy = r % 6
    if strategy == 0:
        scale = 0.005 + 0.003 * (r // 6)
        return h + scale * rng.randn(n)
    elif strategy == 1:

        block = rng.randint(3, min(20, n//4))
        start = rng.randint(0, n - block)
        h_new[start:start+block] = 1.0 - h_new[start:start+block]
        return h_new
    elif strategy == 2:

        for _ in range(rng.randint(2, 8)):
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        return h_new
    elif strategy == 3:

        sorted_idx = np.argsort(h_new)
        h_new[sorted_idx[n//2:]] = np.minimum(1.0, h_new[sorted_idx[n//2:]] + 0.3)
        h_new[sorted_idx[:n//2]] = np.maximum(0.0, h_new[sorted_idx[:n//2]] - 0.3)

    elif strategy == 4:
```

OK, let me also think about whether I should try a different approach entirely. What about using **cvxpy** to solve a related convex problem?

The problem is: minimize t subject to c[k](h) ≤ t, 0≤h≤1, sum(h)=n/2.

The constraints c[k](h) ≤ t are quadratic (non-convex since the Hessian of c[k] is indefinite... wait, let me check).

c[k](h) = sum_i h[i] - sum_i h[i]*h[i-l]

The Hessian is -2 * A where A is a symmetric matrix with A[i,j] = 1/2 if j=i±l (and appropriate boundary handling). This matrix A is positive semi-definite (it's related to a shift operator), so the Hessian is negative semi-definite, meaning c[k] is concave.

So the constraint c[k](h) ≤ t is a constraint of the form "concave function ≤ constant", which is a non-convex constraint. So cvxpy can't solve this directly.

However, we could try a **sequential convex approximation**: linearize the c[k] functions at the current point and solve the resulting LP. This is the cutting plane / linearization approach.

At point h*, the linearization of c[k] is:
c[k](h) ≈ c[k](h*) + ∇c[k](h*)^T (h - h*)

The constraint c[k](h) ≤ t becomes:
c[k](h*) + ∇c[k](h*)^T (h - h*) ≤ t

This is linear! So we can solve:
minimize t
subject to c[k](h*) + ∇c[k](h*)^T (h - h*) ≤ t for all k
           0 ≤ h ≤ 1
           sum(h) = n/2

This is a linear program! We can solve it with cvxpy.

The idea is:
1. Start at h* (current best)
2. Solve the LP to get a new point h'
3. Update h* = h' (or h* = convex combination)
4. Repeat

This is essentially the **cutting plane method** or ** Kelley's method** for non-convex optimization. It's guaranteed to converge to a local minimum (under some conditions).

This could be much faster than SLSQP because:
- Each LP solve is fast (especially with a good solver)
- We can solve many iterations cheaply
- The linearization gives a global lower bound at each step

```python
import cvxpy as cp

def cutting_plane(h, n, max_iters=100):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Compute correlations and gradients at h
        corr = np.correlate(h, 1 - h, mode="full")
        n_corr = 2 * n - 1
        
        # Solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0
        ]
        
        for k in range(n_corr):
            l = k - (n - 1)
            # Gradient of c[k] at h
            grad_k = np.zeros(n)
            j1_start = max(0, l)
            j1_end = min(n, n + l)
            if j1_start < j1_end:
                grad_k[j1_start:j1_end] = 1.0 - h[j1_start - l:j1_end - l]
            j2_start = max(0, -l)
            j2_end = min(n, n - l)
            if j2_start < j2_end:
                grad_k[j2_start:j2_end] -= h[j2_start + l:j2_end + l]
            
            # c[k](h) + grad^T (h_var - h) <= t_var
            constraints.append(corr[k] + grad_k @ (h_var - h) <= t_var)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)  # or ECOS, or CBC
        except:
            try:
                prob.solve(solver=cp.ECOS)
            except:
                break
        
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c5
```

```python
def simulated_annealing(h, n, budget_s=600, seed=42):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    
    T = 0.01  # Initial temperature
    T_min = 0.0001
    cooling = 0.999
    
    while time.time() - start_time < budget_s:
        T *= cooling
        if T < T_min:
            T = T_min  # Stay at minimum temperature
        
        # Generate perturbation
        strategy = rng.randint(0, 5)
        if strategy == 0:
            h_try = current_h + 0.01 * rng.randn(n)
        elif strategy == 1:
            h_try = current_h.copy()
            idx = rng.choice(n, size=rng.randint(1, 4), replace=False)
            h_try[idx] = 1.0 - h_try[idx]
        elif strategy == 2:
            h_try = current_h.copy()
            i, j = rng.choice(n, 2, replace=False)
            h_try[i], h_try[j] = h_try[j], h_try[i]
        elif strategy == 3:
            h_try = current_h.copy()
            start = rng.randint(0, n-5)
            length = rng.randint(2, 6)
            h_try[start:start+length] = 1.0 - h_try[start:start+length]
        else:
            h_try = current_h + 0.005 * rng.randn(n)
        
        h_try = project_feasible(h_try, n)
        c5_try = evaluate_c5(h_try)
        
        delta = c5_try - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_h = h_try
            current_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    return best_h, best_c5
```

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - (n - 1)
        j1s = max(0, l)
        j1e = min(n, n + l)
        if j1s < j1e:
            grad[j1s:j1e] += w * (1.0 - x[j1s - l:j1e - l])
        j2s = max(0, -l)
        j2e = min(n, n - l)
        if j2s < j2e:
            grad[j2s:j2e] += w * (-x[j2s + l:j2e + l])
    grad *= dx
    return grad

    def obj(x):
        return smooth_max_obj(x, n, eps)
    def grad(x):
        return smooth_max_grad(x, n, eps)
    def cons(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n

        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': cons}],

    )
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h.copy(), evaluate_c5(h)
