def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    
    if abs(h.sum() - target) < 1e-10:
        return h
    
    # Bisection on lambda: h_proj[i] = clip(h[i] + lambda, 0, 1)
    # We need sum(clip(h[i] + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def evaluate_c5(h):

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Bisection: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def make_lse_objective_and_grad(n):
    dx = 2.0 / n
    
    def objective_and_grad(h, beta):
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = corr * dx  # length 2n-1
        
        # Log-sum-exp
        max_g = np.max(g)
        exp_g = np.exp(beta * (g - max_g))
        Z = np.sum(exp_g)
        w = exp_g / Z  # weights, length 2n-1
        
        obj = (max_g + np.log(Z)) / beta
        
        # Gradient using symmetry
        # W_p for p = 0,...,n-1: W_p = w[n-1+p] + w[n-1-p] (for p>0), W_0 = w[n-1]
        W = np.zeros(n)
        W[0] = w[n-1]
        for p in range(1, n):
            W[p] = w[n-1+p] + w[n-1-p]
        
        # conv1[j] = sum_{p=0}^{j} W_p * (1-h[j-p])
        conv1 = np.convolve(1.0 - h, W, mode='full')[:n]
        
        # conv2[j] = sum_{p=0}^{n-1-j} W_p * h[j+p]
        W_rev = W[::-1]
        conv_full = np.convolve(h, W_rev, mode='full')
        conv2 = conv_full[n-1:2*n-1]
        
        grad = (dx / beta) * (conv1 - conv2)
        
        return obj, grad

        corr = np.correlate(x, 1.0 - x, mode="full")

            # Final repair
            r.x = project_feasible(r.x, n)
            return r.x, evaluate_c5(r.x)
    except Exception:
        pass
    return h, evaluate_c5(h)

def subgradient_polish(h, n, steps=2000, base_lr=0.01):
    """Subgradient method for the max problem"""
    dx = 2.0 / n
    best = h.copy()
    best_c5 = evaluate_c5(best)
    
    for t in range(steps):
        lr = base_lr / (1 + t * 0.01)
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        g = corr * dx
        max_g = np.max(g)
        
        # Find active lags (within 0.005 of max)
        active_mask = g >= max_g - 0.005
        num_active = np.sum(active_mask)
        
        if num_active == 0:
            break
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for idx in np.where(active_mask)[0]:
            lag = idx - (n - 1)
            p = abs(lag)
            if p == 0:
                grad += (1.0 - 2.0 * h) * dx
            else:
                # For lag = p (positive):
                # d/dh[j] = dx * [1_{p<=j} (1-h[j-p]) - 1_{j<=n-1-p} h[j+p]]
                if p <= j:  # this is per-element, let me vectorize
                    pass
            
        # Vectorized version:
        grad = np.zeros(n)
        for idx in np.where(active_mask)[0]:
            lag = idx - (n - 1)
            p = abs(lag)
            if p == 0:
                grad += (1.0 - 2.0 * h)
            else:
                # contribution for |lag| = p
                mask1 = np.arange(n) >= p
                mask2 = np.arange(n) <= n - 1 - p
                grad[mask1] += 1.0 - h[np.arange(n)[mask1] - p]
                grad[mask2] -= h[np.arange(n)[mask2] + p]
        
        grad *= dx
        grad /= num_active
        
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        
        h = h_new
    
    return best, best_c5
