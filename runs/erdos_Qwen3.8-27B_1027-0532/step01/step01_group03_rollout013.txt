```python
    """Find optimal h[j] given h[jp] is adjusted to maintain sum."""
    # h[j] + h[jp] must remain constant
    const = h[j] + h[jp]
    L = max(0, const - 1)
    U = min(1, const)
    if L >= U:
        return h[j]

    # new result[k] = R_k + delta_k[j] * (t - h[j]) + delta_k[jp] * ((const - t) - h[jp])

    # = R_k + (delta_k[j] - delta_k[jp]) * (t - h[j])
    # = (delta_k[j] - delta_k[jp]) * t + (R_k - (delta_k[j] - delta_k[jp]) * h[j])

    # First compute current correlations
    one_minus_h = 1.0 - h
    R = np.correlate(h, one_minus_h, mode='full')

    # For index i, the valid range of k is:
    #   First term valid: 0 ≤ i+k-n+1 ≤ n-1 → n-1-i ≤ k ≤ n-1+i
    #   Second term valid: 0 ≤ i-k+n-1 ≤ n-1 → i ≤ k ≤ i+n-1
    
    def compute_delta(i):
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _logsumexp_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return 0.0
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _logsumexp_max(corr, alpha) * (2.0 / n)

def _compute_deltas(h, n):
    """Compute delta_k[i] for all i and k.
    delta[k, i] = (1-h[i+k-n+1])_valid - h[i-k+n-1]_valid
    """
    m = 2 * n - 1
    deltas = np.zeros((m, n))
    for i in range(n):
        # First term: 1-h[i+k-n+1] for valid k
        k_start1 = max(0, n - 1 - i)
        k_end1 = min(m - 1, n - 1 + i)
        if k_start1 <= k_end1:
            idx = (i + np.arange(k_start1, k_end1 + 1) - n + 1)
            deltas[k_start1:k_end1+1, i] += 1.0 - h[idx]
        # Second term: -h[i-k+n-1] for valid k
        k_start2 = max(0, i)
        k_end2 = min(m - 1, i + n - 1)
        if k_start2 <= k_end2:
            idx = (i - np.arange(k_start2, k_end2 + 1) + n - 1)
            deltas[k_start2:k_end2+1, i] -= h[idx]
    return deltas

def _coord_descent_sweep(h, n, target_sum, max_sweeps=20, rng=None):
    """Coordinate descent with pivot strategy."""
    if rng is None:
        rng = np.random.RandomState(0)
    m = 2 * n - 1
    best_c5 = _c5_of(h)
    
    for sweep in range(max_sweeps):
        # Choose pivot
        pivot = rng.randint(n)
        order = [i for i in range(n) if i != pivot]
        rng.shuffle(order)
        
        for j in order:
            jp = pivot
            const = h[j] + h[jp]
            L = max(0.0, const - 1.0)
            U = min(1.0, const)
            if L >= U - 1e-12:
                continue
            
            # Compute current correlations
            R = np.correlate(h, 1.0 - h, mode='full')
            
            # Compute effective slopes for this pair
            # new_R[k] = R[k] + delta[k,j]*(t - h[j]) + delta[k,jp]*((const-t) - h[jp])
            # = R[k] + delta[k,j]*(t - h[j]) - delta[k,jp]*(t - h[j])
            # = R[k] + (delta[k,j] - delta[k,jp])*(t - h[j])
            # = a[k]*t + b[k]
            # where a[k] = delta[k,j] - delta[k,jp]
            #       b[k] = R[k] - a[k]*h[j]
            
            # Compute deltas for j and jp
            # delta[k, i] = (1-h[i+k-n+1])_valid - h[i-k+n-1]_valid
            dkj = np.zeros(m)
            dkjp = np.zeros(m)
            
            # For index j:
            k1s = max(0, n - 1 - j)
            k1e = min(m - 1, n - 1 + j)
            if k1s <= k1e:
                idx = j + np.arange(k1s, k1e + 1) - n + 1
                dkj[k1s:k1e+1] += 1.0 - h[idx]
            k2s = max(0, j)
            k2e = min(m - 1, j + n - 1)
            if k2s <= k2e:
                idx = j - np.arange(k2s, k2e + 1) + n - 1
                dkj[k2s:k2e+1] -= h[idx]
            
            # For index jp:
            k1s = max(0, n - 1 - jp)
            k1e = min(m - 1, n - 1 + jp)
            if k1s <= k1e:
                idx = jp + np.arange(k1s, k1e + 1) - n + 1
                dkjp[k1s:k1e+1] += 1.0 - h[idx]
            k2s = max(0, jp)
            k2e = min(m - 1, jp + n - 1)
            if k2s <= k2e:
                idx = jp - np.arange(k2s, k2e + 1) + n - 1
                dkjp[k2s:k2e+1] -= h[idx]
            
            a = dkj - dkjp
            b = R - a * h[j]
            
            # Find min of max_k (a[k]*t + b[k]) for t in [L, U]
            # Compute breakpoints
            # For efficiency, use numpy
            A_diff = a[:, None] - a[None, :]  # m x m
            B_diff = b[None, :] - b[:, None]  # m x m
            with np.errstate(divide='ignore', invalid='ignore'):
                T = B_diff / A_diff
            # Keep upper triangle valid breakpoints
            row_idx, col_idx = np.triu_indices(m, k=1)
            T_upper = T[row_idx, col_idx]
            T_upper = T_upper[np.isfinite(T_upper)]
            T_upper = T_upper[(T_upper >= L) & (T_upper <= U)]
            
            candidates = np.concatenate([[L, U], T_upper])
            # Evaluate objective at candidates
            # obj[t] = max_k (a[k]*t + b[k])
            vals = np.max(a[:, None] * candidates[None, :] + b[:, None], axis=0)
            best_idx = np.argmin(vals)
            best_t = candidates[best_idx]
            
            # Update
            h[j] = best_t
            h[jp] = const - best_t
            h[jp] = np.clip(h[jp], 0.0, 1.0)
            # If clipping changed jp, fix j
            if h[jp] != const - h[j]:
                h[j] = const - h[jp]
                h[j] = np.clip(h[j], 0.0, 1.0)
            
            # Check improvement
            c5_now = _c5_of(h)
            if c5_now < best_c5 - 1e-10:
                best_c5 = c5_now
        # End sweep
    return h, best_c5

def _do_slsqp(h_start, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_obj(x, alpha, n)
    def obj_grad(x):
        # Gradient of smooth max
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = np.exp(alpha * (corr - np.max(corr)))
        w /= np.sum(w)
        # Gradient of corr[k] w.r.t. x[i]
        # corr[k] = sum_i x[i]*(1-x[i+k-n+1])
        # d(corr[k])/dx[i] = (1-x[i+k-n+1])_valid - x[i-k+n-1]_valid
        m = 2 * n - 1
        grad = np.zeros(n)
        for i in range(n):
            k1s = max(0, n - 1 - i)
            k1e = min(m - 1, n - 1 + i)
            if k1s <= k1e:
                idx = i + np.arange(k1s, k1e + 1) - n + 1
                grad[i] += np.sum(w[k1s:k1e+1] * (1.0 - x[idx]))
            k2s = max(0, i)
            k2e = min(m - 1, i + n - 1)
            if k2s <= k2e:
                idx = i - np.arange(k2s, k2e + 1) + n - 1
                grad[i] -= np.sum(w[k2s:k2e+1] * x[idx])
        grad *= (2.0 / n)
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start
```

```python
def _delta_single(h, i, n):
    """Compute delta[k, i] for a single index i."""
    m = 2 * n - 1
    delta = np.zeros(m)
    # First term: 1-h[i+k-n+1] for valid k
    k1s = max(0, n - 1 - i)
    k1e = min(m - 1, n - 1 + i)
    if k1s <= k1e:
        idx = i + np.arange(k1s, k1e + 1) - n + 1
        delta[k1s:k1e+1] += 1.0 - h[idx]
    # Second term: -h[i-k+n-1] for valid k
    k2s = max(0, i)
    k2e = min(m - 1, i + n - 1)
    if k2s <= k2e:
        idx = i - np.arange(k2s, k2e + 1) + n - 1
        delta[k2s:k2e+1] -= h[idx]
    return delta
```

```python
def _min_max_linear(a, b, L, U):

    m = len(a)
    # Evaluate at boundaries
    fL = np.max(a * L + b)
    fU = np.max(a * U + b)
    
    # Find breakpoints in [L, U]
    if m < 2:
        return L if fL <= fU else U
    
    # Compute all pairwise intersection points
    A = a[:, None] - a[None, :]
    B = b[None, :] - b[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):

    # Upper triangle
    ri, ci = np.triu_indices(m, k=1)
    T_upper = T[ri, ci]

    T_upper = T_upper[(T_upper >= L) & (T_upper <= U)]

    # Evaluate objective at each candidate
    # f(t) = max_k (a[k]*t + b[k])
    vals = np.max(a[:, None] * candidates[None, :] + b[:, None], axis=0)
    best_idx = np.argmin(vals)
```

For the smooth objective:
S_α(h) = (2/n) * (1/α) * log(Σ_k exp(α * R_k(h)))

= (2/n) * Σ_k w_k * delta[k,j]

So the gradient is:
∂S/∂h[j] = (2/n) * Σ_k w_k * delta[k,j]

This can be computed as: (2/n) * (D @ w) where D[k,j] = delta[k,j]

Or equivalently, for each j:
grad[j] = (2/n) * Σ_k w[k] * delta[k,j]

Let me compute delta matrix D of shape (m, n) once, then:

```python
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _delta_single(h, i, n):
    m = 2 * n - 1
    delta = np.zeros(m)
    k1s = max(0, n - 1 - i)
    k1e = min(m - 1, n - 1 + i)
    if k1s <= k1e:
        idx = i + np.arange(k1s, k1e + 1) - n + 1
        delta[k1s:k1e + 1] += 1.0 - h[idx]
    k2s = max(0, i)
    k2e = min(m - 1, i + n - 1)
    if k2s <= k2e:
        idx = i - np.arange(k2s, k2e + 1) + n - 1
        delta[k2s:k2e + 1] -= h[idx]
    return delta

def _min_max_linear(a, b, L, U):
    m = len(a)
    if m < 2:
        return (L + U) / 2.0
    A = a[:, None] - a[None, :]
    B = b[None, :] - b[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        T = B / A
    ri, ci = np.triu_indices(m, k=1)
    T_upper = T[ri, ci]
    T_upper = T_upper[np.isfinite(T_upper)]
    T_upper = T_upper[(T_upper >= L) & (T_upper <= U)]
    candidates = np.concatenate(([L, U], T_upper))
    if len(candidates) > 5000:
        candidates = np.unique(candidates)
    vals = np.max(a[:, None] * candidates[None, :] + b[:, None], axis=0)
    best_idx = np.argmin(vals)
    return candidates[best_idx]

def _coord_descent(h, n, target_sum, max_sweeps=30, rng=None, time_limit=None, t_start=None):
    m = 2 * n - 1
    best_c5 = _c5_of(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        pivot = rng.randint(n)
        order = list(range(n))
        order.remove(pivot)
        rng.shuffle(order)
        R = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for j in order:
            jp = pivot
            const = h[j] + h[jp]
            L = max(0.0, const - 1.0)
            U = min(1.0, const)
            if L >= U - 1e-12:
                continue
            dkj = _delta_single(h, j, n)
            dkjp = _delta_single(h, jp, n)
            a_vec = dkj - dkjp
            b_vec = R - a_vec * h[j]
            best_t = _min_max_linear(a_vec, b_vec, L, U)
            if abs(best_t - h[j]) > 1e-10:
                delta_t = best_t - h[j]
                h[j] = best_t
                h[jp] = const - best_t
                R += a_vec * delta_t
                improved = True
        if improved:
            c5_now = _c5_of(h)
            if c5_now < best_c5:
                best_c5 = c5_now
        else:
            break
    return h, best_c5

def _slsqp_smooth(h_start, n, target_sum, alpha, maxiter=300, ftol=1e-13):
    m = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return _logsumexp_max(corr, alpha) * (2.0 / n)

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mcorr = np.max(corr)
        w = np.exp(np.clip(alpha * (corr - mcorr), -500, 500))
        w /= np.sum(w)
        g = np.zeros(n)
        for j in range(n):
            k1s = max(0, n - 1 - j)
            k1e = min(m - 1, n - 1 + j)
            if k1s <= k1e:
                idx = j + np.arange(k1s, k1e + 1) - n + 1
                g[j] += np.sum(w[k1s:k1e + 1] * (1.0 - x[idx]))
            k2s = max(0, j)
            k2e = min(m - 1, j + n - 1)
            if k2s <= k2e:
                idx = j - np.arange(k2s, k2e + 1) + n - 1
                g[j] -= np.sum(w[k2s:k2e + 1] * x[idx])
        g *= (2.0 / n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def run(seed=42, budget_s=1300, **kwargs):
    n = 86
    target_sum = n / 2.0
    t_start = time.time()
    deadline = t_start + budget_s - 10

    def elapsed_ok(margin=20):
        return time.time() < deadline - margin

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(n, 0.5)

    h0 = _fix_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = _c5_of(h0)

    rng = np.random.RandomState(seed)

    if elapsed_ok(60):
        h_cd, c5_cd = _coord_descent(best_h, n, target_sum, max_sweeps=50, rng=rng,
                                     time_limit=50, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(120):
        for alpha in [50, 100, 200, 400]:
            if not elapsed_ok(25):
                break
            h_new = _slsqp_smooth(best_h, n, target_sum, alpha, maxiter=200, ftol=1e-13)
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    if elapsed_ok(60):
        h_cd, c5_cd = _coord_descent(best_h, n, target_sum, max_sweeps=60, rng=rng,
                                     time_limit=50, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(300):
        for attempt in range(40):
            if not elapsed_ok(15):
                break
            scale = 0.02 + 0.06 * rng.rand()
            h_pert = best_h + scale * rng.randn(n)
            h_pert = _fix_sum(h_pert, target_sum)
            h_cd, c5_cd = _coord_descent(h_pert, n, target_sum, max_sweeps=10, rng=rng,
                                         time_limit=15, t_start=t_start)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    if elapsed_ok(120):
        for alpha in [800, 2000, 5000]:
            if not elapsed_ok(25):
                break
            h_new = _slsqp_smooth(best_h, n, target_sum, alpha, maxiter=400, ftol=1e-14)
            h_new = _fix_sum(h_new, target_sum)
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    if elapsed_ok(60):
        h_cd, c5_cd = _coord_descent(best_h, n, target_sum, max_sweeps=60, rng=rng,
                                     time_limit=50, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(60):
        h_block = np.zeros(n)
        h_block[:int(target_sum)] = 1.0
        h_block = _fix_sum(h_block, target_sum)
        h_cd, c5_cd = _coord_descent(h_block, n, target_sum, max_sweeps=25, rng=rng,
                                     time_limit=30, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(50):
        x = np.linspace(0, 2, n, endpoint=False)
        h_sine = 0.5 + 0.3 * np.cos(np.pi * x)
        h_sine = _fix_sum(h_sine, target_sum)
        h_cd, c5_cd = _coord_descent(h_sine, n, target_sum, max_sweeps=25, rng=rng,
                                     time_limit=30, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(40):
        h_alt = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * np.linspace(0, 1, n, endpoint=False))
        h_alt = _fix_sum(h_alt, target_sum)
        h_cd, c5_cd = _coord_descent(h_alt, n, target_sum, max_sweeps=20, rng=rng,
                                     time_limit=25, t_start=t_start)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed_ok(30):
        h_new = _slsqp_smooth(best_h, n, target_sum, 10000, maxiter=500, ftol=1e-15)
        h_new = _fix_sum(h_new, target_sum)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)

    return (best_h, best_c5, n)
```
