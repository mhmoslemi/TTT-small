import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = tau * (corr - cmax)
    exp_vals = np.exp(shifted)
    logsumexp = cmax + np.log(np.sum(exp_vals)) / tau
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    j = np.arange(n)
    n_active = 0
    for ki in range(len(corr)):
        wi = w[ki]
        if wi < 1e-10:
            continue
        n_active += 1
        shift = ki - (n - 1)
        a = j - shift
        b = j + shift
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += wi * (1.0 - h[a[m1]])
        grad[m2] -= wi * h[b[m2]]
    grad *= dx
    return logsumexp, grad

def nesterov_smoothed(h0, tm, tau_init=200.0, tau_end=40.0, n_iter=8000):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    y = h.copy()
    t_curr = 1.0
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > tm:
            break
        frac = min(1.0, it / max(1, n_iter))
        tau = tau_init * (tau_end / tau_init) ** frac
        obj, grad = smooth_obj_and_grad(y, tau)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_curr * t_curr)) / 2.0
        beta = (t_curr - 1.0) / t_new
        lr = 2.0 / (1.0 + tau)
        y_new = y - lr * grad
        x_new = project_feasible(y_new, tgt)
        y = (1.0 + beta) * x_new - beta * h
        h = x_new
        if it % 20 == 0:
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    c5 = eval_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best = h.copy()
    return best

def adaptive_grad_descent(h0, tm, tau=80.0, n_iter=10000):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    lr = 1.0 / tau
    for it in range(n_iter):
        if time.time() - t0 > tm:
            break
        obj, grad = smooth_obj_and_grad(h, tau)
        h_new = project_feasible(h - lr * grad, tgt)
        obj_new, _ = smooth_obj_and_grad(h_new, tau)
        tries = 0
        while obj_new > obj + 1e-6 * lr * np.dot(grad, h - h_new) and tries < 8:
            lr *= 0.5
            h_new = project_feasible(h - lr * grad, tgt)
            obj_new, _ = smooth_obj_and_grad(h_new, tau)
            tries += 1
        if obj_new < obj:
            lr = min(lr * 1.05, 2.0 / tau)
        h = h_new
        if it % 10 == 0:
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    return best

def block_refine(h0, tm, block_size=8):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    tau = 80.0
    outer = 0
    while time.time() - t0 < tm and outer < 50:
        outer += 1
        improved = False
        for start in range(0, n, block_size):
            if time.time() - t0 > tm:
                break
            end = min(start + block_size, n)
            indices = np.arange(start, end)
            h_block = h[indices].copy()
            h_fixed = h.copy()
            def block_obj(xb):
                h_temp = h_fixed.copy()
                h_temp[indices] = np.clip(xb, 0.0, 1.0)
                penalty = 0.5 * (h_temp.sum() - tgt) ** 2
                obj, _ = smooth_obj_and_grad(h_temp, tau)
                return obj + penalty
            try:
                res = minimize(block_obj, h_block, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * len(indices),
                               options={'maxiter': 100, 'ftol': 1e-10})
                if res.success and res.fun < block_obj(h_block) - 1e-10:
                    h[indices] = np.clip(res.x, 0.0, 1.0)
                    h = project_feasible(h, tgt)
                    c5 = eval_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best = h.copy()
                        improved = True
            except Exception:
                pass
        if not improved:
            break
    return best

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float)
    xd = np.linspace(0.0, float(n_src), n_dst, endpoint=False)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def make_structured_init(n, rng):

    elif r == 1:

    elif r == 3:

    elif r == 4:

        for level in range(rng.randint(2, 5)):

    return project_feasible(np.clip(h, 0.0, 1.0), n / 2.0)

def binary_push(h, strength=0.3):
    h_pushed = np.where(h > 0.5,
                        np.minimum(1.0, h + strength * (1.0 - h)),
                        np.maximum(0.0, h - strength * h))
    return project_feasible(h_pushed, len(h) / 2.0)

    n = len(h0)
    h = h0 + rng.normal(0, scale, size=n)
