def smooth_obj(h, n, t):
    corr = get_corr_scaled(h, n)
    m = np.max(corr)
    shifted = t * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    return m + np.log(np.sum(np.exp(shifted))) / t


def smooth_grad(h, n, t):
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    corr = corr_raw * dx
    m = np.max(corr)
    shifted = t * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    e = np.exp(shifted)
    s = np.sum(e)
    w = e / s

        wi = w[idx]

            continue

            if n - lag > 0:
                grad[:n - lag] += wi * (1.0 - h[lag:])
                grad[lag:] += -wi * h[:n - lag]
        else:
            l = -lag
            if n - l > 0:
                grad[l:] += wi * (1.0 - h[:n - l])
                grad[:n - l] += -wi * h[l:]
    return grad * dx


def slsqp_opt(h0, n, t, max_iter=800):
    h0 = np.clip(h0, 0.0, 1.0)

    def obj(x):
        return smooth_obj(x, n, t)

    def grad(x):
        return smooth_grad(x, n, t)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraint(h_opt, n)
        return h_opt
    except Exception:
        return h0.copy()


def projected_gradient(h0, n, t, n_iter=200, lr_init=0.05):
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = lr_init
    prev_obj = smooth_obj(h, n, t)
    for it in range(n_iter):
        g = smooth_grad(h, n, t)
        step = lr
        for _ in range(20):
            h_new = project_onto_constraint(h - step * g, n)
            obj_new = smooth_obj(h_new, n, t)
            if obj_new <= prev_obj - 1e-12 * step * np.linalg.norm(g):
                break
            step *= 0.5
        h = h_new
        prev_obj = obj_new

def nesterov_subgrad(h0, n, t_start=5.0, t_end=200.0, n_iter=500):
    h = h0.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        g = smooth_grad(h, n, t)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        v_new = ((it / beta) * v + g) / (1.0 + it / beta)
        alpha = 1.0 / (t ** 0.5 * (it + 1) ** 0.5 + 1e-8)
        h_new = project_onto_constraint(h - alpha * v_new, n)
        v = v_new
        h = h_new

            best_h = h.copy()
