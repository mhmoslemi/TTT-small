    f_hi = f(x_hi)

        active = vals >= m - 1e-15
        if np.any(active):
            a_active = a[active]
            # subgradient is [min(a_active), max(a_active)]
            return np.min(a_active), np.max(a_active)
        return 0, 0

            hi = mid
        elif smin > 0:  # f is increasing at mid, go right
            lo = mid
        else:  # 0 is in subgradient, mid is optimal
            return mid

def c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        if diff > 0:
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_corr(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = corr - m
    return float(m + np.log(np.sum(np.exp(shifted))) / beta)

def constraint_sum(h, n):
    return float(np.sum(h) - n * 0.5)

def run_slsqp(h0, n, dx, beta, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    res = minimize(smooth_max_corr, h0, args=(dx, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return repair_h(res.x, n)
    return h0
