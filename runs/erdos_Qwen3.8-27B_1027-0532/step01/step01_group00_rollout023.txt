    n = len(h)
    dx = 2.0 / n

    corr = np.correlate(h, one_minus_h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if abs(np.sum(h) - target_sum) < 1e-13:
        return h.copy()
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_try = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_try) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / alpha

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    return e / np.sum(e)

def grad_smooth_max(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:

    return grad
