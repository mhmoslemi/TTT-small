    lo = -1.0
    hi = 1.0
    for _ in range(80):

        p = np.clip(h - mid, 0.0, 1.0)

def soft_max_grad(h, tau):

    tau_corr = tau * corr

In fictitious play for this problem:
1. Maintain a weight vector w over the 2n-1 lags
2. Compute the "average" overlap: F(h) = sum_k w[k] * corr[k] * dx
3. This is a quadratic function of h: F(h) = dx * sum_k w[k] * sum_i h[i]*(1-h[i+k])
4. Optimize F(h) subject to 0≤h≤1, sum(h)=n/2 using L-BFGS-B with penalty
5. Update w: w[k] proportional to exp(corr[k] * dx / temperature) or just put weight on the top few

The key advantage: the weighted sum F(h) is a quadratic form in h, so its gradient is linear in h, making each step very fast.

Let me work out the quadratic form:
F(h) = dx * sum_k w[k] * sum_i h[i] - h[i]*h[i+k]

The first term: dx * sum_k w[k] * sum_i h[i] = dx * (sum_k w[k]) * (sum_i h[i])
= dx * W_total * sum_i h[i]  (linear in h)

The second term: -dx * sum_k w[k] * sum_i h[i]*h[i+k]
This is a quadratic form: -dx * h^T * A * h

where A is a (2n-1) block matrix... actually it's simpler.

For each lag k_m = m - (n-1):
sum_i h[i] * h[i+k_m] = sum_i h[i] * h[i+k_m]

This is the sum of products of h with itself shifted by k_m. So:
Q(h) = sum_m w[m] * sum_i h[i]*h[i+m-(n-1)]

This can be written as h^T M h / 2 where M is a circulant-like matrix.

Actually, let's think of it as:
Q(h) = sum_{i,j} h[i] * h[j] * (sum_m w[m] * 1_{j-i = m-(n-1)})
= sum_{i,j} h[i] * h[j] * w[j-i+(n-1)]

where w[k] = 0 for k < 0 or k > 2n-2.

So Q(h) = h^T W_mat h where (W_mat)[i,j] = w[j-i+(n-1)].

The gradient is: ∇Q(h) = 2 * W_mat * h (since W_mat is symmetric when w is symmetric, but it might not be).

Actually, W_mat[i,j] = w[j-i+(n-1)]. For this to be symmetric, we need w[j-i+(n-1)] = w[i-j+(n-1)], which means w must be symmetric: w[k] = w[2(n-1)-k]. The weights from soft-max won't necessarily be symmetric.

But the gradient of h^T M h is (M + M^T) h. So:
∇Q(h) = 2 * ((W_mat + W_mat^T)/2) * h = (W_mat + W_mat^T) * h

And W_mat + W_mat^T has entries:
[(W_mat + W_mat^T)][i,j] = w[j-i+(n-1)] + w[i-j+(n-1)]

This is a symmetric matrix that's a function of (j-i), so it's a **symmetrized circulant-like** matrix (more precisely, a Hankel matrix in the lag domain).

We can compute the matrix-vector product (W_mat + W_mat^T) * h efficiently using FFT:

Then [(W_mat + W_mat^T)] * h = conv(s, h) evaluated appropriately.

This is the cross-correlation of w_centered with h, where w_centered[l] = w[(n-1)+l].

Using FFT: cross-corr can be computed as ifft(fft(w_centered_padded) * conj(fft(h_padded)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_value(val, tau):
    m = np.max(val)
    return m + np.log(np.sum(np.exp(val - m))) / tau

def soft_max_weights(val, tau):
    tau_val = tau * val
    m = np.max(tau_val)
    w = np.exp(tau_val - m)
    w /= np.sum(w)
    return w

def weighted_corr_grad(h, weights, dx):
    """Compute gradient of sum_k weights[k] * corr[k] * dx w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        lag = m - (n - 1)
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        j2_lo = max(0, -lag)
        j2_hi = min(n - 1, n - 1 - lag)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] += w * (-h[j2_lo + lag:j2_hi + lag + 1])
    return grad * dx

def lbfgsb_optimize(h0, tau, deadline, penalty=1e6):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        tau_corr = tau * corr
        max_tc = np.max(tau_corr)
        lse = max_tc + np.log(np.sum(np.exp(tau_corr - max_tc)))
        sm = (lse / tau) * dx
        pen = penalty * (x.sum() - target) ** 2
        return sm + pen
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        w = soft_max_weights(corr, tau)
        g = weighted_corr_grad(x, w, dx)
        g += 2 * penalty * (x.sum() - target) * np.ones(n)
        return g
    
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8,
                                'maxls': 50, 'disp': False})
        h_opt = np.clip(res.x, 0, 1)
        h_opt = project_onto_constraint(h_opt, target)
        c5 = compute_c5(h_opt)
        return h_opt, c5
    except:
        return h0, compute_c5(h0)

def fictitious_play(h0, deadline, max_iter=2000):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Initialize weights uniformly
    K = 2 * n - 1
    w = np.ones(K) / K
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute current correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_current = np.max(corr) * dx
        
        if c5_current < best_c5 - 1e-12:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Compute gradient using current weights
        g = weighted_corr_grad(h, w, dx)
        
        # Line search
        alpha = 0.01
        for _ in range(20):
            h_new = h - alpha * g
            h_new = project_onto_constraint(h_new, target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            c5_new = np.max(corr_new) * dx
            if c5_new < c5_current:
                h = h_new
                break
            alpha *= 0.7
        else:
            h = project_onto_constraint(h - alpha * g, target)
        
        # Update weights: put more on worst lags
        tau_w = 50.0
        tau_corr = tau_w * corr
        max_tc = np.max(tau_corr)
        w = np.exp(tau_corr - max_tc)
        w /= np.sum(w)
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    sorted_lags = np.argsort(corr)[::-1][:5]
    n_selected = len(sorted_lags)
    w = np.zeros(2 * n - 1)
    for idx in sorted_lags:
        w[idx] = 1.0 / n_selected
    g = weighted_corr_grad(h0, w, dx)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.04
    for it in range(3000):
        if time.time() > deadline:
            break
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            lr *= 0.95
            if lr < 0.003:
                corr = np.correlate(h, 1.0 - h, mode="full")
                sorted_lags = np.argsort(corr)[::-1][:5]
                w = np.zeros(2 * n - 1)
                for idx in sorted_lags:
                    w[idx] = 1.0 / n_selected
                g = weighted_corr_grad(h, w, dx)
                lr = 0.04
                h = best_h
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 25

    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is not None:
        n0 = len(h_init)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = project_onto_constraint(h_init, n0 / 2.0)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)
    else:
        n0 = 120
        best_h = np.full(n0, 0.5)
        best_h = project_onto_constraint(best_h, n0 / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at native resolution with increasing tau
    for tau in [40, 80, 120, 180]:
        if time.time() > deadline - 400:
            break
        h_opt, c5_opt = lbfgsb_optimize(best_h, tau, deadline, penalty=1e5, maxiter=400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: PGD refinement at native resolution
    if time.time() < deadline - 300:
        n = len(best_h)
        h_opt, c5_opt = pgd_refine(best_h, deadline - 150, tau=150, max_iter=3000, lr_init=0.04)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Multi-scale to higher resolution
    for target_n in [200, 300]:
        if time.time() > deadline - 200:
            break
        n_cur = len(best_h)
        if target_n <= n_cur:
            continue
        h_interp = interpolate_h(best_h, target_n)
        h_interp = project_onto_constraint(h_interp, target_n / 2.0)
        for tau in [60, 120]:
            if time.time() > deadline - 120:
                break
            h_opt, c5_opt = lbfgsb_optimize(h_interp, tau, deadline, penalty=1e5, maxiter=300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_interp = h_opt.copy()

    # Phase 4: Fictitious play at best resolution
    if time.time() < deadline - 150:
        h_opt, c5_opt = fictitious_play(best_h, deadline - 80, max_iter=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Multi-lag focus
    if time.time() < deadline - 80:
        h_opt, c5_opt = multi_lag_focus(best_h, deadline - 40)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Final PGD polish
    if time.time() < deadline - 30:
        h_opt, c5_opt = pgd_refine(best_h, deadline - 15, tau=200, max_iter=2000, lr_init=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Worst-case PGD
    if time.time() < deadline - 5:
        h_opt, c5_opt = worst_case_pgd(best_h, deadline, max_iter=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    n_final = len(best_h)
    best_h = project_onto_constraint(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
