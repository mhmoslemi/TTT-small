```python
def compute_swap_delta(h, i, j, n):
    # Returns delta for all lags (as an array of length 2n-1)
    # delta[k + n - 1] = h[i+k] + h[i-k] - h[j-k] - h[j+k]
    # using h[k]=0 for k outside [0, n-1]
    
    # h[i+k] for k from -(n-1) to n-1:
    # When k >= 0: h[i+k] valid if i+k < n
    # When k < 0: h[i+k] valid if i+k >= 0
    # This is h convolved with a delta at position i, evaluated at all offsets
    
    # Simpler approach:
    # Create a helper: h_ext[k] = h[k] if 0<=k<n else 0
    # delta[m] for m = 0 to 2n-2 (correlate index):
    # k = m - (n-1)
    # delta[m] = h_ext[i+k] + h_ext[i-k] - h_ext[j-k] - h_ext[j+k]
    
    # Let's use a vectorized approach
    k = np.arange(-(n-1), n)  # lags
    i_k = i + k
    i_k2 = i - k
    j_k = j + k
    j_k2 = j - k
    
    def safe_index(idx):
        return np.where((idx >= 0) & (idx < n), idx, 0)
    
    def safe_val(idx):
        return np.where((idx >= 0) & (idx < n), h[safe_index(idx)], 0.0)
    
    delta = safe_val(i_k) + safe_val(i_k2) - safe_val(j_k) - safe_val(j_k2)
    return delta
```

```python
    """Efficient swap-based local search for binary (or near-binary) sequences."""
    n = len(h)

            break

        j = zeros[rng.randint(len(zeros))]

        ik = i + k_arr
        ikm = i - k_arr
        jk = j + k_arr
        jkm = j - k_arr

            zeros = zeros[zeros != j]
```
