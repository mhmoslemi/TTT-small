import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_onto_constraint(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def lse_value(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(corr - m)))) / beta

def lse_grad(h, beta, threshold=1e-14):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = corr - m
    log_sum = np.log(np.sum(np.exp(shifted)))
    weights = np.exp(shifted - log_sum)
    
    grad = np.zeros(n)
    
    for j in range(2 * n - 1):
        w = weights[j]
        if w < threshold:
            continue
        s = j - (n - 1)
        wd = w * dx
        
        if s > 0:
            grad[:n-s] += wd * (1.0 - h[s:])
            grad[s:] -= wd * h[:n-s]
        elif s < 0:
            t = -s
            grad[t:] += wd * (1.0 - h[:n-t])
            grad[:n-t] -= wd * h[t:]
        else:
            grad += wd * (1.0 - 2.0 * h)
    
    return grad

def projected_gradient_step(h, grad, alpha, n):
    """Take a projected gradient step."""
    x_new = h - alpha * grad
    return project_onto_constraint(x_new, n)
