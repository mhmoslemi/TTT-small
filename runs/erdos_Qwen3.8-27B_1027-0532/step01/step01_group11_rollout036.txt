import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_refine(h_init, eps_list, maxiter_per_eps, time_limit):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    start = time.time()
    for eps in eps_list:
        if time.time() - start > time_limit:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        try:
            result = minimize(smooth_objective, h, args=(eps,),
                              jac=smooth_gradient, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter_per_eps, 'ftol': 1e-14})
            h = project_onto_constraint(result.x, target_sum)
        except Exception:
            pass
    return h, compute_c5(h)

def de_step(pop, fitness, F, CR, rng, target_sum, n):
    pop_size = len(pop)
    for i in range(pop_size):
        mask = np.ones(pop_size, dtype=bool)
        mask[i] = False
        candidates = np.where(mask)[0]
        r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
        v = pop[r1] + F * (pop[r2] - pop[r3])
        trial = pop[i].copy()
        j_rand = rng.integers(n)
        for j in range(n):
            if rng.random() < CR or j == j_rand:
                trial[j] = v[j]
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_onto_constraint(trial, target_sum)
        trial_fit = compute_c5(trial)
        if trial_fit < fitness[i]:
            pop[i] = trial
            fitness[i] = trial_fit
    return pop, fitness

def subgrad_polish(h, n_steps, lr_init=0.01):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_steps + 1):
        lr = lr_init / np.sqrt(t)
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        k_star = np.argmax(corr)
        grad = np.zeros(n)
        s = k_star - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] = dx * omh[:n-s]
                if n-s > 0:
                    grad[:n-s] -= dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n-r] = dx * omh[r:]
                grad[r:] -= dx * h[:n-r]
        h = project_onto_constraint(h - lr * grad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_polish(h, rng, max_rounds=20):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for _ in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for idx in order:
            for d in deltas:
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + d, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wk * dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= wk * dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:n - r]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_refine(h_init, eps_list, maxiter_per_eps, time_limit):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    start = time.time()
    for eps in eps_list:
        if time.time() - start > time_limit:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
        try:
            result = minimize(smooth_objective, h, args=(eps,),
                              jac=smooth_gradient, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter_per_eps, 'ftol': 1e-14})
            h = project_onto_constraint(result.x, target_sum)
        except Exception:
            pass
    return h, compute_c5(h)

def de_step(pop, fitness, F, CR, rng, target_sum, n):
    pop_size = len(pop)
    for i in range(pop_size):
        mask = np.ones(pop_size, dtype=bool)
        mask[i] = False
        candidates = np.where(mask)[0]
        r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
        v = pop[r1] + F * (pop[r2] - pop[r3])
        trial = pop[i].copy()
        j_rand = rng.integers(n)
        for j in range(n):
            if rng.random() < CR or j == j_rand:
                trial[j] = v[j]
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_onto_constraint(trial, target_sum)
        trial_fit = compute_c5(trial)
        if trial_fit < fitness[i]:
            pop[i] = trial
            fitness[i] = trial_fit
    return pop, fitness

def subgrad_polish(h, n_steps, lr_init=0.005):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_steps + 1):
        lr = lr_init / np.sqrt(t)
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        k_star = np.argmax(corr)
        grad = np.zeros(n)
        s = k_star - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] = dx * omh[:n - s]
                if n - s > 0:
                    grad[:n - s] -= dx * h[s:]
        else:
            r = -s
            if r < n:
                grad[:n - r] = dx * omh[r:]
                grad[r:] -= dx * h[:n - r]
        h = project_onto_constraint(h - lr * grad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_polish(h, rng, max_rounds=15):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.005, -0.005]
    for _ in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for idx in order:
            for d in deltas:
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + d, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def init_population(best_h, n, pop_size, rng, target_sum):
    pop = np.zeros((pop_size, n))
    fitness = np.full(pop_size, np.inf)
    pop[0] = best_h
    fitness[0] = compute_c5(best_h)
    for i in range(1, pop_size):
        strategy = i % 6
        if strategy == 0:
            cand = best_h + 0.1 * rng.standard_normal(n)
        elif strategy == 1:
            cand = best_h + 0.2 * (rng.random(n) - 0.5) * 2.0
        elif strategy == 2:
            bits = (rng.random(n) < 0.5).astype(float)
            cand = bits * rng.uniform(0.5, 1.5, n)
        elif strategy == 3:
            cand = best_h.copy()
            idx = rng.choice(n, size=max(1, n // 4), replace=False)
            cand[idx] = 1.0 - cand[idx]
        elif strategy == 4:
            alpha = rng.uniform(0.2, 0.8)
            cand = alpha * best_h + (1.0 - alpha) * (1.0 - best_h)
        else:
            cand = best_h[rng.permutation(n)]
        cand = project_onto_constraint(cand, target_sum)
        pop[i] = cand
        fitness[i] = compute_c5(cand)
    return pop, fitness
