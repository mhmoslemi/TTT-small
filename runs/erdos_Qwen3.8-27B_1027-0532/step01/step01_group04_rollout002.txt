import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_gs_target_sum = 0.0
_gs_n = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj_lse(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def proj(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = int(np.sum(mask))
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def _con_fun(h):
    return np.sum(h) - _gs_target_sum


def _con_jac(h):
    return np.ones(_gs_n)


def _de_obj(x, ts):
    h = proj(x, ts)

    return h_new
