So there are O(n) candidate points to check. This is feasible.

But this is O(n² * n) = O(n³) per sweep which is too slow for n=350.

Let me think of a faster approach. Instead of full coordinate descent, I'll use:
1. Subgradient descent with a smart step size
2. Projected gradient (L-BFGS style) with the subgradient
3. Simulated annealing with structured moves

Actually, let me take a step back and think about what the previous code did well and what it didn't:

The previous code used:
- SLSQP (which handles constraints well but struggles with non-smooth objectives)
- Softmax smoothing (to make the max differentiable)
- Random restarts

The issue is that SLSQP with 350 variables and a non-smooth objective is slow and can get stuck.

My plan:
1. Use a smooth approximation of the max (log-sum-exp) with adaptive temperature
2. Use L-BFGS-B (which handles bound constraints efficiently) for the smooth objective
3. Use a "boiling" schedule for the temperature (start hot, cool down)
4. Intersperse with random perturbations to escape local minima
5. At the end, do a final non-smooth optimization step

Additionally, I'll try a different structural approach:
- Start from analytically motivated initial guesses
- Use the Fourier structure to generate good starting points

Let me also think about the "dual" approach. The problem:
minimize max_k f_k(h)
s.t. 0 ≤ h ≤ 1, sum(h) = n/2

can be written as:
minimize t
s.t. f_k(h) ≤ t for all k
     0 ≤ h ≤ 1
     sum(h) = n/2

By LP duality (but this isn't an LP since f_k is quadratic), the Lagrangian is:
L(h, t, λ, μ, ν) = t + Σ_k λ_k (f_k(h) - t) + μ^T(-h) + ν^T(h-1) + α(sum(h) - n/2)

At optimality, Σ_k λ_k = 1 and the KKT conditions hold.

The key insight: at the optimum, there are typically only a FEW active constraints (shifts achieving the max). If we can identify them, we can solve a smaller problem.

Let me implement a "active set" approach:
1. Find the current active set K* = {k : f_k(h) ≈ max}
2. Minimize max_{k∈K*} f_k(h) (a smaller problem)
3. Check if any other constraint is violated
4. If yes, add it to K* and repeat

For step 2, minimizing max of a few quadratics can be done with SOCP or as a sequence of QPs.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def projection(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0)
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0)
            h[mask] = np.maximum(0.0, h[mask] - step)
    return np.clip(h, 0.0, 1.0)

def subgradient_c5(h, top_m=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_val = float(np.max(corr) * dx)
    top_idx = np.argsort(corr)[-top_m:]
    w_max = np.max(corr)
    active = np.zeros(len(corr))
    active[top_idx] = 1.0 / top_m
    grad = np.zeros(n)
    for k_i in top_idx:
        s = k_i - (n - 1)
        w = active[k_i]
        if s == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif s > 0:
            n_valid = n - s
            grad[:n_valid] += -w * h[s:n]
            grad[s:] += w * (1.0 - h[:n_valid])
        else:
            s_abs = -s
            n_valid = n + s
            grad[s_abs:] += w * (1.0 - h[:n_valid])
            grad[:n_valid] += -w * h[s_abs:]
    return c5_val, grad * dx

def lse_smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * beta
    m = np.max(scaled)
    logsum = m + np.log(np.sum(np.exp(scaled - m)))
    return logsum / beta * dx

def lse_smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * beta
    m = np.max(scaled)
    e = np.exp(scaled - m)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k_i in range(len(corr)):
        if w[k_i] < 1e-12:
            continue
        s = k_i - (n - 1)
        ww = w[k_i] * dx
        if s == 0:
            grad[:] += ww * (1.0 - 2.0 * h)
        elif s > 0:
            n_valid = n - s
            grad[:n_valid] += -ww * h[s:n]
            grad[s:] += ww * (1.0 - h[:n_valid])
        else:
            s_abs = -s
            n_valid = n + s
            grad[s_abs:] += ww * (1.0 - h[:n_valid])
            grad[:n_valid] += -ww * h[s_abs:]
    return grad

def make_lse_funcs(n, beta):
    def obj(x):
        return lse_smooth_obj(x, n, beta)
    def obj_grad(x):
        g = lse_smooth_grad(x, n, beta)
        s = (x - np.sum(x) / n)
        return g + s
    return obj, obj_grad

def annealing_lse(h, n, betas, maxiter_each, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    for beta in betas:
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break
        obj, obj_grad = make_lse_funcs(n, beta)
        try:
            res = minimize(
                obj, best_h, jac=obj_grad, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0,
                              'jac': lambda x, n=n: np.ones(n) / n}],
                options={'maxiter': maxiter_each, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            h_new = projection(res.x, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad_descent(h, n, n_iter, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    alpha = 0.01
    h_curr = h.copy()
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        c5_val, grad = subgradient_c5(h_curr, top_m=5)
        h_curr = h_curr - alpha * grad
        h_curr = projection(h_curr, n)
        if c5_val < best_c5:
            best_h, best_c5 = h_curr.copy(), c5_val
        alpha = max(0.001, alpha * 0.999)
    return best_h, best_c5

def coordinate_descent_c5(h, n, max_sweeps, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_curr = h.copy()
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            h_old = h_curr[j]
            if h_old <= 1e-10 or h_old >= 1.0 - 1e-10:
                continue
            target_sum = np.sum(h_curr)
            best_x = h_old
            best_val = np.inf
            candidates = np.concatenate([
                [h_old * 0.5, h_old * 0.75, h_old * 1.25, h_old * 1.5],
                [0.0, 1.0],
                np.array([h_old + d for d in [-0.1, -0.05, 0.05, 0.1]])
            ])
            candidates = candidates[(candidates >= 0.0) & (candidates <= 1.0)]
            for xc in candidates:
                adj = target_sum - h_old + xc
                if adj > n / 2.0 + 1e-9 or adj < n / 2.0 - 1e-9:
                    continue
                h_curr[j] = xc
                v = evaluate_c5(h_curr)
                if v < best_val:
                    best_val = v
                    best_x = xc
            if best_x != h_old:
                h_curr[j] = best_x
                improved = True
            else:
                h_curr[j] = h_old
        h_curr = projection(h_curr, n)
        c5_curr = evaluate_c5(h_curr)
        if c5_curr < best_c5:
            best_h, best_c5 = h_curr.copy(), c5_curr
        if not improved:
            break
    return best_h, best_c5

def simulated_annealing(h, n, time_limit, n_iter_max=5000):

    best_c5 = evaluate_c5(best_h)

        j = np.random.randint(n)

        h_curr[j] = new_val
        adj = h_curr - (np.sum(h_curr) - n / 2.0) / n

        new_c5 = evaluate_c5(h_curr)
        if new_c5 < curr_c5 or np.random.random() < np.exp(-(new_c5 - curr_c5) / max(temp, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_curr.copy(), new_c5

def pair_move_annealing(h, n, time_limit, n_iter_max=3000):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    h_curr = h.copy()
    curr_c5 = best_c5
    t_init = 0.05
    t_final = 0.001
    for it in range(n_iter_max):
        if time.time() - t_start > time_limit:
            break
        cool = it / max(n_iter_max, 1)
        temp = t_init * (t_final / t_init) ** cool
        i, j = np.random.choice(n, 2, replace=False)
        amount = np.random.uniform(0.02, 0.15)
        if np.random.random() < 0.5:
            h_curr[i] -= amount
            h_curr[j] += amount
        else:
            h_curr[i] += amount
            h_curr[j] -= amount
        h_curr = projection(h_curr, n)
        new_c5 = evaluate_c5(h_curr)
        if new_c5 < curr_c5 or np.random.random() < np.exp(-(new_c5 - curr_c5) / max(temp, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_curr.copy(), new_c5
        else:
            h_curr[i] += amount if h_curr[i] < np.clip(h_curr[j] + amount, 0, 1) else -amount
            h_curr[j] -= amount if h_curr[j] > np.clip(h_curr[i] - amount, 0, 1) else amount
            h_curr = projection(h_curr, n)
    return best_h, best_c5
