import numpy as np
from scipy.optimize import minimize
import time

_CURRENT_T = 1.0

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def lse_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    if _CURRENT_T < 1e-14:
        return dx * cmax
    scaled = (corr - cmax) / _CURRENT_T
    return dx * (cmax + _CURRENT_T * np.log(np.sum(np.exp(scaled))))

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

    _CURRENT_T = T
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_con}],
                      options={'maxiter': max_iter, 'ftol': 1e-15})

            x_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            return x_new, score_c5(x_new)

    return h.copy(), score_c5(h)
