```python
from scipy.optimize import minimize
import time

def project_feasible(x, target):
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0-h, mode='full') * dx).max())

def smooth_max_obj(h, tau, penalty, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    m = corr.max()
    shifted = (corr - m) / tau
    logsum = np.log(np.exp(shifted).sum())
    obj = m + tau * logsum
    viol = h.sum() - target_sum
    obj += penalty * viol * viol
    return obj

def smooth_max_grad(h, tau, penalty, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    m = corr.max()
    shifted = (corr - m) / tau
    w = np.exp(shifted)
    w /= w.sum()
    
    j = np.arange(n)
    grad = np.zeros(n)
    for ki, wi in enumerate(w):
        if wi < 1e-10:
            continue
        k = ki - (n-1)
        a_idx = j + k
        b_idx = j - k
        ma = (a_idx >= 0) & (a_idx < n)
        mb = (b_idx >= 0) & (b_idx < n)
        grad[ma] += wi * dx * (1.0 - h[a_idx[ma]])
        grad[mb] -= wi * dx * h[b_idx[mb]]
    
    viol = h.sum() - target_sum
    grad += 2 * penalty * viol * np.ones(n)
    return grad

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.arange(n_src, dtype=float) + 0.5
    xd = np.linspace(0.5, n_src - 0.5, n_dst)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

                        penalty_init, penalty_growth, tm, target_sum):

    n = len(h0)
    h = project_feasible(h0, target_sum)
    best = h.copy()

    t0 = time.time()

    penalty = penalty_init

            break

            return smooth_max_obj(x, tau, penalty, target_sum)
        
        def grad(x):
            return smooth_max_grad(x, tau, penalty, target_sum)

            best = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_feasible(x, target):
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0-h, mode='full') * dx).max())

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0-h, mode='full') * dx

def corr_grad(h, k):
    """Gradient of C_k = dx * sum_i h[i](1-h[i+k]) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    a_idx = j + k
    b_idx = j - k
    ma = (a_idx >= 0) & (a_idx < n)
    mb = (b_idx >= 0) & (b_idx < n)
    grad[ma] += dx * (1.0 - h[a_idx[ma]])
    grad[mb] -= dx * h[b_idx[mb]]
    return grad

def active_set_lp(h_star, corr_star, n_active, target_sum):
    """Solve the LP relaxation at h_star using the top n_active shifts."""
    n = len(h_star)
    # Sort by correlation value (descending)
    idx = np.argsort(corr_star)[::-1]
    active_k = idx[:n_active]
    
    # Build LP:
    # Variables: [h_0, ..., h_{n-1}, t] (n+1 variables)
    # Minimize: t (i.e., c = [0,...,0,1])
    # Subject to:
    #   t - sum_i grad_k[i] * h_i >= corr_k - sum_i grad_k[i] * h_star[i]  for each active k
    #   Rearranged: -t + sum_i grad_k[i] * h_i <= -(corr_k - sum_i grad_k[i] * h_star[i])
    #   0 <= h_i <= 1
    #   sum(h_i) = target_sum
    
    # Inequality constraints (A_ub @ x <= b_ub):
    # For each active k:
    #   -t + sum_i grad_k[i] * h_i <= -(corr_k - sum_i grad_k[i]*h_star[i])
    #   = sum_i grad_k[i]*h_star[i] - corr_k
    
    A_ub = np.zeros((n_active, n+1))
    b_ub = np.zeros(n_active)
    
    for ai, ki in enumerate(active_k):
        k = int(ki) - (n-1)  # actual shift
        g = corr_grad(h_star, k)
        rhs = g @ h_star - corr_star[ki]
        A_ub[ai, :n] = g
        A_ub[ai, n] = -1.0
        b_ub[ai] = rhs
    
    # Equality constraint: sum(h_i) = target_sum
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: 0 <= h_i <= 1, t unbounded below (but will be >= max corr)
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Objective: minimize t
    c = np.zeros(n+1)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h_star
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_feasible(x, target):
    x = np.clip(x.astype(float), 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())

def compute_corr_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_single_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    a_idx = j + k
    b_idx = j - k
    ma = (a_idx >= 0) & (a_idx < n)
    mb = (b_idx >= 0) & (b_idx < n)
    grad[ma] += dx * (1.0 - h[a_idx[ma]])
    grad[mb] -= dx * h[b_idx[mb]]
    return grad

def active_set_lp(h_star, corr_star, n_active, target_sum):
    n = len(h_star)
    idx = np.argsort(corr_star)[::-1]
    active_idx = idx[:min(n_active, len(corr_star))]
    n_act = len(active_idx)
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    for ai, ki in enumerate(active_idx):
        k = int(ki) - (n - 1)
        g = corr_single_grad(h_star, k)
        rhs = g @ h_star - corr_star[ki]
        A_ub[ai, :n] = g
        A_ub[ai, n] = -1.0
        b_ub[ai] = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, 1)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h_star.copy()

def active_set_optimize(h0, n_iters, target_sum, time_limit, k_start, k_end):
    n = len(h0)
    h = project_feasible(h0, target_sum)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        frac = it / max(1, n_iters - 1)
        k_cur = int(k_start + (k_end - k_start) * frac)
        k_cur = max(3, k_cur)
        corr = compute_corr_arr(h)
        h_new = active_set_lp(h, corr, k_cur, target_sum)
        h_new = project_feasible(h_new, target_sum)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        h = h_new
    return best, best_c5

def smooth_max_obj_grad(h, tau, penalty, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = (corr - m) / tau
    max_shift = shifted.max()
    exp_shifted = np.exp(shifted - max_shift)
    logsum = max_shift + np.log(exp_shifted.sum())
    obj = m + tau * logsum
    viol = h.sum() - target_sum
    obj += penalty * viol * viol
    w = exp_shifted / exp_shifted.sum()
    j = np.arange(n)
    grad = np.zeros(n)
    for ki, wi in enumerate(w):
        if wi < 1e-10:
            continue
        k = ki - (n - 1)
        a_idx = j + k
        b_idx = j - k
        ma = (a_idx >= 0) & (a_idx < n)
        mb = (b_idx >= 0) & (b_idx < n)
        grad[ma] += wi * dx * (1.0 - h[a_idx[ma]])
        grad[mb] -= wi * dx * h[b_idx[mb]]
    grad += 2.0 * penalty * viol
    return obj, grad

def smooth_max_optimize(h0, tau_start, tau_end, n_tau, maxiter_per_tau,
                        penalty_init, penalty_growth, time_limit, target_sum):
    n = len(h0)
    h = project_feasible(h0, target_sum)
    best = h.copy()
    best_c5 = eval_c5(h)
    t0 = time.time()
    taus = np.geomspace(tau_start, tau_end, n_tau)
    penalty = penalty_init
    for tau in taus:
        if time.time() - t0 > time_limit:
            break
        def obj_fn(x):
            o, _ = smooth_max_obj_grad(x, tau, penalty, target_sum)
            return o

        def grad_fn(x):
            _, g = smooth_max_obj_grad(x, tau, penalty, target_sum)
            return g

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter_per_tau,
                                                   'gtol': 1e-8, 'ftol': 1e-12})
            h = project_feasible(res.x, target_sum)
        except Exception:
            pass
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        penalty *= penalty_growth
    return best, best_c5

def resample_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(float).copy()
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    h_new = np.interp(xd, xs, h_src)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_dst / 2.0)

def selpolish(h0, tm, target_sum):
    n = len(h0)
    h = project_feasible(h0, target_sum)
    t0 = time.time()

    def f(x):
        if time.time() - t0 > tm:
            raise Exception("timeout")
        return float((np.correlate(x, 1.0 - x, 'full') * (2.0 / n)).max())

    def gfun(x):
        return x.sum() - target_sum

    def gjac(x):
        return np.ones(n)

    try:
        r = minimize(f, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': gfun, 'jac': gjac}],
                     options={'maxiter': 500, 'ftol': 1e-14})
        h1 = project_feasible(r.x, target_sum)
        if eval_c5(h1) < eval_c5(h):
            return h1
    except Exception:
        pass
    return h

def subgrad_opt(h0, tm, lr_init, beta, target_sum):
    n = len(h0)
    dx = 2.0 / n
    h = project_feasible(h0, target_sum)
    best = h.copy()
    best_c5 = eval_c5(h)
    mom = np.zeros(n)
    lr = lr_init
    t0 = time.time()
    j = np.arange(n)
    for it in range(50000):
        if time.time() - t0 > tm:
            break
        corr = np.correlate(h, 1.0 - h, 'full') * dx
        ki = np.argmax(corr)
        k = ki - (n - 1)
        a = j - k
        b = j + k
        grad = np.zeros(n)
        m1 = (a >= 0) & (a < n)
        m2 = (b >= 0) & (b < n)
        grad[m1] += 1.0 - h[a[m1]]
        grad[m2] -= h[b[m2]]
        grad *= dx
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - lr * mom, target_sum)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= 0.9995
        if lr < 1e-8:
            break
    return best

def make_binary_rand(n, rng):
    positions = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

def sa_binary(h0, n, tm, target_sum, rng):
    h = h0.copy()
    h = np.clip(h, 0.0, 1.0)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

    best_c5 = eval_c5(h)
    T = 0.01
    t0 = time.time()
    dx = 2.0 / n
    n_iter = 0
    while time.time() - t0 < tm:
        n_iter += 1
        if n_iter % 200 == 0:
            T *= 0.97
        i_one = ones[rng.randint(len(ones))]
        i_zero = zeros[rng.randint(len(zeros))]
        h[i_one] = 0.0
        h[i_zero] = 1.0
        c5_new = eval_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        else:
            h[i_one] = 1.0
            h[i_zero] = 0.0
    return best, best_c5
```
