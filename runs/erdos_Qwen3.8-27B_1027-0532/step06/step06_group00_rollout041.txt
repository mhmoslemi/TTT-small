def multi_shift_grad(corr, h, n, mid, idxs, n_top=5, power=8):
    gradients = np.zeros(n)
    top_shifts = np.argsort(corr)[-n_top:]
    weights = corr[top_shifts] ** power
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.ones(n_top) / n_top
    else:
        weights /= w_sum
    for s_idx, w in zip(top_shifts, weights):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        gradients[v1] += w * (1.0 - h[idxs[v1] - k])
        gradients[v2] -= w * h[idxs[v2] + k]
    gradients -= np.mean(gradients)
    return gradients

def fast_pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
                    n_cand_start=30, n_cand_end=100, max_no_improve=500,
                    n_top=5, grad_power=8):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        grad = multi_shift_grad(corr, h, n, mid, idxs, n_top=n_top, power=grad_power)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(13):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5

def subgrad_momentum(h_init, n, target, time_limit, lr_init=0.005, momentum=0.9,
                     max_iter=30000, decay=0.99995, n_top=5):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = multi_shift_grad(corr, h, n, mid, idxs, n_top=n_top, power=10)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new
        lr *= decay

    return best_h, best_c5

def simulated_annealing_pairs(h_init, n, target, time_limit, max_steps=80000,
                              T_init=0.05, T_min=0.001, cooling=0.9998):
    start = time.time()
    rng = np.random.RandomState(777)
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    shifts_arr = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T_init
    change = compute_change_matrix(h, n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break

        n_pair = int(rng.choice([1, 2, 3, 5]))
        n_pair = min(n_pair, n)
        i_list = rng.choice(n, size=n_pair, replace=False)
        j_list = rng.choice(n, size=n_pair, replace=False)

        h_try = h.copy()
        for ii in range(n_pair):
            i = int(i_list[ii])
            j = int(j_list[jj])
            if i == j:
                continue
            delta = float(rng.uniform(-0.15, 0.15))
            delta = max(delta, h[i] - 1.0, -h[j])
            delta = min(delta, h[i], 1.0 - h[j])
            if abs(delta) < 1e-12:
                continue
            h_try[i] -= delta
            h_try[j] += delta
            update_change_row(change, i, h_try, n, shifts_arr)
            update_change_row(change, j, h_try, n, shifts_arr)

        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        new_c5 = compute_c5(h_try)

        if new_c5 < cur_c5:
            h = h_try
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5 - 1e-13:
                best_c5 = new_c5
                best_h = h_try.copy()
            change = compute_change_matrix(h, n)
        else:
            accept_prob = np.exp(-(new_c5 - cur_c5) / max(T, 1e-10))
            if rng.rand() < accept_prob:
                h = h_try
                cur_c5 = new_c5
                change = compute_change_matrix(h, n)
            no_improve += 1

        T = max(T * cooling, T_min)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)

    if tleft() > 40:

    if tleft() > 200:
