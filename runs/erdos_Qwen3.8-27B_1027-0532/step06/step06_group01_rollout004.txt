    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        hp, sc = slsqp_optimize(best_h, n, target, [100, 500, 2000, 10000, 50000], 3000, 1e-16, min(100, tleft()))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 180:
        cd_time = min(250, tleft() * 0.3)
        hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=cd_time, n_cand=25, max_sweeps=100000, n_bisect=13)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 80:
        hp, sc = slsqp_optimize(best_h, n, target, [5000, 20000, 100000, 500000], 2000, 1e-16, min(60, tleft()))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 150:
        cd_time = min(180, tleft() * 0.35)
        hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=cd_time, n_cand=35, max_sweeps=100000, n_bisect=14)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hsg, c5sg = subgrad_multi(best_h, n, target, time_limit=min(40, tleft() * 0.4), n_shifts=5, base_lr=0.015, alpha_w=20.0)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 50:
        hp, sc = slsqp_optimize(best_h, n, target, [20000, 100000, 500000, 2000000], 1500, 1e-16, min(40, tleft()))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 350:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_optimize(h_interp, n_new, target_new, [100, 500, 2000, 10000, 50000], 2000, 1e-16, min(150, tleft() * 0.4))
        if tleft() > 150:
            hcd, c5cd = pair_cd_fast(hp, n_new, target_new, time_limit=min(150, tleft() * 0.4), n_cand=30, max_sweeps=50000, n_bisect=13)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(120, tleft() * 0.45)
        hcd, c5cd = pair_cd_fast(best_h, n_use, target_use, time_limit=cd_time, n_cand=25, max_sweeps=100000, n_bisect=13)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_optimize(best_h, n_use, target_use, [50000, 200000, 1000000], 1000, 1e-16, min(40, tleft()))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 60:
        hsg, c5sg = subgrad_multi(best_h, len(best_h), len(best_h) / 2.0, time_limit=min(35, tleft() * 0.5), n_shifts=7, base_lr=0.01, alpha_w=30.0)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_fast(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.6), n_cand=30, max_sweeps=50000, n_bisect=15)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_optimize(best_h, n_use, target_use, [200000, 1000000, 5000000], 500, 1e-16, min(20, tleft()))
        if sc < best_score:
            best_score = sc
            best_h = hp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
