import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _power_obj_func(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.mean(corr ** p) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=70, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_nesterov(h_init, n, target, time_limit, n_active=8, alpha_base=20.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    h_y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 0.9
    t_count = 1.0
    no_improve = 0

    for it in range(1000000):
        if time.time() - start > time_limit:
            break

        corr_y = np.correlate(h_y, 1.0 - h_y, mode='full')
        c5_y = float(np.max(corr_y)) * dx

        if c5_y < best_c5 - 1e-13:
            best_c5 = c5_y
            best_h = h_y.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                h = best_h.copy()
                h_y = best_h.copy()
                v = np.zeros(n)
                no_improve = 0
                continue

        c_max = np.max(corr_y)
        weights = np.exp(alpha_base * (corr_y - c_max))
        weights /= weights.sum()

        grad = np.zeros(n)
        active_shifts = np.argsort(corr_y)[-n_active:]
        for s_idx in active_shifts:
            w = weights[s_idx]
            if w < 1e-6:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_y[idxs[v1] - k])
            grad[v2] -= w * h_y[idxs[v2] + k]
        grad -= np.mean(grad)

        grad_norm = np.sqrt(np.sum(grad ** 2))
        if grad_norm < 1e-12:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_count ** 2)) / 2.0
        beta_new = (1.0 - t_new) / (t_count + 1.0)

        step = 2.0 / (1.0 + t_new)
        h_new = h - step * (grad / grad_norm + beta_new * v)
        h_new = project_feasible(h_new, target)
        v = h_new - h
        h = h_new
        t_count = t_new
        h_y = h + beta_new * (h - h_y)

    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, n_shifts=5, alpha=15.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_size = 0.05
    no_improve = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            step_size *= 0.9
            if step_size < 1e-5:
                break
            no_improve = 0

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step_size = min(step_size * 1.05, 0.1)
        else:
            no_improve += 1

        c_max = np.max(corr)
        weights = np.exp(alpha * (corr - c_max))
        w_sum = weights.sum()
        if w_sum < 1e-10:
            break
        weights /= w_sum

        grad = np.zeros(n)
        top_idx = np.argsort(corr)[-n_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-5:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        gnorm = np.sqrt(np.sum(grad ** 2))
        if gnorm < 1e-12:
            break

        h_try = h - step_size * grad / gnorm
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)

        if c5_try < c5_val:
            h = h_try
        else:
            step_size *= 0.95

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, rng, mode=0):
    h_p = h.copy()
    if mode == 0:
        freq = rng.randint(1, 5)
        amp = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        h_p += amp * np.sin(freq * x + phase)
    elif mode == 1:
        block_len = rng.randint(n // 10, n // 4)
        start = rng.randint(0, n - block_len)
        shift_amt = rng.uniform(-0.15, 0.15)
        h_p[start:start + block_len] += shift_amt
    elif mode == 2:
        h_p = np.roll(h_p, rng.randint(-n // 8, n // 8))
    elif mode == 3:
        h_p = h[::-1].copy()
    elif mode == 4:
        sigma = rng.uniform(1.0, 3.0)
        from scipy.ndimage import gaussian_filter1d
        h_p = gaussian_filter1d(h_p, sigma=sigma, mode='wrap')
    elif mode == 5:
        n_pert = rng.randint(n // 5, n // 2)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_p[idxs] += rng.randn(n_pert) * 0.1
    else:
        h_p += rng.randn(n) * 0.05
    return project_feasible(h_p, target)
