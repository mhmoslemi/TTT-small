def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def adam_projected(h_init, n, target, time_limit, alphas, top_K=200,
                   beta1=0.9, beta2=0.999, lr=0.1, eps=1e-8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    for alpha in alphas:
        while time.time() - start < time_limit:
            g, overlaps = compute_smooth_gradient(h, dx, alpha, top_K)
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            # Check convergence
            if c5_new > best_c5 + 0.01:
                h = best_h.copy()
                break
    
    return best_h, best_c5

**Alternative: Use the "duality gap" to guide optimization**

The dual problem is:
max_{v ∈ simplex, ||v||_1 ≤ 1} min_{h: constraints} sum_k v_k * O(k)

= min_h dx * [S * sum_i h[i] - sum_{i,k} v_k * h[i] * h[i+k]]

This is a quadratic function of h. We can write it as:
= min_h dx * [S * 1^T h - h^T A v h]

where A is some matrix depending on v. This is a QP! We can solve it efficiently with scipy or cvxpy.

But the outer problem (choosing v) is over the simplex, which is high-dimensional (799 dimensions).

This dual approach might be useful for getting a lower bound, but solving the full saddle point problem is complex.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, dx, alpha, top_K=200):
    n = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    n_half = n - 1
    if top_K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(overlaps)[::-1][:top_K]
    top_vals = overlaps[top_idx]
    c_max = top_vals[0]
    shifted = alpha * (top_vals - c_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    g = np.zeros(n)
    for i in range(len(w)):
        w_k = w[i]
        if w_k < 1e-12:
            continue
        k = top_idx[i] - n_half
        if k == 0:
            g += w_k * (1.0 - 2.0 * h) * dx
        elif 0 < k < n:
            g[:n - k] += w_k * (1.0 - h[k:]) * dx
            g[k:] += w_k * (-h[:n - k]) * dx
        elif -n < k < 0:
            m = -k
            g[m:] += w_k * (1.0 - h[:n - m]) * dx
            g[:n - m] += w_k * (-h[m:]) * dx
    return g

def nesterov_projected_grad(h_init, n, target, time_limit, alpha=100,
                             top_K=200, lr=0.3, max_iter=200000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Nesterov step
        beta = float(t) / (t + 2.0)
        y = h + beta * (h - h_prev)
        y = project_feasible(y, target)
        
        g = smooth_max_grad(y, dx, alpha, top_K)
        h_new = y - lr * g
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        
        h_prev = h
        h = h_new
        
        # Adaptive learning rate
        if stale > 1000:
            lr *= 0.5
            stale = 0
            if lr < 1e-6:
                break
        elif stale == 0 and t > 100:
            # Try to increase lr slightly
            lr *= 1.05
    
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    
    def obj(x):
        overlaps = np.correlate(x, 1.0 - x, mode='full') * dx
        c_max = np.max(overlaps)
        shifted = np.clip(alpha * (overlaps - c_max), -50.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        return smooth_max_grad(x, dx, alpha, top_K=400)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h_start

def alternating_lp(h_init, n, target, time_limit, n_top=300,
                   damping=0.85, reg=1e-4, max_iter=3000):
    start = time.time()

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 100 and stuck > 200:
            break
        
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        # Build LP: minimize t + reg * ||h_var - h||^2
        # s.t. sum(h_var) = target, 0 <= h_var <= 1
        #      for each lag k: sum_i h_var[i] * (1-h[i+k]) * dx <= t
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
                constraints.append(coeff @ h_var <= t_var)
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = (1.0 - h[k:]) * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n - m]) * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5
