def compute_grad_matrix(h, n):
    """G[i, k_idx] = ∂c_k/∂h[i]"""
    n_corr = 2*n - 1
    mid = n - 1
    G = np.zeros((n, n_corr))
    
    # For each lag k, ∂c_k/∂h[i] = [1-h[i+k] if i+k valid] - [h[i-k] if i-k valid]
    # This is: for each i, G[i, :] is determined by h shifted
    
    # ∂c_k/∂h[i] where k = k_idx - mid:
    # Term 1: 1 - h[i+k] if 0 <= i+k < n, i.e., i in [-k, n-1-k]
    # Term 2: -h[i-k] if 0 <= i-k < n, i.e., i in [k, n-1+k]
    
    # Let's think of it differently:
    # G[i, k_idx] = (1 - h[i + (k_idx - mid)]) * 1[0 <= i + k_idx - mid < n]
    #              - h[i - (k_idx - mid)] * 1[0 <= i - k_idx + mid < n]
    
    # For term 1: i + k_idx - mid is in [0, n-1]
    #   => i is in [mid - k_idx, mid - k_idx + n - 1] ∩ [0, n-1]
    #   Let k = k_idx - mid, then i + k in [0, n-1], i in [-k, n-1-k]
    #   G[i, k_idx] += 1 - h[i+k] for valid i
    
    # For term 2: i - k_idx + mid is in [0, n-1]
    #   => i - k in [0, n-1] where k = k_idx - mid
    #   i in [k, n-1+k] ∩ [0, n-1]
    #   G[i, k_idx] -= h[i-k] for valid i
    
    # Better approach: use the fact that this is a convolution-like operation
    # G[i, k_idx] = (1 - h[i+k]) * valid - h[i-k] * valid
    
    # For fixed i: G[i, :] as a function of k
    # G[i, mid+k] = (1-h[i+k]) * 1[0<=i+k<n] - h[i-k] * 1[0<=i-k<n]
    
    # This is: for k from -(n-1) to (n-1):
    #   if 0 <= i+k < n: G[i, mid+k] += 1-h[i+k]
    #   if 0 <= i-k < n: G[i, mid+k] -= h[i-k]
    
    # Vectorized: for each i, the valid k range for term 1 is [-(i), n-1-i]
    #              and for term 2 is [i-n+1, i]
    
    for i in range(n):
        # Term 1: k in [-i, n-1-i]
        k_min1 = max(-(n-1), -i)
        k_max1 = min(n-1, n-1-i)
        if k_min1 <= k_max1:
            k_range = np.arange(k_min1, k_max1+1)
            h_vals = h[i + k_range]
            G[i, k_range + mid] += 1.0 - h_vals
        
        # Term 2: k in [i-n+1, i]
        k_min2 = max(-(n-1), i-n+1)
        k_max2 = min(n-1, i)
        if k_min2 <= k_max2:
            k_range = np.arange(k_min2, k_max2+1)
            h_vals = h[i - k_range]
            G[i, k_range + mid] -= h_vals
    
    return G

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_grad_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    G = np.zeros((n, n_corr))
    for i in range(n):
        k_min1 = max(0, -i)
        k_max1 = min(n - 1, n - 1 - i)
        if k_min1 <= k_max1:
            kr = np.arange(k_min1, k_max1 + 1)
            G[i, mid + kr] += 1.0 - h[i + kr]
        k_min2 = max(0, i - (n - 1))
        k_max2 = min(n - 1, i)
        if k_min2 <= k_max2:
            kr = np.arange(k_min2, k_max2 + 1)
            G[i, mid - kr] -= h[i - kr]
    return G


def find_optimal_delta(corr, d, delta_max, n_bisect=12):
    if delta_max < 1e-14:
        return 0.0
    new_max_try = np.max(corr + delta_max * d)
    if new_max_try >= np.max(corr) - 1e-14:
        a = 0.0
        b = delta_max
        for _ in range(n_bisect):
            mid_d = 0.5 * (a + b)
            vals = corr + mid_d * d
            f_mid = np.max(vals)
            active = vals >= f_mid - 1e-14
            slope = np.max(d[active]) if np.any(active) else 0.0
            if slope > 1e-13:
                b = mid_d
            else:
                a = mid_d
        return 0.5 * (a + b)
    return delta_max


def vectorized_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=50, n_bisect=12):
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break

        G = compute_grad_matrix(h, n)

        grad = G.mean(axis=1)
        grad -= np.mean(grad)

        tol = 0.02
        c_max_raw = np.max(corr)
        active_mask = corr >= c_max_raw * (1.0 - tol / (c_max_raw * dx + 1e-30))
        if np.sum(active_mask) > 0:
            grad_active = G[:, active_mask].mean(axis=1)
            grad_active -= np.mean(grad_active)
            combined = 0.5 * grad + 0.5 * grad_active
        else:
            combined = grad

        sorted_idx = np.argsort(combined)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

                d_max = min(h[i], 1.0 - h[j])

                delta_star = find_optimal_delta(corr, d, d_max, n_bisect)
                if delta_star < 1e-12:
                    continue
                approx = np.max(corr + delta_star * d) * dx

                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)
        else:
            if no_improve > 40:
                break

    return best_h, best_c5

def structured_perturb(h, n, target, rng, n_modes=5, scale=0.05):

        h_new += amp * np.sin(np.pi * m * x)
        h_new += amp * np.cos(np.pi * m * x + phase)
    return project_feasible(h_new, target)

            best_score = sc
            best_h = h_pol

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
