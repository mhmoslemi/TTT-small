```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_G(h, n):
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    for k in range(-(n - 1), n):
        s = n - 1 + k
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            G[s, lo:hi] += 1.0 - h[lo + k:hi + k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            G[s, lo2:hi2] -= h[lo2 - k:hi2 - k]
    return G


def lp_step(h, n, target, active_lags=None):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_score = float(np.max(corr)) * dx

    G = compute_G(h, n)

    if active_lags is not None:
        idxs = active_lags
        G_sub = G[idxs, :]
        corr_sub = corr[idxs]
    else:
        G_sub = G
        corr_sub = corr

    n_constraints = len(corr_sub)
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = G_sub @ h - corr_sub

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds_lp = [(0, 1)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs',
                  options={'time_limit': 10})

    if not res.success:
        return None, current_score

    h_lp = res.x[:n]
    direction = h_lp - h

    if np.max(np.abs(direction)) < 1e-14:
        return h.copy(), current_score

    alpha = 1.0
    for _ in range(25):
        h_try = project_feasible(h + alpha * direction, target)
        score_try = compute_c5(h_try)
        if score_try < current_score - 1e-13:
            return h_try, score_try
        alpha *= 0.5
        if alpha < 1e-8:
            break

    return None, current_score


def lp_method(h0, n, target, time_limit, max_iters=5000, use_active=False, n_active=30):
    start = time.time()
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    stale = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        if use_active:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            threshold = c_max - 0.05 * (c_max - np.min(corr))
            active = np.where(corr >= threshold)[0]
            if len(active) > n_active:
                top = np.argsort(-corr, axis=0)[:n_active]
                active = top
            h_new, sc = lp_step(h, n, target, active_lags=active)
        else:
            h_new, sc = lp_step(h, n, target)

        if h_new is not None:
            h = h_new
            if sc < best_score - 1e-13:
                best_score = sc
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
        else:
            stale += 1

        if stale > 200:
            break

    return best_h, best_score


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    alpha_scaled = alpha / (2.0 / len(h))
    shifted = alpha_scaled * corr
    shifted -= np.max(shifted)
    return float(np.max(shifted) / alpha_scaled + np.log(np.sum(np.exp(shifted))) / alpha_scaled * (2.0 / len(h)))


def slsqp_cons_fun(h):
    return np.sum(h) - _global_target


def slsqp_cons_jac(h):
    return np.ones(_global_n)


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur, compute_c5(h_cur)

                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5


def subgradient_method(h0, n, target, time_limit, max_iters=100000):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h0, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    step_base = 0.01
    gamma = 0.9995
    gamma_momentum = 0.9
    momentum = np.zeros(n)
    step = step_base
    it_count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        it_count += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-13:
            best_score = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        k = k_star - mid
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        grad_normed = grad / grad_norm
        momentum = gamma_momentum * momentum + (1 - gamma_momentum) * grad_normed
        direction = momentum / (np.linalg.norm(momentum) + 1e-15)

        h_try = project_feasible(h - step * direction, target)
        score_try = compute_c5(h_try)

        if score_try < c5_val - 1e-14:
            h = h_try
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.9
            if step < 1e-6:
                break

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_lp(h_base, n, target, time_limit, n_starts, seed_val):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_base.copy()
    best_score = compute_c5(h_base)
    time_per_start = time_limit / n_starts

    for s in range(n_starts):
        if time.time() - start > time_limit - 10:
            break
        noise_scale = 0.05 + 0.1 * s / max(1, n_starts - 1)
        h_pert = project_feasible(h_base + rng.randn(n) * noise_scale, target)
        sc_pert = compute_c5(h_pert)

        if sc_pert < best_score:
            best_h = h_pert
            best_score = sc_pert

        t_sub = min(time_per_start * 0.6, time.time() - start + time_per_start * 0.6 - (time.time() - start))
        h_lp, sc_lp = lp_method(h_pert, n, target, time_limit=0.7 * (time_limit / n_starts), max_iters=2000)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

        if time.time() - start > time_limit - 20:
            break

        h_cd, sc_cd = pair_cd(h_lp if sc_lp < sc_pert else h_pert, n, target,
                              time_limit=0.2 * (time_limit / n_starts), max_sweeps=1000, n_cand=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol, sc_pol = slsqp_polish(best_h, n, target, [2000, 8000], 800)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 250:
        t_lp = min(350, tleft() * 0.35)
        h_lp, sc_lp = lp_method(best_h, n, target, time_limit=t_lp, max_iters=8000, use_active=False)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    if tleft() > 80:
        h_cd, sc_cd = pair_cd(best_h, n, target, time_limit=min(120, tleft() * 0.25), max_sweeps=4000, n_cand=45)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol, sc_pol = slsqp_polish(best_h, n, target, [20000, 80000], 1200)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 250:
        t_lp = min(300, tleft() * 0.3)
        h_lp, sc_lp = lp_method(best_h, n, target, time_limit=t_lp, max_iters=8000, use_active=True, n_active=40)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    if tleft() > 100:
        h_cd, sc_cd = pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.3), max_sweeps=3000, n_cand=50)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 200:
        t_ms = min(250, tleft() * 0.3)
        h_ms, sc_ms = multi_start_lp(best_h, n, target, time_limit=t_ms, n_starts=4, seed_val=seed + 1)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    if tleft() > 100:
        t_lp = min(120, tleft() * 0.3)
        h_lp, sc_lp = lp_method(best_h, n, target, time_limit=t_lp, max_iters=5000, use_active=False)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    if tleft() > 80:
        h_sg, sc_sg = subgradient_method(best_h, n, target, time_limit=min(80, tleft() * 0.3), max_iters=50000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 80:
        h_cd, sc_cd = pair_cd(best_h, n, target, time_limit=min(80, tleft() * 0.3), max_sweeps=3000, n_cand=55)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc_pol = slsqp_polish(best_h, n, target, [50000, 200000], 1500)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 40:
        h_cd, sc_cd = pair_cd(best_h, n, target, time_limit=min(40, tleft() * 0.5), max_sweeps=2000, n_cand=50)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    best_score = compute_c5(best_h)

    return best_h, best_score, n_use
```
