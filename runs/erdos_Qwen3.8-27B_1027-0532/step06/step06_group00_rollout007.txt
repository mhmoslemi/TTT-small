            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand_start=10,
            n_cand_end=50, max_no_improve=200, bisection_iters=10):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_active_cd(h_init, n, target, time_limit, max_sweeps=3000,
                    n_cand=30, n_active=3, bisection_iters=10):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-n_active:]
        max_corr = np.max(corr)
        tol = max_corr - (max_corr * 0.02 + 1e-6)
        active_mask = corr >= tol
        active_shifts = np.where(active_mask)[0]
        if len(active_shifts) == 0:
            active_shifts = top_idx

        grad = np.zeros(n)
        for s in active_shifts:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]
        grad /= len(active_shifts)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, max_steps=20000, base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 80:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [50, 200, 1000, 5000], 1000)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Upscale to n=800 and do CD
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h_cd, c5_cd = fast_cd(h1, n1, target1,
                                  time_limit=min(250, tleft() * 0.5),
                                  max_sweeps=4000, n_cand_start=10,
                                  n_cand_end=50, max_no_improve=300,
                                  bisection_iters=10)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    # Phase 3: Multi-active shift CD
    if tleft() > 150:
        h_ma, c5_ma = multi_active_cd(best_h, n0, target0,
                                      time_limit=min(120, tleft() * 0.5),
                                      max_sweeps=3000, n_cand=35,
                                      n_active=3, bisection_iters=10)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [2000, 10000, 30000], 1200)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: More CD at higher resolution
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 180:
            h_cd2, c5_cd2 = fast_cd(h2, n2, target2,
                                    time_limit=min(160, tleft() * 0.55),
                                    max_sweeps=3000, n_cand_start=10,
                                    n_cand_end=40, max_no_improve=200,
                                    bisection_iters=8)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2

    # Phase 6: Stochastic refinement
    if tleft() > 80:
        h_st, c5_st = stochastic_refine(best_h, n0, target0,
                                        time_limit=min(40, tleft() * 0.4),
                                        max_steps=20000, base_scale=0.02)
        if c5_st < best_score:
            best_score = c5_st
            best_h = h_st

    # Phase 7: Multi-start from perturbations
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc = slsqp_optimize(h_pert, n0, target0,
                                       [100, 1000, 5000], 500)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Final CD
    if tleft() > 80:
        h_fcd, c5_fcd = fast_cd(best_h, n0, target0,
                                time_limit=min(60, tleft() * 0.5),
                                max_sweeps=3000, n_cand_start=20,
                                n_cand_end=60, max_no_improve=150,
                                bisection_iters=10)
        if c5_fcd < best_score:
            best_score = c5_fcd
            best_h = h_fcd

    # Phase 9: Final SLSQP
    if tleft() > 40:
        h_fp, sc = slsqp_optimize(best_h, n0, target0,
                                  [10000, 50000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_fp

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
