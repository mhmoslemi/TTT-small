import numpy as np

from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_linearize_step(h, n, target, n_active=20):
    """One LP linearization step."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    
    # Identify active lags (top n_active)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:n_active]
    
    # Build LP:
    # Variables: h[0..n-1], T (total n+1 variables)
    # Minimize T
    # Subject to:
    #   sum(h) = target
    #   0 <= h[i] <= 1
    #   For each active lag s: corr[s] + sum_i grad_s[i]*(h[i]-h[i]) <= T
    #   => sum_i grad_s[i]*h[i] <= T - corr[s]
    
    # Compute gradients for active lags
    # For shift k = s - mid:
    # grad_s[i] = (1-h[i+k])*1[0<=i+k<n] - h[i-k]*1[0<=i-k<n]
    
    # Build constraint matrix A for linprog:
    # A_ub @ x ≤ b_ub
    # A_eq @ x = b_eq
    
    # A_ub: n_active × (n+1), b_ub: n_active
    # For each active lag s (shift k = s - mid):
    #   sum_i grad_s[i]*h[i] - T <= -corr[s]
    #   Row: [grad_s[0], ..., grad_s[n-1], -1]
    #   b: -corr[s]
    
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    
    for idx, s in enumerate(active):
        k = int(s) - mid
        for i in range(n):
            g = 0.0
            if 0 <= i + k < n:
                g += (1.0 - h[i + k])
            if 0 <= i - k < n:
                g -= h[i - k]
            A_ub[idx, i] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -corr[s]
    
    # A_eq: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # c: minimize T
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h[i] <= 1, T unbounded below (but effectively bounded)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    return result.x[:n], result

import numpy as np
from scipy.optimize import minimize, linprog
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_lp_for_active(h, corr, active_indices, mid, n, target):
    n_act = len(active_indices)
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    i_arr = np.arange(n)
    for idx, s in enumerate(active_indices):
        k = int(s) - mid
        valid_p = (i_arr + k >= 0) & (i_arr + k < n)
        valid_m = (i_arr - k >= 0) & (i_arr - k < n)
        grad_s = np.zeros(n)
        grad_s[valid_p] = (1.0 - h[i_arr[valid_p] + k])
        grad_s[valid_m] -= h[i_arr[valid_m] - k]
        A_ub[idx, :n] = grad_s
        A_ub[idx, n] = -1.0
        b_ub[idx] = -corr[s]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    return A_ub, b_ub, A_eq, b_eq, c_obj, bounds


def lp_linearize_iterate(h, n, target, time_limit, max_iters=500, tol_factor=0.005):
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    dx = 2.0 / n
    mid = n - 1
    i_arr = np.arange(n)

    for iteration in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        c_max_dx = c_max * dx

        if c_max_dx < best_c5 - 1e-14:
            best_c5 = c_max_dx
            best_h = h_cur.copy()

        tol = tol_factor / dx
        threshold = c_max - tol
        active = np.where(corr >= threshold)[0]
        n_act = len(active)

        if n_act == 0:
            break
        if n_act > 80:
            sorted_idx = np.argsort(corr)[::-1]
            active = sorted_idx[:80]
            n_act = 80

        A_ub, b_ub, A_eq, b_eq, c_obj, bounds = build_lp_for_active(
            h_cur, corr, active, mid, n, target)

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if result.status != 0:
            break

        h_lp = result.x[:n]
        improved = False

            h_try = (1.0 - t_frac) * h_cur + t_frac * h_lp

                improved = True
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_cur.copy()
                break

        if not improved:
            break

        if abs(best_c5 - compute_c5(h_cur)) < 1e-13:
            break

    return best_h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 60:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                g = g_i.copy()
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 20:
                break

    return best_h, best_c5

def subgradient_step(h, n, target, n_sub=100, step_decay=0.9999, init_step=0.05):
    """Subgradient descent on max of I(k)."""
    dx = 2.0 / n
    mid = n - 1
    i_arr = np.arange(n)
    step = init_step
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    for s in range(n_sub):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        c_max_dx = c_max * dx
        if c_max_dx < best_c5 - 1e-14:
            best_c5 = c_max_dx
            best_h = h_cur.copy()

        tol = 0.02 / dx
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            vp = (i_arr + k >= 0) & (i_arr + k < n)
            vm = (i_arr - k >= 0) & (i_arr - k < n)
            g = np.zeros(n)
            g[vp] = (1.0 - h_cur[i_arr[vp] + k])
            g[vm] -= h_cur[i_arr[vm] - k]
            grad += g * dx
        grad /= len(active)
        grad -= np.mean(grad)

        h_try = project_feasible(h_cur - step * grad, target)
        h_cur = h_try
        step *= step_decay

    return best_h, best_c5

                                   max_sweeps=3000, n_cand=60, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final LP polish
    if tleft() > 20:
        h_lp = lp_linearize_iterate(best_h, n, target, time_limit=min(15, tleft()*0.6),
                                    max_iters=100, tol_factor=0.001)
        sc = compute_c5(h_lp)
        if sc < best_score:
            best_score = sc
            best_h = h_lp

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
