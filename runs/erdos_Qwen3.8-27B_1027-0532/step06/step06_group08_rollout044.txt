        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-10:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    if mask.any():
        result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    if mask.any():
        result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    if mask.any():
        result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    if mask.any():
        result[mask] -= h[idx[mask]]
    return result

def greedy_2swap(h, max_iters, time_limit, start_time, rng, topk=3):
    n = len(h)
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    positions = np.arange(n)
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        max_val = float(np.max(corr))
        topk_idx = np.argsort(corr)[-topk:]
        k_stars = [int(ki) - (n - 1) for ki in topk_idx]
        k_stars = [k for k in k_stars if k != 0]
        if not k_stars:
            break
        k_star = k_stars[-1]

        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        ones_sorted = ones_arr[np.argsort(A[ones_arr])]
        zeros_sorted = zeros_arr[np.argsort(-A[zeros_arr])]

        found = False
        C1 = min(40, len(ones_sorted))
        C2 = min(40, len(zeros_sorted))

        for ci in range(C1):
            for cj in range(C2):
                i = int(ones_sorted[ci])
                j = int(zeros_sorted[cj])
                if A[i] >= A[j]:
                    continue
                delta = compute_delta(h, i, j)
                new_max = float(np.max(corr + delta))
                if new_max < max_val - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    ones.remove(i)
                    zeros.remove(j)
                    zeros.append(i)
                    ones.append(j)
                    stall = 0
                    new_c5 = float(np.max(corr)) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            C3 = min(15, len(ones_sorted))
            C4 = min(15, len(zeros_sorted))
            cand_i = ones_sorted[:C3]
            cand_j = zeros_sorted[:C4]

            deltas = np.zeros((C3, C4, 2 * n - 1))
            valid_pairs = np.zeros((C3, C4), dtype=bool)
            for a in range(C3):
                for b in range(C4):
                    i = int(cand_i[a])
                    j = int(cand_j[b])
                    if A[i] < A[j]:
                        deltas[a, b] = compute_delta(h, i, j)
                        valid_pairs[a, b] = True

            if valid_pairs.any():
                best_pair_max = max_val
                best_a, best_b = -1, -1
                for a in range(C3):
                    for b in range(C4):
                        if not valid_pairs[a, b]:
                            continue
                        for a2 in range(a, C3):
                            for b2 in range(C4):
                                if not valid_pairs[a2, b2]:
                                    continue
                                i1, j1 = int(cand_i[a]), int(cand_j[b])
                                i2, j2 = int(cand_i[a2]), int(cand_j[b2])
                                if i1 == i2 or j1 == j2:
                                    continue
                                dm = float(np.max(corr + deltas[a, b] + deltas[a2, b2]))
                                if dm < best_pair_max - 1e-12:
                                    best_pair_max = dm
                                    best_a, best_b = (a, b, a2, b2)
                                    best_i1, best_j1 = i1, j1
                                    best_i2, best_j2 = i2, j2

                if best_a >= 0:
                    h[best_i1] = 0.0
                    h[best_j1] = 1.0
                    h[best_i2] = 0.0
                    h[best_j2] = 1.0
                    corr += compute_delta(h, best_i1, best_j1) + compute_delta(
                        np.where((h == 1.0) & (np.arange(n) == best_i2), 1.0, np.where(h == 1.0, 1.0, 0.0)),
                        best_i2, best_j2)
                    # Recompute corr to be safe
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    ones = list(np.where(h == 1.0)[0])
                    zeros = list(np.where(h == 0.0)[0])
                    stall = 0
                    new_c5 = float(np.max(corr)) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True

        if not found:
            stall += 1

    return best_h, best_c5

def sa_binary(h, time_limit, start_time, rng, n_perturb=3, t_init=0.03, t_final=0.00002):
    n = len(h)
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    positions = np.arange(n)

    for perturb_round in range(n_perturb):
        if time.time() - start_time > time_limit:
            break

        cur_c5 = float(np.max(corr)) * dx
        n_total = int(max(1, (time_limit - (time.time() - start_time)) * 3000))
        t_cur = t_init

        for it in range(n_total):
            if time.time() - start_time > time_limit:
                break

            ratio = it / max(n_total - 1, 1)
            t = t_init * (t_final / t_init) ** ratio

            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break

            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)

            if k_star != 0 and rng.random() < 0.7:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_pool = ones[A[ones] <= 1.0]
                if len(i_pool) == 0:
                    i_pool = ones
                j_pool = zeros[A[zeros] >= 1.0]
                if len(j_pool) == 0:
                    j_pool = zeros
            else:
                i_pool = ones
                j_pool = zeros

            i = int(i_pool[rng.randint(len(i_pool))])
            j = int(j_pool[rng.randint(len(j_pool))])

            delta = compute_delta(h, i, j)
            new_max = float(np.max(corr + delta))
            new_c5 = new_max * dx
            dc = new_c5 - cur_c5

            if dc < 0 or (t > 1e-15 and rng.random() < np.exp(-dc / t)):
                h[i] = 0.0
                h[j] = 1.0
                corr += delta
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()

        # Perturb for next round
        if perturb_round < n_perturb - 1:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
            n_swaps = rng.randint(3, 8)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            for _ in range(n_swaps):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def continuous_optimize(h_start, time_limit, start_time, rng):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)

    def smooth_obj(x, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(c))
        shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    alphas = [30, 100, 500, 2000]
    h_cur = h.copy()
    cur_score = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full'))) * dx

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break

        # Projected gradient descent with smoothing
        n_iters = int(min(500000, max(5000, (time_limit - (time.time() - start_time)) * 5000)))
        lr = 0.02
        h_pg = h_cur.copy()
        stall = 0

        for it in range(n_iters):
            if time.time() - start_time > time_limit:
                break
            if stall > 5000:
                break

            ratio = it / max(n_iters - 1, 1)
            lr_cur = 0.02 * (0.0005 / 0.02) ** ratio

            c = np.correlate(h_pg, 1.0 - h_pg, mode='full')
            c_max = float(np.max(c))
            shifted_w = alpha * (c - c_max)
            shifted_w = np.clip(shifted_w, -400.0, 0.0)
            exp_w = np.exp(shifted_w)
            W = exp_w / exp_w.sum()

            grad = np.zeros(n)
            for k in range(2 * n - 1):
                wk = W[k]
                if wk < 1e-12:
                    continue
                s = shifts[k]
                if s >= 0:
                    grad[:n - s] -= wk * (1.0 - h_pg[s:])
                    grad[s:] -= wk * h_pg[:n - s]
                else:
                    abs_s = -s
                    grad[abs_s:] -= wk * (1.0 - h_pg[:n - abs_s])
                    grad[:n - abs_s] -= wk * h_pg[abs_s:]

            h_pg = h_pg - lr_cur * grad
            h_pg = project_box_sum(h_pg, target)

            c = np.correlate(h_pg, 1.0 - h_pg, mode='full')
            sc = float(np.max(c)) * dx
            if sc < cur_score - 1e-13:
                cur_score = sc
                h_cur = h_pg.copy()
                stall = 0
            else:
                stall += 1

        h_cur = project_box_sum(h_cur, target)
        cur_score = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full'))) * dx
        if cur_score < best_c5:
            best_c5 = cur_score
            best_h = h_cur.copy()

    # SLSQP polish
    if time.time() - start_time < time_limit - 10:
        bounds = [(0.0, 1.0)] * n
        for alpha in [200, 1000]:
            if time.time() - start_time > time_limit - 10:
                break
            try:
                def obj_fn(x):
                    return smooth_obj(x, alpha)
                def con_fun(x):
                    return np.sum(x) - target
                def con_jac(x):
                    return np.ones(n)
                res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                               options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_box_sum(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass

    if cur_score < best_c5:
        best_c5 = cur_score
        best_h = h_cur.copy()

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h_bin = project_binary(h_init, target_init)
    best_score_bin = compute_c5(best_h_bin)
    best_h_cont = project_box_sum(h_init, float(n_init) / 2.0)
    best_score_cont = compute_c5(best_h_cont)

    # Phase 1: Greedy 1-swap + 2-swap
    allowed = min(80.0, tleft() * 0.12)
    if allowed > 15:
        h_g, c5_g = greedy_2swap(best_h_bin, max_iters=100000, time_limit=allowed,
                                  start_time=start_time, rng=rng, topk=3)
        if c5_g < best_score_bin:
            best_score_bin = c5_g
            best_h_bin = h_g.copy()

    # Phase 2: Intensive SA
    allowed = min(400.0, tleft() * 0.35)
    if allowed > 60:
        h_sa, c5_sa = sa_binary(best_h_bin, time_limit=allowed, start_time=start_time,
                                 rng=rng, n_perturb=4, t_init=0.025, t_final=0.00001)
        if c5_sa < best_score_bin:
            best_score_bin = c5_sa
            best_h_bin = h_sa.copy()

    # Phase 3: Another greedy pass from SA result
    allowed = min(60.0, tleft() * 0.10)
    if allowed > 15:
        h_g2, c5_g2 = greedy_2swap(best_h_bin, max_iters=50000, time_limit=allowed,
                                    start_time=start_time, rng=rng, topk=5)
        if c5_g2 < best_score_bin:
            best_score_bin = c5_g2
            best_h_bin = h_g2.copy()

    # Phase 4: Continuous optimization
    allowed = min(200.0, tleft() * 0.20)
    if allowed > 30:
        h_co, c5_co = continuous_optimize(best_h_bin, time_limit=allowed,
                                           start_time=start_time, rng=rng)
        if c5_co < best_score_cont:
            best_score_cont = c5_co
            best_h_cont = h_co.copy()

    # Phase 5: Use continuous solution to perturb binary
    if tleft() > 60:
        h_pert = best_h_bin.copy()
        # Identify positions where continuous and binary differ most
        diff = best_h_cont - h_pert
        top_diff = np.argsort(np.abs(diff))[-10:]
        for idx in top_diff:
            if h_pert[int(idx)] == 0.0 and best_h_cont[int(idx)] > 0.5:
                zeros = np.where(h_pert == 0.0)[0]
                ones = np.where(h_pert == 1.0)[0]
                # Swap: set this position to 1, find a 1 to set to 0
                if len(ones) > 0:
                    h_pert[int(idx)] = 1.0
                    # Find the 1 to remove that minimizes damage
                    best_j = ones[0]
                    for j in ones[:20]:
                        d = compute_delta(h_pert, j, int(idx))
                        if float(np.max(np.correlate(h_pert, 1.0 - h_pert, mode='full')[:-1] + d)) < 200:
                            pass
                    # Just pick the one with smallest A value
                    n = len(h_pert)
                    corr_tmp = np.correlate(h_pert, 1.0 - h_pert, mode='full')
                    max_k = int(np.argmax(corr_tmp)) - (n - 1)
                    positions = np.arange(n)
                    A = np.zeros(n)
                    idx_s = positions + max_k
                    valid = (idx_s >= 0) & (idx_s < n)
                    A[valid] += h_pert[idx[valid]]
                    idx_s = positions - max_k
                    valid = (idx_s >= 0) & (idx_s < n)
                    A[valid] += h_pert[idx[valid]]
                    best_j = int(ones[np.argmin(A[ones])])
                    h_pert[best_j] = 0.0
        h_pert = project_binary(h_pert, target_init)
        sc_pert = compute_c5(h_pert)
        if sc_pert < best_score_bin:
            best_score_bin = sc_pert
            best_h_bin = h_pert.copy()

    # Phase 6: More SA rounds
    if tleft() > 50:
        allowed = min(100.0, tleft() * 0.15)
        if allowed > 30:
            h_sa2, c5_sa2 = sa_binary(best_h_bin, time_limit=allowed, start_time=start_time,
                                       rng=rng, n_perturb=3, t_init=0.015, t_final=0.00001)
            if c5_sa2 < best_score_bin:
                best_score_bin = c5_sa2
                best_h_bin = h_sa2.copy()

    # Phase 7: Final greedy
    if tleft() > 20:
        allowed = min(30.0, tleft() * 0.3)
        if allowed > 10:
            h_g3, c5_g3 = greedy_2swap(best_h_bin, max_iters=30000, time_limit=allowed,
                                        start_time=start_time, rng=rng, topk=3)
            if c5_g3 < best_score_bin:
                best_score_bin = c5_g3
                best_h_bin = h_g3.copy()

    # Phase 8: Continuous polish
    if tleft() > 30:
        allowed = min(40.0, tleft() * 0.4)
        if allowed > 15:
            h_cp, c5_cp = continuous_optimize(best_h_bin, time_limit=allowed,
                                               start_time=start_time, rng=rng)
            if c5_cp < best_score_cont:
                best_score_cont = c5_cp
                best_h_cont = h_cp.copy()

    # Final: choose best
    n_final = len(best_h_bin)
    target_final = float(n_final) / 2.0
    h_bin_final = project_binary(best_h_bin, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    n_cont = len(best_h_cont)
    h_cont_final = project_box_sum(best_h_cont, float(n_cont) / 2.0)
    cont_score = compute_c5(h_cont_final)

    if bin_score <= cont_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_cont_final, cont_score, n_cont
