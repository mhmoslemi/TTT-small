def project_box_sum(h, target):

def project_binary(h, target_sum):

def swap_delta(h, ia, ib, n):

def smooth_max_obj(h, beta, target, penalty):

    d = float(np.sum(h)) - target

    return val

    N = len(h)
    dx = 2.0 / N

    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1

        idx1_valid = idx1[mask1]
        t_valid = t_arr[mask1]
        grad[j] += np.dot(one_minus_h[np.minimum(np.maximum(t_valid, 0), N-1)], weights[idx1_valid] * mask1)
