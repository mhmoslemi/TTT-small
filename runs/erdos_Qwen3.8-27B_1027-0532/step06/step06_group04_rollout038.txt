    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-14 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def lp_solve_active(h, n, target, active_k, reg, scs_eps=1e-11):
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in active_k:
        k = k_idx - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo + k:hi + k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=200000)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    h_sol = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    t_val = float(t_var.value) if t_var.value is not None else None
    return h_sol, t_val


def alternating_lp(h_init, n, target, max_iter, time_limit, n_active, scs_eps=1e-11):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = sorted(np.argsort(corr)[-n_active:].tolist())
    reg = 1e-4
    damping = 0.9

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        h_sol, t_val = lp_solve_active(h, n, target, active_k, reg, scs_eps)
        if h_sol is None:
            break

        h_new = (1.0 - damping) * h + damping * h_sol
        h = project_feasible(h_new, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_set = set(np.argsort(corr)[-n_active:].tolist())

        t_check = best_c5
        check_set = set()
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in top_set:
                if corr[k_idx] * dx > t_check + 1e-6:
                    check_set.add(k_idx)
        new_active = top_set | check_set
        if len(new_active) > 3 * n_active:
            new_active = set(np.argsort(corr)[-3 * n_active:].tolist())
        if new_active != set(active_k):
            active_k = sorted(new_active)

        if it % 30 == 29:
            reg = max(1e-10, reg * 0.6)
            damping = min(0.99, damping + 0.01)

    return best_h, best_c5


def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.005

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - (n - 1)
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-12)[0]
            else:
                idx = np.where(h > 1e-12)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        new_active = np.argsort(corr)[-n_active:]
        active_k = new_active

        if p % 50 == 49:
            lam_rate = max(0.001, lam_rate * 0.9)

    return best_h, best_c5


def fista_smooth(h_init, n, target, alpha, max_iter, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    t_mom = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    step = 0.01

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        y = h + ((t_mom - 1.0) / (t_mom + 1.0)) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        C_vals = corr * dx
        c_max = np.max(C_vals)
        shifted = alpha * (C_vals - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights = weights / wsum

        active_k = np.where(weights > 1e-10)[0]

        grad = np.zeros(n)
        for k_idx in active_k:
            k = k_idx - half_n
            w = weights[k_idx] * dx
            L = max(0, -k)
            R = min(n - 1, n - 1 - k)
            mask1 = (idxs >= L) & (idxs <= R)
            mask2 = (idxs - k >= L) & (idxs - k <= R) & (idxs >= 0) & (idxs < n)
            if np.any(mask1):
                j1 = idxs[mask1]
                grad[mask1] += w * (1.0 - y[j1 + k])
            if np.any(mask2):
                j2 = idxs[mask2]
                grad[mask2] -= w * y[j2 - k]

        step_back = step
        for _ in range(10):
            h_try = project_feasible(y - step_back * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try <= c_max + step_back * step_back * np.dot(grad, grad) / (2.0 * alpha):
                break
            step_back *= 0.5
        step = max(0.01 * step_back, 1e-6)

        h_prev = h.copy()
        h = project_feasible(y - step_back * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        t_mom = t_new

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def binary_local_search(h, target, max_iters, time_limit):
    start = time.time()
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        mid_mask = (h > 0.05) & (h < 0.95)
        mid_idx = np.where(mid_mask)[0]
        if len(mid_idx) == 0:
            break
        improved = False
        rng_order = np.random.permutation(mid_idx)
        for i in rng_order[:30]:
            for val in [0.0, 1.0, 0.5]:
                if abs(h[i] - val) < 1e-6:
                    continue
                h_try = h.copy()
                h_try[i] = val
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for i in mid_idx[:15]:
                for delta in [-0.02, 0.02, -0.05, 0.05, -0.1, 0.1]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h[i] + delta, 0, 1)
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        h = h_try
                        best_h = h_try.copy()
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_perturb(h, mode, amplitude, rng):
    n = len(h)
    x = np.linspace(0.0, 2.0, n)
    phase = rng.uniform(0, 2 * np.pi)
    pert = amplitude * np.cos(np.pi * mode * x + phase)
    h_new = h + pert
    return project_feasible(h_new, sum(h))


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: LP linearization with large active set
    if tleft() > 200:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=min(300, tleft() * 0.35),
                                     n_active=80, scs_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent
    if tleft() > 100:
        h_cd, c5_cd = coord_descent(best_h, target, n_active=80,
                                    max_passes=30000, time_limit=min(120, tleft() * 0.15), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: FISTA with smooth approximation
    if tleft() > 120:
        alpha_list = [200, 1000, 5000]
        for a in alpha_list:
            if tleft() < 40:
                break
            h_f, c5_f = fista_smooth(best_h, n, target, alpha=a,
                                     max_iter=2000, time_limit=min(50, tleft() * 0.2))
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_f

    # Phase 4: Binary local search
    if tleft() > 60:
        h_bl, c5_bl = binary_local_search(best_h, target,
                                          max_iters=2000, time_limit=min(50, tleft() * 0.3))
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl

    # Phase 5: LP again with tighter settings
    if tleft() > 150:
        h_lp2, c5_lp2 = alternating_lp(best_h, n, target,
                                       max_iter=50000, time_limit=min(150, tleft() * 0.25),
                                       n_active=100, scs_eps=1e-12)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 6: Multi-start with Fourier perturbations
    for mode in [1, 2, 3, 5]:
        if tleft() < 100:
            break
        for amp in [0.03, 0.08]:
            if tleft() < 60:
                break
            h_pert = fourier_perturb(best_h, mode, amp, rng)
            h_cd_p, c5_p = coord_descent(h_pert, target, n_active=60,
                                         max_passes=10000, time_limit=min(30, tleft() * 0.25), rng=rng)
            if c5_p < best_score:
                best_score = c5_p
                best_h = h_cd_p

    # Phase 7: SLSQP polish
    if tleft() > 80:
        h_ref = slsqp_round(best_h, n, target, [2000, 10000, 50000], 1500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Try different n
    if tleft() > 200:
        for n_try in [396, 404, 410]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            sc_int = compute_c5(h_interp)
            if sc_int >= best_score:
                continue
            h_cd_n, c5_n = coord_descent(h_interp, target_try, n_active=60,
                                         max_passes=15000, time_limit=min(60, tleft() * 0.3), rng=rng)
            if c5_n < best_score:
                best_score = c5_n
                best_h = h_cd_n

    # Phase 9: Final LP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp3, c5_lp3 = alternating_lp(best_h, n_use, target_use,
                                       max_iter=20000, time_limit=min(40, tleft() * 0.6),
                                       n_active=100, scs_eps=1e-13)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 10: Final coordinate descent
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd_f, c5_cd_f = coord_descent(best_h, target_use, n_active=100,
                                        max_passes=20000, time_limit=min(25, tleft() * 0.7), rng=rng)
        if c5_cd_f < best_score:
            best_score = c5_cd_f
            best_h = h_cd_f

    # Phase 11: Final binary search
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bl_f, c5_bl_f = binary_local_search(best_h, target_use,
                                              max_iters=1000, time_limit=min(15, tleft() * 0.8))
        if c5_bl_f < best_score:
            best_score = c5_bl_f
            best_h = h_bl_f

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
