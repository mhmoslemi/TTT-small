Let me try a different strategy: **use the subgradient of the exact (non-smooth) max, combined with a projection method**.

The subgradient of max_k corr[k] at a point h is:
- If there's a unique maximizer k*, the subgradient is ∂corr[k*]/∂h
- If there are multiple maximizers, the subgradient is any convex combination of their gradients

We can use the "steepest descent" direction: the subgradient that points most in the direction of decreasing the feasible set's "distance to the sublevel set."

**Algorithm: Subgradient + Projection**
1. Compute corr, find active set A = {k : corr[k] = max(corr)}
2. Compute subgradient g = mean of {∂corr[k]/∂h : k ∈ A}
3. Project g onto the tangent space of the constraints (i.e., subtract mean to satisfy sum constraint)
4. Take a step: h_new = project(h - lr * g)
5. Use backtracking line search to ensure descent

This should work well because:
- It directly targets the active constraints
- The projection ensures feasibility
- Backtracking ensures we make progress

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_subgradient(h, corr, n):
    """Compute subgradient of max(corr) w.r.t. h using active set."""
    c_max = np.max(corr)
    active = np.where(corr >= c_max - 1e-12)[0]
    n_corr = len(corr)
    mid = n - 1
    grad = np.zeros(n)
    
    for s in active:
        k = int(s) - mid
        if k >= 0:
            # corr[mid+k] = sum_{i=0}^{n-k-1} h(i+k)*(1-h(i))
            # d/dh[j] = [k<=j<=n-1]*(1-h(j-k)) + [0<=j<=n-k-1]*(-h(j+k))
            lo1, hi1 = k, n-1
            lo2, hi2 = 0, n-k-1
            if lo1 <= hi1:
                grad[lo1:hi1+1] += (1.0 - h[lo1-k:hi1+1-k])
            if lo2 <= hi2:
                grad[lo2:hi2+1] += -h[lo2+k:hi2+1+k]
        else:
            kp = -k
            # corr[mid-kp] = sum_{i=0}^{n-kp-1} h(i)*(1-h(i+kp))
            # d/dh[j] = [0<=j<=n-kp-1]*(1-h(j+kp)) + [kp<=j<=n-1]*(-h(j-kp))
            lo1, hi1 = 0, n-kp-1
            lo2, hi2 = kp, n-1
            if lo1 <= hi1:
                grad[lo1:hi1+1] += (1.0 - h[lo1+kp:hi1+1+kp])
            if lo2 <= hi2:
                grad[lo2:hi2+1] += -h[lo2-kp:hi2+1-kp]
    
    grad /= len(active)
    return grad

def subgradient_method(h_init, n, target, time_limit, max_iters):
    """Subgradient method with backtracking line search."""
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        # Compute subgradient
        grad = exact_subgradient(h, corr, n)
        grad *= dx
        grad -= np.mean(grad)
        
        # Backtracking line search
        improved = False
        for trial in range(30):
            step = lr / (trial + 1)
            h_try = project_feasible(h - step * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-16:
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            lr *= 0.7
        else:
            lr = min(lr * 1.1, 1.0)
    
    return best_h, best_c5

def compute_sensitivities(h, n, candidates):
    """Compute d(corr)/dh[candidate] for all candidates simultaneously."""
    n_corr = 2 * n - 1
    mid = n - 1
    m = len(candidates)
    sens = np.zeros((m, n_corr))
    
    for idx, c in enumerate(candidates):
        # For k = 0 to n-1 (positive lags):
        # sens[idx, mid+k] = (1-h[c-k])*1[k<=c] - h[c+k]*1[c+k<n]
        ks = np.arange(n)  # k = 0, 1, ..., n-1
        mask1 = ks <= c  # k <= c
        mask2 = c + ks < n  # c+k < n
        vals1 = np.where(mask1, 1.0 - h[np.clip(c - ks, 0, n-1)], 0.0)
        vals2 = np.where(mask2, -h[c + ks], 0.0)
        sens[idx, mid + ks] = vals1 + vals2
        
        # For k = 1 to n-1 (negative lags):
        # sens[idx, mid-k] = (1-h[c+k])*1[c+k<n] - h[c-k]*1[k<=c]
        ks2 = np.arange(1, n)  # k = 1, ..., n-1
        mask3 = c + ks2 < n
        mask4 = ks2 <= c
        vals3 = np.where(mask3, 1.0 - h[c + ks2], 0.0)
        vals4 = np.where(mask4, -h[c - ks2], 0.0)
        sens[idx, mid - ks2] = vals3 + vals4
    
    return sens

Actually, I want to also try something new: **alternating block optimization**. 

Divide h into two halves: h = [h1, h2]. Optimize h1 with h2 fixed (this is a smaller problem), then optimize h2 with h1 fixed. This might escape local optima that full coordinate descent gets stuck in.

For optimizing h1 with h2 fixed:
The objective is max_k ∫ h(x)(1-h(x+k)) dx, which depends on both h1 and h2.
With h2 fixed, this is a function of h1 only. The constraint is 0 ≤ h1 ≤ 1 and sum(h1) = n/2 - sum(h2_fixed_in_range).

This is a smaller optimization problem (dimension n/2 instead of n), which might be easier to solve. We can use SLSQP or even a custom method.

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    
    for j in range(n):
        # d(corr)/dh[j]
        # For s = mid + k (k >= 0): d/dh[j] = (1-h[j-k])*1[k<=j] - h[j+k]*1[j+k<n]
        # For s = mid - k (k > 0): d/dh[j] = (1-h[j+k])*1[j+k<n] - h[j-k]*1[k<=j]
        
        # We need: grad[j] = sum_s weights[s] * d(corr[s])/dh[j] * dx
        
        # Positive lags k = 0, 1, ..., n-1:
        # s = mid + k
        # contribution: weights[mid+k] * ((1-h[j-k])*1[k<=j] - h[j+k]*1[j+k<n])
        
        # Negative lags k = 1, ..., n-1:
        # s = mid - k
        # contribution: weights[mid-k] * ((1-h[j+k])*1[j+k<n] - h[j-k]*1[k<=j])
        
        pos_k = np.arange(n)  # k = 0 to n-1
        s_pos = mid + pos_k
        mask1 = pos_k <= j
        mask2 = j + pos_k < n
        contrib_pos = weights[s_pos] * (np.where(mask1, 1.0 - h[j - pos_k], 0.0) + np.where(mask2, -h[j + pos_k], 0.0))
        
        neg_k = np.arange(1, n)  # k = 1 to n-1
        s_neg = mid - neg_k
        mask3 = j + neg_k < n
        mask4 = neg_k <= j
        contrib_neg = weights[s_neg] * (np.where(mask3, 1.0 - h[j + neg_k], 0.0) + np.where(mask4, -h[j - neg_k], 0.0))
        
        grad[j] = (np.sum(contrib_pos) + np.sum(contrib_neg)) * dx
    
    return obj, grad

def _con_fun(h):
    return np.sum(h) - _global_target

def _con_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj_grad, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000):
    """Efficient pair coordinate descent."""
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    no_improve = 0
    
    while time.time() - start_t < time_limit and no_improve < 2000:
        c_max = np.max(corr)
        
        # Active set (top 2)
        top_vals = np.sort(corr)[-3:]
        active = np.where(corr >= top_vals[-2] - 1e-12)[0]
        
        # Gradient from active set
        grad = np.zeros(n)
        for s in active:
            k = int(s) - mid
            if k >= 0:
                lo1, hi1 = k, n - 1
                lo2, hi2 = 0, n - k - 1
                if lo1 <= hi1:
                    grad[lo1:hi1+1] += (1.0 - h[lo1-k:hi1+1-k])
                if lo2 <= hi2:
                    grad[lo2:hi2+1] += -h[lo2+k:hi2+1+k]
            else:
                kp = -k
                lo1, hi1 = 0, n - kp - 1
                lo2, hi2 = kp, n - 1
                if lo1 <= hi1:
                    grad[lo1:hi1+1] += (1.0 - h[lo1+kp:hi1+1+kp])
                if lo2 <= hi2:
                    grad[lo2:hi2+1] += -h[lo2-kp:hi2+1-kp]
        
        grad /= len(active)
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        half = n_cand // 2
        low_set = sorted_idx[:half]
        high_set = sorted_idx[-(n_cand - half):]
        
        # Precompute sensitivities for all candidates
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_sens = len(all_cands)
        sens = np.zeros((n_sens, n_corr))
        cand_to_idx = {c: i for i, c in enumerate(all_cands)}
        
        for idx, c in enumerate(all_cands):
            c = int(c)
            # Positive lags: s = mid + k, k = 0..n-1
            for k in range(n):
                val = 0.0
                if k <= c:
                    val += (1.0 - h[c - k])
                if c + k < n:
                    val += -h[c + k]
                sens[idx, mid + k] = val
            # Negative lags: s = mid - k, k = 1..n-1
            for k in range(1, n):
                val = 0.0
                if c + k < n:
                    val += (1.0 - h[c + k])
                if k <= c:
                    val += -h[c - k]
                sens[idx, mid - k] = val
        
        # Find best pair
        best_max_after = c_max
        best_i, best_j, best_delta = -1, -1, 0.0
        
        for i in high_set:
            i_idx = cand_to_idx[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                j_idx = cand_to_idx[int(j)]
                g = sens[j_idx] - sens[i_idx]
                
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-14:
                    continue
                
                # Binary search for optimal delta
                a, b = 0.0, delta_max
                for _ in range(18):
                    mid_d = 0.5 * (a + b)
                    new_max = np.max(corr + mid_d * g)
                    if new_max < c_max - 1e-15:
                        active_new = np.where(corr + mid_d * g >= new_max - 1e-14)[0]
                        if len(active_new) > 0 and np.max(g[active_new]) > 1e-14:
                            b = mid_d
                        else:
                            a = mid_d
                    else:
                        a = mid_d
                
                delta_star = 0.5 * (a + b)
                if delta_star < 1e-14:
                    continue
                new_max = np.max(corr + delta_star * g)
                if new_max < best_max_after:
                    best_max_after = new_max
                    best_i, best_j, best_delta = int(i), int(j), delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Enforce sum constraint
            diff = target - np.sum(h)
            if abs(diff) > 1e-14:
                if diff > 0:
                    mask = h < 1.0 - 1e-10
                else:
                    mask = h > 1e-10
                if np.any(mask):
                    h[mask] += diff / np.sum(mask)
                    h = np.clip(h, 0.0, 1.0)
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
            no_improve = 0 if np.max(corr) < c_max - 1e-14 else no_improve + 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def subgradient_optimize(h_init, n, target, time_limit, max_iters=50000):
    """Subgradient method with Polyak step size."""
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    f_star = best_c5  # running best (for Polyak step)
    lr_base = 0.5
    no_improve = 0
    t_counter = 0
    
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            f_star = c5_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        
        t_counter += 1
        # Polyak step size
        gap = c5_val - f_star + 1e-6
        lr = lr_base / (np.linalg.norm(np.ones(n)) + 1e-10) * gap / max(t_counter, 1) * 100
        
        # Compute subgradient
        c_max = np.max(corr)
        active = np.where(corr >= c_max - 1e-12)[0]
        grad = np.zeros(n)
        for s in active:
            k = int(s) - mid
            if k >= 0:
                lo1, hi1 = k, n - 1
                lo2, hi2 = 0, n - k - 1
                if lo1 <= hi1:
                    grad[lo1:hi1+1] += (1.0 - h[lo1-k:hi1+1-k])
                if lo2 <= hi2:
                    grad[lo2:hi2+1] += -h[lo2+k:hi2+1+k]
            else:
                kp = -k
                lo1, hi1 = 0, n - kp - 1
                lo2, hi2 = kp, n - 1
                if lo1 <= hi1:
                    grad[lo1:hi1+1] += (1.0 - h[lo1+kp:hi1+1+kp])
                if lo2 <= hi2:
                    grad[lo2:hi2+1] += -h[lo2-kp:hi2+1-kp]
        grad /= len(active)
        grad *= dx
        grad -= np.mean(grad)
        
        # Take step
        h_try = project_feasible(h - lr * grad, target)
        h = h_try
    
    return best_h, best_c5
