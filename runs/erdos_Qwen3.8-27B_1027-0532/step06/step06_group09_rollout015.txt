import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_S(h):
    """Precompute S[i][m] = h[i-k] + h[i+k] for k = m-(n-1)"""
    n = len(h)
    idx = np.arange(n)
    shifts = np.arange(-(n-1), n)
    i_plus = idx[:, None] + shifts[None, :]
    i_minus = idx[:, None] - shifts[None, :]
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    S = np.where(valid_plus, h[np.clip(i_plus, 0, n-1)], 0.0) + \
        np.where(valid_minus, h[np.clip(i_minus, 0, n-1)], 0.0)
    return S

def smooth_obj(h, alpha, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    # Add constraint penalty
    penalty = 1000.0 * (np.sum(h) - target) ** 2
    return obj + penalty

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_obj(x, alpha, target), h_start,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

    no_improve = 0
    n_cand = n_cand_start
    
    # Precompute S

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_S(h):
    n = len(h)
    idx = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    i_plus = idx[:, None] + shifts[None, :]
    i_minus = idx[:, None] - shifts[None, :]
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    S = np.where(valid_plus, h[np.clip(i_plus, 0, n - 1)], 0.0) + \
        np.where(valid_minus, h[np.clip(i_minus, 0, n - 1)], 0.0)
    return S

def update_S(S, h, p, delta, n, mid, row_idx):
    S[row_idx, p - row_idx + mid] += delta
    S[row_idx, row_idx - p + mid] += delta

def smooth_obj_func(h, alpha, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    penalty = 500.0 * (np.sum(h) - target) ** 2
    return obj + penalty

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_func, h_start, args=(alpha, target),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def fast_pair_cd(h, n, target, time_limit, max_sweeps=200000,
                 n_cand=50, n_top_pairs=80, patience=400):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    row_idx = np.arange(n)

    S = compute_S(h)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star = int(np.argmax(corr)) - mid
        k_star_pos = k_star + mid
        idxs = np.arange(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        S_at_star = S[:, k_star_pos]

        pair_scores = []
        for i in high_set:
            si_val = S_at_star[i]
            for j in low_set:
                if i == j:
                    continue
                score = S_at_star[j] - si_val
                if score > 1e-10:
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi > 1e-10:
                        pair_scores.append((score, int(i), int(j)))

        if not pair_scores:
            continue

        pair_scores.sort(reverse=True)
        top_pairs = pair_scores[:n_top_pairs]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for score, i, j in top_pairs:
            g = S[i] - S[j]
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            a_d = 0.0
            b_d = delta_hi
            for _ in range(15):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            update_S(S, h, best_i, -best_delta, n, mid, row_idx)
            update_S(S, h, best_j, best_delta, n, mid, row_idx)

            if sweep % 200 == 0:
                h = project_feasible(h, target)
                S = compute_S(h)

    return best_h, best_c5

def multi_active_cd(h, n, target, time_limit, max_sweeps=50000,
                    n_cand=40, m_active=5, patience=300):
    """Pair CD targeting multiple active correlations simultaneously."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    row_idx = np.arange(n)

    S = compute_S(h)

    for sweep in range(max_sweeps):

            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        c_max = np.max(corr)
        threshold = c_max - 0.05 * c_max
        active_positions = np.where(corr >= threshold)[0]
        if len(active_positions) > m_active:
            top_active = np.argsort(corr)[-m_active:]
        else:
            top_active = active_positions

        k_first = int(top_active[0]) - mid
        idxs = np.arange(n)
        v1 = (idxs - k_first >= 0) & (idxs - k_first < n)
        v2 = (idxs + k_first >= 0) & (idxs + k_first < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_first])
        grad[v2] -= h[idxs[v2] + k_first]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        pair_scores = []
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-10:
                    continue
                g = S[i] - S[j]
                worst_at_active = np.max(g[top_active])
                if worst_at_active < -1e-10:
                    pair_scores.append((-worst_at_active, int(i), int(j)))

        if not pair_scores:
            continue

        pair_scores.sort(reverse=True)
        top_pairs = pair_scores[:60]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for sc, i, j in top_pairs:
            g = S[i] - S[j]
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            a_d = 0.0
            b_d = delta_hi
            for _ in range(15):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            update_S(S, h, best_i, -best_delta, n, mid, row_idx)
            update_S(S, h, best_j, best_delta, n, mid, row_idx)

            if sweep % 200 == 0:
                h = project_feasible(h, target)
                S = compute_S(h)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def balanced_subgradient(h, n, target, time_limit, max_steps=50000,
                         mu=0.85, base_lr=0.02, m_top=8):
    """Subgradient method with momentum targeting top-m correlations."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        threshold = c_max - 0.1 * c_max
        active = np.where(corr >= threshold)[0]
        if len(active) > m_top:
            active = np.argsort(corr)[-m_top:]

        grad = np.zeros(n)
        for pos in active:
            k = int(pos) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad *= dx / len(active)
        grad -= np.mean(grad)

        v = mu * v + grad
        lr = base_lr * max(0.1, 1.0 - step / max_steps)
        h_new = project_feasible(h - lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        h = h_new

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B polish
    if tleft() > 100:
        for alpha in [100, 500]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 2: Main fast pair CD
    if tleft() > 300:
        cd_time = min(350, tleft() * 0.35)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=200000,
                                   n_cand=50, n_top_pairs=80,
                                   patience=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B with higher alpha
    if tleft() > 100:
        for alpha in [2000, 10000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Multi-active CD
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                      time_limit=cd_time,
                                      max_sweeps=80000,
                                      n_cand=40, m_active=5,
                                      patience=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: More pair CD with different params
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=200000,
                                   n_cand=70, n_top_pairs=120,
                                   patience=350)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Balanced subgradient for escape from local optimum
    if tleft() > 100:
        h_sub, c5_sub = balanced_subgradient(best_h, n, target,
                                              time_limit=min(80, tleft() * 0.2),
                                              max_steps=30000,
                                              mu=0.85, base_lr=0.015,
                                              m_top=8)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub
            h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                       time_limit=min(80, tleft() * 0.2),
                                       max_sweeps=100000,
                                       n_cand=50, n_top_pairs=80,
                                       patience=200)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 7: Perturbation + re-optimize
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfags_polish(h_pert, n, target, 2000, maxiter=800)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                           time_limit=min(60, tleft() * 0.2),
                                           max_sweeps=80000,
                                           n_cand=50, n_top_pairs=80,
                                           patience=150)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 8: Try higher resolution
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, target_try, 1000, maxiter=800)
            h_hires, c5_hires = fast_pair_cd(h_hires, n_try, target_try,
                                             time_limit=min(80, tleft() * 0.25),
                                             max_sweeps=100000,
                                             n_cand=50, n_top_pairs=80,
                                             patience=200)
            h_hires = lbfags_polish(h_hires, n_try, target_try, 5000, maxiter=500)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 9: Final polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = lbfags_polish(best_h, n_use, target_use, 20000, maxiter=1000)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

        if tleft() > 30:
            h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                       time_limit=min(25, tleft() * 0.5),
                                       max_sweeps=50000,
                                       n_cand=60, n_top_pairs=100,
                                       patience=100)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
