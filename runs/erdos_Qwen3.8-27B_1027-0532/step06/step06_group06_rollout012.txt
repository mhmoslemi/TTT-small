    # These are different (one is w[n+j], the other is w[j+n-1-n'])
    
    # OK I think the cleanest way is:
    # A[j] = sum_p (1-h[p]) * w[j+n-1-p]
    # This is the convolution of (1-h) and w, reversed:
    # convolve((1-h), w)[j+n-1] = sum_p (1-h[p]) * w[j+n-1-p] ✓
    
    # So A = np.convolve(1-h, w, 'full')[n-1 : 2*n-1]
    
    # Similarly for B[j] = sum_p h[p] * w[j+n-1-p] (wait, let me check)
    # Second part: sum_m w[m] * h[j+(m-n+1)] * I(0<=j+m-n+1<n)
    # Let q = j+m-n+1, m = q-j+n-1
    # = sum_q h[q] * w[q-j+n-1] * I(0<=q<n and 0<=q-j+n-1<2n-1)
    # = sum_q w[q-j+n-1] * h[q]
    # Let p' = q-j+n-1, q = p'-j+n-1... this is getting circular.
    # 
    # B[j] = sum_q h[q] * w[q-j+n-1] for valid q and w index
    # = convolve(h, w_reversed)[j + something]... 
    # 
    # Actually: sum_q h[q] * w[q-j+n-1]
    # = sum_q h[q] * w_reversed[2n-2-(q-j+n-1)]
    # = sum_q h[q] * w_reversed[n-1-j+q]
    # = convolve(h, w_reversed)[n-1-j]... 
    
    # This is getting too complex. Let me just use FFT-based cross-correlation.
    # 
    # Actually, the simplest efficient approach for n~600:
    # Use the fact that grad = correlate(1-h, w_forward) - correlate(h, w_forward) evaluated appropriately.
    # 
    # Let me just verify numerically and use a practical approach.
    # For n=606, I can compute the gradient in O(n^2) which is ~367k ops per gradient evaluation.
    # With L-BFGS-B doing maybe 200 iterations, that's ~73M ops total, which should be fine.
    
    # Actually wait - I can use a much simpler approach:
    # grad[j] = sum_m w[m] * [(1-h[j-(m-n+1)]) - h[j+(m-n+1)]]
    # 
    # Let k = m - (n-1), so m = k + n - 1, k ranges from -(n-1) to (n-1)
    # grad[j] = sum_k w[k+n-1] * [(1-h[j-k]) - h[j+k]]
    #          = sum_k w[k+n-1] * (1-h[j-k]) - sum_k w[k+n-1] * h[j+k]
    
    # First term: sum_k w[k+n-1] * (1-h[j-k])
    # For fixed j, k ranges from -(n-1) to (n-1), and we need 0<=j-k<n, so k in [j-n+1, j]
    # = sum_{k=max(-(n-1),j-n+1)}^{min(n-1,j)} w[k+n-1] * (1-h[j-k])
    # Let p = j-k, so k = j-p, p ranges from 0 to min(j, n-1)
    # = sum_{p=0}^{min(j,n-1)} w[j-p+n-1] * (1-h[p])
    
    # Second term: sum_k w[k+n-1] * h[j+k]
    # For fixed j, need 0<=j+k<n, so k in [-j, n-1-j]
    # Let q = j+k, so k = q-j, q ranges from 0 to min(n-1-j, ...)
    # = sum_{q=0}^{n-1-j} w[q-j+n-1] * h[q]
    
    # So: grad[j] = sum_{p=0}^{j} w[j-p+n-1] * (1-h[p]) - sum_{q=0}^{n-1-j} w[q-j+n-1] * h[q]
    #              (with the understanding that w indices must be in [0, 2n-2])
    
    # This can be computed as:
    # first[j] = sum_{p=0}^{j} w[j-p+n-1] * (1-h[p])  = (1-h) correlated with w, at offset j+n-1
    # second[j] = sum_{q=0}^{n-1-j} w[q-j+n-1] * h[q] = h correlated with w... 
    
    # You know what, for n=606, let me just do the O(n^2) computation with numpy vectorization:
    
    # Create matrix M where M[j, m] = d(corr[m])/dh[j]
    # Then grad = M^T @ weights (times dx)
    
    # M[j, m] = (1-h[j-(m-n+1)]) * I(0<=j-(m-n+1)<n) - h[j+(m-n+1)] * I(0<=j+(m-n+1)<n)
    
    # For efficiency, I'll use the following:
    # The first term is a Toeplitz-like matrix based on (1-h)
    # The second term is a Toeplitz-like matrix based on h
    
    # Actually, the cleanest way: use FFT correlation.
    # The gradient of corr[m] w.r.t. h[j] can be computed as:
    # First part: (1-h) shifted by (m-(n-1)) and correlated with a delta at position j
    # This equals: (1-h)[j - (m-(n-1))] if valid, else 0
    # Second part: -h shifted by (m-(n-1)) and correlated with a delta at position j  
    # This equals: -h[j + (m-(n-1))] if valid, else 0
    
    # So grad[j] = sum_m w[m] * (1-h)[j-(m-n+1)] - sum_m w[m] * h[j+(m-n+1)]
    #            = sum_m w[m] * (1-h)[j-m+n-1] - sum_m w[m] * h[j+m-n+1]
    
    # First sum: let i = m, this is sum_i w[i] * (1-h)[j-i+n-1]
    # = (cross-correlation of w and (1-h), but with w in the "signal" role)
    # = np.correlate(w, (1-h))[n-1+j] ... let me verify
    # np.correlate(a, v, 'full')[k] = sum_n a[n+k] * v[n]
    # np.correlate(w, 1-h, 'full')[k] = sum_n w[n+k] * (1-h[n])
    # We want: sum_i w[i] * (1-h)[j-i+n-1]
    # Let n' = j-i+n-1, so i = j+n-1-n'
    # = sum_n' w[j+n-1-n'] * (1-h)[n']
    # This is sum_n' w[j+n-1-n'] * (1-h)[n']
    # = np.correlate(reversed_w, 1-h)[something]... 
    
    # OK I give up trying to get a clean closed form. Let me just use the direct O(n^2) approach
    # which is fast enough for n=606.
    
    t = np.arange(n)
    # For each j, k ranges such that both j-k and j+k are valid
    # k = m - (n-1) for m = 0, ..., 2n-2
    # So k = -(n-1), ..., n-1
    
    # grad[j] = sum_k w[k+n-1] * [(1-h[j-k]) * I(0<=j-k<n) - h[j+k] * I(0<=j+k<n)]
    
    # Vectorized: for each j, the valid k range is max(-(n-1), j-(n-1)) to min(n-1, n-1-j)
    # i.e., max(2j-n+1, -(n-1)) to min(n-1, n-1-j) ... this is getting complicated.
    
    # Simplest vectorized approach:
    # Create a (n, 2n-1) matrix where entry [j, m] = (1-h[(j-(m-n+1)) % n]) ... no, that's wrong.
    
    # Let me just do it with broadcasting for n~600:
    # j_idx = np.arange(n)[:, None]  # (n, 1)
    # k_vals = np.arange(-(n-1), n)  # (2n-1,)
    # valid1 = (j_idx - k_vals[None, :] >= 0) & (j_idx - k_vals[None, :] < n)
    # valid2 = (j_idx + k_vals[None, :] >= 0) & (j_idx + k_vals[None, :] < n)
    # term1 = np.where(valid1, 1.0 - h[j_idx - k_vals[None, :]], 0.0)
    # term2 = np.where(valid2, h[j_idx + k_vals[None, :]], 0.0)
    # grad = (term1 - term2) @ weights
    
    # This creates a (606, 1211) matrix = ~734k elements, which is fine.
    
    j_idx = np.arange(n)[:, None]  # (n, 1)
    k_vals = np.arange(-(n-1), n)  # (2n-1,)
    
    # j_idx - k_vals: (n, 2n-1)
    idx1 = j_idx - k_vals[None, :]
    idx2 = j_idx + k_vals[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # Build the gradient
    h_at_1 = np.where(valid1, 1.0 - h[idx1.clip(0, n-1)], 0.0)
    h_at_2 = np.where(valid2, h[idx2.clip(0, n-1)], 0.0)
    
    grad = (h_at_1 - h_at_2) @ weights
    grad *= dx
    
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def iterative_reweighting(h_start, n_iters, target, penalty, start_time, time_limit, 
                          tau_init=0.01, tau_final=0.001):
    """Iterative reweighting (multiplicative weights) for the max-correlation problem."""
    n = len(h_start)
    h = h_start.copy()
    weights = np.ones(2*n - 1) / (2*n - 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Solve: min_h sum_k w[k] * corr_k(h) + penalty * (sum(h) - target)^2
        # Use L-BFGS-B
        h_opt = minimize(
            weighted_corr_objective, h,
            args=(weights, target, penalty),
            jac=weighted_corr_gradient,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        ).x
        
        h_opt = project_box_sum(h_opt, target)
        
        # Evaluate
        corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
        c5 = float(corr.max()) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        
        h = h_opt
        
        # Update weights using softmax
        tau = tau_init * (tau_final / tau_init) ** (it / max(n_iters - 1, 1))
        shifted = corr / tau
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
    
    return best_h, best_c5

def fast_binary_local_search(h_start, target_sum, time_limit, start_time, 
                              tabu_size=20, max_stall=300):
    """Binary local search with tabu list and incremental correlation updates."""
    n = len(h_start)
    h = h_start.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stall = 0
    max_stall = max(max_stall, n // 3)
    tabu = set()
    
    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Find the worst correlation
        max_vals = np.sort(corr)[::-1]
        # Consider the top few worst lags
        top_frac = 0.05
        n_top = max(1, int(len(corr) * top_frac))
        top_indices = np.argsort(corr)[::-1][:n_top]
        worst_lag = top_indices[0] - (n - 1)
        
        # For each of the top lags, try to find a good (i, j) pair
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for lag_idx in top_indices[:5]:
            k = lag_idx - (n - 1)
            
            # Find candidates: i where h[i]=1 and h[i+k]=0 (to reduce corr at this lag)
            # j where h[j]=0 and h[j+k]=1 (to reduce corr at this lag)
            ic = np.array([i for i in ones if 0 <= i + k < n and h[i + k] == 0])
            jc = np.array([j for j in zeros if 0 <= j + k < n and h[j + k] == 1])
            
            if len(ic) == 0 or len(jc) == 0:
                continue
            
            # Limit candidates
            if len(ic) > 15:
                # Sort by how much they contribute to the worst corr
                contrib = np.array([corr[lag_idx] for _ in range(len(ic))])  # all same for this lag
                ic = ic[:15]
            if len(jc) > 15:
                jc = jc[:15]
            
            for i in ic[:10]:
                if i in tabu:
                    continue
                for j in jc[:10]:
                    if j in tabu:
                        continue
                    if i == j:
                        continue
                    
                    # Compute new max correlation (approximately)
                    new_max = 0.0
                    for ti in top_indices:
                        tk = ti - (n - 1)
                        delta = 0.0
                        # Effect of flipping h[i] from 1 to 0:
                        if 0 <= i + tk < n:
                            delta -= (1.0 - h[i + tk])
                        if 0 <= i - tk < n:
                            delta += h[i - tk]
                        # Effect of flipping h[j] from 0 to 1:
                        if 0 <= j + tk < n:
                            delta += (1.0 - h[j + tk])
                        if 0 <= j - tk < n:
                            delta -= h[j - tk]
                        
                        new_val = corr[ti] + delta
                        if new_val > new_max:
                            new_max = new_val
                            break
                    
                    if new_max < float(corr[lag_idx]) - best_improve:
                        best_improve = float(corr[lag_idx]) - new_max
                        best_i = i
                        best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = float(corr.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            
            # Update tabu
            if len(tabu) >= tabu_size:
                # Remove oldest (just clear all for simplicity)
                tabu.clear()
            tabu.add(best_i)
            tabu.add(best_j)
        else:
            stall += 1
            if stall > max_stall // 2:
                # Try a random move to escape
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    stall = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)
    
    # Initial best
    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)
    
    h_bin = project_binary(h_init, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Phase 1: Iterative reweighting (new approach)
    if tleft() > 100:
        time_alloc = min(180, tleft() * 0.3)
        h_ir, c5_ir = iterative_reweighting(
            best_h, n_iters=30, target=target, penalty=500.0,
            start_time=start_time, time_limit=time_alloc,
            tau_init=0.05, tau_final=0.005
        )
        if c5_ir < best_c5:
            best_c5 = c5_ir
            best_h = h_ir.copy()
    
    # Phase 2: Log-sum-exp refinement with increasing beta
    if tleft() > 80:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [20, 50, 100, 200, 400, 800]:
            if tleft() < 30:
                break
            time_alloc = min(40, tleft() * 0.1)
            
            def smooth_obj(h_x):
                corr = np.correlate(h_x, 1.0 - h_x, mode='full')
                c_max = float(corr.max())
                shifted = beta * (corr - c_max)
                shifted -= shifted.max()
                val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
                d = float(np.sum(h_x)) - target
                val += 1000.0 * d * d
                return val
            
            def smooth_grad(h_x):
                N = len(h_x)
                dx_l = 2.0 / N
                corr = np.correlate(h_x, 1.0 - h_x, mode='full')
                c_max = float(corr.max())
                shifted = beta * (corr - c_max)
                shifted -= shifted.max()
                weights = np.exp(shifted)
                weights /= weights.sum()
                # Use the weighted gradient
                return weighted_corr_gradient(h_x, weights, target, 1000.0)
            
            h_opt = minimize(
                smooth_obj, h_cont,
                jac=smooth_grad,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': -1}
            ).x
            h_opt = project_box_sum(h_opt, target)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()
    
    # Phase 3: Binary local search with tabu
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.2)
        h_ls, c5_ls = fast_binary_local_search(
            h_bin, target_int, time_limit=time_alloc,
            start_time=start_time, tabu_size=30, max_stall=200
        )
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 4: More iterative reweighting from improved solution
    if tleft() > 80:
        time_alloc = min(120, tleft() * 0.2)
        h_ir2, c5_ir2 = iterative_reweighting(
            best_h, n_iters=20, target=target, penalty=1000.0,

    n = len(h)

    # corr[m] = sum_i h[i+k] * (1-h[i]) where k = m-(n-1)
    # d(corr[m])/dh[j] = (1-h[j-k]) * I(valid) - h[j+k] * I(valid)
