def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    c_max = np.max(C)
    shifted = alpha * (C - c_max)
    weights = np.exp(shifted)
    wsum = weights.sum()
    weights /= wsum
    mid = n - 1
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid  # lag
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            # grad: for j in [0, n-1-k]: 1-h[j+k]; for j in [k, n-1]: -h[j-k]

            else:
                grad[:n-k] += w * (1.0 - h[k:]) * dx
                grad[k:] += w * (-h[:n-k]) * dx
        else:

            # f_{-kp} = sum_{i=kp}^{n-1} h[i](1-h[i-kp])

            grad[:n-kp] += w * (-h[kp:]) * dx
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    cmax = np.max(C)
    shifted = alpha * (C - cmax)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    obj = cmax + np.log(wsum) / alpha
    weights = exp_shifted / wsum
    mid = n - 1
    grad = np.zeros(n)
    significant = weights > 1e-10
    sig_idx = np.where(significant)[0]
    for s in sig_idx:
        w = weights[s]
        k = int(s) - mid
        if k >= 0:
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                grad[:n-k] += w * (1.0 - h[k:])
                grad[k:] += w * (-h[:n-k])
        else:
            kp = -k
            grad[kp:] += w * (1.0 - h[:n-kp])
            grad[:n-kp] += w * (-h[kp:])
    grad *= dx
    if penalty > 0:
        s_h = np.sum(h)
        t = n / 2.0
        obj += penalty * (s_h - t) ** 2
        grad[:] += 2.0 * penalty * (s_h - t)
    return obj, grad
