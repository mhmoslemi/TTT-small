    n = len(h)

    return float(np.max(corr)) * dx

def apply_symmetry(h_red, n_full):
    n_red = n_full // 2
    h_full = np.empty(n_full)
    h_full[:n_red] = h_red
    h_full[n_red:] = 1.0 - h_red[::-1]
    return h_full

def init_symmetric(h_full):
    n_full = len(h_full)
    n_red = n_full // 2
    # h_red[j] = (h_full[j] + 1 - h_full[n-1-j]) / 2
    mirror = h_full[n_full-1: n_full-1-n_red: -1]
    h_red = (h_full[:n_red] + 1.0 - mirror) / 2.0
    h_red = np.clip(h_red, 0.0, 1.0)
    return h_red

def smooth_obj_sym(h_red, n_full, alpha):
    h_full = apply_symmetry(h_red, n_full)
    dx = 2.0 / n_full
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad_sym(h_red, n_full, alpha):
    n_red = n_full // 2
    n = n_full
    h_full = apply_symmetry(h_red, n)
    dx = 2.0 / n
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    mid = n - 1
    grad_full = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-15:

            grad_full[idx] += wm * (1.0 - h_full[idx + s])
        # -h_full[j-s] for valid j: j ranges [max(0,s), min(n-1, n-1+s)]

    grad_red = (grad_full[:n_red] - grad_full[mirror_idx]) * dx

def apply_symmetry(h_red, n_full):
    n_red = n_full // 2
    h_full = np.empty(n_full)
    h_full[:n_red] = h_red
    h_full[n_red:] = 1.0 - h_red[::-1]
    return h_full

def init_symmetric(h_full):
    n_full = len(h_full)
    n_red = n_full // 2
    mirror = h_full[n_full - 1: n_full - 1 - n_red: -1]
    h_red = (h_full[:n_red] + 1.0 - mirror) / 2.0
    h_red = np.clip(h_red, 0.0, 1.0)
    return h_red

def smooth_obj_sym(h_red, n_full, alpha):
    h_full = apply_symmetry(h_red, n_full)
    dx = 2.0 / n_full
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad_sym(h_red, n_full, alpha):
    n_red = n_full // 2
    n = n_full
    h_full = apply_symmetry(h_red, n)
    dx = 2.0 / n
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad_full = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-15:
            continue
        s = m - mid
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_hi >= j_lo:
            idx = np.arange(j_lo, j_hi + 1)
            grad_full[idx] += wm * (1.0 - h_full[idx + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, n - 1 + s)
        if j_hi2 >= j_lo2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad_full[idx2] -= wm * h_full[idx2 - s]
    mirror_idx = np.arange(n_full - 1, n_full - 1 - n_red, -1)
    grad_red = (grad_full[:n_red] - grad_full[mirror_idx]) * dx
    return grad_red

def power_mean_obj(h_red, n_full, p):
    h_full = apply_symmetry(h_red, n_full)
    dx = 2.0 / n_full
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    return float(np.sum(corr ** p) ** (1.0 / p)) * dx

def lbfgsb_reduced(h_red, n_full, alphas, maxiter, ftol, time_limit):
    start_t = time.time()
    n_red = n_full // 2
    bounds = [(0.0, 1.0)] * n_red
    h_cur = h_red.copy()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(
                lambda x: smooth_obj_sym(x, n_full, alpha),
                h_cur,
                jac=lambda x: smooth_grad_sym(x, n_full, alpha),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            if compute_c5(apply_symmetry(h_new, n_full)) <= compute_c5(apply_symmetry(h_cur, n_full)) + 1e-12:
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def slsqp_reduced(h_red, n_full, alphas, maxiter, ftol, time_limit):
    start_t = time.time()
    n_red = n_full // 2
    bounds = [(0.0, 1.0)] * n_red
    h_cur = h_red.copy()
    for alpha in alphas:
        if time.time() - start_t > time_limit:

            res = minimize(
                lambda x: smooth_obj_sym(x, n_full, alpha),

            h_new = np.clip(res.x, 0.0, 1.0)
            if compute_c5(apply_symmetry(h_new, n_full)) <= compute_c5(apply_symmetry(h_cur, n_full)) + 1e-12:

def momentum_reduced(h_red, n_full, time_limit, max_steps, alpha, mu, base_lr):
    start_t = time.time()
    n_red = n_full // 2
    h = h_red.copy()
    v = np.zeros(n_red)
    best_h = h.copy()
    best_c5 = compute_c5(apply_symmetry(h, n_full))
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad_sym(h, n_full, alpha)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = np.clip(h - lr * lr_mult * v, 0.0, 1.0)
            c5_try = compute_c5(apply_symmetry(h_try, n_full))
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
    return best_h, best_c5

def pair_cd_reduced(h_red, n_full, time_limit, max_sweeps, n_cand, top_k):
    start_t = time.time()
    n_red = n_full // 2
    n = n_full
    h = h_red.copy()
    best_c5 = compute_c5(apply_symmetry(h, n))
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    consecutive_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        h_full = apply_symmetry(h, n)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 200:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_alpha = 80.0
        w_top = np.exp(soft_alpha * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()
        grad_full = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
            v2 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            grad_full[v1] += wi * (1.0 - h_full[np.arange(n)[v1] - k])
            grad_full[v2] -= wi * h_full[np.arange(n)[v2] + k]
        mirror_idx = np.arange(n - 1, n - 1 - n_red, -1)
        grad_red = grad_full[:n_red] - grad_full[mirror_idx]
        sorted_idx = np.argsort(grad_red)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        shifts = np.arange(-(n - 1), n)
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                for pos in [i, n - 1 - i, j, n - 1 - j]:
                    sign = 1.0 if pos in (i,) or (pos == n - 1 - i and i != n - 1 - i) else (-1.0 if pos in (j,) or (pos == n - 1 - j and j != n - 1 - j) else 0.0)
                    if pos == i:
                        sign_i = 1.0
                    elif pos == n - 1 - i:
                        sign_i = -1.0
                    elif pos == j:
                        sign_i = -1.0
                    elif pos == n - 1 - j:
                        sign_i = 1.0
                    else:
                        continue
                    vp = (pos + shifts >= 0) & (pos + shifts < n)
                    vm = (pos - shifts >= 0) & (pos - shifts < n)
                    g[vp] += sign_i * h_full[pos + shifts[vp]]
                    g[vm] += sign_i * h_full[pos - shifts[vm]]
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if a_d >= b_d:
                    continue
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * (2.0 / n)
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
    return best_h, best_c5

def power_mean_opt(h_red, n_full, p, time_limit, maxiter):
    start_t = time.time()
    n_red = n_full // 2
    bounds = [(0.0, 1.0)] * n_red
    h_cur = h_red.copy()
    try:
        res = minimize(
            lambda x: power_mean_obj(x, n_full, p),
            h_cur,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        if compute_c5(apply_symmetry(h_new, n_full)) < compute_c5(apply_symmetry(h_cur, n_full)):
            h_cur = h_new
    except Exception:
        pass
    return h_cur

def interpolate_to(h_full_src, n_dst):
    n_src = len(h_full_src)
    if n_src == n_dst:
        return h_full_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_full_src)
    return h_out

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    if n % 2 != 0:
        n = n - 1
        h_init = h_init[:n]

    h_red = init_symmetric(h_init)
    best_h_red = h_red.copy()
    best_c5 = compute_c5(apply_symmetry(best_h_red, n))

    if tleft() > 80:
        h_r = lbfgsb_reduced(best_h_red, n, [10, 30, 100, 300], 3000, 1e-15, min(70, tleft()))
        c5 = compute_c5(apply_symmetry(h_r, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_r

    if tleft() > 60:
        h_r = lbfgsb_reduced(best_h_red, n, [1000, 5000, 20000], 2000, 1e-16, min(50, tleft()))
        c5 = compute_c5(apply_symmetry(h_r, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_r

    if tleft() > 50:
        h_m, c5_m = momentum_reduced(best_h_red, n, min(35, tleft() * 0.4), 50000, 200, 0.9, 0.04)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h_red = h_m

    if tleft() > 50:
        h_r = slsqp_reduced(best_h_red, n, [500, 2000, 10000], 1500, 1e-16, min(40, tleft()))
        c5 = compute_c5(apply_symmetry(h_r, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_r

    if tleft() > 60:
        h_cd, c5_cd = pair_cd_reduced(best_h_red, n, min(45, tleft() * 0.4), 3000, 40, 3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h_red = h_cd

    if tleft() > 40:
        h_r = lbfgsb_reduced(best_h_red, n, [5000, 20000, 100000], 1500, 1e-16, min(30, tleft()))
        c5 = compute_c5(apply_symmetry(h_r, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_r

    if tleft() > 30:
        h_pm = power_mean_opt(best_h_red, n, 20, min(20, tleft()), 1000)
        c5 = compute_c5(apply_symmetry(h_pm, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_pm

    if tleft() > 50:
        h_m, c5_m = momentum_reduced(best_h_red, n, min(30, tleft() * 0.4), 40000, 500, 0.85, 0.02)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h_red = h_m

    if tleft() > 80:
        for n_try in [600, 800]:
            if tleft() < 50:
                break
            h_full_cur = apply_symmetry(best_h_red, n)
            h_interp = interpolate_to(h_full_cur, n_try)
            h_red_try = init_symmetric(h_interp)
            h_r = lbfgsb_reduced(h_red_try, n_try, [10, 50, 200, 1000, 5000], 2000, 1e-15, min(35, tleft()))
            c5 = compute_c5(apply_symmetry(h_r, n_try))
            if c5 < best_c5:
                best_c5 = c5
                best_h_red = h_r
                n = n_try
            if tleft() > 30:
                h_m2, c5_m2 = momentum_reduced(best_h_red, n, min(20, tleft() * 0.4), 20000, 2000, 0.9, 0.02)
                if c5_m2 < best_c5:
                    best_c5 = c5_m2
                    best_h_red = h_m2

    if tleft() > 30:
        h_cd, c5_cd = pair_cd_reduced(best_h_red, n, min(20, tleft() * 0.5), 2000, 50, 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h_red = h_cd

    if tleft() > 20:
        h_r = lbfgsb_reduced(best_h_red, n, [10000, 50000], 1000, 1e-16, min(15, tleft()))
        c5 = compute_c5(apply_symmetry(h_r, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h_red = h_r

    if tleft() > 15:
        h_m, c5_m = momentum_reduced(best_h_red, n, min(10, tleft()), 20000, 5000, 0.9, 0.01)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h_red = h_m

    h_final = apply_symmetry(best_h_red, n)
    final_c5 = compute_c5(h_final)
    return h_final, final_c5, n
