def block_slsqp(h, n, target, block_size, block_start, k_top, alpha, maxiter):
    """Optimize a single block using SLSQP"""
    B = block_size
    block = h[block_start:block_start+B].copy()
    
    # Precompute fixed contributions
    # For the block, the objective is a function of block variables only
    
    def obj(x):
        h_test = h.copy()
        h_test[block_start:block_start+B] = np.clip(x, 0, 1)
        return topk_smooth(h_test, k_top, alpha)
    
    def grad(x):
        h_test = h.copy()
        h_test[block_start:block_start+B] = np.clip(x, 0, 1)
        g = topk_smooth_grad(h_test, k_top, alpha, n)
        return g[block_start:block_start+B]
    
    bounds = [(0.0, 1.0)] * B
    # Sum constraint: sum(block) = sum(h[block_start:block_start+B])
    target_block = np.sum(h[block_start:block_start+B])
    
    def con_fun(x):
        return np.sum(x) - target_block
    
    def con_jac(x):
        return np.ones(B)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, block, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return np.clip(res.x, 0, 1)
    except:
        return block

def lp_refine_topk(h0, n, target, time_limit, n_active=15, max_rounds=30):

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in sorted_idx:

            grad_k = np.zeros(n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_perturb(h, n, target, rng, scale=0.02):
    h_p = h.copy()
    mode = rng.integers(0, 3)
    if mode == 0:
        b = max(4, n // 10)
        s = rng.integers(0, n - b)
        h_p[s:s+b] += scale * rng.normal(0, 1, b)
    elif mode == 1:
        idx = rng.choice(n, size=max(1, n // 8), replace=False)
        h_p[idx] += scale * rng.normal(0, 1, len(idx))
    else:
        b = max(4, n // 10)
        s1 = rng.integers(0, n - b)
        s2 = rng.integers(0, n - b)
        h_p[s1:s1+b], h_p[s2:s2+b] = h_p[s2:s2+b].copy(), h_p[s1:s1+b].copy()
    return project_feasible(h_p, target)

                if tleft() < 30:
                    break
                h_new = slsqp_topk(best_h, n, target, k_top, alpha,
                                    maxiter=3000, ftol=1e-15,
                                    time_limit=min(40.0, tleft() * 0.25),
                                    start_time=start_time)
                c5_new = compute_c5(h_new)
                if c5_new < best_score:
                    best_score = c5_new
                    best_h = h_new

    # Phase 7: CD + PG interleaved
    if tleft() > 150:
        for iteration in range(3):
            if tleft() < 50:
                break
            h_cd, c5_cd = coord_descent_2pt(best_h, n, target,
                                             time_limit=min(50.0, tleft() * 0.2),
                                             max_sweeps=8000, n_cand=20, k_top=5)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
            if tleft() < 40:
                break
            h_pg, c5_pg = projected_grad_descent(best_h, n, target, 6,
                                                  alpha=200.0, time_limit=min(40.0, tleft() * 0.2),
                                                  start_time=start_time, max_steps=3000)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

    # Phase 8: Structured perturb + re-optimize (escape local optima)
    if tleft() > 200:
        for attempt in range(5):
            if tleft() < 80:
                break
            h_pert = structured_perturb(best_h, n, target, rng, scale=0.01 + 0.005 * attempt)
            h_new = slsqp_topk(h_pert, n, target, 5, 200,
                                maxiter=2000, ftol=1e-14,
                                time_limit=min(40.0, tleft() * 0.2),
                                start_time=start_time)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_score:
                best_score = c5_new
                best_h = h_new
            h_cd, c5_cd = coord_descent_2pt(h_new, n, target,
                                             time_limit=min(30.0, tleft() * 0.15),
                                             max_sweeps=5000, n_cand=15, k_top=5)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 9: Multi-resolution
    if tleft() > 250:
        for n_try in [600, 1000]:
            if tleft() < 120:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_topk(h_interp, n_try, target_try, 5, 200,
                                  maxiter=2000, ftol=1e-14,
                                  time_limit=min(80.0, tleft() * 0.3),
                                  start_time=start_time)
            c5_hires = compute_c5(h_hires)
            if tleft() < 60:
                if c5_hires < best_score:
                    best_score = c5_hires
                    best_h = h_hires
                continue
            h_cd_h, c5_cd_h = coord_descent_2pt(h_hires, n_try, target_try,
                                                  time_limit=min(50.0, tleft() * 0.25),
                                                  max_sweeps=8000, n_cand=15, k_top=5)
            if c5_cd_h < best_score:
                best_score = c5_cd_h
                best_h = h_cd_h
            if tleft() > 40:
                h_hires2 = slsqp_topk(best_h, len(best_h), float(len(best_h)) / 2.0,
                                       8, 1000, maxiter=2000, ftol=1e-15,
                                       time_limit=min(30.0, tleft() * 0.25),
                                       start_time=start_time)
                c5_hires2 = compute_c5(h_hires2)
                if c5_hires2 < best_score:
                    best_score = c5_hires2
                    best_h = h_hires2

    # Phase 10: Final LP at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_lp, c5_lp = lp_refine_topk(best_h, n_use, target_use,
                                      time_limit=min(50.0, tleft() * 0.4),
                                      n_active=20, max_rounds=25)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        for k_top, alpha in [(5, 2000), (3, 5000)]:
            if tleft() < 20:
                break
            h_final = slsqp_topk(best_h, n_use, target_use, k_top, alpha,
                                  maxiter=2000, ftol=1e-15,
                                  time_limit=min(20.0, tleft() * 0.4),
                                  start_time=start_time)
            c5_final = compute_c5(h_final)
            if c5_final < best_score:
                best_score = c5_final
                best_h = h_final

    # Phase 12: Final CD
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = coord_descent_2pt(best_h, n_use, target_use,
                                         time_limit=min(10.0, tleft() * 0.7),
                                         max_sweeps=5000, n_cand=20, k_top=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)
    return h_out, final_score, n_final
