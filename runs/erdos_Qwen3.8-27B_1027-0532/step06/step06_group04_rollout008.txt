def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.01
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) > 1e-8:

            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)

# Constraint A: coeff_a[j] * h_new[j] for j in [i_lo, i_hi)
# coeff_a[i] = (1 - h_old[i+k]) * dx
coeff_a = np.zeros(n)
coeff_a[i_lo:i_hi] = (1.0 - h[i_lo + k:i_hi + k]) * dx
constraints.append(coeff_a @ h_var <= t_var)

# Constraint B: 
# j = i + k, so j ranges from i_lo+k to i_hi+k-1
j_lo = i_lo + k
j_hi = i_hi + k  # exclusive
# coeff_b[j] = h_old[j-k] * dx = h_old[i] * dx
coeff_b = np.zeros(n)
coeff_b[j_lo:j_hi] = h[i_lo:i_hi] * dx
sum_h_old = np.sum(h[i_lo:i_hi]) * dx
# Constraint: coeff_b @ h_new >= sum_h_old - t
constraints.append(coeff_b @ h_var >= sum_h_old - t_var)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def build_lp_constraints(h_var, t_var, h, n, target, active_k_set, use_both_cuts):
    dx = 2.0 / n
    half_n = n - 1
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h
    for k_idx in sorted(active_k_set):
        k = k_idx - half_n
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi <= i_lo:
            continue
        coeff_a = np.zeros(n)
        coeff_a[i_lo:i_hi] = g[i_lo + k:i_hi + k] * dx
        constraints.append(coeff_a @ h_var <= t_var)
        if use_both_cuts:
            j_lo = i_lo + k
            j_hi = i_hi + k
            coeff_b = np.zeros(n)
            coeff_b[j_lo:j_hi] = h[i_lo:i_hi] * dx
            sum_h_old = np.sum(h[i_lo:i_hi]) * dx
            constraints.append(coeff_b @ h_var >= sum_h_old - t_var)
    return constraints

def dca_both_cuts(h_init, n, target, max_iter, time_limit, n_active_init, n_active_max, d0, use_both, reg_weight):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    n_active_current = n_active_init
    stall_count = 0

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = set(np.argsort(corr)[-n_active_current:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_current = float(np.max(corr)) * dx

        top_k = np.argsort(corr)[-n_active_current:]
        active_set = set(top_k.tolist())
        threshold = t_current * 0.90
        for k_idx in range(2, len(corr) - 2):
            if corr[k_idx] * dx >= threshold and k_idx not in active_set:
                active_set.add(k_idx)
        if len(active_set) > n_active_max:
            ak_list = list(active_set)
            ak_list.sort(key=lambda x: corr[x], reverse=True)
            active_set = set(ak_list[:n_active_max])

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = build_lp_constraints(h_var, t_var, h, n, target, active_set, use_both)

        if reg_weight > 0:
            objective = t_var + reg_weight * cp.sum_squares(h_var - h)
        else:
            objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved or h_var.value is None:
            if use_both:
                constraints2 = build_lp_constraints(h_var, t_var, h, n, target, active_set, False)
                prob2 = cp.Problem(cp.Minimize(objective if reg_weight > 0 else t_var), constraints2)
                try:
                    prob2.solve(solver=cp.SCS, eps=1e-10, max_iters=100000, verbose=False)
                    if h_var.value is not None:
                        solved = True
                except Exception:
                    pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_damped = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h_damped, target)
        c5_val = compute_c5(h)

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 0:
            damping = max(0.2, damping * 0.96)
        else:
            damping = min(1.0, damping + 0.05)

        if it % 30 == 29 and n_active_current < n_active_max:
            n_active_current = min(n_active_max, n_active_current + 5)

        if it % 200 == 199:
            damping = 1.0
            if reg_weight > 1e-8:
                reg_weight = max(1e-8, reg_weight * 0.5)

    return best_h, best_c5

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.01
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    return best_h, best_c5

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 60:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    penalty = 1e4 * (np.sum(h) - target) ** 2
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n) + penalty

def slsqp_refine(h_init, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

                                       d0=0.99, use_both=True, reg_weight=1e-7)
        if c5_dca < best_score:
            best_score = c5_dca
            best_h = h_dca

    # Phase 11: Final CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(25, tleft() * 0.5)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=150,
                                     max_passes=30000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final subgradient
    if tleft() > 10:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                           max_steps=2000, time_limit=min(8, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
