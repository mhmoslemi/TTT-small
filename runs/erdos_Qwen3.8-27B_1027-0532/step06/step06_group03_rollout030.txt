    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        h_full = make_full_from_reduced(a)
        return smooth_max_obj(h_full, _cur_alpha)

    def grad_red(a):
        h_full = make_full_from_reduced(a)
        gk = smooth_max_grad_full(h_full, _cur_alpha)
        gr = gk[:n_half] - gk[n_half:][::-1]
        return gr

    global _cur_alpha
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        _cur_alpha = alpha
        try:
            res = minimize(obj_red, h_red, jac=grad_red, method='SLSQP',
                          bounds=bounds, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = make_full_from_reduced(res.x)
            h_new = project_feasible(h_new, n / 2.0)
            if compute_c5(h_new) < compute_c5(make_full_from_reduced(h_red)):
                h_red = res.x
        except Exception:
            pass
    return make_full_from_reduced(h_red)

def slsqp_full(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    h_cur = h_start.copy()
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    global _cur_n, _cur_target

    def obj_fun(h):
        return smooth_max_obj(h, _cur_alpha)

    def grad_fun(h):
        return smooth_max_grad_full(h, _cur_alpha)

    def con_fun(h):
        return np.sum(h) - _cur_target

    def con_jac(h):
        return np.ones(_cur_n)

    _cur_n = n
    _cur_target = target
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        _cur_alpha = alpha
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_fun, h_cur, jac=grad_fun, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def sine_basis_search(n, K, alphas, time_limit, seed):
    start_t = time.time()
    x = np.arange(n) * (2.0 / n)
    S = np.zeros((n, K))
    for k in range(K):
        S[:, k] = np.sin((k + 1) * np.pi * x)

    target = n / 2.0
    rng = np.random.RandomState(seed)

    def obj_with_penalty(a, alpha, pen_w):
        h = 0.5 + S @ a
        h = np.clip(h, 0.0, 1.0)
        base = smooth_max_obj(h, alpha)
        raw = 0.5 + S @ a
        penalty = pen_w * (np.sum(np.maximum(raw - 1.0, 0.0)) + np.sum(np.maximum(-raw, 0.0)))
        return base + penalty

    def grad_with_penalty(a, alpha, pen_w):
        h = 0.5 + S @ a
        gh = smooth_max_grad_full(h, alpha)
        raw = 0.5 + S @ a
        gp = pen_w * S.T @ (np.maximum(raw - 1.0, 0.0) - np.maximum(-raw, 0.0))
        return gh + gp

    best_h = None
    best_val = np.inf
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        for attempt in range(3):
            if time.time() - start_t > time_limit:
                break
            scale = [0.05, 0.15, 0.4][attempt % 3]
            a0 = rng.randn(K) * scale
            pen_w = 10.0
            try:
                res = minimize(obj_with_penalty, a0, args=(alpha, pen_w),
                              jac=grad_with_penalty, args=(alpha, pen_w),
                              method='L-BFGS-B', options={'maxiter': 300})
                h_try = 0.5 + S @ res.x
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_val:
                    best_val = sc
                    best_h = h_try.copy()
            except Exception:
                pass
    if best_h is None:
        a0 = np.zeros(K)
        best_h = project_feasible(0.5 + S @ a0, target)
        best_val = compute_c5(best_h)
    return best_h, best_val

def projected_momentum(h_start, n, target, time_limit, alpha, mu, base_lr):
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(50000):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_max_grad_full(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult / (1.0 + 0.005 * step)
            h_try = project_feasible(h - lr * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
    return best_h, best_c5

def fast_pair_cd(h_start, n, target, time_limit, n_cand, top_k, max_sweeps):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_a = 50.0
        w_top = np.exp(soft_a * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = d_lo
                b_d = d_hi
                for _ in range(12):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    active = vals >= f_m - 1e-14
                    if active.any():
                        f_p = np.max(g[active])
                    else:
                        f_p = 0.0
                    if f_p > 1e-14:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial - 1e-12:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cur_alpha, _cur_n, _cur_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 0: Sine basis to explore different shapes
    if tleft() > 100:
        h_sine, sc_sine = sine_basis_search(n, 12, [50, 200, 1000], min(60, tleft()), seed)
        if sc_sine < best_score:
            best_score = sc_sine
            best_h = h_sine

    # Phase 1: Symmetry-reduced SLSQP
    if tleft() > 120:
        n_half = n // 2
        h_red = (best_h[:n_half] + (1.0 - best_h[n_half:][::-1])) / 2.0
        h_red = np.clip(h_red, 0.0, 1.0)
        h_sym = make_full_from_reduced(h_red)
        h_sym = project_feasible(h_sym, target)
        h_pol = slsqp_reduced(h_red, n, [50, 200, 1000, 5000], 2000, 1e-16, min(100, tleft()))
        h_pol = project_feasible(h_pol, target)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Full SLSQP starting from symmetric solution
    if tleft() > 120:
        h_pol = slsqp_full(best_h, n, target, [200, 1000, 5000], 2000, 1e-16, min(100, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Projected momentum
    if tleft() > 80:
        for alpha_m in [200, 1000]:
            if tleft() < 30:
                break
            h_pm, c5_pm = projected_momentum(best_h, n, target,
                                             time_limit=min(40, tleft() * 0.4),
                                             alpha=alpha_m, mu=0.9, base_lr=0.03)
            if c5_pm < best_score:
                best_score = c5_pm
                best_h = h_pm

    # Phase 4: Pair CD
    if tleft() > 80:
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                    time_limit=min(100, tleft() * 0.4),
                                    n_cand=45, top_k=3, max_sweeps=8000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Higher resolution n=800
    if tleft() > 200:
        n2 = max(2 * n, 800)
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        n_half2 = n2 // 2
        h_red2 = (h_interp[:n_half2] + (1.0 - h_interp[n_half2:][::-1])) / 2.0
        h_red2 = np.clip(h_red2, 0.0, 1.0)
        h_sym2 = make_full_from_reduced(h_red2)
        h_sym2 = project_feasible(h_sym2, target2)
        h_pol2 = slsqp_reduced(h_red2, n2, [50, 200, 1000], 1500, 1e-15, min(80, tleft()))
        h_pol2 = project_feasible(h_pol2, target2)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            # Also try full SLSQP at higher res
            if tleft() > 80:
                h_pol2b = slsqp_full(h_pol2, n2, target2, [500, 2000], 1500, 1e-15, min(60, tleft()))
                sc2b = compute_c5(h_pol2b)
                if sc2b < sc2:
                    sc2 = sc2b
                    h_pol2 = h_pol2b
            best_score = sc2
            best_h = h_pol2
            n = n2
            target = target2

    # Phase 6: Pair CD at current resolution
    if tleft() > 80:
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                    time_limit=min(100, tleft() * 0.5),
                                    n_cand=50, top_k=4, max_sweeps=8000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Multi-start perturbation
    rng = np.random.RandomState(seed)
    if tleft() > 150:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = slsqp_full(h_pert, n, target, [200, 2000], 1000, 1e-15, min(40, tleft()))
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if tleft() > 30:
                    h_cd2, c5_cd2 = fast_pair_cd(h_ref, n, target,
                                                  time_limit=min(20, tleft() * 0.3),
                                                  n_cand=40, top_k=3, max_sweeps=3000)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        h_final = slsqp_full(best_h, n, target, [5000, 20000], 1000, 1e-16, min(20, tleft()))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 9: Final pair CD
    if tleft() > 20:
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                    time_limit=min(15, tleft() * 0.5),
                                    n_cand=50, top_k=3, max_sweeps=3000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
