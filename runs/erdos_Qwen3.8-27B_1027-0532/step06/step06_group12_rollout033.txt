def cutting_plane_lp(h_current, n, target, time_limit, max_constraints=100):
    """Cutting plane approach: add constraints one at a time"""
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    active_lags = []
    h = h_current.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Start with top constraints
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:max_constraints]
    active_lags = [(idx - n_half) for idx in top_idx]
    
    for it in range(200):
        if time.time() - start > time_limit:
            break
        
        # Build and solve LP
        try:
            h_var = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [
                cp.sum(h_var) == target,
                h_var >= 0.0,
                h_var <= 1.0
            ]
            
            for lag in active_lags:
                abs_k = abs(lag)
                if abs_k == 0:
                    coeff = (1.0 - h) * dx
                    constraints.append(coeff @ h_var <= t_var)
                elif lag > 0:
                    coeff = np.zeros(n)
                    coeff[:n-lag] = (1.0 - h[lag:]) * dx
                    constraints.append(coeff @ h_var <= t_var)
                else:
                    m = -lag
                    coeff = np.zeros(n)
                    coeff[m:] = (1.0 - h[:n-m]) * dx
                    constraints.append(coeff @ h_var <= t_var)
            
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            prob.solve(solver=cp.GLPK, verbose=False)
            
            if h_var.value is None:
                break
            
            h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            
            # Find most violated constraint
            overlaps_new = compute_overlaps(h_new)
            max_overlap_idx = np.argmax(overlaps_new)

                best_h = h.copy()

                t_var = cp.Variable()

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_pen = 1.0
_g_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _obj(h):
    n = _g_n
    alpha = _g_alpha
    pen = _g_pen
    target = _g_target
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    soft_max = np.sum(exp_vals)
    obj = (c_max + np.log(soft_max) / alpha)
    obj += pen * (np.sum(h) - target) ** 2
    return obj


def _obj_grad(h):
    n = _g_n
    alpha = _g_alpha
    pen = _g_pen
    target = _g_target
    dx = 2.0 / n
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    soft_max = np.sum(exp_vals)
    p = exp_vals / soft_max
    grad = np.zeros(n)
    significant = np.where(p > 1e-10)[0]
    for idx in significant:
        k = idx - n_half
        if k == 0:
            grad += p[idx] * alpha * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += p[idx] * alpha * (1.0 - h[:n-k]) * dx
            grad[:n-k] -= p[idx] * alpha * h[k:] * dx
        else:
            m = -k
            grad[:n-m] += p[idx] * alpha * (1.0 - h[m:]) * dx
            grad[m:] -= p[idx] * alpha * h[:n-m] * dx
    grad += 2.0 * pen * (np.sum(h) - target)
    return grad


def lbfgsb_phase(h_start, n, target, alphas, maxiter=300, penalty=1.0, gtol=1e-8):
    global _g_n, _g_target, _g_pen, _g_alpha
    _g_n = n
    _g_target = target
    _g_pen = penalty
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()

        _g_alpha = alpha
        _g_pen = penalty
        try:
            res = minimize(_obj, h_cur, jac=_obj_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': gtol, 'maxls': 50, 'maxfun': 5000})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h_cur)
            if c5_new < c5_cur:
                h_cur = h_new
        except Exception:
            pass
        h_cur = project_feasible(h_cur, target)
    return h_cur


def frank_wolfe_extrap(h_init, n, target, time_limit, max_iter=300000, K=50):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    z = h.copy()
    dx = 2.0 / n

    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]

        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            if k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[k:] += (1.0 - h[:n-k]) * dx
                g[:n-k] -= h[k:] * dx
            else:
                m = -k
                g[:n-m] += (1.0 - h[m:]) * dx
                g[m:] -= h[:n-m] * dx
        g /= len(top_idx)

        gamma = 2.0 / (t + 2)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)

        if t > 0:
            gamma_prev = 2.0 / (t + 1)
            extrap = min(1.0, (gamma_prev / gamma) - 1.0)
            z_new = h + extrap * (h - z)
            z_new = project_feasible(z_new, target)
            g_z = np.zeros(n)
            for idx in top_idx:
                k = idx - n_half
                if k == 0:
                    g_z += (1.0 - 2.0 * z_new) * dx
                elif k > 0:
                    g_z[k:] += (1.0 - z_new[:n-k]) * dx
                    g_z[:n-k] -= z_new[k:] * dx
                else:
                    m = -k
                    g_z[:n-m] += (1.0 - z_new[m:]) * dx
                    g_z[m:] -= z_new[:n-m] * dx
            g_z /= len(top_idx)
            sort_idx2 = np.argsort(g_z)
            v2 = np.zeros(n)
            v2[sort_idx2[:int(target)]] = 1.0
            h_e = (1.0 - gamma) * z_new + gamma * v2
            h_e = project_feasible(h_e, target)
            c5_e = compute_c5(h_e)
            c5_n = compute_c5(h_new)
            if c5_e < c5_n:
                h_new = h_e
        z = h
        h = h_new

        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            best_t = t
        if t > 15000 and t - best_t > 60000:
            break
    return best_h, best_c5


def cutting_plane_lp(h_current, n, target, time_limit, max_constraints=100, damping=0.5):

def perturb_block(h, n, target, rng, block_len, n_perturb):
    h_new = h.copy()
    for _ in range(n_perturb):
        start = rng.randint(0, n - block_len)
        h_new[start:start + block_len] = 1.0 - h_new[start:start + block_len]
    return project_feasible(h_new, target)
