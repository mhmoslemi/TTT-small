def lsq_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_w = np.exp(shifted)
    w_sum = float(np.sum(exp_w))
    w = exp_w / w_sum
    F = c_max + np.log(w_sum) / alpha
    grad = np.full(n, 1.0)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws < 1e-14:

        if s >= 0:
            grad[:n - s] -= ws * h[s:]
            grad[s:] -= ws * h[:n - s]
        else:
            a = -s
            grad[a:] -= ws * h[:n - a]
            grad[:n - a] -= ws * h[a:]

    need = target_int - curr
    if need > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_cont[zeros])[:need]
        h_bin[zeros[order]] = 1.0
    elif need < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_cont[ones])[:(-need)]
        h_bin[ones[order]] = 0.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    h0 = project_feasible(h_init, target)
    best_h = h0.copy()
    best_score = compute_c5(h0)

    def _remaining():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: Binary greedy + SA
    if _remaining() > 50:
        h_bin = binarize(h0, int(target))
        allow = min(25.0, _remaining() * 0.06)
        h_g, sc_g = binary_greedy(h_bin, 30000, allow, start_time)
        if sc_g < best_score:
            best_score = sc_g
            best_h = h_g.copy()

    if _remaining() > 80:
        allow = min(120.0, _remaining() * 0.2)
        n_sa = int(min(4000000, allow * 55000))
        h_sa, sc_sa = binary_sa(h_g if 'h_g' in dir() else h_bin, n_sa, 0.012, 0.00003, rng, allow, start_time)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()

    # Phase 2: Continuous optimization with log-sum-exp (coarse to fine)
    if _remaining() > 100:
        allow = min(_remaining() * 0.35, 350.0)
        schedule = [(15, 20000), (30, 15000), (60, 12000), (120, 10000), (250, 8000)]
        h_c1, sc_c1 = nesterov_opt(h0, target, schedule, 0.85, 3.0, allow, start_time)
        if sc_c1 < best_score:
            best_score = sc_c1
            best_h = h_c1.copy()

    # Phase 3: Fine continuous optimization
    if _remaining() > 80:
        allow = min(_remaining() * 0.25, 250.0)
        schedule2 = [(300, 10000), (500, 8000), (800, 6000), (1500, 5000)]
        h_c2, sc_c2 = nesterov_opt(best_h, target, schedule2, 0.9, 2.0, allow, start_time)
        if sc_c2 < best_score:
            best_score = sc_c2
            best_h = h_c2.copy()

    # Phase 4: Max-subgradient polishing
    if _remaining() > 60:
        allow = min(_remaining() * 0.2, 150.0)
        n_pg = int(min(80000, allow * 1200))
        h_p, sc_p = max_subgrad_opt(best_h, target, n_pg, 0.9, 0.002, allow, start_time)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p.copy()

    # Phase 5: Multi-resolution
    for n_try in [n + 50, n + 100, n - 50]:
        if _remaining() < 60:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_i = interp_to(best_h, n_try, target_try)
        allow = min(45.0, _remaining() * 0.15)
        if allow < 20:
            continue
        sched_try = [(50, 8000), (100, 6000), (200, 5000), (500, 4000)]
        h_t, sc_t = nesterov_opt(h_i, target_try, sched_try, 0.88, 2.5, allow, start_time)
        if sc_t < best_score:
            best_score = sc_t
            best_h = h_t.copy()

    # Phase 6: Final polish on best solution
    if _remaining() > 30:
        n_f = len(best_h)
        t_f = n_f / 2.0
        allow = min(_remaining() - 10, 60.0)
        n_fin = int(min(50000, allow * 1500))
        h_fin, sc_fin = max_subgrad_opt(best_h, t_f, n_fin, 0.92, 0.001, allow, start_time)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_final
