    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h

    h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def pair_coord_descent(h, n, dx, n_sweeps=500, seed=42, time_limit=None,
                       pair_mode='antipodal'):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    start_time = time.time()
    k_arr = np.arange(-(n - 1), n)

    for sweep in range(n_sweeps):
        if time_limit is not None and time.time() - start_time > time_limit:
            break

        if pair_mode == 'antipodal':
            pairs = [(i, (i + n // 2) % n) for i in range(n // 2)]
        elif pair_mode == 'shifted':
            shift = (sweep * 7 + 1) % n
            pairs = [(i, (i + shift) % n) for i in range(n // 2)]
            pairs = [(a, b) for a, b in pairs if a != b]
        else:
            idx = np.arange(n)
            rng.shuffle(idx)
            pairs = [(idx[i], idx[i + n // 2]) for i in range(n // 2)]

        overlaps = compute_overlaps(h)

        for (i, j) in pairs:
            if i == j:
                continue
            s = h[i] + h[j]
            lo_d = max(-s / 2.0, s / 2.0 - 1.0)
            hi_d = min(1.0 - s / 2.0, s / 2.0)
            if hi_d - lo_d < 1e-14:
                continue

            a = dx * ((1.0 - h[(i + k_arr) % n]) + h[(i - k_arr) % n]
                      - (1.0 - h[(j + k_arr) % n]) - h[(j - k_arr) % n])

            lo = lo_d
            hi = hi_d
            for _ in range(50):
                mid = (lo + hi) / 2.0
                val = overlaps + mid * a
                mx = np.max(val)
                active = np.where(val >= mx - 1e-12)[0]
                min_a = a[active].min()
                max_a = a[active].max()
                if min_a >= -1e-13:
                    hi = mid
                elif max_a <= 1e-13:
                    lo = mid
                else:
                    break

            d_opt = (lo + hi) / 2.0
            h[i] = s / 2.0 + d_opt
            h[j] = s / 2.0 - d_opt
            overlaps += d_opt * a

        c5_val = float(np.max(overlaps))
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def block_coord_descent(h, n, dx, n_sweeps=200, block_size=4, seed=42,
                        time_limit=None):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    start_time = time.time()
    k_arr = np.arange(-(n - 1), n)

    def block_obj(vars_block, h_full, idx_block):
        h_try = h_full.copy()
        h_try[idx_block] = vars_block
        ovl = compute_overlaps(h_try)
        return float(np.max(ovl))

    for sweep in range(n_sweeps):
        if time_limit is not None and time.time() - start_time > time_limit:
            break

        if rng.rand() < 0.5:
            start = rng.randint(0, n - block_size)
            idx_block = np.arange(start, start + block_size)
        else:
            idx_block = np.sort(rng.choice(n, size=block_size, replace=False))

        s_block = h[idx_block].sum()
        bounds = [(0.0, 1.0)] * block_size

        def obj(v):
            h_try = h.copy()
            h_try[idx_block] = v
            h_try = np.clip(h_try, 0, 1)
            ovl = compute_overlaps(h_try)
            return float(np.max(ovl))

        def cons_eq(v):
            return v.sum() - s_block

        try:
            res = minimize(obj, h[idx_block].copy(), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_eq}],
                           options={'maxiter': 200, 'ftol': 1e-15})
            if res.success or (res.fun < obj(h[idx_block])):
                h_new_block = np.clip(res.x, 0, 1)
                diff = s_block - h_new_block.sum()
                if abs(diff) > 1e-10:
                    h_new_block += diff / block_size
                    h_new_block = np.clip(h_new_block, 0, 1)
                c5_new = compute_c5(np.where(np.isin(np.arange(n), idx_block),
                                             np.concatenate([h[:idx_block[0]],
                                                            h_new_block,
                                                            h[idx_block[-1] + 1:]
                                                            ] if len(idx_block) > 1 else h),
                                             h))
                h[idx_block] = h_new_block
                c5_val = compute_c5(h)
                if c5_val < best_c5 - 1e-13:
                    best_c5 = c5_val
                    best_h = h.copy()
        except Exception:
            pass

    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def frank_wolfe(h_init, n, target, time_limit, max_iter=400000, K=40):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    k_arr = np.arange(-(n - 1), n)

    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            if k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n - k] += (1.0 - h[k:]) * dx
                g[k:] += (-h[:n - k]) * dx
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m]) * dx
                g[:n - m] += (-h[m:]) * dx
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 20000 and t - best_t > 150000:
            break
    return best_h, best_c5


def build_lp_linear(h_fixed, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    n_half = n - 1
    for k in lags:
        if k == 0:
            coeff = (1.0 - h_fixed) * dx
            constraints.append(coeff @ h_var <= t_var)
        elif k > 0:
            coeff = np.zeros(n)
            coeff[:n - k] = (1.0 - h_fixed[k:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        else:
            m = -k
            coeff = np.zeros(n)
            coeff[m:] = (1.0 - h_fixed[:n - m]) * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=300,
                   damping_start=0.9, damping_end=0.8,
                   reg_start=1e-3, reg_end=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 100 and stuck > 200:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        progress = min(1.0, it / max(1, int(max_iter * 0.3)))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        h_var, t_var, constraints = build_lp_linear(h, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def annealing_descent(h, n, dx, time_limit, seed=42, n_iterations=100000):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    T_start = best_c5 * 0.01
    T_end = best_c5 * 0.0001
    k_arr = np.arange(-(n - 1), n)
    n_half = n - 1

    for it in range(n_iterations):
        if time.time() - start_time > time_limit:
            break
        progress = it / float(n_iterations)
        T = T_start * (T_end / T_start) ** progress

        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue

        s = h[i] + h[j]
        lo_d = max(-s / 2.0, s / 2.0 - 1.0)
        hi_d = min(1.0 - s / 2.0, s / 2.0)
        if hi_d - lo_d < 1e-14:

        a = dx * ((1.0 - h[(i + k_arr) % n]) + h[(i - k_arr) % n]
                  - (1.0 - h[(j + k_arr) % n]) - h[(j - k_arr) % n])

        d_opt = (lo_d + hi_d) / 2.0
        ov_lap = compute_overlaps(h)
        new_max = float(np.max(ov_lap + d_opt * a))
        delta = new_max - best_c5

            if new_max < best_c5 - 1e-13:

                best_h = h.copy()

                                 [1000, 10000, 100000, 1000000],
                                 maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_slq3)
        if sc < best_score:
            best_score = sc
            best_h = h_slq3

    # Phase 11: Pair CD final
    tl = time_remaining(start_time, budget_s)
    if tl > 50:
        cd_time = min(60, tl * 0.15)
        h_cd6, c5_cd6 = pair_coord_descent(best_h, n, dx, n_sweeps=3000,
                                           seed=seed + 6, time_limit=cd_time,
                                           pair_mode='random')
        if c5_cd6 < best_score:
            best_score = c5_cd6
            best_h = h_cd6

    # Phase 12: Annealing
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        an_time = min(60, tl * 0.15)
        h_an, c5_an = annealing_descent(best_h, n, dx, time_limit=an_time,
                                        seed=seed + 7, n_iterations=50000)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 13: Final SLSQP
    tl = time_remaining(start_time, budget_s)
    if tl > 50:
        h_final = slsqp_optimize(best_h, n, target,
                                  [5000, 50000, 500000, 5000000],
                                  maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 14: Final pair CD
    tl = time_remaining(start_time, budget_s)
    if tl > 30:
        cd_time = min(30, tl * 0.8)
        h_cd7, c5_cd7 = pair_coord_descent(best_h, n, dx, n_sweeps=5000,
                                           seed=seed + 8, time_limit=cd_time,
                                           pair_mode='antipodal')
        if c5_cd7 < best_score:
            best_score = c5_cd7
            best_h = h_cd7

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
