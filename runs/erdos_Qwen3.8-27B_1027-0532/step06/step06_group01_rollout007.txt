import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def multi_shift_gradient(h, n, top_m=5, alpha=50.0):
    """Compute gradient of smoothed max using top-m shifts with softmax weights."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    
    # Get top-m shifts
    top_idx = np.argsort(corr)[-top_m:]
    logits = corr[top_idx] - cur_max
    weights = np.exp(alpha * logits)
    weights = weights / weights.sum()
    
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for w, si in zip(weights, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w * h[idxs[v2] + k]
    
    return grad * dx, cur_max * dx

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

_SLQ_TARGET = 0.0

def improved_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000, top_m=3):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        # Use multi-shift gradient for better direction
        top_idx = np.argsort(corr)[-top_m:]
        alpha_w = 20.0
        logits = corr[top_idx] - cur_max
        weights = np.exp(alpha_w * logits)
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, si in zip(weights, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * h[idxs[v2] + k]

        # Select candidates
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        # Find best pair move
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_method(h_init, n, target, time_limit, max_steps=100000, top_m=5, base_lr=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break

        top_idx = np.argsort(corr)[-top_m:]
        alpha_w = 30.0
        logits = corr[top_idx] - cur_max
        weights = np.exp(alpha_w * logits)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, si in zip(weights, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        # Try several step sizes
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            h_try = h - lr * lr_mult * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            lr *= 0.8
            if lr < 1e-7:
                break

    return best_h, best_c5

def random_restart_optimize(h_best, n, target, time_total, n_restarts=5):
    """Try random perturbations of the best solution."""
    start = time.time()
    per_restart = time_total / n_restarts
    best = h_best.copy()
    best_score = compute_c5(h_best)
    
    for r in range(n_restarts):
        if time.time() - start > time_total - 10:
            break
        
        # Create perturbation
        h_pert = h_best.copy()
        n_perms = n // 10
        perm_idx = np.random.choice(n, size=n_perms, replace=False)
        h_pert[perm_idx] += np.random.uniform(-0.05, 0.05, size=n_perms)
        h_pert = project_feasible(h_pert, target)
        
        # Optimize the perturbed solution
        hp, sc = improved_cd(h_pert, n, target, time_limit=per_restart - 20, n_cand=20, max_sweeps=20000, top_m=3)
        if sc < best_score:
            best_score = sc
            best = hp
    
    return best, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

_SLQ_TARGET = 0.0

def _slq_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slq_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def binary_anneal(n, time_limit, n_ones=None):
    """Simulated annealing for binary h with exactly n_ones ones."""
    if n_ones is None:
        n_ones = n // 2
    
    start = time.time()
    # Start random
    h = np.zeros(n)
    h[:n_ones] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = 0.1  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    while time.time() - start < time_limit and T > T_min:
        # Pick two random positions to swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - best_c5  # positive means worse
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    return best_h, best_c5

For binary h, corr[l] = sum_m h[m] * (1-h[m+l-(n-1)])

This counts the number of pairs (m, m+l-(n-1)) where h[m]=1 and h[m+l-(n-1)]=0.

If we flip h[i] from 1 to 0:
- Pairs where m=i and h[i]=1: these no longer contribute. The number lost is (1-h[i+l-(n-1)]) for valid l, which is 1 if h[i+l-(n-1)]=0.

def _slq_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):

def slsqp_opt(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slq_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new

    n_corr = 2 * n - 1
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:

        cur_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:

        top_idx = np.argsort(corr)[-top_m:]
        logits = corr[top_idx] - cur_max

def subgradient_descent(h_init, n, target, time_limit, max_steps=50000, top_m=5, base_lr=0.015, alpha_w=30.0):

    idxs = np.arange(n)

    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break

        top_idx = np.argsort(corr)[-top_m:]
        logits = corr[top_idx] - cur_max
        weights = np.exp(alpha_w * logits)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, si in zip(weights, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = h - lr * lr_mult * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            lr *= 0.9
            if lr < 1e-8:
                break

    return best_h, best_c5
