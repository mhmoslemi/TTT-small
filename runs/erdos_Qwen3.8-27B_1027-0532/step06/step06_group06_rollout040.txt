    n_corr = 2*n - 1
    k = np.arange(n_corr) - (n-1)  # shape (n_corr,)

    i_idx = np.arange(n)  # shape (n,)

    idx_plus = i_idx[:, None] + k[None, :]  # (n, n_corr)
    idx_minus = i_idx[:, None] - k[None, :]  # (n, n_corr)
    
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    valid_minus = (idx_minus >= 0) & (idx_minus < n)

    # We need to gather h at these indices

After a swap (i: 1→0, j: 0→1):
- h[i] changes from 1 to 0
- h[j] changes from 0 to 1

The delta matrix fd[m, t] = -(1-h[m+k[t]]) * valid + h[m-k[t]] * valid_m

The entries that change are those where m+k[t] = i or m+k[t] = j or m-k[t] = i or m-k[t] = j.

For m+k[t] = i: k[t] = i-m, so t = i-m+(n-1). This gives at most n values of t for each m, or equivalently, for a fixed t, m = i-k[t]. So for each t, exactly one m has m+k[t] = i (if valid).

Similarly for the other conditions.

- fd[m, t] where m+k[t] = i: change by -(1-0) - (-(1-1)) = -1 - 0 = -1
  Wait, fd[m,t] = -(1-h[m+k[t]]) + h[m-k[t]] (for valid indices)

  Change: +1

Similarly for fu (which is -fd for the "flip" direction... wait no, fu is different).

fu[m, t] = (1-h[m+k[t]]) - h[m-k[t]] (for valid indices)

  Before: h[i]=1, so term is (1-1) = 0

  Before: h[j]=0, so term is -h[j] = 0
  After: h[j]=1, so term is -h[j] = -1
  Change: -1

So the updates are:
- fd: -1 where m+k[t]=i or m-k[t]=i; +1 where m+k[t]=j or m-k[t]=j
- fu: +1 where m+k[t]=i or m-k[t]=i; -1 where m+k[t]=j or m-k[t]=j

For a given position p (either i or j), the affected entries are:
- m+k[t] = p: for each t, m = p - k[t]. Valid if 0 <= m < n.
  k[t] = t - (n-1), so m = p - (t-(n-1)) = p - t + n - 1
  Valid if 0 <= p - t + n - 1 < n, i.e., n-1-p <= t <= n-1+p

def project_box_sum(h, target):

def project_binary(h, target_sum):

    current_sum = int(h_r.sum())

        return h_r.astype(np.float64)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

def compute_fd_fu(h, n):
    nc = 2 * n - 1
    k = np.arange(nc) - (n - 1)
    ii = np.arange(n)
    idx_p = ii[:, None] + k[None, :]
    idx_m = ii[:, None] - k[None, :]
    vp = (idx_p >= 0) & (idx_p < n)
    vm = (idx_m >= 0) & (idx_m < n)
    hp = np.where(vp, h[np.clip(idx_p, 0, n-1)], 0.0)
    hm = np.where(vm, h[np.clip(idx_m, 0, n-1)], 0.0)
    fd = -np.where(vp, 1.0 - hp, 0.0) + np.where(vm, hm, 0.0)
    fu = np.where(vp, 1.0 - hp, 0.0) - np.where(vm, hm, 0.0)
    return fd, fu

        h_interp = interpolate_to(best_h, n_try, target_try)
        tl = min(25, tleft() * 0.1)
        h_opt = lbfgs_run(h_interp, 100, target_try, 500.0, 150, tl, start_time)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()
        if tleft() < 30:
            continue
        h_b = project_binary(h_opt, target_try_int)
        tl2 = min(30, tleft() * 0.12)
        h_ls, c5_ls = binary_local_search(h_b, target_try_int, tl2, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 8: Final SA polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_b = project_binary(best_h, target_use)
        tl = min(50, max(tleft() - 10, 10))
        max_it = int(min(200000, tl * 3000))
        h_sf, c5_sf = binary_sa(h_b, target_use, max_it, 0.003, 0.00001, rng, tl, start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 9: Final local search
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_b = project_binary(best_h, target_use)
        tl = min(30, max(tleft() - 10, 5))
        h_ls, c5_ls = binary_local_search(h_b, target_use, tl, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
