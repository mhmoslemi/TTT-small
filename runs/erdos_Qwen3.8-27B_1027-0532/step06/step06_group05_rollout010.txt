def lbfgs_smooth(h0, target, alpha, penalty, time_limit, start_time, maxiter=10000):
    n = len(h0)
    
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        val = float(np.log(np.sum(np.exp(alpha * (corr - corr.max()))))) / alpha + float(np.max(corr))
        if time.time() - start_time > time_limit:
            return val  # signal termination
        penalty_term = penalty * (float(np.sum(x)) - target)**2
        return val + penalty_term
    
    def grad(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha*(corr - c_max), -500, 0)
        exp_vals = np.exp(shifted)
        w = exp_vals / float(np.sum(exp_vals))
        
        g = np.zeros(n)
        omh = 1.0 - x
        for s in range(n):
            wi = w[n-1+s]
            if wi < 1e-14: continue
            if s == 0:
                g[:] += wi * omh
            else:
                g[:n-s] += wi * omh[s:]
                g[s:] -= wi * x[:n-s]
        for p in range(1, n):
            wi = w[n-1-p]
            if wi < 1e-14: continue
            g[p:] += wi * omh[:n-p]
            g[:n-p] -= wi * x[p:]
        
        # Add penalty gradient
        g += 2 * penalty * (float(np.sum(x)) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                               'maxfun': maxiter*2})
    h_new = np.clip(result.x, 0, 1)
    # Project to exact sum
    h_new = project_box_sum(h_new, target)
    return h_new, compute_c5(h_new)

                c = np.correlate(h_bin, 1 - h_bin, mode='full')
                new_max = float(c.max())
                if new_max < max_val - 1e-12:
                    corr = c
                    max_val = new_max
                    best_c5 = max_val * dx
                    best_h = h_bin.copy()
                    found = True
                    break
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
            if found:
                break
        if found:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def multi_start_binary(rng, n, target_sum, n_starts, time_limit, start_time):
    best_h = None
    best_score = float('inf')
    for _ in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        h_rand = np.zeros(n)
        idx = rng.choice(n, size=target_sum, replace=False)
        h_rand[idx] = 1.0
        h_bin, sc = binary_local_search(h_rand, target_sum,
                                        time_limit=5.0, start_time=start_time,
                                        max_stall=50)
        if sc < best_score:
            best_score = sc
            best_h = h_bin
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _lbfgs_n, _lbfgs_target, _lbfgs_alpha, _lbfgs_penalty_val
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing alpha and penalty
    if tleft() > 80:
        phases = [
            (50, 10.0, 8000),
            (200, 50.0, 8000),
            (1000, 200.0, 8000),
            (5000, 1000.0, 8000),
            (20000, 5000.0, 8000),
            (100000, 20000.0, 8000),
            (500000, 100000.0, 8000),
        ]
        for alpha, pen, miter in phases:
            if tleft() < 40:
                break
            h_new, sc = lbfgs_optimize(best_h, target, alpha, pen, miter,
                                       time_limit=max(5.0, tleft() * 0.12),
                                       start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Fine Adam with very small learning rates
    if tleft() > 120:
        h_new, sc = adam_optimize(
            best_h, target,
            alphas=[5000, 20000, 100000, 500000, 2000000],
            lrs=[0.001, 0.0003, 0.0001, 0.00003, 0.00001],
            time_limit=tleft() * 0.25, start_time=start_time,
            stall_limit=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: L-BFGS-B again from current best (with high alpha)
    if tleft() > 80:
        for alpha, pen in [(200000, 50000), (1000000, 200000), (5000000, 1000000)]:
            if tleft() < 30:
                break
            h_new, sc = lbfgs_optimize(best_h, target, alpha, pen, 5000,
                                       time_limit=max(5.0, tleft() * 0.12),
                                       start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Binary local search from best continuous solution
    if tleft() > 60:
        h_bin, sc_bin = binary_local_search(best_h, int(target),
                                            time_limit=tleft() * 0.15,
                                            start_time=start_time, max_stall=150)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        # Refine binary in continuous space
        if tleft() > 50:
            h_new, sc = adam_optimize(h_bin, target,
                                      alphas=[10000, 50000, 200000, 1000000],
                                      lrs=[0.001, 0.0003, 0.0001, 0.00003],
                                      time_limit=tleft() * 0.2, start_time=start_time,
                                      stall_limit=1500)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            if tleft() > 30:
                h_new2, sc2 = lbfgs_optimize(best_h, target, 200000, 50000, 5000,
                                             time_limit=max(5.0, tleft() * 0.15),
                                             start_time=start_time)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new2

    # Phase 5: Multi-restart from perturbed best
    if tleft() > 100:
        for r in range(3):
            if tleft() < 50:
                break
            jitter = 0.003 * (r + 1)
            h_pert = best_h + rng.randn(n) * jitter
            h_pert = project_box_sum(h_pert, target)
            h_new, sc = adam_optimize(
                h_pert, target,
                alphas=[10000, 100000, 1000000],
                lrs=[0.0005, 0.0001, 0.00001],
                time_limit=max(10.0, tleft() * 0.15), start_time=start_time,
                stall_limit=2000)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            if tleft() > 30:
                h_new2, sc2 = lbfgs_optimize(best_h, target, 500000, 100000, 5000,
                                             time_limit=max(5.0, tleft() * 0.1),
                                             start_time=start_time)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new2

    # Phase 6: Ultra-fine final polish
    if tleft() > 20:
        h_new, sc = adam_optimize(
            best_h, target,
            alphas=[1000000, 5000000, 20000000],
            lrs=[0.000005, 0.000002, 0.000001],
            time_limit=max(5.0, tleft() * 0.5), start_time=start_time,
            stall_limit=5000)
        if sc < best_score:
            best_score = sc
            best_h = h_new

        if tleft() > 10:
            h_new2, sc2 = lbfgs_optimize(best_h, target, 5000000, 500000, 3000,
                                         time_limit=max(3.0, tleft() * 0.3),
                                         start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_new2

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
