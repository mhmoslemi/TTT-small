Actually, let me also think about the "swap" move more carefully.

For a swap (i, j, δ): h[i] += δ, h[j] -= δ, all others unchanged.
The change in corr[k_idx] is:
Δcorr[k_idx] = δ * [(1-h[i+k]) - h[j+k]] * dx - δ * [(1-h[i-k]) - h[j-k]] * dx (for valid indices)
= δ * [(1-h[i+k]) - h[j+k] - (1-h[i-k]) + h[j-k]] * dx

This is linear in δ. So the 1D problem is the same as before but with slopes computed for the pair.

Actually, it's simpler to just think of it as: after the swap, the new corr[k_idx] = old_corr[k_idx] + δ * B[k_idx] where B[k_idx] is the coefficient. Then we minimize max_k (corr[k_idx] + δ * B[k_idx]) * dx over δ such that h[i]+δ ∈ [0,1] and h[j]-δ ∈ [0,1].

This is again a 1D piecewise linear minimization, solvable by the same bisection method.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max(slopes, intercepts, lo_bound=0.0, hi_bound=1.0):
    """Solve min_x max_j (slopes[j]*x + intercepts[j]) subject to lo_bound <= x <= hi_bound."""
    if len(slopes) == 0:
        return lo_bound
    if np.all(slopes > 1e-14):
        return lo_bound
    if np.all(slopes < -1e-14):
        return hi_bound
    lo = lo_bound
    hi = hi_bound
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(slopes), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(slopes[active] > 1e-14):
            hi = mid
        elif np.all(slopes[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent_fast(h_init, target, max_passes, time_limit, rng, n_max_active=150, margin_init=0.03):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.1
    margin = margin_init
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = float(np.max(corr)) * dx
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        threshold = c5_curr - margin
        active_mask = corr * dx > threshold
        n_active_count = np.sum(active_mask)
        if n_active_count < 10:
            active_k = np.argsort(corr)[-max(10, n_max_active // 10):]
        elif n_active_count > n_max_active:
            active_k = np.argsort(corr * active_mask)[-n_max_active:]
        else:
            active_k = np.where(active_mask)[0]
        
        ks = active_k - half_n
        corr_active = corr[active_k]
        n_active = len(ks)
        
        order = rng.permutation(n)
        for i in order:
            valid_plus = (i + ks >= 0) & (i + ks < n)
            valid_minus = (i - ks >= 0) & (i - ks < n)
            A = np.zeros(n_active)
            if np.any(valid_plus):
                A[valid_plus] += (1.0 - h[i + ks[valid_plus]]) * dx
            if np.any(valid_minus):
                A[valid_minus] -= h[i - ks[valid_minus]] * dx
            h[i] = solve_1d_max(A + lam * np.ones(n_active), corr_active - A * h[i])
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -3.0, 3.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        c5_curr = c5_val
        margin = max(0.002, margin * 0.997)
        
        if p % 500 == 0:
            lam_rate = max(0.01, lam_rate * 0.8)
    
    return best_h, best_c5

def swap_refinement(h_init, target, n_pairs, time_limit, rng):
    """Try pairwise swaps to improve the solution."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-200:]
    ks = active_k - half_n
    corr_active = corr[active_k]
    n_active = len(ks)
    
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        for _ in range(n_pairs):
            if time.time() - start > time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # For swap: h[i] += delta, h[j] -= delta
            # delta ∈ [-h[i], 1-h[j]]
            lo_d = -h[i]
            hi_d = 1.0 - h[j]
            if lo_d >= hi_d:
                continue
            
            # Compute B[k_idx] for each active k
            # Δcorr[k_idx] = delta * B[k_idx]
            # B[k_idx] = [(1-h[i+k]) - h[j+k]]*dx (if valid) - [(1-h[i-k]) - h[j-k]]*dx (if valid)
            k = ks
            B = np.zeros(n_active)
            valid_ip = (i + k >= 0) & (i + k < n)
            valid_jp = (j + k >= 0) & (j + k < n)
            valid_im = (i - k >= 0) & (i - k < n)
            valid_jm = (j - k >= 0) & (j - k < n)
            # (1-h[i+k]) term
            B[valid_ip] += (1.0 - h[i + k[valid_ip]]) * dx
            # -h[j+k] term
            B[valid_jp] -= h[j + k[valid_jp]] * dx
            # -(1-h[i-k]) term
            B[valid_im] -= (1.0 - h[i - k[valid_im]]) * dx
            # +h[j-k] term
            B[valid_jm] += h[j - k[valid_jm]] * dx
            
            # Minimize max_k (corr_active[k] + delta * B[k]) over delta in [lo_d, hi_d]
            opt_delta = solve_1d_max(B, corr_active, lo_d, hi_d)
            if abs(opt_delta) > 1e-10:
                h[i] += opt_delta
                h[j] -= opt_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    best_h = h.copy()
                    improved = True
                corr_active = corr[active_k]
    
    return best_h, best_c5

def gradient_polish(h_init, target, time_limit, alpha=5000):
    """Use smooth max approximation with projected gradient descent."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(500):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        # Smooth approximation of max
        shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Gradient of the smooth max
        grad = np.zeros(n)
        idxs = np.arange(n)
        for w_idx, w in enumerate(weights):
            if w < 1e-6:
                continue
            k = w_idx - (n - 1)
            valid1 = (0 <= idxs - k) & ((idxs - k) < n)
            valid2 = (0 <= idxs + k) & ((idxs + k) < n)
            grad[valid1] += w * (1.0 - h[idxs[valid1] - k]) * dx
            grad[valid2] -= w * h[idxs[valid2] + k] * dx
        
        # Projected gradient step
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 5e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())
        
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.98, damping + 0.03)
        if it % 60 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)
    
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    """SLSQP with smooth max approximation."""
    global_n = n
    global_target = target
    bounds = [(0.0, 1.0)] * n
    
    def _obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def _cons_fun(h):
        return np.array([np.sum(h) - global_target])
    
    def _cons_jac(h):
        return np.array([np.ones(global_n)])
    
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda h: _obj(h, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.1
    margin = margin_init

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_curr = float(np.max(corr)) * dx

    for p in range(max_passes):
        if time.time() - start_t > time_limit:
            break

        threshold = c5_curr - margin
        active_mask = corr * dx > threshold
        n_act = int(np.sum(active_mask))
        if n_act < 10:
            active_k = np.argsort(corr)[-max(10, n_max_active // 5):]
        elif n_act > n_max_active:
            idxs = np.where(active_mask)[0]
            active_k = idxs[np.argsort(corr[idxs])[-n_max_active:]]
        else:
            active_k = np.where(active_mask)[0]

        ks = active_k - half_n
        corr_active = corr[active_k]
        n_active = len(ks)
        ones_nact = np.ones(n_active)

        order = rng.permutation(n)
        for i in order:
            valid_plus = (i + ks >= 0) & (i + ks < n)
            valid_minus = (i - ks >= 0) & (i - ks < n)
            A = np.zeros(n_active)
            if np.any(valid_plus):
                A[valid_plus] += (1.0 - h[i + ks[valid_plus]]) * dx
            if np.any(valid_minus):
                A[valid_minus] -= h[i - ks[valid_minus]] * dx
            h[i] = solve_1d_max(A + lam * ones_nact, corr_active - A * h[i])

        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -3.0, 3.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        c5_curr = c5_val
        margin = max(0.001, margin * 0.996)

        if p % 500 == 0:
            lam_rate = max(0.005, lam_rate * 0.8)

    return best_h, best_c5


def swap_refine(h_init, target, n_pairs_per_iter, max_iters, time_limit, rng, n_active_set=300):
    start_t = time.time()
    n = len(h_init)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active_set:]
    ks = active_k - half_n
    corr_active = corr[active_k]
    n_active = len(ks)

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        improved_any = False
        for _ in range(n_pairs_per_iter):
            if time.time() - start_t > time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            lo_d = -h[i]
            hi_d = 1.0 - h[j]
            if lo_d >= hi_d - 1e-12:
                continue
            k = ks
            B = np.zeros(n_active)
            valid_ip = (i + k >= 0) & (i + k < n)
            valid_jp = (j + k >= 0) & (j + k < n)
            valid_im = (i - k >= 0) & (i - k < n)
            valid_jm = (j - k >= 0) & (j - k < n)
            if np.any(valid_ip):
                B[valid_ip] += (1.0 - h[i + k[valid_ip]]) * dx
            if np.any(valid_jp):
                B[valid_jp] -= h[j + k[valid_jp]] * dx
            if np.any(valid_im):
                B[valid_im] -= (1.0 - h[i - k[valid_im]]) * dx
            if np.any(valid_jm):
                B[valid_jm] += h[j - k[valid_jm]] * dx
            opt_delta = solve_1d_max(B, corr_active, lo_d, hi_d)
            if abs(opt_delta) > 1e-10:
                h[i] += opt_delta
                h[j] -= opt_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    best_h = h.copy()
                    improved_any = True
                corr_active = corr[active_k]
        if not improved_any and it > 5:
            break
    return best_h, best_c5


def smooth_obj_fn(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_eq_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_eq_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj_fn, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def gradient_polish(h_init, target, time_limit, alpha=3000):
    start_t = time.time()
    n = len(h_init)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for step in range(300):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            break
        weights /= wsum

        grad = np.zeros(n)
        for wi in range(len(weights)):
            if weights[wi] < 1e-5:
                continue
            k = wi - half_n
            valid1 = (idxs - k >= 0) & (idxs - k < n)
            valid2 = (idxs + k >= 0) & (idxs + k < n)
            grad[valid1] += weights[wi] * (1.0 - h[idxs[valid1] - k]) * dx
            grad[valid2] -= weights[wi] * h[idxs[valid2] + k] * dx

        for lr in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11):
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 5e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.98, damping + 0.03)
        if it % 60 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)

    return best_h, best_c5


def sharpen(h, strength=0.3):
    h_new = h + strength * (h - h * h) * np.sign(h - 0.5)
    return np.clip(h_new, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init.copy(), target0)
    best_score = compute_c5(best_h)
    best_n = n0

    # Phase 1: Fast CD on n0
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = coord_descent_fast(best_h, target0, 50000, cd_time, rng,
                                         n_max_active=200, margin_init=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Sharpening + CD restart
    if tleft() > 150:
        h_sharp = project_feasible(sharpen(best_h, 0.4), target0)
        cd_time = min(100, tleft() * 0.2)
        h_cd, c5_cd = coord_descent_fast(h_sharp, target0, 30000, cd_time, rng,
                                         n_max_active=200, margin_init=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = target0 * 2

    # Phase 3: Try n=300
    if tleft() > 200:
        n_try = 300
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        cd_time = min(120, tleft() * 0.25)
        h_cd, c5_cd = coord_descent_fast(h_interp, target_try, 30000, cd_time, rng,
                                         n_max_active=150, margin_init=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = n_try
        elif sc_int < best_score:
            best_score = sc_int
            best_h = h_interp
            best_n = n_try

    # Phase 4: Try n=500
    if tleft() > 250:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        cd_time = min(150, tleft() * 0.25)
        h_cd, c5_cd = coord_descent_fast(h_interp, target_try, 40000, cd_time, rng,
                                         n_max_active=200, margin_init=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = n_try
        elif sc_int < best_score:
            best_score = sc_int
            best_h = h_interp
            best_n = n_try

    # Phase 5: Try n=600
    if tleft() > 250:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        cd_time = min(150, tleft() * 0.25)
        h_cd, c5_cd = coord_descent_fast(h_interp, target_try, 40000, cd_time, rng,
                                         n_max_active=200, margin_init=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = n_try
        elif sc_int < best_score:
            best_score = sc_int
            best_h = h_interp
            best_n = n_try

    # Phase 6: LP polish on best solution
    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(180, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n_use, target_use,
                                     max_iter=5000, time_limit=lp_time,
                                     d0=0.85, reg0=1e-4, n_active=80, scs_eps=1e-11)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Swap refinement
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        swap_time = min(80, tleft() * 0.2)
        h_swap, c5_swap = swap_refine(best_h, target_use, 2000, 50, swap_time,
                                      rng, n_active_set=200)
        if c5_swap < best_score:
            best_score = c5_swap
            best_h = h_swap

    # Phase 8: Another CD round with tighter margin
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(100, tleft() * 0.2)
        h_cd, c5_cd = coord_descent_fast(best_h, target_use, 50000, cd_time, rng,
                                         n_max_active=250, margin_init=0.01)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: SLSQP polish
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(best_h, n_use, target_use, [2000, 10000, 50000], 1000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Gradient polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        gp_time = min(40, tleft() * 0.3)
        h_gp, c5_gp = gradient_polish(best_h, target_use, gp_time, alpha=5000)
        if c5_gp < best_score:
            best_score = c5_gp
            best_h = h_gp

    # Phase 11: Multi-restart with Fourier perturbations
    for freq in [1, 2, 3, 5, 7]:
        if tleft() < 60:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        x = np.linspace(0, 2, n_use)
        pert = 0.01 * np.cos(np.pi * freq * x)
        h_pert = project_feasible(best_h + pert, target_use)
        cd_time = min(40, tleft() * 0.2)
        h_cd, c5_cd = coord_descent_fast(h_pert, target_use, 20000, cd_time, rng,
                                         n_max_active=200, margin_init=0.02)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final LP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(50, tleft() * 0.5)
        h_lp, c5_lp = alternating_lp(best_h, n_use, target_use,
                                     max_iter=3000, time_limit=lp_time,
                                     d0=0.95, reg0=1e-5, n_active=120, scs_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 13: Final swap + CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        swap_time = min(20, tleft() * 0.3)
        h_swap, c5_swap = swap_refine(best_h, target_use, 1000, 20, swap_time,
                                      rng, n_active_set=300)
        if c5_swap < best_score:
            best_score = c5_swap
            best_h = h_swap

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(25, tleft() * 0.6)
        h_cd, c5_cd = coord_descent_fast(best_h, target_use, 30000, cd_time, rng,
                                         n_max_active=300, margin_init=0.005)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
