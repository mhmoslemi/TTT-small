import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    target = int(target_sum)
    h_r = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    h_r[order[:target]] = 1.0
    return h_r


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-10:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-10:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad


def adam_continuous(h0, target_sum, start_time, time_budget, alphas, lr_start, lr_end, beta1, beta2, eps):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)

    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')

        n_iters_max = 500
        for it in range(n_iters_max):
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break

            progress = it / max(n_iters_max - 1, 1)
            lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * progress)) + lr_end
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual - 1e-8:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()

    return best_h, best_actual

def _smooth_slsqp_obj(h, alpha):

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def two_interval_candidate(n, target, a_frac, l1_frac, gap_frac):
    h = np.zeros(n)
    l1 = int(round(l1_frac * target))
    l2 = int(target) - l1
    a = int(round(a_frac * (n - target - 1)))
    gap = int(round(gap_frac * max(n - target, 1)))
    a = max(0, min(a, n - l1 - l2 - gap - 1))
    if a < 0:
        a = 0
    b = a + l1 + max(gap, 0)
    if b + l2 > n:
        b = n - l2
        if b < a + l1:
            return h
    h[a:a + l1] = 1.0
    h[b:b + l2] = 1.0
    return h

        sa_time = min(250, tleft() - 80)
        max_it = int(min(300000, sa_time * 600))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                0.006, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4b: Second SA from a different projection
    if tleft() > 100:
        h_bin3 = project_binary(best_h + rng.randn(n) * 0.3, int(target))
        sa_time2 = min(150, tleft() - 60)
        max_it2 = int(min(200000, sa_time2 * 500))
        h_sa2, c5_sa2 = sa_binary(h_bin3, int(target), max_it2,
                                  0.004, 0.000003, rng, sa_time2, start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: SLSQP polish
    if tleft() > 80:
        slsqp_time = min(180, tleft() - 40)
        alphas_sl = [3000, 8000, 20000, 50000, 100000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     2000, 1e-16, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 6: Final greedy + Adam polish
    if tleft() > 50:
        h_bin4 = project_binary(best_h, int(target))
        g_time2 = min(40, tleft() * 0.3)
        h_g2, c5_g2 = multi_k_greedy_binary(h_bin4, int(target), max_iters=30000,
                                            time_limit=g_time2, start_time=start_time,
                                            top_k=5, scan_w=20)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    if tleft() > 30:
        polish_time = max(15.0, tleft() - 15)
        h_p, s_p = adam_continuous(
            best_h, target, start_time, polish_time,
            alphas=[30000, 80000, 200000],
            lr_start=0.001, lr_end=0.0001,
            beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
