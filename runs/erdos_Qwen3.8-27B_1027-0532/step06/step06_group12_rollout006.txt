def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    if l == 0:

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def gradient_for_lag(h, l, n, dx):
    grad = np.zeros(n)
    if l == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif l > 0:
        grad[l:] += (1.0 - h[:n - l]) * dx
        grad[:n - l] -= h[l:] * dx
    else:
        m = -l
        grad[:n - m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n - m] * dx
    return grad


def successive_lp(h_init, n, target, max_iter, time_limit,
                  step_init=0.2, step_min=0.05, top_k=200):
    start = time.time()
    h = h_init.copy()
    dx = 2.0 / n
    n_half = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        c5 = float(np.max(overlaps))
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break
        if top_k is not None and top_k < 2 * n - 1:
            sorted_idx = np.argsort(overlaps)[::-1][:top_k]
        else:
            sorted_idx = np.arange(2 * n - 1)
        progress = min(1.0, it / max(1, int(max_iter * 0.6)))
        step_size = step_init + (step_min - step_init) * progress
        delta = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(delta) == 0,
            delta >= -h + 1e-12,
            delta <= 1.0 - h - 1e-12,
            delta >= -step_size,
            delta <= step_size,
        ]
        for idx in sorted_idx:
            l = idx - n_half
            grad = gradient_for_lag(h, l, n, dx)
            constraints.append(overlaps[idx] + grad @ delta <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if delta.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        delta_val = np.asarray(delta.value).ravel()
        delta_val = np.clip(delta_val, -step_size, step_size)
        best_alpha = 0.0
        best_obj = best_c5
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        for alpha in alphas:
            h_test = h + alpha * delta_val
            h_test = np.clip(h_test, 0.0, 1.0)
            h_test = project_feasible(h_test, target)
            obj = compute_c5(h_test)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_alpha = alpha
                break
        if best_alpha == 0.0:
            continue
        h = h + best_alpha * delta_val
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    return best_h, best_c5


def frank_wolfe(h_init, n, target, time_limit, max_iter=400000, K=50):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        dx = 2.0 / n
        g = np.zeros(n)
        for idx in top_idx:
            l = idx - n_half
            g += gradient_for_lag(h, l, n, dx)
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 30000 and t - best_t > 200000:
            break
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=3000, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def random_binary_init(n, target, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:int(target)]] = 1.0
    return h


def structured_init(n, target, rng, shift):
    h = np.zeros(n)
    block = int(target // 4)
    offsets = [0, block, 2 * block, 3 * block]
    for off in offsets:
        start = (off + shift) % n
        for j in range(block):
            h[(start + j) % n] = 1.0
    return project_feasible(h, target)


def time_remaining(start_time, budget_s):
    return budget_s - 15.0 - (time.time() - start_time)
