from scipy.optimize import linprog

def solve_lp(h, n, target, reg=None):
    dx = 2.0 / n
    g = 1.0 - h
    c_obj = np.concatenate([np.zeros(n), [1.0]])
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_j h[j]*g[j+k]*dx - t <= 0
    # So row is [g[j+k]*dx for valid j, ..., 0, -1]
    
    A_rows = []
    for k in range(-(n-1), n):
        row = np.zeros(n + 1)
        row[-1] = -1.0  # coefficient of t
        if k >= 0:
            row[:n-k] = g[k:] * dx
        else:
            row[-k:] = g[:n+k] * dx
        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.zeros(2*n - 1)
    
    # Bounds: 0 <= h[i] <= 1, t unbounded below (but will be >= 0)
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_max_line(slopes, intercepts, lam=0.0):
    C = slopes + lam
    max_C = np.max(C)
    min_C = np.min(C)
    if max_C < 1e-13:
        return 1.0
    if min_C > 1e-13:
        return 0.0
    lo = 0.0
    hi = 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-11 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        c_sum = C[active].sum()
        if c_sum > 1e-11:
            hi = mid
        elif c_sum < -1e-11:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def alternating_lp_highs(h_init, n, target, max_iter, time_limit, d0=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        for idx, k in enumerate(range(-(n - 1), n)):
            A_ub[idx, -1] = -1.0
            if k >= 0:
                A_ub[idx, :n - k] = g[k:] * dx
                A_ub[idx, n - k:] = 0.0
            else:
                A_ub[idx, :-k] = 0.0
                A_ub[idx, -k:] = g[:n + k] * dx

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success or res.x is None:
            break

        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.5, damping * 0.97)
        else:
            damping = min(0.99, damping + 0.01)

    return best_h, best_c5


def coord_descent_all(h_init, target, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    k_vals = np.arange(-(n - 1), n)
    corr_idx = (n - 1) - k_vals

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        for i in order:
            i_plus_k = i + k_vals
            i_minus_k = i - k_vals
            valid1 = (i_plus_k >= 0) & (i_plus_k < n)
            valid2 = (i_minus_k >= 0) & (i_minus_k < n)
            slopes = np.zeros(2 * n - 1)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h[i_plus_k[valid1]]) * dx
            if np.any(valid2):
                slopes[valid2] -= h[i_minus_k[valid2]] * dx
            f_vals = corr[corr_idx] * dx
            intercepts = f_vals - slopes * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts)

        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.array([np.sum(h) - _global_target])

    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:

                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:

        except Exception:

def subgradient_improve(h, target, max_steps, time_limit):
    start = time.time()
    n = len(h)

        idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)

        grad = np.zeros(n)
        if np.any(idx_v1):
            grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k])
        if np.any(idx_v2):
            grad[idx_v2] -= h_cur[idxs[idx_v2] + k]
        grad *= dx
        improved = False

                stall = 0
                break

        lp_time = min(200, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp_highs(best_h, n, target,
                                           max_iter=50000, time_limit=lp_time, d0=0.92)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent (all shifts, exact 1D)
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.18)
        h_cd, c5_cd = coord_descent_all(best_h, target,
                                        max_passes=50000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP with HiGHS again (after CD)
    if tleft() > 100:
        lp_time = min(150, tleft() * 0.18)
        h_lp, c5_lp = alternating_lp_highs(best_h, n, target,
                                           max_iter=50000, time_limit=lp_time, d0=0.95)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: More coordinate descent
    if tleft() > 80:
        cd_time = min(120, tleft() * 0.15)
        h_cd, c5_cd = coord_descent_all(best_h, target,
                                        max_passes=50000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP with increasing alpha
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [500, 2000, 10000], 2000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: LP with active set (cvxpy, different approach)
    if tleft() > 120:
        lp_time = min(150, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp_active(best_h, n, target,
                                            max_iter=20000, time_limit=lp_time,
                                            n_active=80, d0=0.92, reg0=5e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Coordinate descent again
    if tleft() > 80:
        cd_time = min(100, tleft() * 0.15)
        h_cd, c5_cd = coord_descent_all(best_h, target,
                                        max_passes=50000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: SLSQP high alpha
    if tleft() > 80:
        h_ref = slsqp_round(best_h, n, target, [20000, 100000, 300000], 1500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Subgradient refinement
    if tleft() > 40:
        h_sg, c5_sg = subgradient_improve(best_h, target,
                                          max_steps=5000, time_limit=min(30, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: LP HiGHS final
    if tleft() > 50:
        lp_time = min(60, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_highs(best_h, n, target,
                                           max_iter=30000, time_limit=lp_time, d0=0.97)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final CD
    if tleft() > 30:
        cd_time = min(30, tleft() * 0.5)
        h_cd, c5_cd = coord_descent_all(best_h, target,
                                        max_passes=30000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 20:
        h_ref = slsqp_round(best_h, n, target, [50000, 200000, 500000], 1000, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 13: Final subgradient
    if tleft() > 10:
        h_sg, c5_sg = subgradient_improve(best_h, target,
                                          max_steps=3000, time_limit=min(8, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
