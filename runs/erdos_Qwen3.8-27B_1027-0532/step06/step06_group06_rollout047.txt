            else:
                sc_cur = compute_c5(h_cur)
                if sc < sc_cur:
                    h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def targeted_binary_search(h_start, target_sum, time_limit, start_time, n_sweeps=200):
    """Binary local search that targets the worst correlations via k-swaps."""
    n = len(h_start)
    h = h_start.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200
    
    for iteration in range(n_sweeps * n):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Find worst few correlations
        top_m = min(5, len(corr))
        worst_indices = np.argsort(corr)[-top_m:]
        
        found_improvement = False
        for max_idx in worst_indices:
            k_star = max_idx - (n - 1)
            max_val = float(corr[max_idx])
            
            # Find candidates: i where h[i]=1 and h[i+k_star]=0
            # These directly contribute to corr[max_idx]
            if k_star >= 0:
                valid_i = np.arange(0, n - k_star)
                candidates_i = valid_i[(h[valid_i] == 1) & (h[valid_i + k_star] == 0)]
                # For each such i, we want to find j where h[j]=0 and h[j+k_star]=1
                # Swapping h[i] and h[j] would reduce corr at k_star
                candidates_j = valid_i[(h[valid_i] == 0) & (h[valid_i + k_star] == 1)]
            else:
                k_pos = -k_star
                valid_i = np.arange(k_pos, n)
                candidates_i = valid_i[(h[valid_i] == 1) & (h[valid_i - k_pos] == 0)]
                candidates_j = valid_i[(h[valid_i] == 0) & (h[valid_i - k_pos] == 1)]
            
            if len(candidates_i) == 0 or len(candidates_j) == 0:
                continue
            
            # Try best swap
            n_try_i = min(len(candidates_i), 20)
            n_try_j = min(len(candidates_j), 20)
            
            best_improve = 0
            best_i = -1
            best_j = -1
            
            for ii in range(n_try_i):
                i = candidates_i[ii]
                for jj in range(n_try_j):
                    j = candidates_j[jj]
                    if i == j:
                        continue
                    # Estimate change in max correlation
                    nm = max_val
                    for t_idx in worst_indices:
                        k = t_idx - (n - 1)
                        d = 0.0
                        # Effect of h[i]: 1->0
                        if 0 <= i + k < n:
                            d -= (1.0 - h[i + k])
                        if 0 <= i - k < n:
                            d += h[i - k]
                        # Effect of h[j]: 0->1
                        if 0 <= j + k < n:
                            d += (1.0 - h[j + k])
                        if 0 <= j - k < n:
                            d -= h[j - k]
                        v = corr[t_idx] + d
                        if v > nm:
                            nm = v
                            break
                    if nm < max_val - best_improve:
                        best_improve = max_val - nm
                        best_i = i
                        best_j = j
            
            if best_i >= 0:
                h[best_i] = 0
                h[best_j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(corr.max()) * dx
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h.copy()
                    stall = 0
                    found_improvement = True
                    break  # Restart with new worst indices
                else:
                    h[best_i] = 1
                    h[best_j] = 0
        
        if not found_improvement:
            stall += 1
    
    return best_h, best_c5

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 3000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        h_new = h.copy()
        moved = False
        
        if r < 0.5:
            # Targeted move: pick from worst correlations
            worst = np.argsort(corr)[-10:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            if k >= 0:
                ip = [i for i in ones if i + k < n]
                jp = [j for j in zeros if j + k < n]
            else:
                kp = -k
                ip = [i for i in ones if i - kp >= 0]
                jp = [j for j in zeros if j - kp >= 0]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.8:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            # 4-swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        
        if not moved:
            continue
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def smooth_max_gradient_v2(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # grad[j] = dot(1-h, weights[j:N-1+N-1]) - dot(h, weights[0:j+N])... 
    # Let me use the fact that:
    # part1[j] = sum_{m=0}^{N-1-j} weights[N-1-j+m] * (1-h[m])
    #           = dot((1-h)[:N-j], weights[N-1-j:2*N-1-j])
    # part2[j] = sum_{m=0}^{N-1} weights[j+m] * h[N-1-m]
    #           = dot(h[::-1], weights[j:j+N])
    
    # This is still O(N^2) but can be done with a single convolve if we're clever
    # For now, the loop version is fine for N=606
    
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    grad = np.zeros(N)
    
    for j in range(N):
        grad[j] = np.dot(one_minus_h[:N-j], weights[N-1-j:]) - np.dot(h_rev, weights[j:j+N])
    
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r

def smooth_max_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_max_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    grad = np.zeros(N)
    for j in range(N):
        grad[j] = np.dot(one_minus_h[:N-j], weights[N-1-j:2*N-1-j]) - np.dot(h_rev, weights[j:j+N])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i // 2] ^ (i % 2)
    return tm

def periodic_pattern(n, period, duty):
    pattern = np.zeros(period)
    n_ones = int(round(duty * period))
    pattern[:n_ones] = 1
    reps = n // period + 1
    result = np.tile(pattern, reps)[:n]
    return result

def lbfgs_optimize(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_max_objective(h_cur, beta, target, penalty)
    for _attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_max_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_max_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            obj_new = smooth_max_objective(h_candidate, beta, target, penalty)
            if obj_new < curr_obj:
                h_cur = h_candidate
                curr_obj = obj_new
            else:
                sc_cur = compute_c5(h_cur)
                if sc < sc_cur:
                    h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def targeted_binary_search(h_start, target_sum, time_limit, start_time, max_sweeps=300):
    n = len(h_start)
    h = h_start.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 150
    for _sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        top_m = min(8, len(corr))
        worst_indices = np.argsort(corr)[-top_m:]
        found_improvement = False
        for max_idx in worst_indices:
            k_star = max_idx - (n - 1)
            max_val = float(corr[max_idx])
            if k_star >= 0:
                valid_range = np.arange(0, n - k_star)
                if len(valid_range) == 0:
                    continue
                candidates_i = valid_range[(h[valid_range] == 1) & (h[valid_range + k_star] == 0)]
                candidates_j = valid_range[(h[valid_range] == 0) & (h[valid_range + k_star] == 1)]
            else:
                k_pos = -k_star
                valid_range = np.arange(k_pos, n)
                if len(valid_range) == 0:
                    continue
                candidates_i = valid_range[(h[valid_range] == 1) & (h[valid_range - k_pos] == 0)]
                candidates_j = valid_range[(h[valid_range] == 0) & (h[valid_range - k_pos] == 1)]
            if len(candidates_i) == 0 or len(candidates_j) == 0:
                continue
            n_try_i = min(len(candidates_i), 30)
            n_try_j = min(len(candidates_j), 30)
            best_improve = 0.0
            best_i = -1
            best_j = -1
            for ii in range(n_try_i):
                i = candidates_i[ii]
                for jj in range(n_try_j):
                    j = candidates_j[jj]
                    if i == j:
                        continue
                    nm = max_val
                    for t_idx in worst_indices:
                        k = t_idx - (n - 1)
                        d = 0.0
                        if 0 <= i + k < n:
                            d -= (1.0 - h[i + k])
                        if 0 <= i - k < n:
                            d += h[i - k]
                        if 0 <= j + k < n:
                            d += (1.0 - h[j + k])
                        if 0 <= j - k < n:
                            d -= h[j - k]
                        v = corr[t_idx] + d
                        if v > nm:
                            nm = v
                            break
                    if nm < max_val - best_improve:
                        best_improve = max_val - nm
                        best_i = i
                        best_j = j
            if best_i >= 0:
                h[best_i] = 0.0
                h[best_j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(corr.max()) * dx
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h.copy()
                    stall = 0
                    found_improvement = True
                    break
                else:
                    h[best_i] = 1.0
                    h[best_j] = 0.0
        if not found_improvement:
            stall += 1
    return best_h, best_c5

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 4000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = float(it) / float(max(max_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.5:
            worst = np.argsort(corr)[-12:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            if k >= 0:
                ip = [i for i in ones if i + k < n]
                jp = [j for j in zeros if j + k < n]
            else:
                kp = -k
                ip = [i for i in ones if i >= kp]
                jp = [j for j in zeros if j >= kp]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.8:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def iterated_local_search(h_start, target_sum, n_restarts, time_limit, start_time, rng, perturb_strength=6):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h_curr = best_h.copy()
        if r > 0:
            n_flip = perturb_strength + rng.randint(0, 4)
            ones = np.where(h_curr == 1)[0]
            zeros = np.where(h_curr == 0)[0]
            if len(ones) > n_flip and len(zeros) > n_flip:
                flip_i = rng.choice(len(ones), n_flip, replace=False)
                flip_j = rng.choice(len(zeros), n_flip, replace=False)
                h_curr[ones[flip_i]] = 0
                h_curr[zeros[flip_j]] = 1
        time_remaining = time_limit - (time.time() - start_time)
        if time_remaining < 5:
            break
        time_alloc = min(40, time_remaining * 0.6)
        h_ls, c5_ls = targeted_binary_search(h_curr, target_sum,
                                               time_limit=time_alloc,
                                               start_time=start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        time_remaining = time_limit - (time.time() - start_time)
        if time_remaining < 10:
            break
        time_alloc_sa = min(30, time_remaining * 0.4)
        max_it = int(min(50000, time_alloc_sa * 1500))
        h_sa, c5_sa = sa_binary(h_ls, target_sum, max_it, 0.006, 0.0001,
                                 rng, time_limit=time_alloc_sa, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    return best_h, best_c5
