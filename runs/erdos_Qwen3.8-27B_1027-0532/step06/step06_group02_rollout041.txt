            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            top_i_list = ones[np.argsort(-A[ones])[:min(20, len(ones))]]
            top_j_list = zeros[np.argsort(-B[zeros])[:min(20, len(zeros))]]
            for i in top_i_list:
                if found:
                    break
                for j in top_j_list[:8]:
                    delta = compute_delta(h, i, j, k_arr)
                    new_top = full[top_lags] + delta[top_lags]
                    new_max = float(new_top.max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        stall = 0
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_adaptive(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr,
                use_2swap, block_frac=0.15, max_block_len=10, top_k_focus=5):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)
    stuck_count = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        if t < 1e-12:
            t = 1e-12
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        move_type = rng.random()
        accepted = False

        if move_type < block_frac:
            blen = rng.randint(3, min(max_block_len, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a + blen] = h[a:a + blen][::-1]
            if abs(h_new.sum() - h.sum()) > 1e-10:
                continue
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            d = c5_new - current_c5
            if d < 0 or rng.random() < np.exp(-d / t):
                h = h_new
                full = full_new
                current_c5 = c5_new
                accepted = True
        elif use_2swap and move_type < block_frac + 0.15 and stuck_count > 50:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    i2 = ones[rng.randint(len(ones))]
                    if i1 == i2:
                        continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    j2 = zeros[rng.randint(len(zeros))]
                    if j1 == j2:
                        continue
                if i1 in (j1, j2) or i2 in (j1, j2):
                    continue
                delta = compute_delta_2swap(h, i1, i2, j1, j2, k_arr)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                d = c5_new - current_c5
                if d < 0 or rng.random() < np.exp(-d / t):
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                    full += delta
                    current_c5 = c5_new
                    accepted = True
        else:
            if it % 5 == 0 and top_k_focus > 0:
                top_lags = np.argsort(full)[::-1][:top_k_focus]
                best_move = None
                best_score = 0
                for ti in top_lags:
                    k_star = int(ti) - (n - 1)
                    if k_star == 0:
                        continue
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    i_cands = ones[np.argsort(-A[ones])[:min(10, len(ones))]]
                    j_cands = zeros[np.argsort(-B[zeros])[:min(10, len(zeros))]]
                    for i in i_cands:
                        for j in j_cands:
                            delta_candidate = compute_delta(h, i, j, k_arr)
                            new_top = full[top_lags] + delta_candidate[top_lags]
                            nm = float(new_top.max())
                            if nm < float(full[top_lags[0]]):
                                score = float(full[top_lags[0]]) - nm
                                if score > best_score:
                                    best_score = score
                                    best_move = (i, j, delta_candidate)
                    if best_move is not None:
                        break
                if best_move is not None:
                    i, j, delta = best_move
                    new_max = float((full + delta).max())
                    c5_new = new_max * dx
                    d = c5_new - current_c5
                    if d < 0 or rng.random() < np.exp(-d / t):
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        current_c5 = c5_new
                        accepted = True
            if not accepted:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                d = c5_new - current_c5
                if d < 0 or rng.random() < np.exp(-d / t):
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    current_c5 = c5_new
                    accepted = True

        if accepted:
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
                stuck_count = 0
            else:
                stall += 1
                stuck_count += 1
        else:
            stall += 1
            stuck_count += 1

        if stall % 10000 == 0 and stuck_count > 200:
            stuck_count = 0
            h = project_binary(h, int(h.sum()))

    return best_h, best_c5

def smooth_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    logs = float(np.log(np.sum(np.exp(shifted))))
    return c_max + logs / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    exp_c = np.exp(alpha * (corr - c_max))
    w = exp_c / exp_c.sum()
    positions = np.arange(n)
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        idx1 = positions + k
        valid1 = (idx1 >= 0) & (idx1 < n)
        idx2 = positions - k
        valid2 = (idx2 >= 0) & (idx2 < n)
        wk = float(w[k + n - 1])
        grad[valid1] += wk * (1.0 - h[idx1[valid1]])
        grad[valid2] -= wk * h[idx2[valid2]]
    return grad

def gradient_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        phase_time = min(40.0, remaining * 0.4)
        lr = 1.5 / (alpha + 1.0)
        momentum = 0.8
        v = np.zeros(n)
        n_iter = int(min(phase_time * 150, 2000))
        n_iter = max(n_iter, 30)

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            c_max = float(corr.max())
            exp_c = np.exp(alpha * (corr - c_max))
            w = exp_c / exp_c.sum()
            positions = np.arange(n)
            grad = np.zeros(n)
            for k in range(-(n - 1), n):
                idx1 = positions + k
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = positions - k
                valid2 = (idx2 >= 0) & (idx2 < n)
                wk = float(w[k + n - 1])
                grad[valid1] += wk * (1.0 - h[idx1[valid1]])
                grad[valid2] -= wk * h[idx2[valid2]]
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 30 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def perturb_diverse(h, rng, n_pert):
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 10) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a + blen] = h[a:a + blen][::-1]
    return h_pert

def perturb_random_block_swap(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 10) + 1)
    a = rng.randint(0, n - blen + 1)
    b = rng.randint(0, n - blen + 1)
    while b == a or abs(b - a) < blen:
        b = rng.randint(0, n - blen + 1)
    temp = h_pert[a:a + blen].copy()
    h_pert[a:a + blen] = h_pert[b:b + blen]
    h_pert[b:b + blen] = temp
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    _g_n = n_init
    _g_target = float(target_init)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Multi-lag greedy
    allowed = min(40, tleft() * 0.06)
    h_g, c5_g = greedy_multi_lag(best_h, max_iters=50000, top_k=8, rng=rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Adaptive SA (main phase)
    allowed = min(250, tleft() * 0.25)
    n_sa = int(min(6000000, allowed * 40000))
    h_sa, c5_sa = sa_adaptive(best_h, n_sa, 0.03, 0.00005, rng,
                              time_limit=allowed, start_time=start_time, k_arr=k_arr,
                              use_2swap=True, block_frac=0.12, max_block_len=12, top_k_focus=5)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Another SA pass from best
    allowed = min(150, tleft() * 0.18)
    n_sa2 = int(min(4000000, allowed * 40000))
    h_sa2, c5_sa2 = sa_adaptive(best_h, n_sa2, 0.02, 0.00002, rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                use_2swap=True, block_frac=0.08, max_block_len=8, top_k_focus=3)
    if c5_sa2 < best_score:
        best_score = c5_sa2
        best_h = h_sa2.copy()

    # Phase 4: Gradient refinement (continuous)
    if tleft() > 100:
        target_use = float(n_init) / 2.0
        alphas = [5, 20, 60, 200, 800, 3000]
        allowed = min(180, tleft() * 0.25)
        h_ref, sc = gradient_refine(best_h, target_use, start_time, allowed, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: Project to binary, focused SA
    if tleft() > 80:
        h_bin = project_binary(best_h, target_init)
        allowed = min(100, tleft() * 0.15)
        n_sa3 = int(min(3000000, allowed * 40000))
        h_sa3, c5_sa3 = sa_adaptive(h_bin, n_sa3, 0.015, 0.00001, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                    use_2swap=True, block_frac=0.1, max_block_len=10, top_k_focus=5)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 6: Multi-start with diverse perturbations
    for restart in range(10):
        if tleft() < 60:
            break
        if restart % 4 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(5, 15))
        elif restart % 4 == 1:
            h_pert = perturb_block(best_h, rng, 15)
        elif restart % 4 == 2:
            h_pert = perturb_random_block_swap(best_h, rng, 12)
        else:
            h_pert = perturb_diverse(best_h, rng, 3)
            h_pert = perturb_block(h_pert, rng, 8)

        allowed = min(50, tleft() * 0.07)
        n_sa_r = int(min(1500000, allowed * 40000))
        h_sr, c5_sr = sa_adaptive(h_pert, n_sa_r, 0.015, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                  use_2swap=True, block_frac=0.15, max_block_len=10, top_k_focus=3)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Coarse-to-fine
    if tleft() > 100:
        for n_coarse in [60, 100, 150]:
            if tleft() < 50:
                break
            target_coarse = n_coarse // 2
            k_arr_c = np.arange(2 * n_coarse - 1) - (n_coarse - 1)
            x_new = np.linspace(0, 1, n_coarse)
            x_old = np.linspace(0, 1, n_init)
            h_coarse = np.interp(x_new, x_old, best_h)
            h_coarse_bin = project_binary(h_coarse, target_coarse)
            allowed = min(25, tleft() * 0.12)
            h_cg, c5_cg = greedy_multi_lag(h_coarse_bin, max_iters=10000, top_k=5,
                                           rng=rng, time_limit=allowed,
                                           start_time=start_time, k_arr=k_arr_c)
            allowed2 = min(25, tleft() * 0.1)
            n_sac = int(min(300000, allowed2 * 30000))
            h_cs, c5_cs = sa_adaptive(h_cg, n_sac, 0.015, 0.0001, rng,
                                      time_limit=allowed2, start_time=start_time, k_arr=k_arr_c,
                                      use_2swap=False, block_frac=0.2, max_block_len=6, top_k_focus=3)
            h_coarse_best = h_cs if c5_cs < c5_cg else h_cg
            # Embed to fine
            h_fine = np.zeros(n_init)
            ratio = n_init / n_coarse
            for i in range(n_coarse):
                s = int(i * ratio)
                e = int((i + 1) * ratio)
                if e > n_init:
                    e = n_init
                if h_coarse_best[i] > 0.5:
                    h_fine[s:e] = 1.0
                else:
                    h_fine[s:e] = 0.0
            h_fine = project_binary(h_fine, target_init)
            allowed3 = min(20, tleft() * 0.08)
            h_fg, c5_fg = sa_adaptive(h_fine, int(min(500000, allowed3 * 40000)),
                                      0.01, 0.00005, rng,
                                      time_limit=allowed3, start_time=start_time, k_arr=k_arr,
                                      use_2swap=False, block_frac=0.1, max_block_len=8, top_k_focus=3)
            if c5_fg < best_score:
                best_score = c5_fg
                best_h = h_fg.copy()

    # Phase 8: Try nearby resolutions
    for n_try in [n_init + 4, n_init + 8, n_init - 4, n_init + 12]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        x_old = np.linspace(0, 1, n_init)
        x_new = np.linspace(0, 1, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(15, tleft() * 0.08)
        h_g3, c5_g3 = sa_adaptive(h_bin_try, int(min(300000, allowed * 30000)),
                                  0.01, 0.0001, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr_try,
                                  use_2swap=False, block_frac=0.1, max_block_len=8, top_k_focus=3)
        dx_try = 2.0 / n_try
        c5_scaled = c5_g3
        if c5_scaled < best_score:
            # Interpolate back to n_init
            x_b = np.linspace(0, 1, n_try)
            x_a = np.linspace(0, 1, n_init)
            h_back = np.interp(x_a, x_b, h_g3)
            h_back = project_binary(h_back, target_init)
            best_score = compute_c5(h_back)
            best_h = h_back.copy()

    # Phase 9: SLSQP refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        if n_use != _g_n:
            _g_n = n_use
            _g_target = target_use
        alphas_slsqp = [30, 100, 400, 1500, 6000]
        allowed = min(tleft() - 20, 150)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas_slsqp, 2000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 10: Final gradient + binary SA
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        if n_use != _g_n:
            _g_n = n_use
            _g_target = target_use
        alphas2 = [50, 200, 1000, 5000]
        allowed = min(tleft() - 20, 80)
        h_ref2, sc2 = gradient_refine(best_h, target_use, start_time, allowed, alphas2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 11: Final binary SA burst
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        k_arr_f = np.arange(2 * n_use - 1) - (n_use - 1)
        h_bin_final = project_binary(best_h, target_use)
        remaining = max(5.0, tleft() - 5)
        n_sa_f = int(min(1000000, remaining * 30000))
        h_sa_f, c5_f = sa_adaptive(h_bin_final, n_sa_f, 0.005, 0.000005, rng,
                                   time_limit=remaining, start_time=start_time, k_arr=k_arr_f,
                                   use_2swap=True, block_frac=0.08, max_block_len=8, top_k_focus=5)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Final: project to feasible
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
