            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -0.5, 0.5)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Update active set: include constraints close to max
        max_corr = np.max(corr)
        threshold = max_corr - 0.05 / dx  # in units of the sum
        # Actually, let's use top-K plus any constraint within a small gap
        new_active = np.argsort(corr)[-n_active:]
        active_k = new_active
        
        if p % 50 == 49:
            lam_rate = max(0.001, lam_rate * 0.98)
    
    return best_h, best_c5

def alternating_lp_v2(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_set)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        # Add constraints that have become "close" to active
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_set and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 3e-5:
                    active_set.add(k_idx)
                    if len(active_set) > n_active * 5:
                        break
        active_set = top_k | active_set
        if len(active_set) > n_active * 5:
            active_set = set(np.argsort(corr)[-n_active * 5:].tolist())
        
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.98, damping + 0.05)
        if it % 40 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.6)
    
    return best_h, best_c5

def smooth_max_obj(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    soft = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(soft) * dx

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol):
    dx = 2.0 / n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            return smooth_max_obj(x, alpha, n, target)
        def obj_grad(x):
            # Numerical gradient would be too slow; use analytical
            # The gradient of smooth max of f_k(x) is (1/Z) sum exp(alpha(f_k - f_max)) * grad f_k
            # For simplicity, use SLSQP's numerical gradient
            return None
        cons = [{'type': 'eq',
                 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_descent(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_scale_cd(h_init, n, target, time_limit, rng, resolutions=None):
    """Try multiple resolutions and return the best."""
    start = time.time()
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    
    if resolutions is None:
        base_n = len(h_init)
        resolutions = [r for r in [base_n, base_n*2, base_n*3, base_n*4] if r <= 2000]
    
    for n_trial in resolutions:
        if time.time() - start > time_limit - 30:
            break
        target_trial = n_trial / 2.0
        h_trial = interpolate_to(best_h, n_trial, target_trial)
        sc = compute_c5(h_trial)
        
        time_left = time_limit - (time.time() - start) - 20
        if time_left < 60:
            if sc < best_score:
                best_score = sc
                best_h = h_trial
            continue
        
        # Coordinate descent
        cd_time = min(time_left * 0.4, 120)
        n_act = min(n_trial, max(80, n_trial // 5))
        h_cd, c5_cd = coord_descent_v2(h_trial, target_trial, n_act,
                                         max_passes=50000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            if len(h_trial) != n_trial:
                h_trial = h_cd
        
        # LP
        time_left = time_limit - (time.time() - start) - 20
        if time_left > 80:
            lp_time = min(time_left * 0.3, 100)
            h_lp, c5_lp = alternating_lp_v2(h_trial, n_trial, target_trial,
                                             max_iter=5000, time_limit=lp_time,
                                             d0=0.9, reg0=1e-4, n_active=n_act, scs_eps=1e-11)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
        
        # More CD
        time_left = time_limit - (time.time() - start) - 15
        if time_left > 40:
            cd_time = min(time_left * 0.5, 60)
            h_cd, c5_cd = coord_descent_v2(best_h, len(best_h)/2.0, n_act,
                                             max_passes=30000, time_limit=cd_time, rng=rng)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
    
    return best_h, best_score

def annealed_cd(h_init, target, n_active, time_limit, rng, n_restarts=3):
    """Coordinate descent with simulated annealing restarts."""
    start = time.time()
    n = len(h_init)
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    
    for r in range(n_restarts):
        time_left = time_limit - (time.time() - start)
        if time_left < 30:
            break
        noise = (r + 1) * 0.03
        h_cur = best_h + rng.randn(n) * noise
        h_cur = project_feasible(h_cur, target)
        
        cd_time = min(time_left * 0.3, 60)
        h_cd, c5_cd = coord_descent_v2(h_cur, target, n_active,
                                         max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    
    return best_h, best_score

And this DOES depend on i (through h[i+k] and h[i-k]).

So we can't fully precompute it as a simple shift. We need h[i+k] and h[i-k] for each i.

But wait: for a fixed k, the vector [slope_k[i] for all i] IS a linear combination of shifted h:
slope_k[i] = dx * (1 - h[i+k] - h[i-k])

For i where both i+k and i-k are valid (0≤i+k<n, 0≤i-k<n):
slope_k[i] = dx * (1 - h[i+k] - h[i-k])

This is: dx * 1 - dx*h[i+k] - dx*h[i-k]

The vector of h[i+k] for all valid i is just h shifted by k (padded with 0).
Similarly for h[i-k].

So: slope_vector_k = dx * (ones - shift(h, k) - shift(h, -k)) (with appropriate padding)

This IS precomputable! Let me write it more carefully:

For k > 0:
- h[i+k] is valid for i = 0, ..., n-1-k → values are h[k], h[k+1], ..., h[n-1]
- h[i-k] is valid for i = k, ..., n-1 → values are h[0], h[1], ..., h[n-1-k]

So:
slope_k[i] = dx * (1 - h[i+k] - h[i-k]) for i where both are valid
           = dx * (1 - h[i+k]) for i where only i+k is valid (i < k, so i-k < 0)
           = dx * (1 - h[i-k]) for i where only i-k is valid (i > n-1-k, so i+k ≥ n)
           = dx * 1 for i where neither is valid

Wait, I need to be more careful about the boundary.

For k > 0:
- i+k < n → i < n-k → i = 0, ..., n-k-1
- i-k ≥ 0 → i ≥ k → i = k, ..., n-1

So:
- i = 0, ..., k-1: only i+k valid (assuming k < n-k, i.e., k < n/2)
  slope = dx*(1 - h[i+k])
- i = k, ..., n-k-1: both valid
  slope = dx*(1 - h[i+k] - h[i-k])
- i = n-k, ..., n-1: only i-k valid
  slope = dx*(1 - h[i-k])

This can be written as:
slope_k = dx * (1 - shift_right(h, k) - shift_left(h, k))

where shift_right(h, k)[i] = h[i+k] if i+k < n else 0
and shift_left(h, k)[i] = h[i-k] if i-k ≥ 0 else 0

So the slope vector for lag k is simply:
dx * (ones - np.roll-like shifts)

Great, so we can precompute the slope matrix (n_active × n) in O(n_active * n) time, which is much faster than the nested loop.
