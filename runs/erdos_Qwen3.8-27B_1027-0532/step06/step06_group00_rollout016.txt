import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

# Smooth objective for SLSQP
_smooth_n = 0
_smooth_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _smooth_target

def _cons_jac(x):
    return np.ones(_smooth_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _smooth_n, _smooth_target
    _smooth_n = n
    _smooth_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

import numpy as np
from scipy.optimize import minimize
import time

_smooth_n = 0
_smooth_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _smooth_target


def _cons_jac(x):
    return np.ones(_smooth_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _smooth_n, _smooth_target
    _smooth_n = n
    _smooth_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            h_pg, c5_pg = projected_gradient(h1, n1, target1,
                                             time_limit=min(40, tleft() * 0.15),
                                             n_shifts=4, max_iters=3000)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

        if tleft() > 250:
            h_cd, c5_cd = enhanced_pair_cd(h1, n1, target1,
                                           time_limit=min(300, tleft() * 0.65),
                                           max_sweeps=30000,
                                           n_cand_start=12, n_cand_end=50,
                                           n_shifts=4, max_no_improve=250,
                                           stochastic=True, temp_init=0.003)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [2000, 10000, 30000],
                                   1000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_pg, c5_pg = projected_gradient(best_h, n0, target0,
                                         time_limit=min(30, tleft() * 0.1),
                                         n_shifts=5, max_iters=2000)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

        h_cd, c5_cd = enhanced_pair_cd(best_h, n0, target0,
                                       time_limit=min(250, tleft() * 0.6),
                                       max_sweeps=25000,
                                       n_cand_start=25, n_cand_end=70,
                                       n_shifts=5, max_no_improve=300,
                                       stochastic=True, temp_init=0.002)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 180:
            h_cd2, c5_cd2 = enhanced_pair_cd(h2, n2, target2,
                                             time_limit=min(180, tleft() * 0.6),
                                             max_sweeps=15000,
                                             n_cand_start=12, n_cand_end=40,
                                             n_shifts=4, max_no_improve=250,
                                             stochastic=False)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2

            if tleft() > 80:
                h_pol2, sc2 = slsqp_optimize(best_h, n0, target0,
                                             [5000, 20000],
                                             800, 1e-14)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_pol2

    if tleft() > 80:
        h_final, sc = slsqp_optimize(best_h, n0, target0,
                                     [10000, 50000],
                                     800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 50:
        h_cd_f, c5_f = enhanced_pair_cd(best_h, n0, target0,
                                        time_limit=min(40, tleft() * 0.7),
                                        max_sweeps=10000,
                                        n_cand_start=40, n_cand_end=80,
                                        n_shifts=5, max_no_improve=200)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    if tleft() > 30:
        h_last, sc_last = slsqp_optimize(best_h, n0, target0,
                                         [50000, 100000],
                                         500, 1e-15)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
