import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix any remaining discrepancy
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            rem = np.where(h_r == 1)[0]
            for p in rem[:final_sum - target_sum]:
                h_r[p] = 0
        else:
            rem = np.where(h_r == 0)[0]
            for p in rem[:target_sum - final_sum]:
                h_r[p] = 1
    return h_r.astype(float)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def smoothed_objective(h, alpha):
    corr = full_corr_fast(h)
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    return c_max + float(np.log(exp_shifted.sum())) / alpha

def smoothed_gradient(h, alpha):
    n = len(h)
    corr = full_corr_fast(h)
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w = w / w.sum()
    
    one_minus_h = 1.0 - h
    positions = np.arange(n)
    grad = np.zeros(n)
    
    for lag_idx in range(2 * n - 1):
        wk = w[lag_idx]
        if wk < 1e-15:
            continue
        k = lag_idx - (n - 1)
        # First term: (1-h[i+k]) for valid i+k
        idx = positions + k
        valid = (idx >= 0) & (idx < n)
        grad[valid] += wk * one_minus_h[idx[valid]]
        # Second term: -h[i-k] for valid i-k
        idx2 = positions - k
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= wk * h[idx2[valid2]]
    
    return grad

def adam_smooth_optimize(h0, target, alphas, time_limit, start_time, lr_base=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    best_h = h0.copy()
    best_score = compute_c5(h0)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(120.0, time_limit - (time.time() - start_time))
        h = h0.copy()
        h = project_box_sum(h, target)
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        n_iter = int(phase_time * 200)
        n_iter = max(n_iter, 100)
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            t += 1
            grad = smoothed_gradient(h, alpha)
            lr = lr_base / (1.0 + 0.01 * t)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h, target)
            
            if it % 20 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                    h0 = h.copy()
    
    return best_h, best_score

def steepest_binary_search(h, k_arr, time_limit, start_time, max_rounds=5000):
    n = len(h)
    h = h.copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        k_star = int(np.argmax(full)) - (n - 1)
        if k_star == 0:
            break
        
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] = h[idx[valid]]
        idx2 = positions - k_star
        valid2 = (idx2 >= 0) & (idx2 < n)
        A[valid2] += h[idx2[valid2]]
        
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] = h[idx[valid]]
        idx2 = positions + k_star
        valid2 = (idx2 >= 0) & (idx2 < n)
        B[valid2] += h[idx2[valid2]]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        top_i = ones[np.argsort(A[ones])[:min(100, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(100, len(zeros))]]
        
        best_imp = 0.0
        bi, bj = -1, -1
        bd = None
        
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                imp = float(full.max()) - new_max
                if imp > best_imp:
                    best_imp = imp
                    bi, bj = i, j
                    bd = delta
        
        if bi >= 0:
            h[bi] = 0.0
            h[bj] = 1.0
            full += bd
            c5 = float(full.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def sa_optimize(h, k_arr, n_iters, t_init, t_final, time_limit, start_time, rng,

    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        if r < 0.7:
            # Single swap biased towards critical lag
            if it % 3 == 0 and len(ones) > 2 and len(zeros) > 2:
                k_star = int(np.argmax(full)) - (n - 1)
                if k_star != 0:
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] = h[idx[valid]]
                    idx2 = positions - k_star
                    valid2 = (idx2 >= 0) & (idx2 < n)
                    A[valid2] += h[idx2[valid2]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] = h[idx[valid]]
                    idx2 = positions + k_star
                    valid2 = (idx2 >= 0) & (idx2 < n)
                    B[valid2] += h[idx2[valid2]]
                    i_score = A[ones]
                    j_score = B[zeros]
                    # Sample with probability proportional to score
                    i_probs = np.clip(i_score - i_score.min(), 0, None)
                    if i_probs.sum() > 0:
                        i_probs = i_probs / i_probs.sum()
                        i = np.random.choice(ones, p=i_probs, random_state=rng)
                    else:
                        i = ones[rng.randint(len(ones))]
                    j_probs = np.clip(j_score - j_score.min(), 0, None)
                    if j_probs.sum() > 0:
                        j_probs = j_probs / j_probs.sum()
                        j = np.random.choice(zeros, p=j_probs, random_state=rng)
                    else:
                        j = zeros[rng.randint(len(zeros))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        else:

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2 * n - 1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

    mask = (idx >= 0) & (idx < n)

    idx = i - k_arr

    n = len(h)
    corr = full_corr_fast(h)
    c_max = corr.max()
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)

    one_minus_h = 1.0 - h

        valid2 = (idx2 >= 0) & (idx2 < n)

            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def sa_optimize(h, k_arr, n_iters, t_init, t_final, time_limit, start_time, rng,
                block_frac=0.25, max_block_len=12):
    n = len(h)
    h = h.copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)
    stall = 0
    max_stall = 20000
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        if r < 1.0 - block_frac:
            if it % 4 == 0:
                k_star = int(np.argmax(full)) - (n - 1)
                if k_star != 0:
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] = h[idx[valid]]
                    idx2 = positions - k_star
                    valid2 = (idx2 >= 0) & (idx2 < n)
                    A[valid2] += h[idx2[valid2]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] = h[idx[valid]]
                    idx2 = positions + k_star
                    valid2 = (idx2 >= 0) & (idx2 < n)
                    B[valid2] += h[idx2[valid2]]
                    i_score = A[ones]
                    j_score = B[zeros]
                    i_thresh = np.percentile(i_score, 70)
                    j_thresh = np.percentile(j_score, 30)
                    i_cand = ones[i_score <= i_thresh]
                    j_cand = zeros[j_score >= j_thresh]
                    if len(i_cand) == 0:
                        i_cand = ones
                    if len(j_cand) == 0:
                        j_cand = zeros
                    i = i_cand[rng.randint(len(i_cand))]
                    j = j_cand[rng.randint(len(j_cand))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            blen = rng.randint(3, min(max_block_len, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a + blen] = h[a:a + blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)
    k_arr = np.arange(2 * n - 1) - (n - 1)

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Adam with analytical gradient (smoothed max)
    if tleft() > 100:
        alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
        allowed = min(120, tleft() * 0.12)
        h_opt, sc = adam_smooth_optimize(best_h, float(target), alphas, allowed, start_time,
                                         lr_base=0.05, beta1=0.9, beta2=0.999)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    # Phase 2: Steepest binary descent
    h_bin = project_binary(best_h, target)
    if tleft() > 80:
        allowed = min(150, tleft() * 0.15)
        h_sd, c5_sd = steepest_binary_search(h_bin, k_arr, allowed, start_time,
                                             max_rounds=20000, n_cand=40)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd.copy()

    # Phase 3: SA with smart moves
    if tleft() > 100:
        allowed = min(200, tleft() * 0.18)
        n_sa = int(min(5000000, allowed * 50000))
        h_sa, c5_sa = sa_optimize(h_bin, k_arr, n_sa, 0.02, 0.00001,
                                   allowed, start_time, rng,
                                   block_frac=0.25, max_block_len=12)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Second Adam round from improved binary
    if tleft() > 120:
        h_cont = project_box_sum(best_h, float(target))
        alphas2 = [100, 500, 2000, 10000, 50000]
        allowed = min(120, tleft() * 0.12)
        h_opt2, sc2 = adam_smooth_optimize(h_cont, float(target), alphas2, allowed, start_time,
                                           lr_base=0.03, beta1=0.9, beta2=0.999)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_opt2.copy()

    # Phase 5: Steepest descent again from latest best
    h_bin2 = project_binary(best_h, target)
    if tleft() > 60:
        allowed = min(100, tleft() * 0.12)
        h_sd2, c5_sd2 = steepest_binary_search(h_bin2, k_arr, allowed, start_time,
                                               max_rounds=15000, n_cand=50)
        if c5_sd2 < best_score:
            best_score = c5_sd2
            best_h = h_sd2.copy()

    # Phase 6: Multi-start SA with diverse perturbations
    for restart in range(6):
        if tleft() < 70:
            break
        if restart % 3 == 0:
            h_pert = best_h.copy()
            n_flips = rng.randint(3, 8)
        elif restart % 3 == 1:
            h_pert = best_h.copy()
            blen = rng.randint(4, 12)
            a = rng.randint(0, n - blen)
            h_pert[a:a + blen] = best_h[a:a + blen][::-1]
        else:
            h_pert = best_h.copy()
            for _ in range(rng.randint(2, 5)):
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
            blen = rng.randint(3, 8)
            a = rng.randint(0, n - blen)
            h_pert[a:a + blen] = h_pert[a:a + blen][::-1]

        allowed = min(50, tleft() * 0.07)
        n_sa_r = int(min(800000, allowed * 40000))
        h_sr, c5_sr = sa_optimize(h_pert, k_arr, n_sa_r, 0.012, 0.000005,
                                   allowed, start_time, rng,
                                   block_frac=0.2, max_block_len=8)
        if tleft() > 40:
            h_sr, c5_sr = steepest_binary_search(h_sr, k_arr, max(3, tleft() * 0.04),
                                                  start_time, max_rounds=3000, n_cand=30)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Try larger resolution
    if tleft() > 180:
        n_large = 1000
        target_large = n_large // 2
        k_arr_large = np.arange(2 * n_large - 1) - (n_large - 1)
        x_new = np.linspace(0, 1, n_large)
        x_old = np.linspace(0, 1, n)
        h_large = np.interp(x_new, x_old, best_h)
        h_large_bin = project_binary(h_large, target_large)

        allowed = min(80, tleft() * 0.2)
        n_sa_l = int(min(2000000, allowed * 15000))
        h_sl, c5_sl = sa_optimize(h_large_bin, k_arr_large, n_sa_l, 0.015, 0.00001,
                                   allowed, start_time, rng, block_frac=0.2, max_block_len=10)
        if tleft() > 50:
            allowed2 = min(40, tleft() * 0.12)
            h_sl, c5_sl = steepest_binary_search(h_sl, k_arr_large, allowed2, start_time,
                                                 max_rounds=5000, n_cand=35)
        if tleft() > 80:
            h_large_cont = project_box_sum(h_sl, float(target_large))
            alphas_l = [50, 200, 1000, 5000, 20000]
            allowed3 = min(80, tleft() * 0.2)
            h_opt_l, sc_l = adam_smooth_optimize(h_large_cont, float(target_large), alphas_l,
                                                  allowed3, start_time, lr_base=0.03)
            if sc_l < c5_sl:
                c5_sl = sc_l
                h_sl = h_opt_l
            if tleft() > 30:
                h_sl_bin = project_binary(h_sl, target_large)
                h_sl_bin, c5_sl2 = steepest_binary_search(h_sl_bin, k_arr_large,
                                                           max(5, tleft() * 0.08), start_time,
                                                           max_rounds=3000, n_cand=30)
                if c5_sl2 < c5_sl:
                    c5_sl = c5_sl2
                    h_sl = h_sl_bin

        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl.copy()
            n = n_large
            k_arr = k_arr_large

    # Phase 8: Final steepest descent + SA burst
    n_final = len(best_h)
    target_final = n_final // 2
    k_arr_final = np.arange(2 * n_final - 1) - (n_final - 1)
    h_bin_final = project_binary(best_h, target_final)

    if tleft() > 40:
        allowed = min(50, tleft() * 0.2)
        h_fs, c5_fs = steepest_binary_search(h_bin_final, k_arr_final, allowed, start_time,
                                             max_rounds=10000, n_cand=50)
        if c5_fs < best_score:
            best_score = c5_fs
            best_h = h_fs.copy()

    if tleft() > 30:
        allowed = min(30, tleft() * 0.3)
        n_sa_f = int(min(1000000, allowed * 50000))
        h_sa_f, c5_sf = sa_optimize(h_bin_final, k_arr_final, n_sa_f, 0.005, 0.000001,
                                     allowed, start_time, rng, block_frac=0.2, max_block_len=8)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sa_f.copy()

    # Phase 9: Final Adam polish
    if tleft() > 30:
        n_out = len(best_h)
        target_out = float(n_out) / 2.0
        h_cont_final = project_box_sum(best_h, target_out)
        alphas_final = [10000, 50000, 200000]
        allowed = min(30, tleft() * 0.4)
        h_opt_f, sc_f = adam_smooth_optimize(h_cont_final, target_out, alphas_final, allowed,
                                              start_time, lr_base=0.02, beta1=0.9, beta2=0.999)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_opt_f.copy()

    # Final feasibility
    n_out = len(best_h)
    target_out = float(n_out) / 2.0
    h_out = project_box_sum(best_h, target_out)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_out
