    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:

    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_grad_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    G = np.zeros((n, n_corr))
    for m in range(n_corr):
        k = m - mid
        j1s = max(0, k)
        j1e = min(n, n + k)
        if j1e > j1s:
            idx = np.arange(j1s, j1e)
            G[idx, m] += (1.0 - h[idx - k])
        j2s = max(0, -k)
        j2e = min(n, n - k)
        if j2e > j2s:
            idx = np.arange(j2s, j2e)
            G[idx, m] -= h[idx + k]
    return G

def lp_iteration(h0, n, target):
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    G = compute_grad_matrix(h0, n)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((n, 1)).T])
    b_ub = -corr + G @ h0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return h0.copy(), float(np.max(corr))

def lp_phase(h_init, n, target, time_limit, max_iters=200):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        h_lp, t_lp_dx = lp_iteration(h, n, target)
        h_new = project_feasible(h_lp, target)
        
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        
        if not improved:
            if it > 10:
                break
    
    return best_h, best_c5
