def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def water_filling_project(y, S):
    """Project y onto {h: 0<=h<=1, sum(h)=S}"""
    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2
        h = np.clip(y - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fix any remaining error
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-h)[:int(abs(diff)) + 1]  # take from largest
        else:
            idx = np.argsort(h)[:int(abs(diff)) + 1]  # take from smallest
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (cmax + np.log(sum_exp) / alpha) * dx
    
    p = exp_s / sum_exp  # softmax weights over correlations
    
    grad = np.zeros(N)
    N_half = N - 1
    
    for k in range(len(corr)):
        if p[k] < 1e-14:
            continue
        lag = k - N_half
        pk = p[k]
        if lag >= 0:
            # First term: (1-h[j-lag]) for j in [lag, N-1]
            if lag < N:
                grad[lag:] += pk * (1.0 - h[:N-lag])
            # Second term: -h[j+lag] for j in [0, N-lag-1]
            if N - lag > 0:
                grad[:N-lag] += pk * (-h[lag:])
        else:
            m = -lag
            # First term: (1-h[j+m]) for j in [0, N-m-1]
            if N - m > 0:
                grad[:N-m] += pk * (1.0 - h[m:])
            # Second term: -h[j-m] for j in [m, N-1]
            if m < N:
                grad[m:] += pk * (-h[:N-m])
    
    grad *= dx
    return obj, grad

def water_filling_project(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(200):
        tau = (lo + hi) / 2.0
        h = np.clip(y - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the residual
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        # Add residual to the "most room" elements
        n_adjust = min(n, max(1, int(abs(diff)) + 1))
        h[order[:n_adjust]] += diff / n_adjust
        h = np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):

    dx = 2.0 / n

    grad = np.zeros(n)
    n_half = n - 1

        if pk < 1e-15:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def water_filling_project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        tau = (lo + hi) * 0.5
        h = np.clip(y - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        n_adj = max(1, min(n, int(abs(diff)) + 1))
        h = h.copy()
        h[order[:n_adj]] += diff / n_adj
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    obj = (cmax + np.log(sum_exp) / alpha) * dx
    p = exp_s / sum_exp
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - n_half
        if lag >= 0:
            if lag < n:
                grad[lag:] += pk * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= pk * h[lag:]
        else:
            m = -lag
            if n - m > 0:
                grad[:n - m] += pk * (1.0 - h[m:])
            if m < n:
                grad[m:] -= pk * h[:n - m]
    grad *= dx
    return obj, grad

    start = time.time()
    state = {'count': 0}

    def obj_grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        obj, grad = smooth_obj_and_grad(x_c, alpha)
        s = np.sum(x_c) - target
        obj += mu * s * s
        grad = grad + 2.0 * mu * s
        state['count'] += 1
        return obj, grad

    res = minimize(obj_grad, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': 200000,
                            'ftol': 1e-18, 'gtol': 1e-14, 'disp': False})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = water_filling_project(h_new, target)
    return h_new
