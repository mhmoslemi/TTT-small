            k = int(si) - (n - 1)
            if k == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif k > 0:
                m_end = n - k
                grad[:m_end] += wi * (1.0 - h[k:m_end + k])
                grad[k:] -= wi * h[:m_end]
            else:
                ak = -k
                m_end = n - ak
                grad[ak:] += wi * (1.0 - h[:m_end])
                grad[:m_end] -= wi * h[ak:]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr in [0.5, 0.3, 0.15, 0.08, 0.04]:
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 1000:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _TARGET
