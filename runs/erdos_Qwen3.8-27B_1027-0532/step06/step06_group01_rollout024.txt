Key insight for the continuous case: at the optimal solution, the "active" shifts (those achieving the max) should all have exactly the same overlap value. If we can identify which shifts are active, we can set up a system of equations to solve for the optimal h.

**New idea: Active set identification + Newton's method**

1. Compute the overlap at all shifts
2. Identify the shifts that are within ε of the max (call this the active set A)
3. We want ∑_i h[i](1-h[i+k]) = C for all k ∈ A (all equal)
4. We also want ∑_i h[i] = n/2
5. And 0 ≤ h[i] ≤ 1

This is a system of |A| + 1 equations in n unknowns (with inequality constraints). If |A| + 1 < n, there are many solutions. We can pick the one that minimizes ∑ h[i]² (or some other regularizer) to get a unique solution.

But this is a nonlinear system (bilinear terms h[i](1-h[i+k])). We can linearize around the current solution and use Newton's method.

Actually, let me think about this differently. The problem is:
minimize t
subject to: ∑_i h[i](1-h[i+k]) * dx ≤ t for all k
            ∑_i h[i] = n/2
            0 ≤ h[i] ≤ 1

At optimality (assuming some regularity), the KKT conditions give:
- ∂L/∂h[i] = 0 for 0 < h[i] < 1
- ∂L/∂t = 0: ∑_{k∈A} λ_k = 1 (where A is the active set)
- ∂L/∂h[i] = ∑_{k∈A} λ_k * (1-h[i+k]) * dx - ∑_{k'∈A: i-k' valid} λ_{k'} * h[i-k'] * dx - μ = 0

This is a linear equation in h (given the active set and multipliers). So we can solve it!

But we need to know the active set A, which depends on the solution. This is a chicken-and-egg problem that can be solved iteratively:
1. Start with a guess for A
2. Solve the KKT system
3. Check if the solution is consistent with A (i.e., g_k = t for k ∈ A, g_k < t for k ∉ A)
4. If not, update A and repeat

This is similar to active set methods in quadratic programming.

def active_set_solve(h_init, n, target, time_limit, tol=1e-10):
    """Solve the KKT system iteratively with active set identification."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    
    for iteration in range(1000):
        if time.time() - start > time_limit:
            break
        
        # Compute all overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        t = float(np.max(overlaps))
        
        # Identify active set (within tol of max)
        active = np.where(overlaps >= t - 1e-8)[0]
        
        if len(active) == 0:
            break
        
        # Try to equalize the active overlaps by adjusting h
        # For each active shift k, compute ∂g_k/∂h
        # and solve for the adjustment that makes all active g_k equal
        
        # Gradient of g_k at shift k:
        # ∂g_k/∂h[i] = (1-h[i+k])*dx - h[i-k]*dx (for valid indices)
        ks = active - (n - 1)  # actual shift values
        
        # Build the gradient matrix G where G[k,i] = ∂g_k/∂h[i]
        # Then we want to find δh such that G @ δh makes all g_k equal
        # i.e., G @ δh = -1 (all active g_k decrease by the same amount)
        # with the constraint that sum(δh) = 0
        
        G = np.zeros((len(active), n))
        idxs = np.arange(n)
        for idx, k in enumerate(ks):
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            G[idx, v1] += (1.0 - h[idxs[v1] - k])
            G[idx, v2] -= h[idxs[v2] + k]
        
        # We want G @ δh = -dx * ones (to reduce all by dx)
        # subject to ones @ δh = 0
        # Use least squares with the constraint
        # [G; ones] @ δh = [-dx*ones; 0]
        A_mat = np.vstack([G, np.ones(n) / np.sqrt(n)])
        b_vec = np.concatenate([np.full(len(active), -1.0), [0.0]])
        
        # Solve least squares
        delta_h, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        
        # Damped update
        for alpha in [0.1, 0.05, 0.01, 0.005]:
            h_try = project_feasible(h + alpha * delta_h, target)
            c5_try = compute_c5(h_try)
            if c5_try < t:
                h = h_try
                break
        else:
            # No improvement, try smaller steps
            for alpha in [0.001, 0.0005, 0.0001]:
                h_try = project_feasible(h + alpha * delta_h, target)
                c5_try = compute_c5(h_try)
                if c5_try < t:
                    h = h_try
                    break
            else:
                break
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def swap_delta_c(h, i, j, n):
    ks = np.arange(-(n - 1), n)
    dc = np.zeros(2 * n - 1)
    m1 = i - ks
    v1 = (m1 >= 0) & (m1 < n)
    dc[v1] -= (1.0 - h[m1[v1]])
    m2 = j - ks
    v2 = (m2 >= 0) & (m2 < n)
    dc[v2] += (1.0 - h[m2[v2]])
    m3 = i + ks
    v3 = (m3 >= 0) & (m3 < n)
    dc[v3] += h[m3[v3]]
    m4 = j + ks
    v4 = (m4 >= 0) & (m4 < n)
    dc[v4] -= h[m4[v4]]
    return dc

def fast_binary_search(h_init, n, target, time_limit, n_cand=15):
    start = time.time()
    dx = 2.0 / n
    h = np.round(h_init).astype(float)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) > int(target):
        extra = len(ones) - int(target)
        to_remove = np.random.choice(ones, extra, replace=False)
        h[to_remove] = 0
    elif len(ones) < int(target):
        need = int(target) - len(ones)
        to_add = np.random.choice(zeros, need, replace=False)
        h[to_add] = 1
    best_h = h.copy()
    best_score = compute_c5(h)
    T = 5.0
    T_min = 0.01
    cooling = 0.9995
    no_improve = 0
    while time.time() - start < time_limit and no_improve < 300:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_cur = cur_max * dx
        if c5_cur < best_score:
            best_score = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        idxs = np.arange(n)
        top_k_idx = np.argsort(corr)[-6:]
        grad = np.zeros(n)
        for ki in top_k_idx:
            k = int(ki) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]

        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        cand_ones = ones[np.argsort(-grad[ones])[:n_cand]]
        cand_zeros = zeros[np.argsort(grad[zeros])[:n_cand]]
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for i in cand_ones:
            for j in cand_zeros:
                if i == j:

                dc = swap_delta_c(h, int(i), int(j), n)
                new_max = float(np.max(corr + dc))

                h[best_j] = 1.0

        except Exception:

        improved = False

            if c5_try < t_cur - 1e-12:

def smooth_obj_fn(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    dx = 2.0 / n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit - 5:
            break
        try:
            res = minimize(smooth_obj_fn, h_cur, args=(alpha, dx),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def multi_shift_gradient_step(h, n, target, n_shifts=5, lr=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    top_idx = np.argsort(corr)[-n_shifts:]
    alpha_w = 50.0
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w = w / w.sum()
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    h_new = project_feasible(h - lr * grad, target)
    return h_new, compute_c5(h_new)

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hae, scae = active_set_equalize(best_h, n_use, target_use, time_limit=min(15, tleft() * 0.5), n_active_max=8)
        if scae < best_score:
            best_score = scae
            best_h = hae

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
