    start = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h < 0.5)[0]
    
    rng = np.random.RandomState(42)
    
    # Precompute index arrays
    ks = np.arange(-(n-1), n)
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        
        # Temperature
        progress = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** progress
        
        # Pick random swap
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        # Compute delta
        idx_i_p = i + ks
        valid_i_p = (idx_i_p >= 0) & (idx_i_p < n)
        idx_i_m = i - ks
        valid_i_m = (idx_i_m >= 0) & (idx_i_m < n)
        idx_j_p = j + ks
        valid_j_p = (idx_j_p >= 0) & (idx_j_p < n)
        idx_j_m = j - ks
        valid_j_m = (idx_j_m >= 0) & (idx_j_m < n)
        
        delta = np.zeros(2*n - 1)
        delta[valid_i_p] += h[idx_i_p[valid_i_p]]
        delta[valid_i_m] += h[idx_i_m[valid_i_m]]
        delta[valid_j_p] -= h[idx_j_p[valid_j_p]]
        delta[valid_j_m] -= h[idx_j_m[valid_j_m]]
        
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        new_c5 = new_max * dx
        
        dE = new_c5 - cur_c5

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj_alpha(alpha):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    return obj

def _smooth_grad(x, alpha, top_m=80):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    top_m = min(top_m, 2 * n - 1)
    top_indices = np.argsort(w)[::-1][:top_m]
    grad = np.zeros(n)
    for ki in top_indices:
        ki = int(ki)
        wi = float(w[ki])
        if wi < 1e-15:
            break
        k = ki - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += wi * (1.0 - x[0:n-k])
                grad[0:n-k] -= wi * x[k:]
        else:
            kk = -k
            if n - kk > 0:
                grad[0:n-kk] += wi * (1.0 - x[kk:])
                grad[kk:] -= wi * x[0:n-kk]
    return grad

def _slq_con_fun(x, target):
    return np.sum(x) - target

def _slq_con_jac(x, target):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter=2000, ftol=1e-15, top_m=80):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
            res = minimize(obj, h_cur, jac=lambda x: _smooth_grad(x, alpha, top_m),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit, n_cand=25, max_sweeps=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def simple_pair_cd(h_init, n, target, time_limit, max_sweeps=200000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            break
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max < 1e-12:
            break
        h[i] -= delta_max
        h[j] += delta_max
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    return best_h, best_c5

def swap_delta_fast(h, i, j, n, ks):
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    idx_ip = i + ks
    vip = (idx_ip >= 0) & (idx_ip < n)
    idx_im = i - ks
    vim = (idx_im >= 0) & (idx_im < n)
    idx_jp = j + ks
    vjp = (idx_jp >= 0) & (idx_jp < n)
    idx_jm = j - ks
    vjm = (idx_jm >= 0) & (idx_jm < n)
    if np.any(vip):
        delta[vip] += h[idx_ip[vip]]
    if np.any(vim):
        delta[vim] += h[idx_im[vim]]
    if np.any(vjp):
        delta[vjp] -= h[idx_jp[vjp]]
    if np.any(vjm):
        delta[vjm] -= h[idx_jm[vjm]]
    return delta

def simulated_annealing(h_bin, n, target, time_limit, n_iter=300000, T_start=0.008, T_end=0.00002):
    start_t = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    ks = np.arange(-(n - 1), n)
    corr_len = 2 * n - 1
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()

    for it in range(n_iter):
        if time.time() - start_t > time_limit:
            break
        if it % 500 == 0:
            progress = it / max(n_iter - 1, 1)
            T = T_start * (T_end / T_start) ** progress
            ones = np.where(h > 0.5)[0].tolist()
            zeros = np.where(h < 0.5)[0].tolist()
        else:
            progress = it / max(n_iter - 1, 1)
            T = T_start * (T_end / T_start) ** progress

        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = swap_delta_fast(h, i, j, n, ks)
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        dE = (new_max - cur_max) * dx

        if dE < 0 or (T > 1e-10 and rng.random() < np.exp(-dE / T)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()

    return best_h, best_c5

def iterated_local_search(h_init, n, target, time_limit, n_rounds=8, perturb_size=4, ls_time=25):
    start_t = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(777)

    for rnd in range(n_rounds):
        round_budget = ls_time + 5
        if time.time() - start_t + round_budget > time_limit:
            break

        h_pert = best_h.copy()
        ones = np.where(h_pert > 0.5)[0]
        zeros = np.where(h_pert < 0.5)[0]
        for _ in range(perturb_size):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 1.0 - h_pert[i]
            h_pert[j] = 1.0 - h_pert[j]
            h_pert = project_feasible(h_pert, target)
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]

        h_ls, c5_ls = simple_pair_cd(h_pert, n, target, time_limit=ls_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_lag_subgrad(h_init, n, target, time_limit, max_steps=30000, n_lags=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    lr = 0.01

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        if cur_max * dx < best_c5 - 1e-13:
            best_c5 = cur_max * dx
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_lags:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1m = (idxs - k >= 0) & (idxs - k < n)
            v2m = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1m):
                grad[v1m] += float(wi) * (1.0 - h[idxs[v1m] - k])
            if np.any(v2m):
                grad[v2m] -= float(wi) * h[idxs[v2m] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = h - lr_mult * lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < cur_max * dx - 1e-14:
                h = h_try
                improved = True
                break

        if not improved:
            lr = max(lr * 0.9, 0.001)
            if step > 5000:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Pair CD from initial
    if tleft() > 180:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=150, n_cand=30, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 2: SLSQP with analytical gradient
    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000, 20000], maxiter=2000, ftol=1e-15, top_m=60)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 3: More pair CD
    if tleft() > 150:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=120, n_cand=35, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 4: Simulated annealing
    if tleft() > 200:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        sa_time = min(150, tleft() * 0.4)
        hsa, c5sa = simulated_annealing(h_bin, n, target, time_limit=sa_time,
                                         n_iter=400000, T_start=0.01, T_end=0.00001)
        if c5sa < best_score:
            hcd2, c5cd2 = simple_pair_cd(hsa, n, target, time_limit=20)
            if c5cd2 < best_score:
                best_score = c5cd2
                best_h = hcd2
            elif c5sa < best_score:
                best_score = c5sa
                best_h = hsa

    # Phase 5: Iterated local search
    if tleft() > 200:
        h_il
        pert_size = 3
        h_il, c5_il = iterated_local_search(best_h, n, target, time_limit=min(180, tleft() * 0.5),
                                             n_rounds=6, perturb_size=pert_size, ls_time=20)
        if c5_il < best_score:
            best_score = c5_il
            best_h = h_il

    # Phase 6: SLSQP polish
    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], maxiter=1500, ftol=1e-15, top_m=50)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Final pair CD
    if tleft() > 80:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(60, tleft() * 0.4), n_cand=40, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 8: Try higher resolution n=800
    if tleft() > 250:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, n_new, target_new, [200, 1000, 5000], maxiter=1000, ftol=1e-14, top_m=50)
        if tleft() > 120:
            hcd, c5cd = fast_pair_cd(hp, n_new, target_new, time_limit=min(100, tleft() * 0.4),
                                     n_cand=30, max_sweeps=50000)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 9: Multi-lag subgradient
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hsg, c5sg = multi_lag_subgrad(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.4),
                                       max_steps=20000, n_lags=5)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 10: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [50000, 200000], maxiter=1000, ftol=1e-15, top_m=40)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=min(20, tleft() * 0.5),
                                 n_cand=30, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
