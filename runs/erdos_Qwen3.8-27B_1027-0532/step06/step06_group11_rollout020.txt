import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, n):
    G = np.zeros((n, 2 * n - 1))
    idx = np.arange(n)
    for i in range(n):
        k_vals = idx - (n - 1)
        p_minus_k = i - k_vals
        mask1 = (p_minus_k >= 0) & (p_minus_k < n)
        G[i, mask1] += (1.0 - h[p_minus_k[mask1]])
        p_plus_k = i + k_vals
        mask2 = (p_plus_k >= 0) & (p_plus_k < n)
        G[i, mask2] -= h[p_plus_k[mask2]]
    return G

def compute_grad_matrix_fast(h, n):
    G = np.zeros((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    lags = idx - (n - 1)
    for i in range(n):
        pmk = i - lags
        m1 = (pmk >= 0) & (pmk < n)
        G[i, m1] += (1.0 - h[pmk[m1]])
        ppk = i + lags
        m2 = (ppk >= 0) & (ppk < n)
        G[i, m2] -= h[ppk[m2]]
    return G

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def find_optimal_delta(corr, dcorr, delta_max):
    if delta_max < 1e-14:
        return 0.0
    a, b = 0.0, delta_max
    for _ in range(16):
        mid = 0.5 * (a + b)
        vals = corr + mid * dcorr
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-13
        if active.any():
            f_prime = np.max(dcorr[active])
        else:
            f_prime = 0.0
        if f_prime > 1e-13:
            b = mid
        else:
            a = mid
    return 0.5 * (a + b)

def fast_pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr))
        G = compute_grad_matrix_fast(h, n)
        grad_active = G[:, k_star]
        grad_active -= np.mean(grad_active)
        sorted_idx = np.argsort(grad_active)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_move = None
        for i in high_set:
            gi = G[i]
            for j in low_set:
                if i == j:
                    continue
                dcorr = G[j] - gi
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-14:
                    continue
                delta = find_optimal_delta(corr, dcorr, delta_max)
                if delta < 1e-12:
                    continue
                trial = float(np.max(corr + delta * dcorr)) * dx
                if trial < best_trial:
                    best_trial = trial
                    best_move = (i, j, delta)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        c_max = np.max(corr)
        weights = np.exp(30.0 * (corr - c_max))
        wsum = weights[top_idx].sum()
        if wsum < 1e-300:
            wsum = 1.0
        top_weights = weights[top_idx] / wsum
        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, top_weights):
            k = int(s_idx) - mid
            grad += w * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def project_max_cd(h_start, n, target, time_limit, max_sweeps, n_cand, n_active):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def random_restart_optimize(h_base, n, target, rng, max_tries, time_limit, refine_alphas):
    start = time.time()
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    for t in range(max_tries):
        if time.time() - start > time_limit:
            break
        noise_scale = float(rng.choice([0.02, 0.05, 0.1, 0.15, 0.2]))
        h_pert = h_base + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_optimize(h_pert, n, target, refine_alphas, 500, 1e-14)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 180:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [200, 1000, 5000, 20000],
                                   1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_adam, sc = adam_optimize(best_h, n, target,
                                   time_limit=min(100, max(30, tleft() * 0.15)),
                                   max_steps=20000, n_top_start=3, n_top_end=20,
                                   lr=0.02, beta1=0.8, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    if tleft() > 250:
        h_cd, sc = fast_pair_cd(best_h, n, target,
                                time_limit=min(120, max(40, tleft() * 0.2)),
                                max_sweeps=10000, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 150:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [5000, 20000, 100000],
                                   1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_scd, sc = smoothed_pair_cd(best_h, n, target,
                                     time_limit=min(120, max(40, tleft() * 0.2)),
                                     max_sweeps=10000, n_cand=18, n_top=8)
        if sc < best_score:
            best_score = sc
            best_h = h_scd

    if tleft() > 150:
        h_pmc, sc = project_max_cd(best_h, n, target,
                                   time_limit=min(100, max(30, tleft() * 0.2)),
                                   max_sweeps=8000, n_cand=18, n_active=6)
        if sc < best_score:
            best_score = sc
            best_h = h_pmc

    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [10000, 50000, 200000],
                                   1200, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        h_rr, sc = random_restart_optimize(best_h, n, target, rng,
                                            max_tries=8, time_limit=min(80, tleft() * 0.4),
                                            refine_alphas=[500, 2000, 10000])
        if sc < best_score:
            best_score = sc
            best_h = h_rr

    if tleft() > 180:
        h_cd2, sc = fast_pair_cd(best_h, n, target,
                                 time_limit=min(100, max(30, tleft() * 0.3)),
                                 max_sweeps=8000, n_cand=25)
        if sc < best_score:
            best_score = sc
            best_h = h_cd2

    if tleft() > 200:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = slsqp_optimize(h_interp, n_try, target_try,
                                     [100, 500, 2000, 8000],
                                     1500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try

    if tleft() > 100:
        h_cd3, sc = fast_pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                 time_limit=min(60, max(20, tleft() * 0.3)),
                                 max_sweeps=6000, n_cand=15)
        if sc < best_score:
            best_score = sc
            best_h = h_cd3

    if tleft() > 80:

        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [50000, 200000, 500000],
                                     1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0

        if sc < best_score:
            best_score = sc
            best_h = h_cd4

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
