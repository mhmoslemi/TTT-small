    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha)) * dx

    return _smooth_n

def _g_target_global():
    return _smooth_target

_smooth_n = 0
_smooth_target = 0.0

def smooth_con_fun(h):
    return np.sum(h) - _smooth_target

def smooth_con_jac(h):
    return np.ones(_smooth_n)

def slsqp_smooth(h_start, n, target, alphas, maxiter=5000, ftol=1e-15):
    global _smooth_n, _smooth_target
    _smooth_n = n
    _smooth_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_con_fun, 'jac': smooth_con_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def frank_wolfe_ls(h_init, n, target, time_limit, K=50, ls_points=20):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    n_half = n - 1
    for t in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * (2.0 / n)
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            if k == 0:
                g += (1.0 - 2.0 * h)
            elif k > 0:
                g[:n - k] += (1.0 - h[k:])
                g[k:] += (-h[:n - k])
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m])
                g[:n - m] += (-h[m:])
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        best_gg = 0
        best_gamma = 0
        for gamma in np.linspace(0.01, 1.0, ls_points):
            h_try = (1.0 - gamma) * h + gamma * v
            h_try = project_feasible(h_try, target)
            gg = float(np.dot(h_try - v, g))
            if gg < best_gg:
                best_gg = gg
                best_gamma = gamma
        h = (1.0 - best_gamma) * h + best_gamma * v
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            best_t = t
        if t > 5000 and t - best_t > 50000:
            break
    return best_h, best_c5

def alternating_min_with_lp(h_init, n, target, time_limit, n_top=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(5000):
        if time.time() - start > time_limit:
            break
        if it > 50 and stuck > 150:
            break
        g_vec = 1.0 - h
        corr = np.correlate(g_vec, h, mode='full')
        overlaps_linear = (np.sum(h) * 1.0) * dx - corr * dx
        n_half = n - 1
        top_idx = np.argsort(overlaps_linear)[::-1][:n_top]
        lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = g_vec * dx
                constraints.append(coeff @ h_var <= t_var)
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = g_vec[k:] * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = g_vec[:n - m] * dx
                constraints.append(coeff @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.5 * h + 0.5 * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def block_refine(h_init, n, target, time_limit, block_size=8):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_blocks = n // block_size
    bounds_list = [(0.0, 1.0)] * block_size
    while time.time() - start < time_limit:
        improved = False
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            s = b * block_size
            e = s + block_size
            if e > n:
                continue
            h_fixed_sum = np.sum(h) - np.sum(h[s:e])
            block_target = target - h_fixed_sum
            if block_target < 0 or block_target > block_size:
                continue
            h_block = h[s:e].copy()
            h_rest = h.copy()
            
            def block_obj(x, s=s, e=e, h_rest_full=h):
                h_try = h_rest_full.copy()
                h_try[s:e] = x
                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                return float(np.max(corr)) * (2.0 / n)
            
            def block_con_fun(x):
                return np.sum(x) - block_target
            
            cons = [{'type': 'eq', 'fun': block_con_fun, 'jac': lambda x: np.ones(block_size)}]
            try:
                res = minimize(block_obj, h_block, method='SLSQP',
                               bounds=bounds_list, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
                if res.success or res.x is not None:
                    h_new_block = np.clip(res.x, 0.0, 1.0)
                    h[s:e] = h_new_block
                    h = project_feasible(h, target)
                    c5_val = compute_c5(h)
                    if c5_val < best_c5 - 1e-14:
                        best_c5 = c5_val
                        best_h = h.copy()
                        improved = True
            except Exception:
                pass
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _smooth_n, _smooth_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Frank-Wolfe with line search (120s)
    tl = budget_s - (time.time() - start_time)
    if tl > 150:
        fw_time = min(120, (tl - 30) * 0.15)
        h_fw, c5_fw = frank_wolfe_ls(best_h, n, target, fw_time, K=40, ls_points=15)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw
    
    # Phase 2: Smart SA on binary (400s)
    tl = budget_s - (time.time() - start_time)
    if tl > 200:
        b_start = to_binary(best_h, target)
        sa_time = min(200, (tl - 30) * 0.25)
        b_best, e_best = sa_binary(b_start, n, target, sa_time, t_start=3.0, t_end=0.01, rng=rng)
        c5_binary = e_best * (2.0 / n)
        if c5_binary < best_score:
            best_score = c5_binary
            best_h = b_best.copy()
    
    # Phase 3: Smart SA with guided moves (300s)
    tl = budget_s - (time.time() - start_time)
    if tl > 150:
        b_start = to_binary(best_h, target)
        sa_time = min(150, (tl - 30) * 0.2)
        b_best2, e_best2 = smart_sa_binary(b_start, n, target, sa_time, rng=rng)
        c5_binary2 = e_best2 * (2.0 / n)
        if c5_binary2 < best_score:

            best_h = b_best2.copy()
    
    # Phase 4: Projected gradient with momentum (200s)

        h_pg, c5_pg = projected_gradient_subgrad(best_h, n, target, pg_time, K=30, lr_init=0.005, lr_decay=0.9997)
        if c5_pg < best_score:

            best_h = h_pg
    
    # Phase 5: SLSQP with smooth max (200s)
    tl = budget_s - (time.time() - start_time)
    if tl > 100:
        _smooth_n = n
        _smooth_target = target
        h_slq = slsqp_smooth(best_h, n, target, [50, 200, 1000, 5000], maxiter=3000, ftol=1e-15)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq
    
    # Phase 6: Block refinement (200s)
    tl = budget_s - (time.time() - start_time)
    if tl > 100:
        bl_time = min(100, (tl - 30) * 0.15)
        h_bl, c5_bl = block_refine(best_h, n, target, bl_time, block_size=6)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl
    
    # Phase 7: More SA from best (if binary helped)
    tl = budget_s - (time.time() - start_time)
    if tl > 150:
        b_start = to_binary(best_h, target)
        sa_time = min(100, (tl - 30) * 0.15)
        b_best3, e_best3 = smart_sa_binary(b_start, n, target, sa_time, rng=rng)
        c5_binary3 = e_best3 * (2.0 / n)
        if c5_binary3 < best_score:
            best_score = c5_binary3
            best_h = b_best3.copy()
    
    # Phase 8: SLSQP high alpha polish (150s)
    tl = budget_s - (time.time() - start_time)
    if tl > 80:
        _smooth_n = n
        _smooth_target = target
        h_polish = slsqp_smooth(best_h, n, target, [2000, 10000, 50000, 200000], maxiter=4000, ftol=1e-16)
        sc = compute_c5(h_polish)
        if sc < best_score:
            best_score = sc
            best_h = h_polish
    
    # Phase 9: FW from current best (100s)
    tl = budget_s - (time.time() - start_time)
    if tl > 80:
        fw_time = min(60, (tl - 30) * 0.15)
        h_fw2, c5_fw2 = frank_wolfe_ls(best_h, n, target, fw_time, K=50, ls_points=15)
        if c5_fw2 < best_score:
            best_score = c5_pw = c5_fw2
            best_h = h_fw2
    
    # Phase 10: Final SLSQP
    tl = budget_s - (time.time() - start_time)
    if tl > 60:
        _smooth_n = n
        _smooth_target = target
        h_final_try = slsqp_smooth(best_h, n, target, [10000, 50000, 200000, 1000000], maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:

            best_h = h_final_try
    
    # Phase 11: Final block refinement

        if c5_bl2 < best_score:
            best_score = c5_bl2
            best_h = h_bl2
    
    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, len(h_final)
