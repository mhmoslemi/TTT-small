def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def minimize_max_linear(A, B, lo=0.0, hi=1.0, iters=40):
    """Minimize f(x) = max_i(A[i] + B[i]*x) for x in [lo, hi]"""
    for _ in range(iters):
        if hi - lo < 1e-15:
            break

        max_val = np.max(vals)

            lo = mid
        else:
            break

def active_set_cd(h_start, n, target, time_limit, max_sweeps, active_thresh_fn):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Determine active set
        c_max = np.max(corr)
        thresh = active_thresh_fn(sweep)
        active = np.where(corr >= c_max - thresh)[0]
        active_shifts = active - mid
        
        # Coordinate descent
        for j in range(n):
            idx_m = j - active_shifts
            idx_p = j + active_shifts
            
            vm = (idx_m >= 0) & (idx_m < n)
            vp = (idx_p >= 0) & (idx_p < n)
            
            h_m = np.where(vm, h[np.clip(idx_m, 0, n-1)], 0.0)
            h_p = np.where(vp, h[np.clip(idx_p, 0, n-1)], 0.0)
            
            B = 1.0 - h_m - h_p
            A = corr[active] - B * h[j]
            
            x_opt = minimize_max_linear(A, B, 0.0, 1.0)
            h[j] = x_opt
        
        h = project_feasible(h, target)
    
    return best_h, best_c5

Hmm, this is getting complicated. Let me think about it more carefully.

C_s = Σ_n h[n+s](1-h[n]) over valid n.

For a block of coordinates {j_start, ..., j_end-1}, let's denote the block variables as x_0, ..., x_{m-1} where x_i = h[j_start + i].

C_s = Σ_n h[n+s](1-h[n])

This sum includes terms where:
- n is in the block: n = j_start + i for some i. Then the term is h[n+s](1-x_i).
  - If n+s is also in the block: h[n+s] = x_{n+s-j_start}, so the term is x_{n+s-j_start}(1-x_i). This is bilinear in block variables!
  - If n+s is not in the block: h[n+s] is fixed, so the term is h[n+s](1-x_i), which is linear.
- n+s is in the block but n is not: n+s = j_start + i, so n = j_start + i - s. The term is x_i(1-h[n]), which is linear (h[n] is fixed since n is not in the block).
- Neither n nor n+s is in the block: term is fixed (contributes to const).

So the issue is terms where BOTH n and n+s are in the block. These are bilinear.

For block size m=1, there are no bilinear terms (as we showed). For m=2, there could be bilinear terms if s is small enough that both n and n+s fall in the block.

To avoid bilinear terms, we need the block to be small relative to the minimum |s| in the active set. If the minimum |s| is, say, 5, then a block of size ≤ 5 would have no bilinear terms.

But in practice, the active set likely includes small shifts (s=0, ±1, ±2, ...), so this approach doesn't work well.

Alternative: use the bilinear form directly. For a block of size m, C_s is a quadratic function of the block variables. The minimax of quadratic functions is not a linear program, but for small m, we can solve it with a general nonlinear solver.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def minimize_max_linear(A, B, lo, hi, iters):
    for _ in range(iters):
        if hi - lo < 1e-15:
            break
        mid = 0.5 * (lo + hi)
        vals = A + B * mid
        max_val = np.max(vals)
        active_mask = (vals >= max_val - 1e-12)
        min_B = np.min(B[active_mask])
        max_B = np.max(B[active_mask])
        if min_B > 1e-10:
            hi = mid
        elif max_B < -1e-10:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def adam_topk(h_start, n, target, time_limit, max_steps, n_top, lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            idx_m = np.arange(n) - k
            idx_p = np.arange(n) + k
            vm = (idx_m >= 0) & (idx_m < n)
            vp = (idx_p >= 0) & (idx_p < n)
            grad[vm] += 1.0 - h[idx_m[vm]]
            grad[vp] -= h[idx_p[vp]]
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)

        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                          max_sweeps=4000,
                                          base_thresh=0.01,
                                          decay=0.002)
        if sc_r < best_score:
            best_score = sc_r
            best_h = h_final_cd

    if tleft() > 150:
        h_cd, sc = active_set_cd(best_h, n, target,
                                  time_limit=min(100, tleft() * 0.4),
                                  max_sweeps=6000,
                                  base_thresh=0.01,
                                  decay=0.001)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 100:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = active_set_cd(h_interp, n_try, target_try,
                                     time_limit=min(80, tleft() * 0.35),
                                     max_sweeps=4000,
                                     base_thresh=0.015,
                                     decay=0.002)
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try
        h_pol, sc = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                    [5000, 50000, 200000],
                                    1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 60:
        n_use = len(best_h)
        h_final, sc = slsqp_optimize(best_h, n_use, n_use / 2.0,
                                      [50000, 200000, 1000000, 5000000],
                                      1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 30:
        n_use = len(best_h)
        h_final, sc = active_set_cd(best_h, n_use, n_use / 2.0,
                                     time_limit=min(25, tleft() * 0.5),
                                     max_sweeps=3000,
                                     base_thresh=0.008,
                                     decay=0.001)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    best_h = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_use
