            i_lo2, i_hi2 = 0, n - 1 + k
        if i_lo2 <= i_hi2:
            G[i_lo2:i_hi2+1, si] -= h[i_lo2-k:i_hi2-k+1]
    return G

def fast_cd(h_start, n, target, time_limit, n_cand=20, bisection_steps=8, beta=30.0, max_no_improve=150):
    start_t = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    no_improve = 0
    while time.time() - start_t < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Compute G
        G = compute_G(h, n)
        
        # Soft gradient
        c_max = np.max(corr)
        shifted = np.clip(beta * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[n-1] = 1.0
        else:
            weights /= wsum
        
        grad = G.T @ weights
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Evaluate all pairs
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_c5 = c5_val
        
        for ii in range(n_cand):
            i = high_set[ii]
            for jj in range(n_cand):
                j = low_set[jj]
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                dg = G[j, :] - G[i, :]
                
                a_d = 0.0
                b_d = delta_max
                for _ in range(bisection_steps):
                    mid_d = 0.5 * (a_d + b_d)
                    new_vals = corr + mid_d * dg
                    f_mid = np.max(new_vals)
                    active = new_vals >= f_mid - 1e-12
                    f_prime = np.max(dg[active])
                    if f_prime > 0:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue
                
                new_c5 = np.max(corr + delta_star * dg) * dx
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Fix sum
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                if diff > 0:
                    idx = np.where(h < 1.0 - 1e-10)[0]
                else:
                    idx = np.where(h > 1e-10)[0]
                if len(idx) > 0:
                    h[idx] += diff / len(idx)
                    h = np.clip(h, 0.0, 1.0)
    
    return best_h, best_c5

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted)) / alpha)) * dx

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    h_cur = h_start.copy()
    start_t = time.time()
    
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def toepmult_fft(lam_arr, h, n):
    """Compute B@h where B is Toeplitz with B[i,j] = lam[j-i]."""
    lam_rev = lam_arr[::-1]
    N = 1 << (3 * n - 2).bit_length()
    H = fft(h, N)
    L = fft(lam_rev, N)
    result = ifft(H * L).real
    return result[n-1 : 2*n-1]

def dual_ascent(h_start, n, target, time_limit, beta_start=5.0, beta_end=200.0, n_outer=50, n_inner=200):
    start_t = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for outer in range(n_outer):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Compute dual weights
        beta = beta_start * (beta_end / beta_start) ** (outer / max(n_outer - 1, 1))
        shifted = np.clip(beta * (corr - np.max(corr)), -500.0, 0.0)
        lam = np.exp(shifted)
        lsum = lam.sum()
        if lsum < 1e-300:
            lam = np.zeros(2*n-1)
            lam[n-1] = 1.0
        else:
            lam /= lsum
        
        # Inner: projected gradient ascent on h^T B h
        step = 0.4  # step size
        for inner in range(n_inner):
            if time.time() - start_t > time_limit:
                break
            Bh = toepmult_fft(lam, h, n)
            grad = 2.0 * Bh
            h_new = h + step * grad
            h_new = project_feasible(h_new, target)
            h = h_new
        
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
