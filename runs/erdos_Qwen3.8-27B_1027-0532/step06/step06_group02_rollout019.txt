So: grad = alpha * [conv(1-h, p_centered) - corr(h, p_centered)]

where conv is standard convolution and corr is cross-correlation.

Both can be computed with FFT in O(n log n)!

Let me verify: 
- conv(1-h, p)[j] = sum_k (1-h)[j-k] * p[k] = sum_k (1-h)[j-k] * p[k]
  This is what I called T1 above (with p[k] = w[k+(n-1)])

- The second term: sum_k p[k] * h[j+k]
  Let's define q[k] = p[-k] (reversed p)
  Then sum_k p[k] * h[j+k] = sum_k q[-k] * h[j+k] = sum_m q[m] * h[j-m] (substituting m=-k)
  = conv(h, q)[j] = conv(h, p_reversed)[j]

So grad = alpha * [conv(1-h, p) - conv(h, p_reversed)]

Both convolutions can be done with FFT!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def sparse_lse_grad(corr, alpha, top_k=100):
    """Compute sparse LSE gradient weights for top_k lags."""
    n = len(corr) // 2 + len(corr) % 2 - 1  # this is just n, let me compute differently
    # corr has length 2n-1
    n = (len(corr) + 1) // 2
    top_idx = np.argsort(corr)[::-1][:top_k]
    max_c = corr.max()
    shifted = alpha * (corr[top_idx] - max_c)
    w = np.exp(shifted)
    w = w / w.sum()
    return top_idx, w

def compute_grad_from_weights(h, top_idx, w, alpha):
    """Compute gradient given pre-computed top indices and weights."""
    n = len(h)
    grad = np.zeros(n)
    n_half = (n - 1)
    for t in range(len(top_idx)):
        idx = top_idx[t]
        wi = w[t]
        if wi < 1e-15:
            continue
        k = idx - n_half
        # T1: (1-h[j-k]) for valid j-k in [0,n)
        if k >= 0:
            js, je = k, n
            ms = 0
        else:
            js, je = 0, n + k
            ms = -k
        L = je - js
        grad[js:je] += wi * (1.0 - h[ms:ms+L])
        # T2: -h[j+k] for valid j+k in [0,n)
        if k >= 0:
            js2, je2 = 0, n - k
            ms2 = k
        else:
            js2, je2 = -k, n
            ms2 = 0
        L2 = je2 - js2
        grad[js2:je2] -= wi * h[ms2:ms2+L2]
    return alpha * grad

def adam_continuous(h0, alpha_schedule, start_time, time_limit, top_k_schedule=None):
    """Adam optimizer for the LSE-smoothed objective."""
    n = len(h0)
    h = h0.astype(float).copy()
    target = n / 2.0
    h = project_box_sum(h, target)
    
    best_h = h.copy()
    corr = full_corr_fast(h)
    best_score = float(corr.max()) * (2.0 / n)
    
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit - 10:
            break
        top_k = top_k_schedule.get(alpha, 100) if top_k_schedule else 100
        
        # Adaptive learning rate
        lr = 2.0 / (alpha + 1.0)
        
        n_iters = 20000
        for it in range(n_iters):
            if time.time() - start_time > time_limit - 10:

            t += 1
            corr = full_corr_fast(h)
            top_idx, w = sparse_lse_grad(corr, alpha, top_k)
            grad = compute_grad_from_weights(h, top_idx, w, alpha)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project
            h = project_box_sum(h, target)
            
            # Check improvement (every 100 iters)

                corr = full_corr_fast(h)
                score = float(corr.max()) * (2.0 / n)
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def sparse_top_weights(corr, alpha, top_k):
    top_idx = np.argsort(corr)[::-1][:top_k]
    max_c = corr.max()
    shifted = alpha * (corr[top_idx] - max_c)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(top_k)
        w[0] = 1.0
    else:
        w = w / s
    return top_idx, w

def compute_grad_weights(h, top_idx, w, alpha):
    n = len(h)
    n_half = n - 1
    grad = np.zeros(n)
    for t in range(len(top_idx)):
        idx = top_idx[t]
        wi = w[t]
        if wi < 1e-15:
            continue
        k = idx - n_half
        if k >= 0:
            js = k
            je = n
            ms = 0
        else:
            js = 0
            je = n + k
            ms = -k
        L = je - js
        grad[js:je] += wi * (1.0 - h[ms:ms+L])
        if k >= 0:
            js2 = 0
            je2 = n - k
            ms2 = k
        else:
            js2 = -k
            je2 = n
            ms2 = 0
        L2 = je2 - js2
        grad[js2:je2] -= wi * h[ms2:ms2+L2]
    return alpha * grad

def adam_continuous(h0, alpha_schedule, top_k_map, start_time, time_limit, n_inner=50000):
    n = len(h0)
    target = n / 2.0
    h = project_box_sum(h0.astype(float).copy(), target)
    dx = 2.0 / n

    corr = full_corr_fast(h)
    best_score = float(corr.max()) * dx
    best_h = h.copy()

    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    stall = 0
    max_stall = 5000

    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit - 10:
            break
        top_k = top_k_map.get(alpha, 80)
        lr = 3.0 / (alpha + 1.0)
        m = np.zeros(n)
        v = np.zeros(n)

        for it in range(n_inner):
            if time.time() - start_time > time_limit - 10:

            t_count += 1
            corr = full_corr_fast(h)
            top_idx, w = sparse_top_weights(corr, alpha, top_k)
            grad = compute_grad_weights(h, top_idx, w, alpha)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_count)
            v_hat = v / (1.0 - beta2 ** t_count)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)

                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0

def sa_swap(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)

    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 40000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t_temp = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if it % 3 == 0:
            max_pos = int(np.argmax(full))
            k_star = max_pos - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx_p = positions + k_star
                valid = (idx_p >= 0) & (idx_p < n)
                A[valid] += h[idx_p[valid]]
                idx_m = positions - k_star
                valid = (idx_m >= 0) & (idx_m < n)
                A[valid] += h[idx_m[valid]]
                i_cands = ones[A[ones] <= 0.5]
                if len(i_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j, k_arr)
        new_full = full + delta
        c5_new = float(new_full.max()) * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t_temp > 1e-15 and rng.random() < np.exp(-delta_c5 / t_temp)):
            h[i] = 0.0
            h[j] = 1.0
            full = new_full
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def sa_block_moves(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t_temp = t_init * (t_final / t_init) ** ratio

        if rng.random() < 0.3:
            blen = rng.randint(3, min(15, n // 10) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t_temp > 1e-15 and rng.random() < np.exp(-delta_c5 / t_temp)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            num_swaps = rng.randint(1, 4)
            h_new = h.copy()
            for _ in range(num_swaps):
                ones_c = np.where(h_new == 1.0)[0]
                zeros_c = np.where(h_new == 0.0)[0]
                if len(ones_c) < 2 or len(zeros_c) < 2:
                    break
                i = ones_c[rng.randint(len(ones_c))]
                j = zeros_c[rng.randint(len(zeros_c))]
                h_new[i] = 0.0
                h_new[j] = 1.0
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t_temp > 1e-15 and rng.random() < np.exp(-delta_c5 / t_temp)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def perturb_binary(h, rng, n_pert):
    h_p = h.copy()
    ones = np.where(h_p == 1.0)[0]
    zeros = np.where(h_p == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_p[i] = 0.0
        h_p[j] = 1.0
        ones = np.where(h_p == 1.0)[0]
        zeros = np.where(h_p == 0.0)[0]
    return h_p

def perturb_continuous(h, rng, strength):
    h_p = h.copy()
    noise = rng.randn(len(h)) * strength
    h_p = h_p + noise
    target = sum_bin(h, len(h))
    return project_box_sum(h_p, target)

def sum_bin(h, n):
    return n / 2.0

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: Strong continuous Adam optimization
    alpha_phase1 = [10, 20, 50, 100, 200, 500, 1000, 2000]
    topk1 = {10: 200, 20: 150, 50: 120, 100: 80, 200: 50, 500: 30, 1000: 20, 2000: 10}
    allowed1 = min(350, remaining() * 0.3)
    if allowed1 > 30:
        h_c1, sc1 = adam_continuous(best_h, alpha_phase1, topk1, start_time, allowed1 + (time.time() - start_time), n_inner=40000)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_c1.copy()

    # Phase 2: Binary refinement
    if remaining() > 60:
        h_bin = project_binary(best_h, int(target_init))
        allowed2 = min(120, remaining() * 0.15)
        n_sa = int(min(4000000, allowed2 * 80000))
        h_s2, c5_2 = sa_swap(h_bin, n_sa, 0.025, 0.00003, rng, allowed2 + (time.time() - start_time), start_time, k_arr)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h_s2.copy()

    # Phase 3: Block SA on binary
    if remaining() > 50:
        h_bin3 = project_binary(best_h, int(target_init))
        allowed3 = min(80, remaining() * 0.1)
        n_sa3 = int(min(2000000, allowed3 * 40000))
        h_s3, c5_3 = sa_block_moves(h_bin3, n_sa3, 0.02, 0.0001, rng, allowed3 + (time.time() - start_time), start_time)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h_s3.copy()

    # Phase 4: Another continuous round with higher alpha
    if remaining() > 80:
        h_cont4 = project_box_sum(best_h, target_init)
        alpha_phase4 = [50, 100, 300, 800, 2000, 5000]
        topk4 = {50: 100, 100: 60, 300: 40, 800: 25, 2000: 15, 5000: 8}
        allowed4 = min(200, remaining() * 0.25)
        h_c4, sc4 = adam_continuous(h_cont4, alpha_phase4, topk4, start_time, allowed4 + (time.time() - start_time), n_inner=50000)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_c4.copy()

    # Phase 5: Binary from new continuous
    if remaining() > 50:
        h_bin5 = project_binary(best_h, int(target_init))
        allowed5 = min(80, remaining() * 0.12)
        n_sa5 = int(min(3000000, allowed5 * 80000))
        h_s5, c5_5 = sa_swap(h_bin5, n_sa5, 0.015, 0.00002, rng, allowed5 + (time.time() - start_time), start_time, k_arr)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h_s5.copy()

    # Phase 6: Multi-start with perturbations
    for restart in range(6):
        if remaining() < 50:
            break
        if restart % 3 == 0:
            h_pert = perturb_binary(best_h, rng, rng.randint(4, 10))
        elif restart % 3 == 1:
            h_pert = perturb_continuous(best_h, rng, 0.05)
        else:
            h_pert = perturb_binary(best_h, rng, 2)
            h_pert = project_box_sum(h_pert, target_init)

        alpha_ms = [50, 200, 800]
        topk_ms = {50: 80, 200: 40, 800: 20}
        allowed_ms = min(50, remaining() * 0.06)
        h_r, sc_r = adam_continuous(h_pert, alpha_ms, topk_ms, start_time, allowed_ms + (time.time() - start_time), n_inner=20000)
        if sc_r < best_score:
            best_score = sc_r
            best_h = h_r.copy()

        h_bin_r = project_binary(h_r, int(target_init))
        allowed_sa = min(25, remaining() * 0.04)
        n_sa_r = int(min(1000000, allowed_sa * 60000))
        h_sr, c5_sr = sa_swap(h_bin_r, n_sa_r, 0.01, 0.00002, rng, allowed_sa + (time.time() - start_time), start_time, k_arr)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Final continuous polish with very high alpha
    if remaining() > 40:
        h_final_c = project_box_sum(best_h, target_init)
        alpha_final = [500, 2000, 8000]
        topk_final = {500: 25, 2000: 12, 8000: 6}
        allowed_f = min(remaining() - 20, 100)
        h_pf, sc_pf = adam_continuous(h_final_c, alpha_final, topk_final, start_time, allowed_f + (time.time() - start_time), n_inner=50000)
        if sc_pf < best_score:
            best_score = sc_pf
            best_h = h_pf.copy()

    # Phase 8: Final SA burst
    if remaining() > 25:
        h_bin_f = project_binary(best_h, int(target_init))
        allowed_sf = max(10, remaining() - 12)
        n_sa_f = int(min(3000000, allowed_sf * 70000))
        h_sf, c5_sf = sa_swap(h_bin_f, n_sa_f, 0.006, 0.000005, rng, allowed_sf + (time.time() - start_time), start_time, k_arr)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Final: check both continuous and binary
    h_cont_out = project_box_sum(best_h, target_init)
    h_bin_out = project_binary(best_h, int(target_init))
    sc_cont = compute_c5(h_cont_out)
    sc_bin = compute_c5(h_bin_out)

    if sc_cont <= sc_bin:
        final_h = h_cont_out
        final_score = sc_cont
    else:
        final_h = h_bin_out
        final_score = sc_bin

    return final_h, final_score, n_init
