def project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} using bisection."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def adam_optimize(h0, target, start_time, time_limit, alphas, lr_initial=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, check_interval=20):
    """Adam optimization of the soft-max smoothed C5 objective."""
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = time_limit - (time.time() - start_time)
        if phase_time < 5:
            break
        
        # Determine iterations for this phase
        n_iter = int(min(phase_time * 300, 50000))
        n_iter = max(n_iter, 100)
        
        # Learning rate schedule: start high, decay
        lr = lr_initial / (1.0 + alpha * 0.01)
        
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        max_stall = 2000
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            if stall > max_stall:
                break
            
            t += 1
            corr = full_corr_fast(h)
            grad = smooth_gradient_fft(h, alpha, corr)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_box_sum(h_new, target)
            h = h_new
            
            if it % check_interval == 0 or it == n_iter - 1:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                # Adaptive lr: if stalling, reduce lr
                if stall == 500:
                    lr *= 0.5
                if stall == 1000:
                    lr *= 0.5
        
        # After each alpha phase, also try binary projection + local search
        h_bin = project_binary(h, int(np.round(target)))
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
    
    return best_h, best_c5

def sa_optimize(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr,
               block_frac=0.2, max_block_len=10):

    n = len(h0)
    h = h0.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

            a = rng.randint(0, n - blen + 1)

            # Try to find a new location
            a2 = rng.randint(0, n - blen + 1)
            if a2 == a:
                stall += 1
                continue
            # Compute the change
            h_new = h.copy()
            h_new[a2:a2+blen] = block
            h_new[a:a+blen] = 1.0 - block  # swap
            s_check = h_new.sum()
            if abs(s_check - h.sum()) > 1e-10:
                stall += 1
                continue
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new

                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    """Greedy descent: always make the best single swap."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        
        if k_star == 0:
            stall += 1
            continue
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find best swap
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                improvement = max_val - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            full += compute_delta(h + np.zeros(n), best_i, best_j, k_arr)  # recompute
            # Actually let me just recompute full
            full = full_corr_fast(h)
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Try random
            found = False
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2 * n - 1][::-1]

def smooth_gradient(h, alpha, corr):
    n = len(h)
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = p[k + n - 1]
        if pk < 1e-15:
            continue
        if k >= 0:
            grad[:n - k] += pk * (1.0 - h[k:])
            grad[k:] -= pk * h[:-k] if k < n else 0.0
        else:
            kk = -k
            grad[kk:] += pk * (1.0 - h[:n - kk])
            grad[:n - kk] -= pk * h[kk:]
    return grad

            if a2 == a:
                stall += 1
                continue
            block = h[a:a + blen].copy()
            h_new = h.copy()
            h_new[a2:a2 + blen] = block
            h_new[a:a + blen] = 1.0 - block
            if abs(h_new.sum() - h.sum()) > 1e-10:
                stall += 1
                continue
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        best_imp = 0.0
        best_i = -1
        best_j = -1
        top_i = ones[np.argsort(A[ones])[:min(40, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(40, len(zeros))]]
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                imp = max_val - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            full = full_corr_fast(h)
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            found = False
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    return best_h, best_c5

def focused_sa(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h0)
    h = h0.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 60000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 5 == 0:
            top_k = 4
            top_indices = np.argsort(full)[::-1][:top_k]
            best_move = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_cands = ones[np.argsort(A[ones])[:min(10, len(ones))]]
                j_cands = zeros[np.argsort(-B[zeros])[:min(10, len(zeros))]]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move = True
                    break
            if not best_move:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def perturb_diverse(h, rng, n_pert):
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 10) + 1)
    a = rng.randint(0, n - blen + 1)
    b = a + blen - 1
    h_pert[a:b + 1] = h[a:b + 1][::-1]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0
    target_int = int(n / 2.0)
    k_arr = np.arange(2 * n - 1) - (n - 1)

    best_h = project_binary(h_init, target_int)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam continuous optimization
    if tleft() > 180:
        alphas = [5, 12, 25, 50, 120]
        allowed = min(200, tleft() * 0.22)
        h_a1, c5_a1 = adam_optimize(best_h, target, start_time, allowed, alphas, lr_initial=0.008)
        if c5_a1 < best_c5:
            best_c5 = c5_a1
            best_h = h_a1.copy()

    # Phase 2: Greedy from current best binary
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        allowed = min(70, tleft() * 0.1)
        h_g1, c5_g1 = fast_greedy(h_bin, max_iters=30000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g1 < best_c5:
            best_c5 = c5_g1
            best_h = h_g1.copy()

    # Phase 3: SA with block moves
    if tleft() > 120:
        allowed = min(160, tleft() * 0.18)
        n_sa = int(min(3000000, allowed * 35000))
        h_s1, c5_s1 = sa_optimize(best_h, n_sa, 0.02, 0.00004, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                  block_frac=0.25, max_block_len=12)
        if c5_s1 < best_c5:
            best_c5 = c5_s1
            best_h = h_s1.copy()

    # Phase 4: Adam from improved binary
    if tleft() > 150:
        h_bin2 = project_binary(best_h, target_int)
        alphas2 = [25, 60, 150, 400]
        allowed = min(180, tleft() * 0.2)
        h_a2, c5_a2 = adam_optimize(h_bin2, target, start_time, allowed, alphas2, lr_initial=0.005)
        if c5_a2 < best_c5:
            best_c5 = c5_a2
            best_h = h_a2.copy()

    # Phase 5: Focused SA from Adam result
    if tleft() > 100:
        h_bin3 = project_binary(best_h, target_int)
        allowed = min(80, tleft() * 0.12)
        h_g2, c5_g2 = fast_greedy(h_bin3, max_iters=15000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    if tleft() > 100:
        allowed = min(120, tleft() * 0.15)
        n_sa2 = int(min(2500000, allowed * 30000))
        h_s2, c5_s2 = focused_sa(best_h, n_sa2, 0.012, 0.00002, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    # Phase 6: Multi-restart with diverse perturbations
    for restart in range(5):
        if tleft() < 80:
            break
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(4, 10))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 14)
        else:
            h_pert = perturb_diverse(best_h, rng, 3)
            h_pert = perturb_block(h_pert, rng, 8)
        allowed = min(60, tleft() * 0.08)
        h_gr, c5_gr = fast_greedy(h_pert, max_iters=8000, rng=rng,
                                  time_limit=allowed * 0.4, start_time=start_time, k_arr=k_arr)
        allowed2 = min(allowed * 0.6, tleft() - 20)
        n_sr = int(min(800000, allowed2 * 25000))
        h_sr, c5_sr = sa_optimize(h_gr, n_sr, 0.01, 0.00003, rng,
                                  time_limit=allowed2, start_time=start_time, k_arr=k_arr,
                                  block_frac=0.2, max_block_len=8)
        if c5_sr < best_c5:
            best_c5 = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Low-temperature SA burst
    if tleft() > 60:
        h_bin4 = project_binary(best_h, target_int)
        allowed = min(80, tleft() * 0.1)
        n_sa3 = int(min(2000000, allowed * 35000))
        h_s3, c5_s3 = focused_sa(h_bin4, n_sa3, 0.006, 0.000005, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_s3 < best_c5:
            best_c5 = c5_s3
            best_h = h_s3.copy()

    # Phase 8: High-alpha Adam polish
    if tleft() > 50:
        alphas3 = [200, 800, 3000, 10000]
        allowed = min(60, tleft() * 0.08)
        h_a3, c5_a3 = adam_optimize(project_binary(best_h, target_int), target,
                                    start_time, allowed, alphas3, lr_initial=0.003)
        if c5_a3 < best_c5:
            best_c5 = c5_a3
            best_h = h_a3.copy()

    # Phase 9: Final SA
    if tleft() > 30:
        h_bin5 = project_binary(best_h, target_int)
        allowed = max(15, tleft() - 12)
        n_sa4 = int(min(1500000, allowed * 30000))
        h_s4, c5_s4 = sa_optimize(h_bin5, n_sa4, 0.004, 0.000002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                  block_frac=0.1, max_block_len=6)
        if c5_s4 < best_c5:
            best_c5 = c5_s4
            best_h = h_s4.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
