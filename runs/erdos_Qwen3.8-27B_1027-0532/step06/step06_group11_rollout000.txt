def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def sbx_crossover(x1, x2, rng, prob, eta):
    if rng.uniform() > prob:
        return x1.copy(), x2.copy()
    n = len(x1)
    c1 = x1.copy()
    c2 = x2.copy()
    for i in range(n):
        if rng.uniform() < 0.5:
            if abs(x1[i] - x2[i]) > 1e-14:
                u = rng.uniform()
                if u <= 0.5:
                    beta = (2.0 * u) ** (1.0 / (eta + 1.0))
                else:
                    beta = (1.0 / (2.0 * (1.0 - u))) ** (1.0 / (eta + 1.0))
                c1[i] = 0.5 * ((1.0 + beta) * x1[i] + (1.0 - beta) * x2[i])
                c2[i] = 0.5 * ((1.0 - beta) * x1[i] + (1.0 + beta) * x2[i])
    return c1, c2

                delta = 1.0 - (2.0 * (1.0 - u)) ** (1.0 / (eta + 1.0))
            x[i] += delta * (bounds_hi[i] - bounds_lo[i])
    return np.clip(x, bounds_lo, bounds_hi)

            best_score = scores[0]

            s2 = compute_c5(c2)

            if len(new_pop) < pop_size:
                new_pop.append(c2)
        
        pop = new_pop
        scores = [compute_c5(p) for p in pop]
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_SLSQP_n = 0
_SLSQP_target = 0.0
_SLSQP_alpha = 0.0

def _obj_sl(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_SLSQP_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _SLSQP_alpha)

def _con_fun(h):
    return np.sum(h) - _SLSQP_target

def _con_jac(h):
    return np.ones(_SLSQP_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol):
    global _SLSQP_n, _SLSQP_target, _SLSQP_alpha
    _SLSQP_n = n
    _SLSQP_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best = compute_c5(h_cur)
    for alpha in alphas:
        _SLSQP_alpha = alpha
        try:
            res = minimize(_obj_sl, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best:
                best = c5
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best

def adam_refine(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end, lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = max(int(n_top_start + (n_top_end - n_top_start) * frac), 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def pairwise_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    no_improve = 0

        if time.time() - start > time_limit:

            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        n_shifts = 2 * n - 1
        f = np.full((n, n_shifts), -1.0)
        for p in range(n):
            idx1 = p + n - 1 - np.arange(n_shifts)
            mask1 = (idx1 >= 0) & (idx1 < n)
            f[p, mask1] += h[idx1[mask1]]
            idx2 = p + np.arange(n_shifts) - n + 1
            mask2 = (idx2 >= 0) & (idx2 < n)
            f[p, mask2] += h[idx2[mask2]]
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = f[i] - f[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])

                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)

                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def simulated_annealing(h_start, n, target, time_limit, max_steps, rng, t_start, t_end, cooling):

        if move_type == 0:
            center = rng.randint(n)
            width = int(rng.choice([3, 5, 8, 12, 20]))

            dist = np.abs(np.arange(n) - center)
            kernel = np.exp(-0.5 * (dist / max(1, width)) ** 2)
            h_try += amp * kernel
        elif move_type == 1:
            freq = rng.choice([2, 4, 5, 8, 10, 12])
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(-0.08, 0.08)
            h_try += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif move_type == 2:
            n_block = int(rng.choice([5, 10, 15, 20, 30]))
            start_idx = rng.randint(0, n - n_block + 1)
            idxs = np.arange(start_idx, start_idx + n_block)
            h_try[idxs] += rng.uniform(-0.1, 0.1)
        else:
            n_pert = int(rng.choice([3, 5, 8, 12, 20]))
            idxs = rng.choice(n, size=n_pert, replace=False)
            h_try[idxs] += rng.randn(n_pert) * 0.05
        h_try = project_feasible(h_try, target)

        delta = new_score - current_score

            current_score = new_score
            if new_score < best_score - 1e-13:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]
        if len(sig_idx) == 0:
            break
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_pol, sc = slsqp_refine(best_h, n, target,
                                  [200, 1000, 5000, 20000, 100000, 500000],
                                  2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        h_adam, sc = adam_refine(best_h, n, target,
                                 time_limit=min(80, tleft() * 0.3),
                                 max_steps=50000, n_top_start=3, n_top_end=20,
                                 lr=0.02, beta1=0.85, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam
        h_adam2, sc = adam_refine(best_h, n, target,
                                  time_limit=min(60, tleft() * 0.2),
                                  max_steps=30000, n_top_start=5, n_top_end=30,
                                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam2

    if tleft() > 100:
        h_cd, sc = pairwise_cd(best_h, n, target,
                               time_limit=min(120, tleft() * 0.3),
                               max_sweeps=5000, n_cand=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 80:
        h_pol, sc = slsqp_refine(best_h, n, target,
                                  [5000, 20000, 100000, 500000, 2000000],
                                  2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_sa, sc = simulated_annealing(best_h, n, target,
                                        time_limit=min(180, tleft() * 0.45),
                                        max_steps=50000, rng=rng,
                                        t_start=0.05, t_end=0.001, cooling=0.999)
        if sc < best_score:
            best_score = sc
            best_h = h_sa

    if tleft() > 150:
        h_pol, sc = slsqp_refine(best_h, n, target,
                                  [1000, 5000, 50000, 200000, 1000000],
                                  1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_ga, sc = ga_optimize(best_h, n, target,
                                time_limit=min(250, tleft() * 0.55),
                                pop_size=35, max_gens=20000,
                                n_elite=5, xover_prob=0.9, xover_eta=15.0,
                                mut_prob=0.5, mut_eta=20.0, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_ga

    if tleft() > 100:
        h_pol, sc = slsqp_refine(best_h, n, target,
                                  [2000, 10000, 50000, 200000, 1000000],
                                  2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 80:
        h_cd, sc = pairwise_cd(best_h, n, target,
                               time_limit=min(80, tleft() * 0.3),
                               max_sweeps=3000, n_cand=35)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 150:
        h_sa2, sc = simulated_annealing(best_h, n, target,
                                         time_limit=min(120, tleft() * 0.4),
                                         max_steps=40000, rng=rng,
                                         t_start=0.03, t_end=0.0005, cooling=0.999)
        if sc < best_score:
            best_score = sc
            best_h = h_sa2
        if sc < best_score + 0.001:
            h_pol, sc2 = slsqp_refine(h_sa2, n, target,
                                       [5000, 50000, 500000],
                                       1500, 1e-16)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol
            h_cd2, sc3 = pairwise_cd(h_sa2, n, target,
                                     time_limit=min(40, tleft() * 0.2),
                                     max_sweeps=2000, n_cand=30)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_cd2

    if tleft() > 120:
        n_coarse = 50
        target_coarse = n_coarse / 2.0
        for attempt in range(4):
            if tleft() < 60:
                break
            h_rand_coarse = np.clip(rng.normal(0.5, 0.3, n_coarse), 0, 1)
            h_rand_coarse = project_feasible(h_rand_coarse, target_coarse)
            h_coarse, sc = slsqp_refine(h_rand_coarse, n_coarse, target_coarse,
                                         [100, 500, 2000, 10000],
                                         500, 1e-14)
            h_interp = interpolate_to(h_coarse, n, target)
            h_fin, sc_fin = slsqp_refine(h_interp, n, target,
                                          [500, 2000, 10000, 50000],
                                          1000, 1e-15)
            if sc_fin < best_score:
                best_score = sc_fin
                best_h = h_fin

    if tleft() > 80:
        h_sg, sc = soft_weighted_subgrad(best_h, n, target,
                                         time_limit=min(50, tleft() * 0.3),
                                         max_steps=30000, alpha_w=50.0, base_lr=0.03)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    if tleft() > 60:
        h_cd, sc = pairwise_cd(best_h, n, target,
                               time_limit=min(40, tleft() * 0.5),
                               max_sweeps=2000, n_cand=40)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_refine(best_h, n_use, target_use,
                                    [50000, 200000, 1000000, 5000000],
                                    1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
