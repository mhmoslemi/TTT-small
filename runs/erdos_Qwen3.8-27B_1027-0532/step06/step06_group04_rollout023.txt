def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:

def coord_descent_full(h_init, target, n_active, max_passes, time_limit, rng):
    """Coordinate descent using top n_active constraints per coordinate."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            idxs = active_k
            slopes = np.empty(len(idxs))
            intercepts = np.empty(len(idxs))
            for idx in range(len(idxs)):
                k_idx = idxs[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            
            new_val = solve_1d_max_line_all(slopes, intercepts, 0.0)
            if abs(new_val - h[i]) > 1e-15:
                delta = new_val - h[i]
                # Update correlation
                for idx in range(len(idxs)):
                    k_idx = idxs[idx]
                    k = k_idx - half_n
                    A = slopes[idx]
                    corr[k_idx] += A * delta
                h[i] = new_val
        
        h = np.clip(h, 0.0, 1.0)
        # Project to feasible
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

            best_h = h.copy()

        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5

        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:

                # But we only track active_k, so this is an approximation

            corr = np.correlate(h, 1.0 - h, mode='full')

        active_k = new_active

Actually, I can do even better by precomputing the indices:

For each k (from -(n-1) to n-1):
- The (1-h[i-k]) term contributes to i in [max(0,k), min(n, n+k))
- The -h[i+k] term contributes to i in [max(0,-k), min(n, n-k))

This is the same as the A matrix structure. So:
grad = A^T @ p * dx

where A is the (2n-1) × n matrix I built before! And A^T @ p is just a matrix-vector product.

So: grad = (A.T @ p) * dx * ... wait, let me be more careful.

A[m, i] = dx * g[i + m - (n-1)] where g = 1-h

So A^T @ p = sum_m p[m] * A[m, i] = sum_m p[m] * dx * g[i + m - (n-1)]

But the gradient of corr[m] with respect to h[i] is:
d(corr[m])/dh[i] = dx * [(1-h[i-k]) if valid - h[i+k] if valid] where k = m-(n-1)

This is NOT the same as dx * g[i-k] = dx * (1-h[i-k]).

The difference is the second term: -dx * h[i+k].

So the full gradient matrix B has:
B[m, i] = dx * [(1-h[i-k]) if 0<=i-k<n - h[i+k] if 0<=i+k<n]

This is different from A (which only has the first term).

Let me build B:
```python
def build_grad_matrix(h, n, dx):
    """Build (2n-1) x n matrix B where B[m,i] = d(corr[m])/dh[i]"""
    B = np.zeros((2 * n - 1, n))
    half_n = n - 1
    for m in range(2 * n - 1):
        k = m - half_n
        # First term: (1-h[i-k]) for valid i-k
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            B[m, lo1:hi1] += (1.0 - h[lo1 + k:hi1 + k]) * dx
        # Second term: -h[i+k] for valid i+k
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            B[m, lo2:hi2] -= h[lo2 + k:hi2 + k] * dx
    return B
```

Then grad = B.T @ p (where p are the softmax weights).

This is a (2n-1)×n matrix times an (2n-1) vector, which is O(n²) = O(160K) for n=400. Very fast.

1. Choose α (start small, increase geometrically)
2. At each L-BFGS-B step:

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

        idx = np.where(h < 1.0 - 1e-10)[0]

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _smooth_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    p = exp_s / total
    grad = np.zeros(n)
    half_n = n - 1
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        k = m - half_n
        w = pm * dx
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 + k:hi2 + k]
    return grad

def _cons_eq(h, target):
    return np.sum(h) - target

def _cons_eq_jac(h, n):
    return np.ones(n)

def slsqp_smooth(h_start, n, target, alphas, maxiter, time_limit):
    global _global_alpha, _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac, 'args': (target, n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _global_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           jac=_smooth_grad,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score
