import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_grad_shift(h, k, n):
    """Gradient of corr at shift k with respect to h."""
    g = np.zeros(n)
    if k >= 0:
        # corr[k+n-1] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        # d/dh[i] = (1-h[i-k]) for i >= k (from h[i] as first factor at position i-k)
        #          - h[i-k] for i >= k ... wait
        # Let me redo:
        # corr at shift k: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j] for j < k: -h[j+k] (j appears as h[i] with i=j, and h[j+k] is the other factor)
        #   Wait: when i=j, the term is h[j+k]*(1-h[j]), d/dh[j] = -h[j+k]
        # d/dh[j] for j >= k and j <= n-1-k+k = n-1: 
        #   j appears as h[i+k] when i=j-k: term is h[j]*(1-h[j-k]), d/dh[j] = (1-h[j-k])
        #   j also appears as h[i] when i=j (if j <= n-1-k): term is h[j+k]*(1-h[j]), d/dh[j] = -h[j+k]
        # So: for j < k: d/dh[j] = -h[j+k] (only if j+k < n, which is true since j < k ≤ n-1)
        #     for k ≤ j ≤ n-1-k: d/dh[j] = (1-h[j-k]) - h[j+k]
        #     for j > n-1-k: d/dh[j] = (1-h[j-k]) (only if j-k >= 0, which is true since j >= k)

        for i in range(n - k):
            g[i] += -h[i + k]  # d/dh[i] of h[i+k]*(1-h[i]) = -h[i+k]
            g[i + k] += (1.0 - h[i])  # d/dh[i+k] of h[i+k]*(1-h[i]) = (1-h[i])
    else:
        k_abs = -k
        for i in range(n - k_abs):
            g[i + k_abs] += -h[i]  # h[i]*(1-h[i+k_abs]): d/dh[i+k_abs] = -h[i]
            g[i] += (1.0 - h[i + k_abs])  # d/dh[i] = (1-h[i+k_abs])
    return g

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(10):

            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]

            break

        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

def slsqp_pnorm(h_start, n, target, p_values, maxiter, ftol):
    """SLSQP optimization using p-norm smoothing."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 
             'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for p in p_values:
        def obj(h):
            corr = compute_correlation(h)
            dx = 2.0 / n
            return float(np.sum(np.abs(corr) ** p) ** (1.0/p) * dx)
        
        def obj_grad(h):
            corr = compute_correlation(h)
            dx = 2.0 / n
            # grad of (sum |corr|^p)^(1/p) * dx
            # = (1/p) * (sum |corr|^p)^(1/p - 1) * p * |corr|^(p-1) * sign(corr) * dcorr/dh * dx
            # = (sum |corr|^p)^(1/p - 1) * |corr|^(p-1) * sign(corr) * dcorr/dh * dx
            factor = (np.sum(np.abs(corr)**p))**(1.0/p - 1.0) * dx
            if factor < 1e-30:
                return np.zeros(n)
            # For each shift, compute dcorr/dh
            mid = n - 1
            grad = np.zeros(n)
            for s in range(2*n - 1):
                w = factor * np.abs(corr[s])**(p-1) * np.sign(corr[s])
                if abs(w) < 1e-15:
                    continue
                k = s - mid
                grad += w * get_grad_shift(h, k, n)
            return grad
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5

Actually, flipping h[i] changes:

- The first index: i appears as h[j] when j=i, affecting corr[m] where i+n-1-m is valid, i.e., m ∈ [i, i+n-1]
- The second index: i appears as h[j+n-1-m] when j+n-1-m=i, i.e., m=i+n-1-j, so m ranges over [i, i+n-1] as j ranges over [0,n-1]

So flipping h[i] affects corr[m] for m in some range of length at most n. We can update these in O(n) time.

This would make each SA step O(n) instead of O(n²), allowing 10x more steps in the same time.

When we flip h[i] from old_val to new_val (change = new_val - old_val):
For each affected m:
  corr[m] changes by: change * (contribution of h[i] to corr[m])

The contribution of h[i] to corr[m]:
- As first factor h[j] with j=i: (1-h[i+n-1-m]) if valid
- As second factor h[j+n-1-m] with j+n-1-m=i: -h[j] = -h[i+n-1-m] ... wait

Let me be more careful. corr[m] = Σ_j h[j] * (1-h[j+n-1-m])

If h[i] changes by δ:
- Term where j=i: h[i]*(1-h[i+n-1-m]) changes by δ*(1-h[i+n-1-m]) [if i+n-1-m is valid]
- Term where j+n-1-m=i, i.e., j = i+n-1-m: h[i+n-1-m]*(1-h[i]) changes by -h[i+n-1-m]*δ [if i+n-1-m is in [0,n-1] and j=i+n-1-m is in [0,n-1]]

Wait, j = i+n-1-m must be in [0, n-1], and also j+n-1-m = i must be in [0, n-1] (which is true since i is in [0,n-1]).

So the condition for the second term is: 0 ≤ i+n-1-m ≤ n-1, i.e., n-1-i ≤ m ≤ n-1+i.

And the condition for the first term is: 0 ≤ i+n-1-m ≤ n-1, same condition! n-1-i ≤ m ≤ n-1+i.

So both terms are active for m ∈ [max(0, n-1-i), min(2n-2, n-1+i)].

The change in corr[m] when h[i] changes by δ:
Δcorr[m] = δ * (1-h[i+n-1-m]) + δ * (-h[i+n-1-m]) ... wait no.

The first term contributes δ * (1-h[i+n-1-m]) to corr[m].
The second term: the term is h[j] * (1-h[i]) where j = i+n-1-m. When h[i] changes by δ, (1-h[i]) changes by -δ, so the term changes by h[j] * (-δ) = -h[i+n-1-m] * δ.

So total: Δcorr[m] = δ * (1-h[i+n-1-m]) - δ * h[i+n-1-m] = δ * (1 - 2*h[i+n-1-m])

Wait, that's only if both terms are present. But actually, for a given m, h[i] can appear in the sum at most twice:
1. As h[j] with j=i: the term is h[i]*(1-h[i+n-1-m]), present when i+n-1-m ∈ [0,n-1]
2. As h[j+n-1-m] with j=i+n-1-m: the term is h[i+n-1-m]*(1-h[i]), present when i+n-1-m ∈ [0,n-1]

Both are present under the same condition! So:
Δcorr[m] = δ*(1-h[i+n-1-m]) + h[i+n-1-m]*(-δ) = δ*(1-2*h[i+n-1-m])

This is valid for m ∈ [n-1-i, n-1+i] (clipped to [0, 2n-2]).

So the update is:
For m from max(0, n-1-i) to min(2n-2, n-1+i):
  corr[m] += δ * (1 - 2*h[i+n-1-m])

This is O(n) per flip, which is great!

        else:
            no_improve += 1
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, target)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP polish at current resolution
    if tleft() > 100:
        h_pol, sc = slsqp_opt(best_h, n, target,
                              [200, 1000, 5000, 20000, 100000, 500000, 2000000],
                              1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Adam optimization
    if tleft() > 120:
        t_lim = min(100, tleft() * 0.2)
        h_adam, sc = adam_opt(best_h, n, target,
                              time_limit=t_lim, max_steps=50000,
                              n_top_start=3, n_top_end=20,
                              lr=0.01, beta1=0.8, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    # Phase 3: Binary simulated annealing
    if tleft() > 80:
        h_bin, sc = binary_sa(best_h, n, target,
                              time_limit=min(60, tleft() * 0.2), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(h_bin, target)
        # Also try from random binary start
        h_rand_bin = np.zeros(n)
        ones_positions = rng.choice(n, size=int(target), replace=False)
        h_rand_bin[ones_positions] = 1.0
        h_bin2, sc2 = binary_sa(h_rand_bin, n, target,
                                time_limit=min(40, tleft() * 0.15), rng=rng)
        if sc2 < best_score:
            best_score = sc2
            best_h = project_feasible(h_bin2, target)

    # Phase 4: SLSQP after binary SA
    if tleft() > 100:
        h_pol, sc = slsqp_opt(best_h, n, target,
                              [1000, 5000, 20000, 100000, 500000, 2000000],
                              2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Pair coordinate descent
    if tleft() > 120:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(100, tleft() * 0.2),
                           max_sweeps=5000, n_cand=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 6: Adam round 2
    if tleft() > 80:
        t_lim = min(60, tleft() * 0.2)
        h_adam, sc = adam_opt(best_h, n, target,
                              time_limit=t_lim, max_steps=40000,
                              n_top_start=5, n_top_end=15,
                              lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    # Phase 7: Binary SA round 2 (from current best)
    if tleft() > 60:
        h_bin, sc = binary_sa(best_h, n, target,
                              time_limit=min(40, tleft() * 0.2), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(h_bin, target)
        if tleft() > 50:
            h_pol, sc2 = slsqp_opt(best_h, n, target,
                                    [5000, 20000, 100000, 500000, 2000000, 5000000],
                                    1500, 1e-16)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol

    # Phase 8: Higher resolution
    if tleft() > 250:
        n_hires = 600
        target_hires = n_hires / 2.0
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        h_hr, sc = slsqp_opt(h_interp, n_hires, target_hires,
                              [500, 2000, 10000, 50000, 200000, 1000000],
                              1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_hr
            n = n_hires
            target = target_hires

        if tleft() > 100:
            h_cd, sc2 = pair_cd(best_h, n, target,
                                 time_limit=min(60, tleft() * 0.25),
                                 max_sweeps=3000, n_cand=20)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_cd

        if tleft() > 80:
            h_pol, sc3 = slsqp_opt(best_h, n, target,
                                    [5000, 20000, 100000, 500000, 2000000],
                                    1500, 1e-16)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol

    # Phase 9: Try n=500 if we haven't already
    if tleft() > 200 and n == 400:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hr, sc = slsqp_opt(h_interp, n_try, target_try,
                              [500, 2000, 10000, 50000, 200000],
                              1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_hr
            n = n_try
            target = target_try
        if tleft() > 80:
            h_cd, sc2 = pair_cd(best_h, n, target,
                                 time_limit=min(50, tleft() * 0.25),
                                 max_sweeps=2000, n_cand=20)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_cd

    # Phase 10: Noise restarts
    for noise_scale in [0.02, 0.05, 0.1]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_opt(h_pert, len(best_h), target,
                              [1000, 5000, 20000, 100000, 500000],
                              1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_opt(best_h, n_use, target_use,
                                [10000, 50000, 200000, 1000000, 5000000],
                                1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 12: Final Adam fine-tune
    if tleft() > 30:
        h_adam, sc = adam_opt(best_h, len(best_h), len(best_h) / 2.0,
                              time_limit=min(20, tleft() * 0.4), max_steps=20000,
                              n_top_start=10, n_top_end=10,
                              lr=0.002, beta1=0.95, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
