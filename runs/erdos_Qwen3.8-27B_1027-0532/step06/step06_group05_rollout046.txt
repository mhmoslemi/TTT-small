    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def compute_grad(h, weights):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    # For s = 0 to n-1: corr[n-1+s] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        # d/dh[j]: (1-h[j+s]) for j in [0, n-1-s], -h[j-s] for j in [s, n-1]
        if s == 0:
            grad[:] += w * (1.0 - h[:])
            # No second term since s=0
        else:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
    # For s = 1 to n-1: corr[n-1-s] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
    for s in range(1, n):
        w = weights[n - 1 - s]
        if w < 1e-14:
            continue
        # d/dh[j]: (1-h[j-s]) for j in [s, n-1], -h[j+s] for j in [0, n-1-s]
        grad[s:] += w * (1.0 - h[:n-s])
        grad[:n-s] -= w * h[s:]
    return grad
