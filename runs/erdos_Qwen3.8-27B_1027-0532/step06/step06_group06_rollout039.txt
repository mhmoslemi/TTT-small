        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones].astype(float))[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros].astype(float))[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[np.argmin(h[idx].astype(float))]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[np.argmax(h[idx].astype(float))]] = 1
            final_sum += 1
    return h_r.astype(float)

    """Compute ΔC(k) for all k when swapping h[i]=1→0 and h[j]=0→1."""
    delta = np.zeros(2*N - 1)
    # For each k (shift), the change is:
    # ΔC(k) = h[i-k]*1_{valid} + h[i+k]*1_{valid} - h[j+k]*1_{valid} - h[j-k]*1_{valid}
    # k ranges from -(N-1) to (N-1), indexed by k+N-1 in the delta array
    
    # h[i-k] is nonzero (equals h[i-k]) when 0 ≤ i-k < N, i.e., i-N+1 ≤ k ≤ i
    for k in range(max(-(N-1), i-N+1), min(N-1, i)+1):
        delta[k+N-1] += h[i-k]
    
    # h[i+k] is nonzero when 0 ≤ i+k < N, i.e., -i ≤ k ≤ N-1-i

def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    cs = int(h_r.sum())
    if cs == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if cs > target_sum:
        need = cs - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones].astype(float))[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - cs
        if need <= len(zeros):
            order = np.argsort(-h[zeros].astype(float))[:need]
            h_r[zeros[order]] = 1
    cs = int(h_r.sum())
    while cs != target_sum:
        if cs > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[np.argmin(h[idx].astype(float))]] = 0
            cs -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[np.argmax(h[idx].astype(float))]] = 1
            cs += 1
    return h_r.astype(float)

def lbfgs_run(h_init, beta, target, penalty, maxiter, bounds):
    def obj(x):
        corr = compute_corr(x)
        cm = float(corr.max())
        sh = beta * (corr - cm)
        sh -= sh.max()
        v = cm + float(np.log(np.sum(np.exp(sh)))) / beta
        d = float(np.sum(x)) - target
        v += penalty * d * d
        return v
    
    def grad(x):

        dx = 2.0 / N
        corr = compute_corr(x)
        cm = float(corr.max())
        sh = beta * (corr - cm)
        sh -= sh.max()

        tw = float(w.sum())

            w = np.zeros(2*N-1)
            w[int(np.argmax(corr))] = 1.0
        else:

        omh = 1.0 - x

            s2 = slice(j, j+N)
            g[j] = float(np.dot(omh, w[s1])) - float(np.dot(x, w[s2]))
        g *= dx
        g += 2.0 * penalty * (float(np.sum(x)) - target)
        return g
    
    res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})
    return res.x

def binary_ls_focused(h_start, target_sum, max_iters, time_limit, start_time, rng):
    n = len(h_start)
    h = h_start.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    no_improve = 0
    max_no_improve = 200
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        # Find top-5 worst shifts
        top_indices = np.argsort(corr)[-5:][::-1]
        top_shifts = top_indices - (n - 1)
        c_max = float(corr[max(top_indices)])
        
        # Generate candidates: for each top shift k, find i where h[i]=1 and h[i+k]=0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        candidates = []
        for k in top_shifts:
            if k > 0:
                # h[i]=1, h[i+k]=0: i in [0, n-k-1]
                valid_i = np.where((h[:n-k] == 1) & (h[k:] == 0))[0]
                if len(valid_i) > 0:
                    sample = valid_i[rng.randint(len(valid_i), size=min(5, len(valid_i)))]
                    for i in sample:
                        # Find j where h[j]=0 and h[j+k]=1 (to "fix" the pair)
                        valid_j = np.where((h[:n-k] == 0) & (h[k:] == 1))[0]
                        if len(valid_j) > 0:
                            j_sample = valid_j[rng.randint(len(valid_j), size=min(5, len(valid_j)))]
                            for j in j_sample:
                                if j != i:
                                    candidates.append((i, j))
            elif k < 0:
                kk = -k
                # h[i]=1, h[i-kk]=0: i in [kk, n-1]
                valid_i = np.where((h[kk:] == 1) & (h[:-kk] == 0))[0] + kk
                if len(valid_i) > 0:
                    sample = valid_i[rng.randint(len(valid_i), size=min(5, len(valid_i)))]
                    for i in sample:
                        valid_j = np.where((h[kk:] == 0) & (h[:-kk] == 1))[0] + kk
                        if len(valid_j) > 0:
                            j_sample = valid_j[rng.randint(len(valid_j), size=min(5, len(valid_j)))]
                            for j in j_sample:
                                if j != i:
                                    candidates.append((i, j))
        
        if len(candidates) == 0:
            # Fallback: random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                candidates = [(i, j)]
            else:
                break
        
        # Evaluate candidates
        best_improve = 0.0
        best_cand = None
        for (i, j) in candidates[:50]:
            if h[i] != 1 or h[j] != 0:
                continue
            h[i] = 0
            h[j] = 1
            new_corr = compute_corr(h)
            new_max = float(new_corr.max()) * dx
            if new_max < best_c5 - 1e-10:
                improve = best_c5 - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_cand = (i, j, new_corr, new_max)
            h[i] = 1
            h[j] = 0
        
        if best_cand is not None:
            i, j, new_corr, new_max = best_cand
            h[i] = 0
            h[j] = 1
            corr = new_corr
            best_c5 = new_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5
