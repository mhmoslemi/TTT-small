def get_grad_shift_v(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        # First term: j in [0, n-k-1], g[j] += (1-h[j+k])
        g[:n-k] += (1.0 - h[k:])
        # Second term: j in [k, n-1], g[j] -= h[j-k]
        g[k:] -= h[:n-k]
    else:

        g[kk:] += (1.0 - h[:n-kk])
        # Second term: j in [0, n-kk-1], g[j] -= h[j+kk]
        # j+kk ranges from kk to n-1
        g[:n-kk] -= h[kk:]

import numpy as np
from scipy.optimize import minimize
import time

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift_v(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    return g

def _obj_slsqp(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def _cons_fun(h):
    return np.sum(h) - _G_target

def _cons_jac(h):
    return np.ones(_G_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        _G_alpha = alpha
        try:
            res = minimize(_obj_slsqp, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,

    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    rng_local = np.random.RandomState(12345)
    for step in range(max_steps):

            break
        if step > 300 and no_improve > 1500:
            break

        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.5), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift_v(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)

        t = step + 1

        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
