    """Compute gradient of (1/alpha) * log(sum(exp(alpha * corr))) subject to box constraints."""
    n = len(h)
    shifts = np.arange(2*n-1) - (n-1)
    max_corr = corr.max()
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -400, 0)
    exp_w = np.exp(shifted)
    p = exp_w / exp_w.sum()
    
    grad = np.zeros(n)

        pk = p[k_idx]
        if pk < 1e-15:
            continue

        if s >= 0:

import numpy as np
from scipy.optimize import minimize
import time

def _fft_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    return np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft((1.0 - h)[::-1], N), N)[:2 * n - 1]

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_fft_corr(h))) * dx

def _compute_delta(h, i, j, n):
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask_ik = (idx >= 0) & (idx < n)
    result[mask_ik] += h[idx[mask_ik]] - 1.0
    idx = j - k
    mask_jk = (idx >= 0) & (idx < n)
    result[mask_jk] += 1.0 - h[idx[mask_jk]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    result[i - j + n - 1] += 1.0
    result[j - i + n - 1] += 1.0
    result[n - 1] -= 2.0
    return result

def _project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def _project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-10:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        cand_i = ones[np.argsort(A[ones])[:top_candidates]]
        cand_j = zeros[np.argsort(-B[zeros])[:top_candidates]]

        found = False
        for i in cand_i:
            for j in cand_j:
                delta = _compute_delta(h, i, j, n)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    stall = 0
                    c5_val = float(corr.max()) * dx
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            # Random attempts
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = _compute_delta(h, i, j, n)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    stall = 0
                    c5_val = float(corr.max()) * dx
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5
