def project_feasible(h, target_sum):

    h = np.clip(h.astype(float), 0.0, 1.0)

def compute_rh(h, n):
    """Compute R_h(d) for d=0,...,n-1 using FFT"""
    N = 1
    while N < 2 * n:
        N *= 2
    H = fft(h, N)
    ac = np.real(ifft(H * np.conj(H)))

def project_feasible(h, target_sum):

    n = len(h)
    for _ in range(50):

        if abs(diff) < 1e-12:
            break

            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_rh_fft(h, n):
    N = 1
    while N < 2 * n:
        N *= 2
    H = fft(h, N)
    ac = np.real(ifft(H * np.conj(H)))
    return ac[:n]

def smooth_min_rh_and_grad(h, n, alpha):
    rh = compute_rh_fft(h, n)
    x = -alpha * rh
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    q = exp_x / np.sum(exp_x)
    f_val = -(x_max + np.log(np.sum(exp_x))) / alpha
    kernel = np.zeros(2 * n - 1)
    kernel[n - 1] = 2.0 * q[0]

        N *= 2

    K = fft(kernel, N)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                                    max_sweeps=15000, n_cand_start=15, n_cand_end=50,
                                    max_no_improve=300)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
        if tleft() > 60:
            h_pol1 = slsqp_round(best_h, n1, target1, [5000, 20000], 600, 1e-14)
            sc1 = compute_c5(h_pol1)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    if tleft() > 100:
        h_ms, c5_ms = multi_shift_cd(best_h, len(best_h), len(best_h) / 2.0,
                                     time_limit=min(80, tleft() * 0.5),
                                     max_sweeps=10000, n_cand=30, n_shifts=3)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 80:
        h_pg3, _ = fft_projected_gradient(best_h, len(best_h), len(best_h) / 2.0,
                                          time_limit=min(60, tleft() * 0.5),
                                          alpha_start=5000.0, alpha_end=300000.0,
                                          max_iters=80000, step_init=0.01)
        sc = compute_c5(h_pg3)
        if sc < best_score:
            best_score = sc
            best_h = h_pg3

    if tleft() > 60:
        h_sts, c5_sts = stochastic_search(best_h, len(best_h), len(best_h) / 2.0,
                                          time_limit=min(40, tleft() * 0.4), base_scale=0.03)
        if c5_sts < best_score:
            best_score = c5_sts
            best_h = h_sts

    if tleft() > 50:
        h_cd_fin, c5_fin = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                   time_limit=min(35, tleft() * 0.6),
                                   max_sweeps=8000, n_cand_start=30, n_cand_end=80,
                                   max_no_improve=200)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    if tleft() > 30:
        h_final = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                              [10000, 50000], 500, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
