    h = np.asarray(h, dtype=float)

def project_binary(h, target_sum):

    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:

            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1

def project_box_sum(h, target_sum):

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

        while True:

                stall = 0

def _smooth_slsqp_obj(h, alpha):

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

            best_score = s_a
            best_h = h_a.copy()

        # Binary project the continuous result and do local search
        h_bin2 = project_binary(best_h, target_int)
        if tleft() > 60:
            t_b = min(60, tleft() - 60)
            h_b, c5_b = greedy_binary_v2(h_bin2, target_int, 10**7,
                                         start_time + t_b, start_time, rng,
                                         M_worst=5, max_cand=60)
            if c5_b < best_score:
                best_score = c5_b
                best_h = h_b.copy()

    # Phase 5: Multi-restart Adam from perturbed points
    if tleft() > 150:
        h_pert = best_h + rng.randn(n) * 0.008
        h_pert = project_box_sum(h_pert, target)
        rt = min(60, tleft() * 0.25)
        h_ar, s_ar = adam_continuous(
            h_pert, target, start_time, start_time + rt,
            alphas=[200, 1000, 5000, 20000],
            lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_ar < best_score:
            best_score = s_ar
            best_h = h_ar.copy()

    # Phase 6: Final SA polish
    if tleft() > 80:
        h_fin = project_binary(best_h, target_int)
        sa_time = min(0.15 * (budget_s - 30), tleft() * 0.5)
        h_sa2, c5_sa2 = sa_binary_v2(h_fin, target_int, start_time + sa_time,
                                     start_time, rng, t_init=0.002, t_final=0.00002)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 7: SLSQP polish
    if tleft() > 50:
        slsqp_time = min(60, tleft() * 0.5)
        alphas_sl = [5000, 20000, 80000, 200000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     2000, 1e-16, start_time + slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 8: Final greedy polish
    if tleft() > 30:
        h_fin = project_binary(best_h, target_int)
        t_final = min(30, tleft() * 0.5)
        h_gr, c5_gr = greedy_binary_v2(h_fin, target_int, 10**7,
                                       start_time + t_final, start_time, rng,
                                       M_worst=3, max_cand=80)
        if c5_gr < best_score:
            best_score = c5_gr
            best_h = h_gr.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
