from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

# Smoothed objective for SLSQP
_SLQ_TARGET = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def project_feasible(h, target_sum):

        n = n2
        target = target2

    if tleft() > 50:
        cd_time = min(80, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd_v2(best_h, n, target, time_limit=cd_time, n_cand=50, max_sweeps=25000, top_k=4, alpha_w=45.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 30:
        hp, sc = subgrad_refine(best_h, n, target, time_limit=min(30, tleft() * 0.4), max_steps=10000, top_k=5, base_lr=0.01)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 25:
        hp, sc = slsqp_round(best_h, n, target, [50000, 200000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 20:
        hcd, c5cd = fast_pair_cd_v2(best_h, n, target, time_limit=min(18, tleft() * 0.5), n_cand=40, max_sweeps=15000, top_k=3, alpha_w=40.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
