```python
def topk_smooth(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sort_idx = np.argsort(corr)[::-1]
    top_k_idx = sort_idx[:k]
    value = np.mean(corr[top_k_idx])
    # Gradient: for each i in top_k_idx, add (1/k) * grad_corr[i]
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in top_k_idx:
        w = 1.0 / k
        k_shift = int(s) - (n - 1)
        if k_shift >= 0:
            grad[:n - k_shift] += w * one_minus_h[k_shift:]
            grad[k_shift:] -= w * h[:n - k_shift]
        else:
            k_shift = -k_shift
            grad[k_shift:] += w * one_minus_h[:n - k_shift]
            grad[:n - k_shift] -= w * h[k_shift:]
    return value, grad
```
