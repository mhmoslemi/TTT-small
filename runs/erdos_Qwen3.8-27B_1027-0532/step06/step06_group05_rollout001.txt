import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = float(np.sum(h_proj))
        if abs(s - target_sum) < 1e-13:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    return c_max + np.log(float(np.sum(exp_vals))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    one_minus_h = 1.0 - h
    grad_h = np.zeros(n)
    for s in range(n):
        w = weights[n - 1 + s]
        if w > 1e-14:
            grad_h[:n - s] += w * one_minus_h[s:]
            if s > 0:
                grad_h[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w > 1e-14:
            grad_h[p:] += w * one_minus_h[:n - p]
            grad_h[:n - p] -= w * h[p:]

        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h,
                          jac=lambda x: smooth_grad(x, alpha),
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B with gradually increasing alpha
    alphas_1 = [50, 100, 200, 500, 1000, 2000, 5000]
    if tleft() > 200:
        h1, s1 = lbfgs_phase(best_h, target, alphas_1, 3000,
                             min(400, tleft()), start_time)
        if s1 < best_score:
            best_score = s1
            best_h = h1.copy()

    # Phase 2: Nesterov with high alpha for precise convergence
    if tleft() > 200:
        alphas_2 = [5000, 10000, 20000, 50000, 100000, 200000]
        h2, s2 = nesterov_phase(best_h, target, alphas_2,
                                lr=0.003, momentum=0.85,
                                time_limit=min(400, tleft() - 50),
                                start_time=start_time)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    # Phase 3: Binary local search
    if tleft() > 80:
        h_bin = (best_h >= 0.5).astype(float)
        cs = int(h_bin.sum())
        if cs != int(target):
            if cs > int(target):
                ones = np.where(h_bin == 1)[0]
                order = np.argsort(best_h[ones])
                for idx in order[:cs - int(target)]:
                    h_bin[ones[idx]] = 0.0
            else:
                zeros = np.where(h_bin == 0)[0]
                order = np.argsort(-best_h[zeros])
                for idx in order[:int(target) - cs]:
                    h_bin[zeros[idx]] = 1.0
        h3, s3 = binary_local_search(h_bin, min(250, tleft() - 20), start_time)
        if s3 < best_score:
            best_score = s3
            best_h = h3.copy()

    # Phase 4: SLSQP polish with very high alpha
    if tleft() > 60:
        alphas_4 = [20000, 50000, 100000, 200000]
        h4, s4 = slsqp_refine(best_h, target, alphas_4, 2000,
                              min(200, tleft() - 20), start_time)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()

    # Phase 5: Multi-restart with small perturbations
    if tleft() > 80:
        for r in range(3):
            if tleft() < 40:
                break
            scale = 0.002 + 0.002 * r
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_box_sum(h_pert, target)
            alphas_r = [10000, 50000, 100000]
            hr, sr = lbfgs_phase(h_pert, target, alphas_r, 1000,
                                 min(80, tleft() - 20), start_time)
            if sr < best_score:
                best_score = sr
                best_h = hr.copy()

    # Phase 6: Final L-BFGS-B polish
    if tleft() > 30:
        alphas_6 = [100000, 200000, 500000]
        h6, s6 = lbfgs_phase(best_h, target, alphas_6, 5000,
                             max(10.0, tleft() - 5), start_time)
        if s6 < best_score:
            best_score = s6
            best_h = h6.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
