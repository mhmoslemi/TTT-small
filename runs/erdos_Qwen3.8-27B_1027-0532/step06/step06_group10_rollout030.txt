def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum using water-filling."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_round(h0, n, target, time_limit, n_active, max_rounds):

    h = h0.copy()

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

            k = int(s_idx) - mid

        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
        
        if h_var.value is None:
            break
        h_lp = np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-10:

        else:

def perturb_restart(h, n, target, n_perturbs, perturb_size):

    h_out = np.interp(x_new, x_old, h_src)

Another important improvement: **The bisection line search in CD can be replaced with a more efficient method**. Since we're looking for the point where the maximum of `corr + delta * g` stops decreasing, we can use the fact that this is a piecewise linear function of delta and find the exact minimum.

For a piecewise linear function f(δ) = max_s (c_s + δ * g_s), the minimum over δ ∈ [0, D] occurs at:
- δ = 0 (if the function is increasing at 0)
- δ = D (if the function is decreasing at D)  
- At a "kink" where two lines cross

The kinks occur at δ = (c_j - c_i) / (g_i - g_j) for pairs (i,j) where g_i ≠ g_j.

We can find the minimum by:
1. Identifying which s gives the max at δ = 0 (call it s*)
2. Finding the next kink after δ = 0 for the line s*
3. Checking if the function decreases past that kink
4. Repeating until we find the minimum

This is the "active set" method for piecewise linear optimization. It's much faster than bisection.

Actually, for our problem, the function conv(δ) = max_s (corr[s] + δ * g[s]) is convex in δ (max of linear functions). So we want to minimize it over [0, D]. The minimum is where 0 is in the subdifferential, i.e., where there exist s1, s2 with g[s1] < 0 < g[s2] (or g[s*] = 0 for the unique maximizer s*).

More precisely:
- If max_{s: corr[s]+δ*G[s] = f(δ)} G[s] < 0, decrease is possible, set a = δ
- If min_{s: corr[s]+δ*G[s] = f(δ)} G[s] > 0, increase makes it worse, set b = δ
- Otherwise, we're at the minimum

This is exactly what the bisection does, but we can be smarter:
- Start at δ = 0
- Find the active set A = {s: corr[s] = max corr}
- If min_{s∈A} g[s] >= 0, the function is increasing → minimum is at 0
- Otherwise, increase δ until a new element enters the active set
- At each new element, check the condition again

The new elements enter at δ values where corr[s_new] + δ * g[s_new] = current_max.
So δ_new = (current_max - corr[s_new]) / (g[s_new] - slope_of_current_max)

This gives an exact algorithm that takes O(n_corr * n_kinks) steps, where n_kinks is usually small.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:

                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.min(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime < -1e-14:
                        a_d = mid_d
                    else:
                        b_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_refine_fast(h0, n, target, time_limit, n_active, max_rounds):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_lp = np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            best_mix = c5_val
            found = False
            for mix in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < best_mix:
                    best_mix = c5_mix
                    if c5_mix < c5_val - 1e-10:
                        h = h_mix
                        found = True
                        break
            if not found:
                break
    return best_h, best_c5

def slsqp_optimize(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        slsqp_obj_fn.A = alpha
        slsqp_obj_fn.n_val = n
        slsqp_grad_fn.A = alpha
        slsqp_grad_fn.n_val = n
        slsqp_con_fn.target_val = target
        slsqp_con_jac_fn.n_val = n
        cons = [{'type': 'eq', 'fun': slsqp_con_fn, 'jac': slsqp_con_jac_fn}]
        try:
            res = minimize(slsqp_obj_fn, h, jac=slsqp_grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

class _ObjState:

def slsqp_obj_fn(x):
    alpha = slsqp_obj_fn.A
    n = slsqp_obj_fn.n_val
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = smooth_max_value(corr, alpha)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_and_refine(h, n, target, time_limit, n_perturbs, perturb_size):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)
    per_perturb_time = time_limit / max(n_perturbs, 1)
    for p in range(n_perturbs):
        if time.time() > start_perturb + time_limit:
            break
        h_try = h.copy()
        idx = rng.choice(n, size=max(n // 5, 10), replace=False)
        h_try[idx] += rng.uniform(-perturb_size, perturb_size, size=len(idx))
        h_try = project_feasible(h_try, target)
        h_ref, s_ref = weighted_grad_descent(h_try, n, target, per_perturb_time * 0.5, 500, 3, 0.7)
        if s_ref < best_c5:
            best_c5 = s_ref
            best_h = h_ref
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global slsqp_con_fn, slsqp_con_jac_fn, start_perturb
    slsqp_con_fn = _ConState()
    slsqp_con_jac_fn = _ConState()
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    if tleft() > 200:
        h_p, s_p = weighted_grad_descent(best_h, n, target,
                                         time_limit=min(180.0, tleft() * 0.4),
                                         max_iter=5000, n_active=5, decay_rate=0.6)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 300:
        cd_time = min(350.0, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_fast(best_h, n, target,
                                    time_limit=cd_time, max_sweeps=30000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_lp, c5_lp = lp_refine_fast(best_h, n, target,
                                      time_limit=min(100.0, tleft() * 0.3),
                                      n_active=20, max_rounds=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 200:
        slsqp_con_fn.target_val = target
        slsqp_con_jac_fn.n_val = n
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [100, 500, 2000, 10000],
                                   2000, 1e-14, min(150.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 200:
        cd_time = min(250.0, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_fast(best_h, n, target,
                                    time_limit=cd_time, max_sweeps=25000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        start_perturb = time.time()
        h_pert, s_pert = perturb_and_refine(best_h, n, target,
                                             time_limit=min(80.0, tleft() * 0.35),
                                             n_perturbs=8, perturb_size=0.05)
        if s_pert < best_score:
            best_score = s_pert
            best_h = h_pert

    if tleft() > 250:
        for n_try in [600, 1000, 2000]:
            if tleft() < 180:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            s_interp = compute_c5(h_interp)
            if s_interp < best_score:
                best_score = s_interp
                best_h = h_interp
            slsqp_con_fn.target_val = target_try
            slsqp_con_jac_fn.n_val = n_try
            h_hires, s_hires = slsqp_optimize(h_interp, n_try, target_try,
                                              [50, 200, 1000, 5000],
                                              1500, 1e-14,
                                              min(120.0, tleft() * 0.35), start_time)
            if s_hires < best_score:
                best_score = s_hires
                best_h = h_hires
            if tleft() < 80:
                break
            h_cd2, c5_cd2 = pair_cd_fast(best_h, len(best_h), float(len(best_h)) / 2.0,
                                          time_limit=min(80.0, tleft() * 0.3),
                                          max_sweeps=8000, n_cand=25)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_lp, c5_lp = lp_refine_fast(best_h, n_use, target_use,
                                      time_limit=min(60.0, tleft() * 0.4),
                                      n_active=25, max_rounds=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        slsqp_con_fn.target_val = target_use
        slsqp_con_jac_fn.n_val = n_use
        h_final, s_final = slsqp_optimize(best_h, n_use, target_use,
                                           [2000, 10000, 50000],
                                           1500, 1e-15,
                                           min(25.0, tleft() * 0.6), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = pair_cd_fast(best_h, n_use, target_use,
                                    time_limit=min(12.0, tleft() * 0.7),
                                    max_sweeps=5000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
