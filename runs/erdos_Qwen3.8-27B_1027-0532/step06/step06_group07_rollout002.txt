def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def structured_perturbation(h, n, target, rng, mode='sine'):
    h_pert = h.copy()
    if mode == 'sine':
        freq = float(rng.uniform(2, 20))
        phase = float(rng.uniform(0, 2 * np.pi))
        x = np.linspace(0, 2, n)
        pert = 0.05 * np.sin(freq * x + phase)
        h_pert += pert
    elif mode == 'multi_sine':
        x = np.linspace(0, 2, n)
        pert = np.zeros(n)
        for _ in range(3):
            freq = float(rng.uniform(1, 30))
            phase = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.02, 0.08))
            pert += amp * np.sin(freq * x + phase)
        h_pert += pert
    else:
        scale = float(rng.uniform(0.05, 0.2))
        h_pert += rng.randn(n) * scale
    return project_feasible(h_pert, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            pass
        alpha = alpha * 2.5
    return best_h, best_c5


def penalty_bisection(h_init, n, target, time_limit, max_T_iters=10):
    start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1

    T_lo = best_c5 - 0.01
    T_hi = best_c5 + 0.005

    for T_iter in range(max_T_iters):
        if time.time() - start_time > time_limit:
            break
        T_mid = 0.5 * (T_lo + T_hi)
        T_corr = T_mid / dx

        h_cur = best_h.copy()
        lr = 0.02
        for step in range(500):
            if time.time() - start_time > time_limit:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c5_now = float(np.max(corr)) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_cur.copy()

            violations = np.maximum(corr - T_corr, 0.0)
            if violations.max() < 1e-10:
                break

            grad = np.zeros(n)
            active_idx = np.where(corr > T_corr + 1e-8)[0]
            for s_idx in active_idx:
                k = int(s_idx) - mid
                w = (corr[int(s_idx)] - T_corr)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]

            grad *= 2.0 * dx
            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-13:
                break

            improved = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-8:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_cur.copy()

        if c5_test <= T_mid + 1e-6:
            T_lo = T_mid
        else:
            T_hi = T_mid

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP with geometric alpha increase at original resolution
    if tleft() > 120:
        h_pol, sc_pol = multi_alpha_slsqp(best_h, n0, target0, time_limit=min(80, tleft() * 0.4))
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 2: Aggressive pair CD at original resolution
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_advanced(best_h, n0, target0, time_limit=cd_time,
                                       max_sweeps=8000, n_cand=55, n_bisect=14, structured=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to higher resolution and optimize
    best_n = n0
    for n_try in [800, 1000, 1200]:
        if tleft() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_interp = compute_c5(h_interp)

        if tleft() > 120:
            h_pol2, sc_pol2 = multi_alpha_slsqp(h_interp, n_try, target_try,
                                                 time_limit=min(80, tleft() * 0.3))
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2
                best_n = n_try

        if tleft() > 100 and best_n == n_try:
            cd_time2 = min(60, tleft() * 0.25)
            h_cd2, c5_cd2 = pair_cd_advanced(best_h, n_try, target_try,
                                              time_limit=cd_time2,
                                              max_sweeps=4000, n_cand=45, n_bisect=14, structured=True)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                best_n = n_try

    # Phase 4: Structured perturbations + local optimization
    for mode in ['sine', 'multi_sine', 'gaussian']:
        if tleft() < 60:
            break
        h_pert = structured_perturbation(best_h, best_n, best_n / 2.0, rng, mode=mode)
        if tleft() > 40:
            sc_pert = compute_c5(h_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert
            h_cd3, c5_cd3 = pair_cd_advanced(h_pert, best_n, best_n / 2.0,
                                              time_limit=min(30, tleft() * 0.3),
                                              max_sweeps=2000, n_cand=40, n_bisect=13, structured=True)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 5: Penalty bisection
    if tleft() > 80:
        pen_time = min(80, tleft() * 0.3)
        h_pen, c5_pen = penalty_bisection(best_h, best_n, best_n / 2.0,
                                          time_limit=pen_time, max_T_iters=8)
        if c5_pen < best_score:
            best_score = c5_pen
            best_h = h_pen

    # Phase 6: Final SLSQP polish
    if tleft() > 40:
        h_final, sc_final = multi_alpha_slsqp(best_h, best_n, best_n / 2.0,
                                               time_limit=min(30, tleft() * 0.5))
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 7: Final pair CD
    if tleft() > 30:
        h_cd_f, c5_cd_f = pair_cd_advanced(best_h, best_n, best_n / 2.0,
                                            time_limit=min(30, tleft() * 0.6),
                                            max_sweeps=3000, n_cand=60, n_bisect=15, structured=True)
        if c5_cd_f < best_score:
            best_score = c5_cd_f
            best_h = h_cd_f

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
