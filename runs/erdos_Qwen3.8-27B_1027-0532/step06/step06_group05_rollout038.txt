def project_box_sum(h, target_sum):

    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0

        h_proj = np.clip(h - mid, 0.0, 1.0)

# Smooth objective for continuous optimization
def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

Wait, let me re-examine the gradient computation:

The smoothed objective is: f(h) = c_max + (1/alpha) * log(Σ exp(alpha * (corr[k] - c_max)))

The gradient of this with respect to h[i] is:
∂f/∂h[i] = Σ_k weights[k] * ∂corr[k]/∂h[i]

where corr[k] = Σ_j h[j] * (1-h[j+k]) (with appropriate index handling).

∂corr[k]/∂h[i]:
- If i is in the "first" position: ∂corr[k]/∂h[i] = (1 - h[i+k]) if i+k < n
- If i is in the "second" position: ∂corr[k]/∂h[i] = -h[i-k] if i-k >= 0

So the gradient is:
∂f/∂h[i] = Σ_{k: i+k<n} w[k] * (1-h[i+k]) - Σ_{k: i-k>=0} w[k] * h[i-k]

This can be computed as a convolution! Specifically:
∂f/∂h[i] = (w * (1-h))[i] - (flip(w) * h)[i]

where * denotes convolution (with appropriate truncation).

Using FFT, this can be done in O(n log n) instead of O(n²).

    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()

        if time.time() - start_time > time_limit:
            break
        if stall > 300:
            break
        max_val = float(corr.max())
        # Get top-k shifts
        top_shifts = np.argsort(corr)[-top_k:]
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for max_idx in top_shifts:
            k_star = int(max_idx) - (n - 1)
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
            j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
            if len(i_cand) == 0 or len(j_cand) == 0:
                continue
            # Try swaps
            n_i = min(len(i_cand), scan)
            n_j = min(len(j_cand), scan)
            for ii in range(n_i):
                i = i_cand[ii]
                for jj in range(n_j):
                    j = j_cand[jj]
                    # Check if this swap reduces the max
                    nm = max_val
                    for t_idx in top_shifts:
                        k = int(t_idx) - (n - 1)
                        d = 0.0
                        if 0 <= i + k < n:
                            d -= (1.0 - h[i + k])
                        if 0 <= i - k < n:
                            d += h[i - k]
                        if 0 <= j + k < n:
                            d += (1.0 - h[j + k])
                        if 0 <= j - k < n:
                            d -= h[j - k]
                        v = corr[int(t_idx)] + d
                        if v > nm:
                            nm = v
                            break
                    if nm < max_val - best_improve:
                        best_improve = max_val - nm
                        best_i = i
                        best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')

            if stall > 12:
                break
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
    return best_h, best_actual

def subgradient_momentum(h0, target_sum, start_time, time_budget, lr, beta, top_k, rng, eval_every=10):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_actual = compute_c5(h)
    t = 0
    stall = 0
    while True:
        if time.time() - start_time > time_budget:
            break
        t += 1
        c_max, grad = subsample_grad(h, rng, top_k)
        v = beta * v + (1.0 - beta) * grad
        h_new = h - lr * v
        h = project_box_sum(h_new, target_sum)
        if t % eval_every == 0:
            actual = compute_c5(h)
            if actual < best_actual - 1e-13:
                best_actual = actual
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if stall > 30:
            break
    return best_h, best_actual

def binary_greedy_fast(h, target_sum, max_iters, time_limit, start_time, top_k=5, scan=25):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 250:
            break
        max_val = float(corr.max())
        top_shifts = np.argsort(corr)[-top_k:]
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for max_idx in top_shifts:
            k_star = int(max_idx) - (n - 1)
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
            j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
            if len(i_cand) == 0 or len(j_cand) == 0:
                continue
            n_i = min(len(i_cand), scan)
            n_j = min(len(j_cand), scan)
            for ii in range(n_i):
                i = i_cand[ii]
                for jj in range(n_j):
                    j = j_cand[jj]
                    nm = max_val
                    for t_idx in top_shifts:
                        k = int(t_idx) - (n - 1)
                        d = 0.0
                        if 0 <= i + k < n:
                            d -= (1.0 - h[i + k])
                        if 0 <= i - k < n:
                            d += h[i - k]
                        if 0 <= j + k < n:
                            d += (1.0 - h[j + k])
                        if 0 <= j - k < n:
                            d -= h[j - k]
                        v = corr[int(t_idx)] + d
                        if v > nm:
                            nm = v
                            break
                    if nm < max_val - best_improve:
                        best_improve = max_val - nm
                        best_i = i
                        best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        ratio = it / max(max_iters - 1, 1)
        t_val = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.4:
            worst = np.argsort(corr)[-12:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.75:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t_val > 1e-12 and rng.random() < np.exp(-delta / max(t_val, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def make_periodic(n, block_on, block_off, phase):
    h = np.zeros(n)
    period = block_on + block_off
    positions = np.arange(n)
    pos_in_period = (positions + phase) % period
    h[pos_in_period < block_on] = 1.0
    return project_binary(h, n // 2)

def make_two_interval(n, L1, a, c):
    L2 = n // 2 - L1
    if a < 0 or a + L1 > n or c < 0 or c + L2 > n or a + L1 > c:
        return None
    h = np.zeros(n)
    h[a:a + L1] = 1.0
    h[c:c + L2] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 12.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) // 2.0
    half = n // 2

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Structured initializations
    if tleft() > 100:
        candidates = []
        for block in [20, 30, 40, 50, 60, 75, 100, 120, 150, 200]:
            for phase in range(0, min(block, 8)):
                h_pat = make_periodic(n, block, block, phase)
                c5_pat = compute_c5(h_pat)
                candidates.append((c5_pat, h_pat))
        for L1_frac in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
            L1 = int(half * L1_frac)
            L2 = half - L1
            gap = n - half
            for frac in [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]:
                a = int(frac * max(0, gap - L2))
                if a + L1 >= n:
                    a = n - L1 - L2 - 1
                if a < 0:
                    a = 0
                c = a + L1 + int((1.0 - frac) * max(0, gap - L1 - L2))
                if c + L2 > n:
                    c = n - L2
                if a + L1 > c:
                    c = a + L1
                h_ti = make_two_interval(n, L1, a, c)
                if h_ti is not None:
                    c5_ti = compute_c5(h_ti)
                    candidates.append((c5_ti, h_ti))
        candidates.sort(key=lambda x: x[0])
        if candidates and candidates[0][0] < best_score:
            best_h = candidates[0][1].copy()
            best_score = candidates[0][0]
        elif candidates:
            best_h = candidates[0][1].copy()

    # Phase 2: Subgradient momentum (fast exploration)
    if tleft() > 500:
        sg_time = min(200, tleft() - 500)
        h_sg, s_sg = subgradient_momentum(best_h, target, start_time, sg_time,
                                           lr=0.01, beta=0.95, top_k=5, rng=rng, eval_every=8)
        if s_sg < best_score:
            best_score = s_sg
            best_h = h_sg.copy()

    # Phase 3: Smoothed Adam (main optimization)
    if tleft() > 300:
        adam_time = min(350, tleft() - 300)
        h_a, s_a = adam_smooth(best_h, target, start_time, adam_time,
                                alphas=[30, 60, 150, 400, 1000, 4000, 15000, 60000, 200000],
                                lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8, eval_every=12)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 4: Binary greedy local search
    if tleft() > 150:
        h_bin = project_binary(best_h, int(target))
        greedy_time = min(120, tleft() - 150)
        h_g, c5_g = binary_greedy_fast(h_bin, int(target), max_iters=100000,
                                        time_limit=greedy_time, start_time=start_time, top_k=5, scan=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 5: Multi-restart Adam from perturbed points
    if tleft() > 150:
        n_restarts = 3
        for r in range(n_restarts):
            if tleft() < 60:
                break
            jitter = 0.003 + 0.002 * r
            h_pert = best_h + rng.randn(n) * jitter
            h_pert = project_box_sum(h_pert, target)
            rt = min(80, max(25, (tleft() - 60) / (n_restarts - r)))
            h_ar, s_ar = adam_smooth(h_pert, target, start_time, rt,
                                      alphas=[200, 1000, 5000, 30000, 150000],
                                      lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8, eval_every=10)
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    # Phase 6: Binary SA
    if tleft() > 100:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(130, tleft() - 50)
        max_it = int(min(400000, sa_time * 1200))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                 0.004, 0.00001, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 7: Final continuous polish
    if tleft() > 25:
        polish_time = max(8.0, tleft() - 12)
        h_p, s_p = adam_smooth(best_h, target, start_time, polish_time,
                                alphas=[100000, 500000, 2000000],
                                lr=0.0004, beta1=0.9, beta2=0.999, eps=1e-8, eval_every=5)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Phase 8: Final binary greedy polish
    if tleft() > 15:
        h_bin3 = project_binary(best_h, int(target))
        h_g3, c5_g3 = binary_greedy_fast(h_bin3, int(target), max_iters=20000,
                                          time_limit=min(15, tleft() - 5), start_time=start_time,
                                          top_k=3, scan=15)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
