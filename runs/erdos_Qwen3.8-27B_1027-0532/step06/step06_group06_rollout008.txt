            hi1 = min(n, n - k)
            if lo1 < hi1:
                g[lo1:hi1] += w * (1.0 - h[lo1 + k:hi1 + k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if lo2 < hi2:
                g[lo2:hi2] -= w * h[lo2 - k:hi2 - k]
        g *= dx
    if penalty > 0:
        g += 2.0 * penalty * (float(np.sum(h)) - target)
    return g

def tm_objective(x, m, target, penalty):
    n = len(x) - 1
    h = x[:n]
    t = x[n]
    corr = compute_correlation(h)
    val = m * t + float(np.sum(np.maximum(corr - t, 0.0)))
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def tm_gradient(x, m, target, penalty):
    n = len(x) - 1
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = compute_correlation(h)
    active = np.where(corr > t)[0]
    
    gh = np.zeros(n)
    if len(active) > 0:
        k_vals = active - (n - 1)
        for idx in range(len(k_vals)):
            k = int(k_vals[idx])
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            if lo1 < hi1:
                gh[lo1:hi1] += (1.0 - h[lo1 + k:hi1 + k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if lo2 < hi2:
                gh[lo2:hi2] -= h[lo2 - k:hi2 - k]
        gh *= dx
    
    if penalty > 0:
        gh += 2.0 * penalty * (float(np.sum(h)) - target)
    
    gt = m - float(np.sum(corr > t))
    
    return np.concatenate([gh, [gt]])

def lbfgs_optimize(obj_fn, grad_fn, x0, bounds, maxiter, time_limit, start_time):
    best_x = x0.copy()
    best_f = obj_fn(x0)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_fn, x0, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                                   'gtol': 1e-10, 'iprint': -1})
            f = obj_fn(res.x)
            if f < best_f:
                best_f = f
                best_x = res.x.copy()
                x0 = res.x.copy()
            else:
                break
        except Exception:
            break
    return best_x, best_f

def greedy_binary(h_start, target_sum, time_limit, start_time, scan_width=30):
    n = len(h_start)
    h = h_start.copy()
    corr = compute_correlation(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        top_thresh = max_val * 0.85
        top_indices = np.where(corr >= top_thresh)[0]
        
        i_cand = []
        j_cand = []
        for ti in top_indices[:20]:
            k = ti - (n - 1)
            for i in range(n):
                if h[i] == 1 and 0 <= i + k < n and h[i + k] == 0:
                    i_cand.append(i)
            for j in range(n):
                if h[j] == 0 and 0 <= j + k < n and h[j + k] == 1:
                    j_cand.append(j)
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            # Try k=0 neighborhood
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i_cand = ones[:scan_width].tolist()
            j_cand = zeros[:scan_width].tolist()
        
        i_cand = list(set(i_cand))[:scan_width]
        j_cand = list(set(j_cand))[:scan_width]
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for i in i_cand:
            for j in j_cand:
                nm = max_val
                for ti in top_indices:
                    k = ti - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[ti] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = compute_correlation(h)
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                # Revert
                h[best_i] = 1
                h[best_j] = 0
                corr = compute_correlation(h)
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
