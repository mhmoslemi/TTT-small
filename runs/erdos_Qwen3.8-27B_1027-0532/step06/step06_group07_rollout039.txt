```python
def compute_grad_for_shift(h, s):
    """Compute gradient of corr at shift s (s = index - (n-1)) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    x = np.arange(n)
    # (1-h[x+s]) for valid x+s
    mask1 = (x + s >= 0) & (x + s < n)
    grad[mask1] += (1.0 - h[x[mask1] + s])
    # -h[x-s] for valid x-s
    mask2 = (x - s >= 0) & (x - s < n)
    grad[mask2] -= h[x[mask2] - s]
    return grad
```

```python
import cvxpy as cp

def solve_minimax_lp(h, active_grads, n, target_sum):
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for g in active_grads:
        constraints.append(g @ h_var <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP, verbose=False)  # or CBC, GLPK
    if h_var.value is not None:

    return None
```

```python
    bounds = [(0, 1)] * n + [(None, None)]

                  bounds=bounds, method='highs')

    return None
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_all_grads(h, n):
    dx = 2.0 / n
    x = np.arange(n)
    grads = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        j = k + n - 1
        xm = x - k
        mask1 = (xm >= 0) & (xm < n)
        grads[j, mask1] += (1.0 - h[xm[mask1]])
        xp = x + k
        mask2 = (xp >= 0) & (xp < n)
        grads[j, mask2] -= h[xp[mask2]]
    grads *= dx
    return grads


def solve_minimax_lp(grads, corr, h0, n, target):
    rhs = np.zeros(2 * n - 1)
    for j in range(2 * n - 1):
        rhs[j] = -corr[j] + grads[j] @ h0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def solve_minimax_lp_active(grads, corr, h0, n, target, n_active, tol):
    max_corr = np.max(corr)
    active_idx = np.where(corr > max_corr - tol)[0]
    if len(active_idx) > n_active:
        active_idx = np.argsort(corr[active_idx])[::-1][:n_active] + active_idx
    active_idx = np.unique(active_idx)
    rhs = -corr[active_idx].copy() + grads[active_idx] @ h0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = grads[active_idx]
    A_ub[:, -1] = -1.0
    b_ub = rhs
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_minimax_phase(h_start, n, target, time_limit, n_active=80, tol=0.01):
    start = time.time()
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    x = np.arange(n)
    mid = n - 1
    no_improve = 0
    step_frac = 1.0

    for iteration in range(200000):
        if time.time() - start > time_limit:
            break
        if no_improve > 50:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        grads = compute_all_grads(h, n)

        h_lp = solve_minimax_lp_active(grads, corr, h, n, target, n_active, tol)
        if h_lp is None:
            h_lp = solve_minimax_lp(grads, corr, h, n, target)
        if h_lp is None:
            break

        h_lp = project_feasible(h_lp, target)
        improved = False
        for alpha in [step_frac, step_frac * 0.5, step_frac * 0.25, 0.1]:
            h_try = project_feasible(h + alpha * (h_lp - h), target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-14:
                h = h_try
                step_frac = max(alpha, 0.1)
                improved = True
                break
        if not improved:
            step_frac = max(step_frac * 0.8, 0.05)
            no_improve += 1

    return best_h, best_c5


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        sorted_idx = np.argsort(h)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        low_set = np.where(h[low_set] < 0.999)[0]
        high_set = np.where(h[high_set] > 0.001)[0]
        if len(low_set) == 0 or len(high_set) == 0:
            break
        low_set = sorted_idx[low_set]
        high_set = sorted_idx[high_set]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 30:
                break

    return best_h, best_c5


def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, n, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def diverse_restart_lp(h_best, n, target, time_limit, seed_val):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)

    for noise_scale in [0.02, 0.05, 0.1, 0.15, 0.2]:
        if time.time() - start > time_limit:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        lp_time = min(60, (time_limit - (time.time() - start)) * 0.3)
        if lp_time < 10:
            break
        h_lp, c5_lp = lp_minimax_phase(h_pert, n, target, time_limit=lp_time,
                                       n_active=60, tol=0.005)
        if c5_lp < best_c5 - 1e-12:
            best_c5 = c5_lp
            best_h = h_lp

    return best_h, best_c5

            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [2000, 10000, 50000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        lp_time = min(200, tleft() * 0.25)
        h_lp, c5_lp = lp_minimax_phase(best_h, n, target, time_limit=lp_time,
                                       n_active=50, tol=0.002)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 100:
        dr_time = min(150, tleft() * 0.2)
        h_dr, c5_dr = diverse_restart_lp(best_h, n, target, time_limit=dr_time,
                                         seed_val=seed)
        if c5_dr < best_score:
            best_score = c5_dr
            best_h = h_dr

    if tleft() > 80:
        cd_time = min(100, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=4000, n_cand=60, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [10000, 50000, 200000], 1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 50:
        lp_time = min(40, tleft() * 0.5)
        h_lp, c5_lp = lp_minimax_phase(best_h, n, target, time_limit=lp_time,
                                       n_active=40, tol=0.001)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 30:
        cd_time = min(25, tleft() * 0.5)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=2000, n_cand=70, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
