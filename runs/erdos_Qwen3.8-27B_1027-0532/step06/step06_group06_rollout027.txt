def sa_binary_v2(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 3000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = min(it / max(max_iters - 1, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        h_new = h.copy()
        
        if r < 0.5:
            # Targeted move: pick from worst lags
            worst_indices = np.argsort(corr)[-10:]
            k_idx = worst_indices[rng.randint(len(worst_indices))]
            k = int(k_idx) - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k < n]
            j_pool = [j for j in zeros if 0 <= j + k < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                h_new[i_pool[rng.randint(len(i_pool))]] = 0
                h_new[j_pool[rng.randint(len(j_pool))]] = 1
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
        elif r < 0.80:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
        elif r < 0.90:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
        else:
            # Block flip: flip a small contiguous block
            block_size = rng.randint(2, 6)
            start_pos = rng.randint(0, max(1, n - block_size))
            # Find a block that has mixed values
            for attempt in range(5):
                block = h_new[start_pos:start_pos+block_size]
                if 0 < block.sum() < block_size:
                    h_new[start_pos:start_pos+block_size] = 1.0 - block
                    # Fix sum
                    diff = int(h_new.sum()) - target_sum
                    if diff != 0:
                        pass  # Complex to fix, skip for now
                    break
                start_pos = rng.randint(0, max(1, n - block_size))
        
        if np.array_equal(h_new, h):
            continue
        
        # Ensure sum constraint
        if int(h_new.sum()) != target_sum:
            diff = target_sum - int(h_new.sum())
            if diff > 0:
                z_idx = np.where(h_new < 0.5)[0]
                if len(z_idx) >= diff:
                    h_new[z_idx[:diff]] = 1
                else:
                    continue
            else:
                o_idx = np.where(h_new > 0.5)[0]
                if len(o_idx) >= (-diff):
                    h_new[o_idx[:(-diff)]] = 0
                else:
                    continue
            h_new = np.clip(h_new, 0, 1)
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
