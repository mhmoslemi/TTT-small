import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - current_sum + (current_sum - target_sum)
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def lse_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def lse_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    omh = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(omh, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def topk_objective(h, k, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sv = np.sort(corr)[::-1]
    val = float(np.mean(sv[:k]))
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def topk_gradient(h, k, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:k]
    weights = np.zeros(2 * N - 1)
    weights[top_idx] = 1.0 / k
    t_arr = np.arange(N)
    omh = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(omh, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def quartic_objective(h, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mean_c = np.mean(corr)
    val = float(np.mean((corr - mean_c) ** 4))
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def quartic_gradient(h, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    mean_c = np.mean(corr)
    wc = 4.0 * (corr - mean_c) ** 3 / (2 * N - 1)
    t_arr = np.arange(N)
    omh = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(omh, wc[idx1]) - np.dot(h, wc[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def run_continuous(h_start, obj_fn, grad_fn, obj_args, grad_args, target, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_fn, h_cur, args=obj_args,
                          jac=grad_fn, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1})
            h_cand = project_box_sum(res.x, target)
            if compute_c5(h_cand) < compute_c5(h_cur):
                h_cur = h_cand
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def make_sinusoidal(n, freq, target):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x)
    return project_box_sum(h, target)

def make_perturbed(h_base, sigma, target, rng):
    pert = rng.normal(0, sigma, len(h_base))
    return project_box_sum(h_base + pert, target)

            # Quick continuous optimization
            ta_cont = min(25, remaining() * 0.15)
            h_c = run_continuous(h_start, lse_objective, lse_gradient,
                                 (100, target, 500.0), (100, target, 500.0),
                                 target, 100, ta_cont, start_time)
            sc_c = compute_c5(h_c)

            # Binary + local search
            if remaining() < 30:
                if sc_c < best_c5:
                    best_c5 = sc_c
                    best_h = h_c.copy()
                continue
            h_b = project_binary(h_c, target_int)
            ta_ls = min(30, remaining() * 0.15)
            h_l, c5_l = enhanced_local_search(h_b, target_int, ta_ls, start_time, rng, max_stall=1000)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
            elif sc_c < best_c5:
                best_c5 = sc_c
                best_h = h_c.copy()

    # Phase 7: SA from best with tighter schedule
    if remaining() > 80:
        h_bin = project_binary(best_h, target_int)
        ta = min(100, remaining() * 0.30)
        max_it = int(min(200000, ta * 3500))
        h_sa2, c5_sa2 = sa_diverse(h_bin, target_int, max_it, 0.003, 0.000005,
                                    rng, ta, start_time)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 8: Final local search
    if remaining() > 40:
        h_bin = project_binary(best_h, target_int)
        ta = min(60, remaining() - 15)
        h_ls3, c5_ls3 = enhanced_local_search(h_bin, target_int, ta, start_time, rng, max_stall=4000)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3.copy()

    # Phase 9: Quick continuous polish from best binary
    if remaining() > 30:
        h_cont = project_box_sum(best_h.copy(), target)
        ta = min(25, remaining() - 10)
        h_polished = run_continuous(h_cont, lse_objective, lse_gradient,
                                     (500, target, 1000.0), (500, target, 1000.0),
                                     target, 100, ta, start_time)
        sc = compute_c5(h_polished)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_polished.copy()

    # Phase 10: Try different n
    if remaining() > 50:
        for n_try in [n - 4, n + 4, n - 8, n + 8]:
            if remaining() < 30:
                break
            if n_try < 100 or n_try % 2 != 0:
                continue
            t_try = float(n_try) / 2.0
            t_try_int = int(n_try) / 2
            h_interp = interpolate_to(best_h, n_try, t_try)
            ta = min(20, remaining() * 0.2)
            h_opt = run_continuous(h_interp, lse_objective, lse_gradient,
                                    (100, t_try, 500.0), (100, t_try, 500.0),
                                    t_try, 100, ta, start_time)
            sc = compute_c5(h_opt)
            h_b = project_binary(h_opt, t_try_int)
            ta2 = min(15, remaining() * 0.1)
            h_l, c5_l = enhanced_local_search(h_b, t_try_int, ta2, start_time, rng, max_stall=500)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
            elif sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Final feasibility and evaluation
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
