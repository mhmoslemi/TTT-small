def run_continuous_logit(h_init, n, target, alpha, lr, n_iterations, time_limit, start_time, rng):
    # Initialize z from h_init
    eps = 1e-6
    h_init = np.clip(h_init, eps, 1-eps)
    z = np.log(h_init / (1 - h_init))  # logit transform
    
    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    for t in range(n_iterations):
        if time.time() - start_time > time_limit:
            break
        
        h = 1.0 / (1.0 + np.exp(-z))  # sigmoid
        h = np.clip(h, 1e-10, 1-1e-10)
        
        # Compute corr and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_score = float(corr.max()) * (2.0/n)
        
        # Log-sum-exp gradient
        max_corr = corr.max()
        shifted = alpha * (corr - max_corr)
        shifted = np.clip(shifted, -400, 0)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()  # softmax weights
        
        # Gradient of log-sum-exp w.r.t. h
        n_corr = len(corr)
        shifts = np.arange(n_corr) - (n - 1)
        grad_h = np.zeros(n)
        for k in range(n_corr):
            if p[k] < 1e-12:
                continue
            s = shifts[k]
            if s >= 0:
                grad_h[:n-s] += p[k] * (1.0 - h[s:])
                grad_h[s:] += -p[k] * h[:n-s]
            else:
                s_abs = -s
                grad_h[s_abs:] += p[k] * (1.0 - h[:n-s_abs])
                grad_h[:n-s_abs] += -p[k] * h[s_abs:]
        
        # Gradient w.r.t. z (chain rule): dh/dz = h*(1-h)
        grad_z = grad_h * h * (1.0 - h)
        
        # Add sum constraint penalty gradient
        sum_h = h.sum()
        sum_grad_z = (sum_h - target) * h * (1.0 - h) * lambda_sum
        
        grad_z += sum_grad_z  # or use Lagrange multiplier
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Check improvement
        h_new = 1.0 / (1.0 + np.exp(-z))
        h_new = np.clip(h_new, 0, 1)
        # Project sum
        h_new = project_box_sum(h_new, target)
        
        score_new = compute_c5(h_new)
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
    
    return best_h, best_score

    lo = h.min() - 1.0  # if λ < this, all h_i - λ > 1, so sum = n

    return h_proj

def lse_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr.max())
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -400.0, 0.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = float(p[k])
        if pk < 1e-14:
            continue
        s = int(shifts[k])
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += pk * (1.0 - h[s:])
                grad[s:] += -pk * h[:n-s]
        else:
            sa = -s
            if n - sa > 0:
                grad[sa:] += pk * (1.0 - h[:n-sa])
                grad[:n-sa] += -pk * h[sa:]
    return grad, corr, max_c
