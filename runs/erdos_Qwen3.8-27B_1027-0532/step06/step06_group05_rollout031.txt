        grad[:n - s] -= w * h[s:]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[:n - p] += w * one_minus_h[p:]
        grad[p:] -= w * h[:n - p]
    return value, grad


def _slsqp_smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _slsqp_smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[s:] += w * one_minus_h[:n - s]
        grad[:n - s] -= w * h[s:]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[:n - p] += w * one_minus_h[p:]
        grad[p:] -= w * h[:n - p]
    return grad


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def adam_optimize(h0, target, start_time, time_budget, alphas, lr, beta1, beta2, eps, lr_decay=0.95, decay_every=200):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        cur_lr = lr
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 400:
                break
            if stall > 0 and stall % decay_every == 0:
                cur_lr *= lr_decay
                if cur_lr < 1e-7:
                    break
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
    return best_h, best_actual


def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width=30):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 300
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        top_threshold = max_val * 0.9
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.45:
            worst = np.argsort(corr)[-10:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.80:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_slsqp_smooth_obj, h_cur, args=(alpha,),
                           jac=lambda x, a: _slsqp_smooth_grad(x, a),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_bin = project_binary(best_h, int(target))
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=30000,
                                  time_limit=min(60, tleft() * 0.08),
                                  start_time=start_time, scan_width=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    if tleft() > 200:
        adam_time = min(500, tleft() - 200)
        alphas_main = [50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600]
        h_a, s_a = adam_optimize(
            best_h, target, start_time, adam_time,
            alphas=alphas_main, lr=0.003, beta1=0.9, beta2=0.999,
            eps=1e-8, lr_decay=0.9, decay_every=150
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    if tleft() > 300:
        n_restarts = 6
        for ri in range(n_restarts):
            if tleft() < 80:
                break
            jitter = 0.003 + 0.004 * ri
            h_pert = best_h + rng.randn(n) * jitter
            h_pert = project_box_sum(h_pert, target)
            rt = min(120, max(30, (tleft() - 80) / (n_restarts - ri) - 10))
            h_ar, s_ar = adam_optimize(
                h_pert, target, start_time, rt,
                alphas=[200, 500, 1000, 3000, 8000, 20000, 50000],
                lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8,
                lr_decay=0.92, decay_every=100
            )
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    if tleft() > 150:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(180, tleft() - 60)
        max_it = int(min(300000, sa_time * 800))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                0.01, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 80:
        slsqp_time = min(200, tleft() - 40)
        alphas_sl = [3000, 8000, 20000, 50000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     2000, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    if tleft() > 40:
        polish_time = max(10.0, tleft() - 10)
        h_p, s_p = adam_optimize(
            best_h, target, start_time, polish_time,
            alphas=[50000, 100000, 200000],
            lr=0.0005, beta1=0.9, beta2=0.999, eps=1e-9,
            lr_decay=0.95, decay_every=100
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    if tleft() > 30:
        h_bin3 = project_binary(best_h, int(target))
        h_g2, c5_g2 = greedy_binary(h_bin3, int(target), max_iters=50000,
                                    time_limit=min(30, tleft() - 10),
                                    start_time=start_time, scan_width=40)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n
