def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0).copy()

        return y
    if diff < 0:

            if remaining <= 1e-12:
                break
            can_reduce = y[idx]

                remaining = 0
            else:

Now for the main binary local search, I want to do best-improvement. The key is to efficiently find the best swap.

For a given h and active shift k*, the delta at k* for swap (i,j) is:
delta[k*] = h[i+k*] + h[i-k*] - h[j+k*] - h[j-k*]

We want delta[k*] to be as negative as possible (to reduce the max).

But we also need to make sure other shifts don't increase above the current max. A simple heuristic: minimize the new max correlation.

For efficiency, I'll:
1. Compute A[i] = h[i+k*] + h[i-k*] for all i (candidates to turn off)
2. Compute B[j] = h[j+k*] + h[j-k*] for all j (candidates to turn on)
3. Sort: best i's have largest A[i] (among ones), best j's have smallest B[j] (among zeros)
4. Try the top P candidates for i and top Q for j
5. For each (i,j) pair, compute the full delta and check if it improves the max

If P=Q=50, that's 2500 delta computations per iteration, each O(n). Total: 2500*600 = 1.5M per iteration. That's fast.

But we can do even better: for a fixed i, the best j is the one that minimizes B[j] + (effect on other shifts). If we ignore other shifts, the best j is simply argmin B[j] among zeros.

Let me use a tiered approach:
- Tier 1: Try top 10 i's with the best j (10 evaluations)
- Tier 2: Try top 30 i's with top 10 j's (300 evaluations)
- Tier 3: Try top 50 i's with top 50 j's (2500 evaluations)
- If still stuck, try random swaps or 2-swaps

This should be efficient and effective.

Let me also think about the 2-swap. For a 2-swap (i1, i2 → 0, j1, j2 → 1):
The delta is approximately delta(i1,j1) + delta(i2,j2), but there are cross-terms.

The exact delta can be computed as:
delta_2 = delta(i1,j1)_original + delta(i2,j2)_after_first_swap

But computing delta(i2,j2) after the first swap requires updating h first, which is expensive.

Alternative: compute the exact delta for the 2-swap directly.
When h[i1]→0, h[i2]→0, h[j1]→1, h[j2]→1:
At shift s, the change is:
-(1-h_orig[i1+s]) + h_orig[i1-s] -(1-h_orig[i2+s]) + h_orig[i2-s]
+ (1-h_orig[j1+s]) - h_orig[j1-s] + (1-h_orig[j2+s]) - h_orig[j2-s]
PLUS cross-terms where the new values interact:
+ h_new[i1+s'] for terms where i1-s' = i2 or j1 or j2 (but h changed there)
... this gets complicated.

For simplicity, I'll use the approximation delta_2 ≈ delta(i1,j1) + delta(i2,j2) and verify after applying.

Actually, the cross-terms only matter if i1, i2, j1, j2 are within distance n of each other (which they always are for correlation purposes). But the correction is small (at most 4 in absolute value for any single shift), so the approximation is good for screening purposes.

    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]

    n = len(y)

                remaining = 0.0

                                if new_max < best_2max - 1e-12:

                                    best_2pair = (i1, j1, i2, j2, d1, d2)

                if best_2pair is not None:
                    i1, j1, i2, j2, d1, d2 = best_2pair
                    h[i1] = 0.0
                    h[j1] = 1.0
                    h[i2] = 0.0
                    h[j2] = 1.0
                    corr += d1 + d2
                    stall = 0
                    total_improvements += 1
                    curr_max = float(corr.max())
                    if curr_max < best_corr_max - 1e-12:
                        best_corr_max = curr_max
                        best_h = h.copy()
                        best_c5 = best_corr_max * dx
                    done_2swap = True
                    break

            if not done_2swap:
                stall += 1
                if stall > 200:
                    # Perturbation: random 5-10 swaps
                    n_pert = rng.randint(3, 8)
                    ones_cur = np.where(h == 1.0)[0]
                    zeros_cur = np.where(h == 0.0)[0]
                    for _ in range(n_pert):
                        if len(ones_cur) == 0 or len(zeros_cur) == 0:
                            break
                        i = ones_cur[rng.randint(len(ones_cur))]
                        j = zeros_cur[rng.randint(len(zeros_cur))]
                        delta = compute_delta_binary(h, i, j)
                        h[i] = 0.0
                        h[j] = 1.0
                        corr += delta
                        ones_cur = np.where(h == 1.0)[0]
                        zeros_cur = np.where(h == 0.0)[0]
                    # Walk back: only keep if not too bad
                    if float(corr.max()) > best_corr_max + 3:
                        h = best_h.copy()
                        corr = np.correlate(h, 1.0 - h, mode='full')
                    stall = 0

        if time.time() - start_time > max_time:
            break

    return best_h, best_c5

def smooth_gradient_descent(h, max_time, start_time, alpha_start, alpha_end, lr_start, lr_end, mom=0.7):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    v = np.zeros(n)
    it = 0

    while time.time() - start_time < max_time:
        ratio = min(1.0, it / 5000.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        lr = lr_start * (lr_end / lr_start) ** ratio

        max_corr = float(corr.max())
        shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()

        grad = np.zeros(n)
        active_shifts = np.where(W > 1e-10)[0]
        for k_idx in active_shifts:
            wk = W[k_idx]
            if wk < 1e-12:
                continue
            s = shifts[k_idx]
            if s >= 0:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += wk * (-h[abs_s:])

        v = mom * v - lr * grad
        h_new = h + v
        h_new = project_box_sum(h_new, target)

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        corr = corr_new
        it += 1

    return best_h, best_c5

def slsqp_refine(h_start, max_time, start_time):
    n = len(h_start)
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    alpha_schedule = [50, 200, 500, 2000, 10000]
    for alpha in alpha_schedule:
        if time.time() - start_time > max_time:
            break
        try:
            res = minimize(lambda x: obj_func(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-15:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_h_bin = project_binary(h_init, int(target_init))
    best_score_bin = compute_c5(best_h_bin)
    best_h_cont = project_box_sum(h_init, target_init)
    best_score_cont = compute_c5(best_h_cont)
    overall_best_h = best_h_cont.copy()
    overall_best_score = min(best_score_bin, best_score_cont)

    # Phase 1: Binary best-improvement local search
    t1 = min(250.0, tleft() * 0.25)
    if t1 > 30:
        h_b1, sc_b1 = binary_best_improve(best_h_bin, t1, rng, start_time)
        if sc_b1 < best_score_bin:
            best_score_bin = sc_b1
            best_h_bin = h_b1.copy()
        if sc_b1 < overall_best_score:
            overall_best_score = sc_b1
            overall_best_h = h_b1.copy()

    # Phase 2: Second binary search from perturbed start
    t2 = min(150.0, tleft() * 0.15)
    if t2 > 30:
        h_pert = best_h_bin.copy()
        n_pert = rng.randint(8, 15)
        ones_p = np.where(h_pert == 1.0)[0]
        zeros_p = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones_p) == 0 or len(zeros_p) == 0:
                break
            i = ones_p[rng.randint(len(ones_p))]
            j = zeros_p[rng.randint(len(zeros_p))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones_p = np.where(h_pert == 1.0)[0]
            zeros_p = np.where(h_pert == 0.0)[0]
        h_b2, sc_b2 = binary_best_improve(h_pert, t2, rng, start_time)
        if sc_b2 < best_score_bin:
            best_score_bin = sc_b2
            best_h_bin = h_b2.copy()
        if sc_b2 < overall_best_score:
            overall_best_score = sc_b2
            overall_best_h = h_b2.copy()

    # Phase 3: Continuous smooth gradient descent
    t3 = min(200.0, tleft() * 0.20)
    if t3 > 30:
        h_cont_start = best_h_bin.copy() if best_score_bin <= best_score_cont else best_h_cont.copy()
        h_c3, sc_c3 = smooth_gradient_descent(h_cont_start, t3, start_time,
                                               alpha_start=30.0, alpha_end=5000.0,
                                               lr_start=0.04, lr_end=0.001)
        if sc_c3 < best_score_cont:
            best_score_cont = sc_c3
            best_h_cont = h_c3.copy()
        if sc_c3 < overall_best_score:
            overall_best_score = sc_c3
            overall_best_h = h_c3.copy()

    # Phase 4: SLSQP polish
    t4 = min(120.0, tleft() * 0.12)
    if t4 > 30:
        h_s4, sc_s4 = slsqp_refine(best_h_cont, t4, start_time)
        if sc_s4 < best_score_cont:
            best_score_cont = sc_s4
            best_h_cont = h_s4.copy()
        if sc_s4 < overall_best_score:
            overall_best_score = sc_s4
            overall_best_h = h_s4.copy()

    # Phase 5: Multi-resolution attempts
    for n_try in [n_init * 2, n_init + 100, n_init - 100]:
        if tleft() < 120:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h_cont, n_try, target_try)

        t_gs = min(40.0, tleft() * 0.10)
        h_g, sc_g = smooth_gradient_descent(h_interp, t_gs, start_time,
                                            alpha_start=20.0, alpha_end=2000.0,
                                            lr_start=0.03, lr_end=0.0005)
        h_sl, sc_sl = slsqp_refine(h_g, min(30.0, tleft() * 0.08), start_time)

        if sc_sl < overall_best_score:
            overall_best_score = sc_sl
            overall_best_h = h_sl.copy()
            best_h_cont = h_sl.copy()
            best_score_cont = sc_sl

    # Phase 6: Iterated local search on binary with more restarts
    if tleft() > 80:
        for restart in range(4):
            if tleft() < 40:
                break
            h_rs = best_h_bin.copy()
            n_drift = rng.randint(12, 25)
            ones_r = np.where(h_rs == 1.0)[0]
            zeros_r = np.where(h_rs == 0.0)[0]
            for _ in range(n_drift):
                if len(ones_r) == 0 or len(zeros_r) == 0:
                    break
                i = ones_r[rng.randint(len(ones_r))]
                j = zeros_r[rng.randint(len(zeros_r))]
                h_rs[i] = 0.0
                h_rs[j] = 1.0
                ones_r = np.where(h_rs == 1.0)[0]
                zeros_r = np.where(h_rs == 0.0)[0]
            t_rs = min(30.0, tleft() * 0.10)
            h_rs2, sc_rs2 = binary_best_improve(h_rs, t_rs, rng, start_time)
            if sc_rs2 < best_score_bin:
                best_score_bin = sc_rs2
                best_h_bin = h_rs2.copy()
            if sc_rs2 < overall_best_score:
                overall_best_score = sc_rs2
                overall_best_h = h_rs2.copy()

    # Phase 7: Final continuous refinement from best binary
    if tleft() > 50:
        h_final_start = best_h_bin.copy()
        t_f = min(60.0, tleft() * 0.20)
        h_f, sc_f = smooth_gradient_descent(h_final_start, t_f, start_time,
                                            alpha_start=50.0, alpha_end=10000.0,
                                            lr_start=0.02, lr_end=0.0003)
        if sc_f < best_score_cont:
            best_score_cont = sc_f
            best_h_cont = h_f.copy()
        if sc_f < overall_best_score:
            overall_best_score = sc_f
            overall_best_h = h_f.copy()

    # Phase 8: Final SLSQP
    if tleft() > 30:
        h_fin, sc_fin = slsqp_refine(best_h_cont, min(25.0, tleft() - 10), start_time)
        if sc_fin < overall_best_score:
            overall_best_score = sc_fin
            overall_best_h = h_fin.copy()

    # Final projection and evaluation
    n_final = len(overall_best_h)
    target_final = float(n_final) / 2.0
    h_out = project_box_sum(overall_best_h, target_final)
    score_out = compute_c5(h_out)

    if score_out > overall_best_score:
        h_out = overall_best_h.copy()
        score_out = overall_best_score

    return h_out, score_out, n_final
