        local_best = float('inf')

        while time.time() - start_time < time_budget:
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > stall_limit:
                break

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()

    return best_h, best_actual


def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 150:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                delta = 0.0
                if 0 <= i + k_star < n:
                    delta -= (1.0 - h[i + k_star])
                if 0 <= i - k_star < n:
                    delta += h[i - k_star]
                if 0 <= j + k_star < n:
                    delta += (1.0 - h[j + k_star])
                if 0 <= j - k_star < n:
                    delta -= h[j - k_star]
                new_val = max_val + delta
                if new_val < max_val - best_improve:
                    best_improve = max_val - new_val
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Two-interval structured search
    if tleft() > 120:
        ti_time = min(12.0, tleft() * 0.08)
        params_c, c5_c = two_interval_coarse(n, ti_time + (time.time() - start_time), start_time)
        if params_c is not None:
            ti_fine_time = min(5.0, tleft() * 0.04)
            params_f, c5_f = two_interval_fine(n, params_c[0], params_c[1], params_c[2],
                                                ti_fine_time + (time.time() - start_time), start_time)
            if params_f is not None:
                h_ti = two_interval_h(n, params_f[0], params_f[1], params_f[2])
                if h_ti is not None:
                    c5_ti = compute_c5(h_ti)
                    if c5_ti < best_score:
                        best_score = c5_ti
                        best_h = h_ti.copy()
                    # Also use two-interval as extra starting point for Adam

    # Phase 2: Greedy binary from initial projection
    if tleft() > 150:
        h_bin = project_binary(best_h, int(target))
        g_time = min(60.0, tleft() * 0.15)
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=30000,
                                  time_limit=g_time + (time.time() - start_time),
                                  start_time=start_time, scan_width=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 3: Continuous Adam optimization (main)
    if tleft() > 250:
        adam_time = min(500.0, tleft() - 200)
        alphas_main = [25, 50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600, 50000]
        h_a, s_a = adam_continuous(
            best_h, target, start_time, adam_time + (time.time() - start_time),
            alphas=alphas_main, lr=0.0015, beta1=0.9, beta2=0.999,
            eps=1e-8, stall_limit=250
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 4: Multi-restart Adam with perturbed starting points
    if tleft() > 250:
        n_restarts = 3
        for ri in range(n_restarts):
            if tleft() < 100:
                break
            jitter = 0.003 + 0.004 * ri
            h_pert = best_h + rng.randn(len(best_h)) * jitter
            h_pert = project_box_sum(h_pert, target)
            rt = min(120.0, (tleft() - 80) / (n_restarts - ri))
            alphas_r = [50, 200, 800, 3200, 12800, 50000]
            h_ar, s_ar = adam_continuous(
                h_pert, target, start_time, rt + (time.time() - start_time),
                alphas=alphas_r, lr=0.0012, beta1=0.9, beta2=0.999,
                eps=1e-8, stall_limit=200
            )
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    # Phase 5: Greedy binary from best solution's projection
    if tleft() > 100:
        h_bin2 = project_binary(best_h, int(target))
        g2_time = min(80.0, tleft() * 0.25)
        h_g2, c5_g2 = greedy_binary(h_bin2, int(target), max_iters=50000,
                                    time_limit=g2_time + (time.time() - start_time),
                                    start_time=start_time, scan_width=35)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    # Phase 6: SLSQP polish
    if tleft() > 80:
        sl_time = min(150.0, tleft() - 30)
        h_ref, sc_ref = slsqp_refine(
            best_h, len(best_h), target,
            alphas=[5000, 20000, 80000],
            maxiter=3000, ftol=1e-18,
            time_limit=sl_time + (time.time() - start_time),
            start_time=start_time
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 7: Final short Adam polish with very high alpha
    if tleft() > 40:
        polish_time = max(10.0, tleft() - 10)
        h_p, s_p = adam_continuous(
            best_h, target, start_time, polish_time + (time.time() - start_time),
            alphas=[50000, 100000, 200000],
            lr=0.0008, beta1=0.9, beta2=0.999,
            eps=1e-8, stall_limit=150
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
