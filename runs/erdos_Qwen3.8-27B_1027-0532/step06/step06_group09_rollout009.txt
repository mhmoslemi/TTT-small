import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha)

def lbfags_round(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def exact_pair_cd(h, n, target, time_limit, max_sweeps=5000,
                  n_cand_start=20, n_cand_max=60, patience=100,
                  top_k=1):
    """Coordinate descent that considers top-k lags simultaneously."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        # Adaptive candidate count
        if no_improve > patience // 3 and n_cand < n_cand_max:
            n_cand = min(n_cand + 3, n_cand_max)
        elif no_improve < patience // 5 and n_cand > n_cand_start:
            n_cand = max(n_cand - 2, n_cand_start)

        # Get top-k lags
        top_indices = np.argsort(corr)[-top_k:]
        
        # Build combined gradient for top-k lags (weighted by proximity to max)
        c_max_val = np.max(corr)
        weights = np.zeros(n_corr)
        for ti in top_indices:
            weights[ti] = 1.0  # Equal weight for top-k
        # Normalize
        w_sum = weights[top_indices].sum()
        weights[top_indices] /= w_sum

        grad = np.zeros(n)
        for s_idx in top_indices:
            k = int(s_idx) - mid
            w = weights[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            # Precompute the effect of changing h[i] on all correlations
            h_i_sp = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            h_i_sp[v_ip] = h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            h_i_sm = np.zeros(n_corr)
            h_i_sm[v_im] = h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm_arr = np.zeros(n_corr)
                jpm_arr[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm_arr = np.zeros(n_corr)
                jsm_arr[jm] = h[j - shifts[jm]]

                g = h_i_sp + h_i_sm - jpm_arr - jsm_arr

                # Bisection for optimal delta
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d < a_d:
                    b_d = a_d + 1e-10

                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def simulated_annealing(h_init, n, target, time_limit, 
                        t_start=0.5, t_end=0.001, cooling=0.97,
                        max_steps=5000):
    """Simulated annealing with pair moves."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        # Temperature
        T = t_start * (cooling ** step)
        if T < t_end:
            T = t_end
        
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Compute delta range
        delta_lo = max(-h[i], -(1.0 - h[j]))
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        
        # Pick random delta in range
        delta = rng.uniform(delta_lo, delta_hi)
        
        # Compute effect on correlations
        i_sp = np.zeros(n_corr)
        v_ip = (i + shifts >= 0) & (i + shifts < n)
        i_sp[v_ip] = h[i + shifts[v_ip]]
        v_im = (i - shifts >= 0) & (i - shifts < n)
        i_sm = np.zeros(n_corr)
        i_sm[v_im] = h[i - shifts[v_im]]
        
        j_sp = np.zeros(n_corr)
        v_jp = (j + shifts >= 0) & (j + shifts < n)
        j_sp[v_jp] = h[j + shifts[v_jp]]
        v_jm = (j - shifts >= 0) & (j - shifts < n)
        j_sm = np.zeros(n_corr)
        j_sm[v_jm] = h[j - shifts[v_jm]]
        
        g = i_sp + i_sm - j_sp - j_sm
        
        new_corr = corr + delta * g
        new_max = np.max(new_corr)
        old_max = np.max(corr)
        
        delta_obj = (new_max - old_max) * dx
        
        if delta_obj <= 0 or rng.random() < np.exp(-delta_obj / T):
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = new_corr
            # Recompute if projection changed things
            c5_now = np.max(corr) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
    
    return best_h, best_c5

def multi_scale_optimize(h_init, n, target, time_limit):
    """Coarse-to-fine optimization."""
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Downsample to various resolutions, optimize, then upsample back
    for n_coarse in [100, 200, 300]:
        if time.time() - start > time_limit * 0.7:
            break
        if n_coarse >= n:
            continue
        
        target_coarse = n_coarse / 2.0
        x_old = np.linspace(0, 1, n)
        x_coarse = np.linspace(0, 1, n_coarse)
        h_coarse = np.interp(x_coarse, x_old, best_h)
        h_coarse = project_feasible(h_coarse, target_coarse)
        
        # Optimize at coarse resolution
        h_optimized = exact_pair_cd(h_coarse, n_coarse, target_coarse,
                                     time_limit=30, max_sweeps=2000,
                                     n_cand_start=15, n_cand_max=40, patience=50,
                                     top_k=1)[0]
        
        # Upsample back
        x_fine = np.linspace(0, 1, n)
        h_up = np.interp(x_fine, x_coarse, h_optimized)
        h_up = project_feasible(h_up, target)
        
        # Refine at fine resolution
        h_refined = exact_pair_cd(h_up, n, target,
                                   time_limit=20, max_sweeps=2000,
                                   n_cand_start=20, n_cand_max=50, patience=50,
                                   top_k=1)[0]
        
        c5_ref = compute_c5(h_refined)
        if c5_ref < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def lbfags_round(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def exact_pair_cd(h, n, target, time_limit, max_sweeps=5000,
                  n_cand_start=25, n_cand_max=80, patience=150,
                  top_k=1):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        if no_improve > patience // 3 and n_cand < n_cand_max:
            n_cand = min(n_cand + 5, n_cand_max)
        elif no_improve < patience // 5 and n_cand > n_cand_start:
            n_cand = max(n_cand - 3, n_cand_start)

        top_indices = np.argsort(corr)[-top_k:]

        grad = np.zeros(n)
        for s_idx in top_indices:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            i_sp = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            i_sp[v_ip] = h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            i_sm = np.zeros(n_corr)
            i_sm[v_im] = h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(-h[i], -(1.0 - h[j]))
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm = np.zeros(n_corr)
                jpm[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm = np.zeros(n_corr)
                jsm[jm] = h[j - shifts[jm]]

                g = i_sp + i_sm - jpm - jsm

                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d < a_d:
                    b_d = a_d + 1e-10

                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def iterated_local_search(h_init, n, target, time_limit, seed=99,
                          max_rounds=200, cd_time_per_round=5.0):

        if time.time() - start > time_limit:
            break
        if no_improve > 40:
            break

        scale = base_scale * (1.0 + 0.5 * np.sin(rnd * 0.3))
        n_pert = int(rng.choice([3, 5, 8, 12, 20, 30]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)

def multi_start_momentum(h_init, n, target, time_limit, max_steps=20000,
                         mu=0.85, base_lr=0.04, seed=55):

    dx = 2.0 / n

    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    rng = np.random.RandomState(seed)

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-3:]
        k_star = int(top_k[-1]) - mid

            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        lr = base_lr * max(0.1, 1.0 - step / max_steps)

                    best_c5 = compute_c5(h_new)
                    best_h = h_new.copy()
                no_improve = 0
            if no_improve > 500:
                break
        h = h_new
