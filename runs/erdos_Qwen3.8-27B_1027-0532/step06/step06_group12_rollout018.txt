```python
                k = p - n_half
                if k == 0:
                    slopes[p] = (1.0 - 2.0 * h[j]) * dx
                elif k > 0:
                    if j >= k and j < n:
                        slopes[p] += (1.0 - h[j-k]) * dx
                    if j < n - k:
                        slopes[p] += (-h[j+k]) * dx
                else:
                    m = -k
                    if j < n - m:
                        slopes[p] += (1.0 - h[j+m]) * dx
                    if j >= m:
                        slopes[p] += (-h[j-m]) * dx
            
            # The max overlaps as function of t: max_p (O_p + slopes[p] * (t - h[j]))
            # = max_p ((O_p - slopes[p]*h[j]) + slopes[p]*t)
            # This is the max of linear functions in t, hence convex
            # Minimum is where 0 is in the subgradient
            
            base = compute_overlaps(h) - slopes * old_val
            # Find t in [0,1] that minimizes max_p (base[p] + slopes[p]*t)
            # The minimum of max of linear functions is at a "breakpoint"
            # i.e., where two of the lines intersect
            
            # Simple approach: try t=0, t=1, and the breakpoints
            candidates = [0.0, 1.0]
            # Add breakpoints: where base[a] + slopes[a]*t = base[b] + slopes[b]*t
            # t = (base[b] - base[a]) / (slopes[a] - slopes[b])
            # Only add if in [0,1]
            # For efficiency, only check top few overlaps
            top_idx = np.argsort(base)[::-1][:10]
            for ia in range(len(top_idx)):
                for ib in range(ia+1, len(top_idx)):
                    a, b = top_idx[ia], top_idx[ib]
                    ds = slopes[a] - slopes[b]
                    if abs(ds) > 1e-12:
                        t_candidate = (base[b] - base[a]) / ds
                        if 0.0 <= t_candidate <= 1.0:
                            candidates.append(t_candidate)
            
            # Also maintain sum constraint: if we change h[j] from old_val to t,
            # we need to compensate in another coordinate
            # For simplicity, skip the sum constraint in coordinate descent
            # and project at the end of each sweep
            
            best_t = old_val
            best_obj = compute_c5(h)
            for t in candidates:
                h[j] = t
                obj = compute_c5(h)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_t = t
            h[j] = best_t
            
            if best_t != old_val:
                improved = True
        
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_c5
```

```python
    """Thue-Morse sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        # Count number of 1s in binary representation of i
        h[i] = bin(i).count('1') % 2

def van_der_corput(n):
    """Van der Corput sequence (low discrepancy)."""
    h = np.zeros(n)
    for i in range(n):
        # Radix-2 van der Corput
        v = 0.0
        f = 0.5
        j = i
        while j > 0:
            v += f * (j % 2)
            j //= 2
            f /= 2
        h[i] = 1.0 if v > 0.5 else 0.0
    return h
```

```python
import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h_proj > 1e-12)[0]
        if len(idx) > 0:
            h_proj = h_proj.copy()
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def grad_weighted(h, weights, n, dx):
    n_half = n - 1
    g = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-14:
            continue
        k = p - n_half
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    return g * dx


def smooth_max_grad(h, alpha, n, dx):
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - m), -300.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    return grad_weighted(h, weights, n, dx)


def fista_nesterov(h_init, n, target, alpha, time_limit, L=None, restart_every=30000):
    start = time.time()
    if L is None:
        L = 1.0 + 0.05 * alpha
    step = 1.0 / L
    dx = 2.0 / n
    rng = np.random.RandomState(12345)

    h = h_init.copy()
    z = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for t in range(2000000):
        if time.time() - start > time_limit:
            break

        g = smooth_max_grad(h, alpha, n, dx)
        h_new = project_feasible(z - step * g, target)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        t_k = t_new
        h = h_new

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > restart_every:
            perturb = 0.02 / np.sqrt(no_improve / restart_every)
            h = h + perturb * rng.randn(n)
            h = project_feasible(h, target)
            z = h.copy()
            t_k = 1.0
            no_improve = 0

    return best_h, best_c5


def worst_lag_coordinate_descent(h_init, n, target, time_limit, max_sweeps=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(777)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_k = np.argsort(overlaps)[::-1][:8]
        improved = False

        for lag_p in top_k:
            k = lag_p - n_half
            for trial in range(3):
                if time.time() - start > time_limit:
                    break
                j = rng.randint(0, n)
                old_val = h[j]

                slope = 0.0
                if k == 0:
                    slope = (1.0 - 2.0 * h[j]) * dx
                elif k > 0:
                    if j >= k:
                        slope += (1.0 - h[j - k]) * dx
                    if j < n - k:
                        slope += (-h[j + k]) * dx
                else:
                    m = -k
                    if j < n - m:
                        slope += (1.0 - h[j + m]) * dx
                    if j >= m:
                        slope += (-h[j - m]) * dx

                if abs(slope) < 1e-14:
                    continue

                current_overlap = overlaps[lag_p]
                if slope > 0:
                    new_val = max(0.0, old_val - 0.5)
                else:
                    new_val = min(1.0, old_val + 0.5)

                h[j] = new_val
                new_c5 = compute_c5(h)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] = old_val

        if not improved:
            break

        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def multiplicative_weights_descent(h_init, n, target, time_limit, eta0=0.1, max_iter=500000):
    start = time.time()
    dx = 2.0 / n
    h = np.clip(h_init.copy(), 0.01, 0.99)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta0

    for t in range(max_iter):
        if time.time() - start > time_limit:
            break

        overlaps = compute_overlaps(h)
        m = np.max(overlaps)
        alpha = 50.0 + 50.0 * (t / max_iter)
        shifted = np.clip(alpha * (overlaps - m), -300.0, 0.0)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        g = grad_weighted(h, weights, n, dx)

        h_raw = h * np.exp(-eta * g)
        h_new = project_feasible(h_raw, target)

        t_k = t + 1
        eta = eta0 / np.sqrt(t_k + 1)

        h = h_new
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def subgradient_averaging(h_init, n, target, time_limit, c5_target=None):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if c5_target is None:
        c5_target = best_c5 - 0.005
    else:
        c5_target = min(c5_target, best_c5 - 0.001)

    h_avg = h.copy()
    count = 0

    for t in range(2000000):
        if time.time() - start > time_limit:
            break

        overlaps = compute_overlaps(h)
        max_idx = np.argmax(overlaps)
        weights = np.zeros(2 * n - 1)
        weights[max_idx] = 1.0
        g = grad_weighted(h, weights, n, dx)

        g_norm_sq = np.sum(g * g)
        if g_norm_sq < 1e-18:
            break

        current_val = overlaps[max_idx]
        gap = current_val - c5_target
        if gap < 0:
            gap = 0.01

        step = min(1.0, gap / g_norm_sq)
        h_new = project_feasible(h - step * g, target)

        count += 1
        h_avg = h_avg + (h_new - h_avg) / count
        h = h_new

        if t % 500 == 0:
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5 - 1e-15:
                best_c5 = c5_avg
                best_h = h_avg.copy()

    c5_val = compute_c5(h)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h.copy()
    c5_avg = compute_c5(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg.copy()

    return best_h, best_c5


def structured_perturb(h, n, target, rng, mode=0):
    h_new = h.copy()
    if mode == 0:
        block = rng.randint(5, 20)
        start = rng.randint(0, n - block)
        h_new[start:start + block] = 1.0 - h_new[start:start + block]
    elif mode == 1:
        for _ in range(rng.randint(10, 30)):
            j = rng.randint(0, n)
            h_new[j] = 1.0 - h_new[j]
    else:
        j1 = rng.randint(0, n // 2)
        j2 = rng.randint(n // 2, n)
        h_new[j1], h_new[j2] = h_new[j2], h_new[j1]
    return project_feasible(h_new, target)


def multi_alpha_fista(h_init, n, target, time_limit):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alphas = [200.0, 100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
    total_tl = time_limit

    for alpha in alphas:
        tl = total_tl - (time.time() - start)
        if tl < 10:
            break
        phase_time = min(tl * 0.15, 120.0)
        h_new, c5_new = fista_nesterov(h, n, target, alpha, phase_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new if c5_new <= compute_c5(h) else h

    return best_h, best_c5
```
