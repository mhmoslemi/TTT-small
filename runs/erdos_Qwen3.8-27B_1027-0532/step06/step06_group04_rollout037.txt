                k_idx = active_k[idx]

                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
        if p % 30 == 29:
            lam_rate = max(0.001, lam_rate * 0.95)
    
    return best_h, best_c5

def block_coordinate_descent(h_init, target, n_active, block_size, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    sorted_active = sorted(active_k)
    
    n_blocks = n // block_size
    remaining = n - n_blocks * block_size
    
    for iteration in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Process each block
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            block_start = b * block_size
            block_end = block_start + block_size
            idx = np.arange(block_start, block_end)
            
            # Compute slopes and intercepts for this block
            slopes = np.zeros((len(sorted_active), block_size))
            intercepts = np.zeros(len(sorted_active))
            
            for ai, k_idx in enumerate(sorted_active):
                k = k_idx - (n - 1)
                for bi, i in enumerate(idx):
                    A = 0.0
                    if 0 <= i + k < n:
                        A += (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        A -= h[i - k]
                    slopes[ai, bi] = A * dx
                intercepts[ai] = corr[k_idx] - slopes[ai, :] @ h[idx]
            
            # Solve LP: min t s.t. slopes @ h_block + intercepts <= t, 0 <= h_block <= 1
            # Variables: [t, h_block[0], ..., h_block[block_size-1]]
            # Objective: [1, 0, ..., 0]
            n_var = block_size + 1
            c_obj = np.zeros(n_var)
            c_obj[0] = 1.0
            
            # Inequality constraints: slopes @ h_block - t <= -intercepts
            # => [-slopes, 1] @ [t, h_block] <= -intercepts
            A_ub = np.zeros((len(sorted_active), n_var))
            A_ub[:, 0] = -1.0
            A_ub[:, 1:] = -slopes
            b_ub = -intercepts
            
            # Bound constraints: 0 <= t, 0 <= h_block[i] <= 1
            bounds = [(0.0, None)] + [(0.0, 1.0)] * block_size
            
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                             method='highs')
                if res.status == 0:
                    h[idx] = res.x[1:]
            except Exception:
                pass
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(sorted_active)):
            sorted_active = sorted(new_active)
    
    return best_h, best_c5

def smooth_objective(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400.0, 0.0)
    log_sum = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return log_sum

def smooth_gradient(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(-n+1, n):
        w = weights[k + n - 1]
        if w < 1e-15:
            continue
        idx_v1 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(idx_v1):
            grad[idx_v1] += w * (1.0 - h[idxs[idx_v1] + k]) * dx
        idx_v2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(idx_v2):
            grad[idx_v2] -= w * h[idxs[idx_v2] - k] * dx
    return grad

def lbfgs_smooth(h_init, target, alphas, time_limit):
    start = time.time()
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 5:
            break
        
        def obj(x):
            return smooth_objective(x, alpha, target)
        
        def grad(x):
            return smooth_gradient(x, alpha, target)
        
        def obj_con(x):
            return np.sum(x) - target
        
        def grad_con(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': obj_con, 'jac': grad_con}]
        
        maxiter = max(50, int(remaining * 2))
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def lp_refine(h_init, n, target, max_iter, time_limit, n_active, d0=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    reg = 1e-4
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = sorted(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in active_k:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-11, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
            pass
        if not solved:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in set(active_k) and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 3e-5:
                    active_k.append(k_idx)
                    break
        active_k = sorted(top_k | set(active_k))
        if len(active_k) > 3 * n_active:
            active_k = sorted(np.argsort(corr)[-3 * n_active:].tolist())
        
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.5, damping * 0.95)
        else:
            damping = min(0.99, damping + 0.02)
        if it % 40 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.8)
    
    return best_h, best_c5

def subgradient_momentum(h_init, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    v = np.zeros(n)
    mu = 0.9
    base_lr = 0.05
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_v1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_v2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h[idxs[idx_v1] - k])
        grad[idx_v2] -= h[idxs[idx_v2] + k]
        grad *= dx
        
        v = mu * v - base_lr * grad
        h_try = project_feasible(h + v, target)
        sc = compute_c5(h_try)
        
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h


def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-13 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.005
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)

                k_idx = active_k[idx]

                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        h = np.clip(h, 0.0, 1.0)

                h = np.clip(h, 0.0, 1.0)

def smooth_obj_func(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad_func(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights = weights / sw
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(-n + 1, n):
        w = weights[k + n - 1]
        if w < 1e-15:
            continue
        v1 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - h[idxs[v1] + k]) * dx
        v2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(v2):
            grad[v2] -= w * h[idxs[v2] - k] * dx
    return grad

def lbfgs_round(h_init, target, alphas, time_limit):

    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]

    for alpha in alphas:
        _global_alpha = alpha
        remaining = time_limit - (time.time() - start)
        if remaining < 5:
            break

        def obj(x):
            return smooth_obj_func(x, alpha)

        def grad(x):
            return smooth_grad_func(x, alpha)

        maxiter = max(30, int(remaining * 3))
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

def lp_refine(h_init, n, target, max_iter, time_limit, n_active, d0=0.9):

    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = sorted(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for k_idx in active_k:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:

                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:

            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:

def subgrad_nesterov(h_init, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    t_act = 1.0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        half_n = n - 1
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        v1 = (0 <= idxs - k) & ((idxs - k) < n)
        v2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - y[idxs[v1] - k])
        grad[v2] -= y[idxs[v2] + k]
        grad *= dx
        t_act = (1.0 + np.sqrt(1.0 + 4.0 * t_act * t_act)) / 2.0
        lr = 2.0 / (1.0 + t_act) * 0.3
        beta = (1.0 - t_act) / (1.0 + t_act + 1.0)
        h_new = project_feasible(h - lr * grad, target)
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        sc = compute_c5(h)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h.copy()
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: LP refinement
    if tleft() > 220:
        h_lp, c5_lp = lp_refine(best_h, n, target, max_iter=6000,
                                time_limit=min(200, tleft() * 0.2),
                                n_active=80, d0=0.85)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Block coordinate descent
    if tleft() > 130:
        h_bcd, c5_bcd = block_cd(best_h, target, n_active=80,
                                 block_size=8, time_limit=min(110, tleft() * 0.14), rng=rng)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    # Phase 3: L-BFGS smooth
    if tleft() > 130:
        h_lb, c5_lb = lbfgs_round(best_h, target,
                                  alphas=[200, 600, 2000],
                                  time_limit=min(110, tleft() * 0.14))
        if c5_lb < best_score:
            best_score = c5_lb
            best_h = h_lb

    # Phase 4: Coordinate descent
    if tleft() > 100:
        h_cd, c5_cd = coord_descent(best_h, target, n_active=80,
                                    max_passes=50000, time_limit=min(80, tleft() * 0.13), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: LP with more constraints
    if tleft() > 130:
        h_lp, c5_lp = lp_refine(best_h, n, target, max_iter=5000,
                                time_limit=min(110, tleft() * 0.17),
                                n_active=120, d0=0.9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: Block CD larger blocks
    if tleft() > 100:
        h_bcd, c5_bcd = block_cd(best_h, target, n_active=100,
                                 block_size=12, time_limit=min(80, tleft() * 0.13), rng=rng)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    # Phase 7: Higher resolution
    if tleft() > 250:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        if tleft() > 200:
            h_lp500, c5_lp500 = lp_refine(h_interp, n_try, target_try,
                                          max_iter=4000, time_limit=min(120, tleft() * 0.3),
                                          n_active=80, d0=0.85)
            if c5_lp500 < best_score:
                best_score = c5_lp500
                best_h = h_lp500
                n = n_try
                target = target_try
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp
                n = n_try
                target = target_try
        if tleft() > 100:
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_lb500, c5_lb500 = lbfgs_round(best_h, target_use,
                                            alphas=[1000, 5000],
                                            time_limit=min(80, tleft() * 0.25))
            if c5_lb500 < best_score:
                best_score = c5_lb500
                best_h = h_lb500

    # Phase 8: Nesterov subgradient
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        h_ng, c5_ng = subgrad_nesterov(best_h, n_use, target_use,
                                       max_steps=10000, time_limit=min(40, tleft() * 0.4))
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    # Phase 9: Perturbation restarts
    for noise_scale in [0.03, 0.08, 0.15]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise_scale
        h_pert = project_feasible(h_pert, target_use)
        h_cd_p, c5_cd_p = coord_descent(h_pert, target_use, n_active=80,
                                        max_passes=20000, time_limit=min(30, tleft() * 0.2), rng=rng)
        if c5_cd_p < best_score:
            best_score = c5_cd_p
            best_h = h_cd_p

    # Phase 10: Final LP polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        h_lp, c5_lp = lp_refine(best_h, n_use, target_use, max_iter=4000,
                                time_limit=min(50, tleft() * 0.5),
                                n_active=120, d0=0.95)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final block CD
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 40:
        h_bcd, c5_bcd = block_cd(best_h, target_use, n_active=120,
                                 block_size=6, time_limit=min(30, tleft() * 0.5), rng=rng)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    # Phase 12: Final coord descent
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 25:
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=120,
                                    max_passes=50000, time_limit=min(20, tleft() * 0.6), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
