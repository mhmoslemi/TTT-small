def project_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}."""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to entries with most room
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h += (diff / total) * margins
    else:
        # Need to decrease sum: reduce entries proportionally
        total = h.sum()
        if total > 1e-10:
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
