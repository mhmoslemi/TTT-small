def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_constraints(h, n, active_k, target):
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in sorted(active_k):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def solve_lp(h, n, target, active_k, solver='SCS', eps=1e-10, max_iters=200000):
    h_var, t_var, constraints = build_constraints(h, n, active_k, target)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=eval('cp.' + solver), eps=eps, max_iters=max_iters, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=max_iters)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), t_var.value

def cutting_plane(h_init, n, target, time_limit, max_iter=10000, n_active=200, eps=1e-10):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        h_new, t_val = solve_lp(h, n, target, active_k, eps=eps)
        if h_new is None:
            break
        
        h_new = project_feasible(h_new, target)
        h = h_new
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find violated constraints
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_ref = t_val if t_val is not None else c5_val
        violated = False
        for k_idx in range(len(corr)):
            if k_idx not in active_k:
                if corr[k_idx] * (2.0/n) > t_ref + 1e-7:
                    active_k.add(k_idx)
                    violated = True
        
        if not violated:
            # Try to remove constraints that are no longer tight
            if len(active_k) > 100:
                tight = set()
                for k_idx in active_k:
                    if corr[k_idx] * (2.0/n) > t_ref - 1e-5:
                        tight.add(k_idx)
                if len(tight) >= 50:
                    active_k = tight
        
        # Cap active set
        if len(active_k) > 400:
            active_k = set(np.argsort(corr)[-400:].tolist())
    
    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.array([np.sum(h) - _global_target])

def cons_jac(h):
    return np.array([np.ones(_global_n)])

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def local_perturb(h, n, target, time_limit, delta_list=None):
    start = time.time()
    if delta_list is None:
        delta_list = [1e-4, 1e-3, 1e-2, 5e-2, 0.1]
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    n_half = n - 1
    
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in delta_list:
                for sign in [1, -1]:
                    h_try = h_cur.copy()
                    h_try[i] = np.clip(h_try[i] + sign * d, 0, 1)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_c5 - 1e-14:
                        best_c5 = sc
                        best_h = h_try.copy()
                        h_cur = h_try
                        improved = True
                        break
                if improved:
                    break
    return best_h, best_c5

def gradient_step(h, n, target, time_limit, lr_list=None):
    start = time.time()
    if lr_list is None:
        lr_list = [1e-4, 1e-3, 5e-3, 0.01, 0.05, 0.1]
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    n_half = n - 1
    idxs = np.arange(n)
    
    stall = 0
    while time.time() - start < time_limit:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        idx_v1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        idx_v2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k_star])
        grad[idx_v2] -= h_cur[idxs[idx_v2] + k_star]]
        grad *= dx
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        improved = False
        for lr in lr_list:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5

def coord_descent_cd(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            # Solve min max_j (slopes[j]*h_i + intercepts[j]) s.t. 0 <= h_i <= 1
            # This is a 1D subdivision problem
            # At h_i = 0: max of intercepts
            # At h_i = 1: max of (slopes + intercepts)
            # Find the minimax point
            val_0 = np.max(intercepts)
            val_1 = np.max(slopes + intercepts)
            if val_1 <= val_0:
                h[i] = 1.0
            elif val_0 < val_1:
                # Binary search for the crossing point
                lo, hi = 0.0, 1.0
                for _ in range(40):
                    mid = 0.5 * (lo + hi)
                    vals = slopes * mid + intercepts
                    mv = np.max(vals)
                    # Subgradient at mid
                    active = np.abs(vals - mv) < 1e-12 * max(1.0, abs(mv))
                    sg = np.sum(slopes[active])
                    if sg > 1e-10:
                        hi = mid
                    elif sg < -1e-10:
                        lo = mid
                    else:
                        break
                h[i] = 0.5 * (lo + hi)
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5
