import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def full_softmin_grad(h, n, alpha):
    """Compute gradient of soft-min objective using all shifts."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        # Fallback to argmax
        s_idx = int(np.argmax(corr))
        k = s_idx - (n - 1)
        grad = get_grad_shift(h, k, n)
        return grad * dx
    weights /= wsum
    # Only use significant weights
    mid = n - 1
    sig_idx = np.where(weights > 1e-6)[0]

    for s_idx in sig_idx:
        k = int(s_idx) - mid
        grad += weights[s_idx] * get_grad_shift(h, k, n)

    return grad

    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def precompute_f(h, n):
    f = np.zeros((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    for p in range(n):
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += h[idx1[mask1]]
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    return f

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
