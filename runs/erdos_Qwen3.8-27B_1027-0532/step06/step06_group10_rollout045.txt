def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_lse(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(alpha * (corr - m), -50, 0)))) / alpha

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    return smooth_lse(corr, alpha) * dx

def smooth_grad(h, alpha):

    corr = get_corr(h)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -50.0, 0.0)

    idxs = np.arange(n)

            return smooth_grad(x, alpha)

            return np.ones(n)

def barrier_obj(x, alpha, nu, rho, target):
    val = smooth_obj(x, alpha)
    barrier = -nu * np.sum(np.log(np.clip(x, 1e-15, None) * np.clip(1-x, 1e-15, None)))
    penalty = rho * (np.sum(x) - target)**2
    return val + barrier + penalty

def barrier_grad(x, alpha, nu, rho, target):
    g = smooth_grad(x, alpha)
    xc = np.clip(x, 1e-15, 1-1e-15)
    g += -nu * (1 - 2*xc) / (xc * (1 - xc))
    g += 2 * rho * (np.sum(x) - target)
    return g

# In the optimization:
res = minimize(barrier_obj, h, args=(alpha, nu, rho, target),
               jac=barrier_grad, method='L-BFGS-B',
               bounds=bounds, options={'maxiter': maxiter})
```

This is clean and avoids closures.

Now let me also think about the **simulated annealing CD**. The idea:
1. Compute corr and find the max shift k*
2. Generate random candidate moves (change h[i] and h[j] for random i, j)
3. Evaluate the change in the objective
4. Accept with probability exp(-ΔE/T) where T is the temperature
5. Decrease T over time

The key difference from greedy CD is that we accept worse moves sometimes.

For efficiency, I need to quickly evaluate the change in the max correlation when making a move. This is hard because the max can change.

A simpler approach: instead of evaluating the exact change, use an approximate change based on the gradient at the active shifts.

Let me implement a simpler version:
1. Compute corr
2. Find top m active shifts
3. For each active shift, compute the gradient
4. Combine gradients with weights
5. Take a step in the negative gradient direction
6. Clip and project
7. Repeat with decreasing step size

This is basically a subgradient method with multiple active constraints.
