```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

    return np.sum(h) - _G_target

    return np.ones(_G_n)

def _slsqp_obj(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:

                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5
```

s.t. for each k: sum_j [(1-h₀[j+k])*I(j+k<n) - h₀[j-k]*I(j-k≥0)] * h[j] ≤ t/dx - f_k(h₀)
     sum_j h[j] = n/2
     0 ≤ h[j] ≤ 1

This is a LP with 2n-1 constraints (one for each k) and n+1 variables (h[0],...,h[n-1], t). This is very solvable with cvxpy!

The advantage of this approach:
1. Each LP solve is fast (polynomial time)
2. We get a "linearized" step that's guaranteed to decrease the linearized objective
3. We can use a barrier or trust-region to control the step size

This is essentially the "cutting plane" or "piecewise linear" approach to the max problem.

Let me implement this:

```python
import cvxpy as cp

def lp_subgradient_step(h0, n, target, max_t=None):
    """Solve the LP subproblem to get a new h."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
    
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    mid = n - 1
    
    for s in range(2 * n - 1):
        k = s - mid
        grad = get_grad_shift(h0, k, n)
        # Linearized: corr0[s] + grad · (h - h0) <= t / dx
        # => grad · h <= t/dx - corr0[s] + grad · h0
        constraints.append(grad @ h_var <= t_var / dx - corr0[s] * dx + grad @ h0 * dx)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)  # or ECOS, or CLARABEL
    
    if h_var.value is not None:
        h_new = project_feasible(h_var.value, target)
        return h_new
    return h0
```

Actually, the gradient of corr with respect to h[p] can be computed more efficiently. Let me think...

corr[s] = sum_i h[i] * (1 - h[i - k_s]) where k_s = s - (n-1).

∂corr[s]/∂h[p] = (1 - h[p - k_s]) * I(0 ≤ p-k_s < n) + (-h[p + k_s]) * I(0 ≤ p+k_s < n)

Wait, the second term comes from the case where i - k_s = p, i.e., i = p + k_s. Then the term h[p+k_s] * (1-h[p]) has derivative -h[p+k_s] with respect to h[p].

So: ∂corr[s]/∂h[p] = (1 - h[p - k_s]) * I(0 ≤ p-k_s < n) - h[p + k_s] * I(0 ≤ p+k_s < n)

For a fixed p, as s varies, k_s = s - (n-1) varies from -(n-1) to n-1.
Let k = s - (n-1). Then:
∂corr[s]/∂h[p] = (1 - h[p-k]) * I(0 ≤ p-k < n) - h[p+k] * I(0 ≤ p+k < n)

The first term (1-h[p-k]) is non-zero when 0 ≤ p-k < n, i.e., p-n+1 ≤ k ≤ p.
The second term -h[p+k] is non-zero when 0 ≤ p+k < n, i.e., -p ≤ k < n-p.

So for a fixed p, the gradient vector (over k) is:
g_p[k] = (1-h[p-k]) for k ∈ [max(-(n-1), p-(n-1)), min(n-1, p)]  [from first term]
       - h[p+k] for k ∈ [max(-(n-1), -p), min(n-1, n-1-p)]  [from second term]

Let me just compute this vectorized:

```python
def compute_grad_vec(h, p, n):
    """Compute ∂corr/∂h[p] as a vector of length 2n-1."""
    g = np.zeros(2 * n - 1)
    k = np.arange(-(n-1), n)  # shift values
    
    # First term: (1-h[p-k]) for valid p-k
    idx1 = (p - k >= 0) & (p - k < n)
    g[idx1] += (1.0 - h[p - k[idx1]])
    
    # Second term: -h[p+k] for valid p+k
    idx2 = (p + k >= 0) & (p + k < n)
    g[idx2] -= h[p + k[idx2]]
    
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g


def compute_grad_vec_all(h, p_list, n):
    k = np.arange(-(n - 1), n)
    grads = np.zeros((len(p_list), 2 * n - 1))
    for idx, p in enumerate(p_list):
        idx1 = (p - k >= 0) & (p - k < n)
        grads[idx, idx1] += (1.0 - h[p - k[idx1]])
        idx2 = (p + k >= 0) & (p + k < n)
        grads[idx, idx2] -= h[p + k[idx2]]
    return grads


_G_n = 0
_G_target = 0.0
_G_alpha = 0.0


def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target


def _slsqp_cons_jac(h):
    return np.ones(_G_n)


def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5


def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 300 and no_improve > 1000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def threshold_try(h, n, target):
    n_ones = int(target)
    if n_ones <= 0 or n_ones >= n:
        return None
    best_score = compute_c5(h)
    best = h.copy()
    thresholds = np.sort(h)[::max(1, n // 50)]
    for t in thresholds:
        h_bin = (h >= t).astype(float)
        n1 = int(np.sum(h_bin))
        if n1 != n_ones:
            idxs = np.argsort(-h)
            h_bin = np.zeros(n)
            h_bin[idxs[:n_ones]] = 1.0
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best = h_bin
    return best, best_score


def local_perturbation(h_init, n, target, max_tries, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.15, 0.25]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at original resolution
    if tleft() > 500:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [500, 2000, 10000, 50000, 200000],
                                   1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Multi-resolution refinement
    resolutions = [600, 800, 1000, 1200, 1600]
    for n_try in resolutions:
        if tleft() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        # Adam at this resolution
        adam_budget = min(tleft() * 0.25, 150)
        h_adam, sc = adam_optimize(h_interp, n_try, target_try,
                                   time_limit=adam_budget, max_steps=80000,
                                   n_top_start=5, n_top_end=40,
                                   lr=0.012, beta1=0.85, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

        # Pair CD for fine-tuning
        if tleft() > 150:
            cd_budget = min(tleft() * 0.15, 80)
            nc = 12 if n_try >= 1000 else 18
            h_cd, sc = pair_cd_fast(h_adam, n_try, target_try,
                                    time_limit=cd_budget, max_sweeps=8000, n_cand=nc)
            if sc < best_score:
                best_score = sc
                best_h = h_cd

        # Weighted subgradient for escaping plateaus
        if tleft() > 130:
            sg_budget = min(tleft() * 0.1, 40)
            h_sg, sc = weighted_subgrad(h_adam, n_try, target_try,
                                        time_limit=sg_budget, max_steps=20000,
                                        alpha_w=50.0, base_lr=0.015)
            if sc < best_score:
                best_score = sc
                best_h = h_sg

        # SLSQP polish at this resolution
        if tleft() > 100:
            n_alpha = 4 if n_try <= 800 else 3
            alphas = [5000, 20000, 100000, 500000, 2000000][:n_alpha]
            h_pol, sc = slsqp_optimize(best_h, n_try, target_try,
                                       alphas, 800, 1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

        # Try thresholding
        if tleft() > 80:
            h_bin, sc_bin = threshold_try(best_h, n_try, target_try)
            if h_bin is not None and sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin
                # Re-optimize from binary
                h_adam2, sc2 = adam_optimize(h_bin, n_try, target_try,
                                             time_limit=min(30, tleft() * 0.1),
                                             max_steps=20000, n_top_start=5,
                                             n_top_end=30, lr=0.01,
                                             beta1=0.85, beta2=0.999, eps=1e-8)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_adam2

    # Phase 3: Random restarts to escape local minima
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    for noise_scale in [0.03, 0.08, 0.15]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n_cur) * noise_scale
        h_pert = project_feasible(h_pert, target_cur)
        h_adam, sc = adam_optimize(h_pert, n_cur, target_cur,
                                   time_limit=min(30, tleft() * 0.15),
                                   max_steps=30000, n_top_start=5, n_top_end=25,
                                   lr=0.015, beta1=0.85, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam
        h_pol, sc = slsqp_optimize(h_adam, n_cur, target_cur,
                                   [10000, 50000, 200000], 600, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Final polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        h_cd, sc = pair_cd_fast(best_h, n_use, target_use,
                                time_limit=min(30, tleft() * 0.3),
                                max_sweeps=5000, n_cand=10)
        if sc < best_score:
            best_score = sc
            best_h = h_cd
    if tleft() > 40:
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [20000, 100000, 500000], 500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Local perturbation at the end
    if tleft() > 30:
        h_lp, sc = local_perturbation(best_h, n_use, n_use / 2.0,
                                      max_tries=2000,
                                      time_limit=min(20, tleft() * 0.4), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_lp
            if tleft() > 20:
                h_fp, sc = slsqp_optimize(best_h, n_use, n_use / 2.0,
                                          [50000, 200000], 400, 1e-16)
                if sc < best_score:
                    best_score = sc
                    best_h = h_fp

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n_final
```
