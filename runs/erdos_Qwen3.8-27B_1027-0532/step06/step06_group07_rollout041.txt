import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def subgradient_momentum(h_init, n, target, time_limit, n_active=5, 
                          lr_init=0.01, momentum=0.9, max_iter=5000):
    """Subgradient method with momentum for max of quadratic functions."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.1)
        else:
            no_improve += 1
            if no_improve > 200:
                break
            if no_improve % 50 == 0:
                lr *= 0.95
        
        # Get top n_active shifts
        top_k = np.argsort(corr)[-n_active:]
        
        # Average gradient of active functions
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(top_k)
        grad -= np.mean(grad)  # enforce sum constraint
        
        # Momentum update
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        
        # Check if this improves
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(np.max(corr_new)) * dx
        if c5_new < c5_val:
            h = h_new
        else:
            # Try smaller step
            v = momentum * 0.5 * v - lr * 0.1 * grad
            h_new = project_feasible(h + v, target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            c5_new = float(np.max(corr_new)) * dx
            if c5_new < c5_val:
                h = h_new
            else:
                lr *= 0.9
                v *= 0.5
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n

        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        c_new = compute_c5(h_new)
        if c_new > c5_val:
            v = 0.5 * (momentum * v - 0.2 * lr * grad)
            h_new = project_feasible(h + v, target)
            c_new = compute_c5(h_new)
            if c_new > c5_val:
                lr *= 0.8
                v *= 0.5
            else:
                h = h_new
        else:
            h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [300, 1000], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        lev_time = min(120, tleft() * 0.2)
        h_lev, c5_lev = active_set_optimize(best_h, n, target, time_limit=lev_time, n_active=8, max_iter=4000)
        if c5_lev < best_score:
            best_score = c5_lev
            best_h = h_lev

    if tleft() > 200:
        cd_time = min(250, tleft() * 0.35)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=6000, n_cand=55, n_bisect=15, top_shifts=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [5000, 15000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=5000, n_cand=65, n_bisect=16, top_shifts=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        mo_time = min(80, tleft() * 0.15)
        h_mo, c5_mo = momentum_subgrad(best_h, n, target, time_limit=mo_time,
                                        n_active=6, lr_init=0.008, momentum=0.9, max_iter=3000, patience=150)
        if c5_mo < best_score:
            best_score = c5_mo
            best_h = h_mo

    if tleft() > 60:
        sa_time = min(80, tleft() * 0.15)
        h_sa, c5_sa = sa_pair_cd(best_h, n, target, time_limit=sa_time,
                                  max_sweeps=2000, n_cand=40, n_bisect=12, T_init=0.005, T_final=0.0005)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    if tleft() > 120:
        for noise_scale in [0.04, 0.08, 0.12]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            cd_t = min(60, tleft() * 0.25)
            h_try, c5_try = vectorized_pair_cd(h_pert, n, target,
                                                time_limit=cd_t,
                                                max_sweeps=2000, n_cand=45, n_bisect=13, top_shifts=3)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try

    if tleft() > 100:
        h_lev, c5_lev = active_set_optimize(best_h, n, target, time_limit=min(80, tleft()*0.2), n_active=12, max_iter=3000)
        if c5_lev < best_score:
            best_score = c5_lev
            best_h = h_lev

    if tleft() > 80:
        cd_time = min(80, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=5000, n_cand=50, n_bisect=15, top_shifts=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 40:
        h_pol = slsqp_round(best_h, n, target, [20000, 80000], 600, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 30:
        cd_time = min(30, tleft() * 0.5)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=3000, n_cand=40, n_bisect=14, top_shifts=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
