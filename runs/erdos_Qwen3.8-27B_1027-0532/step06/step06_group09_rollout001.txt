import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_mult(h, alpha, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (c_max + log_sum / alpha) * (2.0 / len(h))

def multi_shift_pair_cd(h, n, target, time_limit, max_sweeps=30000,
                        n_cand_start=40, n_cand_max=120, m_top=8,
                        patience=300, weight_temp=20.0):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        if no_improve > patience // 3 and n_cand < n_cand_max:
            n_cand = min(n_cand + 10, n_cand_max)
        elif no_improve < patience // 5 and n_cand > n_cand_start:
            n_cand = max(n_cand - 5, n_cand_start)

        m = min(m_top, n_corr)
        top_idx = np.argsort(corr)[-m:]
        top_vals = corr[top_idx]
        c_max_val = top_vals[-1]
        weights = np.exp(-weight_temp * (c_max_val - top_vals))
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(m) / m
        else:
            weights /= wsum

        grad = np.zeros(n)
        for w, si in zip(weights, top_idx):
            k = int(si) - mid
            if k == 0:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        pre_j = []
        for j in low_set:
            jp = (j + shifts >= 0) & (j + shifts < n)
            j_pm = np.zeros(n_corr)
            j_pm[jp] = h[j + shifts[jp]]
            jm = (j - shifts >= 0) & (j - shifts < n)
            j_sm = np.zeros(n_corr)
            j_sm[jm] = h[j - shifts[jm]]
            pre_j.append((j, j_pm + j_sm, j_pm, j_sm))

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            ip = (i + shifts >= 0) & (i + shifts < n)
            i_sp = np.zeros(n_corr)
            i_sp[ip] = h[i + shifts[ip]]
            im = (i - shifts >= 0) & (i - shifts < n)
            i_sm = np.zeros(n_corr)
            i_sm[im] = h[i - shifts[im]]
            i_pos = i_sp + i_sm

            for j, j_total, j_pm, j_sm in pre_j:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = i_pos - j_total
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, delta_lo + 1e-10)

                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subspace_optimize(h, n, target, time_limit, n_sub=30, tolerance=0.05):
    start = time.time()
    in_play = np.where((h > tolerance) & (h < 1.0 - tolerance))[0]
    if len(in_play) == 0:
        return h, compute_c5(h)

    # Also include neighbors of in-play coordinates
    extended = set()
    for idx in in_play:
        for d in range(-2, 3):
            ni = idx + d
            if 0 <= ni < n:
                extended.add(ni)
    sub_idx = np.array(sorted(extended))
    n_sub_actual = min(n_sub, len(sub_idx))
    # Pick the most "in play" ones
    in_play_score = np.zeros(n)
    in_play_score[in_play] = 1.0 - np.abs(h[in_play] - 0.5) * 2.0
    sub_idx_full = np.argsort(in_play_score)[-n_sub_actual:]
    sub_idx = np.sort(sub_idx_full)

    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)

    for trial in range(50):
        if time.time() - start > time_limit:
            break
        h_sub = h_cur[sub_idx].copy()
        h_sub_orig = h_sub.copy()

        def sub_obj(x):
            h_full = h_cur.copy()
            h_full[sub_idx] = x
            h_full = project_feasible(h_full, target)
            return compute_c5(h_full)

        try:
            res = minimize(sub_obj, h_sub, method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
            h_try = h_cur.copy()
            h_try[sub_idx] = res.x
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                h_cur = h_try
            else:
                break
        except Exception:
            break

    return h_cur, best_c5

def smooth_anneal(h, n, target, time_limit, alphas=(100, 500, 2000, 8000, 30000, 100000)):
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        time_alloc = max(10, (time_limit - (time.time() - start)) / len(alphas))

        def obj(x):
            return smooth_obj_mult(x, alpha, target)

        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_shifted = np.exp(shifted)
            log_sum = np.log(np.sum(exp_shifted))
            val = (c_max + log_sum / alpha) * (2.0 / n)
            # Numerical gradient
            eps = 1e-7
            grad = np.zeros(n)
            for i in range(n):
                x_p = x.copy()
                x_p[i] += eps
                grad[i] = (smooth_obj_mult(x_p, alpha, target) - val) / eps
            return grad

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': 2000})
            h_try = project_feasible(res.x, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
        except Exception:
            pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def game_theoretic_descent(h_init, n, target, time_limit, max_steps=20000,
                           eta_base=1.5, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.03
    no_improve_count = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 400:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / (lam_top.sum() + 1e-30)

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            if k == 0:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        improved = False
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.2, 0.5)
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.85
            if best_lr < 0.0005:
                break

        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / (lam_new.sum() + 1e-30)
        lam = 0.7 * lam + 0.3 * lam_new

    return best_h, best_c5
