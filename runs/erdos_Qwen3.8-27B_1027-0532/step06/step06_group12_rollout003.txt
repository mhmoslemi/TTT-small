def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h

    else:

        return h

        worst_lag = int(np.argmax(overlaps)) - n_half
        abs_worst = abs(worst_lag)
        
        improved = False
        # Try moving weight from high-overlap-contributing positions
        # At the worst lag, positions where h=1 and h_shifted=0 contribute most
        # Try reducing h at such positions and increasing at others
        if worst_lag >= 0:
            high_contrib = np.where((h_best[:n-worst_lag] > 0.01) & (h_best[worst_lag:] < 0.99))[0]
        else:
            m = -worst_lag
            high_contrib = np.where((h_best[m:] > 0.01) & (h_best[:n-m] < 0.99))[0]
        
        if len(high_contrib) == 0:
            high_contrib = np.where(h_best > 0.01)[0]
        
        for _ in range(30):
            if len(high_contrib) == 0:
                break
            i = high_contrib[np.random.randint(len(high_contrib))]
            # Find a position to increase
            candidates = np.where(h_best < 0.99)[0]
            if len(candidates) == 0:
                break
            j = candidates[np.random.randint(len(candidates))]
            if i == j:
                continue
            delta = 0.01 + 0.05 * np.random.random()
            h_try = h_best.copy()
            h_try[i] = max(0.0, h_try[i] - delta)
            h_try[j] = min(1.0, h_try[j] + delta)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                h_best = h_try
                improved = True
        
        if not improved:
            # Try larger steps
            for _ in range(20):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                delta = 0.1 + 0.3 * np.random.random()
                if h_best[i] < delta:
                    continue
                h_try = h_best.copy()
                h_try[i] -= delta
                h_try[j] = min(1.0, h_try[j] + delta)
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h_best = h_try
                    improved = True
            if not improved:
                break
    return h_best, best_c5

def bisection_lp(h_init, n, target, time_limit, M_max=200, n_bisect=20):
    """Bisection on T with LP feasibility checks."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    overlaps = compute_overlaps(h)
    n_half = n - 1
    top_lags = np.argsort(overlaps)[::-1][:M_max] - n_half
    dx = 2.0 / n
    
    T_low = best_c5 * 0.95
    T_high = best_c5 * 1.01
    
    for bi in range(n_bisect):
        if time.time() - start > time_limit:
            break
        T_mid = 0.5 * (T_low + T_high)
        # Build LP
        h_var = cp.Variable(n)
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for k in top_lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n-k] = (1.0 - h[k:]) * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n-m]) * dx
            constraints.append(coeff @ h_var <= T_mid)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
            except Exception:
                T_high = T_mid
                continue
        
        status = prob.status
        if status == cp.OPTIMAL and h_var.value is not None:
            h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new <= T_mid * 1.05:  # Allow some tolerance
                T_low = T_mid
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                T_high = T_mid
        elif status == cp.INFEASIBLE:
            T_high = T_mid
        else:
            T_high = T_mid
    
    return best_h, best_c5

def alternating_lp_v2(h_init, n, target, time_limit, M=150, n_iter=200):
    """Improved alternating LP with progressive constraint tightening."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_half = n - 1
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:M]
        lags = top_idx - n_half
        current_max = float(np.max(overlaps))
        
        # Build and solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for k in lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n-k] = (1.0 - h[k:]) * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n-m]) * dx
            constraints.append(coeff @ h_var <= t_var)
        
        # Add regularization to avoid wild oscillations
        progress = min(1.0, it / (n_iter * 0.5))
        reg = 0.01 * (1.0 - progress) + 1e-6 * progress
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        
        # Damped update
        damping = 0.7 + 0.2 * (it / n_iter)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter=2000):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    n_half = n - 1
    bounds = [(0.0, 1.0)] * n
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c_max = np.max(c)
        alpha = alphas[np.argmax([a for a in alphas])]  # placeholder
        return c_max
    
    # Use a simpler approach: just minimize the max overlap with penalty
    _target = target
    _n = n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr)) * (2.0 / _n)
    
    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_half = _n - 1
        best_k = int(np.argmax(corr)) - n_half
        g = np.zeros(_n)
        dx = 2.0 / _n
        abs_k = abs(best_k)
        if abs_k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif best_k > 0:
            g[:_n - best_k] += (1.0 - h[best_k:]) * dx
            g[best_k:] += (-h[:_n - best_k]) * dx
        else:
            m = -best_k
            g[m:] += (1.0 - h[:_n - m]) * dx
            g[:_n - m] += (-h[m:]) * dx
        return g
    
    def con_fun(h):
        return np.sum(h) - _target
    
    def con_jac(h):
        return np.ones(_n)
    
    h_cur = h_start.copy()
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    for alpha in alphas:
        def smooth_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / _n)
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return c_max + np.log(np.sum(np.exp(shifted))) / alpha
        
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def spectral_init(n, target, freq_choices):
    """Create initialization based on spectral design."""
    h = np.full(n, 0.5)
    for (freq, amp) in freq_choices:
        h += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    h = project_feasible(h, target)
    return h
