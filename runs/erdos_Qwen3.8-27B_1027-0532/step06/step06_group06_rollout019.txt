So we want h[i+k*] + h[i-k*] < h[j+k*] + h[j-k*].

For binary h, this means: the "context" around position i (at distances ±k*) should have fewer ones than the context around position j.

- For each position j with h[j]=0, compute score_j = h[j+k*] + h[j-k*] (0, 1, or 2)
- We want i with low score_i and j with high score_j

    current_max = float(corr.max())
    best_c5 = current_max * dx
    best_h = h.copy()
    
    for it in range(1000000):
        if time.time() - start_time > time_limit:
            break
        
        # Find top shifts
        top_k = 5
        top_indices = np.argsort(corr)[-top_k:][::-1]
        top_shifts = top_indices - (n-1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute combined score for each position
        # For a move (i,j), the change at shift k is:
        # Δ[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k] (with boundary)
        # We want Δ[k] < 0 for all critical k

        # For each j in zeros, compute "goodness" = sum over critical k of h[j+k] + h[j-k]
        # We want i with low badness and j with high goodness
        
        # But we also need to ensure the new max < current_max

        best_j = -1

                    if 0 <= j - k < n:

            break

2. The objective is max_k corr[k]. As a function of h[i*], each corr[k] is:

   corr[k] = sum_j h[j]*(1-h[j+k])

   - j = i*: h[i*]*(1-h[i*+k])
   - j+k = i*: j = i*-k: h[i*-k]*(1-h[i*])
   
   So corr[k] = h[i*]*(1-h[i*+k]) + h[i*-k]*(1-h[i*]) + (other terms)
   = h[i*] - h[i*]*h[i*+k] + h[i*-k] - h[i*-k]*h[i*] + c_k
   = h[i*]*(1 - h[i*+k] - h[i*-k]) + h[i*-k] + c_k

   This is linear in h[i*]! (Because h[i*+k] and h[i*-k] are fixed when we're optimizing h[i*].)

   So corr[k] = a_k * h[i*] + b_k, where:
   a_k = 1 - h[i*+k] - h[i*-k]
   b_k = h[i*-k] + c_k

   And we want to minimize max_k (a_k * h[i*] + b_k) subject to 0 ≤ h[i*] ≤ 1.

   This is the minimax of linear functions, which has a closed-form solution!

   The minimum of max_k (a_k * x + b_k) over x ∈ [0,1] is:
   - If all a_k = 0: min = max_k b_k (independent of x)
   - Otherwise: the optimal x is where the max is minimized

   This can be solved by finding the lower envelope of the lines and finding its minimum point.

   Actually, this is a 1D convex optimization (max of linear functions is convex). We can solve it by:
   - Binary search on the target value C
   - For a given C, check if there exists x ∈ [0,1] such that a_k*x + b_k ≤ C for all k
   - This is equivalent to: x ≤ (C-b_k)/a_k for all k with a_k > 0, and x ≥ (C-b_k)/a_k for all k with a_k < 0
   - So we need: max_k (C-b_k)/a_k (over a_k<0) ≤ min_k (C-b_k)/a_k (over a_k>0), and the intersection with [0,1] is non-empty

   This is O(n) per binary search step, and we need O(log(precision)) steps, so O(n log(precision)) per coordinate update.

   For n=606 and, say, 50 iterations of binary search, that's about 30,000 operations per coordinate update. With 606 coordinates, one full pass is about 18 million operations. That's fast!

        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Fix any remaining discrepancy
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
        final_sum = int(h_r.sum())
    return h_r

def coord_descent(h, target, max_passes, time_limit, start_time):
    n = len(h)
    h = project_box_sum(h.copy(), target)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_vals = np.arange(2*n-1) - (n-1)
    
    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        
        max_improve = 0.0
        for i in range(n):
            # Compute a_k for position i
            a_k = np.ones(2*n-1)
            valid_plus = (i + k_vals >= 0) & (i + k_vals < n)
            a_k[valid_plus] -= h[i + k_vals[valid_plus]]
            valid_minus = (i - k_vals >= 0) & (i - k_vals < n)
            a_k[valid_minus] -= h[i - k_vals[valid_minus]]
            
            b_k = corr - a_k * h[i]
            
            # Exclude k=0 (index n-1) to avoid the quadratic issue
            # Just use the linear approximation for all k
            # (For near-binary h, the k=0 term is negligible)
            
            # Minimize max(m) (a_k[m]*x + b_k[m]) over x in [0,1]
            # Bisection on C
            f0 = float(np.max(b_k))
            f1 = float(np.max(a_k + b_k))
            hi_C = min(f0, f1)
            lo_C = float(np.min(np.minimum(b_k, a_k + b_k))) - 1.0
            
            for _ in range(50):
                mid_C = (lo_C + hi_C) / 2.0
                # Check: exists x in [0,1] with a_k[m]*x + b_k[m] <= mid_C for all m?
                neg_a = a_k < -1e-12
                pos_a = a_k > 1e-12
                zero_a = np.abs(a_k) <= 1e-12
                
                feasible = True
                lower = 0.0
                upper = 1.0
                
                if np.any(zero_a) and float(np.max(b_k[zero_a])) > mid_C:
                    feasible = False
                
                if feasible and np.any(neg_a):
                    lb = float(np.max((mid_C - b_k[neg_a]) / a_k[neg_a]))
                    lower = max(lower, lb)
                
                if feasible and np.any(pos_a):
                    ub = float(np.min((mid_C - b_k[pos_a]) / a_k[pos_a]))
                    upper = min(upper, ub)
                
                if not feasible or lower > upper:
                    lo_C = mid_C
                else:
                    hi_C = mid_C
            
            # Find x at C = hi_C
            neg_a = a_k < -1e-12
            pos_a = a_k > 1e-12
            
            lower = 0.0
            upper = 1.0
            if np.any(neg_a):
                lower = max(lower, float(np.max((hi_C - b_k[neg_a]) / a_k[neg_a])))
            if np.any(pos_a):
                upper = min(upper, float(np.min((hi_C - b_k[pos_a]) / a_k[pos_a])))
            
            if lower > upper + 1e-12:
                x_new = h[i]
            else:
                x_new = max(0.0, min(1.0, (lower + upper) / 2.0))
            
            delta = x_new - h[i]
            if abs(delta) > 1e-10:
                # Update corr incrementally
                corr_new_max = float(np.max(corr + delta * a_k))
                old_max = float(np.max(corr))
                if corr_new_max <= old_max + 1e-10:
                    corr += delta * a_k
                    h[i] = x_new
                    max_improve = max(max_improve, old_max - float(np.max(corr)))
        
        # Project to maintain sum constraint
        h_proj = project_box_sum(h, target)
        if np.max(np.abs(h_proj - h)) > 1e-8:
            h = h_proj
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        if max_improve < 1e-8:
            break
    
    return h
