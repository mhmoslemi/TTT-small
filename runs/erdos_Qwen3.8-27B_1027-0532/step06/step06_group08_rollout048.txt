def project_box_sum(h, target):
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)

        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        if s != 0 and rng.random() < 0.6:
            si = np.zeros(len(ones))
            m1 = (ones + s) >= 0
            m1 &= (ones + s) < n
            si[m1] -= (1.0 - h[ones[m1] + s])
            m2 = (ones - s) >= 0
            m2 &= (ones - s) < n
            si[m2] += h[ones[m2] - s]
            i = ones[np.argsort(si)[:min(10, len(ones))][rng.randint(min(10, len(ones)))]]
        else:
            i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        corr = compute_corr(h)
        curr = float(corr.max()) * dx
        delta_c = curr - float((best_c5))
        if curr < best_c5 or (t > 1e-15 and rng.random() < np.exp(-(curr - float(corr.max() * 0 + best_c5)) / t)):
            pass
        accept = curr < best_c5 or (t > 1e-15 and rng.random() < np.exp(-(curr - best_h_score) / t)) if 'best_h_score' in dir() else curr < best_c5
        # Simpler SA: accept if better or with probability
        curr_score = float(corr.max()) * dx
        prev_score = best_c5 if curr_score >= best_c5 else best_c5
        # Use current track
        accept = curr_score < best_c5 or (t > 1e-15 and rng.random() < np.exp((best_c5 - curr_score) / t))
        if accept:
            if curr_score < best_c5:
                best_c5 = curr_score
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            corr = compute_corr(h)
            stall += 1
    return best_h, best_c5

def fw_smoothed(h0, n_iters, alpha, time_limit, start_time):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    for k in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break
        corr = compute_corr(h)
        max_c = corr.max()
        shifted = np.clip(alpha * (corr - max_c), -400.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(corr)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= w_sum
        g = np.zeros(n)
        for si in range(2 * n - 1):
            wi = w[si]
            if wi < 1e-8:
                continue
            s = si - (n - 1)
            g += wi * grad_shift(h, s)
        v = np.zeros(n)
        v[np.argsort(g)[:int(target)]] = 1.0
        gamma = 2.0 / (k + 2.0)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = proj_box_sum(h_new, target)
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)
        h = h_new
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def lp_relax(h0, max_active, time_limit, start_time):
    import cvxpy as cp
    n = len(h0)
    target = n / 2.0
    corr = compute_corr(h0)
    dx = 2.0 / n
    top_k = int(np.argmax(corr))
    active_set = [int(top_k) - (n - 1)]
    active_set = [s for s in active_set if s != 0]
    sorted_shifts = np.argsort(corr)[::-1]
    for idx in sorted_shifts[:max_active * 3]:
        s = int(idx) - (n - 1)
        if s != 0 and abs(s) <= n // 3:
            if s not in active_set:
                active_set.append(s)
        if len(active_set) >= max_active:
            break
    if len(active_set) == 0:
        return None, None
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    z_vars = {}
    for s in active_set:
        s_abs = abs(s)
        z_len = n - s_abs
        if z_len <= 0:
            continue
        z = cp.Variable(z_len)
        z_vars[s] = z
        if s >= 0:
            constraints.append(z <= h_var[:n - s])
            constraints.append(z <= h_var[s:])
            constraints.append(z >= h_var[:n - s] + h_var[s:] - 1)
            constraints.append(z >= 0)
            constraints.append(cp.sum(h_var[:n - s]) - cp.sum(z) <= t_var)
        else:
            t_pos = -s
            constraints.append(z <= h_var[t_pos:])
            constraints.append(z <= h_var[:n - t_pos])
            constraints.append(z >= h_var[t_pos:] + h_var[:n - t_pos] - 1)
            constraints.append(z >= 0)
            constraints.append(cp.sum(h_var[t_pos:]) - cp.sum(z) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers_to_try = ['MOSEK', 'GLOP', 'CBC', 'GLPK']
    result = None
    for solver in solvers_to_try[:3]:
        if time.time() - start_time > time_limit - 10:
            break
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                result = h_var.value.copy()
                break
        except Exception:
            continue
    if result is None:
        return None, None
    h_sol = np.clip(result, 0.0, 1.0)
    h_sol = proj_box_sum(h_sol, target)
    score = compute_c5(h_sol)
    return h_sol, score

def slsqp_refine(h_start, n, maxiter, ftol, time_limit, start_time):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(100.0 * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / 100.0
    def cons_fun(h):
        return np.sum(h) - target
    def cons_jac(h):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h = h_start.copy()
    curr_score = compute_c5(h)
    for alpha in [50, 200, 1000]:
        if time.time() - start_time > time_limit:
            break
        def smooth_obj(h, alpha=alpha):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        try:
            res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = proj_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h = h_new
        except Exception:
            pass
    return h, curr_score

def binary_local_from_continuous(h_cont, time_limit, start_time, rng):
    n = len(h_cont)
    h_bin = proj_binary(h_cont, n // 2)
    corr = compute_corr(h_bin)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h_bin.copy()
    stall = 0
    while time.time() - start_time < time_limit and stall < 300:
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        if s == 0:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        si = np.zeros(len(ones))
        m1 = (ones + s) >= 0
        m1 &= (ones + s) < n
        si[m1] -= (1.0 - h_bin[ones[m1] + s])
        m2 = (ones - s) >= 0
        m2 &= (ones - s) < n
        si[m2] += h_bin[ones[m2] - s]
        sj = np.zeros(len(zeros))
        m1j = (zeros + s) >= 0
        m1j &= (zeros + s) < n
        sj[m1j] += (1.0 - h_bin[zeros[m1j] + s])
        m2j = (zeros - s) >= 0
        m2j &= (zeros - s) < n
        sj[m2j] -= h_bin[zeros[m2j] - s]
        top_i = ones[np.argsort(si)[:min(30, len(ones))]]
        top_j = zeros[np.argsort(sj)[:min(30, len(zeros))]]
        best_d = 0.0
        bi = -1
        bj = -1
        for i in top_i:
            for j in top_j:
                if i == j:
                    continue
                d = 0.0
                if 0 <= i + s < n:
                    d -= (1.0 - h_bin[i + s])
                if 0 <= j + s < n:
                    d += (1.0 - h_bin[j + s])
                if 0 <= i - s < n:
                    d += h_bin[i - s]
                if 0 <= j - s < n:
                    d -= h_bin[j - s]
                if d < best_d:
                    best_d = d
                    bi = i
                    bj = j
        if bi >= 0:
            h_bin[bi] = 0.0
            h_bin[bj] = 1.0
            corr = compute_corr(h_bin)
            curr = float(corr.max()) * dx
            if curr < best_c5:
                best_c5 = curr
                best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def multi_swap_sa(h, n_iters, t_init, t_final, n_swaps, time_limit, start_time, rng):
    n = len(h)
    h = proj_binary(h.copy(), n // 2)
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 50000:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < n_swaps or len(zeros) < n_swaps:
            break
        i_idx = rng.choice(len(ones), size=n_swaps, replace=False)
        j_idx = rng.choice(len(zeros), size=n_swaps, replace=False)
        i_set = ones[i_idx]
        j_set = zeros[j_idx]
        h_new = h.copy()
        h_new[i_set] = 0.0
        h_new[j_set] = 1.0
        corr_new = compute_corr(h_new)
        curr = float(corr_new.max()) * dx
        if curr < best_c5 or (t > 1e-15 and rng.random() < np.exp((best_c5 - curr) / t)):
            h = h_new
            corr = corr_new
            if curr < best_c5:
                best_c5 = curr
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    if best_bin_score < 0.38085:

    if best_cont_score < 0.38085:
        return best_h_cont, best_cont_score, n_init

    # Phase 1: Binary greedy (fast)

        h_g, c5_g = binary_greedy(best_h_bin, allowed, start_time, rng)
        if c5_g < best_bin_score:
            best_bin_score = c5_g
            best_h_bin = h_g

    # Phase 2: Binary SA
    allowed = min(180.0, tleft() * 0.20)
    if allowed > 30:
        n_sa = int(min(5000000, allowed * 50000))
        h_sa, c5_sa = sa_binary(best_h_bin, n_sa, 0.015, 0.00003, allowed, start_time, rng)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_h_bin = h_sa

    # Phase 3: Multi-swap SA
    allowed = min(80.0, tleft() * 0.10)
    if allowed > 20:
        n_ms = int(min(200000, allowed * 3000))
        h_ms, c5_ms = multi_swap_sa(best_h_bin, n_ms, 0.05, 0.001, 3, allowed, start_time, rng)
        if c5_ms < best_bin_score:
            best_bin_score = c5_ms
            best_h_bin = h_ms

    # Phase 4: Frank-Wolfe smoothed (continuous)
    allowed = min(200.0, tleft() * 0.20)
    if allowed > 40:
        n_fw = int(min(100000, allowed * 500))
        h_fw, c5_fw = fw_smoothed(best_h_cont, n_fw, alpha=100.0, time_limit=allowed, start_time=start_time)
        if c5_fw < best_cont_score:
            best_cont_score = c5_fw
            best_h_cont = h_fw

    # Phase 5: Frank-Wolfe active
    allowed = min(120.0, tleft() * 0.12)
    if allowed > 30:
        n_fw2 = int(min(50000, allowed * 400))
        h_fw2, c5_fw2 = fw_active(best_h_cont, n_fw2, time_limit=allowed, start_time=start_time)
        if c5_fw2 < best_cont_score:
            best_cont_score = c5_fw2
            best_h_cont = h_fw2

    # Phase 6: LP relaxation
    allowed = min(250.0, tleft() * 0.25)
    if allowed > 60:
        h_lp, c5_lp = lp_relax(best_h_cont, max_active=20, time_limit=allowed, start_time=start_time)
        if h_lp is not None:
            if c5_lp < best_cont_score:
                best_cont_score = c5_lp
                best_h_cont = h_lp
            h_lp_bin, c5_lp_bin = binary_local_from_continuous(h_lp, min(60.0, tleft() * 0.1), start_time, rng)
            if c5_lp_bin < best_bin_score:
                best_bin_score = c5_lp_bin
                best_h_bin = h_lp_bin
            h_fw3, c5_fw3 = fw_smoothed(h_lp, int(min(30000, tleft() * 0.1 * 500)), alpha=200.0,
                                         time_limit=min(60.0, tleft() * 0.1), start_time=start_time)
            if c5_fw3 < best_cont_score:
                best_cont_score = c5_fw3
                best_h_cont = h_fw3

    # Phase 7: SLSQP polish
    allowed = min(150.0, tleft() * 0.15)
    if allowed > 30:
        h_ref, sc_ref = slsqp_refine(best_h_cont, n_init, maxiter=2000, ftol=1e-14,
                                      time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_h_cont = h_ref

    # Phase 8: Iterative improvement
    for iteration in range(5):
        if tleft() < 100:
            break
        # Binary local from continuous
        allowed = min(40.0, tleft() * 0.08)
        if allowed > 15:
            h_bl, c5_bl = binary_local_from_continuous(best_h_cont, allowed, start_time, rng)
            if c5_bl < best_bin_score:
                best_bin_score = c5_bl
                best_h_bin = h_bl
        # FW from best binary (smoothed)
        if tleft() < 80:
            break
        h_smooth = best_h_bin.copy()
        allowed = min(60.0, tleft() * 0.10)
        if allowed > 20:
            n_fw4 = int(min(20000, allowed * 300))
            h_fw4, c5_fw4 = fw_smoothed(h_smooth, n_fw4, alpha=50.0, time_limit=allowed, start_time=start_time)
            if c5_fw4 < best_cont_score:
                best_cont_score = c5_fw4
                best_h_cont = h_fw4
        # SLSQP
        if tleft() < 60:
            break
        allowed = min(50.0, tleft() * 0.08)
        if allowed > 20:
            h_ref2, sc_ref2 = slsqp_refine(best_h_cont, n_init, maxiter=1500, ftol=1e-15,
                                            time_limit=allowed, start_time=start_time)
            if sc_ref2 < best_cont_score:
                best_cont_score = sc_ref2
                best_h_cont = h_ref2

    # Phase 9: Try nearby resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8]:
        if tleft() < 80:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, n_init)
        h_interp = np.interp(x_new, x_old, best_h_cont)
        h_interp = proj_box_sum(h_interp, target_try)
        h_interp_bin = proj_binary(h_interp, n_try // 2)
        allowed = min(30.0, tleft() * 0.10)
        if allowed > 10:
            h_g3, c5_g3 = binary_greedy(h_interp_bin, allowed * 0.5, start_time, rng)
            h_fw5, c5_fw5 = fw_smoothed(h_g3, int(min(10000, allowed * 300)), alpha=80.0,
                                         time_limit=allowed, start_time=start_time)
            if c5_fw5 < best_cont_score:
                best_cont_score = c5_fw5
                best_h_cont = h_fw5
        allowed2 = min(30.0, tleft() * 0.10)
        if allowed2 > 15:
            h_ref3, sc_ref3 = slsqp_refine(h_interp, n_try, maxiter=1000, ftol=1e-14,
                                            time_limit=allowed2, start_time=start_time)
            if sc_ref3 < best_cont_score:

    # Phase 10: Final SLSQP burst
    if tleft() > 40:

                                              ftol=1e-15, time_limit=max(20.0, tleft() - 10),
                                              start_time=start_time)
        if sc_final < best_cont_score:
            best_cont_score = sc_final
            best_h_cont = h_final_try

    # Choose best
    n_final = len(best_h_cont)
    target_final = n_final / 2.0
    h_final_cont = proj_box_sum(best_h_cont, target_final)
    final_cont_score = compute_c5(h_final_cont)

    n_bin = len(best_h_bin)
    h_final_bin = proj_binary(best_h_bin, n_bin // 2)
    final_bin_score = compute_c5(h_final_bin)

    if final_bin_score <= final_cont_score:
        return h_final_bin, final_bin_score, n_bin
    else:
        return h_final_cont, final_cont_score, n_final
