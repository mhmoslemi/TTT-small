```python
def solve_full_lp_direct(h, n, target, dx):
    """Solve the alternating LP directly using scipy linprog."""
    g = 1.0 - h
    half_n = n - 1
    num_constraints = 2 * half_n + 1  # 2n-1
    
    # Build constraint matrix A (num_constraints x (n+1))
    # Each row corresponds to a lag k
    # A[k_idx, j] = dx * g[j - k] if valid, else 0
    # A[k_idx, n] = -1 (for the t variable)
    
    A = np.zeros((num_constraints, n + 1))
    b = np.zeros(num_constraints)
    
    for k_idx in range(num_constraints):
        k = k_idx - half_n  # k ranges from -(n-1) to (n-1)
        # Valid j: max(0, k) <= j <= min(n-1, n-1+k)
        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_hi >= j_lo:
            A[k_idx, j_lo:j_hi+1] = (g[j_lo - k:j_hi - k + 1] * dx)
        A[k_idx, n] = -1.0
    
    # Equality constraint: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Objective: minimize t (which is at index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h[i] <= 1, t is free (but should be >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_val = res.x[n]
        return h_new, t_val
    
    return None, None
```

```python
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
```

```python
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    b_ub = np.zeros(num_k)
    bounds_list = [(0.0, 1.0)] * n + [(0.0, None)]
    
    start = time.time()
    temp = 0.01
    temp_decay = 0.995
    
    for restart in range(max_restarts):
        if time.time() - start > time_limit:
            break
        
        # Generate perturbation
        # Mix of random and gradient-based
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        grad = np.zeros(n)
        j_lo = max(0, k_star)
        j_hi = min(n - 1, n - 1 + k_star)
        grad[j_lo:j_hi+1] = (1.0 - h[j_lo - k_star:j_hi - k_star + 1]) * dx
        
        # Random perturbation + gradient step
        scale = rng.uniform(0.01, 0.3) * temp
        noise = rng.randn(n) * scale
        lr = rng.uniform(0.1, 1.0) * temp
        h_pert = h - lr * grad + noise
        h_pert = project_feasible(h_pert, target)
        
        # Run a few LP iterations from the perturbed solution
        h_cur = h_pert.copy()
        for lp_it in range(20):
            if time.time() - start > time_limit:
                break
            g = 1.0 - h_cur
            for k_idx in range(num_k):
                j_lo2, j_hi2, k = ranges[k_idx]
```

```python
def _obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')

def _jobj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    logits = np.clip(alpha * (corr - c_max), -400.0, 0.0)

        h_ref = slsqp_refine(best_h, n, target, [500, 2000, 8000], 500,
                             min(35, tleft() * 0.7))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final micro-LP
    if tleft() > 20:
        h_lp, c5_lp = alternating_lp_fast(best_h, n, target,
                                          min(18, tleft() * 0.8), max_iter=20000)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Final projection
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
```
