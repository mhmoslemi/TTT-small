import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_target = 0.0

    return np.ones(_G_n)

def _slsqp_obj(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n

    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def precompute_f(h, n):

    return f

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    M = float(np.max(C))
    alpha = _G_alpha
    shifted = np.clip(alpha * (C - M), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = float(np.sum(exp_vals))
    obj = M + np.log(exp_sum) / alpha
    
    if exp_sum > 1e-300:
        weights = exp_vals / (alpha * exp_sum)
    else:
        weights = np.zeros(len(C))
    j_star = int(np.argmax(C))
    weights[j_star] += 1.0
    
    grad = np.zeros(n)
    mid = n - 1
    for j in range(len(weights)):
        wj = weights[j]
        if abs(wj) < 1e-14:
            continue
        k = j - mid
        grad += wj * get_grad_shift(h, k, n)
    grad *= dx
    
    return obj, grad

One more thought: should I try a completely different approach, like using CVXPY?

The problem is:
minimize t
subject to:
  dx * corr[j] ≤ t for j = 0, ..., 2n-2
  sum(h) = n/2
  0 ≤ h ≤ 1

This is non-convex because corr[j] = sum h[i]*(1-h[i+k]) is bilinear in h.

However, I can linearize it:
Let u_j = dx * corr[j]. Then u_j = dx * sum h[i] - dx * sum h[i]*h[i+k] = 1 - dx * sum h[i]*h[i+k] (since sum h[i] = n/2, dx*n/2 = 1).

So u_j = 1 - dx * autocorr[j] where autocorr[j] = sum h[i]*h[i+k].

We want to minimize max_j u_j = 1 - min_j dx*autocorr[j].
Equivalently, maximize min_j dx*autocorr[j].

The autocorr[j] is a quadratic form in h. The problem of maximizing the minimum of quadratic forms is non-convex.

I don't think CVXPY can handle this directly. Let me stick with the gradient-based approach.
