import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.where(h_proj < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h_proj > 1e-14)[0]
        if len(idx) > 0:
            h_proj[idx] += diff / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_obj_val(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad = grad - np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            return smooth_obj_val(x, alpha, n)
        def _grad(x):
            return smooth_obj_grad(x, alpha, n)
        def _con(x):
            return np.sum(x) - target
        def _con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lp_refine_advanced(h0, n, target, time_limit, n_active=30, max_rounds=80, eps_frac=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        c_max = np.max(corr)
        threshold = c_max * (1.0 - eps_frac)
        active_mask = corr >= threshold
        active = np.where(active_mask)[0]
        if len(active) > n_active:
            sorted_active = active[np.argsort(corr[active])[::-1]][:n_active]
        else:
            sorted_active = active
        if len(sorted_active) < 2:
            sorted_active = np.argsort(corr)[::-1][:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in sorted_active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.CVXOPT, cp.GLPK]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_lp = np.clip(np.asarray(h_var.value), 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            improved = False
            for mix in [0.7, 0.5, 0.3, 0.1]:
                h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    improved = True
                    break
            if not improved:
                eps_frac = max(0.01, eps_frac * 0.7)
                if eps_frac < 0.015:
                    break
    return best_h, best_c5

def fast_cd(h0, n, target, time_limit, max_sweeps=15000, n_cand=20):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_max = min(1.0 - h[j], h[i])
                if d_max < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                v_im = (i - shifts >= 0) & (i - shifts < n)
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                g[v_im] += h[i - shifts[v_im]]
                g[v_jp] -= h[j + shifts[v_jp]]
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d, b_d = 0.0, d_max
                for _ in range(28):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def gradient_descent_smooth(h0, n, target, time_limit, max_steps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_cnt = 1.0
    stagnation = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        alpha = 100.0 * (1.0 + step * 0.02)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 3000:
            break
        grad = smooth_obj_grad(h, alpha, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_cnt * t_cnt)) / 2.0
        beta = (t_cnt - 1.0) / t_new
        t_cnt = t_new
        base_lr = 2.0 / (1.0 + t_cnt)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h + beta * v - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-16:
                v = beta * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
            if step > 2000:
                break
    return best_h, best_c5

def perturb_restart(h0, n, target, n_perturb, time_limit):
    start = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    target_sum = target
    for p in range(n_perturb):
        if time.time() - start > time_limit:
            break
        rng = np.random.default_rng(42 + p * 1000)
        h_pert = h0.copy()
        n_change = max(5, n // 20)
        idx = rng.choice(n, size=n_change, replace=False)
        h_pert[idx] += rng.uniform(-0.3, 0.3, size=n_change)
        h_pert = project_feasible(h_pert, target_sum)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            h_sub, c5_sub = slsqp_round(h_pert, n, target_sum, [100, 500, 2000, 10000],
                                         500, 1e-13, min(30.0, (time_limit - (time.time() - start)) * 0.3), start)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub
    return best_h, best_c5
