def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h

    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

            h = np.clip(h, 0.0, 1.0)

    c[n] = 1.0  # minimize t

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    return h_cur

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.fun)
    except Exception:
        pass
    return h_cur, np.inf


def lp_minimax_descent(h_init, target, time_limit, deadline, max_iters=200000):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    stall_count = 0

    for it in range(max_iters):
        elapsed = time.time()
        if elapsed > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break

        G = compute_all_grads(h)
        h_lp, t_star = solve_lp_minimax(G, corr, h, target)

        if np.isfinite(t_star) and t_star < float(np.max(corr)) - 1e-10:
            h_candidate = project_feasible(h_lp, target)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5_now - 1e-14:
                h = h_candidate
            else:
                frac = 0.5
                for _ in range(6):
                    h_frac = (1.0 - frac) * h + frac * h_lp
                    h_frac = project_feasible(h_frac, target)
                    c5_frac = compute_c5(h_frac)
                    if c5_frac < c5_now - 1e-14:
                        h = h_frac
                        break
                    frac *= 0.5
        else:
            stall_count += 1
            if stall_count > 50:
                top_k = np.argsort(corr)[-10:]
                Grad = np.zeros(n)
                for ki in top_k:
                    Grad += G[ki, :]
                Grad /= float(len(top_k))
                Grad *= dx
                Grad -= np.mean(Grad)
                improved = False
                for lr in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                    h_try = project_feasible(h - lr * Grad, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_now - 1e-14:
                        h = h_try
                        improved = True
                        stall_count = 0
                        break
                if not improved and stall_count > 150:
                    rhs = best_c5
                    rng_esc = np.random.RandomState(it)
                    noise = rng_esc.randn(n) * 0.03
                    h_pert = project_feasible(h + noise, target)
                    c5_pert = compute_c5(h_pert)
                    if c5_pert < rhs + 0.001:
                        h = h_pert
                        stall_count = 0

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def subgradient_polish(h_init, target, deadline, max_steps=50000):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    m_top = 8

    for step in range(max_steps):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        top_m = np.argsort(corr)[-m_top:]
        grad = np.zeros(n)
        G = compute_all_grads(h)
        for ki in top_m:
            grad += G[ki, :]
        grad /= float(m_top)
        grad *= dx
        grad -= np.mean(grad)

        lr = 0.03 * max(0.1, 1.0 - step / max_steps)
        improved = False
        for _ in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            lr *= 0.5

        if not improved:
            if no_improve > 200:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP-based linearized minimax (main optimization)
    if time.time() < deadline - 300:
        lp_deadline = time.time() + 450
        h_lp, c5_lp = lp_minimax_descent(best_h, target, 450, lp_deadline)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: L-BFGS-B polish with multiple smoothings
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for alpha in [200, 1000, 5000, 20000]:
            if time.time() > deadline - 100:
                break
            h_try = lbfags_polish(best_h, n_cur, target_cur, alpha, maxiter=2000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 3: Second round of LP descent after polish
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        lp_deadline2 = time.time() + 200
        h_lp2, c5_lp2 = lp_minimax_descent(best_h, target_cur, 200, lp_deadline2)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 4: Subgradient polish
    if time.time() < deadline - 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        sg_deadline = time.time() + 100
        h_sg, c5_sg = subgradient_polish(best_h, target_cur, sg_deadline)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Higher resolution attempt
    if time.time() < deadline - 150:
        n_cur = len(best_h)
        for n_try in [600, 800]:
            if time.time() > deadline - 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            sc_interp = compute_c5(h_interp)
            lp_dl = time.time() + min(80, deadline - time.time() - 40)
            h_hires, c5_hires = lp_minimax_descent(h_interp, target_try,
                                                   min(80, deadline - time.time() - 40),
                                                   lp_dl, max_iters=50000)
            if c5_hires < best_score:
                best_score = c5_hires
                best_h = h_hires
                n_cur = n_try
            elif sc_interp < best_score:
                best_score = sc_interp
                best_h = h_interp
                n_cur = n_try

    # Phase 6: Final LP polish at current resolution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if time.time() < deadline - 30:
        lp_fnl = time.time() + min(60, deadline - time.time() - 10)
        h_fnl, c5_fnl = lp_minimax_descent(best_h, target_final,
                                           min(60, deadline - time.time() - 10),
                                           lp_fnl, max_iters=50000)
        if c5_fnl < best_score:
            best_score = c5_fnl
            best_h = h_fnl

    # Phase 7: Final L-BFGS-B
    if time.time() < deadline - 10:
        h_pol = lbfags_polish(best_h, n_final, target_final, 50000, maxiter=3000)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
