import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_c5_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    return float((m + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def smooth_c5_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    return grad


def slsqp_eq_fun(x, target):
    return np.sum(x) - target


def slsqp_eq_jac(x, target):
    return np.ones(len(x))


def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        obj_func = partial(smooth_c5_val, alpha=alpha)
        grad_func = partial(smooth_c5_grad, alpha=alpha)
        eq_fun = partial(slsqp_eq_fun, target=target)
        eq_jac = partial(slsqp_eq_jac, target=target)
        try:
            res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5


def cd_compute_gradient_multi(h, n, top_shifts):
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for s_idx in top_shifts:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad


def cd_eval_pair(corr, g, delta):
    vals = corr + delta * g
    return float(np.max(vals))


def cd_bisect_delta(corr, g, d_max, n_iters=25):
    a = 0.0
    b = d_max
    for _ in range(n_iters):
        mid_d = 0.5 * (a + b)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-13
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-13:
            b = mid_d
        else:
            a = mid_d
    return 0.5 * (a + b)


def coord_descent(h0, n, target, time_limit, max_sweeps=10000, n_cand=25, n_shifts=3):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    shifts_arr = np.arange(-(n - 1), n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        top_shifts = np.argsort(corr)[::-1][:n_shifts]
        grad = cd_compute_gradient_multi(h, n, top_shifts)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(2 * n - 1)
                v_ip = (i + shifts_arr >= 0) & (i + shifts_arr < n)
                g[v_ip] += h[i + shifts_arr[v_ip]]
                v_im = (i - shifts_arr >= 0) & (i - shifts_arr < n)
                g[v_im] += h[i - shifts_arr[v_im]]
                v_jp = (j + shifts_arr >= 0) & (j + shifts_arr < n)
                g[v_jp] -= h[j + shifts_arr[v_jp]]
                v_jm = (j - shifts_arr >= 0) & (j - shifts_arr < n)
                g[v_jm] -= h[j - shifts_arr[v_jm]]
                delta_star = cd_bisect_delta(corr, g, d_hi, 22)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = cd_eval_pair(corr, g, delta_star) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5


def lp_refine(h0, n, target, time_limit, n_active=15, max_rounds=50):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fista_optimize(h0, n, target, time_limit, max_steps=15000, alpha_start=200, alpha_end=100000):
    start = time.time()
    h = h0.copy()
    y = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        progress = min(step / max_steps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        grad = smooth_c5_grad(y, alpha)
        lr = 2.0 / (1.0 + t_k)
        h_new = project_feasible(y - lr * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        t_k = t_new
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
    return best_h, best_c5


def shake_and_reopt(h0, n, target, time_limit, rng):
    h = h0.copy()
    noise = rng.normal(0, 0.03, n)
    h_pert = project_feasible(h + noise, target)
    alphas = [200, 1000, 5000, 20000]
    h_opt, s_opt = slsqp_optimize(h_pert, n, target, alphas, 1500, 1e-14,
                                   time_limit * 0.6, time.time())
    if s_opt < compute_c5(h0):
        return h_opt, s_opt
    return h0, compute_c5(h0)
