import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse(h, alpha, n):
    """Log-sum-exp approximation of max, scaled by dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 50.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_grad(h, alpha, n):
    """Gradient of the log-sum-exp smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 50.0)
    w = np.exp(shifted)
    ws = w.sum()
    w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    # Only iterate over significant weights for efficiency
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)  # Zero the component along the constraint normal
    return grad

def fista_smooth(h0, n, target, alpha_seq, time_limit, start_time):
    """FISTA with adaptive smoothing."""
    dx = 2.0 / n
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0
    
    for alpha in alpha_seq:
        if time.time() - start_time > time_limit:
            break
        max_iters = 2000
        for it in range(max_iters):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            beta = (t_val - 1.0) / t_new
            y = h + beta * (y - h)
            t_val = t_new
            
            # Compute gradient at y
            grad_y = smooth_grad(y, alpha, n)
            
            # Lipschitz estimate (rough)
            # f is bilinear with values in [0,1], so Lipschitz constant is bounded
            # |f'| <= 2 * dx * n (very loose)
            # Better: L ~ 2 * max(corr) / dx (from the quadratic structure)
            # Use backtracking
            L = 2.0 / dx * 1.5  # Initial guess
            x_trial = y - (1.0 / L) * grad_y
            x_trial = project_feasible(x_trial, target)
            
            # Backtracking line search
            f_trial = smooth_lse(x_trial, alpha, n)
            f_y = smooth_lse(y, alpha, n)
            back = 0
            while back < 30:
                if f_trial <= f_y + np.dot(grad_y, x_trial - y) + 0.5 * L * np.dot(x_trial - y, x_trial - y):
                    break
                L *= 2.0
                x_trial = project_feasible(y - (1.0 / L) * grad_y, target)
                f_trial = smooth_lse(x_trial, alpha, n)
                back += 1
            
            h = x_trial
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
    
    return best_h, best_c5

def sa_kick(h, n, target, kick_size, rng):
    """Apply a random perturbation to escape local optima."""
    h_new = h.copy()
    idx = rng.choice(n, size=kick_size, replace=False)
    h_new[idx] += rng.uniform(-0.05, 0.05, size=kick_size)
    return project_feasible(h_new, target)

        else:

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                    continue

                    best_delta = delta_star
                    best_i = i

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    """SLSQP with smooth max objective."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def _con(x):
        return np.sum(x) - target
    
    def _con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def _obj(x):
            return smooth_lse(x, alpha, n)
        
        def _grad(x):
            return smooth_grad(x, alpha, n)
        
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

    h_out = np.interp(x_new, x_old, h_src)

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse_val(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 50.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_lse_grad(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 50.0)
    w = np.exp(shifted)
    ws = w.sum()
    w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def _slsqp_obj(x):
    return smooth_lse_val(x, _slsqp_state['alpha'], _slsqp_state['n'])

def _slsqp_grad(x):
    return smooth_lse_grad(x, _slsqp_state['alpha'], _slsqp_state['n'])

def _slsqp_con(x):
    return np.sum(x) - _slsqp_state['target']

def _slsqp_con_jac(x):
    return np.ones(_slsqp_state['n'])

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    _slsqp_state['n'] = n
    _slsqp_state['target'] = target
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        _slsqp_state['alpha'] = alpha
        try:
            res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def fista_smooth(h0, n, target, alpha_seq, time_limit, start_time):
    dx = 2.0 / n
    h = h0.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0
    for alpha in alpha_seq:
        if time.time() - start_time > time_limit:
            break
        max_iters = 3000
        for it in range(max_iters):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            beta = (t_val - 1.0) / t_new
            y = h + beta * (y - h)
            t_val = t_new
            grad_y = smooth_lse_grad(y, alpha, n)
            L = 4.0 / dx
            back = 0
            while back < 25:
                x_trial = project_feasible(y - (1.0 / L) * grad_y, target)
                f_trial = smooth_lse_val(x_trial, alpha, n)
                f_y = smooth_lse_val(y, alpha, n)
                quad = 0.5 * L * np.dot(x_trial - y, x_trial - y)
                if f_trial <= f_y + np.dot(grad_y, x_trial - y) + quad:
                    break
                L *= 2.0
                back += 1
            h = x_trial
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5

def fast_cd_polish(h0, n, target, time_limit, n_cand=20):

    no_improve = 0

        grad[v2] -= h[idxs[v2] + k_star]

                    best_i = i
                    best_j = j

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sa_kick(h, n, target, kick_fraction, rng):
    h_new = h.copy()
    kick_size = max(1, n // 10)
    idx = rng.choice(n, size=kick_size, replace=False)
    h_new[idx] += rng.uniform(-0.03, 0.03, size=kick_size)
    return project_feasible(h_new, target)

def multi_start_slsqp(best_h, n, target, n_restarts, time_limit, start_time, rng):
    current_best = best_h.copy()
    current_score = compute_c5(best_h)
