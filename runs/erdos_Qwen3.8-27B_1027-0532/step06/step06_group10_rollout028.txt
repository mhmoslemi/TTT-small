def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Map h to [0,1] and adjust sum to target_sum
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Handle residual from clipping
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx2 = np.where(h > 1e-8)[0]
        if len(idx2) > 0:
            per = residual / len(idx2)
            h[idx2] = np.clip(h[idx2] + per, 0.0, 1.0)
    return h

            no_improve += 1
        if no_improve > 300:
            break
        
        # Compute gradient for the worst shift
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Select candidate coordinates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                
                # Direction: decrease h[i], increase h[j]
                # Compute the effect on all correlation values
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                
                # Binary search for optimal delta
                a_d = 0.0
                b_d = d_hi
                for _ in range(28):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = int(i)
                    best_j = int(j)
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def slsqp_optimize(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    """SLSQP with smooth max objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    idxs = np.arange(n)
    mid = n - 1
    
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit + 20:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_val(corr, alpha) * (2.0 / n)
        
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            w = smooth_max_grad(corr, alpha, n)
            g = np.zeros(n)
            active = np.where(w > 1e-8)[0]
            for s in active:
                k = int(s) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += w[s] * (1.0 - x[idxs[v1] - k])
                g[v2] -= w[s] * x[idxs[v2] + k]
            g *= 2.0 / n
            g -= np.mean(g)
            return g
        
        def con_fun(x):
            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_perturb(h, n, n_modes, amplitude, rng):
    """Add Fourier perturbations to h"""
    x = np.arange(n)
    h_new = h.copy()
    for k in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        h_new += a * np.cos(2 * np.pi * k * x / n) + b * np.sin(2 * np.pi * k * x / n)
    return project_feasible(h_new, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx2 = np.where(h > 1e-6)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def smooth_max_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        g[v2] -= w[s] * x[idxs[v2] + k]
    g *= 2.0 / n
    g -= np.mean(g)
    return g

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_budget, ref_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > ref_time + time_budget:
            break
        res = minimize(smooth_max_obj, h, args=(alpha, n),
                       jac=lambda x: smooth_max_grad(x, alpha, n),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq',

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_eq_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx2 = np.where(h > 1e-6)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _con_eq(x):
    return np.sum(x) - _eq_target

def _con_eq_jac(x):
    return np.ones(len(x))

def _smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def _smooth_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:

        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        g[v2] -= w[s] * x[idxs[v2] + k]
    g *= 2.0 / n
    g -= np.mean(g)
    return g

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_budget, ref_time):

    _eq_target = target

def active_set_lp(h0, n, target, time_limit, max_rounds, init_active):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        n_active = min(n_corr, init_active + 5 * rnd)
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.CLARABEL, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iter=10000)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)

        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:

        else:

                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    improved = True
                    break

def coord_descent(h0, n, target, time_limit, max_sweeps, n_cand):

    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for ii in range(len(high_set)):
            for jj in range(len(low_set)):
                i = int(high_set[ii])
                j = int(low_set[jj])
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(26):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_perturb(h, n, n_modes, amplitude, rng):
    x = np.arange(n, dtype=float)
    h_new = h.copy()
    for k in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        h_new += a * np.cos(2.0 * np.pi * k * x / n) + b * np.sin(2.0 * np.pi * k * x / n)
    return project_feasible(h_new, float(n) / 2.0)

def random_feasible_init(n, rng):
    h = rng.uniform(0.0, 1.0, n)

            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [5000, 20000, 80000, 300000],
                                3000, 1e-15, min(100.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 150:
        h_lp, c5_lp = active_set_lp(best_h, n, target,
                                     time_limit=min(130.0, tleft() * 0.35),
                                     max_rounds=70, init_active=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 120:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     time_limit=min(100.0, tleft() * 0.3),
                                     max_sweeps=10000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 200:
        for n_coarse in [200, 150]:
            if tleft() < 100:
                break
            target_c = float(n_coarse) / 2.0
            h_c = interpolate_to(best_h, n_coarse, target_c)
            h_c, s_c = active_set_lp(h_c, n_coarse, target_c,
                                      time_limit=min(40.0, tleft() * 0.15),
                                      max_rounds=30, init_active=20)
            h_c, s_c = coord_descent(h_c, n_coarse, target_c,
                                      time_limit=min(20.0, tleft() * 0.08),
                                      max_sweeps=4000, n_cand=15)
            h_up = interpolate_to(h_c, n, target)
            for blend_frac in [0.5, 0.7, 0.3]:
                h_blend = project_feasible(blend_frac * best_h + (1.0 - blend_frac) * h_up, target)
                c5_blend = compute_c5(h_blend)
                if c5_blend < best_score:
                    best_score = c5_blend
                    best_h = h_blend

    if tleft() > 200:
        for pert_amp in [0.08, 0.04, 0.02]:
            if tleft() < 80:
                break
            h_pert = fourier_perturb(best_h, n, 6, pert_amp, rng)
            h_opt, s_opt = slsqp_round(h_pert, n, target,
                                        [200, 1000, 5000, 20000],
                                        2000, 1e-14, min(50.0, tleft() * 0.2), start_time)
            h_opt, s_opt2 = coord_descent(h_opt, n, target,
                                           time_limit=min(25.0, tleft() * 0.1),
                                           max_sweeps=5000, n_cand=20)
            if s_opt2 < best_score:
                best_score = s_opt2
                best_h = h_opt

    if tleft() > 150:
        h_rand = random_feasible_init(n, rng)
        h_opt, s_opt = slsqp_round(h_rand, n, target,
                                    [100, 500, 2000, 10000],
                                    2000, 1e-14, min(60.0, tleft() * 0.3), start_time)
        h_opt, s_opt2 = coord_descent(h_opt, n, target,
                                       time_limit=min(30.0, tleft() * 0.12),
                                       max_sweeps=5000, n_cand=20)
        if s_opt2 < best_score:
            best_score = s_opt2
            best_h = h_opt

    if tleft() > 80:
        h_lp, c5_lp = active_set_lp(best_h, n, target,
                                     time_limit=min(60.0, tleft() * 0.5),
                                     max_rounds=50, init_active=50)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 50:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     time_limit=min(40.0, tleft() * 0.6),
                                     max_sweeps=8000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [10000, 50000, 200000],
                                2000, 1e-15, min(25.0, tleft() * 0.7), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 15:
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     time_limit=min(12.0, tleft() * 0.7),
                                     max_sweeps=5000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
