def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    """Compute smooth max of overlaps and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx  # length 2n-1
    
    if top_k is not None and top_k < len(overlaps):
        top_idx = np.argsort(overlaps)[::-1][:top_k]
    else:

    ov = overlaps[top_idx]
    ov_max = np.max(ov)
    shifted = alpha * (ov - ov_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    probs = exps / Z
    
    # Smooth max value
    val = ov_max + np.log(Z) / alpha
    
    # Gradient: sum of probs * grad(overlap_k)
    grad = np.zeros(n)

        return h

    h = np.clip(h, 0.0, 1.0)

        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    Z = np.sum(exps)

            B = 0.0

                    # = dx * (-t^2 + t + rest)
                    A += -dx
                    B += dx

                    #   contribution: -h[j+m]
                    a = 0.0

                        a += (1.0 - h[j - m])
                    if j <= n - 1 - m:
                        a -= h[j + m]

                h[j] = t_new
                improved = True
        
        if not improved:
            break
    h = project_feasible(h, target)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    n_half = n - 1
    ov_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - ov_max), -500.0, 0.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    probs = exps / Z
    val = ov_max + np.log(Z) / alpha
    grad = np.zeros(n)
    for idx in range(len(overlaps)):
        p = probs[idx]
        if p < 1e-9:
            continue
        k = idx - n_half
        if k == 0:
            grad += p * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[:n - k] += p * (1.0 - h[k:]) * dx
            grad[k:] += p * (-h[:n - k]) * dx
        else:
            m = -k
            grad[m:] += p * (1.0 - h[:n - m]) * dx
            grad[:n - m] += p * (-h[m:]) * dx
    return val, grad


def lbfgs_phase(h0, n, target, alpha, beta, maxiter):
    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        val, g = smooth_max_val_grad(h, alpha, n)
        sum_diff = np.sum(h) - target
        val += beta * sum_diff * sum_diff
        g = g + 2.0 * beta * sum_diff
        return val, g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h0, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                           'gtol': 1e-12, 'iprint': -1, 'maxls': 60})
    h_new = project_feasible(res.x, target)
    return h_new, compute_c5(h_new)


def coordinate_descent(h, n, target, n_sweeps=5, top_k=30):
    h = h.copy()
    dx = 2.0 / n
    n_half = n - 1
    for sweep in range(n_sweeps):
        for j in range(n):
            overlaps = compute_overlaps(h)
            top_idx = np.argsort(overlaps)[::-1][:top_k]
            A = 0.0
            B = 0.0
            for idx in top_idx:
                k = idx - n_half
                if k == 0:
                    A += -dx
                    B += dx
                elif k > 0:
                    a = 0.0
                    if j <= n - 1 - k:
                        a += (1.0 - h[j + k])
                    if k <= j:
                        a -= h[j - k]
                    B += dx * a
                else:
                    m = -k
                    a = 0.0
                    if m <= j:
                        a += (1.0 - h[j - m])
                    if j <= n - 1 - m:
                        a -= h[j + m]
                    B += dx * a
            if abs(A) < 1e-15:
                t_new = 0.0 if B > 0 else 1.0
            else:
                t_new = -B / (2.0 * A)
                t_new = np.clip(t_new, 0.0, 1.0)
            h[j] = t_new
        h = project_feasible(h, target)
    return h

            best_score = c5_fw
            best_h = h_fw
        tl = time_remaining(start_time, budget_s)
        if tl > 30:
            best_h = coordinate_descent(best_h, n, target, n_sweeps=5, top_k=30)
            sc = compute_c5(best_h)
            if sc < best_score:
                best_score = sc

    # Phase 6: Alternating LP
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        lp_time = min(180, tl * 0.2)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=2000, time_limit=lp_time,
                                      n_top=300, damping=0.9,
                                      reg_start=1e-3, reg_end=1e-9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
        tl = time_remaining(start_time, budget_s)
        if tl > 30:
            best_h = coordinate_descent(best_h, n, target, n_sweeps=5, top_k=30)
            sc = compute_c5(best_h)
            if sc < best_score:
                best_score = sc

    # Phase 7: Simulated annealing for exploration
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        sa_time = min(100, tl * 0.12)
        h_sa, c5_sa = simulated_annealing(best_h, n, target, sa_time, rng)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
        tl = time_remaining(start_time, budget_s)
        if tl > 40:
            h_new, score = lbfgs_phase(best_h, n, target, 20000, beta=10000,
                                        maxiter=min(int(tl * 3), 1500))
            h_new = coordinate_descent(h_new, n, target, n_sweeps=5, top_k=30)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 8: Final L-BFGS-B with very high alpha
    tl = time_remaining(start_time, budget_s)
    if tl > 50:
        for alpha in [200000, 1000000]:
            tl = time_remaining(start_time, budget_s)
            if tl < 30:
                break
            maxiter = min(int(tl * 3), 2000)
            h_new, score = lbfgs_phase(best_h, n, target, alpha, beta=20000, maxiter=maxiter)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 9: Final coordinate descent
    tl = time_remaining(start_time, budget_s)
    if tl > 20:
        best_h = coordinate_descent(best_h, n, target, n_sweeps=15, top_k=25)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    # Phase 10: One more L-BFGS-B at very high alpha
    tl = time_remaining(start_time, budget_s)
    if tl > 30:
        h_new, score = lbfgs_phase(best_h, n, target, 2000000, beta=50000,
                                    maxiter=min(int(tl * 2), 1000))
        h_new = coordinate_descent(h_new, n, target, n_sweeps=10, top_k=20)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
