```python
def full_lp(h_old, n, target, reg, eps=1e-10, max_iters=500000):
    """Solve one LP iteration with ALL correlation constraints."""
    dx = 2.0 / n
    g = 1.0 - h_old  # fixed
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Build constraint matrices efficiently
    # For lag k (k from -(n-1) to (n-1)):
    # overlap(k) = sum_{valid i} h[i] * g[i+k] * dx <= t
    # where valid means 0 <= i < n and 0 <= i+k < n
    
    # For k >= 0: i ranges from 0 to n-1-k, and g index is i+k (ranges from k to n-1)
    # For k < 0: i ranges from -k to n-1, and g index is i+k (ranges from 0 to n-1+k)
    
    half_n = n - 1
    for k_idx in range(1, 2 * n - 1):
        k = k_idx - half_n
        if k >= 0:
            # h indices: 0 to n-1-k
            # g indices: k to n-1
            h_idx = np.arange(0, n - k)
            g_idx = np.arange(k, n)
        else:
            # h indices: -k to n-1
            # g indices: 0 to n-1+k
            h_idx = np.arange(-k, n)
            g_idx = np.arange(0, n + k)
        
        if len(h_idx) == 0:
            continue
        
        coeff = np.zeros(n)
        coeff[h_idx] = g[g_idx] * dx
        constraints.append(coeff @ h_var <= t_var)
    
    objective = reg * cp.sum_squares(h_var - h_old) + t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=max_iters)
        except Exception:
            return None, None
    
    if h_var.value is None:
        return None, None
    
    h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    t_val = float(t_var.value) if t_var.value is not None else compute_c5(h_new)
    return h_new, t_val

def alternating_lp_all_constraints(h_init, n, target, max_iter, time_limit, 
                                    reg_start, reg_end, damping_start, damping_end):
    """Alternating LP with all constraints and geometric regularization reduction."""
    start_time = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Geometric reduction of regularization
        frac = it / max(1, max_iter)
        reg = reg_start * (reg_end / reg_start) ** frac
        damping = damping_start + (damping_end - damping_start) * frac
        
        h_new, t_val = full_lp(h, n, target, reg=reg)
        if h_new is None:
            break
        
        h_next = (1 - damping) * h + damping * h_new
        h_next = project_feasible(h_next, target)
        
        c5_val = compute_c5(h_next)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_next.copy()
        
        h = h_next
    
    return best_h, best_c5
```

```python
from scipy.sparse import csr_matrix
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_A_matrix(g, n, dx):
    """Build the constraint matrix A where (A @ h)[k_idx] = overlap at lag k_idx."""
    half_n = n - 1
    rows = []
    cols = []
    vals = []
    for k_idx in range(1, 2 * n - 1):

        mask = gz > 1e-15

            rows.extend((np.full(np.sum(mask), k_idx)).tolist())
            cols.extend(h_idx[mask].tolist())
            vals.extend((gz[mask] * dx).tolist())
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_A(g, n, dx):
    half_n = n - 1
    row_list = []
    col_list = []
    val_list = []
    for k_idx in range(1, 2 * n - 1):
        k = k_idx - half_n
        if k >= 0:
            hi = n - k
        else:
            hi = n + k
        if hi <= 0:
            continue
        if k >= 0:
            # h[i] * g[i+k] for i = 0..hi-1
            g_seg = g[k:k+hi]
            nz = np.nonzero(g_seg > 1e-14)[0]
            if len(nz) > 0:
                row_list.append(np.full(len(nz), k_idx))
                col_list.append(nz)
                val_list.append(g_seg[nz] * dx)
        else:
            # h[i] * g[i+k] for i = -k..n-1
            g_seg = g[0:hi]
            nz = np.nonzero(g_seg > 1e-14)[0]
            if len(nz) > 0:
                row_list.append(np.full(len(nz), k_idx))
                col_list.append(nz - k)  # shift by -k
                val_list.append(g_seg[nz] * dx)
    if len(row_list) == 0:
        return csr_matrix((2*n-1, n))
    rows = np.concatenate(row_list)
    cols = np.concatenate(col_list)
    vals = np.concatenate(val_list)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_full_lp(h_old, n, target, reg, eps=1e-10, max_iters=500000):
    dx = 2.0 / n
    g = 1.0 - h_old
    A = build_A(g, n, dx)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        A @ h_var <= t_var
    ]
    objective = reg * cp.sum_squares(h_var - h_old) + t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters)
    except Exception:
        return None, None
    if h_var.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    h_new = np.asarray(h_var.value).ravel()
    t_val = float(t_var.value) if t_var.value is not None else 0
    return h_new, t_val

def full_lp_phase(h_init, n, target, time_limit, n_iters=None):
    """Run the full-constraint LP phase with geometric schedule."""
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if n_iters is None:
        n_iters = 2000
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        frac = min(1.0, (it + 1) / n_iters)
        # Reg decreases from 1.0 to 0.01
        reg = 1.0 * (0.01) ** frac
        # Damping increases from 0.7 to 1.0
        damping = 0.7 + 0.3 * frac
        
        h_sol, t_val = solve_full_lp(h, n, target, reg=reg, eps=1e-10)
        if h_sol is None:
            break
        
        h_new = (1 - damping) * h + damping * h_sol
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    start = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            # For coordinate i, find the optimal value that minimizes the max
            # over active constraints
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            
            # Minimize max_j (slopes[j]*x + intercepts[j]) over x in [0,1]
            # via bisection
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = slopes * mid + intercepts
                avg_active = np.mean(vals[np.argsort(vals)[-min(10, n_active):]])
                if np.max(vals) > avg_active + 1e-12:
                    # Not converged
                    pass
                # More robust: check subgradient
                active_mask = np.abs(vals - np.max(vals)) < 1e-10 * max(1.0, abs(np.max(vals)))
                if not np.any(active_mask):
                    active_mask = np.zeros(n_active, dtype=bool)
                    active_mask[np.argmax(vals)] = True
                subgrad = np.sum(slopes[active_mask])
                if subgrad > 1e-14:
                    hi = mid
                elif subgrad < -1e-14:
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)
        
        h = np.clip(h, 0.0, 1.0)
        # Project to feasible
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active

def slsqp_smooth(h0, n, target, alphas, maxiter, ftol):
```

```python
            h_start, h_end = -k, n
            g_start, g_end = 0, n + k
        ranges.append((h_start, h_end, g_start, g_end))
    return ranges


def build_A(g, n, dx, ranges):
    row_parts = []
    col_parts = []
    val_parts = []
    for k_idx, (h_s, h_e, g_s, g_e) in enumerate(ranges):
        length = h_e - h_s
        if length <= 0:
            continue
        g_seg = g[g_s:g_e]
        nz = np.nonzero(g_seg > 1e-14)[0]
        if len(nz) > 0:
            row_parts.append(np.full(len(nz), k_idx, dtype=np.int32))
            col_parts.append((nz + h_s).astype(np.int32))
            val_parts.append(g_seg[nz] * dx)
    if not row_parts:
        return csr_matrix((2 * n - 1, n))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_full_lp(h_old, n, target, reg, eps, max_iters, ranges):
    dx = 2.0 / n
    g = 1.0 - h_old
    A = build_A(g, n, dx, ranges)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        A @ h_var <= t_var
    ]
    objective = reg * cp.sum_squares(h_var - h_old) + t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=max_iters)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    status = prob.status
    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    h_new = np.asarray(h_var.value).ravel()
    t_val = float(t_var.value) if t_var.value is not None else 0.0
    return h_new, t_val


def full_lp_phase(h_init, n, target, time_limit, n_iters, ranges):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        frac = min(1.0, (it + 1) / max(1, n_iters))
        reg = 1.0 * (0.005) ** frac
        damping = 0.6 + 0.4 * frac
        h_sol, t_val = solve_full_lp(h, n, target, reg, 1e-10, 500000, ranges)
        if h_sol is None:
            break
        h_new = (1.0 - damping) * h + damping * h_sol
        h_new = project_feasible(h_new, target)
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 80:
            break
    return best_h, best_c5


def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    start = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = int(active_k[idx])
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            lo = 0.0
            hi = 1.0
            for _ in range(45):
                mid = 0.5 * (lo + hi)
                vals = slopes * mid + intercepts
                mv = np.max(vals)
                tol = 1e-11 * max(1.0, abs(mv))
                mask = np.abs(vals - mv) < tol
                if not np.any(mask):
                    mask = np.zeros(n_active, dtype=bool)
                    mask[np.argmax(vals)] = True
                sg = np.sum(slopes[mask])
                if sg > 1e-12:
                    hi = mid
                elif sg < -1e-12:
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    return best_h, best_c5


def _slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _slsqp_con_fun(x):
    return np.sum(x) - _slsqp_target


def _slsqp_con_jac(x):
    return np.ones(len(x))


_slsqp_target = 0.0


def slsqp_round(h0, n, target, alphas, maxiter, ftol):
    global _slsqp_target
    _slsqp_target = target
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score


def subgradient_step(h, n, target, max_steps, time_limit):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    start = time.time()
    stall = 0
    idxs = np.arange(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if stall > 120:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        valid1 = (idxs - k >= 0) & (idxs - k < n)
        valid2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[valid1] += (1.0 - h_cur[idxs[valid1] - k])
        grad[valid2] -= h_cur[idxs[valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            h_try = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    ranges_n = precompute_ranges(n)

    # Phase 1: Full-constraint LP
    if tleft() > 250:
        lp_time = min(350, tleft() * 0.4)
        h_lp, c5_lp = full_lp_phase(best_h, n, target, lp_time, 800, ranges_n)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent
    if tleft() > 80:
        cd_time = min(100, tleft() * 0.12)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=100,
                                     max_passes=10000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Second full LP round
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = full_lp_phase(best_h, n, target, lp_time, 600, ranges_n)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP
    if tleft() > 80:
        h_ref, c5_ref = slsqp_round(best_h, n, target, [300, 1500, 8000], 2000, 1e-14)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref

    # Phase 5: Try n=500
    if tleft() > 280:
        n_try = 500
        target_try = n_try / 2.0
        ranges_try = precompute_ranges(n_try)
        h_interp = interpolate_to(best_h, n_try, target_try)
        if tleft() > 200:
            lp_time = min(150, tleft() * 0.35)
            h_lp500, c5_lp500 = full_lp_phase(h_interp, n_try, target_try, lp_time, 500, ranges_try)
            if c5_lp500 < best_score:
                best_score = c5_lp500
                best_h = h_lp500
                n = n_try
                target = target_try
                ranges_n = ranges_try
            else:
                sc_int = compute_c5(h_interp)
                if sc_int < best_score:
                    best_score = sc_int
                    best_h = h_interp
                    n = n_try
                    target = target_try
                    ranges_n = ranges_try
        if tleft() > 60:
            h_cd500, c5_cd500 = coord_descent(best_h, target, n_active=100,
                                               max_passes=4000, time_limit=min(50, tleft() * 0.7), rng=rng)
            if c5_cd500 < best_score:
                best_score = c5_cd500
                best_h = h_cd500

    # Phase 6: Subgradient
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_step(best_h, n_use, target_use, 4000, min(35, tleft() * 0.6))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Final LP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        r_use = precompute_ranges(n_use)
        h_lp, c5_lp = full_lp_phase(best_h, n_use, target_use, min(50, tleft() * 0.7), 400, r_use)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 8: Final coordinate descent
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=120,
                                     max_passes=5000, time_limit=min(25, tleft() * 0.6), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final subgradient micro-refinement
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_step(best_h, n_use, target_use, 2000, min(12, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: Very short SLSQP
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, c5_ref = slsqp_round(best_h, n_use, target_use, [20000, 100000], 500, 1e-15)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
