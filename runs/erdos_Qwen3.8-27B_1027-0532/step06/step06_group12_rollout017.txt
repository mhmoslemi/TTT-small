        # Check improvement
        new_c5 = compute_c5(h)
        if new_c5 < best - 1e-14:
            best = new_c5
    
    return h, best

The key function is `find_optimal_delta`. For a convex piecewise linear function g(delta) = max_k (a_k + c_k * delta):

Method 1: Evaluate at breakpoints
- The breakpoints are where two lines cross: (a_l - a_k) / (c_k - c_l) for all pairs with c_k ≠ c_l
- There are O(K^2) breakpoints, but we only need the ones in [lo, hi]
- For K=799, that's ~320K breakpoints - too many

Method 2: Binary search on slope
- g'(delta) = max{c_k : a_k + c_k * delta = g(delta)} (the slope of the active line)
- g' is non-decreasing (convexity)
- Binary search for where g' crosses 0
- Each evaluation of g' requires O(K)
- Total: O(K log((hi-lo)/eps)) ≈ O(K * 40) for good precision

Method 3: Use the fact that the minimum is at a "median" type point
- Sort lines by slope c_k
- The minimum of max is where the "upper envelope" changes from negative-slope to positive-slope lines
- This is related to the concept of the "smallest enclosing" point

Actually, the cleanest approach for a 1D convex PL function:
- The minimum is at a point where the subgradient [g'_-(x), g'_+(x)] contains 0
- g'_-(x) = min{c_k : a_k + c_k*x = g(x)} (smallest slope of active lines)
- g'_+(x) = max{c_k : a_k + c_k*x = g(x)} (largest slope of active lines)
- We need g'_-(x) ≤ 0 ≤ g'_+(x)

Binary search approach:
- At a given x, compute g(x) and the set of active lines (those achieving g(x))
- If the min slope of active lines > 0: minimum is to the left
- If the max slope of active lines < 0: minimum is to the right
- Otherwise: x is optimal (or very close)

This converges in O(log((hi-lo)/eps)) iterations, each O(K).

For the coefficient computation in coordinate descent:
When we change h[i] by +delta and h[j] by -delta:

For lag shift s (where s = k - (n-1)):
f_s = sum_{valid m} h[m] * (1-h[m+s]) * dx

Change in f_s:
- If i is a valid "m" and i+s is valid: +delta * (1-h[i+s]) * dx
- If i-s is a valid "m" and i is valid: -delta * h[i-s] * dx  
  (because h[i-s] * (1-h[i]) changes by h[i-s] * (-delta))
- If j is a valid "m" and j+s is valid: -delta * (1-h[j+s]) * dx
- If j-s is a valid "m" and j is valid: +delta * h[j-s] * dx

So c_s = dx * [(1-h[i+s])*1_{valid} - h[i-s]*1_{valid} - (1-h[j+s])*1_{valid} + h[j-s]*1_{valid}]

where 1_{valid} means both indices are in [0, n-1].

- i+s valid: 0 ≤ i+s < n
- i-s valid: 0 ≤ i-s < n
- j+s valid: 0 ≤ j+s < n  
- j-s valid: 0 ≤ j-s < n

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_pair_coeffs(i, j, h, n, dx):
    """Compute c_k for all lags k when h[i] += delta, h[j] -= delta."""
    # c_k corresponds to shift s = k - (n-1)
    # c_s = dx * [(1-h[i+s])*1_{0≤i+s<n} - h[i-s]*1_{0≤i-s<n} 
    #             - (1-h[j+s])*1_{0≤j+s<n} + h[j-s]*1_{0≤j-s<n}]
    s = np.arange(-(n-1), n)  # all shifts
    c = np.zeros(2*n-1)
    
    # Term 1: (1-h[i+s]) where 0≤i+s<n
    valid1 = (i + s >= 0) & (i + s < n)
    c[valid1] += (1.0 - h[i + s[valid1]]) * dx
    
    # Term 2: -h[i-s] where 0≤i-s<n
    valid2 = (i - s >= 0) & (i - s < n)
    c[valid2] -= h[i - s[valid2]] * dx
    
    # Term 3: -(1-h[j+s]) where 0≤j+s<n
    valid3 = (j + s >= 0) & (j + s < n)
    c[valid3] -= (1.0 - h[j + s[valid3]]) * dx
    
    # Term 4: +h[j-s] where 0≤j-s<n
    valid4 = (j - s >= 0) & (j - s < n)
    c[valid4] += h[j - s[valid4]] * dx
    
    return c

def find_optimal_delta(overlaps, c, lo, hi, n_iter=60):
    """Find delta in [lo, hi] minimizing max_k (overlaps[k] + c[k]*delta).
    
    The function is convex piecewise linear. Use binary search on the subgradient.
    """
    if lo >= hi - 1e-14:
        return 0.0
    
    # Binary search: find where subgradient contains 0
    a, b = lo, hi
    for _ in range(n_iter):
        mid = (a + b) / 2.0
        vals = overlaps + c * mid
        max_val = np.max(vals)
        # Active set: where vals is within tolerance of max
        active = vals >= max_val - 1e-12
        if active.sum() == 0:
            return mid
        slopes = c[active]
        min_slope = np.min(slopes)
        max_slope = np.max(slopes)
        if min_slope > 1e-10:
            # Function is increasing at mid, minimum is to the left
            b = mid
        elif max_slope < -1e-10:
            # Function is decreasing at mid, minimum is to the right
            a = mid
        else:
            # 0 is in the subgradient, mid is optimal
            return mid
    return (a + b) / 2.0

def coordinate_descent(h_init, n, target, time_limit, max_sweeps=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)

    best_h = h.copy()
    n_corr = 2 * n - 1
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        
        # Try n random pairs per sweep
        for _ in range(n):
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi - 1e-12:
                continue
            
            c = compute_pair_coeffs(i, j, h, n, dx)
            delta = find_optimal_delta(overlaps, c, lo, hi)
            
            h[i] += delta
            h[j] -= delta
            
            # Update overlaps incrementally (approximate: just add c*delta)
            overlaps += c * delta
        
        # Check actual C5 periodically
        if sweep % 5 == 0:
            actual_c5 = compute_c5(h)
            if actual_c5 < best - 1e-14:
                best = actual_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Re-sync overlaps (to avoid drift)
            overlaps = compute_overlaps(h)
        
        if no_improve > 200:
            break
    
    return best_h, best

- If all active c_k > 0: g is increasing, move left (hi = delta)
- If all active c_k < 0: g is decreasing, move right (lo = delta)
- Otherwise: delta is optimal (or very close)

- The minimum of max of lines is at a breakpoint where two lines cross
- We can find the "upper envelope" and locate its minimum
- More complex but exact in O(K log K)

            constraints.append(coeff @ h_var <= t_var)
        elif k > 0:
            coeff = np.zeros(n)
            coeff[:n - k] = (1.0 - h_fixed[k:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        else:
            m = -k
            coeff = np.zeros(n)
            coeff[m:] = (1.0 - h_fixed[:n - m]) * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=250,
                   damping_start=0.9, damping_end=0.8,
                   reg_start=1e-3, reg_end=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 50 and stuck > 150:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        h_var, t_var, constraints = build_lp_linear(1.0 - h, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def golden_init(n, target, rng):
    golden = 0.618033988749895
    h = np.zeros(n)
    phases = rng.uniform(0, 1, size=3)
    for j in range(3):
        for i in range(n):
            val = (i * golden + phases[j]) % 1.0
            if val < 0.5:
                h[i] += 1.0 / 3.0
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, target)

def block_init(n, target, rng, n_blocks=8):
    h = np.zeros(n)
    block_size = n // n_blocks
    fill = n_blocks // 2
    offsets = rng.choice(n_blocks, size=fill, replace=False)
    for off in offsets:
        start = off * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return project_feasible(h, target)

def frank_wolfe(h_init, n, target, time_limit, max_iter=300000, K=40):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    n_half = n - 1
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n - k] += (1.0 - h[k:]) * dx
                g[k:] += (-h[:n - k]) * dx
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m]) * dx
                g[:n - m] += (-h[m:]) * dx
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 10000 and t - best_t > 80000:
            break
    return best_h, best_c5

def time_remaining(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick FW to ensure we're in a good region
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        fw_time = min(120, tl * 0.12)
        h_fw, c5_fw = frank_wolfe(best_h, n, target, fw_time, max_iter=200000, K=50)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 2: Coordinate descent (main workhorse)
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        cd_time = min(300, tl * 0.3)
        h_cd, c5_cd = coordinate_descent(best_h, n, target, cd_time, rng, pairs_per_sweep=n)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B smooth optimization
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        h_lgf = lgfssq_optimize(best_h, n, target,
                                 [30, 100, 400, 1500, 5000, 20000, 80000],
                                 maxiter=3000, penalty=200.0)
        sc = compute_c5(h_lgf)
        if sc < best_score:
            best_score = sc
            best_h = h_lgf

    # Phase 4: More coordinate descent from improved point
    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        cd_time = min(250, tl * 0.28)
        h_cd2, c5_cd2 = coordinate_descent_ordered(best_h, n, target, cd_time, rng)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: SLSQP with escalating alpha
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        h_slq = slsqp_optimize(best_h, n, target,
                                [50, 200, 800, 3000, 10000, 40000, 150000],
                                maxiter=2500, ftol=1e-15)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 6: Alternating LP
    tl = time_remaining(start_time, budget_s)
    if tl > 180:
        lp_time = min(200, tl * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=4000, time_limit=lp_time,
                                      n_top=300, damping_start=0.9,
                                      damping_end=0.8, reg_start=1e-3,
                                      reg_end=1e-10)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Coordinate descent after LP
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        cd_time = min(120, tl * 0.35)
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, target, cd_time, rng, pairs_per_sweep=int(n * 1.5))
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: Multi-start with diverse inits + CD
    for attempt in range(3):
        tl = time_remaining(start_time, budget_s)
        if tl < 80:
            break
        if attempt == 0:
            h_s = golden_init(n, target, rng)
        elif attempt == 1:
            h_s = block_init(n, target, rng, n_blocks=6)
        else:
            h_s = block_init(n, target, rng, n_blocks=12)
        cd_time = min(50, tl * 0.2)
        h_s2, c5_s2 = coordinate_descent(h_s, n, target, cd_time, rng, pairs_per_sweep=n)
        if c5_s2 < best_score:
            best_score = c5_s2
            best_h = h_s2

    # Phase 9: Final L-BFGS-B polish
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_final_lgf = lgfssq_optimize(best_h, n, target,
                                       [200, 1000, 5000, 20000, 100000, 500000],
                                       maxiter=4000, penalty=500.0)
        sc = compute_c5(h_final_lgf)
        if sc < best_score:
            best_score = sc
            best_h = h_final_lgf

    # Phase 10: Final SLSQP
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        h_final_slq = slsqp_optimize(best_h, n, target,
                                      [500, 2000, 10000, 50000, 200000, 1000000],
                                      maxiter=3000, ftol=1e-16)
        sc = compute_c5(h_final_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_final_slq

    # Phase 11: Final coordinate descent
    tl = time_remaining(start_time, budget_s)
    if tl > 40:
        h_final_cd, c5_final_cd = coordinate_descent(best_h, n, target, min(35, tl * 0.8), rng,
                                                      pairs_per_sweep=int(n * 2))
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
