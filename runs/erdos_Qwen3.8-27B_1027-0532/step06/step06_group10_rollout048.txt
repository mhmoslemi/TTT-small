import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def compute_influence(h, n):
    inf = np.zeros((n, 2 * n - 1))
    idxs = np.arange(n)
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        m_plus = idxs + shift
        m_minus = idxs - shift
        valid_plus = (m_plus >= 0) & (m_plus < n)
        valid_minus = (m_minus >= 0) & (m_minus < n)
        inf[valid_plus, k] += (1.0 - h[m_plus[valid_plus]])
        inf[valid_minus, k] -= h[m_minus[valid_minus]]
    return inf

def update_influence_rows(inf, h_old, h_new, changed_idx, n):
    idxs = np.arange(n)
    for m in changed_idx:
        for k in range(2 * n - 1):
            shift = k - (n - 1)
            val = 0.0
            mp = m + shift
            mm = m - shift
            if 0 <= mp < n:
                val += (1.0 - h_new[mp])
            if 0 <= mm < n:
                val -= h_new[mm]
            inf[m, k] = val

def solve_1d_minmax(c, g, d_lo, d_hi):
    if d_hi < d_lo:
        return 0.0
    for _ in range(30):
        if d_hi - d_lo < 1e-15:
            break
        d_mid = 0.5 * (d_lo + d_hi)
        vals = c + d_mid * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            g_active = g[active_mask]
            g_min = np.min(g_active)
            g_max = np.max(g_active)
            if g_min <= 0.0 and g_max >= 0.0:
                break
            elif g_min > 0.0:
                d_lo = d_mid
            else:
                d_hi = d_mid
        else:
            break
    return 0.5 * (d_lo + d_hi)

def fast_cd(h0, n, target, time_limit, max_sweeps=10000, n_cand=20):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    n_corr = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        inf = compute_influence(h, n)
        k_star = int(np.argmax(corr)) - (n - 1)
        grad_kstar = inf[:, int(np.argmax(corr))]
        grad_kstar -= np.mean(grad_kstar)
        sorted_idx = np.argsort(grad_kstar)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                d_lo = -min(h[j], 1.0 - h[i])
                if d_hi < 1e-15 and d_lo > -1e-15:
                    continue
                g = inf[i, :] - inf[j, :]
                if d_hi > 1e-15:
                    d_star = solve_1d_minmax(corr, g, 1e-15, d_hi)
                    approx = np.max(corr + d_star * g) * dx
                    if approx < best_approx:
                        best_approx = approx
                        best_delta = d_star
                        best_i = i
                        best_j = j
                if d_lo < -1e-15:
                    d_star_neg = solve_1d_minmax(corr, g, d_lo, -1e-15)
                    approx = np.max(corr + d_star_neg * g) * dx
                    if approx < best_approx:
                        best_approx = approx
                        best_delta = d_star_neg
                        best_i = i
                        best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    idxs = np.arange(n)
    mid = n - 1
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
            return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n))
        def _grad(x):
            return smooth_max_grad(x, alpha, n)
        def _con(x):
            return np.sum(x) - target
        def _con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lp_refine(h0, n, target, time_limit, n_active=15, max_rounds=30):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val * dx <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

            best_h = h_lp

    # Phase 8: Go to 3x or 4x if time allows
    if tleft() > 250:
        n_hires2 = min(4 * len(h_init), int(n * 1.5))
        if n_hires2 > n:
            target_hires2 = float(n_hires2) / 2.0
            h_interp2 = interpolate_to(best_h, n_hires2, target_hires2)
            t8 = min(180.0, tleft() * 0.4)
            h_hires2, s_hires2 = nesterov_optimize(h_interp2, n_hires2, target_hires2, time_limit=t8, max_steps=40000)
            if s_hires2 < best_score:
                best_score = s_hires2
                best_h = h_hires2
                n = n_hires2
                target = target_hires2

    # Phase 9: Final CD
    if tleft() > 80:
        t9 = min(70.0, tleft() * 0.5)
        h_cd, c5_cd = fast_cd(best_h, n, target, time_limit=t9, max_sweeps=8000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final Nesterov polish
    if tleft() > 60:
        t10 = min(50.0, tleft() * 0.5)
        h_p, s_p = nesterov_optimize(best_h, n, target, time_limit=t10, max_steps=30000)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 11: Very final CD
    if tleft() > 20:
        t11 = min(15.0, tleft() * 0.6)
        h_cd, c5_cd = fast_cd(best_h, n, target, time_limit=t11, max_sweeps=5000, n_cand=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Try multi-start with small perturbations
    if tleft() > 60:
        for trial in range(3):
            if tleft() < 25:
                break
            h_pert = project_feasible(best_h + 0.01 * rng.randn(n), target)
            t_trial = min(20.0, tleft() * 0.4)
            h_tr, s_tr = nesterov_optimize(h_pert, n, target, time_limit=t_trial, max_steps=10000)
            if s_tr < best_score:
                best_score = s_tr
                best_h = h_tr
                t_trial2 = min(15.0, tleft() * 0.3)
                h_cd, c5_cd = fast_cd(best_h, n, target, time_limit=t_trial2, max_sweeps=3000, n_cand=15)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
