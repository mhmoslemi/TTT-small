import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha):
    """SLSQP with log-sum-exp smoothing of the max."""
    global_n = n
    global_target = target
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        alpha = obj.alpha
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    obj.alpha = alphas[0]
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    for alpha in alphas:
        obj.alpha = alpha
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    return h_new

                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        elif no_improve > 25:
            break
    return best_h, best_c5

def _slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    g_n = n
    g_t = target

    def _obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def _cons_fun(h):
        return np.sum(h) - g_t

    def _cons_jac(h):
        return np.ones(g_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: _obj(x, alpha), h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def _fourier_perturb(h, n, target, n_modes, amplitude, rng):
    h_pert = h.copy()
    N = len(h)
    H = np.fft.rfft(h)
    n_fft = len(H)
    for m in range(1, min(n_modes + 1, n_fft - 1)):
        if rng.rand() < 0.7:
            dA = rng.randn() * amplitude
            dP = rng.randn() * amplitude * 0.5
            H[m] += dA + 1j * dP
    h_new = np.fft.irfft(H, n=N)
    return project_feasible(h_new, target)

def _triple_cd(h, n, target, time_limit, max_sweeps=2000, n_cand=30):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 60:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        for a in high_set:
            ha = h[a]
            if ha < 1e-10:
                continue
            va_p = (a + shifts >= 0) & (a + shifts < n)
            va_m = (a - shifts >= 0) & (a - shifts < n)
            ga = np.zeros(n_corr)
            ga[va_p] -= (1.0 - h[a + shifts[va_p]])
            ga[va_m] += h[a - shifts[va_m]]

            for b in high_set:
                if b == a:
                    continue
                hb = h[b]
                if hb < 1e-10:
                    continue
                vb_p = (b + shifts >= 0) & (b + shifts < n)
                vb_m = (b - shifts >= 0) & (b - shifts < n)
                gb = np.zeros(n_corr)
                gb[vb_p] -= (1.0 - h[b + shifts[vb_p]])
                gb[vb_m] += h[b - shifts[vb_m]]

                for c in low_set:
                    if c == a or c == b:
                        continue
                    vc_p = (c + shifts >= 0) & (c + shifts < n)
                    vc_m = (c - shifts >= 0) & (c - shifts < n)
                    gc = np.zeros(n_corr)
                    gc[vc_p] += (1.0 - h[c + shifts[vc_p]])
                    gc[vc_m] -= h[c - shifts[vc_m]]

                    g_total = ga + gb + gc
                    delta_max = min(ha, hb, 2.0 * (1.0 - h[c]))
                    if delta_max < 1e-12:
                        continue

                    a_d = 0.0
                    b_d = delta_max
                    for _ in range(12):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g_total
                        f_mid = np.max(vals)
                        active = vals >= f_mid - 1e-14
                        f_prime = np.max(g_total[active]) if active.any() else 0.0
                        if f_prime > 1e-14:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g_total) * dx
                    if approx_max < best_trial - 1e-15:
                        best_trial = approx_max
                        best_move = (a, b, c, delta_star)

        if best_move is not None:
            a, b, c, d = best_move
            h[a] -= d
            h[b] -= d
            h[c] += 2.0 * d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        elif no_improve > 20:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_cd, c5_cd = _pair_cd(best_h, n, target,
                                time_limit=min(80, tleft() * 0.3),
                                max_sweeps=3000, n_cand=50, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 180:
        h_sg, c5_sg = _subgrad_core(best_h, n, target,
                                     time_limit=min(150, tleft() * 0.35),
                                     max_iter=100000, beta=0.85,
                                     base_step=0.4, n_worst=3, use_momentum=True)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
        h_cd, c5_cd = _pair_cd(best_h, n, target,
                                time_limit=min(60, tleft() * 0.2),
                                max_sweeps=3000, n_cand=55, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol = _slsqp_round(best_h, n, target,
                              [3000, 12000, 50000], 600)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_sg2, c5_sg2 = _subgrad_core(best_h, n, target,
                                       time_limit=min(130, tleft() * 0.35),
                                       max_iter=120000, beta=0.8,
                                       base_step=0.25, n_worst=5, use_momentum=True)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2
        h_cd, c5_cd = _pair_cd(best_h, n, target,
                                time_limit=min(50, tleft() * 0.2),
                                max_sweeps=2500, n_cand=60, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for attempt in range(6):
            if tleft() < 30:
                break
            amp = rng.choice([0.04, 0.08, 0.12, 0.18])
            n_modes = int(rng.choice([5, 10, 15, 25]))
            h_pert = _fourier_perturb(best_h, n, target, n_modes, amp, rng)
            cd_t = min(25, tleft() * 0.25)
            h_try, c5_try = _pair_cd(h_pert, n, target,
                                      time_limit=cd_t,
                                      max_sweeps=1500, n_cand=40, n_bisect=12)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try
            sg_t = min(20, tleft() * 0.2)
            h_sg3, c5_sg3 = _subgrad_core(h_try, n, target,
                                           time_limit=sg_t,
                                           max_iter=50000, beta=0.85,
                                           base_step=0.3, n_worst=3,
                                           use_momentum=True)
            if c5_sg3 < best_score:
                best_score = c5_sg3
                best_h = h_sg3

    if tleft() > 80:
        h_triple, c5_triple = _triple_cd(best_h, n, target,
                                          time_limit=min(60, tleft() * 0.3),
                                          max_sweeps=1500, n_cand=25)
        if c5_triple < best_score:
            best_score = c5_triple
            best_h = h_triple
            h_cd, c5_cd = _pair_cd(best_h, n, target,
                                    time_limit=min(30, tleft() * 0.2),
                                    max_sweeps=2000, n_cand=55, n_bisect=14)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    if tleft() > 60:
        h_sg4, c5_sg4 = _subgrad_core(best_h, n, target,
                                       time_limit=min(50, tleft() * 0.3),
                                       max_iter=80000, beta=0.9,
                                       base_step=0.15, n_worst=4, use_momentum=True)
        if c5_sg4 < best_score:
            best_score = c5_sg4
            best_h = h_sg4

    if tleft() > 40:
        h_pol2 = _slsqp_round(best_h, n, target,
                               [15000, 60000, 150000], 500)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    if tleft() > 25:
        h_cd_final, c5_final = _pair_cd(best_h, n, target,
                                         time_limit=min(20, tleft() * 0.7),
                                         max_sweeps=2500, n_cand=65, n_bisect=15)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
