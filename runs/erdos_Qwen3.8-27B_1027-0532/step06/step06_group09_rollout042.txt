    mid = n - 1
    grad = np.zeros(n)
    for wi, s_idx in zip(lam, np.where(lam > 1e-10)[0]):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    return grad

def game_descent(h_init, n, target, time_limit, max_steps=10000, top_k=12, eta=2.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        # Get top-k shifts
        top_k_idx = np.argsort(corr)[-top_k:]
        lam_top = lam[top_k_idx].copy()
        lam_top = lam_top / (lam_top.sum() + 1e-15)

        # Compute subgradient
        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_k_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        # Line search
        improved = False
        lr = best_lr * max(0.2, 1.0 - step / (max_steps * 1.2))
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.15, 0.5)
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.85
            if best_lr < 0.0005:
                break

        # Update lam (multiplicative weights)
        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -40, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / (lam_new.sum() + 1e-15)
        lam = 0.6 * lam + 0.4 * lam_new

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand=40, patience=250):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        # Gradient from top-5 shifts
        k_star_idx = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for s_idx in k_star_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_c5_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_g = None

        # Precompute for high_set
        for i in high_set:
            ip_mask = (i + shifts >= 0) & (i + shifts < n)
            im_mask = (i - shifts >= 0) & (i - shifts < n)
            h_i_sp = np.zeros(n_corr)
            h_i_sp[ip_mask] = h[i + shifts[ip_mask]]
            h_i_sm = np.zeros(n_corr)
            h_i_sm[im_mask] = h[i - shifts[im_mask]]

            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                jp_mask = (j + shifts >= 0) & (j + shifts < n)
                jm_mask = (j - shifts >= 0) & (j - shifts < n)
                jpm = np.zeros(n_corr)
                jpm[jp_mask] = h[j + shifts[jp_mask]]
                jsm = np.zeros(n_corr)
                jsm[jm_mask] = h[j - shifts[jm_mask]]

                g = h_i_sp + h_i_sm - jpm - jsm

                # Bisection for optimal delta
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, delta_lo + 1e-10)

                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_c5_trial - 1e-15:
                    best_c5_trial = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
                    best_g = g

        # Also try negative delta (reverse direction)
        # Actually, the bisection already handles both positive and negative delta
        # since delta_lo can be negative

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def structured_perturbation(h, n, seed, scale=0.05):
    rng = np.random.RandomState(seed)
    h_pert = h.copy()
    pert_type = rng.randint(4)
    if pert_type == 0:
        # Sinusoidal
        freq = rng.randint(1, n // 10 + 1)
        h_pert += scale * np.sin(2 * np.pi * freq * np.arange(n) / n)
    elif pert_type == 1:
        # Bump
        center = rng.randint(n // 4, 3 * n // 4)
        width = rng.randint(n // 20, n // 5)
        h_pert += scale * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif pert_type == 2:
        # Random patch
        patch_size = rng.randint(n // 20, n // 5)
        start_idx = rng.randint(0, n - patch_size)
        h_pert[start_idx:start_idx + patch_size] += scale * rng.randn(patch_size)
    else:
        # Full random
        h_pert += scale * rng.randn(n)
    return project_feasible(h_pert, n / 2.0)

    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def sum_constraint(h):
    return np.sum(h) - _global_target


def sum_constraint_jac(h):
    return np.ones(_global_n)


def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_objective, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def slsqp_polish(h_start, n, target, alpha, maxiter=800):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(smooth_objective, h_start, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def game_descent(h_init, n, target, time_limit, max_steps=10000, top_k=16, eta=2.5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        top_k_idx = np.argsort(corr)[-top_k:]
        lam_top = lam[top_k_idx].copy()
        s = lam_top.sum()
        if s > 1e-15:
            lam_top = lam_top / s

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_k_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = best_lr * max(0.2, 1.0 - step / (max_steps * 1.2))
        for attempt in range(12):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.15, 0.5)
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.85
            if best_lr < 0.0005:
                break

        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -40, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / (lam_new.sum() + 1e-15)
        lam = 0.6 * lam + 0.4 * lam_new

    return best_h, best_c5


def pair_cd_vectorized(h_init, n, target, time_limit, max_sweeps=5000,
                       n_cand=40, patience=300, top_grad_k=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star_idx = np.argsort(corr)[-top_grad_k:]
        grad = np.zeros(n)
        for s_idx in k_star_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        nl = len(low_set)
        jpm_mat = np.zeros((nl, n_corr))
        jsm_mat = np.zeros((nl, n_corr))
        for jj in range(nl):
            j = int(low_set[jj])
            jp_mask = (j + shifts >= 0) & (j + shifts < n)
            jm_mask = (j - shifts >= 0) & (j - shifts < n)
            jpm_mat[jj, jp_mask] = h[j + shifts[jp_mask]]
            jsm_mat[jj, jm_mask] = h[j - shifts[jm_mask]]

        best_c5_trial = c5_val
        best_i = -1
        best_j_idx = -1
        best_delta = 0.0
        best_g = None

        for i in high_set:
            i = int(i)
            ip_mask = (i + shifts >= 0) & (i + shifts < n)
            im_mask = (i - shifts >= 0) & (i - shifts < n)
            h_i_vec = np.zeros(n_corr)
            h_i_vec[ip_mask] = h[i + shifts[ip_mask]]
            h_i_vec[im_mask] += h[i - shifts[im_mask]]

            g_mat = h_i_vec[np.newaxis, :] - jpm_mat - jsm_mat

            delta_lo_arr = np.maximum(h[i] - 1.0, -h[low_set].astype(float))
            delta_hi_arr = np.minimum(h[i], 1.0 - h[low_set].astype(float))

            valid = delta_hi_arr - delta_lo_arr > 1e-12
            if not np.any(valid):
                continue

            a = delta_lo_arr[valid].copy()
            b = delta_hi_arr[valid].copy()
            bv = g_mat[valid, :]
            corr_v = corr[np.newaxis, :]

            for _ in range(22):
                m_d = 0.5 * (a + b)
                vals = corr_v + m_d[:, np.newaxis] * bv
                f_mid = np.max(vals, axis=1)
                active = vals >= (f_mid[:, np.newaxis] - 1e-14)
                f_prime = np.max(np.where(active, bv, -np.inf), axis=1)
                go_left = f_prime > 1e-14
                b[go_left] = m_d[go_left]
                a[~go_left] = m_d[~go_left]

            delta_star = 0.5 * (a + b)
            vals_final = corr_v + delta_star[:, np.newaxis] * bv
            approx_max = np.max(vals_final, axis=1) * dx

            best_local = np.argmin(approx_max)
            if approx_max[best_local] < best_c5_trial - 1e-15:
                valid_indices = np.where(valid)[0]
                best_j_idx = int(valid_indices[best_local])
                best_i = i
                best_delta = float(delta_star[best_local])
                best_g = (np.where(active[:, best_local], bv[best_local, :], -np.inf))
                best_c5_trial = approx_max[best_local]

        if best_i >= 0:
            j = int(low_set[best_j_idx])
            h[best_i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def structured_perturbation(h, n, seed, scale=0.05):
    rng = np.random.RandomState(seed)
    h_pert = h.copy()
    pert_type = rng.randint(5)
    if pert_type == 0:
        freq = rng.randint(1, n // 8 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h_pert += scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif pert_type == 1:
        center = rng.randint(n // 5, 4 * n // 5)
        width = rng.randint(n // 25, n // 4)
        h_pert += scale * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif pert_type == 2:
        patch_size = rng.randint(n // 15, n // 4)
        start_idx = rng.randint(0, max(1, n - patch_size))
        h_pert[start_idx:start_idx + patch_size] += scale * rng.randn(patch_size)
    elif pert_type == 3:
        half = n // 2
        h_pert[:half] += scale * rng.randn(half)
        h_pert[half:] -= scale * rng.randn(n - half)
    else:
        h_pert += scale * rng.randn(n)
    return project_feasible(h_pert, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B polish
    if tleft() > 80:
        for alpha in [500, 2000, 5000]:
            if tleft() < 20:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1200)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 2: Game-theoretic descent (wide active set)
    if tleft() > 150:
        gt_time = min(180, tleft() * 0.2)
        h_gt, c5_gt = game_descent(best_h, n, target,
                                   time_limit=gt_time,
                                   max_steps=12000, top_k=20, eta=2.5)
        if c5_gt < best_score:
            best_score = c5_gt
            best_h = h_gt

    # Phase 3: Vectorized pair CD (main refinement)
    if tleft() > 200:
        cd_time = min(350, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_vectorized(best_h, n, target,
                                         time_limit=cd_time,
                                         max_sweeps=10000, n_cand=40,
                                         patience=350, top_grad_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: L-BFGS-B + SLSQP polish
    if tleft() > 100:
        for alpha in [10000, 50000]:
            if tleft() < 25:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
        if tleft() > 50:
            h_pol = slsqp_polish(best_h, n, target, 50000, maxiter=600)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 5: Second round of pair CD with wider candidate set
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_vectorized(best_h, n, target,
                                         time_limit=cd_time,
                                         max_sweeps=8000, n_cand=50,
                                         patience=250, top_grad_k=7)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Game descent with wider set
    if tleft() > 120:
        h_gt, c5_gt = game_descent(best_h, n, target,
                                   time_limit=min(100, tleft() * 0.2),
                                   max_steps=10000, top_k=24, eta=3.0)
        if c5_gt < best_score:
            best_score = c5_gt
            best_h = h_gt

    # Phase 7: Structured perturbations + re-optimization
    rng = np.random.RandomState(seed)
    if tleft() > 180:
        for attempt in range(10):
            if tleft() < 70:
                break
            scale = 0.02 + 0.015 * attempt
            h_pert = structured_perturbation(best_h, n, seed + attempt * 137,
                                             scale=scale)
            h_ref = lbfags_polish(h_pert, n, target, 3000, maxiter=400)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            h_cd, c5_cd = pair_cd_vectorized(best_h, n, target,
                                             time_limit=min(25, tleft() * 0.12),
                                             max_sweeps=3000, n_cand=35,
                                             patience=120, top_grad_k=5)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Final pair CD
    if tleft() > 80:
        h_cd, c5_cd = pair_cd_vectorized(best_h, n, target,
                                         time_limit=min(60, tleft() * 0.25),
                                         max_sweeps=6000, n_cand=45,
                                         patience=180, top_grad_k=6)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final SLSQP polish
    if tleft() > 35:
        h_final.try = None
        h_final_try = slsqp_polish(best_h, n, target, 100000, maxiter=400)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, len(h_out)
