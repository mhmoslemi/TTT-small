import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def lse_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    n_corr = len(corr)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    thresh = 1e-12
    significant = np.where(weights > thresh)[0]
    for k_idx in significant:
        w = weights[k_idx]
        k = int(k_idx) - mid
        v_plus = (idxs + k >= 0) & (idxs + k < n)
        v_minus = (idxs - k >= 0) & (idxs - k < n)
        grad[v_plus] += w * (1.0 - h[idxs[v_plus] + k])
        grad[v_minus] -= w * h[idxs[v_minus] - k]
    return grad


def aug_lag_obj(h_vec, alpha, lam, mu, target):
    val = lse_objective(h_vec, alpha)
    s = np.sum(h_vec) - target
    return val + lam * s + 0.5 * mu * s * s


def aug_lag_grad(h_vec, alpha, lam, mu, target):
    g = lse_gradient(h_vec, alpha)
    s = np.sum(h_vec) - target
    g = g + lam + mu * s
    return g


def aug_lag_lbfgsb(h_init, n, target, alpha, time_limit, mu_init=50.0, max_outer=25, lbfgs_iter=400):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    lam = 0.0
    mu = mu_init
    for outer in range(max_outer):
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(aug_lag_obj, h, args=(alpha, lam, mu, target),
                           jac=lambda x: aug_lag_grad(x, alpha, lam, mu, target),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': lbfgs_iter, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
        except Exception:
            break
        s_viol = abs(np.sum(h_new) - target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lam = lam + mu * (np.sum(h) - target)
        if s_viol < 1e-10:
            break
        mu *= 4.0
        if mu > 1e10:
            break
    return best_h, best_c5


def slsqp_cons_fun(h):
    return np.sum(h) - _global_target

    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        def obj_fn(h):
            return lse_objective(h, alpha)
        def grad_fn(h):
            return lse_gradient(h, alpha)
        try:
            res = minimize(obj_fn, h_cur, jac=grad_fn,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd(h_start, n, target, time_limit, max_sweeps=5000, n_cand=50, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 40:
                break
    return best_h, best_c5

    best_score = compute_c5(best_h)

    if _tleft() > 80:
        h_pol, sc = refined_lbfgsb(best_h, n, target, alpha=500.0,
                                   time_limit=min(40, _tleft() * 0.5),
                                   mu_init=100.0, max_outer=20, lbfgs_iter=300)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tleft() > 60:
        h_slsqp = slsqp_round(best_h, n, target, [1000, 5000, 20000], 800)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp

    if _tleft() > 200:
        cd_time = min(250, _tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                              max_sweeps=8000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tleft() > 80:
        h_pol, sc = refined_lbfgsb(best_h, n, target, alpha=5000.0,
                                   time_limit=min(40, _tleft() * 0.4),
                                   mu_init=200.0, max_outer=25, lbfgs_iter=350)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tleft() > 60:
        ns_time = min(50, _tleft() * 0.3)
        h_ns, c5_ns = active_set_newton(best_h, n, target, time_limit=ns_time,
                                        top_k=10, max_iters=300, reg_init=0.05)
        if c5_ns < best_score:
            best_score = c5_ns
            best_h = h_ns

    if _tleft() > 200:
        cd_time = min(200, _tleft() * 0.2)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                              max_sweeps=6000, n_cand=60, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tleft() > 60:
        bl_time = min(50, _tleft() * 0.2)
        h_bl, c5_bl = block_move_cd(best_h, n, target, time_limit=bl_time,
                                    n_blocks=40, max_sweeps=2000)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl

    if _tleft() > 60:
        h_slsqp = slsqp_round(best_h, n, target, [20000, 80000, 300000], 600)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp

    if _tleft() > 100:
        rs_time = min(80, _tleft() * 0.3)
        h_rs, c5_rs = random_restarts_pair_cd(best_h, n, target,
                                              time_limit=rs_time, seed_val=seed)
        if c5_rs < best_score:
            best_score = c5_rs
            best_h = h_rs

    if _tleft() > 80:
        h_pol, sc = refined_lbfgsb(best_h, n, target, alpha=50000.0,
                                   time_limit=min(40, _tleft() * 0.4),
                                   mu_init=500.0, max_outer=20, lbfgs_iter=400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tleft() > 60:
        cd_time = min(40, _tleft() * 0.4)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                              max_sweeps=4000, n_cand=60, n_bisect=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tleft() > 150:
        for n_try in [800, 1000]:
            if _tleft() < 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires, sc_h = refined_lbfgsb(h_interp, n_try, target_try,
                                           alpha=1000.0,
                                           time_limit=min(30, _tleft() * 0.2),
                                           mu_init=100.0, max_outer=15, lbfgs_iter=300)
            if _tleft() > 40 and sc_h < best_score + 0.002:
                cd_t = min(30, _tleft() * 0.2)
                h_hires2, sc2 = pair_cd(h_hires, n_try, target_try,
                                        time_limit=cd_t,
                                        max_sweeps=2000, n_cand=40, n_bisect=13)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    if _tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slsqp = slsqp_round(best_h, n_use, target_use, [50000, 200000], 500)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp

    if _tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(25, _tleft() * 0.6)
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use, time_limit=cd_time,
                              max_sweeps=3000, n_cand=50, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, best_score, n_use
