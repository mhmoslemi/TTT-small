def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-10 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    cur = int(h_r.sum())
    if cur > target_sum:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:cur - target_sum]
        h_r[ones[order]] = 0
    elif cur < target_sum:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:target_sum - cur]
        h_r[zeros[order]] = 1
    return h_r
