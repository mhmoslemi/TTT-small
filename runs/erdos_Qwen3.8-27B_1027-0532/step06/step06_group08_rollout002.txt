    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum} using bisection."""
    n = len(h)
    h = h.astype(float).copy()
    if abs(h.sum() - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    # clip(h + lambda, 0, 1) is non-decreasing in lambda
    lo = target_sum - h.sum() - n  # very negative
    hi = target_sum - h.sum() + n  # very positive
    
    # Actually, let's be smarter
    # When lambda -> -inf: sum -> 0
    # When lambda -> +inf: sum -> n
    # We need sum(clip(h_i + lambda, 0, 1)) = target_sum
    
    def f(lam):
        return np.clip(h + lam, 0.0, 1.0).sum() - target_sum
    
    # Bisection
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    
    lam = (lo + hi) / 2.0

- Its minimum can be found by: find the t where the subgradient contains 0
- The subgradient at t is the set of convex combinations of {a_k : k is active at t}
- 0 ∈ subgradient iff min_{k active} a_k ≤ 0 ≤ max_{k active} a_k

1. Start at some t
2. Find all active k (those achieving the max)
3. If 0 is in [min a_k, max a_k] for active k, we're at the optimum
4. Otherwise, move t in the direction that will make more constraints active
5. Compute the new active set

def compute_c5(h):

def compute_correlation(h):

def project_box_sum(h, target_sum):

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def lse_gradient(h, alpha, topk=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        w = np.zeros_like(corr)
        w[idx] = corr[idx]
        max_w = w.max()
        shifted = alpha * (w - max_w)
        exp_w = np.exp(np.clip(shifted, -400.0, 0.0))
        p = exp_w / exp_w.sum()
    else:
        max_c = corr.max()
        shifted = alpha * (corr - max_c)
        exp_c = np.exp(np.clip(shifted, -400.0, 0.0))
        p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for ki in range(2 * n - 1):
        wk = p[ki]
        if wk < 1e-12:
            continue
        s = shifts[ki]
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
        else:
            sa = -s
            grad[sa:] += wk * (1.0 - h[:n - sa])
            grad[:n - sa] -= wk * h[sa:]
    return grad, p

def lse_objective(h, alpha, topk=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
        max_v = vals.max()
        shifted = alpha * (vals - max_v)
        return max_v + np.log(np.sum(np.exp(np.clip(shifted, -400.0, 0.0)))) / alpha
    else:
        max_c = corr.max()
        shifted = alpha * (corr - max_c)
        return max_c + np.log(np.sum(np.exp(np.clip(shifted, -400.0, 0.0)))) / alpha

def nesterov_optimize(h_start, n_iters, alpha, lr_init, lr_end, momentum, topk, time_limit, start_time, target_sum):
    n = len(h_start)
    h = h_start.copy().astype(float)
    v = np.zeros(n)
    prev_h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        y = h + momentum * (h - prev_h)
        y = np.clip(y, 0.0, 1.0)
        grad, _ = lse_gradient(y, alpha, topk)
        new_h = y - lr * grad
        new_h = project_box_sum(new_h, target_sum)
        prev_h = h
        h = new_h
        score = compute_c5(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_score

def coordinate_descent(h_start, n_sweeps, topk_cd, time_limit, start_time, target_sum):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)

    for sweep in range(n_sweeps):
        if time.time() - start_time > time_limit:
            break
        if time.time() - start_time > time_limit - 5:
            break
        active = np.argsort(corr)[-topk_cd:]
        for j in range(n):
            if time.time() - start_time > time_limit - 3:
                break
            slopes = np.zeros(len(active))
            intercepts = np.zeros(len(active))
            for ai, ki in enumerate(active):
                s = shifts[ki]
                a_val = 0.0
                if s == 0:
                    a_val = 1.0 - 2.0 * h[j]
                elif s > 0:
                    if j < n - s:
                        a_val += 1.0 - h[j + s]
                    if j >= s:
                        a_val -= h[j - s]
                else:
                    sa = -s
                    if j >= sa:
                        a_val += 1.0 - h[j - sa]
                    if j < n - sa:
                        a_val -= h[j + sa]
                slopes[ai] = a_val
                intercepts[ai] = corr[ki] - a_val * h[j]
            candidates = [0.0, 1.0, h[j]]
            for a1 in range(len(active)):
                for a2 in range(a1 + 1, len(active)):
                    denom = slopes[a2] - slopes[a1]
                    if abs(denom) > 1e-12:
                        t_cross = (intercepts[a1] - intercepts[a2]) / denom
                        if 0.0 <= t_cross <= 1.0:
                            candidates.append(t_cross)
            best_val = np.inf
            best_t = h[j]
            for t_c in candidates:
                t_c = max(0.0, min(1.0, t_c))
                vals = slopes * t_c + intercepts
                max_v = vals.max()
                other_vals = np.zeros(2 * n - 1)
                other_vals[active] = vals
                for ki in range(2 * n - 1):
                    if ki not in active:
                        s = shifts[ki]
                        if s == 0:
                            other_vals[ki] = (1.0 - 2.0 * t_c) * 0 + corr[ki] - (1.0 - 2.0 * h[j]) * h[j] + (1.0 - 2.0 * t_c) * t_c
                        else:
                            continue
                fv = max_v
                if fv < best_val:
                    best_val = fv
                    best_t = t_c
            if best_t != h[j]:
                h[j] = best_t
                h = np.clip(h, 0.0, 1.0)
                diff = target_sum - h.sum()
                if abs(diff) > 1e-10:
                    h = project_box_sum(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(corr.max()) * dx
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.argsort(corr)[-topk_cd:]
    return best_h, best_score

def lbfgsb_smooth(h_start, alpha, topk, maxiter, time_limit, start_time, target_sum):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        val = lse_objective(x, alpha, topk)
        grad, _ = lse_gradient(x, alpha, topk)
        penalty = 100.0 * (x.sum() - target_sum)
        val += 0.5 * penalty * (x.sum() - target_sum)
        grad = grad + penalty * np.ones(n)
        return val, grad

    x0 = h_start.copy().astype(float)
    try:
        res = minimize(obj_and_grad, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-14})
        h_new = project_box_sum(res.x, target_sum)
        sc = compute_c5(h_new)
        return h_new, sc
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def slsqp_polish(h_start, target_sum, alphas, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def smooth_obj(x):
            return lse_objective(x, alpha, None)
        def smooth_grad(x):
            g, _ = lse_gradient(x, alpha, None)
            return g
        def cons_fun(x):
            return x.sum() - target_sum
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(smooth_obj, h_cur, jac=smooth_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent(best_cont_h, n_sweeps=3, topk_cd=15,
                                         time_limit=min(80.0, tleft()),
                                         start_time=start_time, target_sum=target_init)
        if sc_cd < best_cont_score:
            best_cont_score = sc_cd
            best_cont_h = h_cd.copy()

    # Phase 4: Binary SA from best continuous
    if tleft() > 120:
        h_bin_round = project_binary(best_cont_h, int(n_init / 2.0))
        n_sa = int(min(5000000, max(100000, tleft() * 30000)))
        h_sa, c5_sa = sa_binary(h_bin_round, n_sa, 0.015, 0.00003, rng,
                                 time_limit=min(150.0, tleft()), start_time=start_time)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_bin_h = h_sa.copy()

    # Phase 5: L-BFGS-B polish
    if tleft() > 80:
        h_lb, sc_lb = lbfgsb_smooth(best_cont_h, alpha=100.0, topk=None,
                                     maxiter=5000, time_limit=min(60.0, tleft()),
                                     start_time=start_time, target_sum=target_init)
        if sc_lb < best_cont_score:
            best_cont_score = sc_lb
            best_cont_h = h_lb.copy()

    # Phase 6: Iterative improvement loop
    for iteration in range(8):
        if tleft() < 100:
            break

        # Nesterov from current best
        if tleft() > 80:
            alphas_sub = [50.0, 100.0, 300.0]
            n_it = int(min(100000, max(5000, tleft() * 50)))
            h_n2, sc_n2 = nesterov_anneal(best_cont_h, n_it, alphas_sub,
                                           lr_init=0.01, lr_end=0.001,
                                           momentum=0.92, topk=8,
                                           time_limit=min(60.0, tleft()),
                                           start_time=start_time, target_sum=target_init)
            if sc_n2 < best_cont_score:
                best_cont_score = sc_n2
                best_cont_h = h_n2.copy()

        # Binary round + greedy
        if tleft() > 60:
            h_br = project_binary(best_cont_h, int(n_init / 2.0))
            h_bg, c5_bg = binary_greedy_aggressive(h_br, max_iters=20000, rng=rng,
                                                    time_limit=min(25.0, tleft() * 0.3),
                                                    start_time=start_time, n_candidates=100)
            if c5_bg < best_bin_score:
                best_bin_score = c5_bg
                best_bin_h = h_bg.copy()

        # Short SA
        if tleft() > 50:
            n_sa2 = int(min(2000000, max(50000, tleft() * 20000)))
            h_sa2, c5_sa2 = sa_binary(best_bin_h, n_sa2, 0.008, 0.00002, rng,
                                       time_limit=min(35.0, tleft() * 0.4),
                                       start_time=start_time)
            if c5_sa2 < best_bin_score:
                best_bin_score = c5_sa2
                best_bin_h = h_sa2.copy()

        # Coordinate descent
        if tleft() > 40:
            h_cd2, sc_cd2 = coordinate_descent(best_cont_h, n_sweeps=2, topk_cd=12,
                                               time_limit=min(25.0, tleft() * 0.3),
                                               start_time=start_time, target_sum=target_init)
            if sc_cd2 < best_cont_score:
                best_cont_score = sc_cd2
                best_cont_h = h_cd2.copy()

    # Phase 7: Try slightly different resolutions
    for n_try in [n_init + 2, n_init + 6, n_init - 2]:
        if tleft() < 80:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_cont_h, n_try, target_try)
        n_it = int(min(50000, max(5000, tleft() * 40)))
        alphas_t = [50.0, 150.0]
        h_new_res, sc_new = nesterov_anneal(h_interp, n_it, alphas_t,
                                             lr_init=0.015, lr_end=0.001,
                                             momentum=0.9, topk=10,
                                             time_limit=min(40.0, tleft()),
                                             start_time=start_time, target_sum=target_try)
        if sc_new < best_cont_score:
            best_cont_score = sc_new
            best_cont_h = h_new_res.copy()

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_cont_h)
        t_use = n_use / 2.0
        h_ref, sc_ref = slsqp_polish(best_cont_h, t_use, alphas=[200, 500, 1500],
                                      maxiter=3000, time_limit=max(15.0, tleft() - 10),
                                      start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 9: Final Nesterov burst
    if tleft() > 20:
        n_use = len(best_cont_h)
        t_use = n_use / 2.0
        n_it_f = int(min(50000, max(10000, (tleft() - 5) * 80)))
        h_ff, sc_ff = nesterov_anneal(best_cont_h, n_it_f, [200.0, 500.0],
                                       lr_init=0.005, lr_end=0.0005,
                                       momentum=0.93, topk=6,
                                       time_limit=max(10.0, tleft() - 5),
                                       start_time=start_time, target_sum=t_use)
        if sc_ff < best_cont_score:
            best_cont_score = sc_ff
            best_cont_h = h_ff.copy()

    # Final selection
    n_final = len(best_cont_h)
    target_final = n_final / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    if n_final % 2 == 0:
        h_bin_final = project_binary(best_cont_h, n_final // 2)
        bin_score = compute_c5(h_bin_final)
        if bin_score < final_score:
            return h_bin_final, bin_score, n_final

    return h_final, final_score, n_final
