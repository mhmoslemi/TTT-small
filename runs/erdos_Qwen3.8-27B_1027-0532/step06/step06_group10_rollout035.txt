def pair_gradient(i, j, h, n, shifts):
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    v1 = (i + shifts >= 0) & (i + shifts < n)
    g[v1] -= (1.0 - h[i + shifts[v1]])
    v2 = (i - shifts >= 0) & (i - shifts < n)
    g[v2] += h[i - shifts[v2]]
    v3 = (j + shifts >= 0) & (j + shifts < n)
    g[v3] += (1.0 - h[j + shifts[v3]])
    v4 = (j - shifts >= 0) & (j - shifts < n)
    g[v4] -= h[j - shifts[v4]]
    return g

def subgrad_step(h, n, target, n_top=10, lr=0.03):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1][:n_top]
    alpha_w = 25.0
    w = np.exp(alpha_w * (corr[sorted_idx] - c_max))
    w = w / w.sum()
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for wi, s_idx in zip(w, sorted_idx):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    h_new = project_feasible(h - lr * grad, target)
    return h_new

def nesterov_smooth(h0, n, target, time_limit, max_steps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    stale = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        alpha = 30.0 + 10.0 * step
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 500:
                break
        grad = _smooth_grad(h, alpha, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_t in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_try = project_feasible(h + beta * v - lr_t * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-16:
                v = 0.8 * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return max(0.0, deadline - time.time())

    # Phase 1: SLSQP smoothing
    if tleft() > 120:
        h_p, s_p = slsqp_run(best_h, n, target,
                              [40, 150, 500, 2000],
                              2500, 1e-14, min(180.0, tleft() * 0.3), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Aggressive CD
    if tleft() > 120:
        h_cd, s_cd = cd_optimize(best_h, n, target,
                                  min(300.0, tleft() * 0.35),
                                  n_top=12, n_cand=35, bisect_iters=45)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Phase 3: SLSQP with high alpha
    if tleft() > 100:
        h_p, s_p = slsqp_run(best_h, n, target,
                              [500, 2000, 8000, 30000],
                              3000, 1e-15, min(150.0, tleft() * 0.3), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: CD refinement with more candidates
    if tleft() > 80:
        h_cd, s_cd = cd_optimize(best_h, n, target,
                                  min(200.0, tleft() * 0.3),
                                  n_top=15, n_cand=40, bisect_iters=50)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Phase 5: Nesterov smooth
    if tleft() > 80:
        h_ne, s_ne = nesterov_smooth(best_h, n, target,
                                      min(100.0, tleft() * 0.25), max_steps=15000)
        if s_ne < best_score:
            best_score = s_ne
            best_h = h_ne

    # Phase 6: Subgradient + CD interplay
    if tleft() > 100:
        h_sg = best_h.copy()
        for _ in range(200):
            if time.time() > deadline:
                break
            h_sg = subgrad_step(h_sg, n, target, n_top=8, lr=0.02)
            if compute_c5(h_sg) < best_score - 1e-6:
                break
        h_cd2, s_cd2 = cd_optimize(h_sg, n, target,
                                    min(60.0, tleft() * 0.2),
                                    n_top=10, n_cand=25, bisect_iters=35)
        if s_cd2 < best_score:
            best_score = s_cd2
            best_h = h_cd2

    # Phase 7: Random restarts
    if tleft() > 120:
        for attempt in range(4):
            if tleft() < 40:
                break
            noise_scales = [0.03, 0.08, 0.15, 0.25]
            noise_scale = noise_scales[min(attempt, 3)] / n
            h_pert = best_h + np.random.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_tr, s_tr = slsqp_run(h_pert, n, target,
                                    [100, 500, 2000],
                                    1500, 1e-14, min(35.0, tleft() * 0.2), start_time)
            h_cd3, s_cd3 = cd_optimize(h_tr, n, target,
                                        min(30.0, tleft() * 0.15),
                                        n_top=10, n_cand=25, bisect_iters=35)
            if s_cd3 < best_score:
                best_score = s_cd3
                best_h = h_cd3

    # Phase 8: Higher resolution
    if tleft() > 150:
        n_try_list = [600, 1000]
        for n_try in n_try_list:
            if tleft() < 100:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hr, s_hr = slsqp_run(h_interp, n_try, target_try,
                                    [50, 200, 800, 3000],
                                    2000, 1e-14, min(80.0, tleft() * 0.3), start_time)
            if tleft() < 50:
                if s_hr < best_score:
                    best_score = s_hr
                    best_h = h_hr
                continue
            h_cd4, s_cd4 = cd_optimize(h_hr, n_try, target_try,
                                        min(70.0, tleft() * 0.25),
                                        n_top=12, n_cand=30, bisect_iters=40)
            if s_cd4 < best_score:
                best_score = s_cd4
                best_h = h_cd4

    # Phase 9: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_p, s_p = slsqp_run(best_h, n_use, target_use,
                              [1000, 5000, 20000, 80000],
                              2000, 1e-15, min(40.0, tleft() * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 10: Final CD
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, s_cd = cd_optimize(best_h, n_use, target_use,
                                  min(15.0, tleft() * 0.7),
                                  n_top=10, n_cand=30, bisect_iters=40)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
