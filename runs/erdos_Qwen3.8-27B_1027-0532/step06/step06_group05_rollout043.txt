    v = np.zeros(n)
    t = 0

    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
        
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    return best_h, best_actual

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]

    final_sum = int(h_r.sum())

            final_sum -= 1
        else:

            final_sum += 1

    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

def adam_continuous(h0, target_sum, start_time, time_budget, alphas, lr, beta1, beta2, eps):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
        
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    return best_h, best_actual

def scale_binary(h_src, n_dst):
    """Scale a binary sequence from len(h_src) to n_dst."""
    n_src = len(h_src)
    h = np.zeros(n_dst)
    for i in range(n_dst):
        j = min(int(i * n_src / n_dst), n_src - 1)
        h[i] = h_src[j]
    return h

    best_n = n_init

    # Time budget per n value
    n_nvals = len(n_candidates)
    time_per_n = (budget_s * 0.4) / n_nvals
    
    for idx, n_try in enumerate(n_candidates):
