        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj_fn(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_value(corr, alpha) * (2.0 / len(x))
        def grad_fn(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            w = smooth_max_weights(corr, alpha)
            return compute_grad_weighted(x, w, len(x))
        def con_fun(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(len(x))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def greedy_cd(h0, n, target, time_limit, max_sweeps, n_cand, top_k=3):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        top_indices = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for s_idx in top_indices:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= top_k
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def three_point_cd(h0, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_approx = c5_val
        best_move = None
        n_low = len(low_set)
        n_high = len(high_set)
        for a in range(n_high):
            i = high_set[a]
            if h[i] < 1e-8:
                continue
            for b in range(n_low):
                j = low_set[b]
                if i == j:
                    continue
                d1 = min(h[i], 1.0 - h[j])
                if d1 < 1e-10:
                    continue
                g1 = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g1[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g1[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g1[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g1[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d1
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g1
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g1[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                d_star = 0.5 * (a_d + b_d)
                if d_star < 1e-10:
                    continue
                approx_val = np.max(corr + d_star * g1) * dx
                if approx_val < best_approx:
                    best_approx = approx_val
                    best_move = (i, j, d_star, g1)
        if best_move is not None:
            i, j, d_star, _ = best_move
            h[i] -= d_star
            h[j] += d_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def fourier_perturb(h, n, target, rng, num_modes=3):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    perturbation = np.zeros(n)
    for _ in range(num_modes):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.1)
        perturbation += amp * np.sin(freq * np.pi * x + phase)
    h_new = h + perturbation
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new

def random_perturb(h, n, target, rng, n_perturb=50, mag=0.1):
    h_new = h.copy()
    idx = rng.choice(n, size=n_perturb, replace=False)
    h_new[idx] += rng.uniform(-mag, mag, size=n_perturb)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new

def shift_perturb(h, n, target, rng, max_shift=10):
    s = rng.integers(-max_shift, max_shift + 1)
    if s == 0:
        return h.copy()
    h_new = np.roll(h, s)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    return h_new

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lp_refine(h0, n, target, time_limit, n_active=20, max_rounds=30):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            for mix in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    break
            else:
                break
    return best_h, best_c5

def nesterov_smooth(h0, n, target, time_limit, max_steps=5000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    stuck_count = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        alpha = 30.0 + step * 5.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        w = smooth_max_weights(corr, alpha)
        grad = compute_grad_weighted(h, w, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_trial in [lr, lr * 0.5, lr * 0.25]:
            h_try = project_feasible(h + beta * v - lr_trial * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                v = 0.8 * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
            if stuck_count > 300:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP warm start
    if tleft() > 150:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [30, 100, 400, 1500, 5000],
                                2000, 1e-14, min(80.0, tleft() * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Aggressive CD with top-k gradient
    if tleft() > 200:
        cd_time = min(250.0, tleft() * 0.35)
        h_cd, c5_cd = greedy_cd(best_h, n, target,
                                 time_limit=cd_time, max_sweeps=30000, n_cand=20, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: 3-point CD variant
    if tleft() > 150:
        h_cd3, c5_cd3 = three_point_cd(best_h, n, target,
                                        time_limit=min(120.0, tleft() * 0.2),
                                        max_sweeps=20000, n_cand=12)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 4: Nesterov smooth optimization
    if tleft() > 100:
        h_ne, c5_ne = nesterov_smooth(best_h, n, target,
                                       time_limit=min(80.0, tleft() * 0.2),
                                       max_steps=8000)
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne

    # Phase 5: Perturbation + re-optimization cycles
    n_perturb_cycles = 3
    for cycle in range(n_perturb_cycles):
        if tleft() < 150:
            break
        if cycle == 0:
            h_pert = fourier_perturb(best_h, n, target, rng, num_modes=4)
        elif cycle == 1:
            h_pert = random_perturb(best_h, n, target, rng, n_perturb=80, mag=0.08)
        else:
            h_pert = shift_perturb(best_h, n, target, rng, max_shift=5)
        
        # Quick SLSQP from perturbed point
        h_s, s_s = slsqp_round(h_pert, n, target,
                                [50, 200, 800, 3000],
                                1000, 1e-14, min(40.0, tleft() * 0.2), start_time)
        # Quick CD
        h_c, c_c = greedy_cd(h_s, n, target,
                              time_limit=min(60.0, tleft() * 0.3),
                              max_sweeps=10000, n_cand=20, top_k=3)
        if c_c < best_score:
            best_score = c_c
            best_h = h_c

    # Phase 6: Second round of CD at current best
    if tleft() > 200:
        cd_time = min(200.0, tleft() * 0.3)
        h_cd, c5_cd = greedy_cd(best_h, n, target,
                                 time_limit=cd_time, max_sweeps=30000, n_cand=25, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: LP refinement
    if tleft() > 100:
        h_lp, c5_lp = lp_refine(best_h, n, target,
                                 time_limit=min(80.0, tleft() * 0.25),
                                 n_active=20, max_rounds=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 8: Multi-resolution refinement
    if tleft() > 250:
        for n_try in [600, 1000]:
            if tleft() < 120:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hr, s_hr = slsqp_round(h_interp, n_try, target_try,
                                      [30, 100, 400, 1500],
                                      1500, 1e-14,
                                      min(60.0, tleft() * 0.2), start_time)
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr
            if tleft() < 60:
                break
            h_cd2, c5_cd2 = greedy_cd(best_h, len(best_h), float(len(best_h)) / 2.0,
                                       time_limit=min(60.0, tleft() * 0.25),
                                       max_sweeps=15000, n_cand=20, top_k=3)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 9: Final SLSQP polish at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_round(best_h, n_use, target_use,
                                        [500, 2000, 8000, 30000],
                                        1500, 1e-15,
                                        min(40.0, tleft() * 0.5), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 10: Final CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = greedy_cd(best_h, n_use, target_use,
                                 time_limit=min(20.0, tleft() * 0.7),
                                 max_sweeps=10000, n_cand=25, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Very final Nesterov
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ne, c5_ne = nesterov_smooth(best_h, n_use, target_use,
                                       time_limit=min(15.0, tleft() * 0.7),
                                       max_steps=3000)
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
