from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if mask.sum() == 0:
        return h
    h = h.copy()
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            m2 = h < 1.0
        else:
            m2 = h > 0.0
        if m2.sum() > 0:
            h[m2] += residual / m2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):

def subgradient_step(h, n, target, dx, corr, active_set, lr):
    """Take a subgradient step and return new h."""
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in active_set:
        k = s  # shift
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += (1.0 - h[idxs[v1] + k])
        grad[v2] -= h[idxs[v2] - k]
    grad *= dx
    grad -= np.mean(grad)
    h_new = project_feasible(h - lr * grad, target)
    return h_new

def line_search(h, n, target, dx, direction, max_lr=2.0):
    """Find best step size along direction using bisection on true objective."""
    # direction is the negative gradient (so we go in direction of decrease)
    # We want to find lr that minimizes compute_c5(projected(h + lr*direction))
    # Use a few trial steps
    
    best_lr = 0.0
    best_c5 = compute_c5(h)
    
    # Try a range of step sizes
    for _ in range(30):

        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_lr = max_lr
        max_lr *= 0.7

                best_lr = max_lr

def active_set_descent(h, n, target, time_limit, max_steps=5000, top_k=5):
    """Descent using top-k active set gradient."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:

        # Find top-k active shifts
        top_indices = np.argsort(corr)[-top_k:]
        active_shifts = (top_indices - mid).astype(int)

        grad = np.zeros(n)
        for k in active_shifts:

            v2 = (idxs - k >= 0) & (idxs - k < n)

            grad[v2] -= h[idxs[v2] - k]
        grad *= dx
        grad -= np.mean(grad)
        
        # Line search

        c5_try = compute_c5(h_try)
        
        improved = False
        lr = 1.0

            if c5_try < c5_now - 1e-15:

                improved = True

                best_c5 = c5_pert

def block_cd(h, n, target, time_limit, block_size=10, max_sweeps=3000):
    """Block coordinate descent: optimize blocks of consecutive points."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    n_blocks = n // block_size
    remaining = n - n_blocks * block_size
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Focus on the worst shift
        k_star = int(np.argmax(corr)) - (n - 1)
        
        # For each block, compute the optimal adjustment
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            start_idx = b * block_size
            end_idx = start_idx + block_size
            block = h[start_idx:end_idx].copy()
            
            # The change in corr[k_star] when we change the block:
            # d corr / d h[j] for j in block = (1-h[j+k_star]) - h[j-k_star]
            idxs = np.arange(start_idx, end_idx)
            v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
            v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
            local_grad = np.zeros(block_size)
            if v1.any():
                local_grad[v1] += (1.0 - h[idxs[v1] + k_star])
            if v2.any():
                local_grad[v2] -= h[idxs[v2] - k_star])

def project_feasible(h, target_sum):
