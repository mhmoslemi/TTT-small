Now let me also think about the L-BFGS-B gradient computation more carefully.

The smooth objective:
f(h) = c_max + (1/α) * log(Σ_k exp(α(c_k - c_max)))

where c_k = corr at lag k (before multiplying by dx).

Gradient:
∂f/∂h[i] = Σ_k w_k * ∂c_k/∂h[i]

where w_k = exp(α(c_k - c_max)) / Σ_j exp(α(c_j - c_max))

And ∂c_k/∂h[i]:
c_k = Σ_x h[x] * (1 - h[x+k]) (with truncation)

∂c_k/∂h[i] = (1-h[i+k]) * 1[x=i valid] + h[i-k] * 1[x=i-k valid] * (-1) * ... 

Wait: c_k = Σ_x h[x] * (1 - h[x+k])
∂/∂h[i] c_k = (1-h[i+k]) [from x=i term, if i+k in bounds]
             + h[i-k] * (-1) [from x=i-k term: ∂/∂h[i] of (1-h[i-k+k]) = ∂/∂h[i](1-h[i]) = -1]
             = (1-h[i+k]) - h[i-k]

with the understanding that out-of-bounds terms are 0.

So:
grad[i] = Σ_k w_k * ((1-h[i+k]) - h[i-k])

This can be rewritten as:
grad[i] = Σ_k w_k - Σ_k w_k * h[i+k] - Σ_k w_k * h[i-k]
        = S - (w ⊛ h)[i] - (w ⊛ h_rev)[i]

where S = Σ_k w_k = 1, ⊛ is cross-correlation, and h_rev is h reversed.

Using the FFT-based approach, I can compute this efficiently. But for n=600, even O(n²) is fast enough if vectorized.

    h: array of length n
    w: weight array of length 2*n-1 (for lags -(n-1) to (n-1))
    Returns gradient of smooth objective w.r.t. h

    n = len(h)

    i_idx = np.arange(n)[:, None]  # (n, 1)
    lags_row = lags[None, :]  # (1, 2n-1)

    im = i_idx - lags_row  # (n, 2n-1): i - lag
    
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)

def compute_gradient(h, w):
    n = len(h)
    lags = np.arange(-(n-1), n)
    
    grad = np.zeros(n)
    
    # For each lag k (stored at index j = k + (n-1) in w):
    # Contribution to grad[i] from lag k: w[j] * ((1-h[i+k]) - h[i-k])
    #   where h[x] = 0 if x out of [0, n-1]
    
    # Vectorize: for lag k, the contribution to grad is:
    #   w[j] * ((1 - h shifted by k) - h shifted by -k)
    #   = w[j] * (1 - h[i+k]) - w[j] * h[i-k]
    
    # sum over j:
    # grad[i] = sum_j w[j] - sum_j w[j]*h[i+k_j] - sum_j w[j]*h[i-k_j]
    
    # The first term is sum(w) = 1
    # The second and third are cross-correlations
    
    # Use FFT for efficiency:
    N_fft = 1
    while N_fft < 3*n:
        N_fft <<= 1
    
    w_pad = np.zeros(N_fft)
    # w has length 2n-1, representing lags -(n-1) to (n-1)
    # For the cross-correlation sum_j w[j]*h[i+k_j]:
    # Let k_j = j - (n-1), so i + k_j = i + j - (n-1)
    # This is (correlation of w and h) at offset i
    # Using FFT: ifft(fft(w) * conj(fft(h))) gives cross-correlation
    
    w_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    
    # cross_corr[m] = sum_j w[j] * h[m + j - (n-1)]  (approximately)
    # Actually, the standard cross-correlation formula:
    # (a ⊛ b)[m] = sum_n a[n] * conj(b[n-m])
    # Using FFT: ifft(fft(a) * conj(fft(b)))
    
    # For our case:
    # sum_j w[j] * h[i + j - (n-1)]
    # Let m = i, then: sum_j w[j] * h[m + j - (n-1)]
    # This is the cross-correlation of w and h at a specific offset
    
    # Actually, let me just use scipy or numpy directly for n=600.
    # It's only 600*1199 ≈ 719K multiplications, which numpy does in <1ms.
    
    # Direct computation:
    # For each i, grad[i] = sum_j w[j] * ((1-h[i+k_j] if 0<=i+k_j<n) - (h[i-k_j] if 0<=i-k_j<n))
    
    # Vectorized:
    i_idx = np.arange(n)
    k_vals = lags  # length 2n-1
    
    # ip[i, j] = i + k_vals[j]
    ip = i_idx[:, None] + k_vals[None, :]  # shape (n, 2n-1)
    im = i_idx[:, None] - k_vals[None, :]  # shape (n, 2n-1)
    
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    
    # h at ip positions (0 where out of bounds)
    h_ip = np.zeros_like(ip, dtype=float)
    h_ip[mask_p] = h[ip[mask_p]]
    
    # h at im positions
    h_im = np.zeros_like(im, dtype=float)
    h_im[mask_m] = h[im[mask_m]]
    
    # grad[i] = sum_j w[j] * ((1 - h_ip[i,j]) * mask_p[i,j] - h_im[i,j] * mask_m[i,j])
    weighted = w[None, :]
    grad = (weighted * (1.0 - h_ip) * mask_p).sum(axis=1) - (weighted * h_im * mask_m).sum(axis=1)
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    a = np.zeros(N_fft)
    a[:n] = h
    b = np.zeros(N_fft)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def compute_delta_2swap(h, i1, i2, j1, j2, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    for i in (i1, i2):
        idx = i - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += h[idx[mask]]
        idx = i + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += h[idx[mask]]
    for j in (j1, j2):
        idx = j - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= h[idx[mask]]
        idx = j + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def compute_gradient(h, w):

def gradient_refine(h0, target, start_time, time_limit, alphas):

    best_h = h.copy()
    best_score = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        lr = 0.5 / (1.0 + 0.1 * alpha)
        momentum = 0.6
        v = np.zeros(n)
        n_iter = int(2000)

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = float(corr.max())
            shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                break
            w = w / w_sum
            grad = compute_gradient(h, w)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 100 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def lbfgsb_refine(h0, target, alpha, maxiter, start_time, time_limit):
    n = len(h0)
    h_init = h0.astype(float).copy()

    def objective(h_flat):
        h_test = project_box_sum(h_flat, target)
        corr = full_corr_fast(h_test)
        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

    def gradient(h_flat):
        h_test = project_box_sum(h_flat, target)
        corr = full_corr_fast(h_test)
        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            return np.zeros(n)
        w = w / w_sum
        return compute_gradient(h_test, w)

    bounds = [(0.0, 1.0)] * n
    result = minimize(objective, h_init, jac=gradient, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 10})
    h_out = project_box_sum(result.x, target)
    return h_out, compute_c5(h_out)

            alloc = min(20.0, (time_limit - (time.time() - start_time)) * 0.4)
            h_opt, c5_opt = coarse_optimize(h_c_bin, n_coarse, target_coarse,
                                            k_arr_c, rng, alloc, start_time)
            x_fine = np.linspace(0, 1, n_init)
            h_fine = np.interp(x_fine, x_new, h_opt)
            h_fine_bin = project_binary(h_fine, target_init)
            c5_fine = compute_c5(h_fine_bin)
            candidates.append((h_fine_bin, c5_fine))

    candidates.sort(key=lambda x: x[1])
    return candidates

def block_reverse(h, rng, max_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_len, n // 6) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def diverse_perturb(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    phase_time = min(80.0, tleft() * 0.12)
    candidates = multi_resolution_init(best_h, n_init, target_init, k_arr,
                                       rng, phase_time, start_time)
    if len(candidates) > 0:
        cand_h, cand_score = candidates[0]
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()

    if tleft() > 60:
        alloc = min(400.0, tleft() * 0.35)
        n_sa = int(min(8000000, alloc * 5000))
        h_sa, c5_sa = sa_2swap(best_h, n_sa, 0.02, 0.00005, rng,
                               alloc, start_time, k_arr,
                               two_swap_frac=0.35, top_focus=5)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 100:
        target_use = float(n_init) / 2.0
        alphas = [10, 30, 100, 300, 1000, 3000]
        alloc = min(250.0, tleft() * 0.25)
        h_ref, sc = gradient_refine(best_h, target_use, start_time, alloc, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 80:
        h_bin = project_binary(best_h, target_init)
        alloc = min(150.0, tleft() * 0.2)
        n_sa2 = int(min(5000000, alloc * 40000))
        h_sa2, c5_sa2 = sa_2swap(h_bin, n_sa2, 0.012, 0.00002, rng,
                                 alloc, start_time, k_arr,
                                 two_swap_frac=0.4, top_focus=5)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    if tleft() > 80:
        target_use = float(n_init) / 2.0
        alphas2 = [50, 200, 800, 3000, 10000, 30000]
        alloc = min(200.0, tleft() * 0.2)
        h_ref2, sc2 = gradient_refine(best_h, target_use, start_time, alloc, alphas2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    for restart in range(10):
        if tleft() < 60:
            break
        if restart % 3 == 0:
            h_pert = diverse_perturb(best_h, rng, rng.randint(3, 8))
        elif restart % 3 == 1:
            h_pert = block_reverse(best_h, rng, 12)
        else:
            h_pert = diverse_perturb(best_h, rng, rng.randint(2, 5))
            h_pert = block_reverse(h_pert, rng, 8)

        alloc = min(50.0, tleft() * 0.07)
        h_sr, c5_sr = sa_2swap(h_pert, int(min(1000000, alloc * 40000)),
                               0.01, 0.00002, rng, alloc, start_time,
                               k_arr, two_swap_frac=0.4)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    if tleft() > 60:
        target_use = float(n_init) / 2.0
        h_bin3 = project_binary(best_h, target_init)
        alloc = min(100.0, tleft() * 0.15)
        n_sa3 = int(min(3000000, alloc * 40000))
        h_sa3, c5_sa3 = sa_2swap(h_bin3, n_sa3, 0.006, 0.000005, rng,
                                 alloc, start_time, k_arr,
                                 two_swap_frac=0.35)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    if tleft() > 40:
        target_use = float(n_init) / 2.0
        alphas_final = [200, 1000, 5000, 20000]
        alloc = min(80.0, tleft() * 0.2)
        h_ref3, sc3 = gradient_refine(best_h, target_use, start_time, alloc, alphas_final)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    if tleft() > 25:
        h_bin_final = project_binary(best_h, target_init)
        alloc = max(5.0, tleft() - 10.0)
        n_final_sa = int(min(2000000, alloc * 50000))
        h_sa_f, c5_f = sa_2swap(h_bin_final, n_final_sa, 0.004, 0.000001,
                                rng, alloc, start_time, k_arr, two_swap_frac=0.3)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
