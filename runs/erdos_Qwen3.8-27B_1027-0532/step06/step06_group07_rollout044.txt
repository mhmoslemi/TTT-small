    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

        return h

    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):

    _global_n = n
    _global_target = target

                improved = True
                break
            lr *= 0.5
        
        if not improved:
            lr *= 0.1
            if lr < 1e-10:
                break

    idxs = np.arange(n)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)

def random_pair_cd(h, n, target, time_limit, max_sweeps=3000, n_bisect=12):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    rng = np.random.RandomState(12345)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_pairs = 30
        for _ in range(n_pairs):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g = np.zeros(n_corr)
            g[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g[v_im] += h[i - shifts[v_im]]
            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def vectorized_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=35, n_bisect=12):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 120:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        high_grads = np.zeros((n_cand, n_corr))
        for idx, i in enumerate(high_set):
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            high_grads[idx][v_ip] -= (1.0 - h[i + shifts[v_ip]])
            high_grads[idx][v_im] += h[i - shifts[v_im]]

        low_grads = np.zeros((n_cand, n_corr))
        for idx, j in enumerate(low_set):
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            low_grads[idx][v_jp] += (1.0 - h[j + shifts[v_jp]])
            low_grads[idx][v_jm] -= h[j - shifts[v_jm]]

        combined = high_grads[:, np.newaxis, :] + low_grads[np.newaxis, :, :]

        h_high = h[high_set]
        h_low = h[low_set]
        delta_max = np.minimum(h_high[:, np.newaxis], 1.0 - h_low[np.newaxis, :])
        delta_max = np.maximum(delta_max, 0.0)

        a = np.zeros_like(delta_max)
        b = delta_max.copy()

        for _ in range(n_bisect):
            mid_d = 0.5 * (a + b)
            vals = corr[np.newaxis, np.newaxis, :] + mid_d[:, :, np.newaxis] * combined
            f_mid = np.max(vals, axis=2)
            active = vals >= (f_mid[:, :, np.newaxis] - 1e-14)
            combined_masked = np.where(active, combined, -np.inf)
            f_prime = np.max(combined_masked, axis=2)
            f_prime = np.where(np.isinf(f_prime), 0.0, f_prime)
            shrink = f_prime > 1e-14
            b = np.where(shrink, mid_d, b)
            a = np.where(shrink, a, mid_d)

        delta_star = 0.5 * (a + b)
        approx_vals = corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * combined
        approx_max = np.max(approx_vals, axis=2) * dx
        approx_max = np.where(delta_star > 1e-12, approx_max, c5_val)

        flat_idx = np.argmin(approx_max)
        best_val = approx_max.ravel()[flat_idx]

        if best_val < c5_val - 1e-14:
            p, q = np.unravel_index(flat_idx, approx_max.shape)
            i = high_set[p]
            j = low_set[q]
            h[i] -= delta_star[p, q]
            h[j] += delta_star[p, q]
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 50:
                break

    return best_h, best_c5

def active_set_subgradient(h, n, target, time_limit, max_iter=3000, tol_frac=0.005):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    idxs = np.arange(n)
    mid = n - 1

    lr = 0.05
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_cur.copy()

        c_max = float(np.max(corr))
        threshold = c_max - tol_frac / dx
        active = corr > threshold
        n_active = np.sum(active)

        if n_active == 0:
            break
        if n_active > 30:
            top_idx = np.argsort(corr)[-30:]
            active = np.zeros(len(corr), dtype=bool)
            active[top_idx] = True
            n_active = 30

        grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for s_idx in active_indices:
            k = int(s_idx) - mid
            w = max(corr[int(s_idx)] - threshold, 0.001)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        direction = -grad / grad_norm
        improved = False
        for attempt in range(15):
            step_size = lr * (0.6 ** attempt)
            h_try = project_feasible(h_cur + step_size * direction * grad_norm, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-12:
                h_cur = h_try
                improved = True
                break

            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.4
            if lr < 1e-8:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def random_restarts_cd(h_init, n, target, time_limit, n_restarts=5, n_cand=35, n_bisect=12):
