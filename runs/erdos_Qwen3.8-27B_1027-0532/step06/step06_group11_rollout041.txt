```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    """Gradient of ∑ h[i](1-h[i+k]) with respect to h"""
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def softmax_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    if alpha < 1e-6:
        return float(max_C)
    shifted = np.clip(alpha * (C - max_C), -50.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)

def softmax_grad(h, alpha):
    """Gradient of softmax objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(alpha * (C - max_C), -50.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    weights /= wsum  # These sum to 1
    
    mid = n - 1
    grad = np.zeros(n)
    for s_idx in range(len(weights)):
        if weights[s_idx] < 1e-10:
            continue
        k = s_idx - mid
        grad += weights[s_idx] * get_grad_shift(h, k, n)
    grad *= dx
    return grad

def slsqp_run(h_start, n, target, alphas, iters_per_alpha, ftol=1e-16):
    """Run SLSQP with increasing alpha"""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(x):
            return softmax_objective(x, alpha)
        def obj_grad(x):
            return softmax_grad(x, alpha)
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': iters_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5
```
