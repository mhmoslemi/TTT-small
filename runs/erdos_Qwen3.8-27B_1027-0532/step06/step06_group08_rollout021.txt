    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    di = 1.0 - 2.0*h[i]
    dj = 1.0 - 2.0*h[j]
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += di * (1.0 - h[idx[mask]])
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= di * h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += dj * (1.0 - h[idx[mask]])
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= dj * h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h += (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-10:
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

Actually, one more important consideration: the `np.correlate(h, 1-h, mode='full')` computation. For n=800, this gives a vector of length 1599. The element at index n-1+

0 corresponds to k=0 (no shift), index n-1+k corresponds to shift k, and index n-1-k corresponds to shift -k.

The maximum of this vector (times dx) gives C₅. We want to minimize this.

Note that corr[n-1] = ∑ h[i]·(1-h[i]) = 0 for binary h (since h[i](1-h[i])=0). So the maximum is always at a non-zero shift.

Let me now write the complete code. I'll make it clean and efficient.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_delta(h, i, j, k_arr, n):
    result = np.zeros(2*n-1)
    di = 1.0 - 2.0*h[i]
    dj = 1.0 - 2.0*h[j]
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += di * (1.0 - h[idx[mask]])
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= di * h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += dj * (1.0 - h[idx[mask]])
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= dj * h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def sa_main(h, n_iters, t_init, t_final, rng, time_limit, start_time, guided_frac=0.8):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)
    k_arr = np.arange(2*n-1) - (n-1)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 80000:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if use_guided:
            # Pick from top 5 lags
            top5 = np.argsort(full)[-5:]
            k_idx = top5[rng.randint(len(top5))]
            k_star = int(k_idx) - (n - 1)
            if k_star != 0:
                # Find i that is 1 and has many 1s at distance k_star
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] = h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                # We want h[i]=1 and A[i] high (many 1s at distance k)
                # Actually we want to REDUCE the overlap at k_star
                # Overlap at k: sum_i h[i](1-h[i+k])
                # To reduce: set h[i]=0 where h[i]=1 and h[i+k]=0
                # So we want i with h[i]=1 and h[i+k_star]=0 (or h[i-k_star]=0)
                
                # Simpler: just pick i from ones that maximizes A[i]
                # (A[i] counts neighbors at distance k, high A means more "wasted" 1s)
                candidates_i = ones
                i = candidates_i[rng.randint(len(candidates_i))]
                
                # Actually, let me use the A array to guide
                # A[i] = h[i+k] + h[i-k] (the sum of neighbors at distance k)
                # If A[i] = 0, both neighbors are 0, so changing h[i] from 1 to 0
                # would help at lag k (reduces overlap by 2)
                # If A[i] = 2, both neighbors are 1, so changing h[i] from 1 to 0
                # would hurt at lag k (reduces overlap by 0, but changes other lags)
                
                # So we want i with low A[i] (few 1-neighbors at distance k)
                low_A = ones[A[ones] <= 1.0]
                if len(low_A) > 0:
                    i = low_A[rng.randint(len(low_A))]
                else:
                    i = ones[rng.randint(len(ones))]
                
                # For j: we want to set h[j]=1 where j was 0
                # This helps at lag k if h[j+k]=0 and h[j-k]=0 (i.e., B[j] is low)
                B = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] = h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                
                low_B = zeros[B[zeros] >= 1.0]
                # Wait, I want B[j] to be LOW (few 1-neighbors), so setting h[j]=1
                # won't create new overlaps
                low_B = zeros[B[zeros] <= 1.0]
                if len(low_B) > 0:
                    j = low_B[rng.randint(len(low_B))]
                else:
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        di = 1.0 - 2.0*h[i]
        dj = 1.0 - 2.0*h[j]
        
        result = np.zeros(2*n-1)
        idx = i + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += di * (1.0 - h[idx[mask]])
        idx = i - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= di * h[idx[mask]]
        idx = j + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += dj * (1.0 - h[idx[mask]])
        idx = j - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= dj * h[idx[mask]]
        
        new_full = full + result
        new_max = float(new_full.max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            full = new_full
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
