def l_bfgs_optimize(h_init, n, target, alpha, maxiter, time_limit):
    """Optimize using L-BFGS-B with smooth max objective."""
    start = time.time()
    
    def objective_and_grad(x):

            raise TimeoutError()
        val, grad = compute_gradient_w(x, alpha)
        # Add penalty for sum constraint
        sum_pen = 1000.0 * (np.sum(x) - target)
        grad = grad + sum_pen * np.ones(n) / n  # gradient of penalty
        # Actually, let me use a quadratic penalty
        diff = np.sum(x) - target
        val += 500.0 * diff * diff
        grad += 1000.0 * diff * np.ones(n) / n
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective_and_grad, h_init, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h_init, target)
    
    return h_out, compute_c5(h_out)

def pair_cd_fast(h_init, n, target, time_limit, n_cand=30):
    """Fast pair coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles
        ks = np.arange(-(n - 1), n)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_and_optimize(h_base, n, target, time_limit, rng, alpha=5000):
    """Perturb h_base randomly, then optimize."""
    # Random perturbation
    h_pert = h_base + rng.normal(0, 0.05, size=n)
    h_pert = project_feasible(h_pert, target)
    
    # Optimize
    h_opt, c5 = l_bfgs_optimize(h_pert, n, target, alpha=alpha, maxiter=200, time_limit=time_limit)
    
    # Pair CD polish
    if time_limit > 10:
        h_cd, c5_cd = pair_cd_fast(h_opt, n, target, time_limit=min(15, time_limit*0.5), n_cand=20)
        if c5_cd < c5:
            h_opt, c5 = h_cd, c5_cd
    
    return h_opt, c5

Actually, np.correlate(a, v, 'full') returns:

For a = [a, b] (len 2) and v = [1-a, 1-b] (len 2):

The max of this is what we want to minimize.
