            if k_star >= n:
                continue
            # i candidates: h(i)=1 and h(i+k_star)=0 (reduces corr at k_star)
            valid_i = [i for i in ones[:80] if i + k_star < n and h[i + k_star] == 0]
            # j candidates: h(j)=0 and h(j+k_star)=1 (reduces corr at k_star)
            valid_j = [j for j in zeros[:80] if j + k_star < n and h[j + k_star] == 1]

            if len(valid_i) == 0 or len(valid_j) == 0:
                continue

            # Limit candidates for speed
            valid_i = valid_i[:12]
            valid_j = valid_j[:12]

            for i in valid_i:
                for j in valid_j:
                    if i == j:
                        continue
                    dk = delta_k(h, i, j, k_range)
                    # Full new corr
                    new_corr_val = corr.copy()
                    new_corr_val[n_offset + k_range] += dk
                    new_corr_val[n_offset - k_range] -= dk
                    nm = float(np.max(new_corr_val))
                    improvement = max_val - nm
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_i = i
                        best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def sa_improved(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_offset = n - 1
    k_range = np.arange(1, n)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 3000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = min(it / max(max_iters - 1, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.5:
            # Targeted: pick from worst shift
            worst_idx = np.argmax(corr)
            k_star = abs(worst_idx - n_offset)
            if k_star == 0 or k_star >= n:
                k_star = rng.randint(1, n)
            valid_i = [i for i in ones if i + k_star < n]
            valid_j = [j for j in zeros if j + k_star < n]
            if len(valid_i) > 0 and len(valid_j) > 0:
                h_new[valid_i[rng.randint(len(valid_i))]] = 0
                h_new[valid_j[rng.randint(len(valid_j))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.8:
            # Random single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif r < 0.95:
            # 2-swap move
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        else:
            # Block move: swap a contiguous block
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-20))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def coarse_to_fine(h_fine_init, n_coarse, time_limit, start_time, rng):
    n_fine = len(h_fine_init)
    target_coarse = float(n_coarse / 2)
    target_coarse_int = int(n_coarse / 2)

    # Downsample
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    h_coarse = np.interp(x_coarse, x_fine, h_fine_init)
    h_coarse = project_box_sum(h_coarse, target_coarse)
    h_coarse_bin = project_binary(h_coarse, target_coarse_int)

    # Optimize at coarse level
    c5_frac = 2.0 / n_coarse
    corr_c = np.correlate(h_coarse_bin, 1.0 - h_coarse_bin, mode='full')
    best_c5_c = float(corr_c.max()) * c5_frac

    time_alloc = min(60, (time_limit - (time.time() - start_time)) * 0.5)
    if time_alloc > 10:
        h_coarse_opt, c5_opt = targeted_local_search(h_coarse_bin, target_coarse_int,
                                                       time_limit=time_alloc, start_time=start_time)
        if c5_opt < best_c5_c:
            best_c5_c = c5_opt
            h_coarse_opt_best = h_coarse_opt
        else:
            h_coarse_opt_best = h_coarse_bin

        if time_limit - (time.time() - start_time) > 30:
            time_alloc2 = min(40, (time_limit - (time.time() - start_time)) * 0.4)
            h_sa, c5_sa = sa_improved(h_coarse_opt_best, target_coarse_int, 50000,
                                       0.01, 0.0001, rng, time_limit=time_alloc2, start_time=start_time)
            if c5_sa < best_c5_c:
                best_c5_c = c5_sa
                h_coarse_opt_best = h_sa

    # Upsample back to fine
    h_upsampled = np.interp(x_fine, x_coarse, h_coarse_opt_best)
    return project_box_sum(h_upsampled, float(n_fine / 2))

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_step(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_objective(h_cur, beta, target, penalty)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            obj_cand = smooth_objective(h_candidate, beta, target, penalty)
            if obj_cand < curr_obj:
                h_cur = h_candidate
                curr_obj = obj_cand
            else:
                sc_cur = compute_c5(h_cur)
                if sc < sc_cur:
                    h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(h_init, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Targeted local search (fast, focused on worst shifts)
    if tleft() > 60:
        time_alloc = min(120, tleft() * 0.15)
        h_ls, c5_ls = targeted_local_search(h_bin, target_int, time_limit=time_alloc,
                                             start_time=start_time, max_iters=300000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 2: SA from best binary
    if tleft() > 80:
        time_alloc = min(200, tleft() * 0.2)
        max_it = int(min(200000, time_alloc * 3000))
        h_sa, c5_sa = sa_improved(best_h, target_int, max_it, 0.01, 0.00005,
                                   rng, time_limit=time_alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Coarse-to-fine
    if tleft() > 80:
        n_coarse = max(100, n // 3)
        if n_coarse % 2 != 0:
            n_coarse += 1
        h_upsampled = coarse_to_fine(best_h, n_coarse, time_limit=tleft(),
                                      start_time=start_time, rng=rng)
        h_up_bin = project_binary(h_upsampled, target_int)
        c5_up = compute_c5(h_up_bin)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_bin.copy()

        # Continue optimizing from upsampled
        if tleft() > 40:
            time_alloc = min(60, tleft() * 0.2)
            h_ls2, c5_ls2 = targeted_local_search(h_up_bin, target_int, time_limit=time_alloc,
                                                    start_time=start_time)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()

    # Phase 4: Multi-restart SA with different temperatures
    for restart in range(4):
        if tleft() < 50:
            break
        # Create perturbed version
        h_pert = best_h.copy()
        n_swap = rng.randint(5, 15)
        ones_idx = np.where(h_pert == 1)[0]
        zeros_idx = np.where(h_pert == 0)[0]
        for _ in range(n_swap):
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones_idx = np.where(h_pert == 1)[0]
            zeros_idx = np.where(h_pert == 0)[0]

        time_alloc = min(80, tleft() * 0.15)
        max_it = int(min(100000, time_alloc * 2500))
        t_init_sa = 0.005 + 0.003 * restart
        h_sr, c5_sr = sa_improved(h_pert, target_int, max_it, t_init_sa, 0.00003,
                                   rng, time_limit=time_alloc, start_time=start_time)
        if c5_sr < best_c5:
            best_c5 = c5_sr
            best_h = h_sr.copy()

    # Phase 5: Continuous L-BFGS refinement
    if tleft() > 60:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [50, 150, 400, 1000]:
            if tleft() < 30:
                break
            time_alloc = min(40, tleft() * 0.15)
            h_opt = lbfgs_step(h_cont, beta, target, 500.0,
                                maxiter=200, time_limit=time_alloc, start_time=start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 6: Binary local search again from best
    if tleft() > 40:
        h_bin_final = project_binary(best_h, target_int)
        time_alloc = min(80, tleft() * 0.3)
        h_ls3, c5_ls3 = targeted_local_search(h_bin_final, target_int, time_limit=time_alloc,
                                                start_time=start_time, max_iters=500000)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3.copy()

    # Phase 7: Final SA polish
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target_int)
        time_alloc = min(50, tleft() - 10)
        max_it = int(min(80000, time_alloc * 3000))
        h_sf, c5_sf = sa_improved(h_bin_final, target_int, max_it, 0.003, 0.00001,
                                   rng, time_limit=time_alloc, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 8: Final continuous polish
    if tleft() > 20:
        h_final = project_box_sum(best_h.copy(), target)
        for beta_f in [500, 1500]:
            if tleft() < 10:
                break
            h_polished = lbfgs_step(h_final, beta_f, target, 1000.0,
                                     maxiter=100, time_limit=min(15, tleft()-5), start_time=start_time)
            h_final = h_polished
            sc = compute_c5(h_final)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_final.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    # Ensure exact sum
    if abs(best_h.sum() - target_final) > 1e-6:
        best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
