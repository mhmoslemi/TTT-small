    """Search over two-interval configurations."""
    best_score = float('inf')
    best_params = None
    target_int = int(target)

    max_iters = 500000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        gap = rng.randint(0, n // 2 + 1)
        max_start = n - target_int - gap
        if max_start < 0:
            continue
        start1 = rng.randint(0, max_start + 1)
        len1 = rng.randint(1, target_int)
        len2 = target_int - len1
        start2 = start1 + len1 + gap
        if start2 + len2 > n:
            continue
        
        h = np.zeros(n)
        h[start1:start1+len1] = 1.0
        h[start2:start2+len2] = 1.0
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_params = (start1, len1, gap)
    
    if best_params is not None:
        s1, l1, g = best_params
        l2 = target_int - l1
        s2 = s1 + l1 + g
        h = np.zeros(n)
        h[s1:s1+l1] = 1.0
        h[s2:s2+l2] = 1.0
        return h, best_score
    return None, best_score

def subgradient_method(h0, target, time_limit, start_time, lr_init=0.01):
    """Subgradient method for the max-of-linear-functions objective."""
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(corr.max()) * (2.0/n)
    best_h = h.copy()
    
    t = 0
    for it in range(100000):
        if time.time() - start_time > time_limit:
            break
        t += 1
        lr = lr_init / np.sqrt(t + 1)
        
        # Find top k values
        top_k = 5
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        weights = np.exp(corr[sorted_idx] / (lr * 10))
        weights /= weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for wi, idx in zip(weights, sorted_idx):
            k = int(idx) - (n - 1)
            # grad[n] = (1-h[n-k]) - h[n+k] for valid indices
            if k >= 0:
                # n+k is valid for n in [0, n-1-k]
                # (1-h[n-k]) for n in [k, n-1] (n-k valid)
                # -h[n+k] for n in [0, n-1-k]
                grad[:n-k] -= h[k:]  # -h[n+k] for n=0..n-1-k
                grad[k:] += (1 - h[:n-k])  # (1-h[n-k]) for n=k..n-1
            else:
                mk = -k
                grad[mk:] -= h[:n-mk]  # -h[n+k] = -h[n-mk] for n=mk..n-1
                grad[:n-mk] += (1 - h[mk:])  # (1-h[n-k]) = (1-h[n+mk]) for n=0..n-1-mk
        
        h = project_box_sum(h - lr * grad, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(corr.max()) * (2.0/n)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score

For the swap to be beneficial, we want c_k + Δc_k < current_max for all k where c_k is near the max.

This is a complex condition. A simpler approach: just try the swap and compute the new max. Since computing the full correlation is O(N²) (or O(N log N) with FFT), and we might try O(N) swaps per iteration, the total per iteration is O(N³) or O(N² log N).

For N=606, O(N³) = 223M per iteration. If each iteration takes 0.1s (optimistic), we can do about 100 iterations in 10s. That's not great.

A better approach: only recompute the full correlation when we accept a move. For evaluating a candidate swap, compute Δc_k for all k in O(N) time (using the formula above). Then check if max(c + Δc) < current_max.

This makes each candidate evaluation O(N), and if we try O(N) candidates per iteration, it's O(N²) per iteration. For N=606, that's 367K per iteration, which is very fast.

```python
def eval_swap(h, ones, zeros, i, j):
    """Compute the change in c_k for swapping h[i]:1->0, h[j]:0->1."""
    n = len(h)
    delta = np.zeros(2*n - 1)
    # Effect of h[i] going from 1 to 0:
    # -A_k(i) where A_k(i) = (1-h[i-k])*1[i-k valid] - h[i+k]*1[i+k valid]
    # For k >= 0 (k goes from 0 to n-1, corresponding to s = k + (n-1)):
    for k in range(-(n-1), n):
        # i-k valid?
        ik = i - k
        if 0 <= ik < n:
            delta[n-1+k] -= (1 - h[ik])
        # i+k valid?
        ik2 = i + k
        if 0 <= ik2 < n:
            delta[n-1+k] += h[ik2]
        # j-k valid?
        jk = j - k
        if 0 <= jk < n:
            delta[n-1+k] += (1 - h[jk])
        # j+k valid?
        jk2 = j + k
        if 0 <= jk2 < n:
            delta[n-1+k] -= h[jk2]
    return delta
```
