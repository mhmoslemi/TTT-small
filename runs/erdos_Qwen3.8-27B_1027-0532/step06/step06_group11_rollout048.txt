import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    # Gradient of C(k) = sum_i h[i]*(1-h[i+k]) * dx w.r.t. h
    # d/dh[j] of C(k) = dx * [(1-h[j-k]) if j-k is valid] - dx * [h[j+k] if j+k is valid]
    idxs = np.arange(n)
    g = np.zeros(n)
    # Term 1: h[j] appears in C(k) as h[j]*(1-h[j+k]) for j+k < n
    # d/dh[j] = (1-h[j+k]) for j+k < n, and 0 otherwise
    mask1 = (idxs + k < n) & (idxs + k >= 0)
    g[mask1] += (1.0 - h[idxs[mask1] + k])
    # Term 2: h[j] appears in C(k) as h[j-k]*(1-h[j]) for j-k >= 0
    # d/dh[j] = -h[j-k] for j-k >= 0, and 0 otherwise
    mask2 = (idxs - k >= 0) & (idxs - k < n)
    g[mask2] -= h[idxs[mask2] - k]
    return g

def power_sum_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    return float(np.sum(c**p)) ** (1.0/p)

def slsqp_power_sum(h_start, n, target, ps_list, maxiter_per_p, ftol):
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    for p in ps_list:
        def obj(h):
            return power_sum_obj(h, p)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_p, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def chase_max_optimize(h_start, n, target, time_limit, max_steps, lr_init):
    """Iteratively chase the maximum shift with projected gradient descent."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    lr = lr_init
    no_improve = 0
    patience = 500
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > patience:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top shifts
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:]
        
        # Gradient is the average gradient over top shifts
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= top_k
        grad *= dx
        grad -= np.mean(grad)
        
        # Backtracking line search
        improved = False
        for backtracking in range(15):
            lr_try = lr * (0.5 ** backtracking)
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            no_improve += 3
        else:
            lr *= 1.1
    
    return best_h, best_c5

def mirror_descent_opt(h_start, n, target, time_limit, max_steps, lr):

def pair_cd_fast(h_start, n, target, time_limit, max_sweeps, n_cand):
    """Faster pair coordinate descent using precomputed influence vectors."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 150:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # For each pair (i in high, j in low), compute the change in corr
        # Moving delta from i to j changes corr by delta * (influence[i] - influence[j])
        # where influence[p] = d(corr)/d(h[p]) for the current h
        # We need to find the pair that most reduces max(corr)
        
        best_trial = c5_val
        best_move = None
        
        # Sample pairs to save time
        n_sample = min(n_cand * n_cand, 500)
        pairs = []
        for i in high_set:
            for j in low_set:
                if i != j:
                    pairs.append((i, j))
        
        if len(pairs) > n_sample:
            rng_local = np.random.RandomState(sweep)
            indices = rng_local.choice(len(pairs), size=n_sample, replace=False)
            pairs = [pairs[i] for i in indices]
        
        for i, j in pairs:
            d = grad[i] - grad[j]
            if abs(d) < 1e-14:
                continue
            # Find optimal delta by bisection
            a_d = 0.0
            b_d = min(h[i], 1.0 - h[j])
            if b_d <= 1e-14:
                continue
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * d
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(d[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            trial_c5 = float(np.max(corr + delta_star * d)) * dx
            if trial_c5 < best_trial:
                best_trial = trial_c5
                best_move = (i, j, delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()

def project_sum_box(y, target):
    """Project y onto {0≤h≤1, sum(h)=target} using bisection on threshold."""
    # h = clip(y - tau, 0, 1), find tau such that sum(h) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += residual / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_influence_matrix(h, n):
    """Compute f[p, s] = d(corr[s])/d(h[p]) for all p, s."""
    f = np.zeros((n, 2*n-1))
    idxs = np.arange(n)
    for s in range(2*n-1):
        k = s - (n-1)
        # f[:, s] = get_grad_shift(h, k, n)
        mask1 = (idxs + k >= 0) & (idxs + k < n)
        mask2 = (idxs - k >= 0) & (idxs - k < n)
        f[mask1, s] += (1.0 - h[idxs[mask1] + k])

def get_influence_row(h, p, n):
    """Compute f[p, s] for all s (one row of the influence matrix)."""
    idxs = np.arange(2*n-1)

    f_row = np.zeros(2*n-1)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
