        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute gradient focused on worst shifts
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            valid_pk = (idx + k >= 0) & (idx + k < n)
            valid_mk = (idx - k >= 0) & (idx - k < n)
            grad[valid_pk] += (1.0 - h[idx[valid_pk] + k])
            grad[valid_mk] -= h[idx[valid_mk] - k]
        grad /= len(top_idx)
        grad -= np.mean(grad)  # maintain sum
        
        # Find best pair to update
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i, best_j, best_delta = -1, -1, 0.0
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Compute the effect on correlation
                g = np.zeros(n_corr)
                vp = (i + shifts >= 0) & (i + shifts < n)
                g[vp] += h[i + shifts[vp]]
                vm = (i - shifts >= 0) & (i - shifts < n)
                g[vm] += h[i - shifts[vm]]
                vp2 = (j + shifts >= 0) & (j + shifts < n)
                g[vp2] -= h[j + shifts[vp2]]
                vm2 = (j - shifts >= 0) & (j - shifts < n)
                g[vm2] -= h[j - shifts[vm2]]
                
                # Binary search for best delta
                a_d, b_d = delta_lo, delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i, best_j, best_delta = i, j, delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

    # Phase 1: SLSQP on symmetric variables with increasing alphas
    if tleft() > 100:

                         ftol=1e-18, time_limit=min(200, tleft()))

            return np.ones(_gn)

            best_h = h_cd
