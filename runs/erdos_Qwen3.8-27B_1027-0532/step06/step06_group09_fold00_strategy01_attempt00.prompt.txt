```python
def momentum_subgradient(h, n, target, time_limit, max_steps=20000,

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

### Objective
Minimize C5 = max_{k} ∫ h(x)(1-h(x+k)) dx over [0,2]
- Discretized as: C5 = (1/2) * max_{lag} |∑ h_i * (1-h_{(i+k)%n})|

### Variables
- h: Piecewise constant function represented on n equally spaced points in [0,2]
- Lag index k ∈ {-(n-1), -(n-2)..., n-1} (allowing any shift)

### Constraints
- 0 ≤ h[i] ≤ 1 for all i
- ∑_{i=0}^{n-1} h[i] = n/2 exactly

```python
def project_feasible(h, target):
    h_curr = np.clip(h.astype(float), 0.0, 1.0)
    s_curr = np.sum(h_curr)
    diff = s_curr - target
    
    if abs(diff) < 1e-10:
        return h_curr
    
    # First pass: adjust interior points
    direction = 1 if diff > 0 else -1
    
    # Strategy: shift adjustment to least-recently-accessed areas
    idx_order = np.argsort(np.arange(len(h)))  # This provides a fixed order per run
    
    # Gradual adjustment
    h_adjusted = h_curr.copy()
    step = 1.0 / abs(diff)
    step = min(1e-4 * len(h), step)
    
    # First adjustment by brute-force
    idx = idx_order[:int(abs(diff))]
    h_adjusted[idx] += step * direction * idx.shape[0]
    h_adjusted = np.clip(h_adjusted, 0.0, 1.0)
    s_adjusted = np.sum(h_adjusted)
    residual = s_adjusted - target
    
    if abs(residual) > 1e-10:
        # Second adjustment for residual error
        direction2 = 1 if residual > 0 else -1
        idx2 = idx_order[10:10+int(abs(residual))]
        h_adjusted[idx2] += abs(residual) * direction2
        h_adjusted = np.clip(h_adjusted, 0.0, 1.0)
        s_adjusted = np.sum(h_adjusted)
        residual2 = abs(s_adjusted - target)
        
        # Verification: if adjustment failed, use random patch
        if residual2 > 1e-5:
            mask = np.random.choice(len(h), min(int(residual2*len(h))/2, len(h)), replace=False)
            h_adjusted[mask] = np.random.uniform(target/len(h), 1-target/len(h), size=mask.shape)
    
    return h_adjusted
```

```python
    current_score = compute_c5(h_feas)
    
    # Evaluate gradients for multiple Lag regions
    grads = []
    for lag in range(8):  # Focus on 8 most relevant Lag regions
        mid = len(h_feas)//2
        window = np.arange(mid-lag, mid+lag+1)
        if len(window)>=2 and lag>=1:
            grad = 0.0
            # Simplified gradient estimation
            for i in window:
                grad += h_feas[i] * (1.0 - 2.0*h_feas[i]) * (dx) 
            grads.append(grad/len(window))
    total_grad = np.mean(grads) * dx
    
    # Negative gradient direction
    new_h = h_feas - alpha * total_grad
    new_h = project_feasible(new_h, target)
    
    # Perform line search
    best_score = float('inf')
    best_alpha_dir = None
    alpha_values = [0.1, 0.01, 0.001, 0.0001, 0.00001]
    
    for alpha_dir in alpha_values:
        candidate = h_feas - alpha_dir * total_grad
        candidate = project_feasible(candidate, target)
        score_candidate = compute_c5(candidate)
        if score_candidate < best_score:
            best_score = score_candidate
            best_alpha_dir = alpha_dir
        if best_score < current_score - 1e-6:
            break
    
    if best_alpha_dir is not None:
        new_h = h_feas - best_alpha_dir * total_grad
        new_h = project_feasible(new_h, target)
        if compute_c5(new_h) < current_score + 1e-6:
            h_cur = new_h
            
    # Line search determines if descent occurred
    return h_cur, compute_c5(h_cur)
```
