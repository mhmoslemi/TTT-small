def smooth_grad_descent(h_init, n, target, time_limit, alpha_start=200, alpha_end=5000, n_alphas=6, max_steps=500):
    start = time.time()
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        lr = 1.0 / alpha * 0.5  # conservative step size
        g_prev = np.zeros(n)
        
        for step in range(max_steps):
            if time.time() - start > time_limit:
                break
            
            # Compute smooth gradient
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
            weights = np.exp(shifted)
            weights = weights / np.sum(weights)
            
            # Only use significant weights
            thresh = 1e-10
            active = np.where(weights > thresh)[0]
            
            grad = np.zeros(n)
            for s in active:
                k = int(s) - mid
                w = weights[int(s)]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad *= dx
            grad -= np.mean(grad)  # handle equality constraint
            
            # Nesterov update
            v = 0.5 * v + grad  # momentum coefficient 0.5
            h_new = project_feasible(h - lr * v, target)
            
            # Check improvement

            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            g_prev = grad.copy()
            
            # Early stopping if gradient is small
            if np.linalg.norm(grad) < 1e-10:
                break
    
    return best_h, best_c5

Wait, f(δ) = max_s (corr[s] + δ * effect[s]). For small δ > 0:

If max_s{effect[s] : corr[s] = M} < 0, then f decreases. The rate is this max.

The new max is reached when some inactive constraint catches up: corr[s'] + δ*effect[s'] = M + δ*max_active_effect

- Or the maximum possible δ (bounded by h[i] and 1-h[j])

    inactive = ~active
    if not np.any(inactive):
        return max_delta
    
    # For inactive s: corr[s] + delta * effect[s] = M + delta * max_active_eff
    # delta * (effect[s] - max_active_eff) = M - corr[s]
    # delta = (M - corr[s]) / (effect[s] - max_active_eff)

    valid = denom > 1e-14
    if not np.any(valid):
        return max_delta
    
    inactive_idx = np.where(inactive)[0]
    deltas = (M - corr[inactive_idx[valid]]) / denom[valid]
    min_delta = np.min(deltas)

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def fast_optimal_delta(corr, effect, max_delta):
    M = float(np.max(corr))
    active = corr >= M - 1e-14
    max_active_g = float(np.max(effect[active]))
    if max_active_g >= -1e-15:
        return 0.0
    inactive = ~active
    if not np.any(inactive):
        return max_delta
    denom = effect[inactive] - max_active_g
    valid = denom > 1e-14
    if not np.any(valid):
        return max_delta
    inactive_idx = np.where(inactive)[0]
    deltas = (M - corr[inactive_idx[valid]]) / denom[valid]
    min_delta = float(np.min(deltas))
    return min(max(0.0, min_delta), max_delta)


def compute_pair_effect(h, i, j, n):
    n_corr = 2 * n - 1
    k_vals = np.arange(-(n - 1), n)
    effect = np.zeros(n_corr)
    ip = (i + k_vals >= 0) & (i + k_vals < n)
    im = (i - k_vals >= 0) & (i - k_vals < n)
    jp = (j + k_vals >= 0) & (j + k_vals < n)
    jm = (j - k_vals >= 0) & (j - k_vals < n)
    effect[ip] -= (1.0 - h[i + k_vals[ip]])
    effect[im] += h[i - k_vals[im]]
    effect[jp] += (1.0 - h[j + k_vals[jp]])
    effect[jm] -= h[j - k_vals[jm]]
    return effect


def pair_cd_fast(h, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    k_vals = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 120:
            break

        M = float(np.max(corr))
        active = np.where(corr >= M - 1e-14)[0]

        grad = np.zeros(n)
        for s in active[:min(len(active), 8)]:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            if h[i] < 1e-12:
                continue
            for j in low_set:
                if i == j:
                    continue
                if h[j] > 1.0 - 1e-12:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue

                effect = compute_pair_effect(h, i, j, n)
                delta_opt = fast_optimal_delta(corr, effect, delta_max)
                if delta_opt <= 1e-14:
                    continue

                new_max = float(np.max(corr + delta_opt * effect)) * dx
                if new_max < best_trial - 1e-15:
                    best_trial = new_max
                    best_i = i
                    best_j = j
                    best_delta = delta_opt

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 40:
                break

    return best_h, best_c5


def smooth_obj_slsqp(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun_slsqp(h):
    return np.sum(h) - _global_target


def cons_jac_slsqp(h):
    return np.ones(_global_n)


def slsqp_polish(h, n, target, alphas, maxiter_per_alpha):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun_slsqp, 'jac': cons_jac_slsqp}]
    h_cur = h.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_slsqp, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_smooth_fd(h_init, n, target, time_limit, alpha_start, alpha_end, n_alphas, max_steps_per_alpha):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        lr = 2.0 / (alpha + 10.0)
        t_count = 0
        for step in range(max_steps_per_alpha):
            if time.time() - start > time_limit:
                break
            t_count += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-300:
                break
            weights = weights / wsum

            thresh = 1e-12
            active = np.where(weights > thresh)[0]
            grad = np.zeros(n)
            for s in active:
                k = int(s) - mid
                w = weights[int(s)]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad *= dx
            grad -= np.mean(grad)

            beta = (1.0 + np.sqrt(1.0 + 4.0 * t_count * t_count)) / 2.0
            v = beta * v + (1.0 - beta) * grad
            h_try = project_feasible(h - lr * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
            h = h_try

            if np.linalg.norm(grad) < 1e-12:
                break

    return best_h, best_c5
