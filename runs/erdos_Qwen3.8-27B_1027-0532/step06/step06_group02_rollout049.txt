def compute_gradient_fft(h, alpha, corr_full):
    n = len(h)
    c_max = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - c_max))
    w = exp_c / exp_c.sum()  # length 2n-1
    
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    
    W_flipped = w[::-1]
    
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    omh_pad = np.zeros(N_fft)
    omh_pad[:n] = 1.0 - h
    W_pad = np.zeros(N_fft)
    W_pad[:2*n-1] = w
    Wf_pad = np.zeros(N_fft)
    Wf_pad[:2*n-1] = W_flipped
    
    h_fft = np.fft.fft(h_pad)
    omh_fft = np.fft.fft(omh_pad)
    W_fft = np.fft.fft(W_pad)
    Wf_fft = np.fft.fft(Wf_pad)
    
    conv_omh_W = np.fft.ifft(omh_fft * W_fft).real
    conv_h_Wf = np.fft.ifft(h_fft * Wf_fft).real
    
    offset = n - 1
    term1 = conv_omh_W[offset : offset + n]
    term2 = conv_h_Wf[offset : offset + n]
    
    grad = term1 - term2
    return grad

    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_gradient_fft(h, alpha, corr_full):
    n = len(h)
    c_max = corr_full.max()
    shifted = alpha * (corr_full - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    W_flipped = w[::-1]
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    omh_pad = np.zeros(N_fft)
    omh_pad[:n] = 1.0 - h
    W_pad = np.zeros(N_fft)
    W_pad[:2*n-1] = w
    Wf_pad = np.zeros(N_fft)
    Wf_pad[:2*n-1] = W_flipped
    h_fft = np.fft.fft(h_pad)
    omh_fft = np.fft.fft(omh_pad)
    W_fft = np.fft.fft(W_pad)
    Wf_fft = np.fft.fft(Wf_pad)
    conv_omh_W = np.fft.ifft(omh_fft * W_fft).real
    conv_h_Wf = np.fft.ifft(h_fft * Wf_fft).real
    offset = n - 1
    term1 = conv_omh_W[offset : offset + n]
    term2 = conv_h_Wf[offset : offset + n]
    return term1 - term2

def soft_obj(h, alpha, penalty):
    corr = full_corr_fast(h)
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted))) 
    obj = (c_max + log_sum) / alpha
    s = np.sum(h)
    obj += penalty * (s - h.shape[0] / 2.0) ** 2
    return obj

def soft_grad(h, alpha, penalty):
    corr = full_corr_fast(h)
    grad = compute_gradient_fft(h, alpha, corr)
    s = np.sum(h)
    grad += 2.0 * penalty * (s - h.shape[0] / 2.0)
    return grad

def lbfgsb_refine(h_start, n, target, alphas, maxiter_per_alpha, penalty, time_limit, start_time):
    h_cur = h_start.astype(float).copy()
    bounds = [(0.0, 1.0)] * n
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        pen = penalty
        for attempt in range(3):
            def obj_func(x):
                return soft_obj(x, alpha, pen)
            def grad_func(x):
                return soft_grad(x, alpha, pen)
            try:
                res = minimize(obj_func, h_cur, jac=grad_func, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14,
                                                       'gtol': 1e-10, 'maxfun': maxiter_per_alpha * 3})
                h_new = res.x
            except Exception:
                break
            h_new = np.clip(h_new, 0.0, 1.0)
            s = h_new.sum()
            if abs(s - target) > 0.5:
                pen *= 10.0
                h_cur = project_box_sum(h_new, target)
                continue
            h_cur = project_box_sum(h_new, target)
            break
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += (1.0 - h[idx[valid]])
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] -= h[idx[valid]]
        B = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += (1.0 - h[idx[valid]])
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] -= h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_targeted(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if it % 6 < 2:
            top_k = 8
            top_indices = np.argsort(full)[::-1][:top_k]
            found_move = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += (1.0 - h[idx[valid]])
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] -= h[idx[valid]]
                i_cands = ones[A[ones] <= 0.5]
                B = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += (1.0 - h[idx[valid]])
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] -= h[idx[valid]]
                j_cands = zeros[B[zeros] >= 0.5]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    found_move = True
                    break
            if not found_move:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        elif it % 6 == 2:
            blen = rng.randint(3, min(15, n // 10) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            continue
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def two_swap_move(h, full, i1, j1, i2, j2, k_arr):
    n = len(h)
    delta = compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
    new_full = full + delta
    return new_full, delta

def perturb_diverse(h, rng, n_pert):
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_box_sum(h_init, target_init)
    best_score = compute_c5(best_h)

    phase1_time = min(30, tleft() * 0.04)
    h_g, c5_g = fast_greedy(best_h, max_iters=50000, rng=rng,
                            time_limit=phase1_time, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    if tleft() > 100:
        alphas = [3, 8, 20, 50, 120, 300, 800, 2000, 5000, 15000]
        phase2_time = min(400, tleft() * 0.35)
        h_ref, sc_ref = lbfgsb_refine(best_h, n_init, target_init, alphas,
                                       maxiter_per_alpha=150, penalty=50.0,
                                       time_limit=phase2_time, start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    if tleft() > 80:
        h_bin = project_binary(best_h, int(target_init))
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()

    if tleft() > 100:
        phase3_time = min(250, tleft() * 0.22)
        n_sa = int(min(5000000, phase3_time * 50000))
        h_sa, c5_sa = sa_targeted(best_h, n_sa, 0.02, 0.00005, rng,
                                   time_limit=phase3_time, start_time=start_time, k_arr=k_arr)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 80:
        alphas2 = [50, 150, 500, 1500, 5000, 20000]
        phase4_time = min(200, tleft() * 0.18)
        h_ref2, sc_ref2 = lbfgsb_refine(best_h, n_init, target_init, alphas2,
                                         maxiter_per_alpha=200, penalty=100.0,
                                         time_limit=phase4_time, start_time=start_time)
        if sc_ref2 < best_score:
            best_score = sc_ref2
            best_h = h_ref2.copy()

    h_bin2 = project_binary(best_h, int(target_init))
    sc_bin2 = compute_c5(h_bin2)
    if sc_bin2 < best_score:
        best_score = sc_bin2
        best_h = h_bin2.copy()

    for restart in range(10):
        if tleft() < 60:
            break
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(8, 16))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 20)
        else:
            h_pert = perturb_diverse(best_h, rng, rng.randint(4, 8))
            h_pert = perturb_block(h_pert, rng, 10)
        allowed = min(40, tleft() * 0.06)
        n_sa_r = int(min(1500000, allowed * 40000))
        h_sr, c5_sr = sa_targeted(h_pert, n_sa_r, 0.012, 0.00005, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()
        if tleft() < 60:
            break
        alphas_r = [20, 80, 300, 1000, 4000]
        allowed2 = min(30, tleft() * 0.05)
        h_rr, sc_rr = lbfgsb_refine(h_sr, n_init, target_init, alphas_r,
                                     maxiter_per_alpha=100, penalty=100.0,
                                     time_limit=allowed2, start_time=start_time)
        if sc_rr < best_score:
            best_score = sc_rr
            best_h = h_rr.copy()

    if tleft() > 100:
        n_coarse = 150
        target_coarse = float(n_coarse) / 2.0
        k_arr_c = np.arange(2 * n_coarse - 1) - (n_coarse - 1)
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        h_coarse_bin = project_binary(h_coarse, int(target_coarse))
        allowed = min(20, tleft() * 0.08)
        h_cg, c5_cg = fast_greedy(h_coarse_bin, max_iters=10000, rng=rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr_c)
        alphas_c = [15, 50, 200, 800]
        allowed_c2 = min(20, tleft() * 0.06)
        h_cr, sc_cr = lbfgsb_refine(h_cg, n_coarse, target_coarse, alphas_c,
                                     maxiter_per_alpha=100, penalty=50.0,
                                     time_limit=allowed_c2, start_time=start_time)
        h_coarse_best = h_cr if sc_cr < c5_cg else h_cg
        h_fine = interpolate_to(h_coarse_best, n_init, target_init)
        h_fine_bin = project_binary(h_fine, int(target_init))
        allowed_f = min(20, tleft() * 0.06)
        h_fg, c5_fg = fast_greedy(h_fine_bin, max_iters=15000, rng=rng,
                                   time_limit=allowed_f, start_time=start_time, k_arr=k_arr)
        allowed_f2 = min(20, tleft() * 0.05)
        n_saf = int(min(600000, allowed_f2 * 30000))
        h_fs, c5_fs = sa_targeted(h_fg, n_saf, 0.01, 0.00005, rng,
                                   time_limit=allowed_f2, start_time=start_time, k_arr=k_arr)
        if c5_fs < best_score:
            best_score = c5_fs
            best_h = h_fs.copy()

    if tleft() > 50:
        alphas_final = [200, 800, 3000, 10000, 30000]
        phase_final_time = min(100, tleft() * 0.25)
        h_ref3, sc_ref3 = lbfgsb_refine(best_h, n_init, target_init, alphas_final,
                                         maxiter_per_alpha=300, penalty=200.0,
                                         time_limit=phase_final_time, start_time=start_time)
        if sc_ref3 < best_score:
            best_score = sc_ref3
            best_h = h_ref3.copy()

    if tleft() > 30:
        h_bin_final = project_binary(best_h, int(target_init))
        sc_final_bin = compute_c5(h_bin_final)
        if sc_final_bin < best_score:
            best_score = sc_final_bin
            best_h = h_bin_final.copy()
        n_sa_f = int(min(800000, max(30000, (tleft() - 10) * 40000)))
        h_sa_f, c5_f = sa_targeted(best_h, n_sa_f, 0.005, 0.00001, rng,
                                    time_limit=max(5.0, tleft() - 8),
                                    start_time=start_time, k_arr=k_arr)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    if tleft() > 15:
        alphas_ultimate = [5000, 20000, 80000]
        h_ref4, sc_ref4 = lbfgsb_refine(best_h, n_init, target_init, alphas_ultimate,
                                         maxiter_per_alpha=500, penalty=500.0,
                                         time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc_ref4 < best_score:
            best_score = sc_ref4
            best_h = h_ref4.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
