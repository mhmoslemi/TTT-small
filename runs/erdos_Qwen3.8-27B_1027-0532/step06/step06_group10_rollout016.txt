def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            return smooth_obj_val(x, alpha)
        def _grad(x):
            return smooth_obj_grad(x, alpha, n)
        def _con(x):
            return np.sum(x) - target
        def _con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active=20, max_rounds=60):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        c_max = np.max(corr)
        eps = 0.02 * c_max
        active = np.where(corr >= c_max - eps)[0]
        if len(active) > n_active:
            active = active[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            g_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g_k[v1] += (1.0 - h[idxs[v1] - k])
            g_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + g_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:

def perturb_search(h0, n, target, time_limit, n_trials=2000, sigma=0.05):

            h = h_pert
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbbfgs_smooth(h0, n, target, alphas, maxiter_per_alpha, time_limit, start_time):

            h_new = project_feasible(res.x, target)

    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_lp, c5_lp = lp_refine(best_h, n_use, target_use,
                                 time_limit=min(50.0, tleft() * 0.4),
                                 n_active=30, max_rounds=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_p, s_p = slsqp_round(best_h, n_use, target_use,
                                 [5000, 20000, 100000, 500000],
                                 2000, 1e-15,
                                 min(30.0, tleft() * 0.6), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = gradient_line_search_cd(best_h, n_use, target_use,
                                               time_limit=min(25.0, tleft() * 0.7),
                                               max_sweeps=20000, n_top=5, n_cand=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_p, s_p = perturb_search(best_h, n_use, target_use,
                                   time_limit=min(15.0, tleft() * 0.7),
                                   n_trials=5000, sigma=0.02)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
