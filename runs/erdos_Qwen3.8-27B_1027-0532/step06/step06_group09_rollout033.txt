```python
def solve_lp_for_h(v, n):
    """Given v = 1-h, solve the LP to find optimal h."""
    # corr_k = sum_i h[i] * v[i+k-(n-1)] for k = 0, ..., 2n-2
    # We want: sum_i h[i] * v_pad[i+k] <= t for all k
    # where v_pad is v zero-padded appropriately
    
    # The correlation at lag k (shift = k - (n-1)):
    # For shift s = k - (n-1), the valid i are those where both i and i+s are in [0, n-1]
    # corr[s] = sum_{i=max(0,-s)}^{min(n-1, n-1-s)} h[i] * v[i+s]
    
    # Let's build the constraint matrix A where A[k, i] = v[i + s_k] if valid, else 0
    # s_k = k - (n-1) for k = 0, ..., 2n-2
    
    # Actually, let's think of it differently:
    # For each shift s in [-(n-1), n-1]:
    #   corr[s] = sum_i h[i] * v[i+s] (over valid i, where v[i+s] = 0 if i+s out of [0,n-1])
    
    # So the constraint is: sum_i h[i] * v[i+s] <= t for each s
    
    # Build the matrix: for each s, the row is [v[0+s], v[1+s], ..., v[n-1+s]] (with 0 for out-of-bounds)
    
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    
    # A matrix: n_corr x n
    A = np.zeros((n_corr, n))
    for row, s in enumerate(shifts):
        # corr[s] = sum_i h[i] * v[i+s] for valid i
        # i+s must be in [0, n-1], so i in [max(0, -s), min(n-1, n-1-s)]
        i_start = max(0, -s)
        i_end = min(n, n - s)  # exclusive
        A[row, i_start:i_end] = v[i_start+s:i_end+s]
    
    # Objective: minimize t
    # Variables: [h[0], ..., h[n-1], t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A @ h - t <= 0, i.e., A @ h + [-1]*t <= 0
    A_ineq = np.hstack([A, -np.ones((n_corr, 1))])
    b_ineq = np.zeros(n_corr)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        h_opt = res.x[:n]
        t_opt = res.x[n]
        return h_opt, t_opt
    else:
        return None, None
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n:
        N *= 2
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(a) * np.conj(np.fft.fft(b))))
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_for_h(v, n):
    """Solve the LP: min t s.t. corr_k(h; v) ≤ t, 0≤h≤1, sum(h)=n/2"""
    shifts = np.arange(-(n-1), n)
    n_corr = 2 * n - 1
    
    A = np.zeros((n_corr, n))
    for row, s in enumerate(shifts):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_end > i_start:
            A[row, i_start:i_end] = v[i_start+s:i_end+s]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ineq = np.hstack([A, -np.ones((n_corr, 1))])
    b_ineq = np.zeros(n_corr)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def alternating_lp(h_init, n, target, time_limit, max_iter=200, damping=1.0):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        v = 1.0 - h
        h_lp = solve_lp_for_h(v, n)
        
        if h_lp is None:
            break
        
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Check convergence
        if abs(c5_new - compute_c5(h)) < 1e-12:
            break
