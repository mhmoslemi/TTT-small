import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, mu, target):
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    overlaps = compute_overlaps(h)
    c_max = np.max(overlaps)
    shifted = alpha * (overlaps - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    total_exp = np.sum(exp_shifted)
    weights = exp_shifted / total_exp
    val = c_max + np.log(total_exp) / alpha + 0.5 * mu * (np.sum(h) - target) ** 2
    
    # Gradient
    grad = np.full(n, dx)  # from the "1" in (1 - h[...])
    
    # Subtract weighted h terms
    # For each overlap k with nonzero weight:
    # A_k[j] = h_valid(j + s_k) where s_k = k - (n-1)
    # B_k[j] = h_valid(j - s_k)
    active_mask = weights > 1e-12
    active_indices = np.where(active_mask)[0]
    
    for idx in active_indices:
        w_k = weights[idx]
        s = idx - n_half
        if s == 0:
            grad -= w_k * h * dx
            grad -= w_k * h * dx
        elif s > 0:
            grad[:n-s] -= w_k * h[s:] * dx
            grad[s:] -= w_k * h[:n-s] * dx
        else:
            m = -s
            grad[m:] -= w_k * h[:n-m] * dx
            grad[:n-m] -= w_k * h[m:] * dx
    
    # Penalty gradient
    grad += mu * (np.sum(h) - target)
    
    return val, grad

def lbfgsb_solve(h_start, n, target, alpha, mu, time_limit, maxiter=5000):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[0]
    
    def grad(x):
        return smooth_obj_and_grad(x, alpha, mu, target)[1]
    
    def callback(xk):
        return time.time() - start > time_limit
    
    try:
        res = minimize(fun, h_start, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter,
                                              'maxfun': 50000,
                                              'ftol': 1e-14,
                                              'gtol': 1e-10,
                                              'callback': callback})
    except Exception:
        return h_start.copy()
    
    h_new = project_feasible(res.x, target)
    return h_new

def binary_local_search(h, n, target, max_sweeps=100, time_limit=60):
    start = time.time()
    h = h.copy()
    overlaps = compute_overlaps(h)
    current_max = np.max(overlaps)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        # Try setting some 0s to 1s and some 1s to 0s
        zeros = np.where(h < 0.01)[0]
        ones = np.where(h > 0.99)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Random subset for speed
        if len(zeros) > 50:
            zeros_try = rng_choice(zeros, 30, start + sweep)
        else:
            zeros_try = zeros
        if len(ones) > 50:
            ones_try = rng_choice(ones, 30, start + sweep)
        else:
            ones_try = ones
        
        for j in zeros_try:
            for j2 in ones_try:
                # Try h[j] = 1, h[j2] = 0
                h[j] = 1.0
                h[j2] = 0.0
                new_overlaps = compute_overlaps(h)
                new_max = np.max(new_overlaps)
                if new_max < current_max - 1e-12:
                    current_max = new_max
                    overlaps = new_overlaps
                    improved = True
                else:
                    h[j] = 0.0
                    h[j2] = 1.0
        
        if not improved:
            break
    
    return h, current_max

    n_half = n - 1
    overlaps = compute_overlaps(h)
    c_max = float(np.max(overlaps))

    exp_shifted = np.exp(shifted)
    total_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / total_exp
    val = c_max + np.log(total_exp) / alpha + 0.5 * mu * (np.sum(h) - target) ** 2

    grad = np.full(n, dx * total_exp / total_exp)
    grad = np.full(n, dx)

    active_mask = weights > 1e-12
    active_idx = np.where(active_mask)[0]

    for idx in active_idx:

        s = idx - n_half

                grad[m:] -= w_k * h[:n - m] * dx
            grad[:n - m] -= w_k * h[m:] * dx

    grad += mu * (np.sum(h) - target)
    return val, grad


def lbfgsb_optimize(h_start, n, target, alpha, mu, time_limit, maxiter=10000):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return _smooth_obj_grad(x, alpha, mu, target)[0]

    def fprime(x):
        return _smooth_obj_grad(x, alpha, mu, target)[1]

    def cbk(xk):
        return (time.time() - t0) > time_limit

    try:
        res = minimize(fun, h_start, jac=fprime, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'maxfun': 100000,
                                'ftol': 1e-16, 'gtol': 1e-12,
                                'callback': cbk, 'disp': False})
    except Exception:
        return h_start.copy()
    if res.x is None:
        return h_start.copy()
    return project_feasible(res.x, target)


def random_binary_init(n, target, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)

def coord_descent_binary(h, n, target, max_sweeps=50, time_limit=20):

    h = h.copy()
    overlaps = compute_overlaps(h)
    cur_max = float(np.max(overlaps))

        improved = False
