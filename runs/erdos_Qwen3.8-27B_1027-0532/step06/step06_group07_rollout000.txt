    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 60:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = grad_for_lag(h, k_star, n, idxs)
        grad -= np.mean(grad)

        active_mask = corr >= np.max(corr) - 0.1 * np.max(corr)
        grad_weighted = grad.copy()
        for s_idx in np.where(active_mask)[0][:min(8, len(np.where(active_mask)[0]))]:
            k = int(s_idx) - mid
            grad_weighted += 0.3 * grad_for_lag(h, k, n, idxs)
        grad_weighted -= np.mean(grad_weighted)

        sorted_idx = np.argsort(grad_weighted)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 20:
                break

    return best_h, best_c5


def gradient_direction_cd(h, n, target, time_limit, max_sweeps=3000, n_active=5):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        max_corr = np.max(corr)
        active_lags = np.argsort(corr)[-n_active:]

        grad = np.zeros(n)
        for s_idx in active_lags:
            k = int(s_idx) - mid
            grad += grad_for_lag(h, k, n, idxs)
        grad /= n_active
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_k = sorted_idx
        high_k = sorted_idx[::-1]

        best_trial_c5 = c5_val
        best_move = None

        for trial in range(20):
            i = int(high_k[trial]) if trial < len(high_k) else -1
            j = int(low_k[trial]) if trial < len(low_k) else -1
            if i < 0 or j < 0 or i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            n_corr_local = 2 * n - 1
            shifts = np.arange(-(n-1), n)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g = np.zeros(n_corr_local)
            g[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g[v_im] += h[i - shifts[v_im]]
            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            break

    return best_h, best_c5


def strategic_perturb_and_restart(h_init, n, target, time_limit, seed_val=12345):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)

    for restart in range(30):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.02 * max_corr)[0]

        n_active_pts = min(len(active) * 5, n // 2)
        perturb_idx = rng.choice(n, size=n_active_pts, replace=False)
        scale = rng.choice([0.02, 0.04, 0.06, 0.08])
        h_try = h.copy()
        h_try[perturb_idx] += rng.randn(n_active_pts) * scale
        h_try = project_feasible(h_try, target)

        cd_time = min(30, (time_limit - (time.time() - start)) * 0.3)
        if cd_time < 5:
            break
        h_new, c5_new = fast_pair_cd(h_try, n, target,
                                     time_limit=cd_time,
                                     max_sweeps=2000, n_cand=30, n_bisect=12)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    return best_h, best_c5


def iterative_refinement(h_init, n, target, time_limit):
    start = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)

    for iteration in range(50):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx if (dx := 2.0 / n) else float(np.max(corr)) * 2.0 / n
        c5_val = float(np.max(corr)) * 2.0 / n
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.03 * max_corr)[0]

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - (n - 1)
            grad += grad_for_lag(h, k, n, idxs)
        grad /= max(1, len(active))
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        step = 0.01 / grad_norm
        for trial in range(10):
            h_try = project_feasible(h - step * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val:
                h = h_try
                break
            step *= 0.5

    return best_h, best_c5


def interpolation_based_search(h_init, n_src, target_src, time_limit):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    n_targets = [int(n_src * 1.2), int(n_src * 1.4), int(n_src * 1.6)]
    for n_tgt in n_targets:
        if time.time() - start > time_limit:
            break
        if n_tgt < 100:
            continue
        target_tgt = n_tgt / 2.0
        x_old = np.linspace(0, 1, n_src)
        x_new = np.linspace(0, 1, n_tgt)
        h_interp = np.interp(x_new, x_old, h_init)
        h_interp = project_feasible(h_interp, target_tgt)

        h_opt = slsqp_round(h_interp, n_tgt, target_tgt,
                            [2000, 8000, 30000], 600, 1e-15)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            cd_t = min(60, (time_limit - (time.time() - start)) * 0.4)
            if cd_t > 10:
                h_opt2, sc2 = fast_pair_cd(h_opt, n_tgt, target_tgt,
                                           time_limit=cd_t,
                                           max_sweeps=2000, n_cand=40, n_bisect=13)
                if sc2 < best_c5:
                    best_c5 = sc2
                    best_h = h_opt2
                elif sc < best_c5:
                    best_c5 = sc
                    best_h = h_opt

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [500, 2000], 500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=4000, n_cand=40, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        grd_time = min(80, tleft() * 0.2)
        h_grd, c5_grd = gradient_direction_cd(best_h, n, target,
                                              time_limit=grd_time,
                                              max_sweeps=2000, n_active=5)
        if c5_grd < best_score:
            best_score = c5_grd
            best_h = h_grd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        cd_time = min(200, tleft() * 0.35)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=5000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_ir, c5_ir = iterative_refinement(best_h, n, target,
                                           time_limit=min(60, tleft() * 0.3))
        if c5_ir < best_score:
            best_score = c5_ir
            best_h = h_ir

    if tleft() > 80:
        sp_time = min(100, tleft() * 0.3)
        h_sp, c5_sp = strategic_perturb_and_restart(best_h, n, target,
                                                     time_limit=sp_time,
                                                     seed_val=seed)
        if c5_sp < best_score:
            best_score = c5_sp
            best_h = h_sp

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [20000, 80000], 800, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        cd_time = min(100, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=4000, n_cand=60, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        h_interp, c5_interp = interpolation_based_search(best_h, n, target,
                                                         time_limit=min(100, tleft() * 0.5))
        if c5_interp < best_score:
            best_score = c5_interp
            best_h = h_interp.copy()

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(30, tleft() * 0.5)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=2000, n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [50000, 200000], 500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
