    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_actual = compute_c5(h)
    total_time = time_budget
    
    for alpha_idx, alpha in enumerate(alphas):
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        # Adaptive learning rate
        cur_lr = lr
        
        while True:
            elapsed = time.time() - start_time
            if elapsed > start_time + total_time:
                break
            
            val, grad = smooth_obj_grad_fft(h, alpha, nfft)
            
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            
            if stall > 500:
                break
            if t > 50000:
                break
            
            t += 1
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            h = project_box_sum(h_new, target)

            if stall > 200:
                cur_lr *= 0.5

        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()

            best_score = s_a2
            best_h = h_a2.copy()

    # Phase 4: Multi-restart from perturbed points
    if tleft() > 150:
        n_restarts = 6
        for ri in range(n_restarts):
            if tleft() < 50:
                break
            jitter = 0.003 + 0.004 * ri
            h_pert = best_h + rng.randn(n) * jitter
            h_pert = project_box_sum(h_pert, target)
            rt = min(80, tleft() * 0.4)
            h_r, s_r = adam_fft(h_pert, target, start_time, rt,
                                alphas=[500, 2000, 10000, 50000], lr=0.002, nfft=nfft)
            if s_r < best_score:
                best_score = s_r
                best_h = h_r.copy()

    # Phase 5: L-BFGS-B refinement with very high alpha
    if tleft() > 80:
        h_f, s_f = lbfgs_run(best_h, target, 100000, 500000.0, nfft,
                              1000, start_time, start_time + min(80, tleft() * 0.5))
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Phase 6: Binary local search
    if tleft() > 60:
        h_b, s_b = binary_local_search(best_h, target_int, start_time,
                                       start_time + min(60, tleft() * 0.5), max_rounds=100)
        if s_b < best_score:
            best_score = s_b
            best_h = h_b.copy()

    # Phase 7: Final Adam polish
    if tleft() > 30:
        h_p, s_p = adam_fft(best_h, target, start_time, min(40, tleft() * 0.6),
                            alphas=[50000, 200000], lr=0.0005, nfft=nfft,
                            max_iters_per_alpha=50000, stall_limit=800)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
