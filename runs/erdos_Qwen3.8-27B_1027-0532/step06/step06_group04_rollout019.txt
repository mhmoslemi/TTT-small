def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):

def subgradient_bb(h, n, target, max_steps, time_limit):

        k = k_star

        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)

def lp_solve(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps):

    start = time.time()

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    
    corr = compute_corr(h)
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False

            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)

            break

def smooth_obj(h, alpha, n, target):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, n, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

```python
def compute_all_slopes(h, n):
    """For each i, compute slopes[i, k_idx] = contribution of h[i] to corr[k_idx].
    This is a (n, 2n-1) matrix.
    slopes[i, k_idx] = ((1-h[i+k] if valid) - (h[i-k] if valid)) * dx
    """
    dx = 2.0 / n
    half_n = n - 1
    n_corr = 2 * n - 1
    slopes = np.zeros((n, n_corr))
    
    idxs = np.arange(n)
    # For all i and k:
    # i+k valid: 0 <= i+k < n
    # i-k valid: 0 <= i-k < n
    
    # Create meshgrid
    I, K = np.meshgrid(idxs, idxs, indexing='ij')  # I: (n, n), K: (n, n)
    
    # Forward part: k >= 0
    # i+k < n: K_fwd < n - I
    # k ranges from 0 to n-1
    K_fwd = idxs  # 0, 1, ..., n-1
    valid_fwd = (I + K_fwd[None, :] < n)
    # slopes[i, (n-1) + k] += (1 - h[i+k]) * dx for valid
    # k_idx = (n-1) + k
    for k in range(n):
        mask = (np.arange(n) + k < n)
        k_idx = half_n + k
        slopes[mask, k_idx] += (1.0 - h[np.arange(n)[mask] + k]) * dx
    
    # Backward part: k < 0
    # k = -1, -2, ..., -(n-1)
    for k in range(1, n):
        # i - k >= 0: i >= k
        mask = (np.arange(n) >= k)
        k_idx = half_n - k
        slopes[mask, k_idx] -= h[np.arange(n)[mask] - k] * dx
    
    return slopes
```

This is O(n²) to precompute, and then the coordinate descent inner loop is just looking up slopes[i, :].

But memory: n * (2n-1) = 400 * 799 ≈ 320,000 floats = 2.5 MB. That's fine.

```python

```

This is O(n) per bisection iteration, O(n * 35) per coordinate, O(n² * 35) per pass.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
