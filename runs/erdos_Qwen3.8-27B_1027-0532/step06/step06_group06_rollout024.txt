def swap_delta(i, j, h, n, off):
    delta = np.zeros(2 * n - 1)
    ci_lo = max(0, off - i)
    ci_hi = min(2 * n - 2, off + n - 1 - i)
    if ci_lo <= ci_hi:
        delta[ci_lo:ci_hi + 1] += h[i + ci_lo - off:i + ci_hi - off + 1]
    ci_lo2 = max(0, off + i - n + 1)
    ci_hi2 = min(2 * n - 2, off + i)
    if ci_lo2 <= ci_hi2:
        idx = np.arange(ci_lo2, ci_hi2 + 1)
        h_idx = i - idx + off
        delta[ci_lo2:ci_hi2 + 1] -= (1.0 - h[h_idx])
    ci_lo3 = max(0, off - j)
    ci_hi3 = min(2 * n - 2, off + n - 1 - j)
    if ci_lo3 <= ci_hi3:
        delta[ci_lo3:ci_hi3 + 1] -= h[j + ci_lo3 - off:j + ci_hi3 - off + 1]

        delta[ci_lo4:ci_hi4 + 1] += (1.0 - h[h_idx])

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)

    for j in range(N):
        idx1 = j - t_arr + N - 1
        mask1 = (idx1 >= 0) & (idx1 < 2 * N - 1)
        idx2 = t_arr - j + N - 1
        mask2 = (idx2 >= 0) & (idx2 < 2 * N - 1)
        g1 = np.dot(one_minus_h, weights[idx1][mask1])
        g2 = np.dot(h, weights[idx2][mask2])
        grad[j] = g1 - g2

    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_continuous(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()

    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

        h_ls, c5_ls = aggressive_local_search(best_h, target_int, time_limit=time_alloc, start_time=start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 2: 2-swap local search
    if tleft() > 60:
        time_alloc = min(80, tleft() * 0.15)
        h_2s, c5_2s = two_swap_local_search(best_h, target_int, time_limit=time_alloc, start_time=start_time)
        if c5_2s < best_c5:
            best_c5 = c5_2s
            best_h = h_2s.copy()

    # Phase 3: SA with informed moves
    if tleft() > 80:
        time_alloc = min(120, tleft() * 0.2)
        max_it = int(min(300000, time_alloc * 3000))
        h_sa, c5_sa = sa_binary(best_h, target_int, max_it, 0.006, 0.00001, rng,
                                time_limit=time_alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Another round of local search after SA
    if tleft() > 60:
        time_alloc = min(80, tleft() * 0.15)
        h_ls2, c5_ls2 = aggressive_local_search(best_h, target_int, time_limit=time_alloc, start_time=start_time)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 5: Continuous refinement
    if tleft() > 80:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [50, 150, 400, 1000, 2000]:
            if tleft() < 30:
                break
            time_alloc = min(35, tleft() * 0.15)
            h_opt = lbfgs_continuous(h_cont, beta, target, penalty=800.0,
                                      maxiter=200, time_limit=time_alloc, start_time=start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 6: Grid size exploration
    n_current = len(best_h)
    n_options = []
    for delta in [50, 100, 150, 200, 300, 400, 500, -100, -200]:
        n_try = n_current + delta
        if n_try >= 300 and n_try % 2 == 0 and n_try != n_current:
            n_options.append(n_try)
    n_options = list(set(n_options))
    n_options.sort(key=lambda x: abs(x - n_current))

    for n_try in n_options[:8]:
        if tleft() < 80:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)

        time_alloc = min(40, tleft() * 0.2)
        h_loc, c5_loc = aggressive_local_search(h_interp, target_try_int,
                                                 time_limit=time_alloc, start_time=start_time)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()

        if tleft() > 60:
            time_alloc2 = min(30, tleft() * 0.12)
            h_sa2, c5_sa2 = sa_binary(h_loc, target_try_int,
                                       int(min(100000, time_alloc2 * 2500)),
                                       0.005, 0.00001, rng,
                                       time_limit=time_alloc2, start_time=start_time)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Phase 7: Final SA polish on best solution
    if tleft() > 50:
        n_use = len(best_h)
        if n_use % 2 == 0:
            target_use = int(n_use / 2)
            h_bin_final = project_binary(best_h, target_use)
            time_alloc = min(80, tleft() - 10)
            max_it = int(min(200000, time_alloc * 2500))
            h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.003, 0.000005,
                                     rng, time_limit=time_alloc, start_time=start_time)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Phase 8: Final local search polish
    if tleft() > 40:
        n_use = len(best_h)
        if n_use % 2 == 0:
            target_use = int(n_use / 2)
            time_alloc = min(50, tleft() - 5)
            h_final_ls, c5_final = aggressive_local_search(best_h, target_use,
                                                            time_limit=time_alloc, start_time=start_time)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final_ls.copy()

    # Phase 9: Final continuous polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_fc = project_box_sum(best_h.copy(), target_use)
        for beta_f in [500, 1500, 3000]:
            if tleft() < 15:
                break
            time_alloc = min(20, tleft() * 0.4)
            h_pol = lbfgs_continuous(h_fc, beta_f, target_use, penalty=1500.0,
                                      maxiter=150, time_limit=time_alloc, start_time=start_time)
            h_fc = h_pol
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
