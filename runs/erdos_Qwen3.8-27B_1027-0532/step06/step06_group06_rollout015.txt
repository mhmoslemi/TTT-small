import numpy as np

def fft_corr_max(h):
    """Compute max correlation using FFT. Returns (max_val, corr_array)."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(V), N)
    # The linear correlation values are in corr[0:n-1] and corr[N-n+1:N]
    # But for finding the max, we can just take max of the relevant portion
    # Actually, for non-negative signals, the max is in the correct region
    # Let's extract the full linear correlation
    linear_corr = np.concatenate([corr[N-n+1:], corr[:n]])  # length 2n-1
    return float(np.max(linear_corr)), linear_corr

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx
