                #   Actually: max(0, s) <= m <= min(n-1, s+n-1) = max(0,-abs_s) <= m <= min(n-1, n-1-abs_s)
                #   = 0 <= m <= n-1-abs_s
                # Second term: same range, m-s = m+abs_s in [abs_s, n-1], valid since n-1 < n
                # Third term: m+s = m-abs_s in [0, n-1], i.e., m in [abs_s, n-1]
                valid1 = slice(0, n - abs_s)
                valid2 = slice(abs_s, n)
                grad[0:n-abs_s] += (1.0 - h[abs_s:])  # (1-h[m+abs_s]) for m in [0, n-1-abs_s]
                grad[abs_s:n] += -h[:n-abs_s]  # -h[m-abs_s] for m in [abs_s, n-1]
        
        grad /= len(active)
        
        # Polyak step size
        f_curr = c_max * dx
        gap = f_curr / dx - f_star  # gap in overlap units
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-20:
            break
        step = gap / (grad_norm_sq + 1e-10)
        step = min(step, 0.5)  # cap step size
        
        h = h - step * grad
        h = project_box_sum(h, target)
        
        # Update average
        avg_h += h / (it + 1)
        
        # Periodically check the average
        if (it + 1) % 100 == 0:
            avg_test = project_box_sum(avg_h, target)
            avg_corr = np.correlate(avg_test, 1.0 - avg_test, mode='full')
            avg_c5 = float(avg_corr.max()) * dx
            if avg_c5 < best_c5:
                best_c5 = avg_c5
                best_h = avg_test.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Final check on average
    avg_test = project_box_sum(avg_h, target)
    avg_corr = np.correlate(avg_test, 1.0 - avg_test, mode='full')
    avg_c5 = float(avg_corr.max()) * dx
    if avg_c5 < best_c5:
        best_c5 = avg_c5
        best_h = avg_test.copy()
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def projected_grad_descent(h, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 5000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        
        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += (-h[abs_s:])
        grad /= topk
        
        h = h - lr * grad
        h = project_box_sum(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0).copy()
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break

            if can_inc > remaining:

                remaining = 0.0

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0).copy()

                remaining = 0.0

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    best_cont_h = project_box_sum(h_init.copy(), float(n_init) / 2.0)
    best_cont_score = compute_c5(best_cont_h)

    # Phase 1: Greedy
    allowed = min(35.0, tleft() * 0.06)
    if allowed > 10:
        h_g, c5_g = greedy_improve(best_h, rng, n_cand=50, time_limit=allowed,
                                    start_time=start_time)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: SA with multi-swap (main workhorse)
    allowed = min(200.0, tleft() * 0.28)
    if allowed > 30:
        n_sa = int(max(1000000, allowed * 50000))
        h_sa, c5_sa = sa_search(best_h, n_sa, 0.025, 0.00002, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: First continuous optimization
    allowed = min(100.0, tleft() * 0.14)
    if allowed > 20:
        n_it = int(max(200000, allowed * 20000))
        h_pg, c5_pg = projected_grad_descent(best_cont_h, n_it, topk=5,
                                              lr_init=0.06, lr_end=0.001,
                                              time_limit=allowed,
                                              start_time=start_time)
        if c5_pg < best_cont_score:
            best_cont_score = c5_pg
            best_cont_h = h_pg.copy()

    # Phase 4: Softmax gradient
    allowed = min(80.0, tleft() * 0.10)
    if allowed > 15:
        n_it = int(max(100000, allowed * 12000))
        h_sm, c5_sm = softmax_grad_descent(best_cont_h, n_it, alpha=40.0,
                                            lr_init=0.04, lr_end=0.0003,
                                            time_limit=allowed,
                                            start_time=start_time)
        if c5_sm < best_cont_score:
            best_cont_score = c5_sm
            best_cont_h = h_sm.copy()

    # Phase 5: SLSQP
    allowed = min(60.0, tleft() * 0.08)
    if allowed > 15:
        h_ref, sc_ref = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                      alphas=[80, 300, 1000, 3000],
                                      maxiter=3000, ftol=1e-14,
                                      time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 6: Iterated local search
    for iteration in range(10):
        if tleft() < 50:
            break

        # Binary SA with perturbation
        allowed = min(35.0, tleft() * 0.07)
        if allowed > 15:
            n_pert = rng.randint(2, 7)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            n_sa2 = int(max(500000, allowed * 45000))
            h_sa2, c5_sa2 = sa_search(h_pert, n_sa2, 0.015, 0.00001, rng,
                                       time_limit=allowed, start_time=start_time)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

        if tleft() < 35:
            break

        # Continuous from binary
        allowed = min(35.0, tleft() * 0.07)
        if allowed > 10:
            h_seed = project_box_sum(best_h, float(n_init) / 2.0)
            n_it = int(max(100000, allowed * 20000))
            h_pg2, c5_pg2 = projected_grad_descent(h_seed, n_it, topk=3,
                                                    lr_init=0.05, lr_end=0.0005,
                                                    time_limit=allowed,
                                                    start_time=start_time)
            if c5_pg2 < best_cont_score:
                best_cont_score = c5_pg2
                best_cont_h = h_pg2.copy()

        if tleft() < 25:
            break

        # SLSQP polish
        allowed = min(25.0, tleft() * 0.05)
        if allowed > 10:
            h_ref2, sc_ref2 = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                            alphas=[200, 800, 3000],
                                            maxiter=2000, ftol=1e-14,
                                            time_limit=allowed, start_time=start_time)
            if sc_ref2 < best_cont_score:
                best_cont_score = sc_ref2
                best_cont_h = h_ref2.copy()

    # Phase 7: Multi-resolution
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8]:
        if tleft() < 45:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, n_init)
        h_interp = np.interp(x_new, x_old, best_cont_h)
        h_interp = project_box_sum(h_interp, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(12.0, tleft() * 0.04)
        if allowed < 5:
            continue
        h_g3, c5_g3 = greedy_improve(h_bin_try, rng, n_cand=40,
                                      time_limit=allowed, start_time=start_time)

        allowed2 = min(18.0, tleft() * 0.05)
        if allowed2 < 8:
            continue
        n_it3 = int(max(50000, allowed2 * 20000))
        h_pg3, c5_pg3 = projected_grad_descent(h_g3, n_it3, topk=4,
                                                lr_init=0.05, lr_end=0.0005,
                                                time_limit=allowed2,
                                                start_time=start_time)
        if c5_pg3 < best_cont_score:
            best_cont_score = c5_pg3
            best_cont_h = h_pg3.copy()

    # Phase 8: Final SA burst on best binary
    if tleft() > 25:
        allowed = tleft() - 15
        if allowed > 15:
            n_sa_f = int(max(500000, allowed * 50000))
            h_sa_f, c5_sa_f = sa_search(best_h, n_sa_f, 0.01, 0.000005, rng,
                                         time_limit=allowed * 0.6, start_time=start_time)
            if c5_sa_f < best_score:
                best_score = c5_sa_f
                best_h = h_sa_f.copy()

    # Phase 9: Final continuous
    if tleft() > 15:
        n_use = len(best_cont_h)
        n_it_f = int(max(50000, (tleft() - 10) * 25000))
        h_pgf, c5_pgf = projected_grad_descent(best_cont_h, n_it_f, topk=3,
                                                lr_init=0.03, lr_end=0.0002,
                                                time_limit=max(5.0, tleft() - 8),
                                                start_time=start_time)
        if c5_pgf < best_cont_score:
            best_cont_score = c5_pgf
            best_cont_h = h_pgf.copy()

    # Final selection
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
