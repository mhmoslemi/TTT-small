        k_star = max_idx - (n - 1)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        a_k = ones - k_star
        b_k = ones + k_star
        s_a = np.zeros(len(ones))
        va = (a_k >= 0) & (a_k < n)
        vb = (b_k >= 0) & (b_k < n)
        s_a[va] += (1.0 - h[a_k[va]])
        s_a[vb] += h[b_k[vb]]
        best_a = int(ones[np.argmin(s_a)])

        c_k = zeros - k_star
        d_k = zeros + k_star
        s_b = np.zeros(len(zeros))
        vc = (c_k >= 0) & (c_k < n)
        vd = (d_k >= 0) & (d_k < n)
        s_b[vc] += (1.0 - h[c_k[vc]])
        s_b[vd] += h[d_k[vd]]
        best_b = int(zeros[np.argmax(s_b)])

        if best_a == best_b:
            break

        delta = compute_delta_corr(h, best_a, best_b, n)
        new_corr = corr + delta
        new_max = float(new_corr.max())

        if new_max < max_val - 1e-12:
            h[best_a] = 0.0
            h[best_b] = 1.0
            corr = new_corr
            sc = new_max * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
        else:
            break

    return best_h, best_c5

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    k = np.arange(-(N - 1), N)

    for j in range(N):
        v1 = (j - k >= 0) & (j - k < N)
        v2 = (j + k >= 0) & (j + k < N)
        grad[j] = np.dot(weights[v1], one_minus_h[j - k[v1]]) - np.dot(weights[v2], h[j + k[v2]])

    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_optimize(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()

    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            if h_candidate is not None:
                h_cur = h_candidate
        except Exception:
            break
    return project_box_sum(h_cur, target)

def interpolate_to(h_src, n_dst, target_dst):
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def coarse_to_fine(h_fine_init, n_fine, target_fine, time_limit, start_time, rng):
    n_coarse = max(100, n_fine // 6)
    if n_coarse % 2 != 0:
        n_coarse += 1
    target_coarse = float(n_coarse // 2)

    h_coarse = interpolate_to(h_fine_init, n_coarse, target_coarse)

    alphas = [5.0, 15.0, 40.0, 100.0]
    for alpha in alphas:
        if time.time() - start_time > time_limit * 0.5:
            break
        h_coarse = lbfgs_optimize(h_coarse, alpha, target_coarse, 200.0,
                                   maxiter=200, time_limit=time_limit * 0.3, start_time=start_time)

    h_up = interpolate_to(h_coarse, n_fine, target_fine)
    return h_up

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    target = float(n // 2)
    target_int = int(n // 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(best_h, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Continuous optimization (short, to get a good starting point)
    if tleft() > 80:
        h_cont = project_box_sum(h_init.copy(), target)
        for beta in [5.0, 10.0, 25.0]:
            if tleft() < 30:
                break
            alloc = min(25, tleft() * 0.15)
            h_cont = lbfgs_optimize(h_cont, beta, target, 500.0,
                                     maxiter=150, time_limit=alloc, start_time=start_time)
            sc = compute_c5(h_cont)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont.copy()

    # Phase 2: Long SA with fast updates (main workhorse)
    if tleft() > 100:
        h_bin = project_binary(best_h, target_int)
        alloc = min(350, tleft() * 0.35)
        max_it = int(alloc * 8000)
        h_sa, c5_sa = sa_binary_fast(h_bin, target_int, max_it, 0.003, 0.000005,
                                      rng, time_limit=alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Greedy on the result
    if tleft() > 60:
        h_bin = project_binary(best_h, target_int)
        alloc = min(40, tleft() * 0.15)
        h_g, c5_g = greedy_binary_fast(h_bin, target_int, time_limit=alloc, start_time=start_time, max_rounds=500)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 4: SA again from improved point (different temperature)
    if tleft() > 100:
        h_bin = project_binary(best_h, target_int)
        alloc = min(250, tleft() * 0.3)
        max_it = int(alloc * 8000)
        h_sa2, c5_sa2 = sa_binary_fast(h_bin, target_int, max_it, 0.0015, 0.000002,
                                        rng, time_limit=alloc, start_time=start_time, burn_in_frac=0.05)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Continuous refinement
    if tleft() > 80:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [30.0, 80.0, 200.0, 500.0]:
            if tleft() < 25:
                break
            alloc = min(30, tleft() * 0.2)
            h_cont = lbfgs_optimize(h_cont, beta, target, 1000.0,
                                     maxiter=200, time_limit=alloc, start_time=start_time)
            sc = compute_c5(h_cont)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont.copy()

    # Phase 6: Binary + SA from continuous result
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        h_g6, c5_6 = greedy_binary_fast(h_bin, target_int, time_limit=min(20, tleft() * 0.1),
                                         start_time=start_time, max_rounds=300)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h_g6.copy()

        if tleft() > 60:
            alloc = min(150, tleft() * 0.3)
            max_it = int(alloc * 8000)
            h_sa3, c5_sa3 = sa_binary_fast(h_bin, target_int, max_it, 0.001, 0.000001,
                                            rng, time_limit=alloc, start_time=start_time, burn_in_frac=0.03)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()

    # Phase 7: Hierarchical coarse-to-fine
    if tleft() > 120:
        h_coarse_init = project_box_sum(best_h.copy(), target)
        alloc = min(80, tleft() * 0.25)
        h_up = coarse_to_fine(h_coarse_init, n, target, time_limit=alloc, start_time=start_time, rng=rng)
        sc_up = compute_c5(h_up)
        if sc_up < best_c5:
            best_c5 = sc_up
            best_h = h_up.copy()

        # Refine on fine grid
        if tleft() > 40:
            h_cont7 = project_box_sum(h_up, target)
            for beta7 in [50.0, 150.0, 400.0]:
                if tleft() < 20:
                    break
                alloc7 = min(25, tleft() * 0.2)
                h_cont7 = lbfgs_optimize(h_cont7, beta7, target, 800.0,
                                          maxiter=150, time_limit=alloc7, start_time=start_time)
                sc7 = compute_c5(h_cont7)
                if sc7 < best_c5:
                    best_c5 = sc7
                    best_h = h_cont7.copy()

    # Phase 8: Multiple random restarts in binary space
    if tleft() > 100:
        for restart in range(3):
            if tleft() < 40:
                break
            # Perturb binary solution
            h_bin_r = best_h.copy()
            n_flip = int(n * 0.02)
            ones_r = np.where(h_bin_r > 0.5)[0]
            zeros_r = np.where(h_bin_r < 0.5)[0]
            flip_ones = rng.choice(ones_r, size=min(n_flip, len(ones_r)), replace=False)
            flip_zeros = rng.choice(zeros_r, size=min(n_flip, len(zeros_r)), replace=False)
            h_bin_r[flip_ones] = 0.0
            h_bin_r[flip_zeros] = 1.0
            h_bin_r = project_binary(h_bin_r, target_int)

            alloc_r = min(40, tleft() * 0.2)
            max_it_r = int(alloc_r * 8000)
            h_sr, c5_sr = sa_binary_fast(h_bin_r, target_int, max_it_r, 0.002, 0.00001,
                                          rng, time_limit=alloc_r, start_time=start_time)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()

    # Phase 9: Final greedy + SA polish
    if tleft() > 50:
        h_bin_final = project_binary(best_h, target_int)
        h_gf, c5_gf = greedy_binary_fast(h_bin_final, target_int, time_limit=min(15, tleft() * 0.2),
                                          start_time=start_time, max_rounds=1000)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

        if tleft() > 30:
            alloc_f = min(50, tleft() - 5)
            max_it_f = int(alloc_f * 8000)
            h_sf, c5_sf = sa_binary_fast(h_bin_final, target_int, max_it_f, 0.0005, 0.0000005,
                                          rng, time_limit=alloc_f, start_time=start_time, burn_in_frac=0.02)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final // 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    # If projection changed things, try binary too
    h_bin_check = project_binary(best_h, int(target_final))
    c5_check = compute_c5(h_bin_check)
    if c5_check < final_c5:
        best_h = h_bin_check
        final_c5 = c5_check

    return best_h, final_c5, n_final
