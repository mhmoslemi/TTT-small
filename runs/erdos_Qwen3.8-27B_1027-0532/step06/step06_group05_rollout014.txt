        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_binary(h, target):
    """Project to binary with exactly target ones"""
    h_b = (h >= 0.5).astype(float)
    cur = int(h_b.sum())
    if cur == target:
        return h_b
    if cur > target:
        need = cur - target
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_b[ones[order]] = 0
    else:
        need = target - cur
        zeros = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_b[zeros[order]] = 1
    return h_b

def smooth_obj_and_grad(h, alpha):
    """Compute log-sum-exp smooth max and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(exp_vals.sum())
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    grad = np.zeros(n)
    one_mh = 1.0 - h
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-10:
            continue
        l = m - (n - 1)
        if l >= 0:
            if n > l:
                grad[:n - l] += w * one_mh[l:]
            if l > 0:
                grad[l:] -= w * h[:n - l]
        else:
            p = -l
            if n > p:
                grad[p:] += w * one_mh[:n - p]
                grad[:n - p] -= w * h[p:]
    return value, grad
