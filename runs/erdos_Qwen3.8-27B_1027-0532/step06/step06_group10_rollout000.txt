def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_logsumexp(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)

    w = np.exp(shifted)

        w[int(np.argmax(corr))] = 1.0

        w = w / ws

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_logsumexp(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(corr, h, alpha, n):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, deadline):

    best_h = h.copy()
    best_c5 = compute_c5(h)

    _slsqp_state['n'] = n

    h = h0.copy()
    best_h = h.copy()

    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline or time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:

        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:

def adam_smooth(h0, n, target, time_limit, deadline, max_steps=50000, alpha_init=150.0):

    v = np.zeros(n)

    eps = 1e-8
    t_count = 0
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline or time.time() - start > time_limit:
            break
        t_count += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 2: CD at base
    if tleft() > 200:
        h_cd, c5_cd = greedy_cd(best_h, n_base, target_base,
                                 time_limit=min(120.0, tleft() * 0.2),
                                 deadline=deadline, max_sweeps=15000, n_cand=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP at base
    if tleft() > 150:
        h_lp, c5_lp = lp_successive(best_h, n_base, target_base,
                                     time_limit=min(80.0, tleft() * 0.15),
                                     deadline=deadline, n_active=25, max_rounds=60)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: Move to n=800
    n_hi = 800
    target_hi = float(n_hi) / 2.0
    h_hi = interpolate_to(best_h, n_hi, target_hi)
    c5_hi = compute_c5(h_hi)
    if c5_hi < best_score:
        best_score = c5_hi
        best_h = h_hi

    # Phase 5: LP at high resolution (primary driver)
    if tleft() > 300:
        h_lp, c5_lp = lp_successive(best_h, n_hi, target_hi,
                                     time_limit=min(180.0, tleft() * 0.35),
                                     deadline=deadline, n_active=30, max_rounds=150)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: CD at high resolution
    if tleft() > 200:
        h_cd, c5_cd = greedy_cd(best_h, n_hi, target_hi,
                                 time_limit=min(100.0, tleft() * 0.2),
                                 deadline=deadline, max_sweeps=8000, n_cand=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: SLSQP at high resolution
    if tleft() > 150:
        h_p, s_p = slsqp_round(best_h, n_hi, target_hi,
                                [100, 400, 1500, 6000],
                                1500, 1e-14, min(80.0, tleft() * 0.25), deadline)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 8: ADAM at high resolution
    if tleft() > 100:
        h_ad, c5_ad = adam_smooth(best_h, n_hi, target_hi,
                                   time_limit=min(60.0, tleft() * 0.2),
                                   deadline=deadline, max_steps=30000, alpha_init=200.0)
        if c5_ad < best_score:
            best_score = c5_ad
            best_h = h_ad

    # Phase 9: Try n=1000
    if tleft() > 250:
        n_hi2 = 1000
        target_hi2 = float(n_hi2) / 2.0
        h_hi2 = interpolate_to(best_h, n_hi2, target_hi2)
        c5_hi2 = compute_c5(h_hi2)
        if c5_hi2 < best_score:
            best_score = c5_hi2
            best_h = h_hi2
        if tleft() > 150:
            h_lp2, c5_lp2 = lp_successive(best_h, n_hi2, target_hi2,
                                           time_limit=min(100.0, tleft() * 0.3),
                                           deadline=deadline, n_active=30, max_rounds=80)
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2
        if tleft() > 80:
            h_cd2, c5_cd2 = greedy_cd(best_h, n_hi2, target_hi2,
                                       time_limit=min(50.0, tleft() * 0.2),
                                       deadline=deadline, max_sweeps=5000, n_cand=18)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
        if tleft() > 60:
            h_p2, s_p2 = slsqp_round(best_h, n_hi2, target_hi2,
                                      [200, 800, 3000],
                                      1500, 1e-14, min(40.0, tleft() * 0.3), deadline)
            if s_p2 < best_score:
                best_score = s_p2
                best_h = h_p2

    # Phase 10: Random restarts at best resolution
    if tleft() > 120:
        n_cur = len(best_h)
        tgt_cur = float(n_cur) / 2.0
        for restart in range(4):
            if tleft() < 30:
                break
            scale = 0.03 * (restart + 1)
            h_pert = perturb_restart(best_h, n_cur, tgt_cur, rng, scale=scale)
            h_r, s_r = lp_successive(h_pert, n_cur, tgt_cur,
                                      time_limit=min(30.0, tleft() * 0.2),
                                      deadline=deadline, n_active=30, max_rounds=40)
            if s_r < best_score:
                best_score = s_r
                best_h = h_r
            h_cd_r, c5_cd_r = greedy_cd(h_r, n_cur, tgt_cur,
                                         time_limit=min(15.0, tleft() * 0.1),
                                         deadline=deadline, max_sweeps=4000, n_cand=18)
            if c5_cd_r < best_score:
                best_score = c5_cd_r
                best_h = h_cd_r

    # Phase 11: Final LP polish
    if tleft() > 50:
        n_cur = len(best_h)
        tgt_cur = float(n_cur) / 2.0
        h_lp, c5_lp = lp_successive(best_h, n_cur, tgt_cur,
                                     time_limit=min(30.0, tleft() * 0.4),
                                     deadline=deadline, n_active=35, max_rounds=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final CD
    if tleft() > 20:
        n_cur = len(best_h)
        tgt_cur = float(n_cur) / 2.0
        h_cd, c5_cd = greedy_cd(best_h, n_cur, tgt_cur,
                                 time_limit=min(15.0, tleft() * 0.6),
                                 deadline=deadline, max_sweeps=4000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
