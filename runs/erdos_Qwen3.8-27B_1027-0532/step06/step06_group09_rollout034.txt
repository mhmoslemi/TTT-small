import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj_constrained(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection on offset."""
    h = h.astype(float).copy()
    # We want: find t such that sum(clip(h - t, 0, 1)) = target_sum
    # f(t) = sum(clip(h - t, 0, 1)) is decreasing in t
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    result = np.clip(h - t, 0.0, 1.0)
    # Fix numerical issues
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_smooth(h, alpha):
    """Log-sum-exp smooth approximation of max(corr)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_gradient(h, alpha):
    """Gradient of log-sum-exp smooth approximation of max(corr)."""
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for s in range(n_corr):
        if weights[s] < 1e-15:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[s] * h[idxs[v2] + k]
    
    return grad * dx

def frank_wolfe(h_init, n, target, time_limit, max_iter=5000, decreasing_gamma=True):
    """Frank-Wolfe (conditional gradient) method."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if decreasing_gamma:
            gamma = 2.0 / (it + 2)
        else:
            gamma = 0.1
        
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        idxs = np.arange(n)
        dx = 2.0 / n
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        # Subtract mean to account for sum constraint (since sum is fixed)
        grad = grad - np.mean(grad)
        
        # Linear oracle: min grad · h' s.t. 0<=h'<=1, sum(h')=target
        # Sort by grad, assign 1 to smallest, 0 to largest, one fractional
        sort_idx = np.argsort(grad)
        h_vw = np.zeros(n)
        remaining = target
        for idx in sort_idx:
            if remaining >= 1.0 - 1e-12:
                h_vw[idx] = 1.0
                remaining -= 1.0

                h_vw[idx] = remaining
                remaining = 0.0

            no_improve = 0

def fista_lse(h_init, n, target, time_limit, max_iter=5000, alpha_start=10.0, alpha_end=50000.0, n_anneal=5):
    """FISTA with log-sum-exp smoothing and annealing."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    no_improve = 0
    
    alphas = np.geomspace(alpha_start, alpha_end, n_anneal)
    steps_per_alpha = max_iter // n_anneal
    
    for alpha_idx, alpha in enumerate(alphas):
        if len(alphas) > 1:
            alpha_current = alphas[alpha_idx + 1]
        else:
            alpha_current = alpha
        
        for it in range(steps_per_alpha):
            if time.time() - start > time_limit:
                break
            
            # FISTA update
            y = h_prev + ((t_k - 1) / (t_k + 1)) * (h_prev - h)
            
            # Gradient step
            grad = lse_gradient(y, alpha_current)
            step_size = 1.0 / (2.0 * alpha_current * n)  # Lipschitz constant bound
            h_new = y - step_size * grad
            h_new = proj_constrained(h_new, target)
            
            # Update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            t_k = t_new
            h_prev = h
            h = h_new
            
            # Track best
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 1000:
                    break
    
    return best_h, best_c5

def subgradient_nesterov(h_init, n, target, time_limit, max_steps=20000, base_lr=0.05):
    """Subgradient method with Nesterov acceleration."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    t_k = 1.0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        # Compute subgradient at y
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr_y)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - y[idxs[v1] - k_star])
        grad[v2] -= y[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        
        # Nesterov update
        lr = base_lr / (1.0 + step * 0.01)
        h_new = proj_constrained(y - lr * grad, target)
        
        # Update y
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h_new + ((t_k - 1) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        t_k = t_new
        h = h_new
        
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    """SLSQP with log-sum-exp smoothing."""
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    for alpha in alphas:
        try:
            def obj(x):
                return lse_smooth(x, alpha)
            
            def obj_grad(x):
                return lse_gradient(x, alpha)
            
            res = minimize(obj, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = proj_constrained(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj_constrained(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    result = np.clip(h - t, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)

def lse_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lse_grad(h, alpha):

    dx = 2.0 / n
    mid = n - 1

    return grad * dx

def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
                       n_cand=30, patience=300, n_bisect=20):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        adj_n_cand = n_cand
        if no_improve > patience // 2:
            adj_n_cand = min(n_cand + 15, 60)

        k_star = int(np.argmax(corr)) - mid
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:adj_n_cand]
        high_set = sorted_idx[-adj_n_cand:]

        J = np.zeros((n, n_corr))
        for i in range(n):
            for s in range(n_corr):
                k = s - mid
                val = 0.0
                if 0 <= i - k < n:
                    val += (1.0 - h[i - k])
                if 0 <= i + k < n:
                    val -= h[i + k]
                J[i, s] = val

        n_lo = len(low_set)
        n_hi = len(high_set)
        G = J[high_set[None, :], :] - J[low_set[:, None], :]

        hi_bounds = np.minimum(h[high_set][None, :], 1.0 - h[low_set][:, None])
        lo_bounds = np.maximum(h[high_set][None, :], -h[low_set][:, None])

        valid = (hi_bounds - lo_bounds) > 1e-12
        np.fill_diagonal(valid, False)

        a_d = np.where(valid, lo_bounds, 0.0)
        b_d = np.where(valid, hi_bounds, 0.0)

        for _ in range(n_bisect):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr[None, :] + mid_d[:, None] * G
            f_mid = np.max(vals, axis=1)
            active = vals >= f_mid[:, None] - 1e-14
            f_prime = np.max(np.where(active, G, -1e18), axis=1)
            inc = (f_prime > 1e-12) & valid
            dec = (f_prime < -1e-12) & valid
            a_d[dec] = mid_d[dec]
            b_d[inc] = mid_d[inc]

        delta_star = 0.5 * (a_d + b_d)
        approx = np.max(corr[None, :] + delta_star[:, None] * G, axis=1) * dx
        approx[~valid] = c5_val + 1.0

        flat_idx = int(np.argmin(approx))
        fi = flat_idx // n_lo
        fj = flat_idx % n_lo

        if approx[flat_idx] < c5_val - 1e-14:
            i_pos = int(high_set[fi])
            j_pos = int(low_set[fj])
            d = delta_star[flat_idx]
            h[i_pos] -= d
            h[j_pos] += d
            h = np.clip(h, 0.0, 1.0)
            h = proj_constrained(h, target)

    return best_h, best_c5

def fista_lse(h_init, n, target, time_limit, max_iter=8000,
              alpha_start=5.0, alpha_end=50000.0, n_anneal=6):
    start = time.time()
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    no_improve = 0
    alphas = np.geomspace(alpha_start, alpha_end, n_anneal)
    steps_per = max_iter // n_anneal

    for ai in range(n_anneal):
        alpha_c = alphas[min(ai + 1, n_anneal - 1)]
        L = 4.0 * alpha_c + 20.0
        step_size = 1.0 / L
        for it in range(steps_per):
            if time.time() - start > time_limit:
                break
            y = h_prev + ((t_k - 1.0) / (t_k + 1.0)) * (h_prev - h)
            grad = lse_grad(y, alpha_c)
            h_new = proj_constrained(y - step_size * grad, target)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            t_k = t_new
            h_prev = h
            h = h_new
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 500:
                    break
    return best_h, best_c5

def frank_wolfe(h_init, n, target, time_limit, max_iter=8000):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        gamma = 2.0 / (it + 2.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad = grad - np.mean(grad)
        sort_idx = np.argsort(grad)
        h_vw = np.zeros(n)
        remaining = target
        for idx in sort_idx:
            if remaining >= 1.0 - 1e-12:
                h_vw[idx] = 1.0
                remaining -= 1.0
            elif remaining > 1e-12:
                h_vw[idx] = remaining
                remaining = 0.0
            else:
                break
        h_new = (1.0 - gamma) * h + gamma * h_vw
        h_new = proj_constrained(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break
        h = h_new
    return best_h, best_c5

def lbfags_smooth(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        def obj(x):
            return lse_val(x, alpha)
        def obj_grad(x):
            return lse_grad(x, alpha)
        res = minimize(obj, h_start, jac=obj_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
        h_new = proj_constrained(res.x, target)
        return h_new
    except Exception:
        return h_start

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    for alpha in alphas:
        try:
            def obj(x):
                return lse_val(x, alpha)
            def obj_grad(x):
                return lse_grad(x, alpha)
            res = minimize(obj, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = proj_constrained(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def multi_scale_search(seed, n_final, budget_remaining):
    rng = np.random.RandomState(seed)
    scales = [50, 100, 200, n_final]
    best_h = None
    best_c5 = 1.0
    for n_s in scales:
        if budget_remaining < 30:
            break
        target_s = n_s / 2.0
        if best_h is not None and len(best_h) != n_s:
            x_old = np.linspace(0.0, 1.0, len(best_h))
            x_new = np.linspace(0.0, 1.0, n_s)
            h_s = np.interp(x_new, x_old, best_h)
        else:
            h_s = 0.5 * np.ones(n_s)
            n_ones = int(round(target_s))
            idxs = rng.choice(n_s, size=n_ones, replace=False)
            h_s = np.zeros(n_s)
            h_s[idxs] = 1.0
            remainder = target_s - n_ones
            if abs(remainder) > 1e-10:
                frac_idx = int(rng.choice(n_s))
                h_s[frac_idx] += remainder
            h_s = proj_constrained(h_s, target_s)
        time_for_stage = min(60, budget_remaining * 0.3)
        h_s = lbfags_smooth(h_s, n_s, target_s, 100.0, maxiter=500)
        h_pc, c5_pc = vectorized_pair_cd(h_s, n_s, target_s,
                                          time_limit=min(30, time_for_stage * 0.5),
                                          max_sweeps=2000, n_cand=20, patience=100)
        h_s = proj_constrained(h_pc, target_s)
        sc = compute_c5(h_s)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_s.copy()
        budget_remaining = budget_remaining - time_for_stage
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = proj_constrained(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: FISTA LSE
    if tleft() > 250:
        ft = min(150, tleft() * 0.2)
        h_f, c5_f = fista_lse(best_h, n, target, time_limit=ft,
                               max_iter=6000, alpha_start=5.0,
                               alpha_end=30000.0, n_anneal=5)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 2: Frank-Wolfe
    if tleft() > 150:
        fw_t = min(100, tleft() * 0.12)
        h_fw, c5_fw = frank_wolfe(best_h, n, target, time_limit=fw_t, max_iter=10000)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 3: Vectorized pair CD (main workhorse)
    if tleft() > 200:
        cd_t = min(300, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target, time_limit=cd_t,
                                          max_sweeps=50000, n_cand=30,
                                          patience=400, n_bisect=22)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: L-BFGS-B with increasing alpha
    if tleft() > 100:
        for alpha in [2000, 10000, 50000]:
            if tleft() < 40:
                break
            h_try = lbfags_smooth(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 5: Second pair CD round with more candidates
    if tleft() > 200:
        cd_t = min(250, tleft() * 0.25)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target, time_limit=cd_t,
                                          max_sweeps=50000, n_cand=40,
                                          patience=300, n_bisect=22)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: SLSQP high-alpha polish
    if tleft() > 80:
        h_slq = slsqp_smooth(best_h, n, target, [20000, 100000, 500000], 600, 1e-15)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        for ns in [0.02, 0.05, 0.1, 0.15]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n) * ns
            h_pert = proj_constrained(h_pert, target)
            h_ref = lbfags_smooth(h_pert, n, target, 5000, maxiter=800)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
                h_cd2, c5_cd2 = vectorized_pair_cd(best_h, n, target,
                                                    time_limit=min(30, tleft() * 0.15),
                                                    max_sweeps=5000, n_cand=30,
                                                    patience=100, n_bisect=18)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2

    # Phase 8: Higher resolution
    if tleft() > 200:
        for n_try in [600, 800, 1000]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            x_old = np.linspace(0.0, 1.0, n)
            x_new = np.linspace(0.0, 1.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = proj_constrained(h_interp, target_try)
            h_hires = lbfags_smooth(h_interp, n_try, target_try, 3000, maxiter=800)
            h_hires, c5_h = vectorized_pair_cd(h_hires, n_try, target_try,
                                                time_limit=min(60, tleft() * 0.3),
                                                max_sweeps=10000, n_cand=25,
                                                patience=150, n_bisect=18)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 9: Final intensive pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_cd, c5_final = vectorized_pair_cd(best_h, n_use, target_use,
                                                   time_limit=min(50, tleft() * 0.5),
                                                   max_sweeps=30000, n_cand=50,
                                                   patience=200, n_bisect=25)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Final
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = proj_constrained(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
