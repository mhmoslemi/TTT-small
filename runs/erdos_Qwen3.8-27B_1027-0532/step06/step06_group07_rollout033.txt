import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    w_norm = weights / w_sum
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = w_norm[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] += w * (-h[idxs[v2] + k])
    val = c_max + np.log(w_sum) / alpha
    return val, grad

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g

                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5

def project_feasible(h, target_sum):

                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5

def active_set_cd(h, n, target, time_limit, n_bisect=16, max_sweeps=4000, n_top=5, n_cand=35):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 70:
            break

        top_k = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for k_idx in top_k:
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = corr[k_idx]
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] += w * (-h[idxs[v2] + k])
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 35:
                break

    return best_h, best_c5

def subgradient_polish(h_init, n, target, time_limit, max_iters=30000):
    start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    step = 0.08
    patience = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience > 400:
            step *= 0.7
            if step < 0.001:
                break
            patience = 0

        c_max = float(np.max(corr))
        threshold = c_max - 0.03 * c_max
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            break
        weights = np.maximum(corr[active] - threshold + 1e-8, 0.0)
        w_sum = np.sum(weights)
        if w_sum < 1e-10:
            break
        weights = weights / w_sum

        grad = np.zeros(n)
        for idx in range(len(active)):
            k_idx = active[idx]
            w = weights[idx]
            if w < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] += w * (-h[idxs[v2] + k])

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - mult * step * grad / grad_norm, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            step *= 0.75

    return best_h, best_c5

def multi_pair_cd(h, n, target, time_limit, max_sweeps=2000, n_cand=30, n_bisect=14, n_pairs=3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        candidates = []
        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                improvement = c5_val - approx_max
                if improvement > 1e-15:
                    candidates.append((improvement, i, j, delta_star, g))

        if not candidates:
            if no_improve > 25:
                break
            continue

        candidates.sort(key=lambda x: -x[0])
        h_new = h.copy()
        corr_new = corr.copy()
        used_pts = set()
        applied = 0

        for imp, i, j, delta, g in candidates[:n_pairs * 4]:
            if i in used_pts or j in used_pts:
                continue
            test_corr = corr_new + delta * g
            if np.max(test_corr) < np.max(corr_new) - 1e-14:
                h_new[i] -= delta
                h_new[j] += delta
                corr_new = test_corr
                used_pts.add(i)
                used_pts.add(j)
                applied += 1
            if applied >= n_pairs:
                break

        if applied > 0:
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: SLSQP moderate alpha
    if remaining() > 200:
        h_pol = slsqp_round(best_h, n, target,
                            [200, 500, 1000, 3000],
                            800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Fast pair CD (main workhorse)
    if remaining() > 300:
        cd_time = min(250, remaining() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=8000, n_cand=50, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP high alpha
    if remaining() > 150:
        h_pol = slsqp_round(best_h, n, target,
                            [5000, 10000, 30000, 80000],
                            1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Active set CD
    if remaining() > 200:
        as_time = min(180, remaining() * 0.25)
        h_as, c5_as = active_set_cd(best_h, n, target,
                                    time_limit=as_time, n_bisect=16,
                                    max_sweeps=4000, n_top=5, n_cand=35)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 5: SLSQP very high alpha
    if remaining() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [50000, 100000, 200000, 500000],
                            1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Multi-pair CD
    if remaining() > 120:
        mp_time = min(100, remaining() * 0.2)
        h_mp, c5_mp = multi_pair_cd(best_h, n, target,
                                    time_limit=mp_time,
                                    max_sweeps=1500, n_cand=30, n_bisect=14,
                                    n_pairs=3)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    # Phase 7: Subgradient polish
    if remaining() > 80:
        sg_time = min(60, remaining() * 0.15)
        h_sg, c5_sg = subgradient_polish(best_h, n, target,
                                         time_limit=sg_time, max_iters=30000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Another fast CD round
    if remaining() > 100:
        cd_time = min(80, remaining() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=5000, n_cand=55, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final SLSQP
    if remaining() > 60:
        h_pol = slsqp_round(best_h, n, target,
                            [100000, 300000, 500000],
                            600, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Random restarts
    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.08, 0.12]:
        if remaining() < 60:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(40, remaining() * 0.12)
        h_try, c5_try = fast_pair_cd(h_pert, n, target,
                                     time_limit=cd_t,
                                     max_sweeps=1500, n_cand=40, n_bisect=14)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 11: Active set CD final
    if remaining() > 60:
        as_time = min(50, remaining() * 0.2)
        h_as, c5_as = active_set_cd(best_h, n, target,
                                    time_limit=as_time, n_bisect=16,
                                    max_sweeps=3000, n_top=7, n_cand=40)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 12: Final SLSQP polish
    if remaining() > 30:
        h_pol = slsqp_round(best_h, n, target,
                            [300000, 500000],
                            500, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
