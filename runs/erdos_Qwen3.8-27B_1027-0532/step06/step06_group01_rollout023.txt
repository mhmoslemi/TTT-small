def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

    n_iter = 0

        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        n_iter += 1
    return h

def _smooth_max_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    penalty += 1000.0 * s * s
    return float(c_max + logsum) + penalty

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Compute profiles efficiently
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

Wait, I realize the multi_shift_cd has an O(n_cand² * n_corr) inner loop which is too slow. Let me go back to the profile-based approach which is O(n_c * n_corr + n_cand²) per sweep.

Let me redesign the approach:

The key insight from the previous code is that the profile-based computation is efficient:
- Precompute profiles for all candidates: O(n_c * n_corr)
- For each pair (i,j): compute g = profiles[i] - profiles[j], then binary search: O(n_corr * log(1/eps))
- Total per sweep: O(n_c * n_corr + n_cand² * n_corr * log(1/eps))

With n=600, n_c≈50, n_corr=1199, n_cand=25:
- Profile computation: 50 * 1199 ≈ 60K
- Pair evaluation: 625 * 1199 * 12 ≈ 9M

That's about 9M operations per sweep. At ~100M ops/sec in Python (with numpy), that's about 0.1s per sweep. So in 300s we can do about 3000 sweeps.

The previous code did this and got to 0.380872. To improve further, I need either:
1. More iterations (more time)
2. Better moves (larger improvements per iteration)
3. A different approach that can make larger jumps

Let me think about option 3. What if we use a "bilinear" move: reduce two high points and increase two low points simultaneously? This gives 4D moves instead of 2D, which can escape some local optima.

Actually, let me think about a completely different approach: **simulated annealing on the binary structure**.

For the binary case (h ∈ {0,1}), the problem becomes: find a binary sequence of length n with exactly n/2 ones that minimizes the maximum correlation.

This is a combinatorial optimization problem. Simulated annealing or tabu search might work well:
- Start from a good initial solution
- Define moves: flip one bit (but maintain the sum constraint by doing a "swap": change a 1 to 0 and a 0 to 1)
- Accept moves based on a temperature schedule

The key advantage is that binary moves are "large" in the continuous space (each move changes the L1 distance by 2), so they can escape local optima of the continuous relaxation.

But wait, the optimal solution might not be binary. Let me check: for the continuous case, we found C₅ ≈ 0.3808. What's the best binary C₅?

For binary h with n/2 ones, corr at shift 0 is 0 (as computed earlier). The max corr is at small non-zero shifts. For random binary, it's about n/4 = 150, giving C₅ = 0.5.

The best known binary sequences (like m-sequences) have sidelobe values of about n/4 ± sqrt(n), so the max is about n/4 + sqrt(n), giving C₅ ≈ (n/4 + sqrt(n))/(n/2) ≈ 1/2 + 2/sqrt(n). For n=600, that's about 0.5 + 0.08 = 0.58. Still much worse than 0.38.

So the continuous solution is much better than any binary solution. This makes sense - by using fractional values, we can reduce the peaks more evenly.

OK so let me focus on improving the continuous optimization. Key strategies:

1. **Better initialization**: Instead of just using initial_h_values, try several perturbations.

2. **Adaptive pair CD**: Dynamically adjust n_cand based on convergence.

3. **Block moves**: Try moving mass from a group of high-gradient points to a group of low-gradient points.

4. **Resolution change**: Try optimizing at different resolutions and interpolating.

5. **Penalty-based approach**: Use a penalty for constraint violation in the SLSQP objective.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_smax(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smooth = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    pen = 500.0 * (np.sum(x) - target) ** 2
    return smooth + pen

def slsqp_opt(h0, n, target, alphas, maxiter, ftol, timeout):
    t0 = time.time()
    bds = [(0.0, 1.0)] * n
    h = h0.copy()
    best = h.copy()
    bs = compute_c5(h)
    for a in alphas:
        if time.time() - t0 > timeout:
            break
        try:
            r = minimize(lambda x: _obj_smax(x, a, target), h, method='SLSQP',
                         bounds=bds, options={'maxiter': maxiter, 'ftol': ftol})
            hn = project_feasible(r.x, target)
            cs = compute_c5(hn)
            if cs < bs:
                bs = cs
                best = hn.copy()
            h = hn
        except:
            pass
    return best, bs

def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def _smax_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(corr)
    sh = np.clip(alpha * (corr - cm), -500.0, 0.0)
    val = float(cm + np.log(np.sum(np.exp(sh))) / alpha)
    pen = 1000.0 * (np.sum(x) - target) ** 2
    return val + pen

def slsqp_run(h0, n, target, alphas, maxiter, ftol, timeout):
