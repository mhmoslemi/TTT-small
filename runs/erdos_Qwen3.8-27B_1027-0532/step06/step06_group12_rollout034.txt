Now, for the continuous part, let me use a **proximal gradient** method on the soft-min objective:

f(h) = -log(Σ_k exp(-alpha * autocorr(k))) / alpha  [this is a soft-min of autocorr]

Or equivalently, since we want to minimize max overlap:
g(h) = log(Σ_k exp(alpha * overlap(k))) / alpha  [soft-max of overlap]

The gradient of g with respect to h[i] can be computed.

For overlap(k) = Σ_p h[p]*(1-h[p+k])*dx:
∂overlap(k)/∂h[i] = dx * [(1-h[i+k])*1(valid) - h[i-k]*1(valid)]

The gradient of g:
∂g/∂h[i] = (1/alpha) * (1/Σ_k exp(alpha*overlap(k))) * Σ_k alpha*exp(alpha*overlap(k)) * ∂overlap(k)/∂h[i]
= Σ_k softmax_k * ∂overlap(k)/∂h[i]

where softmax_k = exp(alpha*overlap(k)) / Σ_m exp(alpha*overlap(m))

This is a weighted average of the per-lag gradients, weighted by the softmax.

I can compute this efficiently:
- For each lag k, the gradient ∂overlap(k)/∂h is a specific pattern
- The weighted sum gives the total gradient

Let me write this:

```python
def soft_max_gradient(h, n, alpha):
    """Compute gradient of soft-max of overlaps."""
    dx = 2.0 / n
    overlaps = compute_overlaps(h)  # length 2n-1
    # Softmax weights
    max_ov = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient: weighted sum of per-lag gradients
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += w * (1 - 2*h) * dx
        elif lag > 0:
            # ∂overlap(lag)/∂h[i] = dx * [(1-h[i+lag]) - h[i-lag]]
            # For i in [0, n-1-lag]: (1-h[i+lag])*dx
            # For i in [lag, n-1]: -h[i-lag]*dx
            # For i in [lag, n-1-lag]: both terms
            # Consolidated:
            g = np.zeros(n)
            g[:n-lag] += (1 - h[lag:]) * dx
            g[lag:] += (-h[:n-lag]) * dx
            grad += w * g
        else:
            m = -lag
            g = np.zeros(n)
            g[m:] += (1 - h[:n-m]) * dx
            g[:n-m] += (-h[m:]) * dx
            grad += w * g
    
    return grad

```python
def compute_softmax_grad(h, n, alpha):
    """Compute gradient of logsumexp(alpha * overlaps) / alpha."""
    dx = 2.0 / n
    overlaps = compute_overlaps(h)  # length 2n-1
    max_ov = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            # ∂overlap(k)/∂h[i] for k > 0:
            # i in [0, n-1-k]: (1-h[i+k])*dx
            # i in [k, n-1]: -h[i-k]*dx
            # i in [k, n-1-k]: both
            temp = np.zeros(n)
            temp[:n-k] += (1.0 - h[k:]) * dx
            temp[k:] += (-h[:n-k]) * dx
            grad += w * temp
        else:
            m_val = -k
            temp = np.zeros(n)
            temp[m_val:] += (1.0 - h[:n-m_val]) * dx
            temp[:n-m_val] += (-h[m_val:]) * dx
            grad += w * temp
    
    return grad
```

This is O(n * n) = O(n²) per gradient evaluation. For n=400, that's 160K operations. Very fast.

```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary_exact(h, target):
    """Convert to binary with exact sum = target (target must be integer)."""
    h_bin = (h >= 0.5).astype(float)
    diff = int(target) - int(np.sum(h_bin))
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        # Pick the ones with highest h value (before binarization)
        # Actually just pick first diff zeros
        h_bin[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        h_bin[ones[-diff:]] = 0.0
    return h_bin

                      time_limit, start_time, rng_seed=0):

    Binary local search with multi-lag weighting and simulated annealing.
    """
    h = h_init.copy()
    overlaps = compute_overlaps(h)
    best_score = float(np.max(overlaps))
    best_h = h.copy()
    stall = 0
    rng = np.random.RandomState(rng_seed)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Get top worst lags
        top_idx = np.argsort(overlaps)[::-1][:top_k]
        weights = np.array([1.0 / (r + 1.0) for r in range(top_k)])
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Compute weighted cost for each one-position
        total_cost = np.zeros(len(ones_idx))
        total_benefit = np.zeros(len(zeros_idx))
        
        for wi in range(top_k):
            m = top_idx[wi]
            k = m - (n - 1)
            w = weights[wi]
            
            # Cost of removing position i at lag k
            # = h[i+k] (if valid) + h[i-k] (if valid)
            i_plus = ones_idx + k
            i_minus = ones_idx - k
            vp = (i_plus >= 0) & (i_plus < n)
            vm = (i_minus >= 0) & (i_minus < n)
            
            cost_k = np.zeros(len(ones_idx))
            if np.any(vp):
                cost_k[vp] += h[np.clip(i_plus, 0, n-1)][vp]
            if np.any(vm):
                cost_k[vm] += h[np.clip(i_minus, 0, n-1)][vm]
            total_cost += w * cost_k
            
            # Benefit of adding position j at lag k
            j_plus = zeros_idx + k
            j_minus = zeros_idx - k
            vpj = (j_plus >= 0) & (j_plus < n)
            vmj = (j_minus >= 0) & (j_minus < n)
            
            ben_k = np.zeros(len(zeros_idx))
            if np.any(vpj):
                ben_k[vpj] += h[np.clip(j_plus, 0, n-1)][vpj]
            if np.any(vmj):
                ben_k[vmj] += h[np.clip(j_minus, 0, n-1)][vmj]
            total_benefit += w * ben_k
        
        # Best swap: minimize cost, maximize benefit
        best_i_local = np.argmin(total_cost)
        best_j_local = np.argmax(total_benefit)
        best_i = ones_idx[best_i_local]
        best_j = zeros_idx[best_j_local]
        
        # Perform swap
        h[best_i] = 0
        h[best_j] = 1
        new_overlaps = compute_overlaps(h)
        new_score = float(np.max(new_overlaps))

            best_h = h.copy()
