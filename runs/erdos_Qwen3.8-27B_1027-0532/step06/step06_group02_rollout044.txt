def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def compute_smooth_gradient(h, alpha, corr_full):
    n = len(h)
    max_c = corr_full.max()
    shifted = alpha * (corr_full - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_c = np.exp(shifted)
    w = exp_c / (exp_c.sum() + 1e-30)
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    grad = 1.0 - 2.0 * conv
    return grad

        v = np.zeros(n)

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            grad = compute_smooth_gradient(h, alpha, corr)
            lr_t = lr / (1.0 + 0.01 * it)
            v = momentum * v - lr_t * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 30 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def it_ls_with_perturb(h, n_rounds, rng, time_limit, start_time, k_arr):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    positions = np.arange(n)

    for rnd in range(n_rounds):
        if time.time() - start_time > time_limit:
            break
        h_cur = best_h.copy()
        cur_c5 = best_c5
        full = full_corr_fast(h_cur)

        for it in range(5000):
            if time.time() - start_time > time_limit:
                break
            max_val = float(full.max())
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h_cur[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h_cur[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h_cur[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h_cur[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(min(20, len(i_cands)))]
                    j = j_cands[rng.randint(min(20, len(j_cands)))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]

            delta = compute_delta(h_cur, i, j, k_arr)
            new_max = float((full + delta).max())
            if new_max < max_val - 1e-10:
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                full += delta
                cur_c5 = new_max * dx
            elif rng.random() < 0.05:
                h_cur[i] = 0.0
                h_cur[j] = 1.0
                full += delta
                cur_c5 = new_max * dx
            else:
                continue

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()

        n_pert = rng.randint(3, 10)
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        h = h_src.copy()
        return project_box_sum(h, target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy to improve initial
    if tleft() > 50:
        allowed = min(40.0, tleft() * 0.04)
        h_g, c5_g = fast_greedy(best_h, max_iters=50000, rng=rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Parallel tempering SA (main workhorse)
    if tleft() > 100:
        allowed = min(500.0, tleft() * 0.45)
        temps = [0.003, 0.006, 0.012, 0.025, 0.05, 0.1]
        n_rep = 6
        n_steps = int(allowed * 40000)
        h_pt, c5_pt = parallel_tempering_sa(
            best_h, n_rep, temps, n_steps,
            time_limit=allowed, start_time=start_time, k_arr=k_arr,
            rng=rng, block_frac=0.15, max_block_len=12, swap_interval=20
        )
        if c5_pt < best_score:
            best_score = c5_pt
            best_h = h_pt.copy()

    # Phase 3: Focused SA from best
    if tleft() > 80:
        allowed = min(150.0, tleft() * 0.15)
        n_sa = int(min(8000000, allowed * 50000))
        h_sa, c5_sa = sa_focused(best_h, n_sa, 0.008, 0.00002, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Continuous refinement
    if tleft() > 80:
        target_use = float(n_init) / 2.0
        alphas = [5, 10, 30, 100, 300, 1000, 3000]
        allowed = min(200.0, tleft() * 0.2)
        h_ref, sc = continuous_refine(best_h, target_use, start_time, allowed, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: Project to binary and do focused SA again
    if tleft() > 60:
        h_bin = project_binary(best_h, target_init)
        allowed = min(100.0, tleft() * 0.12)
        n_sa2 = int(min(4000000, allowed * 40000))
        h_sa2, c5_sa2 = sa_focused(h_bin, n_sa2, 0.006, 0.00001, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Iterated local search with perturbation
    if tleft() > 60:
        allowed = min(150.0, tleft() * 0.15)
        n_rounds = int(allowed / 15)
        h_ils, c5_ils = it_ls_with_perturb(best_h, n_rounds, rng,
                                           time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

    # Phase 7: Another parallel tempering round from current best
    if tleft() > 100:
        allowed = min(200.0, tleft() * 0.2)
        temps2 = [0.002, 0.004, 0.008, 0.016, 0.032, 0.064]
        n_steps2 = int(allowed * 40000)
        h_pt2, c5_pt2 = parallel_tempering_sa(
            best_h, 6, temps2, n_steps2,
            time_limit=allowed, start_time=start_time, k_arr=k_arr,
            rng=rng, block_frac=0.1, max_block_len=8, swap_interval=15
        )
        if c5_pt2 < best_score:
            best_score = c5_pt2
            best_h = h_pt2.copy()

    # Phase 8: Multi-start at different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20.0, tleft() * 0.05)
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=10000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr_try)
        allowed2 = min(20.0, tleft() * 0.05)
        n_sa3 = int(min(500000, allowed2 * 30000))
        h_s3, c5_s3 = sa_focused(h_g3, n_sa3, 0.005, 0.00003, rng,
                                 time_limit=allowed2, start_time=start_time, k_arr=k_arr_try)
        c5_try = c5_s3 if c5_s3 < c5_g3 else c5_g3
        h_try = h_s3 if c5_s3 < c5_g3 else h_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try
            k_arr = k_arr_try
            n_init = n_try
            target_init = target_try

    # Phase 9: Final gradient polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_final = [50, 200, 800, 3000, 10000]
        allowed = min(tleft() - 10, 100.0)
        h_ref2, sc2 = continuous_refine(best_h, target_use, start_time, allowed, alphas_final)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 10: Final binary SA burst
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        k_arr_f = np.arange(2 * n_use - 1) - (n_use - 1)
        h_bin_f = project_binary(best_h, target_use)
        n_sa_f = int(min(2000000, max(100000, (tleft() - 8) * 40000)))
        h_sa_f, c5_f = sa_focused(h_bin_f, n_sa_f, 0.004, 0.000005,
                                  rng, time_limit=max(5.0, tleft() - 5),
                                  start_time=start_time, k_arr=k_arr_f)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
