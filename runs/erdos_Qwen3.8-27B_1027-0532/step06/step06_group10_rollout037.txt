    # Returns G of shape (2n-1, n) where G[s, j] = d(corr[s])/d(h[j])
    G = np.zeros((2*n-1, n))
    mid = n - 1
    for s in range(2*n-1):
        k = s - mid
        if k >= 0:
            # g[j] = (1-h[j+k]) for j < n-k, minus h[j-k] for j >= k
            G[s, :n-k] = 1.0 - h[k:n]
            G[s, k:] += -h[:n-k]

            G[s, l:] = 1.0 - h[:n-l]
            G[s, :n-l] += -h[l:]
    return G

    mid = n - 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_matrix(h, n):
    G = np.zeros((2*n-1, n))
    mid = n - 1
    for k in range(n):
        s = mid + k
        if k == 0:
            G[s, :] = 1.0 - h
        else:
            G[s, :n-k] = 1.0 - h[k:]
            G[s, k:] += -h[:n-k]
    for l in range(1, n):
        s = mid - l
        G[s, l:] = 1.0 - h[:n-l]
        G[s, :n-l] += -h[l:]
    return G

                    if 0 <= j-k < n:

                a_d = 0.0

For n=400, c=15: 160K + 90K = 250K per sweep. At ~100ns per operation in numpy, that's ~0.025s per sweep. With 5000 sweeps: 125s. That's a lot but doable if I'm careful.

Actually, the sig computation is the bottleneck. Let me see if I can vectorize it.

sig_p[s] for all p and s:
For each p (position), and each s (shift index):
k = s - mid
sig_p[s] = (1-h[p+k])·1_{0≤p+k<n} - h[p-k]·1_{0≤p-k<n}

This is like a "cross-correlation" of h with itself, but with a specific structure.

For fixed p, sig_p is a vector of length 2n-1:
- For s = mid + k (k ≥ 0): sig_p[mid+k] = (1-h[p+k])·1_{p+k<n} - h[p-k]·1_{p-k≥0}
- For s = mid - l (l > 0): sig_p[mid-l] = (1-h[p-l])·1_{p-l≥0} - h[p+l]·1_{p+l<n}

This is essentially the "first-order effect" of changing h[p] on all correlations.

I can compute this using slicing:
```python
def compute_sigs(h, n):
    # Returns sig of shape (n, 2n-1) 
    # sig[p, s] = d(corr[s])/d(h[p])
    mid = n - 1
    sig = np.zeros((n, 2*n-1))
    for p in range(n):
        # For shift k >= 0: contribution (1-h[p+k]) if p+k < n
        for k in range(n - p):  # p + k < n
            sig[p, mid + k] += (1.0 - h[p + k])
        # For shift k >= 0: contribution -h[p-k] if p-k >= 0
        for k in range(1, p + 1):  # p - k >= 0 means k <= p
            sig[p, mid + k] += -h[p - k]
        # For shift -l (l > 0): contribution (1-h[p-l]) if p-l >= 0
        for l in range(1, p + 1):
            sig[p, mid - l] += (1.0 - h[p - l])
        # For shift -l: contribution -h[p+l] if p+l < n
        for l in range(1, n - p):
            sig[p, mid - l] += -h[p + l]
    return sig
```

This is O(n²) but with Python loops it'll be slow. Let me vectorize:

For a fixed p:
- sig[p, mid+k] gets (1-h[p+k]) for k = 0, ..., n-1-p
  This is: sig[p, mid:mid+n-p] += 1 - h[p:n]
- sig[p, mid+k] gets -h[p-k] for k = 1, ..., p
  This is: sig[p, mid+1:mid+1+p] += -h[p-1::-1]  (reversed)
  
  Wait, let me be more careful.
  For k = 1: sig[p, mid+1] += -h[p-1]
  For k = 2: sig[p, mid+2] += -h[p-2]
  ...
  For k = p: sig[p, mid+p] += -h[0]
  
  So sig[p, mid+1:mid+1+p] += -h[0:p][::-1] = -h[p-1::-1]
  
  In numpy: sig[p, mid+1:mid+1+p] += -h[p-1::-1]
  But h[p-1::-1] gives h[p-1], h[p-2], ..., h[0]. Yes, that's correct.

- sig[p, mid-l] gets (1-h[p-l]) for l = 1, ..., p
  For l=1: sig[p, mid-1] += 1-h[p-1]
  For l=2: sig[p, mid-2] += 1-h[p-2]
  ...
  So sig[p, mid-p:mid] += 1 - h[p-1:p-1-p:-1] = 1 - h[p-1::-1]
  
  In numpy: sig[p, mid-p:mid] += 1.0 - h[p-1::-1]
  But h[p-1::-1] has p elements: h[p-1], ..., h[0]. And mid-p:mid has p elements. ✓

- sig[p, mid-l] gets -h[p+l] for l = 1, ..., n-1-p
  For l=1: sig[p, mid-1] += -h[p+1]
  For l=2: sig[p, mid-2] += -h[p+2]
  ...
  So sig[p, mid-(n-1-p):mid] += -h[p+1:n]
  
  Wait, that's not right. Let me re-examine.
  For l=1: sig[p, mid-1] += -h[p+1]
  For l=2: sig[p, mid-2] += -h[p+2]
  ...
  For l = n-1-p: sig[p, mid-(n-1-p)] += -h[n-1]
  
  So sig[p, mid-(n-1-p):mid] gets values -h[p+1], -h[p+2], ..., -h[n-1]
  But in reverse order: sig[p, mid-1] gets -h[p+1], sig[p, mid-2] gets -h[p+2], etc.
  
  This is: sig[p, mid-(n-1-p):mid] += -h[p+1:n][::-1]
  
  Let me verify: h[p+1:n][::-1] = h[n-1], h[n-2], ..., h[p+1]
  And sig[p, mid-(n-1-p)] should get -h[n-1], sig[p, mid-(n-1-p)+1] should get -h[n-2], etc.
  Yes! ✓

So the vectorized version:
```python
def compute_sigs(h, n):
    mid = n - 1
    sig = np.zeros((n, 2*n-1))
    for p in range(n):
        # Positive shifts
        if n - p > 0:
            sig[p, mid:mid+n-p] += 1.0 - h[p:]
        if p > 0:
            sig[p, mid+1:mid+1+p] += -h[p-1::-1]
        # Negative shifts
        if p > 0:
            sig[p, mid-p:mid] += 1.0 - h[p-1::-1]
        if n - p - 1 > 0:
            sig[p, mid-(n-1-p):mid] += -h[p+1:n][::-1]
    return sig
```

This is O(n²) with small constant. For n=400, it's about 160K slice operations (each of average length ~n/2). In Python, each slice operation has some overhead, so this might take ~1-2 seconds. That's too slow for 5000 sweeps.

Let me think of a better approach. Actually, I don't need to recompute sigs every sweep. The sigs only change when h changes. And h changes by a small amount each sweep. So I could use a stale approximation... but that might not be accurate enough.

Alternative: don't use the full sig matrix. Instead, for the CD, just use the gradient at the active shift k* to select candidates, then do the exact line search for only the top few pairs.

Here's a simpler plan:
1. Compute corr and find k* (the worst shift)
2. Compute gradient g at k*: g[j] = (1-h[j+k*])·1_{valid} - h[j-k*]·1_{valid}
3. Select top n_cand "to decrease" (highest g) and bottom n_cand "to increase" (lowest g)
4. For each pair (i from high, j from low), do exact line search
5. Apply the best move

For the exact line search, I need d_corr[s] for each pair. But I only need to compute it for the pairs I'm actually evaluating. With n_cand=10, that's 100 pairs. For each pair, computing d_corr takes O(n). Total: O(n_cand² · n) = O(100 · 400) = 40K per sweep. That's fast!

But I also need to compute the max of corr + delta * d_corr, which is another O(n) per pair. So total per sweep: O(n_cand² · n) = 40K. With 5000 sweeps: 200M. In numpy, this should take maybe 5-10 seconds. Good.

Let me implement this:

```python
def cd_fast(h0, n, target, time_limit, max_sweeps=10000, n_cand=10):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        
        k_star = int(np.argmax(corr)) - mid
        # Gradient at active shift
        g = np.zeros(n)
        if k_star >= 0:
            g[:n-k_star] += 1.0 - h[k_star:]
            g[k_star:] -= h[:n-k_star]
        else:
            l = -k_star
            g[l:] += 1.0 - h[:n-l]
            g[:n-l] -= h[l:]
        g -= np.mean(g)
        
        sorted_idx = np.argsort(g)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_c5_trial = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                
                # Compute d_corr for moving delta from i to j
                # d_corr[s] = sig_j[s] - sig_i[s]
                # where sig_p[s] = d(corr[s])/d(h[p])
                # = (1-h[p+k])·1_{p+k valid} - h[p-k]·1_{p-k valid} for k = s-mid
                
                d_corr = np.zeros(2*n-1)
                
                # For position i (decreasing):
                # sig_i[s] = (1-h[i+k])·1_{i+k<n} - h[i-k]·1_{i-k>=0}
                # d_corr gets -sig_i
                
                # For position j (increasing):
                # sig_j[s] = (1-h[j+k])·1_{j+k<n} - h[j-k]·1_{j-k>=0}
                # d_corr gets +sig_j
                
                # Combined: d_corr[s] = sig_j[s] - sig_i[s]
                # = [(1-h[j+k]) - (1-h[i+k])]*1_{valid} - [h[j-k] - h[i-k]]*1_{valid}
                # = (h[i+k] - h[j+k])*1_{both valid for their respective positions}
                #   - (h[j-k] - h[i-k])*1_{...}
                
                # It's easier to just compute them separately:
                for s in range(2*n-1):
                    k = s - mid
                    # sig_j contribution
                    val_j = 0.0
                    if 0 <= j+k < n:
                        val_j += 1.0 - h[j+k]
                    if 0 <= j-k < n:
                        val_j -= h[j-k]
                    # sig_i contribution
                    val_i = 0.0
                    if 0 <= i+k < n:
                        val_i += 1.0 - h[i+k]
                    if 0 <= i-k < n:
                        val_i -= h[i-k]
                    d_corr[s] = val_j - val_i
                
                # Line search: minimize max_s (corr[s] + delta * d_corr[s]) for delta in [0, d_hi]
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * d_corr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(d_corr[active])
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                
                if delta_star < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * d_corr) * dx
                if approx_max < best_c5_trial - 1e-15:
                    best_c5_trial = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5
```

The inner loop over s (799 iterations in Python) is the bottleneck. For 100 pairs per sweep, that's 80K Python iterations per sweep. With 10000 sweeps, that's 800M iterations. At ~100ns each, that's 80 seconds. Marginal.
