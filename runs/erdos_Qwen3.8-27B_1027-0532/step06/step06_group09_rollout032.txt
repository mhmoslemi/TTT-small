def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not mask.any():
        return h
    h = h.copy()
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

        else:

        if mask2.any():

            h = np.clip(h, 0.0, 1.0)

def smooth_obj_with_alpha(h, alpha, corr_cache=None):

def fast_pair_cd(h, n, target, time_limit, max_sweeps=100000,
                 n_cand=30, top_pairs=30, patience=500, reheat_interval=100):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(12345)
    
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    n_corr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                # Reheat: small random perturbation
                if no_improve > patience * 2:
                    break
                if no_improve % reheat_interval == 0:
                    h_pert = h + rng.randn(n) * 0.01
                    h_pert = project_feasible(h_pert, target)
                    if compute_c5(h_pert) < best_c5:
                        h = h_pert
                        no_improve = 0
                    else:
                        h = h  # keep current
                continue
        
        # Compute weighted contribution for active shifts
        c_max = np.max(corr)
        # Weight shifts by proximity to max
        exp_arg = np.clip((corr - c_max) * 15.0, -30.0, 0.0)
        weights = np.exp(exp_arg)
        w_sum = weights.sum()
        if w_sum < 1e-10:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= w_sum
        
        # Compute combined contribution
        contrib = np.zeros(n)
        # Only use shifts with significant weight
        active_mask = weights > 0.01 / n_corr  # threshold
        active_idx = np.where(active_mask)[0]
        
        for s in active_idx:
            m = int(s) - mid
            w = weights[s]
            # contrib[i] += w * [(1-h[i-m])*1[0<=i-m<n] - h[i+m]*1[0<=i+m<n]]
            ip = idxs - m
            im = idxs + m
            mask_p = (ip >= 0) & (ip < n)
            mask_m = (im >= 0) & (im < n)
            c_vals = np.zeros(n)
            c_vals[mask_p] = (1.0 - h[ip[mask_p]])
            c_vals[mask_m] -= h[im[mask_m]]
            contrib += w * c_vals
        
        # Select candidates
        sorted_idx = np.argsort(contrib)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        hp = high_set
        lp = low_set
        n_h = len(hp)
        n_l = len(lp)
        
        # First-order screening
        fo_matrix = contrib[lp][:, None] - contrib[hp][None, :]
        flat_idx = np.argsort(fo_matrix.ravel())
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_g = None
        
        n_checked = 0
        for flat_i in flat_idx:
            if n_checked >= top_pairs:
                break
            a = flat_i // n_l
            b = flat_i % n_l
            i = int(hp[a])
            j = int(lp[b])
            if i == j:
                continue
            n_checked += 1
            
            delta_lo = max(0.0, -h[j])
            delta_hi = min(1.0 - h[j], h[i])
            if delta_hi - delta_lo < 1e-12:
                continue
            
            # Compute g
            i_sp = i - shifts
            i_sm = i + shifts
            j_sp = j - shifts
            j_sm = j + shifts
            
            g = np.zeros(n_corr)
            mask = (i_sp >= 0) & (i_sp < n)
            g[mask] += (1.0 - h[i_sp[mask]])
            mask = (i_sm >= 0) & (i_sm < n)
            g[mask] -= h[i_sm[mask]]
            mask = (j_sp >= 0) & (j_sp < n)
            g[mask] -= (1.0 - h[j_sp[mask]])
            mask = (j_sm >= 0) & (j_sm < n)
            g[mask] += h[j_sm[mask]]
            
            # Bisection
            a_db = delta_lo
            b_db = delta_hi
            for _ in range(12):
                mid_db = 0.5 * (a_db + b_db)
                vals = corr + mid_db * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-12
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-10:
                    b_db = mid_db
                else:
                    a_db = mid_db
            
            delta_star = 0.5 * (a_db + b_db)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-14:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
                best_g = g
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

Wait, I realize there might be an issue with the "fo_matrix" computation. Let me double-check the sign convention.

When we move delta from i to j (h[i] -= delta, h[j] += delta):
- d(corr[s])/d(delta) = [d/d(delta) h[i]] * [∂corr[s]/∂h[i]] + [d/d(delta) h[j]] * [∂corr[s]/∂h[j]]
= (-1) * contrib[i, s] + (1) * contrib[j, s]
= -contrib[i, s] + contrib[j, s]

So g[s] = -contrib[i, s] + contrib[j, s] = contrib[j, s] - contrib[i, s]

For the first-order screening at the argmax shift s*:
g[s*] = contrib[j, s*] - contrib[i, s*]

We want g[s*] < 0 for improvement, which means contrib[j, s*] < contrib[i, s*].

Since i is in the "high" set (high contrib) and j is in the "low" set (low contrib), we typically have contrib[i, s*] > contrib[j, s*], so g[s*] < 0. Good.

The first-order change in C₅ ≈ delta * g[s*] * dx (for small delta).
Since g[s*] < 0, this is negative (improvement). ✓

Now, the fo_matrix[b, a] = contrib[lp[b]] - contrib[hp[a]] (for the active shifts, weighted).
We want the most negative values, which corresponds to the best pairs.
np.argsort(fo_matrix.ravel()) gives indices sorted from most negative to most positive. ✓

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not mask.any():
        return h
    h = h.copy()
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        if mask2.any():
            h[mask2] += residual / mask2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

            g[mask] -= (1.0 - h[j_sp[mask]])
            mask = (j_sm >= 0) & (j_sm < n)
            g[mask] += h[j_sm[mask]]
            
            # Bisection
            a_db = delta_lo
            b_db = delta_hi
            for _ in range(12):
                mid_db = 0.5 * (a_db + b_db)
                vals = corr + mid_db * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-12
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-10:
                    b_db = mid_db
                else:
                    a_db = mid_db
            
            delta_star = 0.5 * (a_db + b_db)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-14:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
