        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        if damp_mode == 'none':
            h = h_new
        elif damp_mode == 'adaptive':
            damp = min(1.0, 0.5 + 0.005 * it)
            h = (1.0 - damp) * h + damp * h_new
            h = project_feasible(h, target)
        else:
            damp = float(damp_mode)
            h = (1.0 - damp) * h + damp * h_new
            h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.array([np.sum(h) - _global_target])

def cons_jac(h):
    return np.array([np.ones(_global_n)])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.01
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - (n - 1)
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5
