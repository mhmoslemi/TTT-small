2. Finding the lower envelope
3. Finding the minimum of the lower envelope on [t_lo, t_hi]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts, lo=0.0, hi=1.0):
    """Solve: min_t max_k (slopes[k]*t + intercepts[k]) s.t. lo <= t <= hi"""
    # The function f(t) = max_k (slopes[k]*t + intercepts[k]) is convex piecewise linear
    # Its minimum is where the "knee" is, or at a boundary
    # Use binary search on the subgradient
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-13 * max(1.0, abs(max_val))
        active = np.where(np.abs(vals - max_val) <= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        if np.all(slopes[active] > 1e-14):
            hi = mid
        elif np.all(slopes[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def pair_descent(h, target, n_active, max_passes, time_limit, rng):

    start = time.time()
    n = len(h)
    dx = 2.0 / n
    half_n = n - 1
    h = h.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    # For shift k (0-indexed in corr), the valid range is:
    # k_shift = k - half_n
    # valid i: max(0, k_shift) <= i < min(n, n + k_shift)
    
    for p in range(max_passes):

            break
        
        # Random pair selection, biased toward "active" indices
        # Choose pairs more likely to be in the overlap regions
        for sub in range(n // 2):  # n/2 pairs per pass
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:

            # Δh[i] = t, Δh[j] = -t (to maintain sum)

            #               + dx * [(1-h[j+k])*1_{j+k valid} - h[j-k]*1_{j-k valid}] * (-t)
            #   = dx * [(1-h[i+k])*1_{i+k valid} - h[i-k]*1_{i-k valid} - (1-h[j+k])*1_{j+k valid} + h[j-k]*1_{j-k valid}] * t
            
            d = np.zeros(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                # Effect of h[i] change:
                a_i = 0.0
                if 0 <= i + k < n:
                    a_i += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    a_i -= h[i - k]
                # Effect of h[j] change:
                a_j = 0.0
                if 0 <= j + k < n:
                    a_j += (1.0 - h[j + k])
                if 0 <= j - k < n:
                    a_j -= h[j - k]
                d[idx] = dx * (a_i - a_j)  # coefficient of t
                
            # The overlap for shift k at current h:
            base = np.zeros(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                base[idx] = corr[k_idx]  # this is sum h[m](1-h[m+k]) without dx

            # = dx * max_k (base[k] + d[k]*t)
            # So minimize max_k (base[k] + d[k]*t)
            # Subject to: -h[i] <= t <= 1-h[i] AND h[j]-1 <= t <= h[j]
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            
            if t_hi - t_lo < 1e-14:

            t_opt = solve_1d_max_line(d, base, t_lo, t_hi)
            h[i] += t_opt
            h[j] -= t_opt
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:

            best_h = h.copy()

        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5

= overlap_k(h_c) - sum_{i=0}^{n-k-1} h_c[i]

So the constraint is:
sum_j g_j * h[j] ≤ t + overlap_k(h_c) - sum_{i=0}^{n-k-1} h_c[i]

Or equivalently:
sum_j g_j * h[j] - overlap_k(h_c) + sum_{i=0}^{n-k-1} h_c[i] ≤ t

Let me verify at h = h_c:
LHS = sum_j g_j h_c[j] - overlap_k(h_c) + sum_{i=0}^{n-k-1} h_c[i]
= [2*overlap_k(h_c) - sum_{i=0}^{n-k-1} h_c[i]] - overlap_k(h_c) + sum_{i=0}^{n-k-1} h_c[i]
= overlap_k(h_c)

And we want LHS ≤ t, with LHS = overlap_k(h_c) at h = h_c. So if t = overlap_k(h_c), the constraint is tight at h_c. Good.

So the LP is:
minimize t
subject to: for each active k:
    sum_j g_j^k * h[j] - overlap_k(h_c) + sum_{i=0}^{n-|k|-1} h_c[i] ≤ t
    sum(h) = target
    0 ≤ h ≤ 1

This is a standard LP with n+1 variables and (n_active + 3) constraints (plus the box constraints).

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
