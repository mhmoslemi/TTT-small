```python
                # So dB(k) = -d*(h[i0-k] + h[i0+k]) + d*(h[j0-k] + h[j0+k])
                #           = d*(h[j0-k] + h[j0+k] - h[i0-k] - h[i0+k])
                # Change in corr[s] = -dB(k) = -d*(h[j0-k] + h[j0+k] - h[i0-k] - h[i0+k])
                #                   = d*(h[i0-k] + h[i0+k] - h[j0-k] - h[j0+k])
                
                # So dcorr[s] (per unit delta) = h[i0-k] + h[i0+k] - h[j0-k] - h[j0+k]
                # where k = s - mid, with boundary conditions (out of range = 0)
                
                i0 = i
                j0 = j
                for s in range(len(corr)):
                    k = s - mid
                    val = 0.0
                    if 0 <= i0 - k < n:
                        val += h[i0 - k]
                    if 0 <= i0 + k < n:
                        val += h[i0 + k]
                    if 0 <= j0 - k < n:
                        val -= h[j0 - k]
                    if 0 <= j0 + k < n:
                        val -= h[j0 + k]
                    dcorr[s] = val
                
                # Now bisect on delta
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-12
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    elif f_prime < -1e-14:
                        a_d = mid_d
                    else:
                        break
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    global _G_n, _G_target, _G_alpha
    start = time.time()
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand, n_top_shifts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    k_arr = np.arange(2 * n - 1) - mid
    no_improve = 0

    def safe_get(idx):
        mask = (idx >= 0) & (idx < n)
        out = np.zeros(2 * n - 1)
        out[mask] = h[idx[mask]]
        return out

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_top_shifts:]
        best_trial = c5_val
        best_move = None

        for s_top in top_shifts:
            k_star = int(s_top) - mid
            grad = get_grad_shift(h, k_star, n)
            grad -= np.mean(grad)
            sorted_idx = np.argsort(grad)
            low_set = sorted_idx[:n_cand]
            high_set = sorted_idx[-n_cand:]

            for i in high_set:
                if h[i] < 1e-14:
                    continue
                hi_arr = safe_get(i - k_arr) + safe_get(i + k_arr)
                for j in low_set:
                    if i == j or h[j] > 1.0 - 1e-14:
                        continue
                    hj_arr = safe_get(j - k_arr) + safe_get(j + k_arr)
                    dcorr = hi_arr - hj_arr

                    a_d = 0.0
                    b_d = min(h[i], 1.0 - h[j])
                    if b_d <= 1e-14:
                        continue
                    for _ in range(18):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * dcorr
                        f_mid = np.max(vals)
                        active = vals >= f_mid - 1e-12
                        if active.any():
                            f_prime = np.max(dcorr[active])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-14:
                            b_d = mid_d
                        elif f_prime < -1e-14:
                            a_d = mid_d
                        else:
                            break
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                    if trial_c5 < best_trial:
                        best_trial = trial_c5
                        best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def block_cd(h_start, n, target, time_limit, block_size, max_sweeps):
    start = time.time()
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_blocks = n // block_size
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 20 * n_blocks:
            break

        improved = False
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            i0 = b * block_size
            i1 = i0 + block_size
            h_block = h[i0:i1].copy()
            block_sum = np.sum(h_block)

            def block_obj(hb):
                h_trial = h.copy()
                h_trial[i0:i1] = np.clip(hb, 0.0, 1.0)
                h_trial = project_feasible(h_trial, target)
                return compute_c5(h_trial)

            cons_block = [{'type': 'eq',
                           'fun': lambda hb: np.sum(hb) - block_sum,
                           'jac': lambda hb: np.ones(block_size)}]
            bounds_block = [(0.0, 1.0)] * block_size
            try:
                res = minimize(block_obj, h_block, method='SLSQP',
                              bounds=bounds_block, constraints=cons_block,
                              options={'maxiter': 200, 'ftol': 1e-12})
                h_new_block = project_feasible(res.x, block_sum)
                h_trial = h.copy()
                h_trial[i0:i1] = h_new_block
                h_trial = project_feasible(h_trial, target)
                c5_trial = compute_c5(h_trial)
                c5_cur = compute_c5(h)
                if c5_trial < c5_cur - 1e-14:
                    h = h_trial
                    improved = True
            except Exception:
                pass

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def soft_weighted_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 1000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]
        if len(sig_idx) == 0:
            break
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    return best_h, best_c5

                if tleft() > 40:
                    h_pol, sc_pol = slsqp_optimize(best_h, n_try, target_try,
                                                   [50000, 200000, 1000000],
                                                   1000, 1e-16, time_limit=min(30, tleft() * 0.15))
                    if sc_pol < best_score:
                        best_score = sc_pol
                        best_h = h_pol

    # Phase 10: Final intensive pair CD
    if tleft() > 80:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cd, sc = pair_cd(best_h, n_cur, tgt_cur,
                           time_limit=min(50, tleft() * 0.4),
                           max_sweeps=8000, n_cand=35, n_top_shifts=3)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_final, sc = slsqp_optimize(best_h, n_cur, tgt_cur,
                                     [100000, 500000, 2000000],
                                     2000, 1e-16, time_limit=min(25, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
