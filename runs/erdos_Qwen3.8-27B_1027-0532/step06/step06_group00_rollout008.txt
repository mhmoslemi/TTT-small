import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(lo, hi) for lo, hi in zip(np.zeros(n), np.ones(n))]
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=70, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

This is exactly what the CD method uses for its gradient computation! So the CD method is essentially a subgradient method with a clever step size selection.

def nesterov_subgrad(h_init, n, target, time_limit, n_shifts=5, 
                      alpha_start=0.01, decay=0.999, momentum_init=0.9):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    y = h.copy()  # Nesterov's y
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = momentum_init
    
    for t in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Compute subgradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_y = float(np.max(corr)) * dx
        
        # Top shifts
        top_idx = np.argsort(corr)[-n_shifts:]
        # Weights (softmax-like)
        vals = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(100 * vals)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            k = int(s_idx) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Step size (Polyak-like)
        # f(h) - f* ≈ c5_h - c5_lb (some lower bound)
        # For now use fixed decaying step
        alpha = alpha_start * (decay ** t)
        alpha = max(alpha, 1e-6)
        
        # Update
        v = mu * v - alpha * grad
        h_new = project_feasible(h + v, target)
        
        # Nesterov update
        mu_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
        y = h_new + (mu / mu_new - 1.0) * (h_new - h)
        
        h = h_new
        mu = mu_new
        
        # Track best
        c5_h = compute_c5(h)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()
    
    return best_h, best_c5

import cvxpy as cp

def lp_cutting_plane(h_init, n, target, time_limit, max_iters=100, damping=0.5):
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Cut storage: (gradient, value) pairs
    cuts = []
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute current gradient and value for all shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx  # need dx
        dx_local = 2.0 / n
        c5_val = float(np.max(corr)) * dx_local
        
        # Add cut for current point
        # For each k: g_k(h) = corr[k] * dx_local
        # grad_k: ∇_h g_k
        # We need the top few shifts
        top_k = np.argsort(corr)[-20:]
        
        for k_idx in top_k:
            k = int(k_idx) - mid
            gk_val = corr[k_idx] * dx_local
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k]) * dx_local
            grad_k[v2] -= h[idxs[v2] + k] * dx_local
            cuts.append((grad_k, gk_val))
        
        # Keep only the most useful cuts (top 50 by value)
        if len(cuts) > 100:
            # Remove cuts that are far below the current max
            sorted_cuts = sorted(cuts, key=lambda c: c[1], reverse=True)
            cuts = sorted_cuts[:50]
        
        # Solve LP: min t s.t. g_k + grad_k^T (h' - h) <= t, 0<=h'<=1, sum(h')=target
        # Using active cuts only
        if not cuts:
            break
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for grad_k, gk_val in cuts:
            # Linearized: gk_val + grad_k^T (h_var - h) <= t_var
            constraints.append(gk_val + grad_k @ (h_var - h) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, tol=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.array(h_var.value).flatten()
        
        # Damped step
        alpha = damping * 0.5  # half step
        h_new = project_feasible(h + alpha * (h_lp - h), target)
        
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            alpha = 0.25
            h_new = project_feasible(h + alpha * (h_lp - h), target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=300,
            n_top_shifts=1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_top_shifts:]
            top_vals = corr[top_idx]
            w_sum = top_vals.sum()
            grad = np.zeros(n)
            for ti in top_idx:
                k = int(ti) - mid
                w = corr[ti] / w_sum
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def shake_and_refine(h_init, n, target, time_limit, seed=0,
                     noise_scales=None, cd_time_frac=0.6):
    if noise_scales is None:
        noise_scales = [0.05, 0.10, 0.15, 0.08, 0.12, 0.20, 0.06, 0.04]
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_init.copy().astype(float)
    best_c5 = compute_c5(best_h)
    total_time = time_limit

    for ns in noise_scales:
        elapsed = time.time() - start
        remaining = total_time - elapsed
        if remaining < 20:
            break
        cd_time = min(remaining * cd_time_frac, 120.0)
        h_pert = best_h + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=cd_time,
                              max_sweeps=20000, n_cand_start=25, n_cand_end=70,
                              max_no_improve=200, n_top_shifts=2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    return best_h, best_c5


def block_perturb_cd(h_init, n, target, time_limit, seed=1, max_rounds=50):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    change = compute_change_matrix(h, n)
    no_improve = 0
    block_sizes = [3, 5, 7, 4, 6]

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if no_improve > 30:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        best_trial_c5 = c5_val
        found = False

        for _attempt in range(8):
            bs = block_sizes[rng.randint(len(block_sizes))]
            pos = int(rng.randint(0, n - bs + 1))
            block_idx = np.arange(pos, pos + bs)
            block_grad_mean = np.mean(grad[block_idx])
            if block_grad_mean > 0:
                direction = -1.0
                partner_mask = grad < np.percentile(grad, 10)
            else:
                direction = 1.0
                partner_mask = grad > np.percentile(grad, 90)

            partner_idx = np.where(partner_mask)[0]
            if len(partner_idx) == 0:
                continue
            partner = int(partner_idx[rng.randint(len(partner_idx))])
            if partner in block_idx:
                continue

            max_block_change = direction * min(
                np.min(h[block_idx]) if direction < 0 else np.min(1.0 - h[block_idx]),
                h[partner] if direction > 0 else 1.0 - h[partner]
            )
            if max_block_change <= 1e-12:
                continue

            g_total = np.zeros(2 * n - 1)
            for bi in block_idx:
                g_total += direction * (1.0 / bs) * change[bi]
            if direction > 0:
                g_total -= change[partner]
            else:
                g_total += change[partner]

            a_d = 1e-12
            b_d = max_block_change
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_total
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g_total[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g_total) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                found = True
                for bi in block_idx:
                    h[bi] += direction * delta_star / bs
                if direction > 0:
                    h[partner] -= delta_star
                else:
                    h[partner] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for bi in block_idx:
                    update_change_row(change, bi, h, n, shifts)
                update_change_row(change, partner, h, n, shifts)
                break

        if found:
            no_improve = 0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            no_improve += 1

    return best_h, best_c5
