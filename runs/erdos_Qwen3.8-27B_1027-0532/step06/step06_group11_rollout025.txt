def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def quadratic_residue_h(n):

            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 1
    
    h = np.zeros(n)
    for i in range(1, n):
        if pow(i, (p-1)//2, p) == 1:  # i is a QR mod p
            h[i] = 1.0

def simulated_annealing_binary(h_init, n, target, time_limit, rng):
    start = time.time()
    h = h_init.copy().astype(float)
    # Ensure binary
    h = (h > 0.5).astype(float)
    h = project_feasible(h, target)
    # Re-ensure binary after projection
    # project_feasible might make it non-binary, so let's do it differently
    ones = np.where(h > 0.5)[0]
    if len(ones) != n//2:
        # Fix
        if len(ones) > n//2:
            excess = len(ones) - n//2
            idx = rng.choice(ones, size=excess, replace=False)
            h[idx] = 0.0
        else:
            deficit = n//2 - len(ones)
            zeros = np.where(h < 0.5)[0]
            idx = rng.choice(zeros, size=deficit, replace=False)
            h[idx] = 1.0
    
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    
    T = 0.1  # initial temperature
    T_min = 0.001
    cooling = 0.999
    
    ones_mask = (h == 1.0)
    ones_list = np.where(ones_mask)[0].tolist()
    zeros_list = np.where(~ones_mask)[0].tolist()
    
    iter_count = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        if T < T_min:
            break
        if no_improve > 2000:
            T *= 0.5
            no_improve = 0
        
        # Choose a move
        # Strategy: with probability 0.7, target the worst shift
        # With probability 0.3, random swap
        if rng.random() < 0.7:
            corr = np.correlate(h, 1-h, mode='full')
            worst_idx = np.argmax(corr)
            k_star = worst_idx - (n-1)
            
            # Find bad positions (h[i]=1, h[i-k_star]=0) and good positions
            if k_star >= 0:
                valid_i = np.arange(max(k_star, 0), n)
            else:
                valid_i = np.arange(0, min(n, n + k_star))
            
            bad = [i for i in valid_i if h[i] == 1.0 and (0 <= i-k_star < n) and h[i-k_star] == 0.0]
            good = [j for j in range(n) if h[j] == 0.0 and (0 <= j-k_star < n) and h[j-k_star] == 1.0]
            
            if bad and good:
                pos_i = int(rng.choice(bad))
                # Find a good j that minimizes the impact on other shifts
                # For now, just pick randomly from good
                pos_j = int(rng.choice(good))
            else:
                pos_i = int(rng.choice(ones_list))
                pos_j = int(rng.choice(zeros_list))
        else:
            pos_i = int(rng.choice(ones_list))
            pos_j = int(rng.choice(zeros_list))
        
        # Evaluate the swap
        h[pos_i] = 0.0
        h[pos_j] = 1.0
        new_score = compute_c5(h)
        
        # Accept or reject
        delta = new_score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_score = new_score
            # Update lists
            ones_list.remove(pos_i)
            ones_list.append(pos_j)
            zeros_list.remove(pos_j)
            zeros_list.append(pos_i)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[pos_i] = 1.0
            h[pos_j] = 0.0
            no_improve += 1
        
        T *= cooling
        iter_count += 1
    
    return best_h, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
