from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h.copy()
    # Redistribute
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    step = diff / len(idx)
    h[idx] += step
    h = np.clip(h, 0.0, 1.0)
    # Fix residual with a second pass
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx2 = np.where(h > 1e-15)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def autocorr(h, dx):
    """Compute autocorrelation R(k) for all lags."""
    return np.correlate(h, h, mode='full') * dx

def smooth_max_obj(h, dx, alpha):
    """Smooth max of overlap = smooth min of autocorrelation."""
    R = autocorr(h, dx)
    R_min = np.min(R)
    shifted = -alpha * (R - R_min)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    # Smooth min ≈ R_min + log_sum/alpha
    # We want to maximize this, so minimize its negative
    return -(R_min + log_sum / alpha)

def smooth_max_grad(h, dx, alpha):
    """Gradient of smooth max of overlap (= negative of smooth min of autocorr)."""
    n = len(h)
    R = autocorr(h, dx)
    R_min = np.min(R)
    shifted = -alpha * (R - R_min)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # grad of smooth min of R with respect to h[j]:
    # = sum_l w[l] * (h[j-l] + h[j+l]) * dx  (where l ranges over valid lags)
    # = 2 * (W * h)[j] where W[l] = w[l] * dx and w is symmetric
    # Using the indexing: w[m] corresponds to lag l = m - (n-1)
    
    # convolve approach:
    # (W_shift * h)[j] = np.convolve(w_arr, h, 'full')[n-1 + j]
    # where w_arr[m] = w[m] (w is already indexed 0 to 2n-2)
    
    conv = np.convolve(w, h, 'full')
    # We want indices n-1 to 2n-2 of the convolution
    grad_min = 2.0 * conv[n-1 : 2*n-1]
    
    # This is the gradient of the smooth minimum of R
    # We want to minimize the smooth MAX of overlap = minimize (-smooth min of R)
    # So the gradient of our objective is -grad_min
    grad = -grad_min
    
    # Project: subtract mean to respect sum constraint
    grad = grad - np.mean(grad)
    
    return grad

def projected_gradient_optimize(h_init, dx, target, alphas, steps_per_alpha, time_limit):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    
    for alpha in alphas:
        v = np.zeros(n)  # momentum
        for step in range(steps_per_alpha):
            if time.time() - start_t > time_limit:
                return best_h, best_c5
            
            # Compute gradient
            R = np.correlate(h, h, 'full') * dx
            R_min = np.min(R)
            shifted = -alpha * (R - R_min)
            shifted = np.clip(shifted, -500, 0)
            exp_s = np.exp(shifted)
            w = exp_s / np.sum(exp_s)
            
            conv = np.convolve(w, h, 'full')
            grad = -2.0 * dx * conv[n-1 : 2*n-1]
            grad -= np.mean(grad)  # project onto constraint
            
            # Momentum
            v = 0.9 * v + grad
            
            # Line search (simple backtracking)
            lr = 0.1
            base_obj = smooth_max_obj(h, dx, alpha)
            for _ in range(30):
                h_try = project_feasible(h - lr * v, target)
                obj_try = smooth_max_obj(h_try, dx, alpha)
                if obj_try < base_obj:
                    h = h_try
                    break
                lr *= 0.5
            
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h.copy()

Now, for the pair CD (coordinate descent on pairs), I want to make it more efficient. The key insight is:

When we change h[i] by -δ and h[j] by +δ (keeping the sum constant), the change in the autocorrelation is:
ΔR[m] = -δ * (∂R[m]/∂h[i] at current h) + δ * (∂R[m]/∂h[j] at current h)
= δ * (∂R[m]/∂h[j] - ∂R[m]/∂h[i])

And ∂R[m]/∂h[j] = dx * (h[j-l_m] + h[j+l_m]) (with boundary handling)

So ΔR[m] = δ * dx * ((h[j-l_m] + h[j+l_m]) - (h[i-l_m] + h[i+l_m]))

The new smooth max (or true max) is what we want to minimize.

For the true max: we want to find δ that minimizes max_m (R[m] + ΔR[m]).

This is a piecewise-linear function of δ, and the minimum can be found by bisection.

Let me precompute for each pair (i,j) the vector ΔR/δ = dx * ((h[j-l_m] + h[j+l_m]) - (h[i-l_m] + h[i+l_m])) for all m.

But this is O(n) per pair, and there are O(n²) pairs, so O(n³) total. For n=400, that's 64 million operations per sweep, which is feasible.

Actually, let me be smarter. I'll precompute the "influence vector" for each index:
inf[j][m] = ∂R[m]/∂h[j] / dx = h[j-l_m] + h[j+l_m] (with boundary handling)

This is a n × (2n-1) matrix. For n=400, that's 400 × 799 ≈ 320K entries, which is fine.

Then for a pair (i,j), ΔR[δ] = δ * dx * (inf[j] - inf[i]).

The problem is: minimize max_m (R[m] + δ * dx * (inf[j][m] - inf[i][m]))

subject to 0 ≤ h[i] - δ ≤ 1 and 0 ≤ h[j] + δ ≤ 1.

This is a 1D problem in δ, and the objective is piecewise-linear (max of linear functions). The minimum is at a breakpoint.

For efficiency, I can use the same bisection approach as the previous code, but with the precomputed influence vectors.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_influence(h, dx):
    """Compute the influence matrix: inf[j][m] = dR[m]/dh[j] / dx"""
    n = len(h)
    ncorr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)  # lags from -(n-1) to (n-1)
    
    inf = np.zeros((n, ncorr))
    for j in range(n):
        # inf[j][m] = h[j - l_m] + h[j + l_m] where l_m = m - mid
        lags = shifts  # lags[m] = m - mid for m = 0, ..., 2n-2
        # For each lag l, h[j-l] is valid if 0 <= j-l < n
        jp = j - lags  # index for h[j-l]
        jm = j + lags  # index for h[j+l]
        valid_p = (jp >= 0) & (jp < n)
        valid_m = (jm >= 0) & (jm < n)
        val_p = np.zeros(ncorr)
        val_m = np.zeros(ncorr)
        val_p[valid_p] = h[jp[valid_p]]
        val_m[valid_m] = h[jm[valid_m]]
        inf[j] = val_p + val_m
    return inf

def smooth_obj_and_grad(h, dx, alpha):
    """Compute smooth min of autocorrelation (to maximize) and its gradient."""
    n = len(h)
    R = np.correlate(h, h, 'full') * dx
    R_min = np.min(R)
    shifted = -alpha * (R - R_min)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    # Smooth min value
    val = -(R_min + np.log(np.sum(exp_s)) / alpha)
    
    # Gradient of smooth min (to maximize)
    conv = np.convolve(w, h, 'full')
    grad_min = 2.0 * dx * conv[n-1 : 2*n-1]
    grad = -grad_min  # gradient of smooth max (to minimize)
    grad -= np.mean(grad)
    
    return val, grad

def fft_gradient_descent(h_init, dx, target, alphas, steps, time_limit, momentum=0.85, base_lr=0.5):
    """Optimize using FFT-based gradient with momentum and line search."""
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    start_t = time.time()
    
    for alpha in alphas:
        for step in range(steps):
            if time.time() - start_t > time_limit:
                return best_h, best_c5
            
            _, grad = smooth_obj_and_grad(h, dx, alpha)
            v = momentum * v + grad
            
            # Backtracking line search
            base_obj = smooth_obj_and_grad(h, dx, alpha)[0]
            lr = base_lr / (1.0 + 0.05 * step)
            for _ in range(50):
                h_try = project_feasible(h - lr * v, target)
                obj_try = smooth_obj_and_grad(h_try, dx, alpha)[0]
                if obj_try < base_obj - 1e-15:
                    h = h_try
                    break
                lr *= 0.5
            
            if step % 50 == 0:
                c5_cur = compute_c5(h)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
    
    c5_cur = compute_c5(h)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()
    
    return best_h, best_c5

        sorted_idx = np.argsort(-grad)  # indices where we want to increase h

        # - Decrease h where grad is negative

        high_set = sorted_idx[-n_cand:]  # most positive grad → increase h
        low_set = sorted_idx[:n_cand]  # most negative grad → decrease h
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            for j in low_set:
                if i == j or h[i] < 1e-10 or h[j] > 1.0 - 1e-10:
                    continue

                # h[i] += delta, h[j] -= delta
                # delta in (0, min(h[j], 1-h[i])]
                delta_max = min(h[j], 1.0 - h[i])

                # ΔR[m] = delta * ∂R[m]/∂h[i] - delta * ∂R[m]/∂h[j]
                # = delta * dx * (base_inf[i][m] - base_inf[j][m])
                
                deltaR_per_unit = (base_inf[i] - base_inf[j]) * dx
                
                # We want to minimize max_m (1 - R[m] - deltaR_per_unit[m] * delta)
                # = 1 - min_m (R[m] + deltaR_per_unit[m] * delta)
                # So we want to maximize min_m (R[m] + deltaR_per_unit[m] * delta)
                
                # This is a concave function of delta (min of linear functions)
                # Find delta* in [0, delta_max] that maximizes it
                
                # Bisection: the function is concave, so we can use ternary search or bisection on derivative
                lo, hi = 0.0, delta_max
                for _ in range(30):
                    mid_d = 0.5 * (lo + hi)
                    R_new = R + mid_d * deltaR_per_unit
                    # Check derivative at mid_d:

                    # If derivative > 0, function is still increasing → move hi down

                base_inf[j_ch] = val_p + val_m
    
    return best_h, best_c5
