Let me implement a projected subgradient method:

```python
def projected_subgradient(h_start, n, target, time_limit, alpha_start=100, alpha_end=200000, max_iter=100000):
    h = h_start.copy()
    dx = 2.0/n
    best = h.copy()
    best_score = compute_c5(best)
    
    for t in range(max_iter):
        # Compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        c_max = np.max(corr)
        
        # Log-sum-exp gradient
        alpha = alpha_start * (alpha_end/alpha_start) ** (t/max_iter)
        weighted = np.exp(alpha * (corr - c_max))
        weights = weighted / np.sum(weighted)
        
        # Compute gradient as weighted sum of individual gradients
        grad = np.zeros(n)
        idxs = np.arange(n)
        for s_idx, w in enumerate(weights):
            if w < 1e-10: continue
            k = s_idx - (n-1)
            valid_pos = (idxs + k >= 0) & (idxs + k < n)
            valid_neg = (idxs - k >= 0) & (idxs - k < n)
            grad[valid_pos] += w * (1 - h[idxs[valid_pos] + k])
            grad[valid_neg] -= w * h[idxs[valid_neg] - k]
        
        # Step size (diminishing)
        step = 1.0 / (t + 1)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
        
        h = h_new
    
    return best, best_score
```

      return float(np.max(corr)) * dx

  def project_onto_C(x, target):
      """Project x onto {0<=h<=1, sum(h)=target} using bisection on lambda."""
      n = len(x)

          else:

          if mask.sum() > 0:
