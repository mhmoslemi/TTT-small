    active_mask = corr > T_corr
    if not active_mask.any():
        return np.zeros(n), corr
    
    # Weight active shifts by their excess
    weights = np.zeros(2 * n - 1)
    weights[active_mask] = (corr[active_mask] - T_corr)
    
    # Normalize weights
    w_sum = weights.sum()
    if w_sum < 1e-15:

    frac = target - n_fill
    s[sorted_idx[:n_fill]] = 1.0
    if frac > 1e-12 and n_fill < n:

    return s

                    f_mid = np.max(vals)

                    best_j = j

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j + shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 30:
                break

    return best_h, best_c5

def projected_subgradient_threshold(h_init, n, target, T_curr, time_limit, lr_init=0.02, max_iter=3000):
    """Projected subgradient method on threshold objective."""
    start = time.time()
    dx = 2.0 / n
    T_corr = T_curr / dx
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g, corr = threshold_grad(h, T_corr, n_active=10)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        g_normalized = g / g_norm
        h_try = project_feasible(h - lr * g_normalized * g_norm, target)
        c5_now = compute_c5(h_try)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_try.copy()
            h = h_try
        else:
            lr *= 0.8
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    h = h.copy()

    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
