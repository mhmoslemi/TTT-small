import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def newton_refine(h_init, n, target, time_limit, alphas=None, max_iter_per_alpha=200):
    """Newton-type method with log-sum-exp smoothing."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    if alphas is None:
        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for it in range(max_iter_per_alpha):
            if time.time() - start > time_limit:
                break
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            m_val = np.max(corr)
            
            # Softmax weights (numerically stable)
            exp_arg = np.clip(alpha * (corr - m_val), -30, 0)
            w = np.exp(exp_arg)
            S = w.sum()
            w = w / S
            w_idx = np.where(w > 1e-12)[0]
            
            # Gradient
            G = np.zeros(n)
            g_vectors = []
            for k_idx in w_idx:
                k = int(k_idx) - mid
                g_k = np.zeros(n)
                if k == 0:
                    g_k = dx * (1.0 - 2.0 * h)
                else:
                    vp = (idxs + k >= 0) & (idxs + k < n)
                    vm = (idxs - k >= 0) & (idxs - k < n)
                    g_k[vp] += dx * (1.0 - h[idxs[vp] + k])
                    g_k[vm] -= dx * h[idxs[vm] - k]
                G += w[k_idx] * g_k
                g_vectors.append((w[k_idx], g_k, k))
            
            # Project gradient
            G_mean = G.mean()
            G_proj = G - G_mean
            
            if np.max(np.abs(G_proj)) < 1e-10:
                break
            
            # Hessian
            H = np.zeros((n, n))
            E_ggT = np.zeros((n, n))
            for wi, g_k, k in g_vectors:
                # Correlation Hessian contribution
                if k == 0:
                    H[np.arange(n), np.arange(n)] += wi * (-2.0 * dx)
                else:
                    lo = max(0, -k)
                    hi = min(n, n - k)
                    for i in range(lo, hi):
                        H[i, i+k] -= wi * dx
                        H[i+k, i] -= wi * dx
                # E[gg^T] contribution
                E_ggT += wi * np.outer(g_k, g_k)
            
            H += alpha * (E_ggT - np.outer(G, G))
            H = (H + H.T) / 2.0  # enforce symmetry
            
            # Regularize
            mu = 1e-4
            for _ in range(10):
                try:
                    step = np.linalg.solve(H + mu * np.eye(n), -G_proj)
                    break
                except np.linalg.LinAlgError:
                    mu *= 10.0
            else:
                step = -G_proj / (np.linalg.norm(G_proj) + 1e-10)
            
            # Backtracking line search
            c5_cur = compute_c5(h)
            found = False
            for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
                h_try = h + t * step
                if np.min(h_try) >= -0.01 and np.max(h_try) <= 1.01:
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-15:
                        h = h_try
                        found = True
                        break
            
            if not found:

                lr = 0.01
                for lr_try in [0.01, 0.005, 0.001, 0.0005]:
                    h_try = project_feasible(h - lr_try * G_proj, target)
                    if compute_c5(h_try) < c5_cur:
                        h = h_try
                        found = True
                        break
            
            if not found:
                break
            
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def pair_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand=40, patience=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            h_i_sp = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            h_i_sp[v_ip] = h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            h_i_sm = np.zeros(n_corr)
            h_i_sm[v_im] = h[i - shifts[v_im]]
            
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm = np.zeros(n_corr)
                jpm[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm = np.zeros(n_corr)
                jsm[jm] = h[j - shifts[jm]]
                
                g = h_i_sp + h_i_sm - jpm - jsm
                
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, delta_lo + 1e-10)
                
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

def project_feasible(h, target_sum):
