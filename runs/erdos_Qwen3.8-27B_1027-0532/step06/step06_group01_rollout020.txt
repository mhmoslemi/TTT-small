import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha, n_active=5):
    """Smooth max using log-sum-exp, focusing on top n_active"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Use all correlations but with high alpha to focus on max
    shifted = alpha * (corr - np.max(corr))
    shifted = np.clip(shifted, -100.0, 0.0)
    return float(np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha)

def active_set_obj(x, top_k=5):
    """Sum of top-k correlations as a smoother objective"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    sorted_corr = np.sort(corr)[::-1]
    return float(np.sum(sorted_corr[:top_k]) / top_k)

_global_target = 0.0

def _con_fun(x):
    return np.sum(x) - _global_target

def _con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, n_time=0):
    global _global_target
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

import numpy as np
from scipy.optimize import minimize
import time

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj_wrap(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -80.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def _con_fun(x):
    return np.sum(x) - _global_target

def _con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_target
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_obj_wrap, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def topk_mean_obj(x, top_k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    return float(np.mean(corr[idx]))

def slsqp_topk(h_start, n, target, top_k, maxiter, ftol):
    global _global_target
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    try:
        res = minimize(topk_mean_obj, h_cur, args=(top_k,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    except Exception:
        pass
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=100000, n_shifts=1, top_k_weight=2.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    ks_all = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1000:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - (n - 1)
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = top_k_weight
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - (n - 1)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks_all
            idx2 = m + ks_all
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(13):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= (f_mid - 1e-13)
                    if not np.any(active_mask):
                        break
                    f_prime = float(np.max(g[active_mask]))
                    if f_prime > 1e-13:
                        hi_d = mid
                    else:
                        lo = mid

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def round_and_reoptimize(h_init, n, target, time_limit):

        best_h = hcd

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_h(h, target, sigma):
    n = len(h)
    h_pert = h + sigma * np.random.randn(n)
    return project_feasible(h_pert, target)
