When we have h and we want to compute the effect of changing h[a]: 1→0 and h[b]: 0→1:

The correlation is c[k] = Σ_j h[j+k] * (1-h[j]) for valid j, j+k (where 0≤j<n and 0≤j+k<n).

After the change, h'[a]=0, h'[b]=1, h'=h elsewhere.

c'[k] - c[k] = Σ_j (h'[j+k]*(1-h'[j]) - h[j+k]*(1-h[j]))

The only terms that change are those involving index a or b.

Terms involving a:
- j+k = a (i.e., j = a-k): h'[a]*(1-h'[a-k]) - h[a]*(1-h[a-k]) = 0 - 1*(1-h[a-k] or h'[a-k])
  If b≠a-k: = -(1-h[a-k])
  If b=a-k: = -(1-1) = 0 (since h'[b]=1)
  Valid when 0≤a-k<n

- j = a: h'[a+k]*(1-h'[a]) - h[a+k]*(1-h[a]) = h'[a+k]*1 - h[a+k]*0
  If a+k≠b: = h[a+k]
  If a+k=b: = 1 (since h'[b]=1)
  Valid when 0≤a+k<n

Terms involving b:
- j+k = b (i.e., j = b-k): h'[b]*(1-h'[b-k]) - h[b]*(1-h[b-k])
  = 1*(1-h'[b-k]) - 0 = 1-h'[b-k]
  If b-k≠a: = 1-h[b-k]
  If b-k=a: = 1-0 = 1 (since h'[a]=0)
  Valid when 0≤b-k<n

- j = b: h'[b+k]*(1-h'[b]) - h[b+k]*(1-h[b])
  = h'[b+k]*0 - h[b+k]*1 = -h'[b+k]
  If b+k≠a: = -h[b+k]
  If b+k=a: = -0 = 0 (since h'[a]=0)
  Valid when 0≤b+k<n

So the total delta (ignoring the special cases where indices coincide, which are rare):

def subgradient_continuous(h, n_iters, time_limit, start_time, lr_init=None, momentum=0.9):
    """Nesterov-accelerated projected subgradient for min max_k corr_k(h)"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    
    if lr_init is None:
        lr_init = 0.05
    
    corr = get_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2*n-1) - (n-1)
    
    v = np.zeros(n)  # momentum
    stall = 0
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 5000:
            break
        
        # Subgradient: pick the shift with maximum correlation, use its gradient
        # With smoothing: use softmax-weighted average of top shifts
        max_corr = float(corr.max())
        alpha = 20.0 + 5.0 * it / n_iters  # increasing alpha (more focus on max)
        w = np.exp(alpha * (corr - max_corr))
        w /= w.sum()
        
        # Gradient of corr_k w.r.t. h[i]:
        # corr_k = sum_j h[j+s]*(1-h[j]) where s = shifts[k]
        # d(corr_k)/d(h[i]) = (1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n]
        # (from h[i] appearing as h[j+s] with j=i-s, and as (1-h[j]) with j=i)
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-10:
                continue
            s = int(shifts[k])
            # d(corr_k)/d(h[i]) for all i:
            # (1-h[i-s])*[0<=i-s<n] - h[i+s]*[0<=i+s<n]
            if s >= 0:
                # i-s valid: i >= s
                grad[s:] += wk * (1.0 - h[:n-s])
                # i+s valid: i <= n-1-s
                grad[:n-s] -= wk * h[s:]
            else:
                abs_s = -s
                # i-s = i+abs_s valid: i <= n-1-abs_s
                grad[:n-abs_s] += wk * (1.0 - h[abs_s:])
                # i+s = i-abs_s valid: i >= abs_s
                grad[abs_s:] -= wk * h[:n-abs_s]
        
        lr = lr_init / (1.0 + 0.01 * it)
        
        # Nesterov step
        y = h + momentum * v
        grad_y = grad  # approximate
        h_new = y - lr * grad_y
        h_new = project_box_sum(h_new, target)
        v = h_new - h
        h = h_new
        
        corr = get_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_delta_swap(h, a, b, lag, n):
    delta = np.zeros(2 * n - 1)
    idx = a - lag
    m = (idx >= 0) & (idx < n)
    delta[m] -= (1.0 - h[idx[m]])
    idx = a + lag
    m = (idx >= 0) & (idx < n)
    delta[m] += h[idx[m]]
    idx = b - lag
    m = (idx >= 0) & (idx < n)
    delta[m] += (1.0 - h[idx[m]])
    idx = b + lag
    m = (idx >= 0) & (idx < n)
    delta[m] -= h[idx[m]]
    return delta

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current > target_sum:
        need = current - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

    if diff < 0:

        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:

            can = y[idx]
            if can > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can
                y[idx] = 0.0

                    vi = ones[ones < n + s]
                if len(vi) > 0:
                    i = int(vi[rng.randint(len(vi))])
                else:
                    i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

        if i == j:
            continue

        delta = compute_delta_swap(h, i, j, lag, n)
        new_max = float((corr + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    return best_h, best_c5

def subgradient_nesterov(h, n_iters, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    shifts = np.arange(2 * n - 1) - (n - 1)

    corr = get_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    v = np.zeros(n)
    momentum = 0.85
    lr0 = 0.04
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 8000:
            break

        max_corr = float(corr.max())
        alpha = 30.0 + 10.0 * (it / max(n_iters, 1))
        w = np.exp(alpha * (corr - max_corr))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 5e-8:
                continue
            s = int(shifts[k])
            if s >= 0:
                grad[s:] += wk * (1.0 - h[:n - s])
                grad[:n - s] -= wk * h[s:]
            else:
                abs_s = -s
                grad[:n - abs_s] += wk * (1.0 - h[abs_s:])
                grad[abs_s:] -= wk * h[:n - abs_s]

        lr = lr0 / np.sqrt(1.0 + 0.05 * it)
        h = h + momentum * v - lr * grad
        h = project_box_sum(h, target)
        v = h - h + momentum * v
        h_new = project_box_sum(h + momentum * v - lr * grad, target) if False else project_box_sum(h, target)

        corr = get_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-10:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def projected_topk_grad(h, n_iters, topk, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    shifts = np.arange(2 * n - 1) - (n - 1)
    corr = get_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    stall = 0
    lr0 = 0.04

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 6000:
            break

        lr = lr0 / (1.0 + 0.02 * it)
        top_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in top_idx:
            s = int(shifts[k])
            if s >= 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
            else:
                abs_s = -s
                grad[:n - abs_s] += (1.0 - h[abs_s:])
                grad[abs_s:] -= h[:n - abs_s]
        grad /= topk

        h = h - lr * grad
        h = project_box_sum(h, target)

        corr = get_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-10:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_f = n_init / 2.0
    target_i = n_init // 2

    h_bin = project_binary(h_init, target_i)
    best_bin_score = compute_c5(h_bin)
    h_cont = project_box_sum(h_init, target_f)
    best_cont_score = compute_c5(h_cont)

    # Phase 1: Binary local search (aggressive)
    allowed = min(100.0, tleft() * 0.12)
    if allowed > 15:
        h_b, c5_b = binary_local_search(h_bin, max_iters=200000, rng=rng,
                                        time_limit=allowed, start_time=start_time)
        if c5_b < best_bin_score:
            best_bin_score = c5_b
            h_bin = h_b
        h_cont = project_box_sum(h_bin, target_f)
        sc = compute_c5(h_cont)
        if sc < best_cont_score:
            best_cont_score = sc
            h_cont = h_cont.copy()

    # Phase 2: SA binary
    allowed = min(180.0, tleft() * 0.18)
    if allowed > 25:
        n_sa = int(min(8000000, allowed * 50000))
        h_sa, c5_sa = sa_binary(h_bin, n_sa, 0.04, 0.0005, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            h_bin = h_sa
        h_c2 = project_box_sum(h_bin, target_f)
        sc = compute_c5(h_c2)
        if sc < best_cont_score:
            best_cont_score = sc
            h_cont = h_c2.copy()

    # Phase 3: Topk projected gradient
    allowed = min(120.0, tleft() * 0.13)
    if allowed > 15:
        n_it = int(min(3000000, allowed * 25000))
        h_pg, c5_pg = projected_topk_grad(h_cont, n_it, topk=5,
                                          time_limit=allowed, start_time=start_time)
        if c5_pg < best_cont_score:
            best_cont_score = c5_pg
            h_cont = h_pg

    # Phase 4: SLSQP (multi-alpha)
    allowed = min(150.0, tleft() * 0.18)
    if allowed > 25:
        h_ref, sc_ref = slsqp_refine(h_cont, n_init, target_f,
                                     alphas=[20, 80, 300, 1500, 8000],
                                     maxiter=4000, ftol=1e-14,
                                     time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            h_cont = h_ref

    # Phase 5: Nesterov subgradient
    allowed = min(100.0, tleft() * 0.10)
    if allowed > 15:
        n_it = int(min(2000000, allowed * 20000))
        h_ng, c5_ng = subgradient_nesterov(h_cont, n_it,
                                           time_limit=allowed, start_time=start_time)
        if c5_ng < best_cont_score:
            best_cont_score = c5_ng
            h_cont = h_ng

    # Phase 6: Iterated binary refinement
    for iteration in range(4):
        if tleft() < 70:
            break

        allowed = min(35.0, tleft() * 0.08)
        if allowed > 15:
            h_pert = h_bin.copy()
            n_pert = rng.randint(6, 16)
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]

            n_bls = int(min(100000, allowed * 8000))
            h_bl, c5_bl = binary_local_search(h_pert, max_iters=n_bls, rng=rng,
                                              time_limit=allowed, start_time=start_time)
            if c5_bl < best_bin_score:
                best_bin_score = c5_bl
                h_bin = h_bl

        if tleft() < 50:
            break

        allowed = min(35.0, tleft() * 0.08)
        if allowed > 15:
            h_c3 = project_box_sum(h_bin, target_f)
            n_it2 = int(min(1500000, allowed * 20000))
            h_pg2, c5_pg2 = projected_topk_grad(h_c3, n_it2, topk=3,
                                                time_limit=allowed, start_time=start_time)
            if c5_pg2 < best_cont_score:
                best_cont_score = c5_pg2
                h_cont = h_pg2

        if tleft() < 40:
            break

        allowed = min(30.0, tleft() * 0.07)
        if allowed > 10:
            h_ref2, sc_ref2 = slsqp_refine(h_cont, n_init, target_f,
                                           alphas=[500, 2000, 10000],
                                           maxiter=3000, ftol=1e-14,
                                           time_limit=allowed, start_time=start_time)
            if sc_ref2 < best_cont_score:
                best_cont_score = sc_ref2
                h_cont = h_ref2

    # Phase 7: Different resolutions
    for n_try in [n_init + 2, n_init - 2, n_init + 4, n_init - 4]:
        if tleft() < 55:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        target_try_i = n_try // 2
        h_interp = interpolate_to(h_cont, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try_i)

        allowed = min(20.0, tleft() * 0.07)
        if allowed < 10:
            continue
        h_g, c5_g = binary_local_search(h_bin_try, max_iters=80000, rng=rng,
                                        time_limit=allowed, start_time=start_time)
        cont_try = project_box_sum(h_g, target_try)
        sc_try = compute_c5(cont_try)
        if sc_try < best_cont_score:
            best_cont_score = sc_try
            h_cont = cont_try

    # Phase 8: Final SLSQP burst
    if tleft() > 35:
        n_use = len(h_cont)
        t_use = n_use / 2.0
        h_f, sc_f = slsqp_refine(h_cont, n_use, t_use,
                                 alphas=[1000, 5000, 20000],
                                 maxiter=6000, ftol=1e-15,
                                 time_limit=max(15.0, tleft() - 15),
                                 start_time=start_time)
        if sc_f < best_cont_score:
            best_cont_score = sc_f
            h_cont = h_f

    # Phase 9: Final topk gradient sweep
    if tleft() > 20:
        n_use = len(h_cont)
        n_it_f = int(min(1000000, max(100000, (tleft() - 8) * 30000)))
        h_f2, sc_f2 = projected_topk_grad(h_cont, n_it_f, topk=3,
                                          time_limit=max(8.0, tleft() - 8),
                                          start_time=start_time)
        if sc_f2 < best_cont_score:
            best_cont_score = sc_f2
            h_cont = h_f2

    # Final selection
    n_final = len(h_cont)
    target_final = n_final / 2.0
    h_final = project_box_sum(h_cont, target_final)
    final_score = compute_c5(h_final)

    n_bin = len(h_bin)
    h_bin_final = project_binary(h_bin, n_bin // 2)
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_bin
    else:
        return h_final, final_score, n_final
