import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, alpha):
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -400, 0)
    exp_w = np.exp(shifted)
    W = exp_w / exp_w.sum()
    
    # Gradient of smooth max w.r.t. h
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    positions = np.arange(n)
    
    for k in range(2*n-1):
        wk = W[k]
        if wk < 1e-10:
            continue
        s = shifts[k]
        if s >= 0:
            # corr_k = sum_{x=0}^{n-1-s} h[x] * (1-h[x+s])
            # d/dh[i] = (1-h[i+s]) for i in [0, n-1-s] (from first term)
            #           -h[i-s] for i in [s, n-1] (from second term)
            if s == 0:
                # corr_0 = sum h[x]*(1-h[x]), d/dh[i] = 1 - 2h[i]
                grad += wk * (1 - 2*h)
            else:
                # First term: d/dh[i] of h[i]*(1-h[i+s]) = (1-h[i+s]) for i < n-s
                grad[:n-s] += wk * (1 - h[s:])
                # Second term: d/dh[i] of (1-h[i]) * h[i-s] = -h[i-s] for i >= s
                grad[s:] += wk * (-h[:n-s])
        else:
            abs_s = -s
            # By symmetry, same as s = abs_s
            if abs_s == 0:
                grad += wk * (1 - 2*h)
            else:
                grad[abs_s:] += wk * (1 - h[:n-abs_s])
                grad[:n-abs_s] += wk * (-h[abs_s:])
    
    return smooth_max(corr, alpha), grad

def project_box_sum(y, target_sum):
    n = len(y)
    y = y.copy()
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r

def fast_greedy(h, max_iters, rng, time_limit, start_time):

        for k_idx in top3:

            valid = (idx >= 0) & (idx < n)

            idx = positions - k_star

def lbnb_smoothed(h_start, n, target, alphas, time_limit, start_time):
    """Use L-BFGS-B with smoothed max objective and projection."""
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def obj_and_grad(x):
            x = np.clip(x, 0.0, 1.0)
            corr = compute_corr(x)
            c_max = corr.max()
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -400, 0)
            exp_w = np.exp(shifted)
            W = exp_w / exp_w.sum()
            val = c_max + np.log(exp_w.sum()) / alpha
            
            # Gradient
            grad = np.zeros(n)
            shifts = np.arange(2*n-1) - (n-1)
            for k in range(2*n-1):
                wk = W[k]
                if wk < 1e-12:
                    continue
                s = shifts[k]
                if s >= 0:
                    if s == 0:
                        grad += wk * (1 - 2*x)
                    else:
                        grad[:n-s] += wk * (1 - x[s:])
                        grad[s:] += wk * (-x[:n-s])
                else:
                    abs_s = -s
                    if abs_s == 0:
                        grad += wk * (1 - 2*x)
                    else:
                        grad[abs_s:] += wk * (1 - x[:n-abs_s])
                        grad[:n-abs_s] += wk * (-x[abs_s:])
            
            return val, grad
        
        try:
            res = minimize(obj_and_grad, h_cur, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-14,
                                                   'maxls': 50})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def projected_nesterov(h_start, n_iters, alpha, lr_init, lr_end, time_limit, start_time):
    """Nesterov accelerated gradient with projection."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h_start.copy().astype(float)

        # Nesterov lookahead
        y = h + mu * v
        
        # Gradient at y
        corr_y = compute_corr(y)
        c_max = corr_y.max()
        shifted = alpha * (corr_y - c_max)
        shifted = np.clip(shifted, -400, 0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()

                    grad += wk * (1 - 2*y)
                else:
                    grad[abs_s:] += wk * (1 - y[:n-abs_s])
                    grad[:n-abs_s] += wk * (-y[abs_s:])
        
        v_new = mu * v - lr * grad
        h_new = h + v_new
        h_new = project_box_sum(h_new, target)
        v = h_new - h
        h = h_new
        
        corr = compute_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(y, target_sum):
    n = len(y)
    y = y.copy().astype(float)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-13:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0

def make_smoothed_fn(alpha, n):
    """Return (obj, grad) for the smoothed max objective."""
    shifts = np.arange(2*n-1) - (n-1)
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        corr = compute_corr(x)
        c_max = float(corr.max())
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        val = c_max + float(np.log(exp_w.sum())) / alpha
        
        grad = np.zeros(n)
        for k_idx in range(2*n-1):
            wk = W[k_idx]
            if wk < 1e-12:
                continue
            s = shifts[k_idx]
            if s >= 0:
                if s == 0:
                    grad += wk * (1.0 - 2.0*x)
                else:
                    grad[:n-s] += wk * (1.0 - x[s:])
                    grad[s:] += -wk * x[:n-s]
            else:
                abs_s = -s
                if abs_s == 0:
                    grad += wk * (1.0 - 2.0*x)
                else:
                    grad[abs_s:] += wk * (1.0 - x[:n-abs_s])
                    grad[:n-abs_s] += -wk * x[abs_s:]
        
        return val, grad
    
    return obj_and_grad

                remaining = 0.0
            else:

        idx_j = j - k_vec
        mask = (idx_j >= 0) & (idx_j < n)
        delta[mask] -= h[idx_j[mask]]
        idx_j = j + k_vec
        mask = (idx_j >= 0) & (idx_j < n)
        delta[mask] -= h[idx_j[mask]]
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h_bin = project_binary(h_init, int(n_init / 2))
    best_score_bin = compute_c5(best_h_bin)

    if tleft() > 40:
        h_g, c5_g = fast_greedy(best_h_bin, max_iters=20000, rng=rng,
                                  time_limit=min(25, tleft() * 0.05), start_time=start_time)
        if c5_g < best_score_bin:
            best_score_bin = c5_g
            best_h_bin = h_g.copy()

    if tleft() > 60:
        h_sa, c5_sa = sa_binary(best_h_bin, int(min(2500000, tleft() * 0.12 * 40000)),
                                 0.012, 0.00002, rng,
                                 time_limit=min(80, tleft() * 0.12), start_time=start_time)
        if c5_sa < best_score_bin:
            best_score_bin = c5_sa
            best_h_bin = h_sa.copy()

    best_h_cont = project_box_sum(best_h_bin, target_init)
    best_score_cont = compute_c5(best_h_cont)

    if tleft() > 60:
        h_l, sc_l = lbnb_optimize(best_h_cont, n_init, target_init,
                                    alphas=[20, 50, 100, 200],
                                    maxiter_per_alpha=400,
                                    time_limit=min(130, tleft() * 0.25),
                                    start_time=start_time)
        if sc_l < best_score_cont:
            best_score_cont = sc_l
            best_h_cont = h_l.copy()

    if tleft() > 50:
        h_n, sc_n = nesterov_smoothed(best_h_cont, int(min(400000, tleft() * 0.12 * 5000)),
                                        alpha=100, lr_init=0.012, lr_end=0.0004,
                                        time_limit=min(60, tleft() * 0.12),
                                        start_time=start_time)
        if sc_n < best_score_cont:
            best_score_cont = sc_n
            best_h_cont = h_n.copy()

    if tleft() > 50:
        h_l2, sc_l2 = lbnb_optimize(best_h_cont, n_init, target_init,
                                      alphas=[300, 600, 1200, 2500],
                                      maxiter_per_alpha=500,
                                      time_limit=min(120, tleft() * 0.20),
                                      start_time=start_time)
        if sc_l2 < best_score_cont:
            best_score_cont = sc_l2
            best_h_cont = h_l2.copy()

    for n_try in [n_init + 6, n_init + 12, n_init - 6]:
        if tleft() < 70:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h_cont, n_try, target_try)
        if tleft() > 50:
            h_l3, sc_l3 = lbnb_optimize(h_interp, n_try, target_try,
                                          alphas=[50, 150, 400, 800],
                                          maxiter_per_alpha=400,
                                          time_limit=min(50, tleft() * 0.20),
                                          start_time=start_time)
            if sc_l3 < best_score_cont:
                best_score_cont = sc_l3
                best_h_cont = h_l3.copy()

    for it in range(10):
        if tleft() < 35:
            break
        n_cur = len(best_h_cont)
        target_cur = n_cur / 2.0
        if tleft() > 25:
            h_f, sc_f = lbnb_optimize(best_h_cont, n_cur, target_cur,
                                        alphas=[500, 1500, 3000, 6000],
                                        maxiter_per_alpha=600,
                                        time_limit=min(40, tleft() * 0.35),
                                        start_time=start_time)
            if sc_f < best_score_cont:
                best_score_cont = sc_f
                best_h_cont = h_f.copy()
        if tleft() > 20:
            h_n2, sc_n2 = nesterov_smoothed(best_h_cont, int(min(150000, tleft() * 0.15 * 4000)),
                                              alpha=800, lr_init=0.006, lr_end=0.0002,
                                              time_limit=min(20, tleft() * 0.15),
                                              start_time=start_time)
            if sc_n2 < best_score_cont:
                best_score_cont = sc_n2
                best_h_cont = h_n2.copy()
        h_bin_p = project_binary(best_h_cont, int(n_cur / 2))
        sc_p = compute_c5(h_bin_p)
        if sc_p < best_score_cont:
            best_score_cont = sc_p
            best_h_cont = h_bin_p.copy()
            best_h_bin = h_bin_p.copy()
            best_score_bin = sc_p

    n_final = len(best_h_cont)
    target_final = n_final / 2.0
    h_final = project_box_sum(best_h_cont, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(h_final, int(n_final / 2))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        h_bin_final = np.clip(h_bin_final, 0.0, 1.0)
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
