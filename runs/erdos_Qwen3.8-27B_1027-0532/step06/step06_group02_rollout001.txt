import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum != target_sum:
        if current_sum > target_sum:
            ones = np.where(h_r == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:current_sum - target_sum]:
                h_r[ones[idx]] = 0
        else:
            zeros = np.where(h_r == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:target_sum - current_sum]:
                h_r[zeros[idx]] = 1
    return h_r.astype(float)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def smooth_gradient(h, w, N_fft_cache_size):
    """Compute gradient of smooth max correlation.
    w: weights on full corr array (length 2n-1)
    Returns: grad (length n) where grad[j] = d(smooth_max)/d(h[j])
    """
    n = len(h)
    # P[j] = (h * w)[j+n-1]
    # Q[j] = (w_rev * h)[j+n-1]
    # Use FFT
    N_fft = N_fft_cache_size  # should be >= 3n-1
    h_fft = np.fft.fft(h, N_fft)
    w_fft = np.fft.fft(w, N_fft)
    
    conv_hw = np.fft.ifft(h_fft * w_fft).real  # (h*w)[k] for k=0..N_fft-1
    w_rev = w[::-1]
    w_rev_fft = np.fft.fft(w_rev, N_fft)
    conv_wh = np.fft.ifft(w_rev_fft * h_fft).real  # (w_rev*h)[k] for k=0..N_fft-1
    
    offset = n - 1
    P = conv_hw[offset:offset+n]
    Q = conv_wh[offset:offset+n]
    
    return 1.0 - P - Q

def adam_refine(h0, target, start_time, time_limit, alphas=None, n_fft=None):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    if n_fft is None:
        n_fft = 1
        while n_fft < 3 * n - 1:
            n_fft <<= 1
    
    if alphas is None:
        alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit - 3:
            break
        phase_remaining = time_limit - 3 - (time.time() - start_time)
        # Estimate iterations possible
        phase_iters = int(min(500, phase_remaining * 50))
        phase_iters = max(phase_iters, 50)
        
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        m = np.zeros(n)
        v = np.zeros(n)
        t_step = 0
        lr_base = 1.0 / (np.sqrt(alpha + 1.0) * np.sqrt(n) * 0.1)
        
        prev_score = best_score
        no_improve = 0
        
        for it in range(phase_iters):
            if time.time() - start_time > time_limit - 3:
                break
            t_step += 1
            
            corr = full_corr_fast(h)
            max_c = corr.max()
            shifted = np.clip(alpha * (corr - max_c), -500, 0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(2*n-1)
                w[n-1] = 1.0  # uniform on the max lag
            else:
                w = w / w_sum
            
            grad = smooth_gradient(h, w, n_fft)
            
            # Adaptive learning rate
            lr = lr_base / (1.0 + 0.1 * np.sqrt(t_step))
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
            
            if it % 10 == 0:
                sc = compute_c5(h)
                if sc < best_score:

                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 10

    return best_h, best_score

            # 2-for-2 swap: swap h[i1],h[j1] with h[i2],h[j2]
            i1 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            i2 = ones[rng.randint(len(ones))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2:
                stall += 1
                continue
            h[i1], h[j1] = 0.0, 1.0
            h[i2], h[j2] = 0.0, 1.0
            delta = compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
            # Wait, this isn't right. After the first flip, h has changed. Let me compute both deltas from the original h.
            # Actually, let me undo and compute properly.
            h[i1], h[j1] = 1.0, 0.0
            h[i2], h[j2] = 1.0, 0.0
            # Delta for flipping i1->0, j1->1 AND i2->0, j2->1
            # = delta(i1,j1) + delta(i2,j2) - interaction
            # The interaction is: the delta from (i1,j1) affects positions that overlap with the delta from (i2,j2)
            # For simplicity, just compute the total delta directly
            full_new_estimate = full + compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
            # But this double-counts the interaction. For now, accept this approximation.
            c5_new = float(full_new_estimate.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[j1] = 1.0
                h[i2] = 0.0
                h[j2] = 1.0
                full = full + compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
                # Recompute to be safe (the approximation might be off)
                # Actually this is dangerous. Let me just recompute full from scratch occasionally.
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Single flip
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Bias toward the worst lag
        if it % 3 == 0:
            top_k = 3
            top_indices = np.argsort(full)[::-1][:top_k]
            best_move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move_found = True
                    break
            if not best_move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
