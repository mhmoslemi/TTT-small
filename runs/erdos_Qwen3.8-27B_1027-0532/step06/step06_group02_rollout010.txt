def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def gradient_smoothing(h, corr_full, alpha):
    n = len(h)
    c_max = corr_full.max()
    shifted = alpha * (corr_full - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    n_c = 2 * n - 1
    grad = np.zeros(n)
    positions = np.arange(n)
    significant = np.where(weights > 1e-8)[0]
    for k_idx in significant:
        w = weights[k_idx]
        s = k_idx - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
            continue
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        g = np.zeros(n)
        g[i_lo:i_hi+1] = w
        if s > 0:
            if n - s > 0:
                g[:n-s] -= w * h[s:]
            if s < n:
                g[s:] -= w * h[:n-s]
        else:
            s_neg = -s
            if n - s_neg > 0:
                g[s_neg:] -= w * h[:n-s_neg]
            if s_neg < n:
                g[:n-s_neg] -= w * h[s_neg:]
        grad += g
    return grad

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    _cons_ref.n = n
    _cons_ref.target = target

def block_shuffle(h, rng, n_blocks=6):
    n = len(h)
    block_size = n // n_blocks
    remainder = n - block_size * n_blocks
    indices = list(range(n_blocks))
    rng.shuffle(indices)
    h_new = np.zeros(n)
    pos = 0
    for new_idx in range(n_blocks):
        orig_idx = indices[new_idx]
        start = orig_idx * block_size
        end = start + block_size
        if orig_idx == n_blocks - 1:
            end = n
        h_new[pos:pos + (end - start)] = h[start:end]
        pos += (end - start)
    return h_new
