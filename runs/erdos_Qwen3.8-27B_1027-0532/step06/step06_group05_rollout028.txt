import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r

def subgrad_compute(h, n, num_lags):
    """Compute weighted subgradient using top num_lags lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top num_lags
    sorted_idx = np.argsort(corr)[::-1]
    top_indices = sorted_idx[:num_lags]
    top_values = corr[top_indices]
    
    # Exponential weighting
    max_val = top_values[0]
    weights = np.exp(50.0 * (top_values - max_val))
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        s = int(top_indices[idx])
        w = weights[idx]
        if w < 1e-10:
            continue
        k = s - (n - 1)
        if k >= 0:
            # Positive part: (1-h[j-k]) for j in [k, n-1]
            if k < n:
                grad[k:] += w * (1.0 - h[:n-k])
            # Negative part: -h[j+k] for j in [0, n-1-k]
            if n - 1 - k >= 0:
                grad[:n-k] -= w * h[k:]
        else:
            m = -k
            # Positive part: (1-h[j+m]) for j in [0, n-1-m]
            if n - 1 - m >= 0:
                grad[:n-m] += w * (1.0 - h[m:])
            # Negative part: -h[j+m] for j in [m, n-1]
            if m < n:
                grad[m:] -= w * h[:n-m]
    
    return grad, float(np.max(corr))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for num_lags in num_lags_schedule:
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = best_score
        
        while time.time() - start_time < time_budget:
            t += 1
            grad, curr_max = subgrad_compute(h, n, num_lags)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
            
            # Check improvement (every 100 steps to save time)
            if t % 100 == 0:
                score = compute_c5(h)
                if score < local_best - 1e-10:
                    local_best = score
                    stall = 0
                else:
                    stall += 1
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                if stall > 20:
                    break
        
        if time.time() - start_time > time_budget:
            break
    
    return best_h, best_score

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0

                weights = np.ones(2*n-1) / (2*n-1)
            else:
                top_m_vals = np.sort(corr)[::-1][:m]
                thresh = top_m_vals[-1]
                mask = corr >= thresh - 1e-10
                top_idx = np.where(mask)[0]
                # Exponential weighting
                w_raw = np.exp(30.0 * (corr[top_idx] - top_m_vals[0]))
                weights_full = np.zeros(2*n-1)
                weights_full[top_idx] = w_raw / w_raw.sum()
                top_idx = np.where(weights_full > 1e-10)[0]
                weights = weights_full[top_idx]
            
            # Compute gradient
            grad = np.zeros(n)
            for idx in range(len(top_idx)):
                s = int(top_idx[idx])
                w = weights[idx]
                if w < 1e-12:
                    continue
                k = s - (n - 1)
                m_abs = abs(k)
                if m_abs < n:
                    grad[m_abs:] += w * (1.0 - h[:n - m_abs])
                    grad[:n - m_abs] -= w * h[m_abs:]
            
            # Adam step
            m_ema = beta1 * m_ema + (1 - beta1) * grad
            v_ema = beta2 * v_ema + (1 - beta2) * (grad * grad)
            m_hat = m_ema / (1 - beta1**t)
            v_hat = v_ema / (1 - beta2**t)
            h = project_box_sum(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
            
            if t % check_every == 0:
                score = compute_c5(h)
                if score < local_best - 1e-12:
                    local_best = score
                    stall = 0
                else:
                    stall += 1
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                if stall > 50:
                    break
    
    return best_h, best_score
