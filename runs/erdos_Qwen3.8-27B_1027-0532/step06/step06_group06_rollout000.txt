import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_detailed(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr * dx

def project_box_sum(h, target):
    """Project h onto [0,1]^n with sum = target using bisection on shift."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to add a tiny amount
            mask = (result < 1.0) & (result > 0.0)
            if np.any(mask):
                n_eligible = np.sum(mask)
                add = diff / n_eligible
                result[mask] = np.minimum(result[mask] + add, 1.0)
        else:
            mask = (result > 0.0) & (result < 1.0)
            if np.any(mask):
                n_eligible = np.sum(mask)
                sub = -diff / n_eligible
                result[mask] = np.maximum(result[mask] - sub, 0.0)
    return result

def project_binary(h, target_sum):
    """Project to binary with exact target sum."""
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r
    if current > target_sum:
        # Need to flip (current - target_sum) ones to zeros
        # Prefer ones with lowest h values
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:current - target_sum]:
            h_r[ones[idx]] = 0
    else:
        # Need to flip (target_sum - current) zeros to ones
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target_sum - current]:
            h_r[zeros[idx]] = 1
    return h_r

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0
            if np.any(mask):
                add = diff / np.sum(mask)
                result[mask] = np.minimum(result[mask] + add, 1.0)
        else:
            mask = result > 0.0
            if np.any(mask):
                sub = -diff / np.sum(mask)
                result[mask] = np.maximum(result[mask] - sub, 0.0)
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r
    if current > target_sum:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:current - target_sum]:
            h_r[ones[idx]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target_sum - current]:
            h_r[zeros[idx]] = 1
    return h_r

def smooth_obj(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_grad(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    t_arr = np.arange(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_opt(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_obj(h_cur, beta, target, penalty)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(beta, target, penalty),
                           jac=smooth_grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1})
            h_cand = project_box_sum(res.x, target)
            sc = compute_c5(h_cand)
            obj_cand = smooth_obj(h_cand, beta, target, penalty)
            if obj_cand < curr_obj - 1e-10:
                h_cur = h_cand
                curr_obj = obj_cand
            else:
                sc_cur = compute_c5(h_cur)
                if sc < sc_cur:
                    h_cur = h_cand
        except Exception:
            pass
    return project_box_sum(h_cur, target)

    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        n_use_float = float(n_use)
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(60, tleft() - 5)
        max_it = int(min(100000, time_alloc * 3000))
        h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.004, 0.000005,
                                rng, time_limit=time_alloc, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
