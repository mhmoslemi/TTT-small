import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def lse_obj_grad(h, alpha, n, m_top=30):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    top_idx = np.argsort(w)[-m_top:]
    ws = w[top_idx]
    ws = ws / ws.sum()
    idxs = np.arange(n)
    grad = np.zeros(n)
    for wv, si in zip(ws, top_idx):
        s_val = int(si) - (n - 1)
        v1 = (idxs - s_val >= 0) & (idxs - s_val < n)
        v2 = (idxs + s_val >= 0) & (idxs + s_val < n)
        grad[v1] += wv * (1.0 - h[idxs[v1] - s_val])
        grad[v2] -= wv * h[idxs[v2] + s_val]
    grad *= dx
    return grad

def lbfags_optimize(h_start, n, target, alphas, maxiter, m_top=30):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj_func(x):
            return lse_obj(x, alpha, n)
        def jac_func(x):
            return lse_obj_grad(x, alpha, n, m_top)
        try:
            res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                                   'gtol': 1e-8, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd_optimize(h, n, target, time_limit, n_cand_start=40, n_cand_max=100,
                     patience=150, max_sweeps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        if no_improve > patience // 3 and n_cand < n_cand_max:
            n_cand = min(n_cand + 5, n_cand_max)
        elif no_improve < patience // 5 and n_cand > n_cand_start:
            n_cand = max(n_cand - 3, n_cand_start)

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        active_shifts = np.argsort(corr)[-min(5, n_corr):]
        for as_idx in active_shifts[1:]:
            s_as = int(as_idx) - (n - 1)
            va1 = (idxs - s_as >= 0) & (idxs - s_as < n)
            va2 = (idxs + s_as >= 0) & (idxs + s_as < n)
            grad[va1] += 0.3 * (1.0 - h[idxs[va1] - s_as])
            grad[va2] -= 0.3 * h[idxs[va2] + s_as]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            h_i_sp = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            h_i_sp[v_ip] = h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            h_i_sm = np.zeros(n_corr)
            h_i_sm[v_im] = h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm_arr = np.zeros(n_corr)
                jpm_arr[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm_arr = np.zeros(n_corr)
                jsm_arr[jm] = h[j - shifts[jm]]

                g = h_i_sp + h_i_sm - jpm_arr - jsm_arr

                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, a_d + 1e-10)

                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def game_theoretic_descent(h_init, n, target, time_limit, eta_base=2.0, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0

    for step in range(20000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = best_lr * max(0.1, 1.0 - step / 15000)
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = lr * 1.15
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.85
            if best_lr < 0.001:
                break

        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.6 * lam + 0.4 * lam_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_local_search(h_binary, n, target, time_limit, max_flips=50000):

        i = ones[rng.randint(len(ones))]

            h[j] = 0.0

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = pair_cd_optimize(best_h, n_use, target_use,
                                       time_limit=min(50, tleft() * 0.5),
                                       n_cand_start=60, n_cand_max=150,
                                       patience=100, max_sweeps=20000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: SLSQP final polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        bounds = [(0.0, 1.0)] * n_use
        global_n = n_use
        global_target = target_use

        def cons_fun(x):
            return np.sum(x) - global_target

        def cons_jac(x):
            return np.ones(global_n)

        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        for alpha in [5000, 20000]:
            if tleft() < 10:
                break
            try:
                res = minimize(lambda x: lse_obj(x, alpha, n_use), best_h,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_use)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            except Exception:
                pass

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
