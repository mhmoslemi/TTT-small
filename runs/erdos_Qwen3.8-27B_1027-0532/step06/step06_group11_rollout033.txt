def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    g[v2] -= h[idxs[v2] - k]
    return g

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        f_arr = precompute_f(h, n)
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = f_arr[i] - f_arr[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def precompute_f(h, n):
    f = -np.ones((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    for p in range(n):
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += h[idx1[mask1]]
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    return f

def simulate_anneal(h_start, n, target, time_limit, T0=0.01, cooling=0.9995, n_top=5):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    T = T0
    no_improve = 0
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top * dx
        grad -= np.mean(grad)
        accepted = False
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = project_feasible(h - lr * grad + rng.randn(n) * lr * 0.5, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now or (c5_try - c5_now) < T:
                h = h_try
                accepted = True
                break
            if np.random.random() < np.exp(-max(0, (c5_try - c5_now)) / max(T, 1e-10)):
                h = h_try
                accepted = True
                break
        T = max(T * cooling, 1e-8)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_sinusoid(n, freq, phase, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return project_feasible(h, n / 2.0)

def make_random_smooth(n, rng, amp):
    freqs = rng.rand(n) ** 2
    phases = rng.randn(n) * 2.0 * np.pi
    h = 0.5 + amp * np.sin(2.0 * np.pi * freqs * np.arange(n) / n + phases)
    for _ in range(3):
        h = np.convolve(h, np.ones(5) / 5.0, mode='same')
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 150:
        h_pol, sc = slsqp_opt(best_h, n, target, [500, 2000, 10000, 50000], 1500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_lb, sc = lbfgs_auglag(best_h, n, target, time_limit=min(80, tleft() * 0.6),
                                alphas=[1000, 5000, 20000, 100000], max_inner=300, mu0=50.0)
        if sc < best_score:
            best_score = sc
            best_h = h_lb

    if tleft() > 150:
        h_ad, sc = adam_topk(best_h, n, target, time_limit=min(80, tleft() * 0.4),
                             max_steps=20000, n_top_start=3, n_top_end=20, lr=0.012)
        if sc < best_score:
            best_score = sc
            best_h = h_ad

    if tleft() > 200:
        n_hires = 800
        target_hires = n_hires / 2.0
        h_hires = interpolate_to(best_h, n_hires, target_hires)
        sc_h = compute_c5(h_hires)
        if sc_h < best_score:
            best_score = sc_h
            best_h = h_hires
        h_hires, sc = slsqp_opt(h_hires, n_hires, target_hires,
                                [500, 2000, 10000, 50000, 200000], 1200)
        if sc < best_score:
            best_score = sc
            best_h = h_hires

    if tleft() > 150:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_lb, sc = lbfgs_auglag(best_h, n_cur, t_cur, time_limit=min(100, tleft() * 0.5),
                                alphas=[5000, 20000, 100000, 500000], max_inner=200, mu0=100.0)
        if sc < best_score:
            best_score = sc
            best_h = h_lb

    if tleft() > 150:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_cd, sc = pair_cd(best_h, n_cur, t_cur,
                           time_limit=min(120, tleft() * 0.5), max_sweeps=5000, n_cand=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 100:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_sa, sc = simulate_anneal(best_h, n_cur, t_cur,
                                   time_limit=min(70, tleft() * 0.5), T0=0.005, cooling=0.9997)
        if sc < best_score:
            best_score = sc
            best_h = h_sa

    if tleft() > 120:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_pol, sc = slsqp_opt(best_h, n_cur, t_cur,
                              [20000, 100000, 500000, 2000000], 1000)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_ad, sc = adam_topk(best_h, n_cur, t_cur, time_limit=min(60, tleft() * 0.5),
                             max_steps=15000, n_top_start=5, n_top_end=30, lr=0.008)
        if sc < best_score:
            best_score = sc
            best_h = h_ad
        if sc < best_score or True:
            h_pol, sc2 = slsqp_opt(best_h, n_cur, t_cur, [50000, 200000, 1000000], 800)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol

    if tleft() > 80:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise in [0.02, 0.05, 0.1]:
            if tleft() < 40:
                break
            h_pert = project_feasible(best_h + rng.randn(n_cur) * noise, t_cur)
            h_ref, sc = slsqp_opt(h_pert, n_cur, t_cur, [2000, 20000, 200000], 800)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_cd, sc = pair_cd(best_h, n_cur, t_cur,
                           time_limit=min(40, tleft() * 0.6), max_sweeps=3000, n_cand=25)
        if sc < best_score:
            best_score = sc
            best_h = h_cd
        h_pol, sc2 = slsqp_opt(best_h, n_cur, t_cur, [100000, 500000, 2000000], 600)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
