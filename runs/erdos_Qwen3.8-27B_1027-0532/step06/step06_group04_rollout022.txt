import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_full_lp(h_init, n, target, damping, n_iter, time_limit, reg=0.0, scs_eps=1e-11):
    """Alternating LP with ALL constraints."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Add ALL correlation constraints
        for k_idx in range(1, 2*n - 1):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi <= lo:
                continue
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
        
        if reg > 1e-10:
            objective = t_var + reg * cp.sum_squares(h_var - h)
        else:
            objective = t_var
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_full_lp(h_init, n, target, damping, n_iter, time_limit, reg=1e-6):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in range(1, 2 * n - 1):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi <= lo:
                continue
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-11, max_iters=200000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def solve_active_lp(h_init, n, target, damping, n_iter, time_limit, reg, scs_eps, n_active):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi <= lo:
                continue
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else c5_val
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())
        
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.5, damping * 0.95)
        else:
            damping = min(0.99, damping + 0.02)
        if it % 50 == 0 and reg > 1e-8:
            reg = max(1e-8, reg * 0.7)
    
    return best_h, best_c5

def solve_1d_max(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-12):
        return 0.0
    if np.all(C < -1e-12):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            idx_max = np.argmax(vals)
            active = np.zeros(len(C), dtype=bool)
            active[idx_max] = True
        if np.all(C[active] > 1e-12):
            hi = mid
        elif np.all(C[active] < -1e-12):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.01
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - (n - 1)
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max(slopes, intercepts, lam)
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
        if p % 20 == 19:
            lam_rate = max(0.001, lam_rate * 0.95)
    
    return best_h, best_c5

def block_flip_perturb(h, rng, target):
    n = len(h)
    h_new = h.copy()
    block_len = rng.randint(5, max(6, n // 5))
    a = rng.randint(0, n - block_len)
    h_new[a:a + block_len] = 1.0 - h_new[a:a + block_len]
    return project_feasible(h_new, target)

def noise_perturb(h, rng, target, sigma):
    h_new = h + rng.randn(len(h)) * sigma
    return project_feasible(h_new, target)

def swap_perturb(h, rng, target, n_swaps):
    h_new = h.copy()
    n = len(h)
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_feasible(h_new, target)

def ramp_perturb(h, rng, target, amplitude):
    n = len(h)
    x = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.choice([1, 2, 3, 4, 5])
    h_new = h + amplitude * np.sin(2 * np.pi * freq * x + phase)
    return project_feasible(h_new, target)

def sa_ils(h_init, n, target, max_epochs, time_limit, rng, lp_func, lp_kwargs):
    """Simulated annealing with iterated local search."""
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    T = 0.05
    gamma = 0.97
    
    for epoch in range(max_epochs):
        if time.time() - start > time_limit:
            break
        
        # Perturb
        move_type = rng.randint(4)
        if move_type == 0:
            h_pert = block_flip_perturb(h_cur, rng, target)
        elif move_type == 1:
            sigma = rng.uniform(0.05, 0.3)
            h_pert = noise_perturb(h_cur, rng, target, sigma)
        elif move_type == 2:
            h_pert = swap_perturb(h_cur, rng, target, rng.randint(2, 8))
        else:
            h_pert = ramp_perturb(h_cur, rng, target, rng.uniform(0.05, 0.2))
        
        p_c5 = compute_c5(h_pert)
        
        # Local improvement
        if time.time() - start < time_limit - 60:
            h_loc, c5_loc = lp_func(h_pert, n, target, **lp_kwargs)
        else:
            h_loc, c5_loc = h_pert, p_c5
        
        # Acceptance
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
            h_cur = h_loc.copy()
        elif c5_loc < best_c5 + T:
            accept_prob = np.exp(-(c5_loc - best_c5) / max(T, 1e-10))
            if rng.random() < accept_prob:
                h_cur = h_loc.copy()
        
        T *= gamma
    
    return best_h, best_c5

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0

def cons_fun(h):
    return np.array([np.sum(h) - _global_target])

def cons_jac(h):
    return np.array([np.ones(_global_n)])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

Wait, I have a problem. The `sa_ils` function takes `lp_func` and `lp_kwargs`, which is a bit awkward. Let me restructure.

Actually, the issue is that `sa_ils` needs to call a local improvement function, but I can't use a lambda. I can pass the function directly as an argument, which is fine (it's not a closure or lambda, it's a top-level function reference).

Let me also reconsider the SA parameters. The temperature T=0.05 might be too high or too low. Let me think about the scale of C₅ differences.

If the current best is 0.38088 and we want to get to 0.3808, the improvement is 0.00008. So the temperature should be on the order of 1e-4 to 1e-3 for the late stages.

Let me use a temperature schedule:
- T starts at 0.01
- T *= 0.99 each epoch
- After ~400 epochs, T ≈ 0.01 * 0.99^400 ≈ 0.01 * 0.018 ≈ 0.00018

This seems reasonable.

But wait, the local improvement (LP) already makes the solution quite good. The SA perturbation needs to be large enough to escape the basin, but the LP will pull it back. The issue is that the LP might always converge to the same basin.

Let me think about this differently. The key is to make the perturbation large enough that the LP, starting from the perturbed solution, converges to a DIFFERENT basin.

Block flips of length 5-30 should be enough to escape a local basin if the basin is "shallow". But if the basin is "deep" (the local optimum is well-separated from others), we need larger perturbations.

Let me increase the perturbation sizes:
- Block flip: length 10-50
- Noise: sigma = 0.1-0.5
- Swap: 5-20 pairs
- Ramp: amplitude 0.1-0.3

And do more LP iterations after each perturbation (10-20 instead of 5).

Actually, let me reconsider the whole approach. Maybe instead of SA, I should use a more structured approach:

**Iterated Local Search (ILS)**:
1. Start from current best
2. Perturb (large, structured)
3. Local improve (until convergence)
4. If improved, accept; otherwise, reject
5. Go to 2

The key difference from SA is that ILS doesn't accept worse solutions (or only accepts them very rarely). This is more efficient when the landscape has many local optima of similar quality.

Let me implement ILS instead of SA:

def ils(h_init, n, target, max_iters, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Large perturbation
        pert_type = rng.randint(4)
        if pert_type == 0:
            h_pert = block_flip_perturb(h_cur, rng, target)
        elif pert_type == 1:
            h_pert = noise_perturb(h_cur, rng, target, rng.uniform(0.1, 0.4))
        elif pert_type == 2:
            h_pert = swap_perturb(h_cur, rng, target, rng.randint(5, 20))
        else:
            h_pert = ramp_perturb(h_cur, rng, target, rng.uniform(0.1, 0.3))
        
        # Local improvement (full LP)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _solve_lp_obj(h, g, dx, n, target, reg, h_center):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in range(1, 2 * n - 1):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi <= lo:
            continue
        coeff = np.zeros(n)
        coeff[lo:hi] = g[lo - k:hi - k] * dx
        constraints.append(coeff @ h_var <= t_var)
    if reg > 1e-10:
        obj = t_var + reg * cp.sum_squares(h_var - h_center)
    else:
        obj = t_var
    prob = cp.Problem(cp.Minimize(obj), constraints)
    prob.solve(solver=cp.SCS, eps=1e-11, max_iters=200000, verbose=False)
    if h_var.value is None:
        prob.solve(solver=cp.ECOS, max_iter=100000)
    return h_var, t_var
