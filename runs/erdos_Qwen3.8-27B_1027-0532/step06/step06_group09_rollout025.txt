    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _sum_con_fun(x):
    return np.sum(x) - _g_target

def _sum_con_jac(x):
    return np.ones(_g_n)

def _lbfags(h, alpha, maxiter=1000):

    except:

def _slsqp(h, alphas, maxiter=800, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    n = len(h)
    _g_n = n

    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
