    # weights has length 2n-1, indexed by k = -(n-1), ..., n-1
    # C(k) = sum_i h[i]*(1-h[i+k]) for valid i
    # dG/dh[j] = sum_k weights[k]*(1-h[j+k])*1[j+k<n] - sum_k weights[k]*h[j-k]*1[j-k<n]

    for j in range(n):
        # First term: sum over k where j+k is in [0, n)
        k_lo = max(-(n-1), -j)  # j+k >= 0 => k >= -j

        if k_lo <= k_hi:

            grad[j] += np.sum(weights[ks+n-1] * (1 - h[j+k_lo:k_hi+1]))

```python
def compute_gradient_vec(h, weights, n):
    # Vectorized O(n^2) using broadcasting
    j = np.arange(n)
    k = np.arange(-(n-1), n)
    # m = j + k for first term, m = j - k for second term
    m1 = j[:, None] + k[None, :]  # shape (n, 2n-1)
    m2 = j[:, None] - k[None, :]  # shape (n, 2n-1)
    
    valid1 = (m1 >= 0) & (m1 < n)
    valid2 = (m2 >= 0) & (m2 < n)
    
    # h[m1] with boundary
    h_m1 = np.where(valid1, h[np.clip(m1, 0, n-1)], 0.0)
    h_m2 = np.where(valid2, h[np.clip(m2, 0, n-1)], 0.0)
    
    w = weights[None, :]  # shape (1, 2n-1)
    
    grad = np.sum(w * (1.0 - h_m1) * valid1, axis=1) - np.sum(w * h_m2 * valid2, axis=1)
    return grad
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):

        else:
            hi = mid

    zeros = np.where(h_r == 0)[0]

    k = np.arange(-(n-1), n)
    m1 = j[:, None] + k[None, :]
    m2 = j[:, None] - k[None, :]
    valid1 = (m1 >= 0) & (m1 < n)
    valid2 = (m2 >= 0) & (m2 < n)
    h_m1 = np.where(valid1, h[np.clip(m1, 0, n-1)], 0.0)
    h_m2 = np.where(valid2, h[np.clip(m2, 0, n-1)], 0.0)
    w = weights[None, :]
    s1 = np.sum(w * (1.0 - h_m1) * valid1.astype(float), axis=1)
    s2 = np.sum(w * h_m2 * valid2.astype(float), axis=1)
    return s1 - s2

                    stall = 0
                else:
                    stall += 1
                if stall > 50:
                    v *= 0.3
                    lr *= 0.8
                    stall = 0
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    return best_h, best_score

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    positions = np.arange(n)
    end_time = start_time + time_limit

    for it in range(n_iters):
        if time.time() > end_time:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if it % 5 == 0:
            top_k = 4
            top_indices = np.argsort(full)[::-1][:top_k]
            move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    move_found = True
                    break
            if not move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def sa_block(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr, block_frac=0.2, max_block_len=10):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)
    end_time = start_time + time_limit

    for it in range(n_iters):
        if time.time() > end_time:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if rng.random() < block_frac:
            blen = rng.randint(3, min(max_block_len, n // 10) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)
    end_time = start_time + time_limit

    for it in range(max_iters):
        if time.time() > end_time:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    target_int = int(n / 2.0)
    k_arr = np.arange(2 * n - 1) - (n - 1)

    h = project_binary(h_init, target_int)
    best_h = h.copy()
    best_score = compute_c5(h)

    # Phase 1: Quick greedy
    if tleft() > 30:
        allowed = min(25, tleft() * 0.025)
        h_g, c5_g = fast_greedy(h, max_iters=15000, rng=rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Primal-dual continuous optimization (main improvement)
    if tleft() > 200:
        allowed = min(350, tleft() * 0.3)
        h_pd, c5_pd = primal_dual_optimize(best_h, start_time, allowed)
        if c5_pd < best_score:
            best_score = c5_pd
            best_h = h_pd.copy()

    # Phase 3: SMax optimization with multiple alphas
    if tleft() > 150:
        allowed = min(200, tleft() * 0.22)
        alphas = [10, 30, 100, 300, 1000, 3000, 10000]
        h_smax, c5_smax = smax_optimize(best_h, start_time, allowed, alphas)
        if c5_smax < best_score:
            best_score = c5_smax
            best_h = h_smax.copy()

    # Phase 4: Binary projection + focused SA
    if tleft() > 100:
        h_bin = project_binary(best_h, target_int)
        allowed = min(120, tleft() * 0.15)
        n_sa = int(min(2500000, allowed * 50000))
        h_sa, c5_sa = sa_focused(h_bin, n_sa, 0.015, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Second primal-dual from current best
    if tleft() > 150:
        allowed = min(200, tleft() * 0.2)
        h_pd2, c5_pd2 = primal_dual_optimize(best_h, start_time, allowed)
        if c5_pd2 < best_score:
            best_score = c5_pd2
            best_h = h_pd2.copy()

    # Phase 6: Multi-start SA with diverse perturbations
    for restart in range(12):
        if tleft() < 50:
            break
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(4, 10))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 12)
        else:
            h_pert = perturb_diverse(best_h, rng, 5)
            h_pert = perturb_block(h_pert, rng, 6)

        allowed = min(35, tleft() * 0.04)
        n_sa_r = int(min(1000000, allowed * 45000))
        h_sr, c5_sr = sa_focused(h_pert, n_sa_r, 0.01, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: SLSQP refinement
    if tleft() > 80:
        alphas_slsqp = [50, 200, 800, 3000, 10000, 30000]
        allowed = min(tleft() - 30, 150)
        h_ref, sc = slsqp_refine(best_h, n, target, alphas_slsqp, 2000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 8: SMax from SLSQP result
    if tleft() > 80:
        alphas2 = [50, 200, 800, 3000, 10000]
        allowed = min(tleft() - 20, 100)
        h_sm2, sc2 = smax_optimize(best_h, start_time, allowed, alphas2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sm2.copy()

    # Phase 9: Final binary SA burst
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target_int)
        allowed = max(5.0, tleft() - 15)
        n_sa_f = int(min(600000, allowed * 40000))
        h_sf, c5_sf = sa_focused(h_bin_final, n_sa_f, 0.005, 0.00001, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Phase 10: Final SLSQP polish
    if tleft() > 20:
        allowed = max(5.0, tleft() - 8)
        h_ref2, sc2 = slsqp_refine(best_h, n, target, [5000, 20000, 60000], 1500, 1e-15,
                                    time_limit=allowed, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
