```python
def compute_corr_gradient(h, k_star):
    """Compute gradient of corr[k_star] with respect to h."""
    n = len(h)
    mid = n - 1
    k = k_star - mid  # actual lag
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def compute_corr_gradient_multi(h, k_stars, weights):
    """Compute weighted sum of gradients for multiple active lags."""
    n = len(h)

        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]

        #                sum(h) = target, 0 <= h <= 1

        c_obj[n] = 1.0  # minimize t

        # t - f_i - g_iᵀ(h - h_i) >= 0  =>  -(t - f_i - g_iᵀ(h - h_i)) <= 0
        # => -t + f_i + g_iᵀh - g_iᵀh_i <= 0

            A_ineq[i, n] = -1.0
            b_ineq[i] = cuts_g[i] @ cuts_h[i] - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(-1.0, None)]

            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if res.success:
                h_new = res.x[:n]
                # Project to ensure feasibility
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                
                # Check gap
                t_val = float(res.x[n])
                gap = f_val - t_val  # should be >= 0 at optimum
                # Actually the gap is: f(h_new) - t_new
                # If f(h_new) is very close to t_new (the LP objective), we're at minimum

                    best_h = h_new.copy()

                actual_gap = abs(compute_c5(h) / dx - t_val)
                if actual_gap < tol * 10:
                    break
            else:
                break
        except Exception:
            break
        
        # Additional convergence: if h barely changed
        # (can't easily check without storing previous h)
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

    dx = 2.0 / n

    cuts_f = []
    cuts_h = []

    max_cuts = 150

    for it in range(max_iters):

def interceptive_cd(h_init, n, target, time_limit, max_sweeps=30000,
                   n_cand_start=20, n_cand_end=80, max_no_improve=500):

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()

        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_pg(h_init, n, target, time_limit, alpha_init=500.0, alpha_end=50000.0,
                max_iters=20000, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        alpha = alpha_init * (alpha_end / alpha_init) ** (it / max_iters)
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p = p / np.sum(p)

        idxs = np.arange(n)
        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        step_backup = 0.01
        h_new = y - step_backup * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < compute_c5(h_curr) + 1e-10:
            h_prev = h_curr.copy()
            h_curr = h_new
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_curr.copy()
        else:
            step_backup *= 0.5
            h_new = y - step_backup * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            h_prev = h_curr.copy()
            h_curr = h_new
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_curr.copy()

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_sls, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    if tleft() > 250:
        h_cp, sc_cp = cutting_plane_method(best_h, n0, target0,
                                           time_limit=min(350, tleft() * 0.6),
                                           max_iters=10000, tol=1e-9)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if tleft() > 200:
        h_cd, sc_cd = interceptive_cd(best_h, n0, target0,
                                      time_limit=min(250, tleft() * 0.5),
                                      max_sweeps=30000,
                                      n_cand_start=20, n_cand_end=80,
                                      max_no_improve=500)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 150:
        h_cp2, sc_cp2 = cutting_plane_method(best_h, n0, target0,
                                             time_limit=min(200, tleft() * 0.5),
                                             max_iters=8000, tol=1e-10)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if tleft() > 100:
        h_sls2, sc2 = slsqp_round(best_h, n0, target0,
                                  [5000, 20000, 80000], 800, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sls2

    if tleft() > 120:
        h_cd3, sc3 = interceptive_cd(best_h, n0, target0,
                                     time_limit=min(150, tleft() * 0.5),
                                     max_sweeps=20000,
                                     n_cand_start=30, n_cand_end=90,
                                     max_no_improve=400)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_cd3

    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h800 = interpolate_to(best_h, n1, target1)
        sc800 = compute_c5(h800)
        if tleft() > 150:
            h800_cd, sc800_cd = interceptive_cd(h800, n1, target1,
                                                 time_limit=min(150, tleft() * 0.4),
                                                 max_sweeps=20000,
                                                 n_cand_start=15, n_cand_end=50,
                                                 max_no_improve=300)
            if sc800_cd < best_score:
                best_score = sc800_cd
                best_h = h800_cd
                n0 = n1
                target0 = target1
            if tleft() > 60:
                h800_sls, sc800_s = slsqp_round(best_h, n0, target0,
                                                 [2000, 10000], 500, 1e-14)
                if sc800_s < best_score:
                    best_score = sc800_s
                    best_h = h800_sls

    if tleft() > 80:
        h_cp3, sc_cp3 = cutting_plane_method(best_h, n0, target0,
                                             time_limit=min(80, tleft() * 0.5),
                                             max_iters=5000, tol=1e-10)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3

    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.05]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = cutting_plane_method(h_pert, n0, target0,
                                                 time_limit=min(40, tleft() * 0.4),
                                                 max_iters=3000, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 40:
        h_cd4, sc4 = interceptive_cd(best_h, n0, target0,
                                     time_limit=min(35, tleft() * 0.5),
                                     max_sweeps=10000,
                                     n_cand_start=40, n_cand_end=100,
                                     max_no_improve=200)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_cd4

    if tleft() > 30:
        h_final_sls, sc_final = slsqp_round(best_h, n0, target0,
                                            [10000, 50000], 500, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
