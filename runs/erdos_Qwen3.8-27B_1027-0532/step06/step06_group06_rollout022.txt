    hi = 2.0

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = current_sum - target_sum
        idx = np.argsort(h)[:need]
        h_r[idx] = 0.0
    else:
        need = target_sum - current_sum
        idx = np.argsort(-h)[:need]
        h_r[idx] = 1.0
    return h_r

def compute_profile(h):

    valid2 = (idx2 >= 0) & (idx2 < n)

    term2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
