∇F = (1/β) * (∑_k exp(β*f_k) * ∇f_k) / (∑_k exp(β*f_k))
= ∑_k p_k * ∇f_k where p_k = exp(β*f_k) / ∑_j exp(β*f_j)

This is a weighted average of the gradients, with weights concentrated on the active lags.

The Hessian involves both the weighting and the individual Hessians, but for L-BFGS-B we only need the gradient.

Let me implement this approach with careful temperature scheduling:

```python
def smooth_gradient(h, beta, n):
    """Compute gradient of the smooth max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idxs = np.arange(n)
    mid = n - 1
    
    # Compute weights
    shifted = beta * (corr - np.max(corr))
    shifted = np.clip(shifted, -700, 0)  # prevent overflow
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for w, s_idx in zip(weights, range(len(corr))):
        if w < 1e-10:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    grad *= dx
    return grad, np.max(corr) * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_gradient_at_lag(h, k_shift, n):
    """Get the gradient of corr at lag k_shift (= index - (n-1))."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
    v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_shift])
    grad[v2] -= h[idxs[v2] + k_shift]
    return grad

def smooth_logsumexp(h, beta, n):
    """Compute smooth log-sum-exp objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700, 0)
    exp_vals = np.exp(shifted)
    raw_sum = exp_vals.sum()
    obj = max_c + np.log(raw_sum) / beta
    
    weights = exp_vals / raw_sum
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    
    for s_idx in np.where(weights > 1e-10)[0]:

        v1 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v2] -= w * h[idxs[v2] + k]
    
    grad *= dx
    return obj * dx, grad  # scale by dx to get C5 scale

    """Run L-BFGS-B on the smooth objective."""

    dx = 2.0 / n
    penalty = 1e8

        o, g = smooth_logsumexp(x, beta, n)
        o += penalty * s * s
        g += 2 * penalty * s
        return o, g
    
    bounds = [(0.0, 1.0)] * n
    try:

                      bounds=bounds, jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-12, 'maxfun': maxiter * 2})
        h_new = project_feasible(res.x, target)
        return h_new

                       max_sweeps=5000):

    best_h = h.copy()

            if no_improve > patience:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700, 0)
    exp_vals = np.exp(shifted)
    raw_sum = exp_vals.sum()
    obj = (max_c + np.log(raw_sum) / beta) * dx
    weights = exp_vals / raw_sum
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for s_idx in np.where(weights > 1e-10)[0]:
        w = weights[s_idx]
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return obj, grad

def lbfags_smooth(h_start, n, target, beta, maxiter, elapsed_fn):
    dx = 2.0 / n
    penalty = 1e8
    bounds = [(0.0, 1.0)] * n
    def obj_and_grad(x):
        s = np.sum(x) - target
        o, g = smooth_obj_grad(x, beta, n)
        o += penalty * s * s
        g = g + 2.0 * penalty * s
        return o, g
    try:
        res = minimize(obj_and_grad, h_start, method='L-BFGS-B',
                      bounds=bounds, jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-16,
                               'gtol': 1e-10, 'maxfun': maxiter * 3})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def game_theoretic_refined(h_init, n, target, time_limit, max_steps=8000,
                           eta_base=1.5, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.15, 0.5)
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.85
            if best_lr < 0.002:
                break

        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.6 * lam + 0.4 * lam_new

        if step % 300 == 0 and no_improve > 80:

                best_c5 = compute_c5(h)

def subgradient_momentum(h_init, n, target, time_limit, max_steps=15000,

    v = np.zeros(n)

            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        lr = base_lr * max(0.05, 1.0 - step / max_steps)
        h_new = project_feasible(h - lr * v, target)

            h_ref = lbfags_smooth(h_pert, n, target, 5000, maxiter=500,

            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if tleft() > 40:
                    h_cd2, c5_cd2 = pair_cd_refined(best_h, n, target,
                                                    time_limit=min(30, tleft() * 0.2),
                                                    n_cand=40, patience=50,
                                                    max_sweeps=3000)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2

    # Phase 9: Local perturbation search
    if tleft() > 50:
        h_pert, c5_pert = local_perturb_search(best_h, n, target,
                                               max_tries=5000,
                                               time_limit=min(30, tleft() * 0.3),
                                               seed=seed + 1)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 10: Final pair CD polish
    if tleft() > 60:
        cd_time = min(50, tleft() * 0.4)
        h_cd, c5_cd = pair_cd_refined(best_h, n, target,
                                      time_limit=cd_time,
                                      n_cand=40, patience=40,
                                      max_sweeps=5000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    final_h = project_feasible(best_h, target)
    final_score = compute_c5(final_h)
    if final_score > best_score:
        final_h = project_feasible(best_h, target)
        final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
