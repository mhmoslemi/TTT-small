                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def get_grad_shift_vec(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    if k >= 0:
        i_valid = np.arange(0, n - k)
        g[i_valid] += (1.0 - h[i_valid + k])
        g[i_valid + k] -= h[i_valid]
    else:
        ka = -k
        i_valid = np.arange(0, n - ka)
        g[i_valid + ka] += (1.0 - h[i_valid])
        g[i_valid] -= h[i_valid + ka]
    return g

def softmax_subgrad(h_start, n, target, time_limit, max_steps, n_top, base_lr, alpha_sched):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        c5_now = float(np.max(C))
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = np.max(corr)
        alpha_w = 30.0 + alpha_sched * step
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        frac = min(1.0, step / max(1, max_steps * 0.5))
        n_top_cur = int(n_top + (min(2 * n_top, n) - n_top) * frac)
        top_idx = np.argsort(corr)[-n_top_cur:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            w = weights[s_idx]
            if w < 1e-8:
                continue
            grad += w * get_grad_shift_vec(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_cur = base_lr / (1.0 + 0.001 * step)
        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def pair_cd_bisect(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 150:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift_vec(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-10:
                    continue
                lo = 0.0
                hi = max_d
                best_local = c5_val
                best_local_d = 0.0
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi)
                    h_try = h.copy()
                    h_try[i] -= mid_d
                    h_try[j] += mid_d
                    c5_try = compute_c5(h_try)
                    if c5_try < best_local:
                        best_local = c5_try
                        best_local_d = mid_d
                    if c5_try < c5_val:
                        lo = mid_d
                    else:
                        hi = mid_d
                if best_local < best_trial:
                    best_trial = best_local
                    best_i = i
                    best_j = j
                    best_delta = best_local_d
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def basin_hopping_inner(h_init, n, target, time_limit, rng, max_iter):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        scale = 0.03 + 0.08 * (1.0 - it / max(max_iter, 1))
        n_pert = int(rng.choice([2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_pert = h_cur.copy()
        h_pert[idxs] += rng.randn(n_pert) * scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_softmax(h_pert, n, target,
                                   [2000, 10000, 50000],
                                   500, 1e-14, time_limit=15.0)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_ref.copy()
            h_cur = h_ref
    return best_h, best_c5

def multi_start_explore(h_base, n, target, time_limit, rng):
    start = time.time()
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    candidates = []
    for scale in [0.01, 0.03, 0.05, 0.08, 0.12]:
        h_pert = h_base + rng.randn(n) * scale
        candidates.append(project_feasible(h_pert, target))
    for freq in [1, 2, 3]:
        x = np.linspace(0, 2, n, endpoint=False)
        for phase in [0.0, np.pi / 2, np.pi, 3 * np.pi / 2]:
            h_sin = h_base * 0.7 + 0.3 * (0.5 + 0.5 * np.sin(2 * np.pi * freq * x / 2.0 + phase))
            candidates.append(project_feasible(h_sin, target))
    for h_cand in candidates:
        if time.time() - start > time_limit:
            break
        h_ref, sc = slsqp_softmax(h_cand, n, target,
                                   [1000, 5000, 20000],
                                   400, 1e-14, time_limit=min(20, time_limit * 0.15))
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_ref.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP refinement of initial point
    if tleft() > 80:
        h_pol, sc = slsqp_softmax(best_h, n, target,
                                   [500, 2000, 10000, 50000, 200000],
                                   1500, 1e-15, time_limit=min(50, tleft() * 0.35))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Softmax subgradient with momentum
    if tleft() > 60:
        h_sg, sc = softmax_subgrad(best_h, n, target,
                                    time_limit=min(35, tleft() * 0.25),
                                    max_steps=8000, n_top=5,
                                    base_lr=0.012, alpha_sched=0.5)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 3: Multi-start exploration
    if tleft() > 100:
        h_ms, sc = multi_start_explore(best_h, n, target,
                                        time_limit=min(70, tleft() * 0.4), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_ms

    # Phase 4: Pair coordinate descent with bisection
    if tleft() > 80:
        h_cd, sc = pair_cd_bisect(best_h, n, target,
                                   time_limit=min(50, tleft() * 0.3),
                                   max_sweeps=1500, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 5: SLSQP high-alpha polish
    if tleft() > 60:
        h_pol, sc = slsqp_softmax(best_h, n, target,
                                   [50000, 200000, 1000000],
                                   1000, 1e-16, time_limit=min(40, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Basin hopping
    if tleft() > 120:
        h_bh, sc = basin_hopping_inner(best_h, n, target,
                                        time_limit=min(80, tleft() * 0.5),
                                        rng=rng, max_iter=15)
        if sc < best_score:
            best_score = sc
            best_h = h_bh

    # Phase 7: More subgradient after basin hopping
    if tleft() > 50:
        h_sg, sc = softmax_subgrad(best_h, n, target,
                                    time_limit=min(30, tleft() * 0.3),
                                    max_steps=5000, n_top=3,
                                    base_lr=0.005, alpha_sched=1.0)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 8: Resolution increase
    if tleft() > 150:
        n_hi = min(800, 2 * n)
        target_hi = float(n_hi) / 2.0
        h_interp = interpolate_to(best_h, n_hi, target_hi)
        h_hi, sc = slsqp_softmax(h_interp, n_hi, target_hi,
                                  [500, 2000, 10000, 50000, 200000],
                                  800, 1e-15, time_limit=min(100, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_hi
            n = n_hi
            target = float(n_hi) / 2.0

    # Phase 9: Final pair CD on best resolution
    if tleft() > 50:
        n_use = len(best_h)
        tgt_use = float(n_use) / 2.0
        h_cd, sc = pair_cd_bisect(best_h, n_use, tgt_use,
                                   time_limit=min(35, tleft() * 0.4),
                                   max_sweeps=1000, n_cand=15)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 10: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        tgt_use = float(n_use) / 2.0
        h_fin, sc = slsqp_softmax(best_h, n_use, tgt_use,
                                   [100000, 500000, 2000000],
                                   800, 1e-16, time_limit=min(25, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    n_final = len(best_h)
    tgt_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, tgt_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
