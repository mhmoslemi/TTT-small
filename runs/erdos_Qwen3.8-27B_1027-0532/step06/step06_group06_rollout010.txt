def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    h = h.copy()
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
        else:
            h_r[:] = 0
            target_indices = np.argsort(-h)[:target_sum]
            h_r[target_indices] = 1
    else:

        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
        else:
            h_r[:] = 1
            target_indices = np.argsort(h)[:n - target_sum]
            h_r[target_indices] = 0
    return h_r.astype(float)

```python
def lse_gradient_fast(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # w has length 2N-1
    # For each j, grad[j] = dx * sum_{m=-(N-1)}^{N-1} w[m+N-1] * ((1-h[j-m])*1_{valid} - h[j+m]*1_{valid})
    
    # Vectorize: create a (N, 2N-1) matrix where entry (j, m) = w[m] * ((1-h[j-m]) - h[j+m])
    # Then grad = dx * (matrix @ 1)
    
    # But this creates an N x (2N-1) array which for N=606 is 606*1211 ≈ 734K elements
    # That's fine memory-wise.
    
    j_idx = np.arange(N)[:, None]  # (N, 1)
    m_idx = np.arange(-(N-1), N)[None, :]  # (1, 2N-1)
    
    # idx_omh = j - m, valid when 0 <= j-m < N
    idx_omh = j_idx - m_idx  # (N, 2N-1)
    valid_omh = (idx_omh >= 0) & (idx_omh < N)
    
    # idx_h = j + m, valid when 0 <= j+m < N
    idx_h = j_idx + m_idx  # (N, 2N-1)
    valid_h = (idx_h >= 0) & (idx_h < N)
    
    one_minus_h = 1.0 - h
    # part1[j, m] = (1-h[j-m]) if valid else 0
    part1 = np.where(valid_omh, one_minus_h[np.clip(idx_omh, 0, N-1)], 0.0)
    # part2[j, m] = h[j+m] if valid else 0
    part2 = np.where(valid_h, h[np.clip(idx_h, 0, N-1)], 0.0)
    
    # diff[j, m] = part1 - part2
    diff = part1 - part2
    
    # grad[j] = dx * sum_m w[m] * diff[j, m]
    grad = dx * (diff @ w)
    
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad
```

This creates arrays of size N*(2N-1) = 606*1211 ≈ 734K, which is fine. The matmul is O(N^2 * (2N)) = O(N^3) ≈ 606^3 ≈ 223M operations. In numpy, this should take maybe 10-50ms. That's acceptable.

Actually wait, the matmul diff @ w is (N, 2N-1) @ (2N-1,) = (N,), which is O(N * 2N) = O(N^2) ≈ 734K multiply-adds. That's very fast.

The creation of the arrays part1 and part2 is the expensive part: O(N * 2N) = O(N^2) each, plus the clipping. Total maybe a few ms.

def fast_binary_ls(h_init, target_sum, time_limit, start_time, max_stall=300):
    n = len(h_init)
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    stall = 0
    n_lags_to_check = 5
    n_candidates_per_lag = 5
    
    while time.time() - start_time < time_limit:
        if stall > max_stall:
            break
        
        max_val = float(np.max(corr))
        
        # Find active lags (within 2% of max)
        threshold = max_val * 0.98
        active_lags = np.where(corr >= threshold)[0]
        active_lags = active_lags[:n_lags_to_check]
        
        if len(active_lags) == 0:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        candidates = set()
        
        for lag_idx in active_lags:
            k = lag_idx - (n - 1)
            if k == 0:
                continue
            
            # Valid i: h[i]=1, i+k in [0,n), i-k in [0,n)
            mask_i = (ones + k >= 0) & (ones + k < n) & (ones - k >= 0) & (ones - k < n)
            valid_i = ones[mask_i]
            
            mask_j = (zeros + k >= 0) & (zeros + k < n) & (zeros - k >= 0) & (zeros - k < n)
            valid_j = zeros[mask_j]
            
            if len(valid_i) == 0 or len(valid_j) == 0:
                continue
            
            a = h[valid_i + k].astype(float) + h[valid_i - k].astype(float)
            b = h[valid_j + k].astype(float) + h[valid_j - k].astype(float)
            
            top_i = valid_i[np.argsort(-a)[:n_candidates_per_lag]]
            top_j = valid_j[np.argsort(b)[:n_candidates_per_lag]]
            
            for i_c in top_i:
                for j_c in top_j:
                    candidates.add((int(i_c), int(j_c)))
        
        if len(candidates) == 0:
            stall += 1
            break
        
        best_improve = 0.0
        best_pair = None
        
        for i_c, j_c in candidates:
            if h[i_c] != 1 or h[j_c] != 0:
                continue
            h[i_c] = 0
            h[j_c] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr))
            # Revert
            h[i_c] = 1
            h[j_c] = 0
            
            if new_max < max_val - best_improve:
                best_improve = max_val - new_max
                best_pair = (i_c, j_c)
        
        if best_pair is not None:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            stall = 0
        else:
            # Try a random perturbation to escape local optimum
            stall += 1
            if stall > max_stall // 2:
                # Random swap to restart
                i_rand = ones[np.random.randint(len(ones))]
                j_rand = zeros[np.random.randint(len(zeros))]
                h[i_rand] = 0
                h[j_rand] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                # Don't update best
                stall = 0  # reset stall after perturbation
    
    return best_h, best_c5

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    h = h.copy()
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def lse_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def lse_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    j_idx = np.arange(N)[:, None].astype(np.int64)
    m_idx = np.arange(-(N-1), N)[None, :].astype(np.int64)
    
    idx_omh = j_idx - m_idx
    valid_omh = (idx_omh >= 0) & (idx_omh < N)
    idx_h = j_idx + m_idx
    valid_h = (idx_h >= 0) & (idx_h < N)
    
    one_minus_h = 1.0 - h
    part1 = np.where(valid_omh, one_minus_h[np.clip(idx_omh, 0, N-1)], 0.0)
    part2 = np.where(valid_h, h[np.clip(idx_h, 0, N-1)], 0.0)
    
    diff = part1 - part2
    grad = dx * (diff @ w)
    
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_round(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lse_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=lse_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_candidate
        except Exception:
            pass
    return h_cur

def build_s_matrix(h):
    """Build s[p, k_index] = h[p+k] + h[p-k] for all valid p,k"""
    n = len(h)
    s = np.zeros((n, 2*n - 1))
    for k in range(-(n-1), n):
        ki = k + n - 1
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            s[lo:hi, ki] += h[lo+k:hi+k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            s[lo2:hi2, ki] += h[lo2-k:hi2-k]
    return s

def full_polish(h_init, target_sum, max_rounds=50, time_limit=60, start_time=0):
    """Exhaustively try all single swaps to find local optimum."""
    n = len(h_init)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    for round_num in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        s = build_s_matrix(h)
        corr_arr = corr
        
        best_new_max = float(corr_arr.max())
        best_i, best_j = -1, -1
        
        for i in ones:
            f = corr_arr + s[i]
            # For each j in zeros: new_max_j = max_k(f[k] - s[j,k])
            diff = f[None, :] - s[zeros, :]
            new_maks = diff.max(axis=1)
            j_best_local = zeros[int(np.argmin(new_maks))]
            new_max_val = float(new_maks.min())
            
            if new_max_val < best_new_max:
                best_new_max = new_max_val
                best_i = int(i)
                best_j = int(j_best_local)
        
        if best_i < 0:
            break  # No improvement possible
        
        h[best_i] = 0
        h[best_j] = 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(corr)) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5

def fast_binary_ls(h_init, target_sum, time_limit, start_time):
    """Fast binary local search using candidate generation from active lags."""
    n = len(h_init)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    
    stall = 0
    max_stall = 100
    
    while time.time() - start_time < time_limit:
        if stall > max_stall:
            # Escape: random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            idx = rng.choice(len(ones), 2, replace=False)
            jdx = rng.choice(len(zeros), 2, replace=False)
            h[ones[idx[0]]] = 0
            h[ones[idx[1]]] = 0
            h[zeros[jdx[0]]] = 1
            h[zeros[jdx[1]]] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            continue
        
        max_val = float(np.max(corr))
        threshold = max_val * 0.97
        active_lags = np.where(corr >= threshold)[0]
        active_lags = active_lags[:8]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        candidates = set()
        
        for lag_idx in active_lags:
            k = lag_idx - (n - 1)
            if k == 0:
                continue
            mask_i = (ones + k >= 0) & (ones + k < n) & (ones - k >= 0) & (ones - k < n)
            mask_j = (zeros + k >= 0) & (zeros + k < n) & (zeros - k >= 0) & (zeros - k < n)
            valid_i = ones[mask_i]
            valid_j = zeros[mask_j]
            
            if len(valid_i) == 0 or len(valid_j) == 0:
                continue
            
            a = h[valid_i + k] + h[valid_i - k]
            b = h[valid_j + k] + h[valid_j - k]
            
            top_i = valid_i[np.argsort(-a)[:6]]
            top_j = valid_j[np.argsort(b)[:6]]
            
            for i_c in top_i:
                for j_c in top_j:
                    candidates.add((int(i_c), int(j_c)))
        
        if len(candidates) == 0:
            stall += 1
            continue
        
        best_improve = 0.0
        best_pair = None
        
        for i_c, j_c in candidates:
            if h[i_c] != 1 or h[j_c] != 0:
                continue
            h[i_c] = 0
            h[j_c] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr))
            h[i_c] = 1
            h[j_c] = 0
            
            if new_max < max_val - best_improve:
                best_improve = max_val - new_max
                best_pair = (i_c, j_c)
        
        if best_pair is not None:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)
