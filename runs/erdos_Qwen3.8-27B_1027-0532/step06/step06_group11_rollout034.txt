from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float)
    n = len(h)
    lo = h.min() - 2.0
    hi = 1.0 - h.min() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(h + lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        interior = np.where((proj > 1e-12) & (proj < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            proj[interior] += diff / len(interior)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

# Global variables for SLSQP to avoid closure issues
_G_n = 0
_G_target = 0.0
_G_alpha = 0.0
_G_horizon = 0.0

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    # Smooth max via log-sum-exp
    shifted = _G_alpha * (C - max_C)
    shifted = np.clip(shifted, -300.0, 0.0)
    smooth_max = max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha
    # Penalty for sum constraint (backup)
    pen = 100.0 * (np.sum(h) - _G_target) ** 2
    return float(smooth_max + pen)

def _slsqp_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = _G_alpha * (C - max_C)
    shifted = np.clip(shifted, -300.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(2*n-1)
        weights[np.argmax(C)] = 1.0
    else:
        weights /= wsum
    
    mid = n - 1
    grad = np.zeros(n)
    for s_idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        k = int(s_idx) - mid
        grad += w * get_grad_shift(h, k, n)
    grad *= dx
    # Add penalty gradient
    grad += 200.0 * (np.sum(h) - _G_target) * np.ones(n)
    return grad

def slsqp_refine(h_start, n, target, time_limit):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    start = time.time()
    
    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    maxiters = [500, 500, 400, 300, 250, 200, 150]
    
    for alpha, niter in zip(alphas, maxiters):
        if time.time() - start > time_limit:
            break
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad, method='SLSQP',
                         bounds=bounds,
                         options={'maxiter': niter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5
