def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_shift_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def soft_max_grad(h, n, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    max_c = np.max(c)

        if noimp > 150:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cv = float(corr.max()) * dx
        if cv < bs - 1e-13:
            bs = cv
            bh = h.copy()
            noimp = 0
        else:
            noimp += 1
        ks = int(corr.argmax()) - mid
        g = shift_grad(h, ks, n)
        g -= g.mean()
        si = np.argsort(g)
        lo = si[:nc]
        hi = si[-nc:]
        fa = np.zeros((n, 2 * n - 1))
        for p in range(n):
            i1 = p + n - 1 - np.arange(2 * n - 1)
            m1 = (i1 >= 0) & (i1 < n)
            fa[p, m1] += h[i1[m1]]
            i2 = p + np.arange(2 * n - 1) - n + 1
            m2 = (i2 >= 0) & (i2 < n)
            fa[p, m2] += h[i2[m2]]
        bt = cv
        bm = None
        for i in hi:
            for j in lo:
                if i == j:
                    continue
                dc = fa[i] - fa[j]
                ad = 0.0
                bd = min(h[i], 1.0 - h[j])
                if bd <= 1e-14:
                    continue
                for _ in range(18):
                    md = 0.5 * (ad + bd)
                    vals = corr + md * dc
                    fm = vals.max()
                    act = vals >= fm - 1e-14
                    if act.any():
                        fp = dc[act].max()
                    else:
                        fp = 0.0
                    if fp > 1e-14:
                        bd = md
                    else:
                        ad = md
                ds = 0.5 * (ad + bd)
                if abs(ds) < 1e-12:
                    continue
                tc = float((corr + ds * dc).max()) * dx
                if tc < bt - 1e-14:
                    bt = tc
                    bm = (i, j, ds)
        if bm is not None:
            i, j, d = bm
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0, 1)
            h = project_feasible(h, tgt)
    return bh, bs

def interp_to(hs, nd, td):
    ns = len(hs)
    if ns == nd:
        return project_feasible(hs.copy(), td)
    xn = np.linspace(0.0, 1.0, nd)
    xo = np.linspace(0.0, 1.0, ns)
    return project_feasible(np.interp(xn, xo, hs), td)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def tl():
        return budget_s - 30.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    tgt = n / 2.0

    bh = project_feasible(h_init, tgt)
    bs = compute_c5(bh)

    if tl() > 300:
        h_new, sc = lbfgs_run(bh, n, tgt,
                              [20, 50, 150, 400, 1000, 3000, 10000, 30000],
                              300, min(220, tl() * 0.3))
        if sc < bs:
            bs, bh = sc, h_new

    if tl() > 180:
        h_new, sc = adam_run(bh, n, tgt,
                            min(130, tl() * 0.3), 12000, 30, 5000, 0.04)
        if sc < bs:
            bs, bh = sc, h_new

    if tl() > 150:
        h_new, sc = pcd(bh, n, tgt,
                        min(120, tl() * 0.4), 800, 20)
        if sc < bs:
            bs, bh = sc, h_new

    if tl() > 180:
        h_new, sc = lbfgs_run(bh, n, tgt,
                              [10000, 30000, 100000, 300000],
                              200, min(120, tl() * 0.35))
        if sc < bs:
            bs, bh = sc, h_new

    if tl() > 250:
        n_try = 600
        td = n_try / 2.0
        hi = interp_to(bh, n_try, td)
        h_new, sc = lbfgs_run(hi, n_try, td,
                              [20, 50, 150, 400, 1000, 3000],
                              250, min(150, tl() * 0.4))
        if sc < bs:
            bs, bh = sc, h_new
            n = n_try
            tgt = td
        if tl() > 100:
            h_new, sc = adam_run(bh, n, tgt,
                                 min(70, tl() * 0.3), 8000, 50, 3000, 0.03)
            if sc < bs:
                bs, bh = sc, h_new
        if tl() > 60:
            h_new, sc = pcd(bh, n, tgt,
                            min(50, tl() * 0.4), 500, 15)
            if sc < bs:
                bs, bh = sc, h_new

    if tl() > 300:
        n_try = 800
        td = n_try / 2.0
        hi = interp_to(bh, n_try, td)
        h_new, sc = lbfgs_run(hi, n_try, td,
                              [20, 50, 150, 400, 1000],
                              200, min(180, tl() * 0.5))
        if sc < bs:
            bs, bh = sc, h_new
            n = n_try
            tgt = td
        if tl() > 80:
            h_new, sc = adam_run(bh, n, tgt,
                                 min(60, tl() * 0.3), 6000, 50, 2000, 0.02)
            if sc < bs:
                bs, bh = sc, h_new

    if tl() > 80:
        nu = len(bh)
        tu = nu / 2.0
        h_new, sc = pcd(bh, nu, tu,
                        min(60, tl() * 0.5), 600, 18)
        if sc < bs:
            bs, bh = sc, h_new

    rng = np.random.RandomState(seed)
    for ns in [0.02, 0.05, 0.1, 0.2]:
        if tl() < 80:
            break
        nu = len(bh)
        tu = nu / 2.0
        hp = project_feasible(bh + rng.randn(nu) * ns, tu)
        h_new, sc = lbfgs_run(hp, nu, tu,
                              [100, 500, 2000, 10000, 50000],
                              200, min(50, tl() * 0.3))
        if sc < bs:
            bs, bh = sc, h_new

    if tl() > 50:
        nu = len(bh)
        tu = nu / 2.0
        h_new, sc = lbfgs_run(bh, nu, tu,
                              [20000, 100000, 500000],
                              150, min(35, tl() * 0.6))

        if sc < bs:
            bs, bh = sc, h_new

    nu = len(bh)
    tu = nu / 2.0
    bh = project_feasible(bh, tu)
    bf = compute_c5(bh)

    return bh, bf, len(bh)
