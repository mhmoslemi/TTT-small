def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

            idx2 = np.where(h < 1.0)[0]
        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

                best_delta = hi

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except:
        return h_start

def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000, 
                 n_cand_start=25, n_cand_max=60, patience=150):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            # Reduce candidates when improving
            if no_improve < patience // 5 and n_cand > n_cand_start:
                n_cand = max(n_cand - 2, n_cand_start)
        else:
            no_improve += 1
            if no_improve > patience:
                break
            # Increase candidates when stuck
            if no_improve > patience // 3 and n_cand < n_cand_max:
                n_cand = min(n_cand + 3, n_cand_max)
        
        # Multi-lag gradient for candidate selection
        # Use top-3 lags for a more informed gradient
        top_k = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(top_k)
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Precompute effect vectors
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        effects = np.zeros((n_cands, n_corr))
        ks = np.arange(-(n-1), n)
        for ci in range(n_cands):
            c = int(all_cands[ci])
            vm = (c - ks >= 0) & (c - ks < n)
            vp = (c + ks >= 0) & (c + ks < n)
            effects[ci, vm] += (1.0 - h[c - ks[vm]])
            effects[ci, vp] -= h[c + ks[vp]]
        
        cand_map = {int(c): idx for idx, c in enumerate(all_cands)}
        
        best_trial = c5_val
        best_delta = 0.0
        best_i = -1
        best_j = -1
        
        # Evaluate all pairs
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            ei = effects[cand_map[i]]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                ej = effects[cand_map[j]]
                g = ej - ei
                
                lo_d = 0.0
                hi_d = min(h[i], 1.0 - h[j])
                if hi_d - lo_d < 1e-13:
                    continue
                
                # Bisection to find optimal delta
                a, b = lo_d, hi_d
                for _ in range(25):
                    mid_d = 0.5 * (a + b)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        slope = float(np.max(g[active_mask]))
                    else:
                        slope = 0.0
                    if slope < -1e-14:
                        a = mid_d
                    else:
                        b = mid_d
                
                delta_star = 0.5 * (a + b)
                if delta_star < 1e-13:
                    continue
                f_star = float(np.max(corr + delta_star * g)) * dx
                
                if f_star < best_trial:
                    best_trial = f_star
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def subgradient_method(h_init, n, target, time_limit, m=10, lr_init=0.03):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    idxs = np.arange(n)
    mid = n - 1
    lr = lr_init
    no_improve = 0
    
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_cur.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        top_idx = np.argsort(corr)[-m:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= h_cur[idxs[v2] + k]
        grad /= m
        grad *= dx
        grad -= np.mean(grad)
        
        found = False
        lr_try = lr
        for _ in range(10):
            h_try = project_feasible(h_cur - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h_cur = h_try
                found = True
                break
            lr_try *= 0.5
        
        if not found:
            lr *= 0.92
            if lr < 0.001:
                break
    
    return best_h, best_c5
