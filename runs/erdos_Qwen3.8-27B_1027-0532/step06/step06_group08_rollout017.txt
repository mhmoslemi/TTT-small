import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_at_lag(h, k):
    """Compute gradient of f_k(h) = sum h[i]*(1-h[i+k]) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    # f_k(h) = sum_i h[i] * (1 - h[i+k])
    # df_k/dh[i] = (1-h[i+k]) - h[i-k] (with out-of-range = 0)
    if k >= 0:
        # (1-h[i+k]) contributes for i in [0, n-1-k]
        g[:n-k] += (1.0 - h[k:])
        # -h[i-k] contributes for i in [k, n-1]
        g[k:] -= h[:n-k]
    else:
        ak = -k
        # (1-h[i+k]) = (1-h[i-ak]) contributes for i in [ak, n-1]
        g[ak:] += (1.0 - h[:n-ak])
        # -h[i-k] = -h[i+ak] contributes for i in [0, n-1-ak]
        g[:n-ak] -= h[ak:]
    return g

def cutting_plane_solve(h_init, n_iters_max, time_limit, start_time, verbose=False):
    """Solve min max_k f_k(h) using cutting plane (outer approximation) with LP."""

import numpy as np

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
    else:
        ak = -k
        if ak < n:
            g[ak:] += (1.0 - h[:n-ak])
            g[:n-ak] -= h[ak:]
    return g

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff > 0:
        margins = 1.0 - y
        total = margins.sum()
        if total > 1e-10:
            y += (diff / total) * margins
    else:
        total = y.sum()
        if total > 1e-10:
            y += (diff / total) * y
    return np.clip(y, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

    best_h = h.copy()

        if time.time() - start_time > time_limit:
            break
        if stall > 500:
            break

        corr = get_corr(h)
        curr_max = float(corr.max())

        c_obj[n] = 1.0

        b_ub = np.zeros(topk)

        for ii, ki in enumerate(top_idx):
            k = shifts[ki]
            g = gradient_at_lag(h, k)
            A_ub[ii, :n] = g
            A_ub[ii, n] = -1.0
            b_ub[ii] = float(np.dot(g, h)) - corr[ki]

        for ki in range(n_lags):
            if ki not in top_idx:
                if corr[ki] > curr_max - 1.0:
                    pass

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            stall += 1
            continue

        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_box_sum(h_new, target_sum)
        corr_new = get_corr(h_new)
        c5_new = float(corr_new.max()) * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new

    return best_h, best_c5

def slsqp_polish(h_init, time_limit, start_time, n_alphas=(50, 200, 1000)):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy().astype(float)
    curr_score = compute_c5(h)
    best_h = h.copy()
    best_score = curr_score
    bounds = [(0.0, 1.0)] * n

    for alpha in n_alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                cm = float(np.max(corr))
                shifted = np.clip(alpha * (corr - cm), -400.0, 0.0)
                return cm + float(np.log(np.sum(np.exp(shifted)))) / alpha

            def con_fun(x):
                return np.sum(x) - target_sum

            def con_jac(x):
                return np.ones(n)

            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h = h_new
                best_score = sc
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

def projected_gradient(h_init, time_limit, start_time, topk=3, lr_init=0.05, lr_end=0.001, max_iters=500000):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy().astype(float)
    corr = get_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    shifts = np.arange(2*n-1) - (n-1)
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break
        ratio = it / max(max_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        top_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for ki in top_idx:
            k = shifts[ki]
            g = gradient_at_lag(h, k)
            grad += g
        grad /= topk
        h = h - lr * grad
        h = project_box_sum(h, target_sum)
        corr = get_corr(h)
        c5_curr = float(corr.max()) * dx
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_c5 = compute_c5(h_init)
    best_h = project_box_sum(h_init, target)

    # Phase 1: Binary greedy (20s)
    alloc1 = min(20.0, tleft() * 0.08)
    if alloc1 > 5:
        h_bin = project_binary(h_init, n // 2)
        c5_bin0 = compute_c5(h_bin)
        h_bin, c5_bin = fast_greedy(h_bin, max_iters=50000, rng=rng,
                                     time_limit=alloc1, start_time=start_time)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = project_box_sum(h_bin, target)

    # Phase 2: Binary SA (80s)
    alloc2 = min(80.0, tleft() * 0.12)
    if alloc2 > 10:
        n_sa = int(min(3000000, alloc2 * 50000))
        h_sa, c5_sa = sa_binary(h_bin, n_sa, 0.015, 0.00005, rng,
                                 time_limit=alloc2, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = project_box_sum(h_sa, target)
            h_bin = h_sa.copy()

    # Phase 3: Cutting plane LP from continuous parent (400s)
    alloc3 = min(400.0, tleft() * 0.35)
    if alloc3 > 30:
        h_cp1, c5_cp1 = cutting_plane_solve(h_init, alloc3, start_time,
                                             max_iters=50000)
        if c5_cp1 < best_c5:
            best_c5 = c5_cp1
            best_h = h_cp1.copy()

    # Phase 4: Cutting plane from binary solution (200s)
    alloc4 = min(200.0, tleft() * 0.20)
    if alloc4 > 20:
        h_cont_bin = project_box_sum(h_bin, target)
        h_cp2, c5_cp2 = cutting_plane_solve(h_cont_bin, alloc4, start_time,
                                             max_iters=30000)
        if c5_cp2 < best_c5:
            best_c5 = c5_cp2
            best_h = h_cp2.copy()

    # Phase 5: Projected gradient on best (80s)
    alloc5 = min(80.0, tleft() * 0.10)
    if alloc5 > 15:
        h_pg, c5_pg = projected_gradient(best_h, alloc5, start_time,
                                          topk=3, lr_init=0.04, lr_end=0.0005)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 6: More cutting plane from current best (150s)
    alloc6 = min(150.0, tleft() * 0.15)
    if alloc6 > 20:
        h_cp3, c5_cp3 = cutting_plane_solve(best_h, alloc6, start_time,
                                             max_iters=30000)
        if c5_cp3 < best_c5:
            best_c5 = c5_cp3
            best_h = h_cp3.copy()

    # Phase 7: SLSQP polish (60s)
    alloc7 = min(60.0, tleft() * 0.10)
    if alloc7 > 10:
        h_ref, sc_ref = slsqp_polish(best_h, alloc7, start_time,
                                      n_alphas=(100, 500, 2000))
        if sc_ref < best_c5:
            best_c5 = sc_ref
            best_h = h_ref.copy()

    # Phase 8: Try perturbed starts with cutting plane
    for perturb_count in range(3):
        if tleft() < 60:
            break
        alloc8 = min(40.0, tleft() * 0.10)
        if alloc8 < 15:
            break
        h_pert = best_h + 0.01 * rng.randn(n)
        h_pert = project_box_sum(h_pert, target)
        h_pert_bin = project_binary(h_pert, n // 2)
        h_cp_p, c5_cp_p = cutting_plane_solve(h_pert_bin, alloc8 * 0.7, start_time,
                                               max_iters=10000)
        if c5_cp_p < best_c5:
            best_c5 = c5_cp_p
            best_h = h_cp_p.copy()
        if tleft() < 40:
            break
        h_pert2 = project_box_sum(h_pert, target)
        h_cp_p2, c5_cp_p2 = cutting_plane_solve(h_pert2, alloc8 * 0.3, start_time,
                                                 max_iters=5000)
        if c5_cp_p2 < best_c5:
            best_c5 = c5_cp_p2
            best_h = h_cp_p2.copy()

    # Phase 9: Final cutting plane burst
    if tleft() > 30:
        alloc9 = min(50.0, tleft() - 10)
        if alloc9 > 15:
            h_final_cp, c5_final_cp = cutting_plane_solve(best_h, alloc9, start_time,
                                                           max_iters=20000)
            if c5_final_cp < best_c5:
                best_c5 = c5_final_cp
                best_h = h_final_cp.copy()

    # Phase 10: Final projected gradient
    if tleft() > 15:
        h_pg_f, c5_pg_f = projected_gradient(best_h, max(5.0, tleft()-5), start_time,
                                              topk=3, lr_init=0.02, lr_end=0.0001,
                                              max_iters=100000)
        if c5_pg_f < best_c5:
            best_c5 = c5_pg_f
            best_h = h_pg_f.copy()

    # Final validation
    best_h = project_box_sum(best_h, target)
    final_c5 = compute_c5(best_h)

    # Also try binary version
    h_bin_final = project_binary(best_h, n // 2)
    bin_c5 = compute_c5(h_bin_final)

    if bin_c5 < final_c5:
        return h_bin_final, bin_c5, n
    else:
        return best_h, final_c5, n
