def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h.copy()

    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

                Gj[jm_r] -= h[j + shifts[jm_r]]

                g = Gj - Gli

                d_lo = max(-h[j], h[i] - 1.0)
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue

                a = d_lo
                b = d_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a + b)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = np.max(g[active])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = mid_d
                    else:
                        a = mid_d

                delta_star = 0.5 * (a + b)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial - 1e-14:
                    best_trial = approx
                    best_pi = int(i)
                    best_pj = int(j)
                    best_pd = delta_star

        if best_pi >= 0:
            h[best_pi] -= best_pd
            h[best_pj] += best_pd
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def game_descent(h_init, n, target, time_limit, max_steps=5000, eta=1.5, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(8):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = lr * 1.1
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.9
            if best_lr < 0.001:
                break

        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.7 * lam + 0.3 * lam_new

        if step % 300 == 0 and no_improve > 150:
            rng_esc = np.random.RandomState(step + 12345)
            h_pert = h + rng_esc.randn(n) * 0.08
            h_pert = project_feasible(h_pert, target)
            if compute_c5(h_pert) < best_c5:
                h = h_pert
                best_c5 = compute_c5(h)
                best_h = h.copy()
                no_improve = 0

    return best_h, best_c5


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_resolve_optimize(h_init, n, target, n_new, time_limit, seed=42):
    target_new = n_new / 2.0
    h_interp = interpolate_h(h_init, n_new, target_new)
    h_cd, c5_cd = pair_cd(h_interp, n_new, target_new,
                          time_limit=min(time_limit * 0.4, time_limit),
                          n_cand=min(35, n_new // 12), patience=200)
    if time_limit > 100:
        h_gt, c5_gt = game_descent(h_cd, n_new, target_new,
                                   time_limit=min(60, time_limit * 0.2),
                                   max_steps=5000, eta=2.0, m_top=10)
        if c5_gt < c5_cd:
            h_cd = h_gt
            c5_cd = c5_gt
    h_pol = lbfags_polish(h_cd, n_new, target_new, 5000, maxiter=1000)
    c5_pol = compute_c5(h_pol)
    if c5_pol < c5_cd:
        h_cd = h_pol
        c5_cd = c5_pol
    return h_cd, c5_cd


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    best_n = n

    np.random.seed(seed)

    # Phase 1: Quick pair CD at original resolution
    if tleft() > 120:
        p1_time = min(80, tleft() * 0.12)
        h1, c5_1 = pair_cd(best_h, n, target, time_limit=p1_time,
                           n_cand=40, patience=200, bisection_iters=14)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1
            best_n = n

    # Phase 2: Game-theoretic descent at original resolution
    if tleft() > 150:
        h2, c5_2 = game_descent(best_h, n, target, time_limit=min(60, tleft() * 0.1),
                                max_steps=4000, eta=2.0, m_top=12)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2
            best_n = n

    # Phase 3: L-BFGS-B polish at original resolution
    if tleft() > 150:
        for alpha in [500, 2000, 10000]:
            if tleft() < 50:
                break
            h3 = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h3)
            if sc < best_score:
                best_score = sc
                best_h = h3

    # Phase 4: Try n=600
    if tleft() > 300:
        n_try = 600
        t4 = min(250, tleft() * 0.3)
        h4, c5_4 = multi_resolve_optimize(best_h, best_n, best_n / 2.0, n_try, t4, seed=seed)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4
            best_n = n_try

    # Phase 5: Try n=800
    if tleft() > 250:
        n_try = 800
        t5 = min(200, tleft() * 0.25)
        if best_n == 600:
            h_src = best_h
            n_src = best_n
        else:
            h_src = best_h
            n_src = best_n
        target_src = n_src / 2.0
        h5, c5_5 = multi_resolve_optimize(h_src, n_src, target_src, n_try, t5, seed=seed + 1)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5
            best_n = n_try

    # Phase 6: Intensive pair CD at best resolution
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        pc_time = min(200, tleft() * 0.35)
        nc = 40 if n_cur <= 600 else 30
        h6, c5_6 = pair_cd(best_h, n_cur, target_cur, time_limit=pc_time,
                           n_cand=nc, patience=300, bisection_iters=15)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6
            best_n = n_cur

    # Phase 7: Game-theoretic refinement
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h7, c5_7 = game_descent(best_h, n_cur, target_cur,
                                time_limit=min(60, tleft() * 0.15),
                                max_steps=5000, eta=2.5, m_top=15)
        if c5_7 < best_score:
            best_score = c5_7
            best_h = h7

    # Phase 8: More pair CD
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        pc_time = min(60, tleft() * 0.2)
        nc = 45 if n_cur <= 600 else 35
        h8, c5_8 = pair_cd(best_h, n_cur, target_cur, time_limit=pc_time,
                           n_cand=nc, patience=150, bisection_iters=16)
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8

    # Phase 9: L-BFGS-B final polish with high alpha
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for alpha in [10000, 50000, 200000]:
            if tleft() < 30:
                break
            h9 = lbfags_polish(best_h, n_cur, target_cur, alpha, maxiter=800)
            sc = compute_c5(h9)
            if sc < best_score:
                best_score = sc
                best_h = h9

    # Phase 10: Perturbation-based escape and re-optimize
    rng = np.random.RandomState(seed + 99)
    if tleft() > 80:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 40:
                break
            n_cur = len(best_h)
            target_cur = n_cur / 2.0
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_pol = lbfags_polish(h_pert, n_cur, target_cur, 5000, maxiter=800)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
                if tleft() > 30:
                    h_cd2, c5_cd2 = pair_cd(best_h, n_cur, target_cur,
                                            time_limit=min(30, tleft() * 0.2),
                                            n_cand=30, patience=100)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score < best_score:
        best_score = final_score
    best_h = h_final

    return best_h, best_score, len(best_h)
