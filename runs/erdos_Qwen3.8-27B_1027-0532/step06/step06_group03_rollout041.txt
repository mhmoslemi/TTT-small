    """Compute gradient of sum of worst-k correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    top_idx = np.argsort(corr)[-k_worst:]

        # d corr_s / d h[j] = (1 - h[j-k]) - h[j+k] (with bounds)
        for j in range(n):

            term2 = 0.0 if (j + k < 0 or j + k >= n) else h[j + k]
            grad[j] += term1 - term2
    grad /= k_worst
    grad *= dx
    return grad

max_s (corr[s] + delta * d[s]) is a piecewise-linear function of delta (it's the max of linear functions). The optimal delta is where this function is minimized.

The function f(delta) = max_s (corr[s] + delta * d[s]) is convex in delta (max of linear functions). We can find its minimum by:

- Otherwise, binary search for where the slope changes sign

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad


def _con_fun(h):
    return np.sum(h) - _global_target


def _con_jac(h):
    return np.ones(_global_n)


def slsqp_analytic(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    best = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def topk_gradient(h, k_worst):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    top_idx = np.argsort(corr)[-k_worst:]
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in top_idx:
        k = int(s) - mid
        idx_plus = idxs + k
        idx_minus = idxs - k
        valid_plus = (idx_plus >= 0) & (idx_plus < n)
        valid_minus = (idx_minus >= 0) & (idx_minus < n)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        term1[valid_plus] = 1.0 - h[idx_plus[valid_plus]]
        term2[valid_minus] = h[idx_minus[valid_minus]]
        grad += term1 - term2
    grad /= k_worst
    grad *= dx
    return grad


def topk_momentum(h_start, n, target, time_limit, k_schedule, base_lr, mu, max_steps):
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    steps_since_improve = 0
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        frac = min(1.0, step / max(1, max_steps // 3))
        k_use = int(k_schedule[0] + frac * (k_schedule[1] - k_schedule[0]))
        k_use = max(2, k_use)
        grad = topk_gradient(h, k_use)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        moved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                steps_since_improve = 0
                moved = True
                break
            elif c5_try <= compute_c5(h) + 1e-14:
                h = h_try
                moved = True
                break
        if not moved:
            steps_since_improve += 1
            if steps_since_improve > 200:
                break
    return best_h, best_c5


def vectorized_pair_cd(h_start, n, target, time_limit, n_cand, bisection_iters, max_sweeps):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-3:]
        soft_a = 50.0
        w_top = np.exp(soft_a * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            idx_p = idxs - k
            idx_m = idxs + k
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)
            grad[vp] += wi * (1.0 - h[idx_p[vp]])
            grad[vm] -= wi * h[idx_m[vm]]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        g_high = np.zeros((n_cand, n_corr))
        g_low = np.zeros((n_cand, n_corr))
        shifts = np.arange(-(n - 1), n)
        for ii in range(n_cand):
            i = int(high_set[ii])
            kp = i + shifts
            km = i - shifts
            vp = (kp >= 0) & (kp < n)
            vm = (km >= 0) & (km < n)
            g_high[ii][vp] = 1.0 - h[kp[vp]]
            g_high[ii][vm] -= h[km[vm]]
        for ii in range(n_cand):
            j = int(low_set[ii])
            kp = j + shifts
            km = j - shifts
            vp = (kp >= 0) & (kp < n)
            vm = (km >= 0) & (km < n)
            g_low[ii][vp] = 1.0 - h[kp[vp]]
            g_low[ii][vm] -= h[km[vm]]
        d_matrix = g_low[None, :, :] - g_high[:, None, :]
        flat_d = d_matrix.reshape(n_cand * n_cand, n_corr)
        rough_delta = np.zeros(n_cand * n_cand)
        active = corr >= c_max - 1e-12
        flat_corr = corr
        obj_at_zero = np.max(flat_d * 0.0 + flat_corr[None, :])
        for p in range(n_cand * n_cand):
            ai = p // n_cand
            aj = p % n_cand
            hi_val = h[int(high_set[ai])]
            lo_val = h[int(low_set[aj])]
            delta_hi = min(hi_val, 1.0 - lo_val)
            if delta_hi < 1e-12:
                rough_delta[p] = 0.0
                continue
            g_active = flat_d[p]
            f_prime = np.max(g_active[active]) if active.any() else 0.0
            if f_prime > 1e-12:
                rough_delta[p] = min(c_max / f_prime, delta_hi)
            elif f_prime < -1e-12:
                rough_delta[p] = delta_hi
            else:
                rough_delta[p] = delta_hi * 0.5
        sorted_p = np.argsort(rough_delta)[::-1][:50]
        best_trial_c5 = c5_val
        best_p = -1
        best_delta_val = 0.0
        for p in sorted_p:
            ai = p // n_cand
            aj = p % n_cand
            i = int(high_set[ai])
            j = int(low_set[aj])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = flat_d[p]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                act = vals >= f_mid - 1e-14
                if act.any():
                    f_prime = np.max(g[act])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_p = p
                best_delta_val = delta_star
                best_i = i
                best_j = j
        if best_p >= 0:
            h[best_i] -= best_delta_val
            h[best_j] += best_delta_val
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation_search(h_init, n, target, max_tries, time_limit, seed_val):
    start_t = time.time()
    rng_local = np.random.RandomState(seed_val)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.02, 0.04, 0.06, 0.1, 0.15]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5
