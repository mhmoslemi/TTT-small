import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Bisection on alpha: find alpha such that sum(clip(h-alpha, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    result = np.clip(h - alpha, 0.0, 1.0)
    # Fix tiny numerical error
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_gradient(h, corr, beta):
    """Gradient of (1/beta)*log(sum exp(beta*C_k)) w.r.t. h"""
    n = len(h)
    max_corr = corr.max()
    shifted = beta * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_w = np.exp(shifted)
    p = exp_w / exp_w.sum()
    
    A = 1.0 - h
    first_term = np.convolve(A, p, 'full')[(n-1):(n-1)+n]
    second_term = np.convolve(h, p[::-1], 'full')[(n-1):(n-1)+n]
    return first_term - second_term

    """Gradient using only top-k correlations (unnormalized)"""
    n = len(h)
    topk_idx = np.argsort(corr)[-k:]
    p = np.zeros(len(corr))
    p[topk_idx] = 1.0 / k

    first_term = np.convolve(A, p, 'full')[(n-1):(n-1)+n]
    second_term = np.convolve(h, p[::-1], 'full')[(n-1):(n-1)+n]
    return first_term - second_term

def projected_lse_hdescent(h, n_iters, beta_init, beta_end, lr_init, lr_end, 
                            time_limit, start_time, target):
    """Projected gradient descent on log-sum-exp objective"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = float(it) / max(n_iters - 1, 1)
        beta = beta_init * (beta_end / beta_init) ** ratio
        lr = lr_init * (lr_end / lr_init) ** ratio
        
        grad = lse_gradient(h, corr, beta)
        
        # Backtracking line search
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        
        # Simple acceptance: accept if improved, or at high temperature accept sometimes
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            for trial in range(5):
                lr_t = lr * (0.5 ** (trial + 1))
                h_t = h - lr_t * grad
                h_t = project_box_sum(h_t, target)
                corr_t = compute_corr(h_t)
                c5_t = float(corr_t.max()) * dx
                if c5_t < float(corr.max()) * dx:
                    h = h_t
                    corr = corr_t
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h.copy()
                        stall = 0
                    break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    result = np.clip(h - alpha, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_gradient(h, corr, beta):
    n = len(h)
    max_c = corr.max()
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_w = np.exp(shifted)
    p = exp_w / exp_w.sum()
    A = 1.0 - h
    t1 = np.convolve(A, p, 'full')
    t2 = np.convolve(h, p[::-1], 'full')
    return t1[n-1:n-1+n] - t2[n-1:n-1+n]

def topk_subgrad(h, corr, k):

                best_cont_c5 = c7
                best_cont = h7.copy()

        if tleft() < 60:
            break
        # Topk from continuous
        alloc = min(50.0, tleft() * 0.10)
        if alloc > 10:
            n_it = int(min(200000, alloc * 3500))
            h8, c8 = proj_topk_descent(best_cont, n_it, 3, 0.03, 0.0003, target0, alloc, start)
            if c8 < best_cont_c5:
                best_cont_c5 = c8
                best_cont = h8.copy()

    # Phase 8: Try different resolutions
    for n_try in [n0 + 2, n0 - 2, n0 + 6, n0 + 10]:
        if tleft() < 60:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        tgt = float(n_try) / 2.0
        h_interp = interpolate_h(best_cont, n_try)
        h_interp = project_box_sum(h_interp, tgt)
        alloc = min(25.0, tleft() * 0.08)
        if alloc < 12:
            continue
        n_it = int(min(80000, alloc * 2500))
        h_try, c_try = proj_lse_descent(h_interp, n_it, 50.0, 500.0, 0.05, 0.001, tgt, alloc, start)
        if c_try < best_cont_c5:
            best_cont_c5 = c_try
            best_cont = h_try.copy()

    # Final: pick best
    h_final_cont = project_box_sum(best_cont, float(len(best_cont)) / 2.0)
    cont_score = compute_c5(h_final_cont)
    h_final_bin = project_box_sum(best_bin, float(len(best_bin)) / 2.0)
    bin_score = compute_c5(h_final_bin)

    if bin_score <= cont_score:
        return h_final_bin, bin_score, len(h_final_bin)
    else:
        return h_final_cont, cont_score, len(h_final_cont)
