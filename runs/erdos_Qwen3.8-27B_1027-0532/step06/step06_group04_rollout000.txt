            slopes = np.zeros(n_active)
            if np.any(pos):
                slopes[pos] = (1.0 - h[i + j[pos]]) * dx
            if np.any(neg):
                slopes[neg] = -h[i + j[neg]] * dx
            intercepts = corr_vals - slopes * h[i]
            h[i] = solve_1d_fast(slopes, intercepts, lam)
        
        h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -2.0, 2.0)
        h = project_feasible(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx
        pass_count += 1
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if pass_count % 10 == 0:
            new_active = np.argsort(corr)[-n_active:]
            if not np.array_equal(np.sort(new_active), np.sort(active_idx)):
                active_idx = new_active
                active_lags = active_idx - half_n
                corr_vals = corr[active_idx].copy()
        
        if stall > max_stall:
            # Add noise and restart
            h = best_h + 0.05 * rng.randn(n)
            h = project_feasible(h, target)
            stall = 0
            max_stall = max(50, max_stall // 2)
            active_idx = np.argsort(compute_c5(h) * np.ones(2*n-1))[-n_active:]  # This is wrong
            # Recompute
            corr = np.correlate(h, 1.0-h, mode='full')
            active_idx = np.argsort(corr)[-n_active:]
            active_lags = active_idx - half_n
            corr_vals = corr[active_idx].copy()
        
        # Gradually increase n_active
        if pass_count % 100 == 0 and n_active < n:
            n_active = min(n, n_active + 5)
            active_idx = np.argsort(corr)[-n_active:]
            active_lags = active_idx - half_n
            corr_vals = corr[active_idx].copy()
    
    return best_h, best_c5

def lp_polish(h, target, n_active, time_limit, start_time, d0=0.9, reg0=1e-4):
    n = len(h)
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    reg = reg0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = set(np.argsort(corr)[-n_active:].tolist())
    
    while time.time() - start_time < time_limit:
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for k_idx in sorted(active_set):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-11, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
