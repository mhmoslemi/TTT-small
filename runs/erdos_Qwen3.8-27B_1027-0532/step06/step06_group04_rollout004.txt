    # two (or more) functions are simultaneously maximal, or at a boundary.

    # Evaluate at x=0 and x=1
    best_val = max(np.max(slopes * 0 + intercepts), np.max(slopes * 1 + intercepts))
    best_x = 0.0 if best_val == np.max(intercepts) else 1.0

    n = len(slopes)
    if n == 1:
        if slopes[0] < 0: return 1.0
        if slopes[0] > 0: return 0.0
        return 0.0
    
    # All pairwise intersections
    js = np.tile(np.arange(n), n)
    ks = np.repeat(np.arange(n), n)
    ds = slopes[js] - slopes[ks]
    valid = np.abs(ds) > 1e-15
    x_candidates = np.empty(n)
    x_candidates[:] = np.nan
    x_candidates[valid] = (intercepts[ks[valid]] - intercepts[js[valid]]) / ds[valid]
    
    # Filter to (0,1)
    mask = valid & (x_candidates > 0) & (x_candidates < 1)
    if not np.any(mask):
        val_0 = np.max(intercepts)
        val_1 = np.max(slopes + intercepts)
        return 0.0 if val_0 <= val_1 else 1.0
    
    x_cand = x_candidates[mask]
    # Evaluate objective at each candidate
    vals = np.max(slopes[None, :] * x_cand[:, None] + intercepts[None, :], axis=1)
    best_idx = np.argmin(vals)
    best_x = x_cand[best_idx]
    
    # Compare with boundaries
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    if val_0 <= vals[best_idx]:
        best_x = 0.0
    elif val_1 <= vals[best_idx]:
        best_x = 1.0
    return best_x

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_vec(slopes, intercepts):
    n = len(slopes)
    if n == 0:
        return 0.5
    if n == 1:
        if slopes[0] < -1e-15:
            return 1.0
        elif slopes[0] > 1e-15:
            return 0.0
        else:
            return 0.5
    # All pairwise intersections
    js = np.repeat(np.arange(n), n)
    ks = np.tile(np.arange(n), n)
    ds = slopes[js] - slopes[ks]
    mask_valid = np.abs(ds) > 1e-15
    x_cand = np.zeros(n * n)
    x_cand[mask_valid] = (intercepts[ks[mask_valid]] - intercepts[js[mask_valid]]) / ds[mask_valid]
    mask = mask_valid & (x_cand > 1e-12) & (x_cand < 1.0 - 1e-12)
    if np.any(mask):
        xc = x_cand[mask]
        vals = np.max(slopes[None, :] * xc[:, None] + intercepts[None, :], axis=1)
        bi = np.argmin(vals)
        best_val = vals[bi]
        best_x = xc[bi]
    else:
        best_val = np.max(intercepts)
        best_x = 0.0
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    if val_0 < best_val:
        best_val = val_0
        best_x = 0.0
    if val_1 < best_val:
        best_x = 1.0
    return best_x

def coord_descent_exact(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    half_n = n - 1
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_vec(slopes, intercepts)
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5

def lp_full_lags(h_init, n, target, max_iter, time_limit, d0, reg0, scs_eps=1e-11):
    """LP/QP with ALL lags included."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    n_lags = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Add all lag constraints
        for k_idx in range(n_lags):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                # coeff[i] = g[i-k] * dx for valid i
                # g[i-k] is valid when 0 <= i-k < n, i.e., k <= i < k+n
                # lo = max(0, k), hi = min(n, k+n)
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Adapt damping
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.99, damping + 0.02)
        
        # Decay regularization
        if it % 30 == 0 and current_reg > 1e-10:
            current_reg = max(1e-10, current_reg * 0.75)
    
    return best_h, best_c5

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, n, target),
                           h_cur, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_vec(slopes, intercepts)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    return best_h, best_c5
