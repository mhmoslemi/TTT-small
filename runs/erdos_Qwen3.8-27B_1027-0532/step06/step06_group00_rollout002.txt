            h = np.clip(h + step, 0.0, 1.0)

                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def balanced_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=50, n_focus=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argpartition(corr, -n_focus)[-n_focus:]
        top_vals = corr[top_idx]
        mean_top = np.mean(top_vals)

        focus_grad = np.zeros(n)
        for s_idx in top_idx:
            w = max(0.0, corr[s_idx] - mean_top + top_vals.min())
            if w < 1e-10:
                w = 1.0
            kk = int(s_idx) - mid
            v1 = (idxs - kk >= 0) & (idxs - kk < n)
            v2 = (idxs + kk >= 0) & (idxs + kk < n)
            focus_grad[v1] += w * (1.0 - h[idxs[v1] - kk])
            focus_grad[v2] -= w * h[idxs[v2] + kk]
        focus_grad -= np.mean(focus_grad)

        sorted_idx = np.argsort(focus_grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def projected_gradient_bb(h_init, n, target, time_limit, max_iter=10000, top_k=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    x_prev = h.copy()
    f_prev = None
    s_prev = None
    ys_prev = None

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[top_idx]
        c_max = np.max(top_vals)

        alpha = 100.0
        weights = np.exp(np.clip(alpha * (top_vals - c_max), -500, 0))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        mid = n - 1
        for idx in top_idx:
            w = weights[list(top_idx).index(idx)]
            k_shift = int(idx) - mid
            if k_shift >= 0:
                for i in range(n - k_shift):
                    grad[i] += w * (1.0 - h[i + k_shift])
                    grad[i + k_shift] -= w * h[i]
            else:
                ks = -k_shift
                for i in range(ks, n):
                    grad[i] += w * (1.0 - h[i - ks])
                    grad[i - ks] -= w * h[i]

        grad /= alpha

        if f_prev is not None and s_prev is not None:
            ys = np.dot(s_prev, ys_prev)
            ss = np.dot(s_prev, s_prev)
            if ys > 1e-10:
                step = ss / ys
            else:
                step = 0.01
        else:
            step = 0.01

        step = min(step, 0.5)

        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                no_improve = 0
                f_prev = None
                s_prev = None
                continue

        s_new = h_new - h
        h = h_new
        f_curr = sc
        if f_prev is not None:
            s_prev = s_new
            ys_prev = (h_new - h_init) * 0 + (corr - np.correlate(h, 1.0 - h, mode='full')) * 0
            ys_prev = s_prev * (grad - np.zeros_like(grad))
            ys_prev = s_prev * grad
            f_prev = f_curr
        else:
            f_prev = f_curr
            s_prev = s_new
            ys_prev = s_prev * grad

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol, sc = slsqp_smooth(best_h, n0, target0,
                                 [50, 200, 1000, 5000], [10, 20, 50], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 200:
            h_cd, c5_cd = fast_pair_cd(h1, n1, target1,
                                       time_limit=min(350, tleft() * 0.55),
                                       max_sweeps=40000,
                                       n_cand_start=15, n_cand_end=70,
                                       max_no_improve=500, bisect_iters=16)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    if tleft() > 100:
        h_pol, sc = slsqp_smooth(best_h, n0, target0,
                                 [1000, 5000, 20000], [10, 20, 50], 1000)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 250:
            h_cd2, c5_cd2 = fast_pair_cd(h2, n2, target2,
                                         time_limit=min(350, tleft() * 0.5),
                                         max_sweeps=30000,
                                         n_cand_start=20, n_cand_end=80,
                                         max_no_improve=400, bisect_iters=16)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2

    if tleft() > 100:
        h_pol, sc = slsqp_smooth(best_h, n0, target0,
                                 [2000, 10000, 30000], [5, 10, 20], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_bcd, c5_bcd = balanced_cd(best_h, n0, target0,
                                    time_limit=min(200, tleft() * 0.6),
                                    max_sweeps=20000, n_cand=60, n_focus=5)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    if tleft() > 100:
        h_cd3, c5_cd3 = fast_pair_cd(best_h, n0, target0,
                                     time_limit=min(150, tleft() * 0.6),
                                     max_sweeps=20000,
                                     n_cand_start=40, n_cand_end=100,
                                     max_no_improve=300, bisect_iters=18)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_pol, sc = slsqp_smooth(best_h, n0, target0,
                                 [5000, 20000, 50000], [5, 10], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = slsqp_smooth(h_pert, n0, target0,
                                         [200, 2000, 10000], [10, 20], 500)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 80:
        h_cd_final, c5_final = fast_pair_cd(best_h, n0, target0,
                                            time_limit=min(80, tleft() * 0.7),
                                            max_sweeps=15000,
                                            n_cand_start=50, n_cand_end=120,
                                            max_no_improve=200, bisect_iters=20)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 40:
        h_final, sc_final = slsqp_smooth(best_h, n0, target0,
                                         [10000, 30000, 80000], [3, 5], 500)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
