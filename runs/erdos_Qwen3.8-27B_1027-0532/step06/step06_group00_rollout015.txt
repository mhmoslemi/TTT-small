import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_start=30, n_cand_end=70, n_shifts=3,
                       max_no_improve=500, bisect_iters=16):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-30:
            top_sum = 1.0
        weights = corr[top_shifts] / top_sum

        grad = np.zeros(n)
        for s_idx in range(len(top_shifts)):
            w = float(weights[s_idx])
            k = int(top_shifts[s_idx]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        n_h = len(high_set)
        n_l = len(low_set)

        h_i = h[high_set].reshape(-1, 1)
        h_j = h[low_set].reshape(1, -1)

        delta_lo = np.maximum(h_i - 1.0, -h_j)
        delta_hi = np.minimum(h_i, 1.0 - h_j)
        valid_mask = (delta_hi - delta_lo) > 1e-12

        if not valid_mask.any():
            continue

        g_i = change[high_set]
        g_j = change[low_set]
        g_3d = g_j[None, :, :] - g_i[:, None, :]

        d_lo = delta_lo.copy()
        d_hi = delta_hi.copy()

        for _ in range(bisect_iters):
            mid_d = 0.5 * (d_lo + d_hi)
            vals = corr[None, None, :] + mid_d[:, :, None] * g_3d
            f_mid = np.max(vals, axis=2)
            active = vals >= f_mid[:, :, None] - 1e-14
            f_prime = np.where(active, g_3d, -np.inf).max(axis=2)
            go_left = f_prime > 1e-12
            d_hi = np.where(go_left, mid_d, d_hi)
            d_lo = np.where(~go_left, mid_d, d_lo)

        delta_star = 0.5 * (d_lo + d_hi)
        final_vals = corr[None, None, :] + delta_star[:, :, None] * g_3d
        best_c5_2d = np.max(final_vals, axis=2) * dx

        best_c5_2d[~valid_mask] = np.inf
        best_c5_2d[delta_star <= 1e-14] = np.inf

        flat_idx = int(np.argmin(best_c5_2d))
        ii = flat_idx // n_l
        jj = flat_idx % n_l
        best_trial = float(best_c5_2d[ii, jj])

        if best_trial >= c5_val - 1e-14:
            continue

        i = int(high_set[ii])
        j = int(low_set[jj])
        delta = float(delta_star[ii, jj])

        if abs(delta) < 1e-14:
            continue

        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

        if h[i] != h[i] or np.any(h != h):
            h = best_h.copy()
            continue

        update_change_row(change, i, h, n, shifts)
        update_change_row(change, j, h, n, shifts)

    return best_h, best_c5

def single_point_cd(h_init, n, target, time_limit, max_sweeps=20000,
                    n_cand=40, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        cand = sorted_idx[:n_cand]

        best_trial_c5 = c5_val
        best_p = -1
        best_dir = 0
        best_delta = 0.0

        for p in cand:
            p = int(p)
            g = -change[p] * (1.0 if grad[p] > 0 else -1.0)
            direction = -1.0 if grad[p] > 0 else 1.0
            if direction < 0:
                d_lo = max(h[p] - 1.0, 0.0)
                d_hi = h[p]
            else:
                d_lo = 0.0
                d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-13:
                continue
            a_d, b_d = d_lo, d_hi
            for _ in range(14):
                m_d = 0.5 * (a_d + b_d)
                vals = corr + m_d * direction * change[p]
                f_m = np.max(vals)
                act = vals >= f_m - 1e-14
                fp = np.max(direction * change[p][act]) if act.any() else 0.0
                if fp > 1e-12:
                    b_d = m_d
                else:
                    a_d = m_d
            delta = 0.5 * (a_d + b_d)
            if abs(delta) < 1e-13:
                continue
            approx = np.max(corr + delta * direction * change[p]) * dx
            if approx < best_trial_c5:
                best_trial_c5 = approx
                best_p = p
                best_dir = direction
                best_delta = delta

        if best_p >= 0:
            h[best_p] += best_dir * best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_p, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def annealing_polish(h_init, n, target, time_limit, max_steps=20000,
                     base_temp=0.08, cool_rate=0.9995):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    temp = base_temp
    no_improve = 0
    n_pert_options = [1, 2, 3, 4, 5, 7, 10]

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        n_pert = int(rng.choice(n_pert_options))
        n_pert = min(n_pert, n // 2)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * temp
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            delta_e = sc - cur_c5
            if rng.rand() < np.exp(-delta_e / max(temp, 1e-10)):
                h = h_try
                cur_c5 = sc
            else:
                no_improve += 1
                if no_improve > 500:
                    h = best_h.copy()
                    cur_c5 = best_c5
                    no_improve = 0
        temp *= cool_rate
        temp = max(temp, base_temp * 0.05)

    return best_h, best_c5

            if c5_cd_hi < best_score:
                best_score = c5_cd_hi
                best_h = h_cd_hi
                if tleft() > 50:
                    h_pol_hi = slsqp_round(best_h, n_hi, target_hi,
                                           [2000, 10000], 800, 1e-14)
                    sc_hi = compute_c5(h_pol_hi)
                    if sc_hi < best_score:
                        best_score = sc_hi
                        best_h = h_pol_hi

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.05, 0.10]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * ns
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_ref = slsqp_round(h_pert, len(best_h), len(best_h) / 2.0,
                                [500, 2000, 10000], 500, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_fin, c5_fin = vectorized_pair_cd(best_h, len(best_h), len(best_h)/2.0,
                                              time_limit=min(50, tleft() * 0.6),
                                              max_sweeps=20000,
                                              n_cand_start=40, n_cand_end=80,
                                              n_shifts=4, max_no_improve=300,
                                              bisect_iters=16)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    if tleft() > 30:
        h_final = slsqp_round(best_h, len(best_h), len(best_h)/2.0,
                              [10000, 50000, 200000], 800, 1e-15)
        sc_fin = compute_c5(h_final)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    if final_score >= best_score:
        final_h = best_h
        final_score = best_score

    return final_h, final_score, len(final_h)
