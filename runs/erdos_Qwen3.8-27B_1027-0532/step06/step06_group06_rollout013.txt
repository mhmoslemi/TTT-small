    """Compute autocorrelation using FFT, returns C(k) for k=-(n-1) to (n-1)"""

    H = np.fft.rfft(h, N)

    # For our purpose: C(k) = S[k] for k=0..n-1, C(k) = S[N-|k|] for k=-(n-1)..-1
    return S[:n]  # We only need k=0..n-1 since C(k)=C(-k)

def smooth_obj(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = beta * (corr - cmax)
    shifted -= shifted.max()
    logsumexp = cmax + np.log(np.sum(np.exp(shifted))) / beta
    d = h.sum() - target
    return logsumexp + penalty * d * d

def smooth_grad(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = beta * (corr - cmax)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    t = np.arange(n)
    g = np.zeros(n)
    for j in range(n):

        ix2 = t - j + n - 1
        g[j] = np.dot(1.0 - h, w[ix1]) - np.dot(h, w[ix2])
    g *= dx
    if penalty > 0:

    return g

            t_a2 = min(25.0, (tlim - (time.time() - t0)) * 0.3)
            if t_a2 < 10:
                break
            h_opt = lbfgs_opt(h_int, beta, target, 1000.0, 200, t0 + t_a2, time.time())
            h_int = h_opt
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best = h_opt.copy()
    return best, best_c5

def binary_local_search(h, target, t0, tlim, max_stall=200):
    """Fast binary local search"""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy().astype(float)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h_cur.copy()
    stall = 0
    improved = True
    while improved and stall < max_stall and time.time() - t0 < tlim:
        improved = False
        cmax = float(corr.max())
        # Find worst lag
        k_worst = int(np.argmax(corr)) - (n - 1)
        # Find positions where h=1 that contribute to the worst lag
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try to swap a 1 with a 0 to reduce the worst correlation
        best_improve = 0
        best_i = -1
        best_j = -1
        # Only try swaps that affect the worst lag
        for i in ones:
            if i + k_worst < n and h_cur[i + k_worst] == 0:
                # Moving h[i] from 1 to 0 at position i
                # and putting 1 at position j = i + k_worst (or nearby)
                pass  # This gets complex, let me use a simpler approach
        
        # Simpler: just do a greedy 1-swap search on top correlations
        top_k = 5
        top_indices = np.argsort(corr)[-top_k:]
        for ki in top_indices:
            k = ki - (n - 1)
            # corr[ki] = sum_i h[i]*(1-h[i+k-shift])
            # We want to reduce this
            # Options: change a 1 at position i where (1-h[i+k-shift])=1 to 0
            #           change a 0 at position j where h[j+k-shift]=1 to 1
            affected_ones = [i for i in ones if 0 <= i + k - (n-1) < n and h_cur[i + k - (n-1)] == 0]
            affected_zeros = [j for j in zeros if 0 <= j + k - (n-1) < n and h_cur[j + k - (n-1)] == 1]
            
            for i in affected_ones[:10]:
                for j in affected_zeros[:10]:
                    # Evaluate the change
                    # Just do a quick check: would this swap reduce the max?
                    h_test = h_cur.copy()
                    h_test[i] = 0
                    h_test[j] = 1
                    corr_test = np.correlate(h_test, 1.0 - h_test, mode='full')
                    new_max = float(corr_test.max())
                    if new_max < cmax - best_improve:
                        best_improve = cmax - new_max
                        best_i = i
                        best_j = j
                        best_corr = corr_test.copy()
        
        if best_i >= 0:
            h_cur[best_i] = 0
            h_cur[best_j] = 1
            corr = best_corr
            new_c5 = float(corr.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
            stall = 0
            improved = True
        else:
            stall += 1
    
    return best_h, best_c5

Hmm, the binary local search above is O(n²) per iteration because of the inner loops. Let me make it more efficient.

Actually, let me step back and think about what will really make the difference. The gap is only 0.00008. I think the most effective approach is:

1. **Very aggressive continuous optimization** with L-BFGS-B at high β
2. **More grid sizes** explored
3. **Better time allocation**

The previous code's L-BFGS-B gradient computation is O(n²) per call (the loop over j). For n=606, that's about 367,000 operations per gradient call. With 200 iterations of L-BFGS-B, that's 73 million operations per L-BFGS-B call. At ~100M ops/sec, that's about 0.7 seconds per L-BFGS-B call.

In 1300 seconds, we can do about 1000 L-BFGS-B calls. The previous code does about 20-30 L-BFGS-B calls. So there's a lot of room for more optimization calls!

The key insight: **spend almost all the time on L-BFGS-B with high β**, using multiple grid sizes and multiple starting points.
