def cutting_plane_step(h, n, target, active_lags, max_steps_lp=1000):
    """Solve the LP relaxation and return the step direction."""
    # Variables: d[0], ..., d[n-1], t (total n+1 variables)
    # Minimize: t
    # Constraints:
    #   c_k(h) + grad_k . d <= t for each active k
    #   0 <= h[i] + d[i] <= 1 for each i
    #   sum(h) + sum(d) = target (i.e., sum(d) = 0 since sum(h) = target)
    
    c_vals = get_correlations(h)
    mid = n - 1
    
    # Objective: minimize t (coefficient 1 for t, 0 for d's)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # c_k(h) + grad_k . d <= t  =>  grad_k . d - t <= -c_k(h)
    A_rows = []
    b_rows = []
    for k in active_lags:
        g = get_grad_shift(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(-c_vals[k])
    
    # 0 <= h[i] + d[i] <= 1  =>  -h[i] <= d[i] <= 1 - h[i]
    # d[i] >= -h[i]  =>  -d[i] <= h[i]
    # d[i] <= 1 - h[i]  =>  d[i] <= 1 - h[i]
    I = np.eye(n)
    A_rows.append(-I)  # -d <= h, i.e., d >= -h
    b_rows.append(h)
    A_rows.append(I)   # d <= 1-h
    b_rows.append(1.0 - h)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')

    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    
    if use_line_search:
        # Try full step first
        h_full = project_feasible(h + d, target)
        c5_full = compute_c5(h_full)
        
        if c5_full < compute_c5(h):
            return h_full
        
        # Backtrack
        for frac in [0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = project_feasible(h + frac * d, target)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h):
                return h_try
    
    return None

def slsqp_softmax(h_start, n, target, alpha, maxiter, ftol=1e-16):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(h):
        n_local = len(h)
        dx_local = 2.0 / n_local
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx_local
        max_C = np.max(C)
        shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
        return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(h):
        # Numerical gradient (expensive but accurate)
        eps = 1e-7
        grad = np.zeros(n)
        f0 = obj(h)
        for i in range(n):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = project_feasible(h_pert, target)
            grad[i] = (obj(h_pert) - f0) / eps
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}]
    
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h_start, compute_c5(h_start)

def steepest_descent_max(h_start, n, target, time_limit, max_steps, n_active=5):
    """Steepest descent on the max function with line search."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 500:
            break
        
        corr = get_correlations(h)
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = np.max(corr)
        
        # Get gradient of the max (or weighted average of top lags)
        top_idx = np.argsort(corr)[-n_active:]
        weights = np.zeros(len(corr))
        for s in top_idx:
            weights[s] = 1.0 / n_active
        
        grad = np.zeros(n)
        for s in top_idx:
            k = int(s) - mid
            grad += weights[s] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        
        # Line search
        best_lr = None
        for lr in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                best_lr = lr
                h = h_try
                break
        
        if best_lr is None:
            no_improve += 10
    
    return best_h, best_c5

    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0

        if time.time() - start > time_limit:

        if no_improve > 200:

        corr = get_correlations(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Focus on the single worst lag
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        
        # Find best pair: i with highest grad (decrease h[i]), j with lowest grad (increase h[j])
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # For each pair, compute the direction of change in correlations
        # dcorr = grad_k for the specific lag k*
        # But we need to check ALL lags, not just k*
        # Use the full correlation change:
        # If we decrease h[i] by δ and increase h[j] by δ:
        # delta_corr[s] = δ * (d/dh[i] c_s - d/dh[j] c_s) for each s
        
        # Precompute the full gradient matrix for the candidate indices
        # This is expensive, so let's use a simpler approach:
        # For the active lag k*, the change is δ * (grad[i] - grad[j])
        # For other lags, we approximate
        
        best_trial = c5_val
        best_move = None
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                # Direction: decrease h[i], increase h[j]
                # Change in correlation at lag k: d_k = grad_k[i] - grad_k[j]
                # where grad_k is the gradient of correlation at lag k
                # For lag k*: d_k* = grad[i] - grad[j] < 0 (since grad[i] > grad[j])
                # This reduces c_{k*}
                # But it might increase other lags
                
                # Quick check: just use the k* gradient
                d_kstar = grad[i] - grad[j]
                if d_kstar >= -1e-10:
                    continue
                
                # Binary search on δ
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                
                # Use bisection to find optimal δ that minimizes max(corr + δ*d)
                # where d = full correlation change vector
                # But computing d for all lags is expensive
                # Approximation: only check a few key lags
                
                # Simple: try a few δ values
                best_delta = None
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    delta = frac * b_d
                    if delta < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_trial:
                        best_trial = c5_try
                        best_move = (i, j, delta)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def simulated_annealing(h_start, n, target, time_limit, max_steps, T_initial=0.05, T_final=0.001):

        if c5_try < best_c5 - 1e-13:

            best_h = h_try.copy()

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    def _obj(h):
        n_l = len(h)
        dx_l = 2.0 / n_l
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx_l
        max_C = np.max(C)
        shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
        return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)

    def _cons_fun(h):
        return np.sum(h) - target

    def _cons_jac(h):
        return np.ones(len(h))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h_start, compute_c5(h_start)

def multi_start_explore(h_init, n, target, time_limit, n_starts, rng):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for trial in range(n_starts):
        if time.time() - (start_time_global) > time_limit:
            break
        scale = rng.choice([0.02, 0.04, 0.06, 0.08, 0.12])
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_new = active_set_lp_step(h_pert, n, target, n_active=10)
        for _ in range(50):
            if h_new is None:
                break
            h_pert = h_new
            h_new = active_set_lp_step(h_pert, n, target, n_active=10)
        h_final = h_pert
        c5_try = compute_c5(h_final)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_final.copy()
    return best_h, best_c5

start_time_global = 0.0

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global start_time_global
    start_time_global = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time_global)

    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Rapid LP-based optimization
    if tleft() > 200:
        h_cur = best_h.copy()
        c5_cur = best_score
        stall = 0
        for step in range(1000):
            if tleft() < 200:
                break
            if stall > 30:
                break
            n_act = min(10 + step // 50, 30)
            h_new = active_set_lp_step(h_cur, n, target, n_active=n_act)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    h_cur = h_new
                    stall = 0
                    if c5_new < best_score:
                        best_score = c5_new
                        best_h = h_cur.copy()
                else:
                    stall += 1
            else:
                stall += 1

    # Phase 2: Steepest descent with normalized gradient
    if tleft() > 150:
        h_sd, sc = steepest_descent_max(best_h, n, target,
                                        time_limit=min(100, tleft() * 0.2),
                                        max_steps=30000, n_active=5)
        if sc < best_score:
            best_score = sc
            best_h = h_sd

    # Phase 3: More LP steps with adaptive active set
    if tleft() > 150:
        h_cur = best_h.copy()
        c5_cur = best_score
        stall = 0
        for step in range(500):
            if tleft() < 50:
                break
            if stall > 20:
                break
            h_new = active_set_lp_step(h_cur, n, target, n_active=20)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    h_cur = h_new
                    stall = 0
                    if c5_new < best_score:
                        best_score = c5_new
                        best_h = h_cur.copy()
                else:
                    stall += 1
            else:
                stall += 1

    # Phase 4: Pairwise coordinate descent
    if tleft() > 120:
        h_cd, sc = fast_pair_cd(best_h, n, target,
                                 time_limit=min(100, tleft() * 0.2),
                                 max_sweeps=10000, n_cand=10)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 5: SLSQP with increasing temperature
    if tleft() > 100:
        for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
            if tleft() < 30:
                break
            h_slq, sc = slsqp_softmax_opt(best_h, n, target, alpha, maxiter=800, ftol=1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_slq

    # Phase 6: LP + CD combined
    if tleft() > 100:
        h_cur = best_h.copy()
        c5_cur = best_score
        for phase in range(5):
            if tleft() < 30:
                break
            h_lp, _ = (h_cur, 0)
            for _ in range(50):
                if tleft() < 30:
                    break
                h_new = active_set_lp_step(h_lp, n, target, n_active=15)
                if h_new is not None:
                    if compute_c5(h_new) < compute_c5(h_lp) - 1e-13:
                        h_lp = h_new
                    else:
                        break
                else:
                    break
            h_cd2, sc2 = fast_pair_cd(h_lp, n, target,
                                       time_limit=min(30, tleft() * 0.15),
                                       max_sweeps=2000, n_cand=10)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_cd2
                h_cur = h_cd2

    # Phase 7: Multi-resolution exploration
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hr = h_interp
            c5_hr = compute_c5(h_hr)
            for _ in range(100):
                if tleft() < 50:
                    break
                h_new = active_set_lp_step(h_hr, n_try, target_try, n_active=15)
                if h_new is not None and compute_c5(h_new) < c5_hr - 1e-13:
                    h_hr = h_new
                    c5_hr = compute_c5(h_new)
                else:
                    break
            h_hr, sc_hr = slsqp_softmax_opt(h_hr, n_try, target_try, 5000, maxiter=500, ftol=1e-15)
            if sc_hr < best_score:
                best_score = sc_hr
                best_h = h_hr

    # Phase 8: Multi-start perturbation
    if tleft() > 100:
        for trial in range(5):
            if tleft() < 40:
                break
            scale = [0.01, 0.03, 0.05, 0.08, 0.12][trial]
            h_pert = best_h + rng.randn(len(best_h)) * scale
            n_p = len(best_h)
            t_p = n_p / 2.0
            h_pert = project_feasible(h_pert, t_p)
            for _ in range(30):
                h_new = active_set_lp_step(h_pert, n_p, t_p, n_active=15)
                if h_new is not None and compute_c5(h_new) < compute_c5(h_pert) - 1e-13:
                    h_pert = h_new
                else:
                    break
            c5_try = compute_c5(h_pert)
            if c5_try < best_score - 1e-13:
                best_score = c5_try
                best_h = h_pert

    # Phase 9: Final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for alpha in [500000, 2000000]:
            if tleft() < 20:
                break
            h_fin, sc = slsqp_softmax_opt(best_h, n_use, target_use, alpha, maxiter=500, ftol=1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_fin

    # Final LP polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cur = best_h.copy()
        for _ in range(50):
            if tleft() < 10:
                break
            h_new = active_set_lp_step(h_cur, n_use, target_use, n_active=25)
            if h_new is not None and compute_c5(h_new) < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
            else:
                break
        if compute_c5(h_cur) < best_score:
            best_score = compute_c5(h_cur)
            best_h = h_cur

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
