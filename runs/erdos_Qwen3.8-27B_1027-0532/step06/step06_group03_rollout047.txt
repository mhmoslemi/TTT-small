def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    i_lo = max(0, shift)
    i_hi = min(n, n + shift)
    if i_hi <= i_lo:
        return g
    i = np.arange(i_lo, i_hi)
    g[i] += (1.0 - h[i - shift])
    g[i - shift] -= h[i]
    return g

    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.minimum(alpha * (corr - c_max), 0.0)

            continue
        shift = s - mid
        grad += (w[s] / w_sum) * grad_shift(h, shift)
    grad *= dx
    return obj, grad

def pair_direction(h, i, j, mid):
    """Compute g[s] = ∂corr[s]/∂h[j] - ∂corr[s]/∂h[i] for all s."""
    n = len(h)
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    
    # ∂corr[s]/∂h[i]:
    # corr[s] = ∑_i' h[i'] * (1 - h[i' - shift]) where shift = s - mid
    # d/dh[i]: contributes when i' = i (gives 1 - h[i - shift]) or i' - shift = i (gives -h[i'])
    # So ∂corr[s]/∂h[i] = (1-h[i-shift])*[0≤i-shift<n] - h[i+shift]*[0≤i+shift<n]

        di = 0.0

        dj = 0.0
        if 0 <= j - shift < n:
            dj += (1.0 - h[j - shift])
        if 0 <= j + shift < n:
            dj -= h[j + shift]
        g[s] = dj - di
    return g

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    return h


def grad_shift_vec(h, shift):
