        if s_hi >= s_lo:

            k_vals = j - s_vals

            k_vals2 = j + s_vals2

def constrained_subgradient(h_start, n, target, time_limit, max_steps, alpha_smooth, lr_init, lr_decay):

def project_feasible(h, target_sum):

    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    current_max = np.max(corr)
    best_half = half.copy()
    best_max = current_max
    start_t = time.time()
    T = 0.5
    cooling = 0.9995
    rng = np.random.RandomState(12345)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        T *= cooling
        i = rng.randint(n2)
        half[i] = 1.0 - half[i]
        h_full = from_symmetric_half(half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        new_max = np.max(corr)
        delta = new_max - current_max

        if delta < 0 or rng.rand() < np.exp(-max(delta, -100.0) / max(T, 1e-10)):
            current_max = new_max
            if current_max < best_max - 1e-14:
                best_max = current_max
                best_half = half.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            half[i] = 1.0 - half[i]
            no_improve += 1

        if no_improve > 5000:
            break

    h_best = from_symmetric_half(best_half)
    return h_best, best_max


def pair_cd_symmetric(half_start, time_limit, max_sweeps=5000, top_k=3):
    n2 = len(half_start)
    n = 2 * n2
    n_corr = 2 * n - 1
    h_full = from_symmetric_half(half_start)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    current_max = float(np.max(corr))
    best_half = half_start.copy()
    best_max = current_max
    start_t = time.time()
    no_improve = 0
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        alpha_soft = 50.0
        w = np.exp(alpha_soft * (corr[top_idx] - c_max))
        w = w / w.sum()

        grad_full = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            k = int(s_idx) - (n - 1)
            v1 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
            v2 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            grad_full[v1] += wi * (1.0 - h_full[np.arange(n)[v1] - k])
            grad_full[v2] -= wi * h_full[np.arange(n)[v2] + k]

        grad_half = 0.5 * (grad_full[:n2] - grad_full[n - 1 :: -1][:n2])
        step = 0.5 / (1.0 + 0.01 * sweep)

        for trial in range(5):
            h_try = np.clip(half_start - step / (trial + 1) * grad_half, 0.0, 1.0)
            h_try_full = from_symmetric_half(h_try)
            c5_try = float(np.max(np.correlate(h_try_full, 1.0 - h_try_full, mode='full'))) * dx
            if c5_try < best_max:
                best_max = c5_try
                best_half = h_try.copy()
                half_start = h_try.copy()
                break

        corr = np.correlate(from_symmetric_half(half_start), 1.0 - from_symmetric_half(half_start), mode='full')
        if np.max(corr) * dx < best_max * (1 - 1e-12):
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    h_best = from_symmetric_half(best_half)
    return h_best, best_max


def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    h_sym = symmetric_project(best_h)
    best_score_sym = compute_c5(h_sym)
    if best_score_sym < best_score:
        best_score = best_score_sym
        best_h = h_sym

    half = best_h[:n // 2].copy()

    # Phase 1: Binary symmetric local search
    if tleft() > 100:
        h_bin, c5_bin = binary_symmetric_search_fast(half, time_limit=min(60, tleft() * 0.3))
        c5_bin_actual = compute_c5(h_bin)
        if c5_bin_actual < best_score:
            best_score = c5_bin_actual
            best_h = h_bin
            half = best_h[:n // 2].copy()

    # Phase 2: SLSQP on symmetric half
    if tleft() > 120:
        half_opt = slsqp_symmetric(half, [50, 200, 1000, 5000], 2000, 1e-15,
                                   min(80, tleft() * 0.4))
        h_full = from_symmetric_half(half_opt)
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = h_full
            half = h_full[:n // 2].copy()

    # Phase 3: Binary search again after continuous refinement
    if tleft() > 100:
        h_bin, c5_bin = binary_symmetric_search_fast(half, time_limit=min(60, tleft() * 0.3))
        c5_bin_actual = compute_c5(h_bin)
        if c5_bin_actual < best_score:
            best_score = c5_bin_actual
            best_h = h_bin
            half = best_h[:n // 2].copy()

    # Phase 4: SLSQP full (no symmetry) for comparison
    if tleft() > 150:
        h_pol = slsqp_full(best_h, n, target, [100, 500, 2000, 10000],
                           1500, 1e-16, min(100, tleft() * 0.4))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Pair CD on symmetric half
    if tleft() > 100:
        half_cd = best_h[:n // 2].copy()
        h_cd, c5_cd = pair_cd_symmetric(half_cd, time_limit=min(80, tleft() * 0.3),
                                        max_sweeps=8000, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            half = h_cd[:n // 2].copy()

    # Phase 6: Simulated annealing on binary symmetric
    if tleft() > 120:
        half_anneal = best_h[:n // 2].copy()
        h_anneal, c5_anneal = anneal_symmetric_binary(half_anneal,
                                                       time_limit=min(60, tleft() * 0.3),
                                                       max_iters=200000)
        c5_anneal_actual = compute_c5(h_anneal)
        if c5_anneal_actual < best_score:
            best_score = c5_anneal_actual
            best_h = h_anneal
            half = best_h[:n // 2].copy()

    # Phase 7: Final SLSQP polish (symmetric)
    if tleft() > 80:
        half_final = best_h[:n // 2].copy()
        half_opt = slsqp_symmetric(half_final, [1000, 5000, 20000], 2000, 1e-16,
                                   min(50, tleft() * 0.4))
        h_full = from_symmetric_half(half_opt)
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = h_full

    # Phase 8: Binary search on final result
    if tleft() > 60:
        h_bin, c5_bin = binary_symmetric_search_fast(best_h[:n // 2],
                                                      time_limit=min(30, tleft() * 0.4))
        c5_bin_actual = compute_c5(h_bin)
        if c5_bin_actual < best_score:
            best_score = c5_bin_actual
            best_h = h_bin

    # Phase 9: Try higher resolution
    if tleft() > 150:
        for n_try in [n + 50, n + 100]:
            if tleft() < 80:
                break
            h_interp = interpolate_to(best_h, n_try)
            h_interp = symmetric_project(h_interp)
            half_interp = h_interp[:n_try // 2].copy()
            half_opt = slsqp_symmetric(half_interp, [50, 200, 1000], 1500, 1e-15,
                                       min(50, tleft() * 0.3))
            h_full = from_symmetric_half(half_opt)
            sc = compute_c5(h_full)
            if sc < best_score:
                best_score = sc
                best_h = h_full
                n = n_try
                target = n / 2.0

    # Phase 10: Final binary + SLSQP at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        half_use = best_h[:n_use // 2].copy()
        h_bin, c5_bin = binary_symmetric_search_fast(half_use,
                                                      time_limit=min(25, tleft() * 0.3))
        c5_bin_actual = compute_c5(h_bin)
        if c5_bin_actual < best_score:
            best_score = c5_bin_actual
            best_h = h_bin

    if tleft() > 30:
        n_use = len(best_h)
        half_use = best_h[:n_use // 2].copy()
        half_opt = slsqp_symmetric(half_use, [5000, 20000], 1000, 1e-16,
                                   min(20, tleft() * 0.5))
        h_full = from_symmetric_half(half_opt)
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = h_full

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
