    try:
        res = minimize(_slsqp_fun, h_cur, jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_box_sum(res.x, target)
        sc = compute_c5(h_new)
        if sc < curr:
            return h_new, sc
    except Exception:
        pass
    return h_cur, curr


def adam_optimize(h0, target, alphas, time_limit, start_time, lr=0.003,
                  beta1=0.9, beta2=0.999, eps=1e-8, patience=300):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        while True:
            if time.time() - start_time > time_limit:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-11:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > patience:
                break
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr / (1.0 + 0.1 * np.log1p(t))
            h_new = h - step * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
    return best_h, best_actual


def subgradient_polish(h0, target, time_limit, start_time, n_iter=20000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for t in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - 1e-10)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        weights = np.ones(len(active)) / len(active)
        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        for idx in range(len(active)):
            w = weights[idx]
            s = int(active[idx]) - (n - 1)
            if s >= 0:
                grad[:n - s] += w * one_minus_h[s:]
                grad[s:] -= w * h[:n - s]
            else:
                p = -s
                grad[p:] += w * one_minus_h[:n - p]
                grad[:n - p] -= w * h[p:]
        eta = 2.0 / (np.sqrt(t + 1) * np.sqrt(n))
        h_new = h - eta * grad
        h = project_box_sum(h_new, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def binary_local_search(h, target_sum, time_limit, start_time, max_iters=30000, rng=None):
    n = len(h)
    h = (h >= 0.5).astype(float)
    ones_count = int(h.sum())
    target_count = int(round(target_sum))
    if ones_count > target_count:
        ones = np.where(h == 1)[0]
        for i in range(ones_count - target_count):
            h[ones[i]] = 0
    elif ones_count < target_count:
        zeros = np.where(h == 0)[0]
        for i in range(target_count - ones_count):
            h[zeros[i]] = 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 800
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ci = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        cj = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
        if rng is not None:
            if len(ones) > 20:
                ci = ci + list(rng.choice(ones, min(8, len(ones)), replace=False))
            if len(zeros) > 20:
                cj = cj + list(rng.choice(zeros, min(8, len(zeros)), replace=False))
        if len(ci) == 0 or len(cj) == 0:
            stall += 1
            continue
        improved = False
        for i in ci[:15]:
            for j in cj[:15]:
                h[i] = 0.0
                h[j] = 1.0
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                nv = float(corr_new.max())
                if nv < max_val:
                    max_val = nv
                    corr = corr_new
                    stall = 0
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            stall += 1
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    return best_h, best_c5


def generate_periodic_h(n, period, duty, target):
    h = np.zeros(n)
    filled = int(period * duty)
    for start in range(0, n, period):
        end = min(start + filled, n)
        h[start:end] = 1.0
    h = project_box_sum(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_alpha, _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    # Phase 1: Adam with moderate alpha (establish good basin)
    if tleft() > 300:
        t1 = min(250, tleft() - 300)
        alphas_p1 = [30, 60, 150, 400, 1000, 2500, 6000]
        h_a, s_a = adam_optimize(best_h, target, alphas_p1, t1, start_time,
                                 lr=0.003, patience=250)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 2: SLSQP polish
    if tleft() > 250:
        t2 = min(200, tleft() - 250)
        alphas_p2 = [3000, 8000, 20000]
        h_s, s_s = slsqp_optimize(best_h, alphas_p2[0], 2000, 1e-15, t2, start_time)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()
        if tleft() > 150:
            h_s2, s_s2 = slsqp_optimize(best_h, alphas_p2[1], 3000, 1e-15,
                                        min(120, tleft() - 150), start_time)
            if s_s2 < best_score:
                best_score = s_s2
                best_h = h_s2.copy()

    # Phase 3: Multi-restart with diverse initializations
    restart_configs = [
        ('perturb_small', 0.003),
        ('perturb_med', 0.008),
        ('perturb_large', 0.015),
        ('shift', None),
        ('periodic', None),
    ]
    n_restarts = min(len(restart_configs), max(1, int(tleft() // 150)))
    for ri in range(n_restarts):
        if tleft() < 100:
            break
        name, scale = restart_configs[ri]
        if name == 'perturb_small' or name == 'perturb_med' or name == 'perturb_large':
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_box_sum(h_pert, target)
        elif name == 'shift':
            s = rng.randint(1, n // 3)
            h_pert = np.roll(best_h, s)
            h_pert = project_box_sum(h_pert, target)
        elif name == 'periodic':
            period = rng.choice([7, 11, 13, 17, 19, 23, 29, 31])
            duty = rng.uniform(0.35, 0.65)
            h_pert = generate_periodic_h(n, period, duty, target)
        else:
            h_pert = project_box_sum(best_h + rng.randn(n) * 0.01, target)

        rt_adam = min(120, (tleft() - 80) / 2)
        rt_slsqp = min(80, (tleft() - 40) / 2)

        alphas_r = [200, 1000, 4000, 15000, 50000]
        h_r, s_r = adam_optimize(h_pert, target, alphas_r, rt_adam, start_time,
                                 lr=0.002, patience=200)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r.copy()

        if tleft() > 50:
            h_sr, s_sr = slsqp_optimize(h_r, 50000, 3000, 1e-15, rt_slsqp, start_time)
            if s_sr < best_score:
                best_score = s_sr
                best_h = h_sr.copy()

    # Phase 4: Subgradient polish on best solution
    if tleft() > 80:
        t_sub = min(100, tleft() - 50)
        h_sub, s_sub = subgradient_polish(best_h, target, t_sub, start_time, n_iter=30000)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()

    # Phase 5: Final SLSQP with very high alpha
    if tleft() > 40:
        t_fin = min(120, tleft() - 20)
        h_f, s_f = slsqp_optimize(best_h, 200000, 5000, 1e-16, t_fin, start_time)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Phase 6: Quick Adam with very high alpha
    if tleft() > 20:
        t_a6 = max(10.0, tleft() - 10)
        h_a6, s_a6 = adam_optimize(best_h, target,
                                   [100000, 300000, 500000], t_a6, start_time,
                                   lr=0.001, patience=100)
        if s_a6 < best_score:
            best_score = s_a6
            best_h = h_a6.copy()

    # Final feasibility
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
