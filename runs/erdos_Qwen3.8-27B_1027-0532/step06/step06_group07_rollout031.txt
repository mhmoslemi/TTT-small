```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_gradients(h, active_shifts, n):
    """Compute gradients ∂c_k/∂h for all k in active_shifts."""
    idxs = np.arange(n)
    grads = []
    for k in active_shifts:
        g = np.zeros(n)
        v_p = (idxs - k >= 0) & (idxs - k < n)  # j-k valid
        v_m = (idxs + k >= 0) & (idxs + k < n)  # j+k valid
        g[v_p] += (1.0 - h[idxs[v_p] - k])
        g[v_m] -= h[idxs[v_m] + k]
        grads.append(g)
    return np.array(grads)  # shape (|S|, n)

def active_set_lp_step(h, corr, n, active_frac=0.05, max_active=50, step_scale=1.0):
    """Solve the active-set LP to find a descent direction."""
    corr_max = np.max(corr)
    threshold = corr_max - active_frac * (corr_max - np.min(corr))
    active_idx = np.where(corr >= threshold)[0]
    if len(active_idx) > max_active:
        # Take the top max_active
        top_idx = np.argsort(corr)[-max_active:]
        active_idx = top_idx
    if len(active_idx) == 0:
        return None, None
    
    # Convert to shifts: corr[m] corresponds to shift k = m - (n-1)
    mid = n - 1
    active_shifts = active_idx - mid
    
    # Compute gradients
    idxs = np.arange(n)
    G = np.zeros((len(active_shifts), n))
    for ii, k in enumerate(active_shifts):
        v_p = (idxs - k >= 0) & (idxs - k < n)
        v_m = (idxs + k >= 0) & (idxs + k < n)
        G[ii, v_p] += (1.0 - h[idxs[v_p] - k])
        G[ii, v_m] -= h[idxs[v_m] + k]
    
    # LP: minimize t
    # subject to: G @ d <= t - c[active_idx]  →  G @ d - t <= -c[active_idx]
    #             sum(d) = 0
    #             -h[i] <= d[i] <= 1-h[i]
    
    # Decision variables: [d_0, ..., d_{n-1}, t]  (length n+1)
    # c (objective): minimize t → c = [0, 0, ..., 0, 1]
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G @ d - t <= -c[active_idx]
    # For each row i of G: G[i,:] @ d - t <= -c[active_idx[i]]
    A_ub = np.zeros((len(active_shifts), n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -corr[active_idx]
    
    # Equality constraints: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] <= d[i] <= 1-h[i], t unbounded (or t >= -large)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((None, None))
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    d = res.x[:n] * step_scale
    t_opt = res.x[-1]
    
    return d, t_opt

def frank_wolfe_step(h, n, target, active_shifts, corr):
    """Frank-Wolfe (conditional gradient) step."""
    idxs = np.arange(n)
    # Compute subgradient: average of gradients over active set
    g = np.zeros(n)
    for k in active_shifts:
        v_p = (idxs - k >= 0) & (idxs - k < n)
        v_m = (idxs + k >= 0) & (idxs + k < n)
        g[v_p] += (1.0 - h[idxs[v_p] - k])
        g[v_m] -= h[idxs[v_m] + k]
    g /= len(active_shifts)
    
    # Linear oracle: minimize g·x subject to 0≤x≤1, sum(x)=target
    # Optimal: set x[i]=1 for the 'target' indices with smallest g[i]
    # (target = n/2, so we pick n/2 indices)
    n_one = int(round(target))
    sorted_idx = np.argsort(g)
    x_oracle = np.zeros(n)
    x_oracle[sorted_idx[:n_one]] = 1.0
    
    return x_oracle
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_budget_end = 0.0

def get_tleft():
    return _budget_end - time.time()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, target, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-15, time_check=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    for alpha in alphas:
        if time_check is not None and time_check() < 10:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_sc

def active_set_lp_direction(h, corr, n, n_active=25, step_bound=0.5):
    mid = n - 1
    corr_max = np.max(corr)
    corr_range = corr_max - np.min(corr)
    if corr_range < 1e-10:
        return None
    threshold = corr_max - 0.02 * corr_range
    active_idx = np.where(corr >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(corr)])
    if len(active_idx) > n_active:
        active_idx = np.argsort(corr)[-n_active:]
    active_shifts = active_idx - mid
    idxs = np.arange(n)
    G = np.zeros((len(active_shifts), n))
    for ii, k in enumerate(active_shifts):
        v_p = (idxs - k >= 0) & (idxs - k < n)
        v_m = (idxs + k >= 0) & (idxs + k < n)
        G[ii, v_p] += (1.0 - h[idxs[v_p] - k])
        G[ii, v_m] -= h[idxs[v_m] + k]
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0
    A_ub = np.zeros((len(active_shifts), n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -corr[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = np.maximum(-h, -step_bound)
    ub = np.minimum(1.0 - h, step_bound)
    bounds = [(lb[i], ub[i]) for i in range(n)]
    bounds.append((None, None))
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    d = res.x[:n]
    t_opt = res.x[-1]
    if t_opt >= corr_max - 1e-10:
        return None
    return d

def frank_wolfe_direction(h, corr, n, target, n_active=20):
    mid = n - 1
    corr_max = np.max(corr)
    corr_range = corr_max - np.min(corr)
    if corr_range < 1e-10:
        return None
    threshold = corr_max - 0.03 * corr_range
    active_idx = np.where(corr >= threshold)[0]
    if len(active_idx) == 0:

    if len(active_idx) > n_active:
        active_idx = np.argsort(corr)[-n_active:]
    active_shifts = active_idx - mid
    idxs = np.arange(n)
    g = np.zeros(n)
    for k in active_shifts:
        v_p = (idxs - k >= 0) & (idxs - k < n)
        v_m = (idxs + k >= 0) & (idxs + k < n)
        g[v_p] += (1.0 - h[idxs[v_p] - k])
        g[v_m] -= h[idxs[v_m] + k]
    g /= len(active_shifts)
    n_one = int(round(target))
    sorted_idx = np.argsort(g)
    x_oracle = np.zeros(n)
    x_oracle[sorted_idx[:n_one]] = 1.0
    return x_oracle

def pair_cd_step(h, corr, n, target, n_cand=30, n_bisect=12):
    dx = 2.0 / n
    corr_max = np.max(corr) * dx
    mid = n - 1
    k_star = int(np.argmax(corr)) - mid
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad = np.zeros(n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    sorted_idx = np.argsort(grad)
    low_set = sorted_idx[:n_cand]
    high_set = sorted_idx[-n_cand:]
    best_c5 = corr_max
    best_i = -1
    best_j = -1
    best_delta = 0.0
    for i in high_set:
        hi = h[i]
        if hi < 1e-10:
            continue
        v_ip = (i + shifts >= 0) & (i + shifts < n)
        v_im = (i - shifts >= 0) & (i - shifts < n)
        g_i = np.zeros(2 * n - 1)
        g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
        g_i[v_im] += h[i - shifts[v_im]]
        for j in low_set:
            if i == j:
                continue
            delta_hi = min(hi, 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g = g_i.copy()
            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_c5 - 1e-14:
                best_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
    if best_i >= 0:
        h[best_i] -= best_delta
        h[best_j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def labeled_feasibility(h, target):
    if np.abs(np.sum(h) - target) > 1e-6:
        return False
    if np.any(h < -1e-6) or np.any(h > 1.0 + 1e-6):
        return False
    return True
```
