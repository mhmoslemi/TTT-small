        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        top_threshold = max_val * 0.88
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), 20)
        n_j = min(len(j_cand), 20)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 4000:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.45:
            worst = np.argsort(corr)[-6:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.8:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

_slsqp_alpha = 5000.0
_slsqp_n = 0
_slsqp_target = 0.0

def _slsqp_obj(h):
    a = _slsqp_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / a)

def _slsqp_cons(h):
    return float(np.sum(h)) - _slsqp_target

def _slsqp_jac(h):
    return np.ones(_slsqp_n)

def slsqp_refine(h_start, n, target, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    alphas = [2000.0, 5000.0, 12000.0, 30000.0, 60000.0]
    for a in alphas:
        if time.time() - start_time > time_limit:
            break
        _slsqp_alpha = a
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) // 2
    
    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick greedy binary
    if tleft() > 60:
        h_bin = project_binary(best_h, int(target))
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=15000,
                                   time_limit=min(40, tleft() * 0.15), start_time=start_time)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()
    
    # Phase 2: L-BFGS-B main optimization
    if tleft() > 200:
        time2 = min(500, tleft() - 180)
        alphas_main = [20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2500.0, 5000.0, 10000.0, 20000.0]
        h_lb, s_lb = run_lbfgsb(best_h, alphas_main, mu=200.0, n=n,
                                 maxiter_per_alpha=800, time_limit=time2, start_time=start_time)
        if s_lb < best_score:
            best_score = s_lb
            best_h = h_lb.copy()
    
    # Phase 3: Multi-restart L-BFGS-B with diverse perturbations
    if tleft() > 200:
        n_restarts = 8
        for ri in range(n_restarts):
            if tleft() < 60:
                break
            rt = min(100, (tleft() - 40) / max(1, n_restarts - ri))
            jitter = 0.003 + 0.002 * ri
            h_pert = best_h + rng.randn(n) * jitter
            h_pert = project_box_sum(h_pert, target)
            
            if ri % 3 == 0:
                alphas_r = [100.0, 500.0, 2000.0, 8000.0]
            elif ri % 3 == 1:
                alphas_r = [500.0, 2000.0, 8000.0, 20000.0]
            else:
                alphas_r = [200.0, 1000.0, 5000.0, 15000.0]
            
            h_r, s_r = run_lbfgsb(h_pert, alphas_r, mu=100.0, n=n,
                                    maxiter_per_alpha=500, time_limit=rt, start_time=start_time)
            if s_r < best_score:
                best_score = s_r
                best_h = h_r.copy()
    
    # Phase 4: Binary SA from best continuous solution
    if tleft() > 100:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(150, tleft() - 60)
        max_it = int(min(150000, sa_time * 800))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                 0.005, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 5: SLSQP polish with high alpha
    if tleft() > 80:
        slsqp_time = min(120, tleft() - 30)
        h_sp, s_sp = slsqp_refine(best_h, n, target, slsqp_time, start_time)
        if s_sp < best_score:
            best_score = s_sp
            best_h = h_sp.copy()
    
    # Phase 6: Final L-BFGS-B polish with very high alpha
    if tleft() > 40:
        polish_time = max(10.0, tleft() - 10)
        alphas_final = [20000.0, 50000.0, 100000.0]
        h_pf, s_pf = run_lbfgsb(best_h, alphas_final, mu=500.0, n=n,
                                  maxiter_per_alpha=1000, time_limit=polish_time, start_time=start_time)
        if s_pf < best_score:
            best_score = s_pf
            best_h = h_pf.copy()
    
    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score
    
    return h_final, final_score, n
