            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

**Idea: Iterative Water Filling**
The goal is to make corr[k] as uniform as possible. We can use an iteratively reweighted approach:
1. Compute current corr
2. Identify the "active" shifts (those with corr near max)
3. For each active shift, compute the gradient
4. Take a step in the negative gradient direction (weighted by how much each shift exceeds the mean)
5. Project back to feasible set
6. Repeat

This is essentially a gradient descent on the max (or log-sum-exp) with adaptive step sizes. The difference from SLSQP is that we control the step size and can use more aggressive steps.

Actually, this is what the pair CD is doing (greedily). The difference is that pair CD is more constrained (only 2-coordinate moves) but each move is optimal for that pair.

The subgradient of f(h) = max_k corr_k at h is the subgradient of the active corr_k. If multiple shifts are active, the subgradient is a convex combination.

            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Quick SLSQP polish at n=400
    if tleft() > 80:
        deadline = time.time() + min(50, tleft() * 0.4)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000, 5000, 20000],
                                600, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=800 and run CD
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 250:
            h_cd, c5_cd = pair_cd_v2(h1, n1, target1,
                                     time_limit=min(200, tleft() * 0.5),
                                     max_sweeps=30000,
                                     n_cand_start=15, n_cand_end=65,
                                     max_no_improve=400,
                                     n_active_shifts=3,
                                     reheat_interval=120, reheat_scale=0.04)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                current_n = n1
                current_target = target1
        if tleft() > 100:
            deadline = time.time() + min(70, tleft() * 0.4)
            h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                    [1000, 5000, 20000],
                                    1000, 1e-14, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 3: More CD at n=800
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_v2(best_h, current_n, current_target,
                                 time_limit=min(150, tleft() * 0.5),
                                 max_sweeps=25000,
                                 n_cand_start=20, n_cand_end=80,
                                 max_no_improve=500,
                                 n_active_shifts=4,
                                 reheat_interval=100, reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Interpolate to n=1200
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 250:
            h_cd2, c5_cd2 = pair_cd_v2(h2, n2, target2,
                                       time_limit=min(200, tleft() * 0.55),
                                       max_sweeps=20000,
                                       n_cand_start=12, n_cand_end=50,
                                       max_no_improve=350,
                                       n_active_shifts=3,
                                       reheat_interval=100, reheat_scale=0.04)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                current_n = n2
                current_target = target2
        if tleft() > 80:
            deadline = time.time() + min(60, tleft() * 0.4)
            h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                      [2000, 10000, 30000],
                                      800, 1e-14, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 5: Multi-start perturbation
    if tleft() > 200:
        for attempt in range(4):
            if tleft() < 60:
                break
            noise_scale = [0.03, 0.06, 0.1, 0.15][attempt % 4]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(35, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [200, 1000, 5000, 20000],
                                        500, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 6: Subgradient method
    if tleft() > 100:
        h_sg, c5_sg = subgradient_cd(best_h, current_n, current_target,
                                     time_limit=min(60, tleft() * 0.5),
                                     alpha0=0.2, alpha_min=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Final CD
    if tleft() > 120:
        h_cd_f, c5_f = pair_cd_v2(best_h, current_n, current_target,
                                  time_limit=min(80, tleft() * 0.55),
                                  max_sweeps=15000,
                                  n_cand_start=25, n_cand_end=90,
                                  max_no_improve=300,
                                  n_active_shifts=5,
                                  reheat_interval=80, reheat_scale=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 8: Final SLSQP
    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.5)
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [5000, 20000, 80000],
                                            800, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
