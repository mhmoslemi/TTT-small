Actually, for the projected gradient method, a simpler approach is to use a backtracking line search:
1. Start with a step size α
2. If f(h - α·g) > f(h) - c·α·||g||² (Armijo condition), reduce α

Let me think about what will actually be most effective here. The key insight is:

The function F(h) = max_k f_k(h) is piecewise differentiable. At a point where the max is achieved at a unique k*, the subgradient is ∇f_k*. If multiple shifts are active, the subdifferential is the convex hull of their gradients.

For a **decent direction**, we want a direction d such that f_k(h+td) < f_k(h) for all active k. This is equivalent to d being in the negative polar of the subdifferential.

The **steepest descent direction** in the subgradient sense is:
d = -argmin_{g∈∂F} ||g||²

For the max of functions, ∂F = conv{∇f_k : k active}. So:
d = -argmin_{λ≥0, ∑λ=1} ||∑λ_k ∇f_k*||²

This is a QP in the weights λ. But for simplicity, we can just use the average gradient (uniform weights) or the gradient of the single worst shift.

    n = len(h_curr)

    # grad_q_k(h) = 4h + (Q_k + Q_{-k})h

    # = 4h + sum_k w_k * (Q_k + Q_{-k}) h (since sum w = 1)

    s_eff = np.zeros(n)  # effective S_k

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        weights = np.exp(-beta * (c_max_val - c_top) / c_max_val)
        weights /= weights.sum()

        h_new = dca_step(h, active_k, weights, target, mu)

        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h = h_new
        else:
            improved = False
            for t in [0.5, 0.3, 0.1]:
                h_blend = (1.0 - t) * h + t * h_new
                h_blend = project_feasible(h_blend, target)
                c5_blend = compute_c5(h_blend)
                if c5_blend < c5_now - 1e-15:
                    h = h_blend
                    improved = True
                    break
            if not improved:
                no_improve += 20
                if no_improve > 1000:
                    break

    return best_h, best_c5


def gradient_descent(h_init, n, target, time_limit, max_steps=50000):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.05
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        idxs = np.arange(n)
        grad = np.zeros(n)
        if k_star >= 0:
            A = np.arange(0, n - k_star)
        else:
            A = np.arange(-k_star, n)
        grad[A] += 1.0 - h[A + k_star]
        B = A + k_star
        grad[B] -= h[A]
        grad -= np.mean(grad)

        improved = False
        for _ in range(12):
            h_try = project_feasible(h - alpha * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            alpha = min(alpha * 1.08, 1.0)
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break

    return best_h, best_c5


def pair_cd_polish(h_init, n, target, time_limit, n_cand=30, patience=100):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    no_improve = 0

    for _ in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star = int(np.argmax(corr)) - mid
        idxs = np.arange(n)
        if k_star >= 0:
            A = np.arange(0, n - k_star)
        else:
            A = np.arange(-k_star, n)
        grad = np.zeros(n)
        grad[A] += 1.0 - h[A + k_star]
        B = A + k_star
        grad[B] -= h[A]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_score = c5_val

        for i in high_set:
            h_ip = np.zeros(2 * n - 1)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            h_ip[v_ip] = h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            h_im = np.zeros(2 * n - 1)
            h_im[v_im] = h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                dl = max(h[i] - 1.0, -h[j])
                dh = min(h[i], 1.0 - h[j])
                if dh - dl < 1e-12:
                    continue
                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm = np.zeros(2 * n - 1)
                jpm[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm = np.zeros(2 * n - 1)
                jsm[jm] = h[j - shifts[jm]]

                g = h_ip + h_im - jpm - jsm
                a_d = max(dl, -1e-10)
                b_d = min(dh, 1e10)
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_score:
                    best_score = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)

    return best_h, best_c5


def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    corr_init = np.correlate(h_start, 1.0 - h_start, mode='full')

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def multi_shift_gradient_descent(h_init, n, target, time_limit,
                                  m_active=8, beta=2.5):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.1
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        top_idx = np.argsort(corr)[-m_active:]
        active_k = (top_idx - (n - 1)).astype(int)
        c_top = corr[top_idx]
        c_max_val = float(c_top[-1])
        weights = np.exp(-beta * (c_max_val - c_top) / max(c_max_val, 1e-10))
        weights /= weights.sum()

        grad = np.zeros(n)
        for k, w in zip(active_k, weights):
            kk = int(k)
            if kk >= 0:
                A = np.arange(0, n - kk)
            else:
                A = np.arange(-kk, n)
            grad[A] += w * (1.0 - h[A + kk])
            B = A + kk
            grad[B] -= w * h[A]
        grad -= np.mean(grad)

        improved = False
        for _ in range(10):
            h_try = project_feasible(h - alpha * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            alpha *= 0.5
        if improved:
            alpha = min(alpha * 1.1, 2.0)
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: DCA at n=400 (fast, many iterations)
    if tleft() > 100:
        h1, c51 = dca_descent(h, n, target, time_limit=min(80, tleft() * 0.15),
                              M=8, beta=3.0, mu=0.0)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1
            h = h1

    # Phase 2: Multi-shift gradient at n=400
    if tleft() > 100:
        h2, c52 = multi_shift_gradient_descent(h, n, target,
                                               time_limit=min(60, tleft() * 0.1),
                                               m_active=8, beta=2.5)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
            h = h2

    # Phase 3: DCA at n=800
    if tleft() > 400:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(h, n2, target2)
        h2, c52 = dca_descent(h2, n2, target2, time_limit=min(200, tleft() * 0.25),
                              M=10, beta=2.5, mu=0.0)
        n = n2
        target = target2
        h = h2
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 4: Multi-shift gradient at n=800
    if tleft() > 200:
        h3, c53 = multi_shift_gradient_descent(h, n, target,
                                               time_limit=min(100, tleft() * 0.2),
                                               m_active=10, beta=2.0)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3
            h = h3

    # Phase 5: DCA at n=1600
    if tleft() > 350:
        n3 = 1600
        target3 = n3 / 2.0
        h3 = interpolate_to(h, n3, target3)
        h3, c53 = dca_descent(h3, n3, target3, time_limit=min(200, tleft() * 0.3),
                              M=12, beta=2.0, mu=0.0)
        n = n3
        target = target3
        h = h3
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3

    # Phase 6: Multi-shift gradient at highest resolution
    if tleft() > 150:
        h4, c54 = multi_shift_gradient_descent(h, n, target,
                                               time_limit=min(100, tleft() * 0.2),
                                               m_active=12, beta=1.5)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4
            h = h4

    # Phase 7: Pair CD polish
    if tleft() > 100:
        h5, c55 = pair_cd_polish(h, n, target, time_limit=min(80, tleft() * 0.15),
                                 n_cand=30, patience=80)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5
            h = h5

    # Phase 8: L-BFGS-B polish
    if tleft() > 60:
        for alpha in [1000, 5000, 20000]:
            if tleft() < 30:
                break
            h6 = lbfags_polish(h, n, target, alpha, maxiter=1500)
            c56 = compute_c5(h6)
            if c56 < best_c5:
                best_c5 = c56
                best_h = h6
                h = h6

    # Phase 9: Final DCA + gradient round
    if tleft() > 60:
        h7, c57 = dca_descent(h, n, target, time_limit=min(30, tleft() * 0.3),
                              M=10, beta=2.0, mu=0.0)
        if c57 < best_c5:
            best_c5 = c57
            best_h = h7
        if tleft() > 30:
            h8, c58 = multi_shift_gradient_descent(h, n, target,
                                                   time_limit=min(20, tleft() * 0.25),
                                                   m_active=10, beta=1.5)
            if c58 < best_c5:
                best_c5 = c58
                best_h = h8

    # Phase 10: Final pair CD
    if tleft() > 40:
        h9, c59 = pair_cd_polish(h, n, target, time_limit=min(30, tleft() * 0.4),
                                 n_cand=40, patience=60)
        if c59 < best_c5:
            best_c5 = c59
            best_h = h9

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_use
