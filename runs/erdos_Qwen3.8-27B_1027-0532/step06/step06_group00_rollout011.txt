def simulated_annealing(h_init, n, target, time_limit, initial_temp=0.1, final_temp=0.001):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    rng = np.random.RandomState(99)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    n_steps = 0
    while time.time() - start < time_limit:
        n_steps += 1
        progress = min(n_steps / (time_limit * 50.0), 1.0)
        temp = initial_temp * (final_temp / initial_temp) ** progress

        # Random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (rng.random() - 0.5) * (delta_hi - delta_lo)
        if abs(delta) < 1e-10:
            continue

        # Evaluate
        g = change[j] - change[i]
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_corr = corr + delta * g
        new_c5 = float(np.max(new_corr)) * dx
        cur_c5_val = float(np.max(corr)) * dx

        delta_f = new_c5 - cur_c5_val
        accept = (delta_f < 0) or (rng.random() < np.exp(-delta_f / temp))

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip
