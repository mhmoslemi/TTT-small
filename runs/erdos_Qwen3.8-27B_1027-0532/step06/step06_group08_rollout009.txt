def compute_gradient(h, w):
    """w is the weight array of length 2n-1 (same as correlation array)."""
    n = len(h)
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):
        wk = w[k]
        if abs(wk) < 1e-12:
            continue
        s = shifts[k]
        if s >= 0:
            # c[k] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # grad[j] += wk * (1-h[j-s]) for j in [s, n-1]
            # grad[j] -= wk * h[j+s] for j in [0, n-1-s]
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] -= wk * h[s:]
        else:
            t = -s
            # c[k] = sum_{i=0}^{n-1-t} h[i]*(1-h[i+t])
            # grad[j] += wk * (1-h[j+t]) for j in [0, n-1-t]
            # grad[j] -= wk * h[j-t] for j in [t, n-1]
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n-t]
    return grad

def adam_optimize(h_init, target, n, rng, time_budget, start_time):
    h = h_init.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    alpha = 30.0  # softmax temperature
    
    best_h = h.copy()
    best_score = compute_c5(h)
    
    t = 0
    while time.time() - start_time < start_time + time_budget:
        t += 1
        
        # Compute weights (softmax)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        shifted = alpha * (corr - max_corr)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        w = w / w.sum()
        
        # Increase alpha over time
        alpha = min(alpha * 1.001, 500.0)
        
        # Compute gradient
        grad = compute_gradient(h, w, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h = project_box_sum(h, float(target))
        
        # Evaluate
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Adapt learning rate
        if t % 100 == 0:
