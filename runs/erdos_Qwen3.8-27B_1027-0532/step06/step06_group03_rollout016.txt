_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _smooth_grad_h(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    mid = n - 1
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[mid + k]
        if w < 1e-15:
            continue
        j_lo = max(0, -k)
        j_hi = min(n - 1, n - 1 - k)
        if j_hi < j_lo:
            continue
        j = np.arange(j_lo, j_hi + 1)
        grad[j] += w * (1.0 - h[j + k])
        grad[j + k] -= w * h[j]
    grad *= dx
    return grad


def _con_obj(x):
    return np.sum(x) - _global_target


def _con_jac(x):
    return np.ones(_global_n)


def slsqp_run(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_obj, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t >= time_limit:
            break
        try:
            res = minimize(
                lambda v: _smooth_val(np.correlate(v, 1.0 - v, mode='full'), alpha) * (2.0 / n),
                h_cur, jac=lambda v: _smooth_grad_h(v, alpha),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def binary_snap(h_start, n, target, time_limit, max_passes):

            break
        rng_snap = np.random.RandomState(1000 + p)
        order = rng_snap.permutation(n)
        improved = False
        for i in order:
            if time.time() - start_t >= time_limit:
                break
            v = h[i]
            if abs(v) < 1e-8 or abs(1.0 - v) < 1e-8:
                continue
            for target_val in [0.0, 1.0]:
                delta = target_val - v
                if abs(delta) < 1e-10:
                    continue

                g = np.zeros(2 * n - 1)
                for k_val in range(-(n - 1), n):
                    m = mid + k_val
                    contrib = 0.0
                    if 0 <= i + k_val < n:
                        contrib += (1.0 - h[i + k_val])
                    if 0 <= i - k_val < n:
                        contrib -= h[i - k_val]
                    g[m] = contrib

                new_corr = corr + delta * g
                new_c5 = np.max(new_corr) * dx

                if new_c5 < best_c5 - 1e-12:
                    h_new = h.copy()
                    h_new[i] = target_val
                    diff = target - np.sum(h_new)
                    if abs(diff) > 1e-12:
                        slack = np.where((h_new < 1.0 - 1e-10) & (h_new > 1e-10) & (np.arange(n) != i))[0]
                        if len(slack) > 0:

                        h = h_new.copy()
                        corr = corr_new
                        improved = True

            h_cd, c5_cd = pair_cd_incremental(h_best, n_try, target_try,
                                              time_limit=alloc2, n_cand=40, top_k=3)
            if c5_cd < c5_best:
                c5_best = c5_cd
                h_best = h_cd

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Symmetric SLSQP (exploit reflection symmetry)
    if tleft() > 150:
        h_sym = symmetric_slsqp(best_h, n, target,
                                [50, 200, 1000], 800, min(60, tleft()))
        sc = compute_c5(h_sym)
        if sc < best_score:
            best_score = sc
            best_h = h_sym

    # Phase 2: Full SLSQP (break symmetry)
    if tleft() > 200:
        h_pol = slsqp_run(best_h, n, target,
                          [50, 200, 1000, 5000], 1200, 1e-14, min(120, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Nesterov momentum with adaptive alpha
    if tleft() > 200:
        h_nes, c5_nes = nesterov_momentum(best_h, n, target,
                                          time_limit=min(120, tleft()),
                                          alphas=[50, 200, 800, 3000],
                                          base_lr=0.03, mu=0.7)
        if c5_nes < best_score:
            best_score = c5_nes
            best_h = h_nes

    # Phase 4: Pair CD (aggressive)
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_incremental(best_h, n, target,
                                          time_limit=min(200, tleft()),
                                          n_cand=50, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Binary snapping
    if tleft() > 100:
        h_snap, c5_snap = binary_snap(best_h, n, target,
                                      time_limit=min(80, tleft()), max_passes=200)
        if c5_snap < best_score:
            best_score = c5_snap
            best_h = h_snap

    # Phase 6: Another SLSQP round at high alpha
    if tleft() > 120:
        h_pol = slsqp_run(best_h, n, target,
                          [2000, 10000, 50000], 1000, 1e-15, min(80, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Pair CD again
    if tleft() > 120:
        h_cd, c5_cd = pair_cd_incremental(best_h, n, target,
                                          time_limit=min(100, tleft()),
                                          n_cand=65, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Multi-scale (try higher resolution)
    if tleft() > 150:
        h_best, c5_best = multi_scale_optimize(best_h, best_score,
                                               time_limit=min(150, tleft()),
                                               n_candidates=[500, 600])
        if c5_best < best_score:
            best_score = c5_best
            best_h = h_best

    # Phase 9: Final polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 50:
        h_final = slsqp_run(best_h, n_use, target_use,
                            [5000, 20000, 100000], 800, 1e-16, min(40, tleft()))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 10: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = pair_cd_incremental(best_h, n_use, target_use,
                                          time_limit=min(25, tleft()),
                                          n_cand=50, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
