import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r

def compute_corr_fft(h, N):
    """Compute the full correlation of h with 1-h using FFT."""
    n = len(h)
    H = np.fft.rfft(h, N)
    O = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * O, N)
    # Reorder to match np.correlate(h, 1-h, 'full') order
    # Positive shifts (0 to n-1) are at indices 0 to n-1
    # Negative shifts (-(n-1) to -1) are at indices N-n+1 to N-1
    corr_full = np.concatenate([corr[N-n+1:], corr[:n]])
    return corr_full

def compute_corr_direct(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_value_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    return value, weights

def gradient_fft(h, weights, N):
    """Compute gradient of soft-max objective using FFT."""
    n = len(h)
    one_minus_h = 1.0 - h
    
    # w_pos[s] = weights[n-1+s] for s=0,...,n-1
    w_pos = weights[n-1:2*n-1]
    # w_neg[s] = weights[n-1-s] for s=1,...,n-1; w_neg[0] = 0
    w_neg = np.zeros(n)
    w_neg[1:] = weights[n-2::-1]  # weights[n-2], weights[n-3], ..., weights[0]
    
    # Precompute FFTs
    H = np.fft.rfft(h, N)
    O = np.fft.rfft(one_minus_h, N)
    WP = np.fft.rfft(w_pos, N)
    WN = np.fft.rfft(w_neg, N)
    
    # term1 = cross-correlation of w_pos with (1-h): ifft(FFT(w_pos) * conj(FFT(1-h)))
    term1 = np.fft.irfft(WP * np.conj(O), N)[:n]
    
    # term2 = convolution of w_pos with h: ifft(FFT(w_pos) * FFT(h))
    term2 = np.fft.irfft(WP * H, N)[:n]
    
    # term3 = convolution of w_neg with (1-h): ifft(FFT(w_neg) * FFT(1-h))
    term3 = np.fft.irfft(WN * O, N)[:n]
    
    # term4 = cross-correlation of w_neg with h: ifft(FFT(w_neg) * conj(FFT(h)))
    term4 = np.fft.irfft(WN * np.conj(H), N)[:n]
    
    grad = term1 - term2 + term3 - term4
    return grad

def soft_max_obj_grad_fft(h, alpha):
    n = len(h)
    N = 2 * n
    corr = compute_corr_fft(h, N)
    value, weights = soft_max_value_weights(corr, alpha)
    grad = gradient_fft(h, weights, N)
    return value, grad

def adam_fft(h0, target_sum, start_time, time_budget, alphas, lr, beta1, beta2, eps):
    n = len(h0)
    N = 2 * n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = soft_max_obj_grad_fft(h, alpha)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
            
            # Periodic actual evaluation
            if t % 50 == 0:
                actual = compute_c5(h)
                if actual < best_actual:
                    best_actual = actual
                    best_h = h.copy()
    
    actual = compute_c5(h)
    if actual < best_actual:
        best_actual = actual
        best_h = h.copy()
    
    return best_h, best_actual

def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width=20):
    n = len(h)
    h = h.copy()
    corr = compute_corr_direct(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 100
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = []
        for i in ones:
            if 0 <= i + k_star < n and h[i + k_star] < 0.5:
                i_cand.append(i)
            elif 0 <= i - k_star < n and h[i - k_star] < 0.5:
                i_cand.append(i)
        
        # Simplified: try random swaps and keep best
        improved = False
        for _ in range(50):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            corr_new = compute_corr_direct(h)
            c5_new = float(corr_new.max()) * (2.0 / n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stall += 1
            # Recompute from best

            corr = compute_corr_direct(h)

def run(seed=42, budget_s=1300, **kwargs):
