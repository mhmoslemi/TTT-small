def multi_active_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=50, n_bisect=15, 
                    active_frac=0.1, momentum=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    mom_vec = np.zeros(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 120:
            break
        
        # Multi-active gradient
        c_max = float(np.max(corr))
        # Adaptive threshold: start wide, narrow over time
        frac = active_frac * (1.0 + 0.5 * np.exp(-sweep / 200.0))
        threshold = c_max * (1.0 - frac)
        active_mask = corr >= threshold
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(corr)])
        
        # Weighted gradient (softmax weights based on how close to max)
        diffs = c_max - corr[active_indices]
        weights = np.exp(-diffs * 200.0)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for a_idx, s_idx in enumerate(active_indices):
            k = int(s_idx) - mid
            w = weights[a_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Momentum
        if momentum > 0:
            mom_vec = momentum * mom_vec + (1 - momentum) * grad
            grad = mom_vec
            grad -= np.mean(grad)
        
        # Pair selection
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]
            
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]
                
                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_set = vals >= f_mid - 1e-14
                    f_prime = float(np.max(g[active_set])) if active_set.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 50:
                break
    
    return best_h, best_c5

def dual_ascent(h_init, n, target, time_limit, max_iter=5000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    lambda_k = np.ones(n_corr) / n_corr
    lr_h = 0.02
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Update dual variables
        violations = np.maximum(corr - c5_val, 0.0)
        lr_lambda = 0.5 / (1.0 + it / 500.0)
        lambda_k = lambda_k + lr_lambda * violations
        lambda_k = np.maximum(lambda_k, 0.0)
        s = lambda_k.sum()
        if s < 1e-10:
            lambda_k = np.ones(n_corr) / n_corr
        else:
            lambda_k /= s
        
        # Compute gradient
        grad = np.zeros(n)
        active = lambda_k > 1e-8
        for s_idx in np.where(active)[0]:
            k = int(s_idx) - mid
            w = lambda_k[int(s_idx)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        # Projected gradient step with backtracking
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr_h * mult
            h_try = project_feasible(h - lr_try * grad, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            # Check if weighted violation decreased
            viol_try = np.sum(lambda_k * np.maximum(corr_try - c5_val, 0.0))
            viol_old = np.sum(lambda_k * violations)
            if viol_try < viol_old:
                h = h_try
                improved = True
                break
        if not improved:
            lr_h *= 0.5
            if lr_h < 1e-8:
                break
        else:
            lr_h = min(lr_h * 1.1, 0.1)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def _slsqp_cons(h):
    return np.sum(h) - _g_target


def _slsqp_cons_jac(h):
    return np.ones(_g_n)


def slsqp_polish(h_start, n, target, alphas, maxiter):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    return project_feasible(h_out, target_dst)

            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        cd_time = min(280.0, remaining * 0.30)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=10000, n_cand=60, n_bisect=13,
                                       active_frac=0.15, momentum=0.25, weight_temp=150.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        h_pol = slsqp_polish(best_h, n, target, [5000, 20000], 1000)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 120:
        da_time = min(150.0, remaining * 0.20)
        h_da, c5_da = dual_ascent(best_h, n, target, time_limit=da_time, max_iter=5000)
        if c5_da < best_score:
            best_score = c5_da
            best_h = h_da

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 120:
        cd_time = min(180.0, remaining * 0.25)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_cand=55, n_bisect=14,
                                       active_frac=0.08, momentum=0.3, weight_temp=300.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        h_pol = slsqp_polish(best_h, n, target, [20000, 80000], 800)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 180:
        for n_try in [500, 400]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_lo = slsqp_polish(h_interp, n_try, target_try, [500, 2000], 600)
            sc = compute_c5(h_lo)
            if sc < best_score + 0.003:
                cd_t = min(60.0, (budget_s - 30.0 - (time.time() - start_time)) * 0.25)
                h_lo2, sc2 = multi_active_cd(h_lo, n_try, target_try,
                                              time_limit=cd_t,
                                              max_sweeps=5000, n_cand=45, n_bisect=13,
                                              active_frac=0.1, momentum=0.2, weight_temp=200.0)
                if sc2 < best_score:
                    best_h = interpolate_to(h_lo2, n, target)
                    best_score = compute_c5(best_h)

    remaining = budget_s - 30.0 - (time.time() - start_time)
    rng = np.random.RandomState(seed)
    for noise_scale in [0.04, 0.08, 0.15]:
        remaining = budget_s - 30.0 - (time.time() - start_time)
        if remaining < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(50.0, remaining * 0.2)
        h_try, c5_try = multi_active_cd(h_pert, n, target,
                                         time_limit=cd_t,
                                         max_sweeps=4000, n_cand=40, n_bisect=12,
                                         active_frac=0.1, momentum=0.15, weight_temp=100.0)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(70.0, remaining * 0.5)
        h_cd, c5_cd = multi_active_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=6000, n_cand=70, n_bisect=15,
                                       active_frac=0.05, momentum=0.2, weight_temp=500.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_polish(best_h, n_use, target_use, [50000, 200000], 500)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
