def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-15:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-15:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_grad(corr, h, alpha, p_threshold=1e-8):
    n = len(h)
    max_c = float(np.max(corr))
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    e = np.exp(shifted)
    p = e / e.sum()
    
    active = np.where(p > p_threshold)[0]
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for k in active:
        pk = p[k]
        s = shifts[k]
        if s == 0:
            grad += pk * (2.0*h - 1.0)
        elif s > 0:
            grad[:n-s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n-s]
        else:
            a = -s
            grad[a:] += pk * (1.0 - h[:n-a])
            grad[:n-a] -= pk * h[a:]
    return grad

    best_h = h.copy()

        v = np.zeros(n)

            if time.time() - start_time > time_limit:

            lr = lr_init * (lr_end / lr_init) ** ratio
            
            grad = soft_grad(corr, h, alpha)
            
            m = b1*m + (1-b1)*grad
            v = b2*v + (1-b2)*grad*grad
            m_hat = m / (1 - b1**(it+1))
            v_hat = v / (1 - b2**(it+1))
            
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = proj_box_sum(h, target)

            c5 = float(np.max(corr)) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def proj_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-15:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-15:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_grad(corr, h, alpha, threshold=1e-10):
    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-15:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-15:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_grad(corr, h, alpha, threshold=1e-10):
    n = len(h)
    max_c = float(np.max(corr))
    shifted = np.clip(alpha * (corr - max_c), -700.0, 0.0)
    e = np.exp(shifted)
    p = e / e.sum()
    active = np.where(p > threshold)[0]
    grad = np.zeros(n)
    for k in active:
        pk = p[k]
        s = k - (n - 1)
        if s == 0:
            grad += pk * (2.0 * h - 1.0)
        elif s > 0:
            grad[:n-s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n-s]
        else:
            a = -s
            grad[a:] += pk * (1.0 - h[:n-a])
            grad[:n-a] -= pk * h[a:]
    return grad

def run_adam_phase(h0, target, n_iters, alpha, lr_init, lr_end, time_limit, start_time,
                   b1=0.9, b2=0.999, eps=1e-8, max_stall=8000):
    n = len(h0)
    h = h0.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        grad = soft_max_grad(corr, h, alpha)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))

        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = proj_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

_SLQ_ALPHA = 1.0
_SLQ_TARGET = 1.0

def _slsqp_obj(x):
    a = _SLQ_ALPHA
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(a * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / a

def _slsqp_jac(x):
    a = _SLQ_ALPHA
    corr = np.correlate(x, 1.0 - x, mode='full')
    return soft_max_grad(corr, x, a, threshold=1e-12)

def _slsqp_cons_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slsqp_cons_jac(x):
    return np.ones(len(x))

def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    global _SLQ_ALPHA, _SLQ_TARGET
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        _SLQ_ALPHA = alpha
        _SLQ_TARGET = target
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_jac,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = proj_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass

    return h_cur, curr_score

def get_time_left(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)

def binary_project(h, target_int):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_int:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_int:
        need = current_sum - target_int
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_int - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def swap_improve(h, max_iters, time_limit, start_time, rng):
    n = len(h)
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break

        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        A = np.zeros(n)
        idx = np.arange(n) + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = np.arange(n) - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = np.arange(n) - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = np.arange(n) + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = np.zeros(2 * n - 1)
                ks = np.arange(2 * n - 1) - (n - 1)
                idx_i = i + ks
                mask = (idx_i >= 0) & (idx_i < n)
                delta[mask] += h[idx_i[mask]]
                idx_i = i - ks
                mask = (idx_i >= 0) & (idx_i < n)
                delta[mask] += h[idx_i[mask]]
                idx_j = j - ks
                mask = (idx_j >= 0) & (idx_j < n)
                delta[mask] -= h[idx_j[mask]]
                idx_j = j + ks
                mask = (idx_j >= 0) & (idx_j < n)
                delta[mask] -= h[idx_j[mask]]

                new_corr = corr + delta
                new_max = float(np.max(new_corr))
                if new_max < float(np.max(corr)) - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    stall = 0
                    c5 = float(np.max(corr)) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            stall += 1

    return best_h, best_c5
