                c += delta * D
                total_improved += 1

        c5_now = float(np.max(c)) * dx_val
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        max_C = np.max(C)
        alpha = alphas[slsqp_optimize._alpha_idx]
        shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
        return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)

    for idx, alpha in enumerate(alphas):
        slsqp_optimize._alpha_idx = idx
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def gradient_based_polish(h_start, n, target, time_limit, max_steps=10000, lr=0.01, n_top=10):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        if step > 100 and no_improve > 500:
            break
        corr = compute_corr(h)
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            idx_lo = np.arange(n) - k
            mask1 = (idx_lo >= 0) & (idx_lo < n)
            g = np.zeros(n)
            g[mask1] += (1.0 - h[idx_lo[mask1]])
            idx_hi = np.arange(n) + k
            mask2 = (idx_hi >= 0) & (idx_hi < n)
            g[mask2] -= h[idx_hi[mask2]]
            grad += g
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad * grad
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    if tleft() > 200:
        h_cd, sc = paired_cd(best_h, n, target, time_limit=min(200, tleft() * 0.3),
                             max_sweeps=5000, n_pairs_per_sweep=200, seed=seed)
        if sc < best_score:
            best_score, best_h = sc, h_cd

    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [500, 2000, 10000, 50000, 200000], 1500, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_pol

    if tleft() > 150:
        h_cd2, sc = paired_cd(best_h, n, target, time_limit=min(120, tleft() * 0.3),
                              max_sweeps=5000, n_pairs_per_sweep=300, seed=seed + 1)
        if sc < best_score:
            best_score, best_h = sc, h_cd2

    if tleft() > 100:
        h_grad, sc = gradient_based_polish(best_h, n, target, time_limit=min(60, tleft() * 0.25),
                                           max_steps=8000, lr=0.01, n_top=15)
        if sc < best_score:
            best_score, best_h = sc, h_grad

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [5000, 20000, 100000, 500000], 1000, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_pol

    for n_try in [500, 600, 800, 1000]:
        if tleft() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        if tleft() > 80:
            h_cd, sc = paired_cd(h_interp, n_try, target_try, time_limit=min(60, tleft() * 0.4),
                                 max_sweeps=3000, n_pairs_per_sweep=150, seed=seed + n_try)
            if sc < best_score:
                best_score, best_h = sc, h_cd
        if tleft() > 40 and best_score < 0.381:
            h_cd2, sc = paired_cd(best_h, len(best_h), len(best_h) / 2.0,
                                  time_limit=min(40, tleft() * 0.3),
                                  max_sweeps=2000, n_pairs_per_sweep=200, seed=seed + 999)
            if sc < best_score:
                best_score, best_h = sc, h_cd2

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                   [10000, 50000, 200000, 1000000], 800, 1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_pol

    if tleft() > 40:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 20:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_cd, sc = paired_cd(h_pert, len(h_pert), len(h_pert) / 2.0,
                                 time_limit=min(15, tleft() * 0.3),
                                 max_sweeps=1000, n_pairs_per_sweep=200, seed=seed + int(noise_scale * 100))
            if sc < best_score:
                best_score, best_h = sc, h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
