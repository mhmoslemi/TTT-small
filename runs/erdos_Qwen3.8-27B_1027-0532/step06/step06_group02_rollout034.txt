    n = len(h)
    # Compute C(k) for all lags
    C = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # C[k + n - 1] = sum_i h[i] * (1-h[i+k]) for k = -(n-1), ..., n-1
    # Actually, np.correlate(h, 1-h, mode='full')[k] = sum_n h[n+k] * (1-h)[n]
    # = sum_n h[n+k] * (1-h[n])

    exp_C = np.exp(alpha * (C - C_max))
    W = exp_C / exp_C.sum()  # length 2n-1, W[i] corresponds to lag i-(n-1)

def adam_update(params, m, v, t, lr, beta1, beta2, eps):
    m_new = beta1 * m + (1-beta1) * params
    v_new = beta2 * v + (1-beta2) * params**2
    m_hat = m_new / (1 - beta1**t)
    v_hat = v_new / (1 - beta2**t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return update, m_new, v_new
