def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):

    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_gradient_weighted(h, alpha, corr_full):
    n = len(h)
    max_c = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - max_c))

    conv = conv_full[:n]
    # Gradient of weighted sum of f_k
    grad = 1.0 - 2.0 * conv
    return grad, w, max_c

def _cons_fun(h):

def slsqq_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(h, a):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted))) / a)
        
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    _cons_target = target
    bounds = [(0.0, 1.0)] * n

    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(h, a):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted))) / a)

        no_improve = 0
        local_best = best_score
        local_best_h = best_h.copy()

        while time.time() < phase_end:
            t_step += 1
            corr = full_corr_fast(h)
            grad = weighted_subgradient(h, corr, alpha)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)
            step = lr_base * m_hat / (np.sqrt(v_hat) + eps)
            step_norm = np.linalg.norm(step)
            if step_norm > 0.5:
                step = step * (0.5 / step_norm)
            h_new = h - step
            h_new = project_box_sum(h_new, target)
            h = h_new

            if t_step % 200 == 0:
                sc = compute_c5(h)
                if sc < local_best - 1e-8:
                    local_best = sc
                    local_best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                if no_improve >= patience:
                    h = project_box_sum(h + rng.randn(n) * 0.03, target)
                    no_improve = 0
                    m *= 0.5
                    v *= 0.5
                elif no_improve >= patience // 2:
                    lr_base *= 0.7

        if local_best < best_score:
            best_score = local_best
            best_h = local_best_h.copy()
        h = local_best_h.copy()

    return best_h, best_score

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
