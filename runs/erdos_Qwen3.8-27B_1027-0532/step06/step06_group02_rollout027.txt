import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_assign(h, target_sum):
    n = len(h)
    h_bin = np.zeros(n)
    idx = np.argsort(-h)[:target_sum]
    h_bin[idx] = 1.0
    return h_bin

def compute_grad_fft(h, alpha, corr):
    n = len(h)
    max_c = corr.max()
    exp_c = np.exp(alpha * (corr - max_c))
    w = exp_c / exp_c.sum()
    N_fft = 1
    while N_fft < 2 * n:
        N_fft <<= 1
    w_p = np.zeros(N_fft)
    w_p[:2*n-1] = w
    h_p = np.zeros(N_fft)
    h_p[:n] = h
    W = np.fft.fft(w_p)
    H = np.fft.fft(h_p)
    T2 = np.fft.ifft(np.conj(W) * H).real[:n]
    T3 = np.fft.ifft(W * H).real[:n]
    cumw = np.zeros(2*n)
    cumw[1:] = np.cumsum(w)
    j_arr = np.arange(n)
    T1 = cumw[2*n-1-j_arr] - cumw[n-1-j_arr]
    return T1 - T2 - T3

def adam_continuous(h0, target, start_time, time_limit, alphas, lr_base=0.1):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_score = compute_c5(h)
    best_bin = project_binary(h, int(target))
    best_bin_score = compute_c5(best_bin)
    best_h_cont = h.copy()
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_budget = min(40.0, time_limit - (time.time() - start_time))
        n_iter = int(phase_budget * 3000)
        n_iter = max(n_iter, 200)
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            if it % 200 == 0 or it == 0:
                corr = full_corr_fast(h)
            else:
                corr = full_corr_fast(h)
            grad = compute_grad_fft(h, alpha, corr)
            lr = lr_base / (1.0 + 0.01 * alpha)
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_count)
            v_hat = v / (1.0 - beta2 ** t_count)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h, target)
            if it % 500 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h_cont = h.copy()
                h_bin = project_binary(h, int(target))
                bin_sc = compute_c5(h_bin)
                if bin_sc < best_bin_score:
                    best_bin_score = bin_sc
                    best_bin = h_bin.copy()
                h_ga = greedy_assign(h, int(target))
                ga_sc = compute_c5(h_ga)
                if ga_sc < best_bin_score:
                    best_bin_score = ga_sc
                    best_bin = h_ga.copy()
    return best_h_cont, best_score, best_bin, best_bin_score

                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move_found = True
                    break
            if not best_move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    for _ in range(n_pert):
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)

    best_h = project_binary(h_init, target_int)
    best_score = compute_c5(best_h)

    alphas_main = [3, 8, 20, 50, 120, 300, 800, 2000, 5000, 12000]
    allowed = min(350, max(50, tleft() * 0.30))
    h_cont, sc_cont, h_bin1, sc_bin1 = adam_continuous(best_h, target, start_time, allowed, alphas_main, lr_base=0.15)
    if sc_bin1 < best_score:
        best_score = sc_bin1
        best_h = h_bin1.copy()

    if tleft() > 50:
        allowed = min(80, max(20, tleft() * 0.08))
        h_g, c5_g = fast_greedy(best_h, max_iters=50000, rng=rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    if tleft() > 80:
        allowed = min(150, max(30, tleft() * 0.15))

                                time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                block_frac=0.2, max_block_len=12)
        if c5_sa < best_score:

            best_h = h_sa.copy()

    if tleft() > 80:
        alphas2 = [10, 30, 80, 200, 500, 1500, 4000, 10000]

        if sc_bin2 < best_score:
            best_score = sc_bin2
            best_h = h_bin2.copy()
        if sc_cont2 < best_score:

            sc_r = compute_c5(h_rounded)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_rounded.copy()
            h_ga = greedy_assign(h_cont2, target_int)

            if sc_ga < best_score:

                best_h = h_ga.copy()

    if tleft() > 60:
        allowed = min(100, max(20, tleft() * 0.12))
        n_sa2 = int(allowed * 50000)
        h_sa2, c5_sa2 = sa_focused(best_h, n_sa2, 0.015, 0.00003, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    for restart in range(10):
        if tleft() < 60:
            break
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(3, 8))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 12)
        else:
            h_pert = perturb_diverse(best_h, rng, 2)
            h_pert = perturb_block(h_pert, rng, 6)
        allowed = min(50, max(10, tleft() * 0.06))
        n_sa_r = int(allowed * 45000)
        h_sr, c5_sr = sa_refine(h_pert, n_sa_r, 0.01, 0.00002, rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                block_frac=0.25, max_block_len=10)

        alphas3 = [5, 15, 50, 150, 500, 2000, 8000, 20000]
        allowed = min(200, max(40, tleft() * 0.25))
        h_cont3, sc_cont3, h_bin3, sc_bin3 = adam_continuous(best_h, target, start_time, allowed, alphas3, lr_base=0.08)

    n_final = len(best_h)
    target_final = float(n_final) / 2.0

    final_score = compute_c5(h_final)
    if eval_score := compute_c5(project_binary(h_final, int(target_final))):
        if eval_score < final_score:
            h_final = project_binary(h_final, int(target_final))
            final_score = eval_score

    return h_final, final_score, n_final
