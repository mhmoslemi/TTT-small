4. Find the steepest descent direction in the feasible subspace: d = -P(g) where P projects onto {v : sum(v) = 0}.
5. Line search: find α that minimizes C(h - α*d) subject to 0 ≤ h - α*d ≤ 1.

import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

# Pre-imported: initial_h_values, evaluate_erdos_solution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {0≤h≤1, sum(h)=target_sum} efficiently."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Binary search for the threshold
    # We want to add diff/n to each element, but clip at 0 and 1
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_adj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_adj)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Fix any tiny residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    """Gradient of C_k with respect to h, for shift k (offset from center)."""
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def logsumexp_slsqp(h_start, n, target, alphas, maxiter, ftol):
    """SLSQP optimization with log-sum-exp smoothing."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            c = corr
            c_max = np.max(c)
            shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        
        def grad(h):
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            c_max = np.max(corr)
            weights = np.exp(np.clip(alpha * (corr - c_max), -700.0, 0.0))
            wsum = weights.sum()
            g = np.zeros(n)
            for k_idx in range(len(corr)):
                if weights[k_idx] / wsum < 1e-10:
                    continue
                k = k_idx - (n - 1)
                g += (weights[k_idx] / wsum) * get_grad_shift(h, k, n)
            ls = c_max + np.log(wsum) / alpha
            return g / ls
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=grad,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5

def projected_gradient_descent(h_start, n, target, time_limit, max_steps, n_top, lr_init, lr_decay):

    start = time.time()

    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    lr = lr_init
    no_improve = 0

            break
        if step > 100 and no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:

        # Get top shifts
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:

            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        # Project gradient to feasible subspace (sum = 0)
        grad -= np.mean(grad)
        
        # Ensure we're moving in descent direction
        if np.all(np.abs(grad) < 1e-15):

        # Backtracking line search
        improved = False
        for back in range(20):

            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)

                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_h, best_c5

def adam_topk(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
              lr, beta1, beta2, eps):
    """Adam optimizer with top-k gradient."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        frac = min(step / max(1, int(max_steps * 0.3)), 1.0)
        n_top_cur = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top_cur = max(n_top_cur, 1)
        top_idx = np.argsort(corr)[-n_top_cur:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top_cur
        grad *= dx
        grad -= np.mean(grad)
        
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, target)
    
    return best_h, best_c5

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Actually, let's use the formula:
        # If we move delta from h[i] to h[j]:
        # New h[i] = h[i] - delta, new h[j] = h[j] + delta
        # Change in corr at lag d: (h[i]-delta)*(1-h[i-d]-... ) - h[i]*(1-h[i-d]) + similar for j
        # This is complicated. Let's use the gradient-based approach.

        best_trial = c5_val
        best_move = None
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                # Change in h: h[i] -= delta, h[j] += delta
                # Change in corr[k] for each k:
                # corr[k] = sum_m h[m]*(1-h[m-d_k])
                # After change: sum_m h'[m]*(1-h'[m-d_k])

                #   - delta*(1-h'[i-d_k]) ... 
                # This gets complicated because h also appears in the (1-h) factor.

                # where grad_k[i] is the partial of corr[k] w.r.t. h[i]

                max_delta = min(h[i], 1.0 - h[j])

                    continue

                    delta = frac * max_delta
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = np.clip(h_try, 0, 1)
                    c_try = float(np.max(np.correlate(h_try, 1-h_try, mode='full'))) * dx

                        best_trial = c_try
                        best_move = (int(i), int(j), delta)

            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def diverse_starts(h_base, n, target, n_starts, seed):

    h_block = project_feasible(h_block, target)
    starts.append(h_block)
