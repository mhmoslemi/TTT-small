def ilqp_step(h0, n, target, max_lp_iter=100):
    """One step of ILQP: linearize and solve LP."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    
    # Compute current correlations
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Compute gradients for all correlations
    # g[k, j] = d corr[k] / d h[j]
    # This is a (2n-1) x n matrix
    # For efficiency, let's compute it
    A = np.zeros((n_corr, n))
    for s_idx in range(n_corr):
        s = shifts[s_idx]
        v_plus = (idxs + s >= 0) & (idxs + s < n)
        v_minus = (idxs - s >= 0) & (idxs - s < n)
        A[s_idx, v_plus] = (1.0 - h0[idxs[v_plus] + s])
        A[s_idx, v_minus] -= h0[idxs[v_minus] - s]
    A *= dx
    
    # LP: minimize C
    # Variables: [h (n vars), C (1 var)]
    # Objective: c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Wait, let me re-derive.

    # → A[k,:] @ h - C ≤ corr0[k] - A[k,:] @ h0
    # In standard form: A_ineq @ x ≤ b_ineq where x = [h, C]
    A_ineq = np.zeros((n_corr, n + 1))
    A_ineq[:, :n] = A
    A_ineq[:, n] = -1.0
    b_ineq = corr0 - A @ h0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [target]
    
    # Bounds: 0 ≤ h[i] ≤ 1, C ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]

        v_minus = (idxs - s >= 0) & (idxs - s < n)
        A[s_idx, v_plus] = (1.0 - h0[idxs[v_plus] + s])
        A[s_idx, v_minus] -= h0[idxs[v_minus] - s]
    A *= dx
    return A, shifts, idxs

                break

For the pair CD, the key insight is: after changing h_i by δ and h_j by -δ (to maintain the sum constraint), the new correlation at shift k is:

This is a 1D optimization: minimize the max of linear functions of δ. The optimal δ is where two of the linear functions cross (or at a boundary).

We can solve this exactly by:
1. Computing all "break points" where the max changes
2. Evaluating the max at each break point
3. Taking the minimum

Hmm, this is getting complicated. Let me simplify the pair CD. Instead of computing the full gradient for each pair, I'll precompute the "influence vector" for each index.

For index i, the influence vector v_i ∈ R^(2n-1) is:
v_i[k] = d corr[k] / d h[i] = (1-h[i+s_k])*I(i+s_k valid) - h[i-s_k]*I(i-s_k valid)

where s_k = k - (n-1).

For a pair (i, j) with h_i -= δ, h_j += δ:
new_corr[k] = corr[k] + δ * (v_j[k] - v_i[k])

We want to minimize max_k new_corr[k].

This is a 1D problem: minimize max_k (c_k + δ * d_k) where c_k = corr[k] and d_k = v_j[k] - v_i[k].

The optimal δ can be found by:
1. If all d_k = 0, any δ works
2. Otherwise, the function f(δ) = max_k (c_k + d_k * δ) is the max of linear functions, so it's convex piecewise linear in δ
3. The minimum is at a "kink point" where two lines cross, or at a boundary

We can find the optimal δ by:
- Computing all pairwise intersections of the lines c_k + d_k * δ
- Evaluating f at each intersection and at the boundaries

import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_A_matrix(h0, n):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    A = np.zeros((n_corr, n))
    for s_idx in range(n_corr):
        s = shifts[s_idx]
        vp = (idxs + s >= 0) & (idxs + s < n)
        vm = (idxs - s >= 0) & (idxs - s < n)
        A[s_idx, vp] = (1.0 - h0[idxs[vp] + s])
        A[s_idx, vm] -= h0[idxs[vm] - s]
    A *= dx
    return A


def ilqp_step(h0, n, target):
    n_corr = 2 * n - 1
    A = compute_A_matrix(h0, n)
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((n_corr, 1))])
    b_ub = corr0 - A @ h0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return h0.copy(), np.inf


def ilqp_solve(h0, n, target, max_iter, time_limit):
    start_t = time.time()
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        h_lp, C_lp = ilqp_step(h_cur, n, target)
        if not np.isfinite(C_lp):
            break
        improved = False
        for lr in [1.0, 0.75, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]:
            h_try = h_cur + lr * (h_lp - h_cur)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h_cur = h_try
                best_c5 = c5_try
                best_h = h_try.copy()
                improved = True
                no_improve = 0
                break
            elif c5_try < compute_c5(h_cur) - 1e-15:
                h_cur = h_try
                break
        if not improved:
            no_improve += 1
            if no_improve >= 15:
                rng_esc = np.random.RandomState(1000 + it)
                h_cur = project_feasible(h_cur + rng_esc.randn(n) * 0.015, target)

def smooth_max_grad(h, alpha):

    dx = 2.0 / n

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)

    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    grad = np.zeros(n)

                             [200, 1000, 5000], 1200, 1e-16,
                             min(80, tleft()))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Pair CD again
    if tleft() > 120:
        h_cd, c5 = pair_cd(best_h, n, target,
                           time_limit=min(100, tleft() * 0.5),
                           max_sweeps=12000, n_cand=60, top_k=5)
        if c5 < best_score:
            best_score = c5
            best_h = h_cd

    # Phase 6: Higher resolution
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_new, c5 = ilqp_solve(h_interp, n_try, target_try,
                                   max_iter=2000, time_limit=min(40, tleft() * 0.4))
            if c5 < best_score:
                best_score = c5
                best_h = h_new
                n = n_try
                target = target_try
            if tleft() < 80:
                break
            h_cd2, c5b = pair_cd(best_h, n, target,
                                 time_limit=min(30, tleft() * 0.3),
                                 max_sweeps=5000, n_cand=40, top_k=3)
            if c5b < best_score:
                best_score = c5b
                best_h = h_cd2

    # Phase 7: Local perturbation search
    if tleft() > 100:
        h_pert, c5_p = local_perturb(best_h, n, target,
                                     max_tries=5000,
                                     time_limit=min(30, tleft() * 0.3),
                                     seed_val=seed + 1)
        if c5_p < best_score:
            best_score = c5_p
            best_h = h_pert

    # Phase 8: Final ILQP
    if tleft() > 60:
        h_ilqp, c5 = ilqp_solve(best_h, n, target, max_iter=3000,
                                time_limit=min(40, tleft() * 0.6))
        if c5 < best_score:
            best_score = c5
            best_h = h_ilqp

    # Phase 9: Final pair CD
    if tleft() > 30:
        h_cd, c5 = pair_cd(best_h, n, target,
                           time_limit=min(20, tleft() * 0.7),
                           max_sweeps=5000, n_cand=50, top_k=3)
        if c5 < best_score:
            best_score = c5
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 20:
        h_pol = slsqp_polish(best_h, n, target, [10000, 50000],
                             500, 1e-16, min(15, tleft() * 0.8))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
