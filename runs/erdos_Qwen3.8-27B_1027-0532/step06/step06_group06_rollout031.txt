def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    n = len(h)
    lo = max(float(h.min()) - 1e-6, -1.0)
    hi = min(float(h.max()) + 1e-6, 2.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(np.int32)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones].copy())[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros].copy())[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def get_top_lags(corr, m):
    """Get indices of top m correlation values."""
    if m >= len(corr):
        return np.arange(len(corr))
    return np.argsort(corr)[-m:]

def compute_delta_matrices(h, ones, zeros, top_lags, n):
    """Compute out and in delta matrices for the top lags."""
    n1 = len(ones)
    n0 = len(zeros)
    m = len(top_lags)
    out_mat = np.zeros((n1, m), dtype=np.float64)
    in_mat = np.zeros((n0, m), dtype=np.float64)
    
    for li, lag_idx in enumerate(top_lags):
        shift = lag_idx - (n - 1)
        
        # out: effect of turning off a 1 at position ones[a]
        ip = ones + shift
        vpos = (ip >= 0) & (ip < n)
        ineg = ones - shift
        vneg = (ineg >= 0) & (ineg < n)
        out_mat[:, li] = -np.where(vpos, 1.0 - h[ip], 0.0) + np.where(vneg, h[ineg], 0.0)
        
        # in: effect of turning on a 0 at position zeros[b]
        jp = zeros + shift
        vpos = (jp >= 0) & (jp < n)
        jneg = zeros - shift
        vneg = (jneg >= 0) & (jneg < n)
        in_mat[:, li] = np.where(vpos, 1.0 - h[jp], 0.0) - np.where(vneg, h[jneg], 0.0)
    
    return out_mat, in_mat

def vectorized_local_search(h, time_limit, start_time, top_m=30, max_rounds=5000):
    """Fast vectorized binary local search."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 30:
            break

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        top_corr = corr[top_lags].copy()
        
        out_mat, in_mat = compute_delta_matrices(h, ones, zeros, top_lags, n)
        
        # Compute approximate new max for all pairs
        n1 = len(ones)
        n0 = len(zeros)
        delta = out_mat[:, None, :] + in_mat[None, :, :]

        new_max_approx = np.max(new_vals, axis=2)

        order = np.argsort(flat)
        
        improved = False
        tried = 0
        max_tried = 50

                h[j] = 0

def downsample_h(h, n_target):
    """Downsample h to n_target points via averaging."""
    n = len(h)
    if n_target >= n:
        return h.copy()
    # Block averaging
    block_size = n // n_target
    n_use = block_size * n_target
    h_use = h[:n_use].reshape(n_target, block_size)
    h_coarse = h_use.mean(axis=1)
    return h_coarse

def upsample_h(h_coarse, n_target):
    """Upsample h_coarse to n_target points via interpolation."""
    n_c = len(h_coarse)
    x_old = np.linspace(0, 1, n_c)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, h_coarse)
