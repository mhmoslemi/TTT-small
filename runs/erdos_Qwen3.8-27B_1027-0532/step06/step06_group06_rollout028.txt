def fractional_local_search(h, target, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 5000
    lr = 0.001  # Learning rate
    stall = 0
    max_stall = 200
    
    for it in range(100000):
        if time.time() - start_time > start_time + time_limit:
            break
        if stall > max_stall:
            break
        
        # Compute gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        shifted = beta * (corr - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for j in range(n):
            idx1 = j - np.arange(n) + n - 1
            idx2 = np.arange(n) - j + n - 1
            grad[j] = np.dot(1.0 - h, weights[idx1]) - np.dot(h, weights[idx2])
        grad *= dx
        
        # Projected gradient step
        h_new = h - lr * grad
        # Maintain sum constraint
        h_new += (target - np.sum(h_new)) / n
        # Project to [0,1]
        h_new = project_box_sum(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            h = h_new  # Still accept to allow exploration
            stall += 1
            # Reduce learning rate
            lr *= 0.999
    
    return best_h, best_c5

def coordinate_descent(h, target, time_limit, start_time):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    total_start = time.time()
    
    it = 0
    while time.time() - total_start < time_limit:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        k_star = int(np.argmax(corr))
        s_star = k_star - (n - 1)
        
        # Compute derivative of corr[k_star] w.r.t. each h[j]
        # d(corr[k_star])/dh[j] = (1-h[j+s_star])*[valid] - h[j-s_star]*[valid]
        deriv = np.zeros(n)
        valid1 = (np.arange(n) + s_star >= 0) & (np.arange(n) + s_star < n)
        valid2 = (np.arange(n) - s_star >= 0) & (np.arange(n) - s_star < n)
        deriv[valid1] = 1.0 - h[np.arange(n)[valid1] + s_star]
        deriv[valid2] -= h[np.arange(n)[valid2] - s_star]
        
        # We want to decrease h where deriv > 0 and increase where deriv < 0
        # But we must maintain sum(h) = target.
        # Also h must stay in [0,1].
        
        # Find the best single coordinate change:
        # If we increase h[j] by δ, corr[k_star] changes by deriv[j]*δ (approximately)
        # We want to minimize corr[k_star], so we want deriv[j]*δ < 0.
        # If deriv[j] > 0, decrease h[j] (δ < 0).
        # If deriv[j] < 0, increase h[j] (δ > 0).
        
        # But we also need to maintain sum(h) = target.
        # So we need to pair: increase one, decrease another.
        
        # Simple approach: find j_down (deriv > 0, h[j] > 0) and j_up (deriv < 0, h[j] < 1)
        # Move δ from j_down to j_up.
        
        # The change in corr[k_star] is: deriv[j_down]*(-δ) + deriv[j_up]*(+δ) = δ*(deriv[j_up] - deriv[j_down])
        # We want this to be negative, so we need deriv[j_up] < deriv[j_down].
        
        # The best pair minimizes (deriv[j_up] - deriv[j_down])
        # = min deriv among candidates for j_up - max deriv among candidates for j_down
        
        pos = np.where((deriv > 0.01) & (h > 0.01))[0]
        neg = np.where((deriv < -0.01) & (h < 0.99))[0]
        
        if len(pos) == 0 or len(neg) == 0:
            # Try a different approach or break
            break
        
        # For each neg candidate j_up, the improvement is deriv[j_up] - max(deriv[pos])
        # We want the most negative value of deriv[j_up] - max(deriv[pos])
        # = min(deriv[neg]) - max(deriv[pos])
        
        best_j_down = pos[np.argmax(deriv[pos])]
        best_j_up = neg[np.argmin(deriv[neg])]
        improve = deriv[best_j_up] - deriv[best_j_down]
        
        if improve >= 0:
            break
        
        # Determine step size: limited by h[j_down] > 0 and h[j_up] < 1
        # Also check that no other lag becomes the new max.
        
        # Try a few step sizes
        max_step = min(h[best_j_down], 1.0 - h[best_j_up])
        best_step = max_step
        best_new_c5 = float('inf')
        
        for frac in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            step = min(max_step, frac * max_step) if frac < 1 else max_step
            h_trial = h.copy()
            h_trial[best_j_down] -= step
            h_trial[best_j_up] += step
            # Check sum
            # (step - step = 0, so sum is preserved)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_new_c5:
                best_new_c5 = c5_trial
                best_step = step
        
        if best_new_c5 < best_c5 - 1e-12:
            h[best_j_down] -= best_step
            h[best_j_up] += best_step
            best_c5 = best_new_c5
            best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    while current_sum != target_sum:
        if current_sum > target_sum:
            ones = np.where(h_r == 1)[0]
            idx = ones[np.argmin(h[ones])]
            h_r[idx] = 0
            current_sum -= 1
        else:
            zeros = np.where(h_r == 0)[0]
            idx = zeros[np.argmax(h[zeros])]
            h_r[idx] = 1
            current_sum += 1
    return h_r.astype(float)

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(N)
    t_arr = np.arange(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(1.0 - h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def optimize_continuous(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for attempt in range(3):
        if time.time() - start_time > start_time + time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})
            h_cur = project_box_sum(res.x, target)
        except Exception:
            pass
    return h_cur

def steepest_ascent_polish(h, target, time_limit, start_time, beta=5000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    total_start = start_time
    stall = 0
    max_stall = 300
    for it in range(50000):
        if time.time() - total_start > time_limit:
            break
        if stall > max_stall:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        shifted = beta * (corr - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        t_arr = np.arange(n)
        grad = np.zeros(n)
        for j in range(n):
            idx1 = j - t_arr + n - 1
            idx2 = t_arr - j + n - 1
            grad[j] = np.dot(1.0 - h, weights[idx1]) - np.dot(h, weights[idx2])
        grad *= 2.0 / n
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        lr = min(0.1, 0.01 / gnorm)
        h_new = h - lr * grad
        h_new += (target - np.sum(h_new)) / n
        h_new = project_box_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            h = h_new
            stall += 1
    return best_h, best_c5

def coordinate_refine(h, target, time_limit, start_time):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(200000):
        if time.time() - start_time > start_time + time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = float(corr.max())
        top_k = 3
        top_indices = np.argsort(corr)[-top_k:]
        s_list = [int(k) - (n - 1) for k in top_indices]
        deriv = np.zeros(n)
        for s in s_list:
            ar = np.arange(n)
            v1 = (ar + s >= 0) & (ar + s < n)
            v2 = (ar - s >= 0) & (ar - s < n)
            deriv[v1] += 1.0 - h[ar[v1] + s]
            deriv[v2] -= h[ar[v2] - s]
        pos = np.where((deriv > 0.005) & (h > 0.005))[0]
        neg = np.where((deriv < -0.005) & (h < 0.995))[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        best_jd = pos[np.argmax(deriv[pos])]
        best_ju = neg[np.argmin(deriv[neg])]
        improve = deriv[best_ju] - deriv[best_jd]
        if improve >= -1e-8:
            break
        max_step = min(h[best_jd], 1.0 - h[best_ju])
        best_step = 0.0
        best_new_c5 = float('inf')
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = max_step * frac
            if step < 1e-8:
                continue
            h_trial = h.copy()
            h_trial[best_jd] -= step
            h_trial[best_ju] += step
            c5_t = compute_c5(h_trial)
            if c5_t < best_new_c5:
                best_new_c5 = c5_t
                best_step = step
        if best_step > 0 and best_new_c5 < best_c5 - 1e-12:
            h[best_jd] -= best_step
            h[best_ju] += best_step
            best_c5 = best_new_c5
            best_h = h.copy()
        else:
            break
    return best_h, best_c5

def sa_continuous(h_start, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    stall = 0
    max_stall = 3000
    for it in range(max_iters):
        if time.time() - start_time > start_time + time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c5_new = compute_c5(h_new)
        delta_c = c5_new - current_c5
        if delta_c < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c / max(t, 1e-30))):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5
