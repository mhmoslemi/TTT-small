If we change h[i] -> h[i] - δ and h[j] -> h[j] + δ:

  - h[i]*(1-h[i+k]) changes by -δ*(1-h[i+k])
  - h[i-k]*(1-h[i]) changes by +δ*h[i-k]
  So net change from position i for shift k: -δ*(1-h[i+k]) + δ*h[i-k] = -δ*g_k[i]

- Terms involving h[j]:

Total change for shift k: δ * (g_k[j] - g_k[i])

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

# Global variables for SLSQP
_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def precompute_f(h, n):
    """Precompute f[i, s] = change in correlation[s] when h[i] is increased by 1."""
    f = np.zeros((n, 2 * n - 1))
    mid = n - 1
    for i in range(n):
        # h[i] appears in corr[s] when:
        # As the "first" factor: corr[s] includes h[i]*(1-h[i+k]) where k = s-mid
        #   so i+k must be valid: 0 <= i+k < n
        # As the "second" factor: corr[s] includes h[i-k]*(1-h[i]) where k = s-mid
        #   so i-k must be valid: 0 <= i-k < n
        k = 0  # we'll compute for all s at once
        # For shift k = s - mid:
        # corr[s] = sum_j h[j] * (1-h[j+k]) for valid j
        # d corr[s] / d h[i] = (1-h[i+k]) if 0<=i+k<n, plus -h[i-k] if 0<=i-k<n
        # But this is for a specific k. For all s:
        # s = mid + k, so k = s - mid
        # d corr[s] / d h[i] = (1-h[i+s-mid]) * [0<=i+s-mid<n] - h[i-s+mid] * [0<=i-s+mid<n]
        
        # First term: i+s-mid in [0, n-1] => s in [mid-i, mid-i+n-1]
        s1_start = max(0, mid - i)
        s1_end = min(2*n-2, mid - i + n - 1)
        if s1_start <= s1_end:
            idx = np.arange(s1_start, s1_end + 1)
            h_idx = i + idx - mid
            f[i, s1_start:s1_end+1] += (1.0 - h[h_idx])
        
        # Second term: i-s+mid in [0, n-1] => s in [mid-(n-1-i), mid+i]
        s2_start = max(0, mid - (n - 1 - i))
        s2_end = min(2*n-2, mid + i)
        if s2_start <= s2_end:
            idx = np.arange(s2_start, s2_end + 1)
            h_idx = i - idx + mid
            f[i, s2_start:s2_end+1] -= h[h_idx]
    
    return f

def pair_cd_random(h_start, n, target, time_limit, max_sweeps, n_cand, rng, temperature=0.0):
    """Pair coordinate descent with randomization and optional simulated annealing."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get gradient for the worst shift
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        
        # Select candidates: mix of greedy and random
        sorted_idx = np.argsort(grad)
        n_rand = max(5, n_cand // 4)
        n_greedy = n_cand - n_rand
        greedy_low = sorted_idx[:n_greedy]
        greedy_high = sorted_idx[-n_greedy:]
        rand_idx = rng.choice(n, size=2*n_rand, replace=False)
        low_set = np.unique(np.concatenate([greedy_low, rand_idx[:n_rand]]))
        high_set = np.unique(np.concatenate([greedy_high, rand_idx[n_rand:]]))
        
        f_arr = precompute_f(h, n)
        best_trial = c5_val
        best_move = None
        
        # Only check a subset of pairs for speed
        max_pairs = 2000
        pairs_checked = 0
        for i in high_set:
            for j in low_set:
                if i == j or pairs_checked >= max_pairs:
                    continue
                pairs_checked += 1
                dcorr = f_arr[i] - f_arr[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                # Bisection to find optimal delta
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                
                # Simulated annealing: accept with probability exp(-delta/T)
                if trial_c5 < best_trial or (temperature > 0 and rng.rand() < np.exp(-(trial_c5 - best_trial) / temperature)):
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

            no_improve += 1

def adam_topk(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
              lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def precompute_f_fast(h, n):
    mid = n - 1
    f = np.zeros((n, 2 * n - 1))
    rev_h = h[::-1]

    for i in range(n):
        f[i, mid - i: 2 * mid - i + 1] += one_minus_h
        f[i, i: i + n] -= rev_h
    return f

    return np.ones(_G_n)

    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def pair_cd_multi(h_start, n, target, time_limit, max_sweeps, n_shifts, n_cand, rng):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_s_idx = np.argsort(corr)[-n_shifts:]
        top_max = np.max(corr[top_s_idx])
        weights = np.exp(100.0 * (corr[top_s_idx] - top_max))
        wsum = weights.sum()
        if wsum > 1e-300:
            weights /= wsum
        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_s_idx):
            k = int(s_idx) - mid
            grad += w * get_grad_shift(h, k, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        n_greedy = max(10, n_cand * 3 // 4)
        n_rand = n_cand - n_greedy
        greedy_low = sorted_idx[:n_greedy]
        greedy_high = sorted_idx[-n_greedy:]
        rand_idx = rng.choice(n, size=2 * n_rand, replace=False)
        low_set = np.unique(np.concatenate([greedy_low, rand_idx[:n_rand]]))
        high_set = np.unique(np.concatenate([greedy_high, rand_idx[n_rand:]]))
        f_arr = precompute_f_fast(h, n)
        best_trial = c5_val
        best_move = None
        max_pairs = 3000
        pairs_checked = 0
        for ii in range(len(high_set)):
            i = high_set[ii]
            for jj in range(len(low_set)):
                j = low_set[jj]
                if i == j or pairs_checked >= max_pairs:
                    continue
                pairs_checked += 1
                dcorr = f_arr[i] - f_arr[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def exact_max_subgrad(h_start, n, target, time_limit, max_steps, lr0):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 300:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        tol = max(1e-10, 1e-8 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= len(active)
        grad *= dx
        grad -= np.mean(grad)
        lr = lr0 / (1.0 + 0.005 * step)
        h_new = project_feasible(h - lr * grad, target)
        h = h_new
    return best_h, best_c5

def adam_topk(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
              lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 180:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
        h_pol, sc = slsqp_optimize(best_h, n, target, alphas, 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd, sc = pair_cd_multi(best_h, n, target,
                                 time_limit=min(100, tleft() * 0.3),
                                 max_sweeps=3000, n_shifts=5, n_cand=40, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 150:
        alphas = [20000, 100000, 500000, 2000000]
        h_pol, sc = slsqp_optimize(best_h, n, target, alphas, 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, sc = pair_cd_single(best_h, n, target,
                                   time_limit=min(80, tleft() * 0.3),
                                   max_sweeps=5000, n_cand=50, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 80:
        h_sg, sc = exact_max_subgrad(best_h, n, target,
                                      time_limit=min(40, tleft() * 0.3),
                                      max_steps=10000, lr0=0.01)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    if tleft() > 100:
        h_adam, sc = adam_topk(best_h, n, target,
                                time_limit=min(50, tleft() * 0.3),
                                max_steps=15000, n_top_start=2, n_top_end=15,
                                lr=0.015, beta1=0.8, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    if tleft() > 80:
        alphas = [100000, 500000, 2000000]
        h_pol, sc = slsqp_optimize(best_h, n, target, alphas, 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        h_cd, sc = pair_cd_multi(best_h, n, target,
                                 time_limit=min(60, tleft() * 0.35),
                                 max_sweeps=3000, n_shifts=3, n_cand=35, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    for noise_scale in [0.005, 0.01, 0.02, 0.03]:
        if tleft() < 70:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        alphas = [2000, 10000, 50000, 200000]
        h_ref, sc = slsqp_optimize(h_pert, n, target, alphas, 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 150:
        n_hires = 600
        target_hires = n_hires / 2.0
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        alphas = [500, 2000, 10000, 50000, 200000]
        h_hires, sc = slsqp_optimize(h_interp, n_hires, target_hires,
                                      alphas, 800, 1e-15)
        if sc < best_score:
            h_cd2, sc2 = pair_cd_multi(h_hires, n_hires, target_hires,
                                        time_limit=min(40, max(0, tleft() - 60)),
                                        max_sweeps=1500, n_shifts=5, n_cand=30, rng=rng)
            if sc2 < sc:
                sc = sc2
                h_hires = h_cd2
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_hires
                target = target_hires

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, sc = pair_cd_single(best_h, n_use, target_use,
                                   time_limit=min(30, tleft() * 0.4),
                                   max_sweeps=2000, n_cand=40, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas = [500000, 2000000, 10000000]
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                      alphas, 500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
