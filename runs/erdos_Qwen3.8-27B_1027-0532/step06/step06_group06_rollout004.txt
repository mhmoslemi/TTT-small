def compute_c5(h):

def c5_only(h):

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(np.int32)
    current_sum = int(h_r.sum())
    if current_sum != target_sum:
        ones = np.where(h_r == 1)[0]
        zeros = np.where(h_r == 0)[0]
        if current_sum > target_sum:
            need = current_sum - target_sum
            if need <= len(ones):
                order = np.argsort(h[ones])[:need]
                h_r[ones[order]] = 0
        else:
            need = target_sum - current_sum
            if need <= len(zeros):
                order = np.argsort(-h[zeros])[:need]
                h_r[zeros[order]] = 1
        final_sum = int(h_r.sum())
        while final_sum != target_sum:

                h_r[idx[0]] = 0
                final_sum -= 1
            else:
                idx = np.where(h_r == 0)[0]
                h_r[idx[0]] = 1
                final_sum += 1
    return h_r.astype(float)


def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val


def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        m1 = (idx1 >= 0) & (idx1 < 2 * N - 1)
        m2 = (idx2 >= 0) & (idx2 < 2 * N - 1)
        grad[j] = np.dot(one_minus_h, weights[idx1[m1]]) - np.dot(h, weights[idx2[m2]])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad


def lbfgs_run(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_objective(h_cur, beta, target, penalty)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = c5_only(h_candidate)
            sc_cur = c5_only(h_cur)
            if sc < sc_cur:
                h_cur = h_candidate
            elif curr_obj > smooth_objective(h_candidate, beta, target, penalty):
                h_cur = h_candidate
                curr_obj = smooth_objective(h_cur, beta, target, penalty)
        except Exception:
            pass
    return project_box_sum(h_cur, target)

        if r < 0.6:
            worst = np.argsort(corr)[-6:]
            k_star = worst[rng.randint(len(worst))] - (n - 1)
            ip = [x for x in ones if 0 <= x + k_star < n or 0 <= x - k_star < n]
            jp = [x for x in zeros if 0 <= x + k_star < n or 0 <= x - k_star < n]
            if len(ip) > 0 and len(jp) > 0:
                i = ip[rng.randint(len(ip))]
                j = jp[rng.randint(len(jp))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            i2 = ones[rng.randint(len(ones))]
            j2 = zeros[rng.randint(len(zeros))]
            if i2 == i or j2 == j:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            h[i2] = 0
            h[j2] = 1
            h_fixed = {}
            h_fixed[i2] = 0
            h_fixed[j2] = 1
            h_float = h.astype(float)
            for key, val in h_fixed.items():
                h_float[key] = val
            corr_new = np.correlate(h_float, 1.0 - h_float, mode='full')
            c5_new = float(corr_new.max()) * dx
            delta = c5_new - current_c5
            if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
                h[i] = 0
                h[j] = 1
                corr = corr_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i2] = 1
                h[j2] = 0
                stall += 1
            continue

        h[i] = 0
        h[j] = 1
        h_float = h.astype(float)
        corr_new = np.correlate(h_float, 1.0 - h_float, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1
            h[j] = 0
            stall += 1

    return best_h.astype(float), best_c5


def coordinate_descent(h_start, target, time_limit, start_time, n_passes=3):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    for p in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            for step in steps:
                for direction in [1.0, -1.0]:
                    h[i] = h[i] + direction * step
                    if h[i] < 0 or h[i] > 1:
                        h[i] = h[i] + direction * step
                    h[i] = min(1.0, max(0.0, h[i]))
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(new_corr.max()) * dx
                    if new_max < best_c5 - 1e-10:
                        corr = new_corr
                        best_c5 = new_max
                    else:
                        h[i] = h[i] - direction * step
                        h[i] = min(1.0, max(0.0, h[i]))
                    if h[i] < 0 or h[i] > 1:
                        h[i] = min(1.0, max(0.0, h[i]))
    return project_box_sum(h, target), best_c5


def interpolate_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = c5_only(best_h)

    h_bin_init = project_binary(h_init, target_int)
    c5_bin_init = c5_only(h_bin_init)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()

    # Phase 1: Continuous L-BFGS-B with increasing beta
    h_cont = best_h.copy()
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        if tleft() < 80:
            break
        time_alloc = min(50, tleft() * 0.1)
        h_opt = lbfgs_run(h_cont, beta, target, penalty=800.0,
                          maxiter=200, time_limit=time_alloc, start_time=start_time)
        sc = c5_only(h_opt)
        h_cont = h_opt
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

    # Phase 2: Coordinate descent polish
    if tleft() > 60:
        h_cd, c5_cd = coordinate_descent(best_h.copy(), target,
                                          time_limit=min(80, tleft() * 0.15),
                                          start_time=start_time, n_passes=2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Multi-shift greedy binary search
    if tleft() > 100:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(180, tleft() * 0.25)
        h_ls, c5_ls = multi_shift_greedy(h_bin, target_int, time_limit=time_alloc,
                                          start_time=start_time, top_k=5, scan_width=30)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: SA from best binary
    if tleft() > 100:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(200, tleft() * 0.3)
        max_it = int(min(300000, time_alloc * 1500))
        h_sa, c5_sa = sa_binary(h_bin, target_int, max_it, 0.005, 0.00001,
                                 rng, time_limit=time_alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Continuous refinement
    if tleft() > 60:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [200, 500, 1500, 3000]:
            if tleft() < 30:
                break
            time_alloc = min(35, tleft() * 0.15)
            h_opt = lbfgs_run(h_cont, beta, target, penalty=2000.0,
                              maxiter=200, time_limit=time_alloc, start_time=start_time)
            h_cont = h_opt
            sc = c5_only(h_cont)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont.copy()

    # Phase 6: More binary search with different parameters
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.2)
        h_ls2, c5_ls2 = multi_shift_greedy(h_bin, target_int, time_limit=time_alloc,
                                            start_time=start_time, top_k=8, scan_width=20)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 7: Try different grid sizes
    if tleft() > 60:
        n_options = [n + 2, n - 2, n + 4, n + 6, n + 8, n - 4]
        n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]
        for n_try in n_options[:4]:
            if tleft() < 40:
                break
            target_try = float(n_try / 2)
            target_try_int = int(n_try / 2)
            h_interp = interpolate_to(best_h, n_try, target_try)
            time_alloc = min(25, tleft() * 0.12)
            h_opt = lbfgs_run(h_interp, 100, target_try, penalty=500.0,
                              maxiter=150, time_limit=time_alloc, start_time=start_time)
            sc = c5_only(h_opt)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()
            if tleft() < 25:
                continue
            h_bin_try = project_binary(h_opt, target_try_int)
            time_alloc2 = min(20, tleft() * 0.1)
            h_ls3, c5_ls3 = multi_shift_greedy(h_bin_try, target_try_int,
                                                time_limit=time_alloc2, start_time=start_time,
                                                top_k=5, scan_width=20)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()

    # Phase 8: Final SA polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(50, tleft() - 5)
        max_it = int(min(80000, time_alloc * 2000))
        h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.003, 0.000005,
                                 rng, time_limit=time_alloc, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 9: Final continuous polish with high beta
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_final = project_box_sum(best_h.copy(), target_use)
        for beta_f in [1000, 3000, 8000]:
            if tleft() < 8:
                break
            time_alloc = min(15, tleft() * 0.4)
            h_polished = lbfgs_run(h_final, beta_f, target_use,
                                    penalty=5000.0, maxiter=100,
                                    time_limit=time_alloc, start_time=start_time)
            h_final = h_polished
            sc = c5_only(h_polished)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_polished.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = c5_only(best_h)

    return best_h, final_c5, n_final
