```python
def project_feasible_v2(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 ≤ x ≤ 1} using bisection on lambda."""
    def residual(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    
    if residual(0.0) <= 0:
        # sum(h) <= target, need to increase h
        # Find lam < 0 such that sum(clip(h-lam, 0, 1)) = target
        # clip(h-lam, 0, 1) = h - lam for h-lam in [0,1], 0 for h-lam < 0, 1 for h-lam > 1
        # If lam < 0: h-lam > h, so more elements might hit 1
        lo, hi = -1.0, 0.0
    else:

        lo, hi = 0.0, 1.0

        r = residual(mid)
        if abs(r) < 1e-12:
            break
        if r > 0:
            lo = mid
        else:
            hi = mid
    
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)
```

Wait, this isn't quite right. The projection onto {x : sum(x) = t, 0 ≤ x ≤ 1} is:

We need to find λ such that sum(clip(h_i - λ, 0, 1)) = t.

The function f(λ) = sum(clip(h_i - λ, 0, 1)) is:
- Decreasing in λ
- f(-∞) = n (all 1s)
- f(+∞) = 0
- Continuous and piecewise linear

1. Find the top-k shifts (those with the highest c_k values)

```python
def active_set_search(h_start, n, target, time_limit, k_active=5, lr_init=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    lr = lr_init
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c5_now = float(np.max(c))
        
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
            lr = max(lr_init, lr * 0.95)  # slightly reduce lr after improvement
        else:
            no_improve += 1
            if no_improve > 500:
                break
            lr *= 0.99
        
        # Top-k active shifts
        top_idx = np.argsort(c)[-k_active:]
        
        # Average gradient
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= k_active
        grad *= dx
        grad -= np.mean(grad)  # Remove mean component (invariance to constant shifts)
        
        # Line search
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr * lr_mult
            h_try = project_feasible_v2(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    
    return best_h, best_c5
```

Now, let me also think about a **more sophisticated local search**. The pairwise CD in the previous code is O(n²) per sweep which is expensive. Instead, I can use a **block Gauss-Seidel** approach:

For each coordinate i (or block of coordinates), optimize h[i] (or h[block]) given all other coordinates fixed. This is a 1D (or low-dimensional) optimization problem.

For a single coordinate i, the objective is:
f(h_i) = max_k c_k(h)

where c_k depends on h_i through at most 2 terms (the terms where i is one of the two indices). So f is the max of a set of piecewise-linear functions of h_i (each c_k is linear in h_i). We can find the minimum of f over h_i ∈ [0,1] by checking the breakpoints.

Actually, c_k as a function of h_i is:
c_k = (1-h_{i-k})*h_i + h_i*(1-h_{i+k}) + (terms not involving h_i)
     = h_i*(2 - h_{i-k} - h_{i+k}) + const

So it's linear in h_i! The slope is (2 - h_{i-k} - h_{i+k}).

The maximum of linear functions is convex (piecewise linear, convex). We want to minimize this over h_i ∈ [0,1] while maintaining sum(h) = target.

If we fix the constraint by adjusting one other coordinate, we have a 2D problem. But for simplicity, let's just do the 1D optimization and then re-project.

```python
def get_grad_shift(h, k, n):
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    if np.any(v1):
        g[v1] += (1.0 - h[idxs[v1] - k])
    if np.any(v2):
        g[v2] -= h[idxs[v2] + k]
    return g

def topk_weighted_grad(h, n, k_active, corr=None):
    dx = 2.0 / n
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    c = corr * dx
    top_idx = np.argsort(c)[-k_active:]
    grad = np.zeros(n)
    for s_idx in top_idx:
        k = int(s_idx) - mid
        grad += get_grad_shift(h, k, n)
    grad /= k_active
    grad *= dx
    grad -= np.mean(grad)
    return grad, c, top_idx

    best_h = h_cur.copy()

    mid = n - 1

        shifted = np.clip(alpha * (c - c_max), -500.0, 500.0)
        exp_shifted = np.exp(shifted)
        log_sum_exp = c_max + np.log(np.sum(exp_shifted)) / alpha
```

Hmm wait, I realize there might be an issue with the L-BFGS-B approach. The projection after each call to minimize() only happens once at the end, not during the optimization. L-BFGS-B will try to maintain the sum(h) ≈ n/2 through the box constraints, but it doesn't explicitly enforce the equality constraint.

A better approach: use an augmented Lagrangian. Define:
f_μ(h) = f_smooth(h) + μ/2 * (sum(h) - target)²

The gradient is:
∇f_μ(h) = ∇f_smooth(h) + μ * (sum(h) - target) * 1

Then use L-BFGS-B on f_μ with just box constraints. Update μ periodically.

```python
def lbfgsb_smooth_auglag(h_start, n, target, alphas, maxiter_per_alpha, time_limit, 
                          mu_init=100.0, mu_rate=1.5):

    best_h = h_cur.copy()

    mu = mu_init

            c = corr * dx
            c_max = np.max(c)
            shifted = np.clip(alpha * (c - c_max), -500.0, 500.0)
            val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            penalty = 0.5 * mu * (np.sum(h) - target) ** 2
```

```python
def project_feasible_v2(h, target_sum):
    h = np.asarray(h, dtype=float)
    def f(lam):
        return float(np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum)
    lo = -1.0
    hi = 1.0
    if f(lo) < 0:
        lo = -2.0

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    if np.any(v1):

    c = corr * dx

    shifted = np.clip(alpha * (c - c_max), -500.0, 500.0)

        k = int(s_idx) - mid
        grad += weights[s_idx] * get_grad_shift(h, k, n)
    grad *= dx
    return val, grad

            g = g + mu * (float(np.sum(h)) - target)
            return g
```

```python
                                   'gtol': 1e-12, 'maxfun': maxiter_per_alpha * 3})
            h_new = project_feasible_v2(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
            mu *= mu_rate
        except Exception:
            pass
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    mid = n - 1
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj_fn(h):
            v, _ = smooth_obj_grad(h, alpha, dx, n, mid)
            return v
        def grad_fn(h):
            _, g = smooth_obj_grad(h, alpha, dx, n, mid)
            return g
        def cons_fun(h):
            return float(np.sum(h)) - target
        def cons_jac(h):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_fn, h_cur, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible_v2(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def subgrad_polish(h_start, n, target, time_limit, lr_init=0.003):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    lr = lr_init
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c5_now = float(np.max(c))
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 4000:
                break
        c_max = float(np.max(c))
        active = np.where(c >= c_max - 1e-8)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 10:
            active = np.argsort(c)[-10:]
        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= float(len(active))
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = lr * lr_mult
            h_try = project_feasible_v2(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            lr *= 0.97
            no_improve += 2
    return best_h, best_c5

def adam_topk(h_start, n, target, time_limit, n_top_start, n_top_end,
              lr_init, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0

    for step in range(50000):
        if time.time() - start > time_limit:
            break
        if step > 500 and no_improve > 2000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c5_now = float(np.max(c))
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, 15000), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(c)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= float(n_top)
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_init / (1.0 + 0.008 * step)
        h_new = project_feasible_v2(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        h = h_new
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible_v2(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible_v2(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 15.0 - elapsed()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible_v2(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 350:
        h1, s1 = lbfgsb_auglag(best_h, n, target,
                                alphas=[30, 100, 300, 1000, 3000, 10000, 30000],
                                maxiter_per_alpha=300,
                                time_limit=min(250, tleft() * 0.35),
                                mu_init=50.0, mu_rate=1.5)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    if tleft() > 300:
        h2, s2 = slsqp_refine(best_h, n, target,
                               alphas=[3000, 10000, 50000, 200000],
                               maxiter_per_alpha=1500,
                               time_limit=min(200, tleft() * 0.3))
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 200:
        h3, s3 = adam_topk(best_h, n, target,
                            time_limit=min(100, tleft() * 0.25),
                            n_top_start=3, n_top_end=12,
                            lr_init=0.012, beta1=0.88, beta2=0.999, eps=1e-8)
        if s3 < best_score:
            best_score = s3
            best_h = h3
        if tleft() > 100:
            h3b, s3b = slsqp_refine(best_h, n, target,
                                     alphas=[10000, 50000, 200000, 1000000],
                                     maxiter_per_alpha=1000,
                                     time_limit=min(80, tleft() * 0.3))
            if s3b < best_score:
                best_score = s3b
                best_h = h3b

    if tleft() > 80:
        h4, s4 = subgrad_polish(best_h, n, target,
                                 time_limit=min(60, tleft() * 0.3),
                                 lr_init=0.003)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    rng = np.random.RandomState(seed)
    if tleft() > 150:
        for trial in range(4):
            if tleft() < 40:
                break
            noise_scale = 0.03 + 0.02 * trial
            h_pert = project_feasible_v2(best_h + rng.randn(n) * noise_scale, target)
            h5, s5 = lbfgsb_auglag(h_pert, n, target,
                                    alphas=[100, 1000, 10000, 50000],
                                    maxiter_per_alpha=200,
                                    time_limit=min(40, tleft() * 0.2))
            if s5 < best_score:
                best_score = s5
                best_h = h5
            if tleft() > 30:
                h5b, s5b = slsqp_refine(best_h, n, target,
                                         alphas=[10000, 100000],
                                         maxiter_per_alpha=800,
                                         time_limit=min(30, tleft() * 0.3))
                if s5b < best_score:
                    best_score = s5b
                    best_h = h5b

    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h_up = interpolate_to(best_h, n2, target2)
        h6, s6 = lbfgsb_auglag(h_up, n2, target2,
                                alphas=[30, 100, 500, 2000, 8000],
                                maxiter_per_alpha=200,
                                time_limit=min(120, tleft() * 0.4))
        if tleft() > 80:
            h6, s6 = slsqp_refine(h6, n2, target2,
                                   alphas=[5000, 30000, 100000],
                                   maxiter_per_alpha=1000,
                                   time_limit=min(70, tleft() * 0.4))
        if s6 < best_score:
            best_score = s6
            best_h = h6
            n = n2
            target = target2

    if tleft() > 30:
        nf = len(best_h)
        tf = nf / 2.0
        hf, sf = slsqp_refine(best_h, nf, tf,
                               alphas=[20000, 100000, 500000, 2000000],
                               maxiter_per_alpha=1000,
                               time_limit=max(15, tleft() * 0.6))
        if sf < best_score:
            best_score = sf
            best_h = hf

    if tleft() > 10:
        nf = len(best_h)
        tf = nf / 2.0
        hf, sf = subgrad_polish(best_h, nf, tf,
                                 time_limit=max(5, tleft() * 0.5),
                                 lr_init=0.001)
        if sf < best_score:
            best_score = sf
            best_h = hf

    best_h = project_feasible_v2(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
```
