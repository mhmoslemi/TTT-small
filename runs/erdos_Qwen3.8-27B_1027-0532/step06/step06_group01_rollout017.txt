        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_ci = -1
        best_cj = -1

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_map[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_map[int(j)]
                g = profiles[cj] - profiles[ci]

                if np.any(active_mask):
                    if np.min(g[active_mask]) >= -1e-14:
                        continue

                lo = 0.0
                hi_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (lo + hi_d)
                    new_vals = corr + mid_d * g
                    f_mid = float(np.max(new_vals))
                    if f_mid < cur_max - 1e-14:
                        active_pts = new_vals >= (f_mid - 1e-12)
                        if np.any(active_pts) and float(np.max(g[active_pts])) > 1e-14:
                            hi_d = mid_d
                        else:
                            lo = mid_d
                    else:
                        hi_d = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
                    best_ci = ci
                    best_cj = cj

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            corr += best_delta * (profiles[best_cj] - profiles[best_ci])
            total_moves += 1
            if total_moves % recompute_every == 0:
                s = target - np.sum(h)
                if abs(s) > 1e-8:
                    h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            if no_improve_count > 30:
                break

    h = project_feasible(h, target)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return best_h, best_c5

def multi_shift_subgrad(h_init, n, target, time_limit, max_steps=50000, n_shifts=3, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    stalling = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stalling = 0
        else:
            stalling += 1
            if stalling > 300:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[idxs[v1]] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[idxs[v2]] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 200:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 300:
        cd_time = min(450, tleft() * 0.4)
        hcd, c5cd = fast_cd_v2(best_h, n, target, time_limit=cd_time, n_cand=25, recompute_every=10, multi_shift_weight=0.2)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 80000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 200:
        cd_time = min(300, tleft() * 0.35)
        hcd, c5cd = fast_cd_v2(best_h, n, target, time_limit=cd_time, n_cand=35, recompute_every=8, multi_shift_weight=0.4)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    perturb_attempts = 3
    for pa in range(perturb_attempts):
        if tleft() < 150:
            break
        perturbation_scale = 0.02 * (1 - pa * 0.3)
        h_pert = best_h + rng.standard_normal(n) * perturbation_scale
        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)

        hp2, sc2 = slsqp_round(h_pert, n, target, [500, 2000, 10000], 1000, 1e-14)
        if sc2 < sc_pert:
            sc_pert = sc2
            h_pert = hp2

        if tleft() > 100 and sc_pert < best_score:
            cd_t = min(80, tleft() * 0.3)
            hcd2, c5cd2 = fast_cd_v2(h_pert, n, target, time_limit=cd_t, n_cand=20, recompute_every=10)
            if c5cd2 < sc_pert:
                sc_pert = c5cd2
                h_pert = hcd2

        if sc_pert < best_score:
            best_score = sc_pert
            best_h = h_pert

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [20000, 80000, 200000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        cd_time = min(100, tleft() * 0.4)
        hcd, c5cd = fast_cd_v2(best_h, n, target, time_limit=cd_time, n_cand=30, recompute_every=8, multi_shift_weight=0.3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 50:
        hsg, c5sg = multi_shift_subgrad(best_h, n, target, time_limit=min(35, tleft() * 0.3),
                                         max_steps=30000, n_shifts=4, base_lr=0.01)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 40:
        hp, sc = slsqp_round(best_h, n, target, [50000, 200000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        hcd, c5cd = fast_cd_v2(best_h, n, target, time_limit=min(25, tleft() * 0.5), n_cand=20, recompute_every=6)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
