def slsqq_optimize(h_start, n, target, alphas, maxiter=2000):

            cmax = np.max(corr)
            shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
            return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)
        
        def obj_grad(h):

            cmax = np.max(corr)
            weights = np.exp(np.clip(alpha * (corr - cmax), -500, 0))
            weights /= weights.sum()
            # Gradient of each overlap_k with respect to h
            n_half = n - 1
            g = np.zeros(n)
            for idx in range(len(corr)):
                w = weights[idx]
                if w < 1e-15:

                    g += w * (1.0 - 2.0 * h)
                elif k > 0:

                    g[k:] += w * (-h[:n-k])
                else:
                    m = -k
                    g[m:] += w * (1.0 - h[:n-m])
                    g[:n-m] += w * (-h[m:])
            return g * (2.0/n)
        
        def grad(h):
            return obj_grad(h)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                 'jac': lambda h: np.ones(n)}]
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_exact(res.x, target)
            if compute_c5(h_new) <= compute_c5(h_cur) + 1e-12:
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

        if c5_f2 < best_score:
            best_score = c5_f2
            best_h = h_final2

    h_final = project_exact(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
