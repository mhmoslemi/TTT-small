    """
    Compute gradient of log-sum-exp smoothing of max overlap.
    Returns (smoothed_value, gradient)
    """
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    
    # For numerical stability, subtract max
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    log_sum_exp = max_ov + np.log(np.sum(np.exp(shifted))) / alpha
    
    # Soft-max weights
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    
    # If n_top is specified, only use top n_top in gradient
    if n_top is not None and n_top < len(overlaps):
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        weights_restricted = np.zeros(len(overlaps))
        weights_restricted[top_idx] = weights[top_idx]
        weights_restricted /= weights_restricted.sum()
        weights = weights_restricted
    
    # Compute gradient
    g = np.zeros(n)
    n_half = n - 1
    for idx in np.where(weights > 1e-10)[0]:
        k = idx - n_half
        w = weights[idx]
        if abs_k == 0:
            g += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n-k] += w * (1.0 - h[k:]) * dx
            g[k:] += w * (-h[:n-k]) * dx
        else:
            m = -k
            g[m:] += w * (1.0 - h[:n-m]) * dx
            g[:n-m] += w * (-h[m:]) * dx
    
    return float(log_sum_exp), g

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n

        top_idx = np.argsort(overlaps)[::-1][:n_top]

        mask[top_idx] = True
        weights = weights * mask
        w_sum = weights.sum()
        if w_sum > 1e-15:
            weights /= w_sum
        else:

        max_iter_alpha = max_iter // len(alpha_schedule)

def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=200):
    import cvxpy as cp
    start = time.time()

        lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0

            if abs_k == 0:
                coeff = (1.0 - h) * dx
                constraints.append(coeff @ h_var <= t_var)
            elif int(k) > 0:

                coeff[:n - abs_k] = (1.0 - h[abs_k:]) * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                coeff = np.zeros(n)
                coeff[abs_k:] = (1.0 - h[:n - abs_k]) * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + 1e-4 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        damping = 0.85 + 0.1 * min(1.0, it / 50.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:

                                  max_sweeps=80)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3

    # Phase 4: SLSQP polish
    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        h4 = slsqp_optimize(best_h, n, target,
                             [50, 200, 800, 3000, 10000, 40000, 150000,
                              500000, 1500000, 5000000],
                             maxiter=3000, ftol=1e-15)
        sc = compute_c5(h4)
        if sc < best_score:
            best_score = sc
            best_h = h4

    # Phase 5: Random perturbation + SGD
    for attempt in range(3):
        tl = time_remaining(start_time, budget_s)
        if tl < 100:
            break
        h_pert = random_perturb(best_h, n, target, rng, scale=0.05)
        h5, c5_5 = smoothed_pgdescent(h_pert, n, target,
                                       [10, 50, 200, 800, 3000, 10000],
                                       time_limit=min(80, tl * 0.15),
                                       max_iter_total=50000)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5
            h5b = slsqp_optimize(h5, n, target,
                                  [500, 2000, 10000, 50000, 200000],
                                  maxiter=2000, ftol=1e-15)
            sc5 = compute_c5(h5b)
            if sc5 < best_score:
                best_score = sc5
                best_h = h5b

    # Phase 6: Alternating LP
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        h6, c5_6 = alternating_lp(best_h, n, target,
                                   max_iter=2000, time_limit=min(180, tl * 0.3),
                                   n_top=250)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6

    # Phase 7: Final SLSQP with very high alpha
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        h7 = slsqp_optimize(best_h, n, target,
                             [1000, 5000, 20000, 100000, 500000, 2000000,
                              10000000, 50000000],
                             maxiter=4000, ftol=1e-16)
        sc7 = compute_c5(h7)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Another coordinate descent
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        h8, c5_8 = coord_descent(best_h, n, target,
                                  time_limit=min(50, tl * 0.5),
                                  max_sweeps=40)
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8

    # Phase 9: Final SLSQP micro-polish
    tl = time_remaining(start_time, budget_s)
    if tl > 40:
        h9 = slsqp_optimize(best_h, n, target,
                             [10000, 100000, 1000000, 10000000],
                             maxiter=5000, ftol=1e-16)
        sc9 = compute_c5(h9)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
