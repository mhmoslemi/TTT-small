```python
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]

            h_r[idx] = 0
```

```python
def focused_subgradient(h, target, n_iter, alpha_decay, start_time, time_limit):
    n = len(h)
    k_max = n - 1
    corr = np.correlate(h, 1.0-h, mode='full')
    f_vals = corr * (2.0/n)  # actual overlap values
    best_c5 = float(np.max(f_vals))
    best_h = h.copy()
    
    alpha = 0.1
    for it in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Find weighted average of top shifts
        top_k = 10
        sorted_idx = np.argsort(f_vals)[-top_k:]
        weights = np.zeros(len(f_vals))
        max_val = f_vals[sorted_idx[-1]]
        for idx in sorted_idx:
            w = np.exp(-5 * (max_val - f_vals[idx]))
            weights[idx] = w
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for m_idx in range(len(f_vals)):
            if weights[m_idx] == 0:
                continue
            k = m_idx - (n-1)
            # grad[i] += weights[m_idx] * ((1-h[i+k]) - h[i-k])
            for i in range(n):
                g = 0.0
                if 0 <= i+k < n:
                    g += 1-h[i+k]
                if 0 <= i-k < n:
                    g -= h[i-k]
                grad[i] += weights[m_idx] * g
        
        # Projected gradient step
        h_new = h - alpha * grad
        h_new = project_box_sum(h_new, target)
        
        corr_new = np.correlate(h_new, 1.0-h_new, mode='full')
        f_vals_new = corr_new * (2.0/n)
        c5_new = float(np.max(f_vals_new))
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        f_vals = f_vals_new
        alpha *= alpha_decay
    
    return best_h, best_c5
```
