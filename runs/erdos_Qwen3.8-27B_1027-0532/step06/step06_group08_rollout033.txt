    k = np.arange(2*n-1) - (n-1)

    # delta_i[k] = [h[i-k] if in range else 0] - [(1-h[i+k]) if in range else 0]

    delta_i[mask_m] += h[im[mask_m]]

    delta_i[mask_p] -= (1.0 - h[ip[mask_p]])

    delta_j[mask mj] -= h[jm[mask mj]]
    
    return delta_i + delta_j

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

    n = len(y)
    y = np.clip(y, 0.0, 1.0).copy()
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:

        remaining = diff

            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0

                remaining -= can_inc
                y[idx] = 1.0
    return y

    mmj = (jm >= 0) & (jm < n)
    delta[mpj] += (1.0 - h[jp[mpj]])
    delta[mmj] -= h[jm[mmj]]
    return delta

def ga_crossover(p1, p2, rng):
    n = len(p1)
    child = p1.copy()
    n1 = int(p1.sum())
    n2 = int(p2.sum())
    target = n // 2
    if n1 == target:
        return child
    diff = n1 - target
    flip_count = 0
    if diff > 0:
        ones = np.where(p1 == 1)[0]
        zeros_in_p2 = np.where(p2 == 0)[0]
        if len(ones) >= diff and len(zeros_in_p2) >= diff:
            to_flip = rng.choice(ones, size=diff, replace=False)
            child[to_flip] = 0
    else:
        zeros = np.where(p1 == 0)[0]
        ones_in_p2 = np.where(p2 == 1)[0]
        need = -diff
        if len(zeros) >= need and len(ones_in_p2) >= need:
            to_flip = rng.choice(zeros, size=need, replace=False)
            child[to_flip] = 1
    current = int(child.sum())
    if current != target:
        if current > target:
            ones = np.where(child == 1)[0]
            for idx in rng.choice(ones, size=current - target, replace=False):
                child[idx] = 0
        else:
            zeros = np.where(child == 0)[0]
            for idx in rng.choice(zeros, size=target - current, replace=False):
                child[idx] = 1
    return child

def ga_mutation(h, k, rng):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) < k or len(zeros) < k:
        return h
    flip_ones = rng.choice(ones, size=k, replace=False)
    flip_zeros = rng.choice(zeros, size=k, replace=False)
    h[flip_ones] = 0
    h[flip_zeros] = 1
    return h

def genetic_algorithm(h_init, pop_size, n_generations, rng, time_limit, start_time,
                      cx_rate=0.8, mut_rate=0.3, elite_count=3, tour_size=5):
    n = len(h_init)
    target = n // 2
    population = [project_binary(h_init, target)]
    while len(population) < pop_size:
        rnd = np.zeros(n)
        flip_positions = rng.choice(n, size=target, replace=False)
        rnd[flip_positions] = 1
        population.append(rnd)

    fitness = [compute_c5(ind) for ind in population]
    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    for gen in range(n_generations):
        if time.time() - start_time > time_limit:
            break

        order = np.argsort(fitness)
        new_pop = [population[order[i]].copy() for i in range(elite_count)]
        new_fit = [fitness[order[i]] for i in range(elite_count)]

        while len(new_pop) < pop_size:
            p1_idx = _tournament_select(fitness, order, rng, tour_size)
            p2_idx = _tournament_select(fitness, order, rng, tour_size)
            if rng.random() < cx_rate:
                child = ga_crossover(population[p1_idx], population[p2_idx], rng)
            else:
                child = population[p1_idx].copy()
            if rng.random() < mut_rate:
                mk = 1 if rng.random() < 0.6 else 2
                try:
                    child = ga_mutation(child, mk, rng)
                except Exception:
                    pass
            new_pop.append(child)
            new_fit.append(1e9)

        for i in range(elite_count, pop_size):
            new_fit[i] = compute_c5(new_pop[i])

        population = new_pop
        fitness = new_fit

        if fitness[0] < best_f:
            best_f = fitness[0]
            best_h = population[0].copy()

    return best_h, best_f

def _tournament_select(fitness, order, rng, tour_size):
    candidates = rng.randint(0, len(fitness), size=min(tour_size, len(fitness)))
    best_c = candidates[0]
    best_fv = fitness[candidates[0]]
    for c in candidates[1:]:
        if fitness[c] < best_fv:
            best_fv = fitness[c]
            best_c = c
    return best_c

            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += (-h[abs_s:])
        grad /= topk

        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_box_sum(h_new, target)
        h_prev = h
        h = h_new

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def softmax_pg(h, n_iters, alpha, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 4000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** max(ratio, 1e-10)

        max_corr = float(corr.max())
        shifted = np.clip(alpha * (corr - max_corr), -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()

        topk_idx = np.argsort(W)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            wk = W[k]
            if wk < 1e-10:
                continue
            s = int(shifts[k])
            if s >= 0:
                grad[:n - s] -= wk * h[s:]
                grad[s:] -= wk * h[:n - s]
            else:
                abs_s = -s
                grad[abs_s:] -= wk * h[:n - abs_s]
                grad[:n - abs_s] -= wk * h[abs_s:]

        h = h + lr * grad
        h = project_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    best_cont_h = project_box_sum(h_init, float(n_init) / 2.0)
    best_cont_score = compute_c5(best_cont_h)

    allowed = min(80.0, tleft() * 0.08)
    if allowed > 10:
        h_g, c5_g = greedy_swap(best_h, max_iters=50000, rng=rng,
                                time_limit=allowed, start_time=start_time, scan=25)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    allowed = min(120.0, tleft() * 0.10)
    if allowed > 20:
        n_sa = int(min(5000000, allowed * 40000))
        h_sa, c5_sa = sa_binary(best_h, n_sa, 0.03, 0.00003, rng,
                                time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    allowed = min(300.0, tleft() * 0.20)
    if allowed > 30:
        n_iters_ms = int(min(50000, allowed * 80))
        h_ms, c5_ms = multi_swap_local(best_h, n_iters_ms, rng,
                                       time_limit=allowed, start_time=start_time,
                                       kswap=2, scan=6)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    allowed = min(300.0, tleft() * 0.18)
    if allowed > 50:
        n_gens = int(min(200, allowed * 0.5))
        h_ga, c5_ga = genetic_algorithm(best_h, pop_size=60, n_generations=n_gens,
                                        rng=rng, time_limit=allowed, start_time=start_time,
                                        cx_rate=0.7, mut_rate=0.4, elite_count=4, tour_size=5)
        if c5_ga < best_score:
            best_score = c5_ga
            best_h = h_ga.copy()

    allowed = min(150.0, tleft() * 0.12)
    if allowed > 30:
        n_iters = int(min(2000000, allowed * 20000))
        h_nas, c5_nas = nesterov_pg(best_h, n_iters, topk=5, lr_init=0.04, lr_end=0.001,
                                    time_limit=allowed, start_time=start_time)
        if c5_nas < best_cont_score:
            best_cont_score = c5_nas
            best_cont_h = h_nas.copy()

    allowed = min(120.0, tleft() * 0.10)
    if allowed > 20:
        n_iters2 = int(min(1000000, allowed * 10000))
        h_sm, c5_sm = softmax_pg(best_cont_h, n_iters2, alpha=80.0, topk=10,
                                  lr_init=0.02, lr_end=0.0003,
                                  time_limit=allowed, start_time=start_time)
        if c5_sm < best_cont_score:
            best_cont_score = c5_sm
            best_cont_h = h_sm.copy()

    allowed = min(tleft() - 30, 120.0)
    if allowed > 30:
        h_ref, sc_ref = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                      alphas=[50, 200, 1000],
                                      maxiter=2000, ftol=1e-14,
                                      time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    for iteration in range(4):
        if tleft() < 100:
            break

        allowed = min(50.0, tleft() * 0.10)
        if allowed > 15:
            n_pert = rng.randint(4, 10)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]

            n_sa2 = int(min(2000000, allowed * 30000))
            h_sa2, c5_sa2 = sa_binary(h_pert, n_sa2, 0.02, 0.00002, rng,
                                      time_limit=allowed, start_time=start_time)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

        if tleft() < 80:
            break

        allowed = min(60.0, tleft() * 0.12)
        if allowed > 20:
            n_iters3 = int(min(1000000, allowed * 15000))
            h_pg2, c5_pg2 = nesterov_pg(best_h, n_iters3, topk=4,
                                        lr_init=0.03, lr_end=0.0005,
                                        time_limit=allowed, start_time=start_time)
            if c5_pg2 < best_cont_score:
                best_cont_score = c5_pg2
                best_cont_h = h_pg2.copy()

        if tleft() < 60:
            break

        allowed = min(40.0, tleft() * 0.08)
        if allowed > 15:
            h_ref2, sc_ref2 = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                           alphas=[200, 1000],
                                           maxiter=1500, ftol=1e-14,
                                           time_limit=allowed, start_time=start_time)
            if sc_ref2 < best_cont_score:
                best_cont_score = sc_ref2
                best_cont_h = h_ref2.copy()

    for n_try in [n_init + 4, n_init + 8, n_init - 4]:
        if tleft() < 80:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_cont_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(30.0, tleft() * 0.05)
        if allowed < 10:
            continue
        h_g3, c5_g3 = greedy_swap(h_bin_try, max_iters=20000, rng=rng,
                                  time_limit=allowed * 0.4, start_time=start_time, scan=20)
        allowed2 = min(40.0, tleft() * 0.06)
        if allowed2 < 15:
            continue
        n_iters4 = int(min(500000, allowed2 * 10000))
        h_pg3, c5_pg3 = nesterov_pg(h_g3, n_iters4, topk=4,
                                    lr_init=0.03, lr_end=0.0003,
                                    time_limit=allowed2, start_time=start_time)
        if c5_pg3 < best_cont_score:
            best_cont_score = c5_pg3
            best_cont_h = h_pg3.copy()

    if tleft() > 40:
        n_use = len(best_cont_h)
        h_ref3, sc_ref3 = slsqp_refine(best_cont_h, n_use, float(n_use) / 2.0,
                                       alphas=[500, 2000],
                                       maxiter=3000, ftol=1e-15,
                                       time_limit=max(10.0, tleft() - 15),
                                       start_time=start_time)
        if sc_ref3 < best_cont_score:
            best_cont_score = sc_ref3
            best_cont_h = h_ref3.copy()

    if tleft() > 20:
        n_use = len(best_cont_h)
        n_iters_f = int(min(500000, max(30000, (tleft() - 10) * 20000)))
        h_pgf, c5_pgf = nesterov_pg(best_cont_h, n_iters_f, topk=3,
                                    lr_init=0.015, lr_end=0.0001,
                                    time_limit=max(5.0, tleft() - 5),
                                    start_time=start_time)
        if c5_pgf < best_cont_score:
            best_cont_score = c5_pgf
            best_cont_h = h_pgf.copy()

    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
