def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def fourier_eval(coeffs, N, x):
    """Evaluate h(x) = 0.5 + sum_{k=1}^N (a_k cos(pi*k*x) + b_k sin(pi*k*x))"""
    h = np.full_like(x, 0.5)
    for k in range(1, N+1):
        h += coeffs[2*(k-1)] * np.cos(np.pi * k * x)
        h += coeffs[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def fourier_objective_and_grad(coeffs, N, x_eval, alpha):
    """
    Compute smooth objective and gradient.
    Objective: 0.5 - 0.5 * (1/alpha) * log(sum_t exp(alpha * sum_k p_k cos(pi*k*t)))
    where p_k = a_k^2 + b_k^2
    """
    n_eval = len(x_eval)
    # Compute S(t) = sum_k (a_k^2 + b_k^2) cos(pi*k*t) for each t
    S = np.zeros(n_eval)
    for k in range(1, N+1):
        pk = coeffs[2*(k-1)]**2 + coeffs[2*(k-1)+1]**2
        S += pk * np.cos(np.pi * k * x_eval)
    
    # Log-sum-exp
    S_max = np.max(S)
    log_sum_exp = S_max + np.log(np.sum(np.exp(alpha * (S - S_max))))
    obj = 0.5 - 0.5 * log_sum_exp / alpha
    
    # Gradient
    w = np.exp(alpha * (S - S_max))
    w = w / w.sum()  # These are the softmax weights
    
    grad = np.zeros(2*N)
    for k in range(1, N+1):
        wi_cos = np.sum(w * np.cos(np.pi * k * x_eval))
        grad[2*(k-1)] = -coeffs[2*(k-1)] * wi_cos
        grad[2*(k-1)+1] = -coeffs[2*(k-1)+1] * wi_cos
    
    return obj, grad

def fourier_constraints(coeffs, N, x_eval):
    """Return constraint values: 0 <= h(x) <= 1"""
    h_vals = fourier_eval(coeffs, N, x_eval)
    cons = []
    for i in range(len(x_eval)):
        cons.append(h_vals[i])  # h >= 0
        cons.append(h_vals[i] - 1.0)  # h <= 1
    return cons

def optimize_power_spectrum_lp(N, n_t_eval):
    t_eval = np.linspace(0, 2, n_t_eval, endpoint=False)
    # A matrix: A[i,k] = cos(pi*(k+1)*t_i)
    k_idx = np.arange(1, N+1)
    A = np.cos(np.pi * np.outer(k_idx, t_eval))  # shape (N, n_t_eval)
    A = A.T  # shape (n_t_eval, N)
    
    # Variables: x = [p_1, ..., p_N, c]
    # Maximize c => minimize -c
    c_obj = np.zeros(N+1)
    c_obj[N] = -1  # coefficient for c

    # So A_ub @ x <= b_ub where:
    A_ub = np.zeros((n_t_eval, N+1))
    A_ub[:, :N] = -A

    b_ub = np.zeros(n_t_eval)

        grad_pen = np.zeros(2 * N)
        excess = np.where(h_vals > 1.0, 2.0 * (h_vals - 1.0), 0.0) + np.where(h_vals < 0.0, 2.0 * h_vals, 0.0)
        for k in range(N):
            grad_pen[2*k] = 50.0 * cos_mat_x[k] @ excess
            grad_pen[2*k+1] = 50.0 * sin_mat_x[k] @ excess
        return obj, grad_c5 + grad_pen

    best_c5 = np.inf
    best_coeffs = coeffs0.copy()
    start = time.time()
    alphas_list = [100.0, 500.0, 2000.0, 10000.0]
    global _fourier_alpha
    _fourier_alpha = 100.0

    def _obj_wrapper(c):
        a = c[0::2]
        b = c[1::2]
        p = a**2 + b**2
        S = cos_mat_t.T @ p
        al = _fourier_alpha
        S_max = np.max(S)
        logse = S_max + np.log(np.sum(np.exp(al * (S - S_max))))
        obj_c5 = 0.5 - 0.5 * logse / al
        h_vals = 0.5 + cos_mat_x.dot(a) + sin_mat_x.dot(b)
        pen = 50.0 * (np.sum(np.maximum(0.0, h_vals - 1.0)**2) + np.sum(np.maximum(0.0, -h_vals)**2))
        return obj_c5 + pen

    _fourier_alpha = 100.0
    for al in [100.0, 500.0, 2000.0, 10000.0]:
        if time.time() - start > time_limit:
            break
        _fourier_alpha = al
        try:
            res = minimize(_obj_wrapper, coeffs0, method='Nelder-Mead',
                           options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
            h_candidate = fourier_to_grid(res.x, N, n_grid)
            h_candidate = project_feasible(h_candidate, n_grid / 2.0)
            sc = compute_c5(h_candidate)
            if sc < best_c5:
                best_c5 = sc
                best_coeffs = res.x.copy()
                coeffs0 = res.x.copy()
        except Exception:
            pass
    h_final = fourier_to_grid(best_coeffs, N, n_grid)
    h_final = project_feasible(h_final, n_grid / 2.0)
    return h_final, compute_c5(h_final)

def fast_pair_cd(h_init, n, target, time_limit, n_cand=25, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for pp in range(n_c):
            cand_pos[int(all_cands[pp])] = pp
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(15):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def multi_shift_equalize(h_init, n, target, time_limit, max_steps=20000, n_shifts=5, base_lr=0.03):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved and step > 2000:
            break
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    target_sum = target
    bounds = [(0.0, 1.0)] * n

    def _con_fun(x):
        return np.sum(x) - target_sum

    def _con_jac(x):
        return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        def _smooth_obj(x, al=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(al * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / al)
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP power spectrum + phase finding for several N
    for N in [6, 8, 10, 12, 15]:
        if tleft() < 80:
            break
        p_opt, c_opt = lp_power_spectrum(N, n)
        c5_lower = 0.5 - 0.5 * c_opt
        if c5_lower > best_score:
            continue
        theta, viol = find_valid_phases(p_opt, N, n, n_restarts=20, max_iter=600, time_limit=30)
        if theta is None or viol > 1.0:
            continue
        sqrt_p = np.sqrt(np.maximum(p_opt, 0.0))
        coeffs = np.zeros(2 * N)
        for k in range(N):
            coeffs[2*k] = sqrt_p[k] * np.cos(theta[k])
            coeffs[2*k+1] = sqrt_p[k] * np.sin(theta[k])
        h_fourier = fourier_to_grid(coeffs, N, n)
        h_fourier = project_feasible(h_fourier, target)
        sc = compute_c5(h_fourier)
        if sc < best_score:
            best_score = sc
            best_h = h_fourier

    # Phase 2: Direct Fourier optimization (Nelder-Mead in coefficient space)
    for N in [8, 12, 16]:
        if tleft() < 60:
            break
        h_f, sc_f = fourier_optimize_direct(best_h, N, n, time_limit=min(50, tleft() * 0.3))
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 3: SLSQP from current best
    if tleft() > 80:
        hp, sc = slsqp_smooth(best_h, n, target, [200, 1000, 5000, 20000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Pair CD (main workhorse)
    if tleft() > 150:
        cd_time = min(300, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Multi-shift equalize
    if tleft() > 60:
        hsg, c5sg = multi_shift_equalize(best_h, n, target, time_limit=min(50, tleft() * 0.2),
                                         max_steps=30000, n_shifts=5, base_lr=0.03)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 6: SLSQP polish
    if tleft() > 60:
        hp, sc = slsqp_smooth(best_h, n, target, [5000, 20000, 100000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Pair CD with larger candidate set
    if tleft() > 120:
        cd_time = min(200, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 8: Try higher resolution
    if tleft() > 250:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_smooth(h_interp, n_new, target_new, [200, 1000, 5000], 1500, 1e-15)
        if tleft() > 100:
            hcd, c5cd = fast_pair_cd(hp, n_new, target_new, time_limit=min(120, tleft() * 0.4),
                                     n_cand=25, max_sweeps=50000)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 9: More pair CD
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(100, tleft() * 0.4)
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=cd_time, n_cand=20, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: SLSQP with very high alpha
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_smooth(best_h, n_use, target_use, [50000, 200000, 500000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final multi-shift
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hsg, c5sg = multi_shift_equalize(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.5),
                                         max_steps=20000, n_shifts=7, base_lr=0.02)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 12: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=min(25, tleft() * 0.6),
                                 n_cand=30, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final SLSQP
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_smooth(best_h, n_use, target_use, [100000, 500000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
