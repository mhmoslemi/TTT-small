    s_star = np.argmax(corr)
    g_star = g[s_star]
    
    if g_star >= -1e-15:
        # The max shift is not decreasing, so best is delta=0 or very small
        return max(delta_lo, 0.0) if delta_lo >= 0 else 0.0
    
    # Find where another shift catches up
    # Need: corr[s] + delta*g[s] <= corr[s*] + delta*g[s*] for all s
    # => delta * (g[s] - g[s*]) <= corr[s*] - corr[s]
    # For g[s] > g[s*]: delta <= (corr[s*] - corr[s]) / (g[s] - g[s*])
    
    diff_g = g - g_star
    positive_g = diff_g > 1e-15
    if positive_g.any():
        catch_deltas = (corr[s_star] - corr[positive_g]) / diff_g[positive_g]
        delta_catch = np.min(catch_deltas)
    else:
        delta_catch = np.inf
    
    delta_opt = min(delta_hi, max(delta_catch, delta_lo))
    delta_opt = max(delta_lo, min(delta_hi, delta_opt))
    
    return delta_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for s in range(n_corr):
        k = s - (n - 1)
        pk = idxs + k
        valid_pk = (pk >= 0) & (pk < n)
        change[valid_pk, s] += (1.0 - h[pk[valid_pk]])
        pm = idxs - k
        valid_pm = (pm >= 0) & (pm < n)
        change[valid_pm, s] -= h[pm[valid_pm]]
    return change

def update_change_rows(change, p, h, n):
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    change[p, :] = 0.0
    for s in range(n_corr):
        k = s - (n - 1)
        pk = idxs[p] + k
        if 0 <= pk < n:
            change[p, s] += (1.0 - h[pk])
        pm = idxs[p] - k
        if 0 <= pm < n:
            change[p, s] -= h[pm]

def update_change_rows_batch(change, positions, h, n):

    idxs = np.arange(n)

def find_optimal_delta_fast(corr, g, delta_lo, delta_hi):
    if delta_hi - delta_lo < 1e-14:
        return 0.0
    s_star = int(np.argmax(corr))
    g_star = g[s_star]
    if g_star >= -1e-14:
        return max(delta_lo, 0.0)
    diff_g = g - g_star
    positive = diff_g > 1e-14
    if positive.any():
        catches = (corr[s_star] - corr[positive]) / diff_g[positive]
        delta_catch = float(np.min(catches))
    else:
        delta_catch = delta_hi
    delta_opt = min(delta_hi, max(delta_catch, delta_lo))
    delta_opt = max(delta_lo, min(delta_hi, delta_opt))
    if delta_opt <= delta_lo + 1e-14:
        return 0.0
    return delta_opt

def pair_cd(h_init, n, target, time_limit, n_cand_start=20, n_cand_end=60,

    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if sweep % recoffee_interval == 0:
            corr = compute_corr(h)

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        s_star = int(np.argmax(corr))

        n_top = min(n_top_shifts, len(corr))
        top_idx = np.argsort(corr)[-n_top:]
        top_corr = corr[top_idx]
        w = np.exp(20.0 * (top_corr - top_corr.max()))
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(n_top):
            s = int(top_idx[ii])
            grad += w[ii] * change[:, s]
        grad -= grad.mean()

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_g = None

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                delta_opt = find_optimal_delta_fast(corr, g, delta_lo, delta_hi)
                if delta_opt <= 1e-13:
                    continue

                new_max = float(np.max(corr + delta_opt * g)) * dx
                if new_max < best_trial_c5:
                    best_trial_c5 = new_max
                    best_i = i
                    best_j = j
                    best_delta = delta_opt
                    best_g = g

        if temperature > 0 and best_trial_c5 >= c5_val:
            n_try = 5
            for _ in range(n_try):
                i = int(low_set[np.random.randint(len(low_set))])
                j = int(high_set[np.random.randint(len(high_set))])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                delta_opt = find_optimal_delta_fast(corr, change[j] - change[i], delta_lo, delta_hi)
                if delta_opt <= 1e-13:
                    continue
                new_max = float(np.max(corr + delta_opt * (change[j] - change[i]))) * dx
                delta_f = new_max - c5_val
                if np.random.rand() < np.exp(-delta_f / temperature):
                    best_trial_c5 = new_max
                    best_i = i
                    best_j = j
                    best_delta = delta_opt
                    best_g = change[j] - change[i]
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = corr + best_delta * best_g
            update_change_rows(change, best_i, h, n)
            update_change_rows(change, best_j, h, n)

    return best_h, best_c5

def slsqp_polish(h_init, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(100.0 * (corr - c_max), -500.0, 0.0)
        return float(c_max)

    def cons(x):
        return np.sum(x) - target

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            best_n = n2

    if tleft() > 60:
        h_cd4, c5_cd4 = multi_pair_cd(best_h, best_n, best_n / 2.0,
                                      time_limit=min(150, tleft() * 0.5),
                                      n_cand=50, n_top_shifts=5,
                                      max_sweeps=15000, n_pairs_per_sweep=2)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    if tleft() > 50:
        for noise_scale in [0.01, 0.03]:
            if tleft() < 35:
                break
            h_p, c5_p = perturb_and_refine(best_h, best_n, best_n / 2.0,
                                           time_limit=min(40, tleft() * 0.4),
                                           rng=rng, noise_scale=noise_scale)
            if c5_p < best_score:
                best_score = c5_p
                best_h = h_p

    if tleft() > 40 and best_n <= 1000:
        h_pol_final = slsqp_polish(best_h, best_n, best_n / 2.0, [200, 500], 300,
                                   time_limit=min(35, tleft() * 0.7),
                                   start_time=start_time)
        sc = compute_c5(h_pol_final)
        if sc < best_score:
            best_score = sc
            best_h = h_pol_final

    if tleft() > 30:
        h_final_cd, c5_final_cd = pair_cd(best_h, best_n, best_n / 2.0,
                                          time_limit=min(25, tleft() * 0.6),
                                          n_cand_start=30, n_cand_end=70,
                                          n_top_shifts=3, max_sweeps=10000,
                                          max_no_improve=200)
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
