    valid_jk = (0 <= j+k) & (j+k < n)
    delta[valid_jk + n - 1] += h[j+k[valid_jk]]
    valid_ik2 = (0 <= i-k) & (i-k < n)
    delta[valid_ik2 + n - 1] -= (1 - h[i-k[valid_ik2]])
    valid_jk2 = (0 <= j-k) & (j-k < n)
    delta[valid_jk2 + n - 1] -= (1 - h[j-k[valid_jk2]])
    return delta

def compute_gradient(h, weights, n):
    # weights: array of length 2n-1 (indexed by shift k, where index = k + n - 1)
    # Returns gradient of shape (n,)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(n):
        # shift k >= 0: corr[k] = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
        w = weights[k + n - 1]
        if w < 1e-12:
            continue
        # ∂corr[k]/∂h[p] = (1-h[p-k]) - h[p+k] for valid p
        # (1-h[p-k]) valid when 0<=p-k<n, i.e., k<=p<=n-1+k, intersect [0,n-1] → p in [k, n-1]
        # h[p+k] valid when 0<=p+k<n, i.e., -k<=p<=n-1-k, intersect [0,n-1] → p in [0, n-1-k]
        grad[k:] += w * one_minus_h[:n-k]  # (1-h[p-k]) for p in [k, n-1], h[p-k] = h[0..n-1-k]
        grad[:n-k] -= w * h[k:]  # h[p+k] for p in [0, n-1-k], h[p+k] = h[k..n-1]
    for k in range(1, n):
        # shift -k: corr[-k] = sum_{n=k}^{n-1} h[n-k]*(1-h[n])
        w = weights[n-1-k]
        if w < 1e-12:
            continue
        # ∂corr[-k]/∂h[p]:
        # From h[n-k] term: nonzero when n-k=p, n=p+k. Contribution: (1-h[p+k]) if 0<=p+k<n → p in [0, n-1-k]
        # From (1-h[n]) term: nonzero when n=p. Contribution: -h[p-k] if 0<=p-k<n → p in [k, n-1]
        grad[:n-k] += w * one_minus_h[k:]  # (1-h[p+k]) for p in [0, n-1-k]
        grad[k:] -= w * h[:n-k]  # h[p-k] for p in [k, n-1]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r

def smooth_max_value_and_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    return value, weights

def smooth_grad(h, weights, n):
    # weights: length 2n-1, index corresponds to shift k at position k+n-1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    # Positive shifts k = 0, 1, ..., n-1
    for k in range(n):
        w = weights[n - 1 + k]
        if w < 1e-10:
            continue
        grad[k:] += w * one_minus_h[:n - k]
        grad[:n - k] -= w * h[k:]
    # Negative shifts k = -1, -2, ..., -(n-1)
    for k in range(1, n):
        w = weights[n - 1 - k]
        if w < 1e-10:
            continue
        grad[:n - k] += w * one_minus_h[k:]
        grad[k:] -= w * h[:n - k]
    return grad

def adam_optimize(h0, target, start_time, max_time, alphas, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        rem = max_time - (time.time() - start_time)
        if rem < 2:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        while rem > 1:
            rem = max_time - (time.time() - start_time)
            if rem < 1:
                break
            t += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            val, weights = smooth_max_value_and_weights(corr, alpha)
            
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 200:
                break
            
            grad = smooth_grad(h, weights, n)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Adaptive learning rate
            lr_t = lr * (0.5 ** (t / max(100, t // 2)))  # decay after 100 steps
            
            h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
        
        actual = compute_c5(h)
        if actual < best_score:
            best_score = actual
            best_h = h.copy()
        else:
            h = best_h.copy()
    
    return best_h, best_score

def binary_local_search(h, target, start_time, max_time, max_stall=500):
    """Fast binary local search with correct candidate filtering."""
    n = len(h)

            for i in ones:
                delta = 0.0
                if 0 <= i + k < n:
                    delta += h[i + k]
                if 0 <= i - k < n:
                    delta -= (1.0 - h[i - k])
                if delta < 0:
                    i_cands.append((i, delta))

            # Change in corr[k] from j (0→1): -h[j+k]*I(0<=j+k<n) + (1-h[j-k])*I(0<=j-k<n)

            # h[j+k]=1 and h[j-k]=0: -1+1 = 0
