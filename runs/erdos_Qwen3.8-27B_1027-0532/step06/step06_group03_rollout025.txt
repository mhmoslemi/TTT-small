def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_smooth_obj(h, alph):

    corr = np.correlate(h, 1.0 - h, mode='full')

def get_smooth_grad(h, alph):

    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_fn(h, alph):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alph * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alph)) * dx

def smooth_grad_fn(h, alph):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alph * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        js = j + shifts
        jm = j - shifts
        vp = (js >= 0) & (js < n)
        vm = (jm >= 0) & (jm < n)
        if vp.any():
            grad[j] += np.dot(w[vp], 1.0 - h[js[vp]])
        if vm.any():
            grad[j] -= np.dot(w[vm], h[jm[vm]])
    return grad * dx

def active_set_lp_step(h, target, n_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:min(n_active, len(sorted_idx))]
    mid = n - 1
    rng_arr = np.arange(n)
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_rows = []
    b_vals = []
    for s_idx in active:
        s = int(s_idx) - mid
        a_s = np.zeros(n)
        js = rng_arr + s
        vp = (js >= 0) & (js < n)
        a_s[vp] += (1.0 - h[js[vp]])
        jm = rng_arr - s
        vm = (jm >= 0) & (jm < n)
        a_s[vm] -= h[jm[vm]]
        if s >= 0:
            vi = rng_arr[:n - s]
            b_s = np.sum(h[vi] * h[vi + s])
        else:
            vi = rng_arr[-s:]
            b_s = np.sum(h[vi] * h[vi + s])
        row = np.zeros(n_vars)
        row[:n] = a_s
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(-b_s)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_list = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs',
                      options={'presolve': True})
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            return h_new
    except Exception:
        pass
    return None

def active_set_optimize(h_init, target, time_limit, n_active_seq=None):
    if n_active_seq is None:
        n_active_seq = [5, 10, 20, 30, 50, 80, 120]
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    for na in n_active_seq:
        level_start = time.time()
        level_budget = min(40, (time_limit - (level_start - start_t)) * 0.6)
        for it in range(300):
            if time.time() - start_t > time_limit:
                break
            if time.time() - level_start > level_budget:
                break
            h_lp = active_set_lp_step(h, target, na)
            if h_lp is None:
                break
            c5_lp = compute_c5(h_lp)
            if c5_lp < best_c5 - 1e-14:
                for lam in [1.0, 0.7, 0.4, 0.2]:
                    h_try = (1.0 - lam) * h + lam * h_lp
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        break
            else:
                h_try = h_lp
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                elif time.time() - level_start > 5.0:
                    break
    return best_h, best_c5

def projected_grad_nesterov(h_init, target, alph, max_iter, time_limit, lr_init):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.85
    start_t = time.time()
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad_fn(h, alph)
        lr = lr_init / (1.0 + 0.05 * it)
        v = mu * v + grad
        mu = min(0.999, 0.85 + 0.001 * it)
        obj_cur = smooth_obj_fn(h, alph)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = project_feasible(h - lr * mult * v, target)
            obj_try = smooth_obj_fn(h_try, alph)
            if obj_try < obj_cur - 1e-16:
                h = h_try
                improved = True
                break
        if not improved:
            h = project_feasible(h - lr * 0.01 * v, target)
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
    return best_h, best_c5

def pair_cd_fast(h_init, target, time_limit, max_sweeps, n_cand, top_k):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        sa = 50.0
        w_top = np.exp(sa * (corr[top_idx] - c_max))
        w_top /= w_top.sum()
        grad = np.zeros(n)
        for wi, si in zip(w_top, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                vip = (i + shifts >= 0) & (i + shifts < n)
                g[vip] += h[i + shifts[vip]]
                vim = (i - shifts >= 0) & (i - shifts < n)
                g[vim] += h[i - shifts[vim]]
                vjp = (j + shifts >= 0) & (j + shifts < n)
                g[vjp] -= h[j + shifts[vjp]]
                vjm = (j - shifts >= 0) & (j - shifts < n)
                g[vjm] -= h[j - shifts[vjm]]
                a_d = 0.0
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    if np.max(corr + mid_d * g) <= c_max + 1e-12:
                        a_d = mid_d
                    else:
                        b_d = mid_d
                delta_star = a_d
                if delta_star < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def optimize_at_resolution(h_in, target, time_limit):
    h = h_in.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if time_limit > 30:
        for alph in [100, 500, 2000]:
            if time_limit - (time.time() - (time.time() - time_limit)) < 15:
                break
        for alph in [100, 500]:
            remaining = time_limit - (time.time() - (time.time() - time_limit))
            sub_time = min(20, remaining * 0.25)
            if sub_time < 10:
                break
            h_pg, c5_pg = projected_grad_nesterov(h, target, alph,
                                                   max_iter=3000, time_limit=sub_time,
                                                   lr_init=0.04)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg
                h = h_pg

    remaining = time_limit - (time.time() - (time.time() - time_limit))
    if remaining > 20:
        sub_time = min(remaining * 0.5, 120)
        h_lp, c5_lp = active_set_optimize(h, target, time_limit=sub_time)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    remaining = time_limit - (time.time() - (time.time() - time_limit))
    if remaining > 15:
        sub_time = min(remaining * 0.4, 60)
        h_cd, c5_cd = pair_cd_fast(best_h, target, time_limit=sub_time,
                                    max_sweeps=8000, n_cand=45, top_k=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 25.0 - elapsed()

    # Phase 1: Optimize at initial resolution (n=400)
    if tleft() > 100:
        h_opt, c5_opt = optimize_at_resolution(best_h, target, time_limit=min(120, tleft()))
        if c5_opt < best_score:
            best_score = c5_opt
            best_h = h_opt

    # Phase 2: Scale to 2x resolution
    if tleft() > 200:
        n2 = 2 * len(best_h)
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        c5_interp = compute_c5(h_interp)
        if c5_interp <= best_score + 0.002:
            h_opt, c5_opt = optimize_at_resolution(h_interp, target2,
                                                    time_limit=min(300, tleft()))
            if c5_opt < best_score:
                best_score = c5_opt
                best_h = h_opt

    # Phase 3: Scale to 4x resolution
    if tleft() > 200:
        n_cur = len(best_h)
        n3 = max(4 * n, 2 * n_cur)
        if n3 <= 2400:
            target3 = n3 / 2.0
            h_interp = interpolate_to(best_h, n3, target3)
            c5_interp = compute_c5(h_interp)
            if c5_interp <= best_score + 0.002:
                h_opt, c5_opt = optimize_at_resolution(h_interp, target3,
                                                        time_limit=min(250, tleft()))
                if c5_opt < best_score:
                    best_score = c5_opt
                    best_h = h_opt

    # Phase 4: Additional active-set refinement at best resolution
    n_fin = len(best_h)
    tgt_fin = n_fin / 2.0
    if tleft() > 60:
        h_lp, c5_lp = active_set_optimize(best_h, tgt_fin, time_limit=min(60, tleft()))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: Final pair CD
    if tleft() > 40:
        h_cd, c5_cd = pair_cd_fast(best_h, tgt_fin, time_limit=min(35, tleft()),
                                    max_sweeps=5000, n_cand=50, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Random restarts if time permits
    rng = np.random.RandomState(seed)
    if tleft() > 80:
        for noise_scale in [0.04, 0.08, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, tgt_fin)
            h_opt, c5_opt = optimize_at_resolution(h_pert, tgt_fin,
                                                    time_limit=min(30, tleft()))
            if c5_opt < best_score:
                best_score = c5_opt
                best_h = h_opt

    # Final feasibility
    n_use = len(best_h)
    tgt_use = n_use / 2.0
    h_final = project_feasible(best_h, tgt_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
