def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _effective_gradient(h, mu, n, dx):
    """Compute the effective gradient G_i = sum_k mu_k * dI(k)/dh_i"""
    n_full = 2 * n - 1
    G = np.zeros(n)
    idxs = np.arange(n)
    for k_idx, mu_k in enumerate(mu):
        if mu_k < 1e-15:
            continue
        k = k_idx - (n - 1)
        # dI(k)/dh_i = dx * [(1-h(i-k)) * 1_{0<=i-k<n} - h(i+k) * 1_{0<=i+k<n}]
        v1 = (idxs - k)
        v2 = (idxs + k)
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        G[m1] += mu_k * (1.0 - h[v1[m1]]) * dx
        G[m2] -= mu_k * h[v2[m2]] * dx
    return G

def smoothed_subgradient(h, n, target, max_steps, time_limit, beta0=50.0, beta_max=5000.0):
    """Subgradient method with soft active set"""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    beta = beta0
    stall = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = np.max(corr)
        # Soft active set
        shifted = beta * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        mu = exp_vals / np.sum(exp_vals)
        
        G = _effective_gradient(h_cur, mu, n, dx)
        # Remove the mean to maintain the sum constraint
        G_centered = G - np.mean(G)
        
        # Try line search
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0]:
            h_try = h_cur - lr * G_centered
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                # Increase beta slowly
                if step % 50 == 0:
                    beta = min(beta_max, beta * 1.05)
                break
        if not improved:
            stall += 1
            # Decrease beta to explore more
            beta = max(beta0, beta * 0.8)
            if stall > 200:
                break
    
    return best_h, best_c5

def alternating_lp_v2(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11, over_relax=1.0):
    """Improved alternating LP with over-relaxation"""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=200000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        
        # Over-relaxation
        if over_relax > 1.0:
            h_updated = (1.0 - over_relax) * h + over_relax * h_new
        else:
            h_updated = (1.0 - damping) * h + damping * h_new

            best_c5 = c5_val

    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.array([np.sum(h) - _global_target])

    return np.array([np.ones(_global_n)])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

Let me try a **Newton-based approach** on the KKT conditions.

At the optimum, if the active set is {k_1, ..., k_m} (the shifts that achieve the max), then the KKT conditions are:
1. I(k_j) = t for all j = 1, ..., m (active constraints)
2. I(k) ≤ t for all other k
3. 1^T h = n/2
4. 0 ≤ h ≤ 1 with complementary slackness

The active constraints give us m equations:
Σ_i h[i+k_j](1-h[i]) dx = t for j = 1, ..., m

Plus the sum constraint: 1^T h = n/2

That's m+1 equations in n+1 unknowns (h_1, ..., h_n, t).

If m < n-1, we have more unknowns than equations, and the solution is determined by the box constraints (which h_i are at 0 or 1).

At the optimum, many h_i will be at their bounds (0 or 1) due to the non-convexity. The "free" variables (those strictly between 0 and 1) are determined by the active constraints.

**Algorithm:**
1. Identify the active set A (k values where I(k) ≈ max)
2. Identify the free set F (i values where 0 < h_i < 1)
3. Solve the linear system for the free variables:
   For each active k_j: Σ_{i∈F, valid} (h[i+k_j] * dx) = t + Σ_{i∉F, valid} h[i+k_j](1-h[i]) dx
   (This is linear in h_F because the "1-h[i]" term is fixed for i ∉ F)
   
   Wait, it's not quite linear because h[i+k_j] might also be free. Let me be more careful.
   
   I(k) = Σ_i h[i+k](1-h[i]) dx
   
   If i is free and i+k is also free, then the term h[i+k](1-h[i]) is bilinear.
   If i is free and i+k is fixed, then it's linear in h[i+k]: (1-h[i]) * h[i+k] * dx
   If i is fixed and i+k is free, then it's linear in h[i+k]: h[i+k] * (1-h[i]) * dx
   If both are fixed, it's a constant.

So the system is only linear if we assume that for each free variable h[i], the corresponding h[i+k] for active k values is fixed. This is a strong assumption but might hold approximately.

Alternatively, we can use a **sequential linearization** approach:
1. At current h, identify active set A and free set F
2. Linearize I(k) around current h for k ∈ A
3. Solve the resulting LP (which is the same as the alternating LP!)
4. Update the active and free sets
5. Repeat

This is essentially what the alternating LP does! So the Newton approach doesn't give us much new.

Let me try yet another angle: **exploiting the symmetry of the problem**.

The problem is symmetric under h(x) → h(2-x) (reflection). The optimal solution should either be symmetric or come in symmetric pairs. We can exploit this by:
- Constraining h to be symmetric: h(i) = h(n-1-i)
- This halves the number of variables and might help the optimizer

But this might not help if the optimal solution is asymmetric.

**Another idea: use the dual problem.**

The primal is:
min max_k I(k)
s.t. 1^T h = n/2, 0 ≤ h ≤ 1

The dual (using Lagrange duality) is:
max inf_{h: 1^Th=n/2, 0≤h≤1} [max_k I(k) + λ(1^Th - n/2)]
= max_λ inf_h [max_k I(k) + λ(1^Th - n/2)]

For fixed λ, the inner minimization is:
inf_{h: 0≤h≤1} [max_k I(k) + λ(1^Th - n/2)]

This is still non-convex. But we can further relax:
≤ max_{μ: μ_k ≥ 0, Σμ_k = 1} inf_{h: 0≤h≤1} [Σ_k μ_k I(k) + λ(1^Th - n/2)]

For fixed μ and λ, this is:
inf_{h: 0≤h≤1} [Σ_k μ_k Σ_i h[i+k](1-h[i]) dx + λ(Σh - n/2)]
= inf_{h: 0≤h≤1} [Σ_i (Σ_k μ_k (1-h[i]) h[i+k] dx) + λ(Σh - n/2)]

The inner part is still quadratic in h (bilinear). So we can't solve it directly.

But if we use the alternating approach (fix "1-h" or "h"), we get the same LP.

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_max_line(slopes, intercepts, lam):

    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-13 * max(1.0, abs(max_val))

            active[np.argmax(vals)] = True

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max) - 500.0, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals) + 1e-300)
    obj = float(c_max + (log_sum - 500.0) / alpha)
    p = exp_vals / (np.sum(exp_vals) + 1e-300)
    half_n = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_full := 2 * n - 1):
        if p[k_idx] < 1e-15:
            continue
        k = k_idx - half_n
        v1 = idxs - k
        v2 = idxs + k
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        grad[m1] += p[k_idx] * (1.0 - h[v1[m1]]) * dx
        grad[m2] -= p[k_idx] * h[v2[m2]] * dx
    return obj, grad


def smooth_obj_only(h, alpha):

    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alpha, maxiter, ftol):

    _global_target = target

        return h_new, sc
    except Exception:
        return h_start, compute_c5(h_start)

        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Aggressive alternating LP
    if tleft() > 200:
        lp_time = min(280, tleft() * 0.28)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_time,
                                     d0=0.85, reg0=1e-4, n_active=100,
                                     scs_eps=1e-12, over_relax=1.0)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent with many active constraints
    if tleft() > 100:
        cd_time = min(80, tleft() * 0.12)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=100,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Projected BFGS on smooth approximation
    if tleft() > 150:
        bfgs_time = min(120, tleft() * 0.18)
        h_bf, c5_bf = projected_bfgs(best_h, n, target,
                                     time_limit=bfgs_time,
                                     alphas=[200, 1000, 5000, 20000],
                                     max_iter_per_alpha=300)
        if c5_bf < best_score:
            best_score = c5_bf
            best_h = h_bf

    # Phase 4: LP refinement with tighter settings
    if tleft() > 180:
        lp_time = min(200, tleft() * 0.22)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_time,
                                     d0=0.92, reg0=1e-5, n_active=120,
                                     scs_eps=1e-12, over_relax=1.05)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: Smoothed subgradient
    if tleft() > 80:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=5000, time_limit=min(60, tleft() * 0.1),
                                          beta0=200.0, beta_max=20000.0)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: SLSQP with high alpha
    if tleft() > 100:
        h_ref, sc = slsqp_optimize(best_h, n, target, 5000, 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_ref, sc = slsqp_optimize(best_h, n, target, 20000, 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Try higher resolution
    if tleft() > 300:
        for n_try in [500, 640]:
            if tleft() < 200:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            sc_int = compute_c5(h_interp)
            if tleft() > 180:
                lp_t = min(120, tleft() * 0.35)
                h_lp2, c5_lp2 = alternating_lp(h_interp, n_try, target_try,
                                               max_iter=20000, time_limit=lp_t,
                                               d0=0.9, reg0=5e-5, n_active=80,
                                               scs_eps=1e-11, over_relax=1.0)
                if c5_lp2 < best_score:
                    best_score = c5_lp2
                    best_h = h_lp2
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp

    # Phase 8: Perturbation restarts
    for noise in [0.01, 0.03, 0.07, 0.12, 0.2]:
        if tleft() < 90:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        h_ref, sc = slsqp_optimize(h_pert, n_use, target_use, 2000, 1200, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        if tleft() > 60 and sc < best_score + 0.001:
            h_bf2, c5_bf2 = projected_bfgs(h_ref, n_use, target_use,
                                           time_limit=min(30, tleft() * 0.3),
                                           alphas=[5000, 20000],
                                           max_iter_per_alpha=200)
            if c5_bf2 < best_score:
                best_score = c5_bf2
                best_h = h_bf2

    # Phase 9: Final coordinate descent
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(40, tleft() * 0.5)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=150,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final LP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(35, tleft() * 0.6)
        h_lp, c5_lp = alternating_lp(best_h, n_use, target_use,
                                     max_iter=20000, time_limit=lp_time,
                                     d0=0.98, reg0=1e-7, n_active=150,
                                     scs_eps=1e-13, over_relax=1.0)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final BFGS
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bf, c5_bf = projected_bfgs(best_h, n_use, target_use,
                                     time_limit=min(20, tleft() * 0.6),
                                     alphas=[50000, 200000],
                                     max_iter_per_alpha=300)
        if c5_bf < best_score:
            best_score = c5_bf
            best_h = h_bf

    # Phase 12: Final subgradient micro-polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=2000, time_limit=min(10, tleft() * 0.7),
                                          beta0=5000.0, beta_max=100000.0)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
