def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, dx, target_sum, penalty):
    """Compute smooth max objective + penalty for sum constraint, and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    scaled -= np.max(scaled)  # for numerical stability
    exp_scaled = np.exp(scaled)
    log_sum = np.log(np.sum(exp_scaled))
    obj = log_sum / alpha  # This approximates max(corr*dx)
    
    # Add penalty for sum constraint
    sum_dev = np.sum(h) - target_sum
    obj += penalty * sum_dev * sum_dev
    
    # Gradient
    weights = exp_scaled / np.sum(exp_scaled)
    mid = n - 1
    
    # Compute gradient of smooth max part
    grad = np.zeros(n)
    for j in range(n):
        s_start = max(j - n + 1, -j)
        s_end = min(j, n - 1 - j)
        if s_start <= s_end:
            s_range = np.arange(s_start, s_end + 1)
            w_idx = s_range + mid
            term1 = 1.0 - h[j - s_range]
            term2 = h[j + s_range]
            grad[j] = dx * dx * np.dot(weights[w_idx], term1 - term2)
    
    # Add gradient of penalty
    grad += 2.0 * penalty * sum_dev
    
    # Remove mean (to maintain sum constraint in gradient direction)
    # Actually, the penalty handles this. But for L-BFGS-B, we should keep it.
    
    return obj, grad

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h

    h = np.clip(h, 0.0, 1.0)

        m = np.max(scaled)
        s = np.log(np.sum(np.exp(scaled - m)))
        obj = (m + s) / alpha
        sum_dev = np.sum(x) - target
        obj += penalty * sum_dev * sum_dev
        return obj

        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = alpha * dx * corr
        m = np.max(scaled)
        weights = np.exp(scaled - m)

    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse(h, alpha, dx):
    """Compute log-sum-exp of alpha*dx*corr, return (value, weights)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    val = (m + np.log(np.sum(exp_s))) / alpha
    weights = exp_s / np.sum(exp_s)
    return val, weights

def grad_smooth(h, weights, dx, n, mid):
    """Compute gradient of log-sum-exp objective."""
    g = np.zeros(n)
    for j in range(n):
        s_start = max(j - n + 1, -j)
        s_end = min(j, n - 1 - j)
        if s_start <= s_end:
            s = np.arange(s_start, s_end + 1)
            gi = weights[s + mid]
            g[j] = dx * dx * np.dot(gi, (1.0 - h[j - s]) - h[j + s])
    return g

def optimize_lbfgs(h0, n, target, alpha, penalty, maxiter, gtol, ftime_s):
    """L-BFGS-B optimization with smooth max objective."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    target_f = target  # sum constraint target
    
    def fun(x):
        val, _ = smooth_lse(x, alpha, dx)
        sd = np.sum(x) - target_f
        return val + penalty * sd * sd
    
    def grad(x):
        _, w = smooth_lse(x, alpha, dx)
        g = grad_smooth(x, w, dx, n, mid)
        sd = np.sum(x) - target_f
        g += 2.0 * penalty * sd
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 40,
                                'maxfun': 100000})
        return res.x
    except Exception:
        return h0.copy()

def interpolate_up(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.5 / n_src, 2.0 - 0.5 / n_src, n_src)
    x_new = np.linspace(0.5 / n_dst, 2.0 - 0.5 / n_dst, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    
    def tl():
        return budget_s - 30.0 - (time.time() - t0)
    
    # Get initial solution
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    tgt0 = n0 / 2.0
    
    best_h = project_feasible(h_init, tgt0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at n0 with L-BFGS-B
    alphas1 = [50.0, 200.0, 1000.0, 5000.0, 20000.0]
    h_cur = best_h.copy()
    for alpha in alphas1:
        if tl() < 100:
            break
        h_new = optimize_lbfgs(h_cur, n0, tgt0, alpha, 1e5, 500, 1e-10, tl())
        h_new = project_feasible(h_new, tgt0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            h_cur = h_new
    
    # Phase 2: Upscale and optimize at higher resolution
    if tl() > 250:
        n1 = 2 * n0
        tgt1 = n1 / 2.0
        h_up = interpolate_up(best_h, n1, tgt1)
        
        alphas2 = [100.0, 500.0, 2000.0, 10000.0, 50000.0, 200000.0]
        h_cur2 = h_up
        for alpha in alphas2:
            if tl() < 150:
                break
            h_new = optimize_lbfgs(h_cur2, n1, tgt1, alpha, 1e5, 500, 1e-10, tl())
            h_new = project_feasible(h_new, tgt1)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_cur2 = h_new
    
    # Phase 3: Diverse restarts at current resolution
    if tl() > 200:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        
        for scale in [0.03, 0.08, 0.15]:
            if tl() < 150:
                break
            h_p = best_h + rng.randn(n_cur) * scale
            h_p = project_feasible(h_p, tgt_cur)
            h_new = optimize_lbfgs(h_p, n_cur, tgt_cur, 5000.0, 1e5, 300, 1e-9, tl())
            h_new = project_feasible(h_new, tgt_cur)
