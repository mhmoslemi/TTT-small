    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    w_sum = float(np.sum(exp_vals))

            continue
        for i in range(n - p):
            grad[i + p] += w * (1.0 - h[i])

```python
def smooth_grad_only(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    
    grad = np.zeros(n)
    
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-30:
            continue
        d = i - (n - 1)  # shift
        ad = abs(d)
        if d >= 0:
            # d(output)/dh[j] = (1-h[j-d])*[d≤j≤n-1] - h[j+d]*[0≤j≤n-1-d]
            if d == 0:
                grad[:] += w * (1.0 - h)  # (1-h[0])*[0≤j≤n-1]
                grad[:] -= w * h  # -h[0]*[0≤j≤n-1]
                # Total: (1-2h) * w
            else:
                # (1-h[j-d]) for j from d to n-1
                grad[d:] += w * (1.0 - h[:n-d])
                # -h[j+d] for j from 0 to n-1-d
                grad[:n-d] -= w * h[d:]
        else:
            # |d| = -d
            # d(output)/dh[j] = (1-h[j+|d|])*[0≤j≤n-1-|d|] - h[j-|d|]*[|d|≤j≤n-1]
            # (1-h[j+|d|]) for j from 0 to n-1-|d|
            grad[:n-ad] += w * (1.0 - h[ad:])
            # -h[j-|d|] for j from |d| to n-1
            grad[ad:] -= w * h[:n-ad]
    
    return grad
```
