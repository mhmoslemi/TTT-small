from scipy.optimize import minimize
import time

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_fn(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad_fn(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

def _con_fun(h):
    return np.sum(h) - _global_target

def _con_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj_fn, h_cur, args=(alpha,), jac=smooth_grad_fn,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def fast_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    mid = n - 1
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(-j, -(n-1))
        k_hi = min(n-1-j, n-1)
        if k_hi >= k_lo:
            ks = np.arange(k_lo, k_hi+1)
            grad[j] += np.dot(w[ks+mid], one_minus_h[j+ks])
        k_lo2 = max(j-(n-1), -(n-1))
        k_hi2 = min(j, n-1)
        if k_hi2 >= k_lo2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            grad[j] -= np.dot(w[ks2+mid], h[j-ks2])
    grad *= dx
    return grad

    return np.sum(h) - _global_target

def _con_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_logsumexp, h_cur, args=(alpha,), jac=fast_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_projected_gradient(h_start, n, target, alpha, time_limit, max_steps, mu=0.9, base_lr=0.02):
    """Fast projected gradient with momentum and adaptive step size."""
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve_count = 0
    
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        
        grad = fast_smooth_grad(h, alpha)
        grad -= np.mean(grad)  # preserve sum constraint in gradient direction
        
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        
        # Backtracking line search
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.025]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h) - 1e-15:
                h = h_try
                improved = True
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                break
        
        if not improved:
            no_improve_count += 1
            v *= 0.5  # dampen velocity
            if no_improve_count > 200:
                break
    
    return best_h, best_c5

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()

    h[idx] += diff / len(idx)

        else:

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand=40, top_k=3):

    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0/n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break
        
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]

        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        candidates = list(set(list(high_set) + list(low_set)))
        change_vecs = {}
        for pos in candidates:

            for k_val in shifts:
                if 0 <= pos + k_val < n:
                    cv[k_val + mid] += (1.0 - h[pos + k_val])
                if 0 <= pos - k_val < n:
                    cv[k_val + mid] -= h[pos - k_val]
            change_vecs[pos] = cv
        
        best_delta = 0.0
        best_i = -1
        best_j = -1

        for i in high_set:

                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                
                g = change_vecs[j] - change_vecs[i]
                
                # Bisection for optimal delta
                a_d = 0.0
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = (vals >= f_mid - 1e-14)
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                
                if delta_star < 1e-12:
                    continue

                if approx_max * (2.0/n) < best_new_c5:

                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def smooth_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode='full')

    mid = n - 1

    grad = np.zeros(n)

            ks2 = np.arange(k_lo2, k_hi2 + 1)
            grad[j] -= np.dot(w[ks2 + mid], h[j - ks2])
    grad *= dx
    return grad

        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        candidates = np.unique(np.concatenate([high_set, low_set]))
        change_vecs = {}
        for pos in candidates:
            cv = np.zeros(n_corr)
            valid_p = (pos + shifts >= 0) & (pos + shifts < n)
            valid_m = (pos - shifts >= 0) & (pos - shifts < n)
            cv[shifts[valid_p] + mid] += (1.0 - h[pos + shifts[valid_p]])
            cv[shifts[valid_m] + mid] -= h[pos - shifts[valid_m]]
            change_vecs[int(pos)] = cv

        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_c5 = c5_val

        for i in high_set:
            ii = int(i)
            for j in low_set:
                jj = int(j)
                if ii == jj:
                    continue
                delta_hi = min(h[ii], 1.0 - h[jj])
                if delta_hi < 1e-12:
                    continue

                g = change_vecs[jj] - change_vecs[ii]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = (vals >= f_mid - 1e-14)
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)

                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max * (2.0 / n) < best_new_c5:
                    best_new_c5 = approx_max * (2.0 / n)
                    best_i = ii
                    best_j = jj
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation(h_init, n, target, max_tries, time_limit, rng_seed=12345):
    start_t = time.time()
    rng_local = np.random.RandomState(rng_seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        n_pert = int(rng_local.choice([2, 3, 5, 8, 13, 21]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.02, 0.04, 0.08, 0.12]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n400 = len(h_init)
    target400 = n400 / 2.0

    best_h = project_feasible(h_init, target400)
    best_score = compute_c5(best_h)

    # Phase 1: Active-set projected gradient at n=400
    if tleft() > 150:
        h_pg, c5_pg = projected_gradient_momentum(best_h, n400, target400,
                                                   time_limit=min(60, tleft() * 0.2),
                                                   max_steps=100000,
                                                   use_smooth=False, k_top=5,
                                                   mu=0.9, base_lr=0.04)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 2: Pair CD at n=400
    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n400, target400,
                               time_limit=min(80, tleft() * 0.25),
                               max_sweeps=30000, n_cand=45, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800 and optimize
    if tleft() > 350:
        n800 = 800
        target800 = n800 / 2.0
        h800 = interpolate_to(best_h, n800, target800)
        c5_800 = compute_c5(h800)

        if tleft() > 280:
            h_pg800, c5_pg800 = projected_gradient_momentum(h800, n800, target800,
                                                              time_limit=min(80, tleft() * 0.2),
                                                              max_steps=80000,
                                                              use_smooth=False, k_top=5,
                                                              mu=0.9, base_lr=0.03)
            if c5_pg800 < c5_800:
                h800 = h_pg800
                c5_800 = c5_pg800

        if tleft() > 200:
            h_cd800, c5_cd800 = pair_cd(h800, n800, target800,
                                         time_limit=min(120, tleft() * 0.3),
                                         max_sweeps=15000, n_cand=35, top_k=3)
            if c5_cd800 < c5_800:
                h800 = h_cd800
                c5_800 = c5_cd800

        if tleft() > 120:
            h_pg800, c5_pg800 = projected_gradient_momentum(h800, n800, target800,
                                                              time_limit=min(60, tleft() * 0.2),
                                                              max_steps=80000,
                                                              use_smooth=True, alpha=1000,
                                                              mu=0.85, base_lr=0.015)
            if c5_pg800 < c5_800:
                h800 = h_pg800
                c5_800 = c5_pg800

        if tleft() > 80:
            h_cd800, c5_cd800 = pair_cd(h800, n800, target800,
                                         time_limit=min(60, tleft() * 0.3),
                                         max_sweeps=10000, n_cand=50, top_k=5)
            if c5_cd800 < c5_800:
                h800 = h_cd800
                c5_800 = c5_cd800

        if c5_800 < best_score:
            best_score = c5_800
            best_h = h800

    # Phase 4: Try n=1200 for even higher resolution
    if tleft() > 250:
        n1200 = 1200
        target1200 = n1200 / 2.0
        h1200 = interpolate_to(best_h, n1200, target1200)
        c5_1200 = compute_c5(h1200)

        if tleft() > 180:
            h_pg, c5_pg = projected_gradient_momentum(h1200, n1200, target1200,
                                                       time_limit=min(70, tleft() * 0.25),
                                                       max_steps=50000,
                                                       use_smooth=False, k_top=5,
                                                       mu=0.9, base_lr=0.03)
            if c5_pg < c5_1200:
                h1200 = h_pg
                c5_1200 = c5_pg

        if tleft() > 100:
            h_cd, c5_cd = pair_cd(h1200, n1200, target1200,
                                   time_limit=min(70, tleft() * 0.3),
                                   max_sweeps=8000, n_cand=30, top_k=3)
            if c5_cd < c5_1200:
                h1200 = h_cd
                c5_1200 = c5_cd

        if c5_1200 < best_score:
            best_score = c5_1200
            best_h = h1200

    # Phase 5: SLSQP polish at n=400 (fast enough)
    if tleft() > 120:
        h400 = interpolate_to(best_h, n400, target400)
        h_slq = slsqp_optimize(h400, n400, target400,
                                [200, 1000, 5000],
                                800, 1e-15, min(80, tleft() * 0.4))
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 6: Local perturbation search
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert, c5_pert = local_perturbation(best_h, n_use, target_use,
                                             max_tries=5000,
                                             time_limit=min(40, tleft() * 0.2),
                                             rng_seed=99999)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 7: Final active-set gradient + pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin, c5_fin = projected_gradient_momentum(best_h, n_use, target_use,
                                                     time_limit=min(25, tleft() * 0.3),
                                                     max_steps=50000,
                                                     use_smooth=False, k_top=3,
                                                     mu=0.95, base_lr=0.02)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                               time_limit=min(30, tleft() * 0.4),
                               max_sweeps=10000, n_cand=50, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
