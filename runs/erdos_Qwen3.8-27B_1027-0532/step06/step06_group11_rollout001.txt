def interpolate_to(h_src, n_dst, target_dst):

    h_cur = h_init.copy()

    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.03, 0.05, 0.08, 0.12, 0.18]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [100, 500, 2000, 10000],
                                   2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(120, tleft() * 0.25),
                           max_sweeps=5000, n_cand=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 200:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [10000, 50000, 200000, 1000000],
                                   3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_sg, sc = topk_subgrad(best_h, n, target,
                                time_limit=min(80, tleft() * 0.3),
                                max_steps=50000, n_top_start=5, n_top_end=20,
                                lr=0.01)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    if tleft() > 250:
        n2 = 500
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        h_hires, sc = slsqp_optimize(h_interp, n2, target2,
                                     [200, 1000, 5000, 20000],
                                     2000, 1e-15)
        if sc < best_score:
            h_cd2, sc2 = pair_cd(h_hires, n2, target2,
                                 time_limit=60,
                                 max_sweeps=3000, n_cand=25)
            if sc2 < sc:
                sc = sc2
                h_hires = h_cd2
            best_score = sc
            best_h = h_hires

    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd, sc = pair_cd(best_h, n_cur, target_cur,
                           time_limit=min(100, tleft() * 0.3),
                           max_sweeps=4000, n_cand=35)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pol, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                   [50000, 200000, 1000000, 5000000],
                                   3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.1]:
        if tleft() < 150:
            break
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pert = best_h + rng.randn(n_cur) * noise_scale
        h_pert = project_feasible(h_pert, target_cur)
        h_ref, sc = slsqp_optimize(h_pert, n_cur, target_cur,
                                   [1000, 10000, 50000, 200000],
                                   1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pert, sc = local_perturbation(best_h, n_cur, target_cur,
                                        max_tries=5000,
                                        time_limit=min(40, tleft() * 0.3), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_pert
        if tleft() > 60:
            h_pol, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                       [10000, 50000, 200000, 1000000],
                                       2000, 1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd, sc = pair_cd(best_h, n_cur, target_cur,
                           time_limit=min(60, tleft() * 0.4),
                           max_sweeps=3000, n_cand=40)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc = slsqp_optimize(best_h, n_cur, target_cur,
                                     [50000, 200000, 1000000, 5000000],
                                     2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
