import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _penalty_obj(x, alpha, top_T, target):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top = np.sort(corr)[::-1][:top_T]
    shifted = np.clip(alpha * (top - c_max), -500.0, 0.0)
    penalty = float(np.log(np.sum(np.exp(shifted)) + 1e-300) / alpha)
    return float(c_max + 0.5 * penalty)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, top_T=1):
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(len(x))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lbgfs_smooth(h_init, n, target, time_limit, alphas=(200, 1000, 5000, 20000)):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x):
            h_c = np.clip(x, 0.0, 1.0)
            return _smooth_obj(h_c, alpha, target)
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'gtol': 1e-12, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def perturb_and_refine(h_init, n, target, time_limit, rng, n_perturbs=6):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    start = time.time()
    per_budget = time_limit / n_perturbs
    for p in range(n_perturbs):
        if time.time() - start > time_limit - 10:
            break
        h_pert = h_init + rng.normal(0, 0.05, n)
        h_pert = project_feasible(h_pert, target)
        t_sub = min(per_budget * 0.4, time_limit - (time.time() - start) - 10)
        if t_sub < 5:
            break
        hp, sc = slsqp_round(h_pert, n, target, [500, 5000], 500, 1e-14)
        t_cd = min(per_budget * 0.5, time_limit - (time.time() - start) - 5)
        if t_cd < 5:
            if sc < best_c5:
                best_c5 = sc
                best_h = hp
            continue
        hcd, c5cd = fast_pair_cd(hp, n, target, time_limit=t_cd, n_cand=20, max_sweeps=20000, top_K=2)
        if c5cd < best_c5:
            best_c5 = c5cd
            best_h = hcd
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        hf, scf = fourier_line_search(best_h, n, target, time_limit=min(30, tleft()*0.3))
        if scf < best_score:
            best_score = scf
            best_h = hf

    if tleft() > 80:
        hp, sc = lbgfs_smooth(best_h, n, target, time_limit=min(50, tleft()*0.5), alphas=(500, 3000, 15000))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        cd_time = min(300, tleft() * 0.35)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=100000, top_K=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [2000, 10000, 50000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 200:
        pt = min(200, tleft() * 0.5)
        hptr, scptr = perturb_and_refine(best_h, n, target, time_limit=pt, rng=rng, n_perturbs=5)
        if scptr < best_score:
            best_score = scptr
            best_h = hptr

    if tleft() > 100:
        cd_time = min(150, tleft() * 0.4)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=80000, top_K=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 50:
        hp, sc = lbgfs_smooth(best_h, n, target, time_limit=min(40, tleft()*0.6), alphas=(10000, 50000, 200000))
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 300:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, n_new, target_new, [500, 2000, 10000], 1000, 1e-14)
        if tleft() > 150:
            hcd, c5cd = fast_pair_cd(hp, n_new, target_new, time_limit=min(120, tleft()*0.4), n_cand=25, max_sweeps=40000, top_K=3)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    if tleft() > 80:
        hsg, c5sg = multi_subgrad(best_h, n, target, time_limit=min(50, tleft()*0.4), n_shifts=5, base_lr=0.02)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 40:
        hp, sc = slsqp_round(best_h, n, target, [50000, 200000, 500000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(25, tleft()*0.5), n_cand=35, max_sweeps=30000, top_K=2)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
