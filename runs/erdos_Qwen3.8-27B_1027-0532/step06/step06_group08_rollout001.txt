def sa_refine(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    """Simulated annealing for binary solution"""
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 30000:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        if r < 0.6:
            # Targeted: focus on worst shifts
            top3 = np.argsort(full)[-3:]
            k_idx = top3[rng.randint(len(top3))]
            k_star = int(k_idx) - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                cands_i = ones[A[ones] <= 1.0]
                if len(cands_i) == 0:
                    cands_i = ones
                i = cands_i[rng.randint(len(cands_i))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            # Semi-random with bias
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                # Sample from top 30% of candidates
                thresh = np.percentile(A[ones], 70)
                cands = ones[A[ones] <= thresh]
                if len(cands) == 0:
                    cands = ones
                i = cands[rng.randint(len(cands))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def iterated_local_search(h, n_restarts, time_limit, start_time, rng):
    """ILS: repeatedly perturb and local search"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        
        # Perturb: flip k random bits
        k_perturb = rng.randint(10, 40)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(k_perturb):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        
        # Short greedy + SA
        t_left = time_limit - (time.time() - start_time)
        if t_left < 5:
            break
        t_greedy = min(5.0, t_left * 0.3)
        h_g, c5_g = greedy_refine(h_pert, max_iters=50000, rng=rng,
                                   time_limit=t_greedy, start_time=start_time)
        t_left = time_limit - (time.time() - start_time)
        if t_left < 3:
            break
        t_sa = min(10.0, t_left * 0.7)
        n_sa = int(min(1000000, t_sa * 100000))
        h_sa, c5_sa = sa_refine(h_g, n_sa, 0.01, 0.00001, rng,
                                 time_limit=t_sa, start_time=start_time)
        
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    return best_h, best_c5

def smooth_max(h, alpha):
    """Smooth approximation of max"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

        return h
    if diff > 0:

        total = margins.sum()
        if total > 1e-10:
            h = h + (diff / total) * (1.0 - h)
    else:

        if total > 1e-10:

                for j in top_j:
                    delta = compute_delta(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < max_val - 1e-15:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        max_val = new_max
                        stall = 0
                        new_c5 = max_val * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val - 1e-15:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    max_val = new_max
                    stall = 0
                    new_c5 = max_val * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_refine(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 40000:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        if r < 0.55:
            top3 = np.argsort(full)[-3:]
            k_idx = top3[rng.randint(len(top3))]
            k_star = int(k_idx) - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                cands_i = ones[A[ones] <= 1.0]
                if len(cands_i) == 0:
                    cands_i = ones
                i = cands_i[rng.randint(len(cands_i))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                thresh = np.percentile(A[ones], 60)
                cands = ones[A[ones] <= thresh]
                if len(cands) == 0:
                    cands = ones
                i = cands[rng.randint(len(cands))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * dx

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    exp_w = np.exp(shifted)
    W = exp_w / exp_w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = W[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s == 0:
                grad[:] += wk * (1.0 - h) - wk * h
            else:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:n - s]
        else:
            sp = -s
            if sp == 0:
                grad[:] += wk * (1.0 - h) - wk * h
            else:
                grad[sp:] += wk * (1.0 - h[:n - sp])
                grad[:n - sp] -= wk * h[sp:]
    return grad * dx

def lbfgs_continuous(h_start, alphas, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = project_box_sum(h_start.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        t_left = time_limit - (time.time() - start_time)
        if t_left < 10:
            break
        try:
            res = minimize(
                lambda x: smooth_max_val(x, alpha),
                h_cur,
                jac=lambda x: smooth_max_grad(x, alpha),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-8,
                         'maxfun': 5000, 'disp': False}
            )
        except Exception:
            continue
        h_new = project_box_sum(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
            h_cur = h_new.copy()
    return best_h, best_c5

def topk_gradient_descent(h_start, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 3000
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                if s == 0:
                    grad[:] += (1.0 - h) - h
                else:
                    grad[:n - s] += (1.0 - h[s:])
                    grad[s:] -= h[:n - s]
            else:
                sp = -s
                grad[sp:] += (1.0 - h[:n - sp])
                grad[:n - sp] -= h[sp:]
        grad /= topk
        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def softmax_gradient_descent(h_start, n_iters, alpha, lr_init, lr_end, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 5000
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        c_max = corr.max()
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = W[k]
            if wk < 1e-10:
                continue
            s = shifts[k]
            if s >= 0:
                if s == 0:
                    grad[:] += wk * ((1.0 - h) - h)
                else:
                    grad[:n - s] += wk * (1.0 - h[s:])
                    grad[s:] -= wk * h[:n - s]
            else:
                sp = -s
                grad[sp:] += wk * (1.0 - h[:n - sp])
                grad[:n - sp] -= wk * h[sp:]
        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def periodic_init(n, period, rng):
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 if phase < 0.5 else 0.0
    # The sum might not be exactly n/2, so project
    return project_binary(h, n // 2)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2

    # Try multiple structured initializations
    candidates = [project_binary(h_init, target_init)]
    for period in [3, 4, 5, 6, 7, 8, 10, 12]:
        if period < n_init:
            candidates.append(periodic_init(n_init, period, rng))
    # Random candidates
    for _ in range(3):
        h_rand = (rng.random(n_init) > 0.5).astype(float)
        candidates.append(project_binary(h_rand, target_init))

    # Pick best binary candidate
    best_bin = candidates[0].copy()
    best_bin_score = compute_c5(best_bin)
    for c in candidates[1:]:
        sc = compute_c5(c)
        if sc < best_bin_score:
            best_bin_score = sc
            best_bin = c.copy()

    # Continuous initial from parent
    best_cont = project_feasible(h_init, float(target_init))
    best_cont_score = compute_c5(best_cont)

    # Phase 1: Greedy on best binary
    allowed = min(80.0, tleft() * 0.12)
    if allowed > 15:
        h_g, c5_g = greedy_refine(best_bin, max_iters=100000, rng=rng,
                                   time_limit=allowed, start_time=start_time)
        if c5_g < best_bin_score:
            best_bin_score = c5_g
            best_bin = h_g.copy()

    # Phase 2: SA on best binary
    allowed = min(250.0, tleft() * 0.25)
    if allowed > 30:
        n_sa = int(min(5000000, allowed * 80000))
        h_sa, c5_sa = sa_refine(best_bin, n_sa, 0.03, 0.00003, rng,
                                 time_limit=allowed, start_time=start_time)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_bin = h_sa.copy()

    # Phase 3: ILS with large perturbations
    allowed = min(300.0, tleft() * 0.25)
    if allowed > 30:
        n_restarts = int(min(20, allowed / 15.0))
        for r in range(n_restarts):
            if time.time() - start_time > start_time + allowed:
                break
            t_left = allowed - (time.time() - start_time)
            if t_left < 10:
                break
            k_pert = rng.randint(20, 60)
            h_pert = best_bin.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(k_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            # Short greedy
            t_g = min(8.0, t_left * 0.3)
            h_g2, c5_g2 = greedy_refine(h_pert, max_iters=30000, rng=rng,
                                         time_limit=t_g, start_time=start_time)
            t_left = allowed - (time.time() - start_time)
            if t_left < 5:
                break
            # Short SA
            t_s = min(8.0, t_left * 0.6)
            n_sa2 = int(min(2000000, t_s * 200000))
            h_sa2, c5_sa2 = sa_refine(h_g2, n_sa2, 0.005, 0.00002, rng,
                                       time_limit=t_s, start_time=start_time)
            if c5_sa2 < best_bin_score:
                best_bin_score = c5_sa2
                best_bin = h_sa2.copy()

    # Phase 4: Continuous L-BFGS from best binary
    allowed = min(200.0, tleft() * 0.20)
    if allowed > 20:
        h_l, sc_l = lbfgs_continuous(best_bin, alphas=[30, 100, 300, 1000],
                                      time_limit=allowed, start_time=start_time)
        if sc_l < best_cont_score:
            best_cont_score = sc_l
            best_cont = h_l.copy()

    # Phase 5: Topk gradient from continuous
    allowed = min(100.0, tleft() * 0.12)
    if allowed > 15:
        n_iters = int(min(1000000, allowed * 30000))
        h_pg, c5_pg = topk_gradient_descent(best_cont, n_iters, topk=5,
                                             lr_init=0.05, lr_end=0.001,
                                             time_limit=allowed,
                                             start_time=start_time)
        if c5_pg < best_cont_score:
            best_cont_score = c5_pg
            best_cont = h_pg.copy()

    # Phase 6: Softmax gradient
    allowed = min(100.0, tleft() * 0.12)
    if allowed > 15:
        n_iters = int(min(500000, allowed * 10000))
        h_sg, c5_sg = softmax_gradient_descent(best_cont, n_iters, alpha=50.0,
                                                lr_init=0.03, lr_end=0.0005,
                                                time_limit=allowed,
                                                start_time=start_time)
        if c5_sg < best_cont_score:
            best_cont_score = c5_sg
            best_cont = h_sg.copy()

    # Phase 7: Another L-BFGS round
    allowed = min(tleft() - 30, 150.0)
    if allowed > 30:
        h_l2, sc_l2 = lbfgs_continuous(best_cont, alphas=[200, 500, 2000, 5000],
                                        time_limit=allowed, start_time=start_time)
        if sc_l2 < best_cont_score:
            best_cont_score = sc_l2
            best_cont = h_l2.copy()

    # Phase 8: Multi-resolution
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8, n_init + 16]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_cont, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        # Quick greedy
        t_g3 = min(15.0, tleft() * 0.06)
        if t_g3 < 5:
            continue
        h_g3, c5_g3 = greedy_refine(h_bin_try, max_iters=20000, rng=rng,
                                     time_limit=t_g3, start_time=start_time)
        # L-BFGS
        t_l3 = min(40.0, tleft() * 0.15)
        if t_l3 < 10:
            continue
        h_l3, sc_l3 = lbfgs_continuous(h_g3, alphas=[50, 200, 1000],
                                        time_limit=t_l3, start_time=start_time)
        if sc_l3 < best_cont_score:
            best_cont_score = sc_l3
            best_cont = h_l3.copy()

    # Phase 9: Final topk gradient burst
    if tleft() > 20:
        n_iters_f = int(min(500000, max(50000, (tleft() - 10) * 40000)))
        h_pgf, c5_pgf = topk_gradient_descent(best_cont, n_iters_f, topk=3,
                                               lr_init=0.02, lr_end=0.0001,
                                               time_limit=max(5.0, tleft() - 5),
                                               start_time=start_time)
        if c5_pgf < best_cont_score:
            best_cont_score = c5_pgf
            best_cont = h_pgf.copy()

    # Phase 10: Final L-BFGS
    if tleft() > 15:
        h_ref, sc_ref = lbfgs_continuous(best_cont, alphas=[500, 2000, 10000],
                                          time_limit=max(5.0, tleft() - 5),
                                          start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont = h_ref.copy()

    # Final selection
    n_final = len(best_cont)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_bin, int(n_final / 2.0) if len(best_bin) == n_final else target_init)
    if len(best_bin) == n_final:
        bin_score = compute_c5(h_bin_final)
    else:
        bin_score = 1e9

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
