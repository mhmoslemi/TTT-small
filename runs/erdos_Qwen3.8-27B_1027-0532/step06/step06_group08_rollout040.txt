def project_bisect(h, target_sum):

        return np.clip(h + off, 0.0, 1.0).sum()

            cm = float(corr.max())

            return cm + float(np.log(np.sum(np.exp(sh)))) / alpha
        def con_f(x):
            return np.sum(x) - target
        def con_j(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]

        h_a2, sc_a2 = adam_lse(h_c4, target_init, alpha_sched2,
                                steps_per=6000, lr=0.015, b1=0.85, b2=0.999,
                                eps=1e-8, time_limit=allowed, start_time=t0)
        if sc_a2 < best_sc_cont:
            best_sc_cont = sc_a2
            best_h_cont = h_a2.copy()
            if sc_a2 < best_overall_sc:
                best_overall_sc = sc_a2
                best_overall_h = h_a2.copy()

    # Phase 5: Interactive refinement loop
    for iteration in range(8):
        if tleft() < 100:
            break
        # Perturb continuous, re-optimize
        allowed = min(40.0, tleft() * 0.12)
        if allowed > 15:
            h_pert = best_h_cont.copy()
            noise = rng.randn(len(h_pert)) * 0.05
            h_pert = project_bisect(h_pert + noise, target_init)
            alpha_s3 = [50.0, 200.0, 800.0]
            h_r, sc_r = adam_lse(h_pert, target_init, alpha_s3,
                                  steps_per=3000, lr=0.01, b1=0.9, b2=0.999,
                                  eps=1e-8, time_limit=allowed, start_time=t0)
            if sc_r < best_sc_cont:
                best_sc_cont = sc_r
                best_h_cont = h_r.copy()
                if sc_r < best_overall_sc:
                    best_overall_sc = sc_r
                    best_overall_h = h_r.copy()

        if tleft() < 60:
            break
        # Binary SA with perturbation
        allowed = min(30.0, tleft() * 0.08)
        if allowed > 10:
            h_p = best_h_bin.copy()
            for _ in range(rng.randint(3, 8)):
                ones = np.where(h_p == 1.0)[0]
                zeros = np.where(h_p == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h_p[i] = 0.0
                    h_p[j] = 1.0
            n_sa2 = int(min(1000000, allowed * 30000))
            h_s2, sc_s2 = binary_sa(h_p, n_sa2, 0.01, 0.00001, rng,
                                     time_limit=allowed, start_time=t0)
            if sc_s2 < best_sc_bin:
                best_sc_bin = sc_s2
                best_h_bin = h_s2.copy()
                if sc_s2 < best_overall_sc:
                    best_overall_sc = sc_s2
                    best_overall_h = h_s2.copy()

    # Phase 6: Try different resolutions
    for n_try in [n_init + 4, n_init + 8, n_init - 4, n_init + 16]:
        if tleft() < 80:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = float(n_try) / 2.0
        h_int = interp_to(best_h_cont, n_try, target_try)
        allowed = min(30.0, tleft() * 0.10)
        if allowed < 15:
            continue
        alpha_s4 = [50.0, 200.0]
        h_r4, sc_r4 = adam_lse(h_int, target_try, alpha_s4,
                                steps_per=3000, lr=0.01, b1=0.9, b2=0.999,
                                eps=1e-8, time_limit=allowed, start_time=t0)
        if sc_r4 < best_overall_sc:
            best_overall_sc = sc_r4
            best_overall_h = h_r4.copy()

    # Phase 7: SLSQP polish on best continuous
    if tleft() > 40:
        n_use = len(best_h_cont)
        h_pol, sc_pol = slsqp_polish(best_h_cont, n_use, float(n_use)/2.0,
                                      alphas=[200, 800, 3000],
                                      maxiter=3000, ftol=1e-15,
                                      time_limit=max(20.0, tleft()-10), start_time=t0)
        if sc_pol < best_overall_sc:
            best_overall_sc = sc_pol
            best_overall_h = h_pol.copy()

    # Phase 8: Final Adam burst
    if tleft() > 25:
        alpha_final = [500.0, 2000.0, 5000.0]
        h_f, sc_f = adam_lse(best_overall_h, float(len(best_overall_h))/2.0,
                              alpha_final, steps_per=5000, lr=0.005,
                              b1=0.9, b2=0.999, eps=1e-9,
                              time_limit=max(10.0, tleft()-5), start_time=t0)
        if sc_f < best_overall_sc:
            best_overall_sc = sc_f
            best_overall_h = h_f.copy()

    # Final: project and evaluate
    n_final = len(best_overall_h)
    target_final = float(n_final) / 2.0
    h_final = project_bisect(best_overall_h, target_final)
    final_sc = compute_c5(h_final)

    # Also check binary version
    h_bin_f = project_binary(best_overall_h, int(n_final // 2)) if n_final % 2 == 0 else project_binary(best_overall_h, n_final // 2)
    bin_sc = compute_c5(h_bin_f)

    if bin_sc < final_sc:
        return h_bin_f, bin_sc, len(h_bin_f)
    return h_final, final_sc, n_final
