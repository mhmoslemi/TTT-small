```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def multiplicative_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, lr_init, lr_decay):

    best_h = h.copy()

        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum

        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        
        lr = lr_init / (1.0 + lr_decay * step)
        
        # Multiplicative update in log-odds space
        # h = sigmoid(logit), update logit

        h_new = 1.0 / (1.0 + np.exp(-logit))

        if compute_c5(h_new) < c5_now:
            h = h_new
        else:

            if compute_c5(h_add) < c5_now:
                h = h_add
            else:
                h = project_feasible(h - 0.1 * lr * grad, target)
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -400.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5
```

```python
# For shift k, range k from -(n-1) to (n-1)
# dcorr_p[k] = (1-h[p-k])*[0<=p-k<n] - h[p+k]*[0<=p+k<n]

ks = np.arange(-(n-1), n)  # shape (2n-1,)
# candidates[:, None] - ks[None, :]  → shape (m, 2n-1)
idx_neg = candidates[:, None].astype(int) - ks[None, :]  # p - k
idx_pos = candidates[:, None].astype(int) + ks[None, :]  # p + k

dcorr_matrix = np.zeros((n_cand_actual, 2*n-1))
valid_neg = (idx_neg >= 0) & (idx_neg < n)
valid_pos = (idx_pos >= 0) & (idx_pos < n)
dcorr_matrix[valid_neg] += (1.0 - h[idx_neg[valid_neg]])
dcorr_matrix[valid_pos] -= h[idx_pos[valid_pos]]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -400.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def fast_pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand, max_pairs):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    ks = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = np.max(corr)
        top_count = min(15, 2 * n - 1)
        top_shifts = np.argsort(corr)[-top_count:]
        
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= len(top_shifts)
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        candidates = np.unique(np.concatenate([high_set, low_set]))
        cand_to_idx = {int(c): ci for ci, c in enumerate(candidates)}
        n_cand_actual = len(candidates)
        cand_int = candidates.astype(int)
        
        idx_neg = cand_int[:, None] - ks[None, :]
        idx_pos = cand_int[:, None] + ks[None, :]
        dcorr_matrix = np.zeros((n_cand_actual, 2 * n - 1))
        valid_neg = (idx_neg >= 0) & (idx_neg < n)
        valid_pos = (idx_pos >= 0) & (idx_pos < n)
        dcorr_matrix[valid_neg] += (1.0 - h[idx_neg[valid_neg]])
        dcorr_matrix[valid_pos] -= h[idx_pos[valid_pos]]
        
        pairs = []
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta <= 1e-14:
                    continue
                score = -(grad[j] - grad[i]) * max_delta
                pairs.append((score, int(i), int(j), max_delta))
        
        pairs.sort(reverse=True)
        pairs = pairs[:max_pairs]
        
        best_trial = c5_val
        best_move = None
        
        for score, i, j, max_delta in pairs:
            di = cand_to_idx[i]
            dj = cand_to_idx[j]
            dcorr = dcorr_matrix[dj] - dcorr_matrix[di]
            b_d = max_delta
            a_d = 0.0
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * dcorr
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-15
                if active.any():
                    f_prime = np.max(dcorr[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-15:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
            if trial_c5 < best_trial - 1e-15:
                best_trial = trial_c5
                best_move = (i, j, delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 300 and no_improve > 2000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def multiplicative_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, lr_init, lr_decay):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        
        lr = lr_init / (1.0 + lr_decay * step)
        h_eps = np.clip(h, 1e-6, 1.0 - 1e-6)
        logit = np.log(h_eps / (1.0 - h_eps))
        logit -= lr * grad
        h_new = 1.0 / (1.0 + np.exp(-logit))
        h_new = project_feasible(h_new, target)
        
        if compute_c5(h_new) < c5_now:
            h = h_new
        else:
            h_add = project_feasible(h - lr * 0.5 * grad, target)
            if compute_c5(h_add) < c5_now:
                h = h_add
            else:
                h = project_feasible(h - lr * 0.1 * grad, target)
    
    return best_h, best_c5

def binary_nudge(h_init, n, target, time_limit, rng):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for _ in range(20):
        if time.time() - start > time_limit:
            break
        h_try = h_init.copy()
        threshold = rng.uniform(0.3, 0.7)
        h_try = np.where(h_try > threshold, 1.0, 0.0)
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_try.copy()
    return best_h, best_c5

def diverse_restarts(h_init, n, target, time_limit, rng, n_restarts):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    per_restart = time_limit / max(1, n_restarts)
    
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        h_try = h_init.copy()
        strategy = r % 5
        if strategy == 0:
            h_try += rng.randn(n) * 0.05
        elif strategy == 1:
            h_try += rng.randn(n) * 0.1
        elif strategy == 2:
            block_size = max(1, n // 8)
            offset = int(rng.randint(0, n - block_size))
            h_try[offset:offset + block_size] += rng.choice([-1, 1]) * 0.3
        elif strategy == 3:
            bsize = max(1, n // 6)
            i = int(rng.randint(0, n - 2 * bsize))
            j = int(rng.randint(i + bsize, n - bsize))
            tmp = h_try[i:i + bsize].copy()
            h_try[i:i + bsize] = h_try[j:j + bsize]
            h_try[j:j + bsize] = tmp
        else:
            h_try = np.where(h_init > rng.uniform(0.35, 0.65), 1.0, 0.0)
        
        h_try = project_feasible(h_try, target)
        deadline = min(time.time() + per_restart, start + time_limit)
        h_ref, sc = slsqp_optimize(h_try, n, target,
                                   [500, 2000, 10000],
                                   400, 1e-14)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
