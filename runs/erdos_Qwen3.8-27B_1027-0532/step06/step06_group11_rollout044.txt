```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]

        return h
    h = h.copy()

    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

        else:

        if len(idx2) > 0:

def get_grad_shift(h, k, n):

    g[v2] -= h[np.arange(n)[v2] + k]
```

```python
def lp_subgradient_step(h_cur, n, target, **kwargs):
    """Solve the linearized subproblem."""
    from scipy.optimize import linprog
    
    dx = 2.0 / n
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    n_shifts = 2 * n - 1
    mid = n - 1
    
    # Compute gradients for all shifts
    A = np.zeros((n_shifts, n))
    for s in range(n_shifts):
        k = s - mid
        A[s] = get_grad_shift(h_cur, k, n)
    
    # LP: min t
    # s.t. t - A[s] @ h >= corr[s] - A[s] @ h_cur  for each s
    #      0 <= h_i <= 1
    #      sum(h) = target
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: [0, ..., 0, 1] (minimize t)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A[s] @ h - t <= -(corr[s] - A[s] @ h_cur)
    # i.e., [A[s], -1] @ [h, t] <= A[s] @ h_cur - corr[s]
    
    A_ub = np.zeros((n_shifts, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = A @ h_cur - corr
    
    # Bounds
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = res.x[:n]
            return project_feasible(h_new, target)
    except Exception:
        pass
    
    return h_cur
```

```python
import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_target = 0.0
_G_alpha = 0.0

    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def precompute_f(h, n):
    f = np.zeros((n, 2 * n - 1))
    for p in range(n):
        idx = np.arange(2 * n - 1)
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += (1.0 - h[idx1[mask1]])
        idx2 = p + idx - (n - 1)
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] -= h[idx2[mask2]]
    return f

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 180:
        h_pol, sc = lbfgsb_optimize(best_h, n, target,
                                    [100, 500, 5000, 50000, 500000],
                                    maxiter=5000, time_limit=min(120, tleft() * 0.6))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [2000, 20000, 200000, 2000000],
                                   3000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_lp, sc = lp_active_loop(best_h, n, target,
                                  time_limit=min(180, tleft() * 0.6),
                                  max_iters=5000, n_active=100)
        if sc < best_score:
            best_score = sc
            best_h = h_lp

    if tleft() > 120:
        h_adam, sc = adam_optimize(best_h, n, target,
                                   time_limit=min(80, tleft() * 0.4),
                                   max_steps=50000, n_top_start=3, n_top_end=20,
                                   lr=0.01, beta1=0.8, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    if tleft() > 100:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(90, tleft() * 0.6),
                           max_sweeps=5000, n_cand=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 100:
        h_lp, sc = lp_active_loop(best_h, n, target,
                                  time_limit=min(80, tleft() * 0.5),
                                  max_iters=5000, n_active=60)
        if sc < best_score:
            best_score = sc
            best_h = h_lp

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [10000, 100000, 1000000, 10000000],
                                   2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(80, tleft() * 0.5),
                           max_sweeps=5000, n_cand=40)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 80:
        h_sg, sc = soft_weighted_subgrad(best_h, n, target,
                                         time_limit=min(50, tleft() * 0.4),
                                         max_steps=20000, alpha_w=50.0, base_lr=0.015)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    rng = np.random.RandomState(seed)
    if tleft() > 120:
        h_ms, sc = multi_start_slsqp(best_h, n, target, rng,
                                     time_limit=min(80, tleft() * 0.6), n_starts=3)
        if sc < best_score:
            best_score = sc
            best_h = h_ms

    if tleft() > 150:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = lbfgsb_optimize(h_interp, n_try, target_try,
                                      [100, 1000, 10000, 100000],
                                      maxiter=3000, time_limit=min(100, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try

    if tleft() > 60:
        h_lp, sc = lp_active_loop(best_h, len(best_h), len(best_h) / 2.0,
                                  time_limit=min(50, tleft() * 0.5),
                                  max_iters=3000, n_active=80)
        if sc < best_score:
            best_score = sc
            best_h = h_lp

    if tleft() > 40:
        h_cd, sc = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                           time_limit=min(35, tleft() * 0.5),
                           max_sweeps=3000, n_cand=35)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [50000, 500000, 5000000],
                                     1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
