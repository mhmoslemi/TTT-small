import cvxpy as cp
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_of_overlap_k(h, k, n):
    """Compute gradient of overlap at shift k with respect to h.
    k is the lag: k=0 means no shift, k>0 means positive shift.
    For numpy correlate 'full' mode, output index j corresponds to lag k = j - (n-1).
    """
    grad = np.zeros(n)
    if k >= 0:
        # O = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        for i in range(n - k):
            # dO/dh[i] from (1-h[i]) term: -h[i+k]
            grad[i] -= h[i + k]
            # dO/dh[i+k] from h[i+k] term: (1-h[i])
            grad[i + k] += (1.0 - h[i])
    else:
        m = -k
        # O = sum_{i=m}^{n-1} h[i] * (1-h[i-m])
        for i in range(m, n):
            grad[i] -= h[i - m]
            grad[i - m] += (1.0 - h[i])
    return grad

**LP approach (improved):**

At each iteration:
1. Identify the top-K active overlaps
2. Linearize each active overlap around the current h:
   O_k(h_new) ≈ O_k(h) + grad_k @ (h_new - h)
3. Solve: minimize t subject to:
   O_k(h) + grad_k @ (h_new - h) ≤ t for k in active set
   0 ≤ h_new ≤ 1
   sum(h_new) = n/2
4. Update h = damping * h + (1-damping) * h_new (or just h = h_new)

This is a linear program with K+1 minimization constraints and box/sum constraints. For K=200-400 and n=400, this should be solvable in < 1 second with SCS.

```python
def solving_lp(h_current, n, target, active_indices, dx, solver='SCS', eps=1e-8):
    """Solve the LP relaxation at the current point."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target
    ]
    
    n_half = n - 1
    for idx in active_indices:
        k = idx - n_half
        # Linearize O_k around h_current
        O_k_current = dx * np.correlate(h_current, 1.0 - h_current, mode='full')[idx]
        # Gradient
        grad = np.zeros(n)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] -= h_current[k:]
                grad[k:] += (1.0 - h_current[:n-k])
        else:
            m = -k
            if n - m > 0:
                grad[m:] -= h_current[:n-m]
                grad[:n-m] += (1.0 - h_current[m:])
        # Constraint: O_k_current + grad @ (h_var - h_current) <= t_var
        # O_k_current + grad @ h_var - grad @ h_current <= t_var
        # (O_k_current - grad @ h_current) + grad @ h_var <= t_var
        offset = O_k_current - grad @ h_current
        constraints.append(offset + grad @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, eps=eps, verbose=False)
    except:
        return None
    
    if h_var.value is None:
        return None

The gradient of O_k with respect to h is:
For k ≥ 0: grad[:n-k] -= h[k:]; grad[k:] += (1-h[:n-k])
For k < 0: grad[|k|:] -= h[:n-|k|]; grad[:n-|k|] += (1-h[|k|:])

This can be written as:
grad = convolve(h, something) + convolve(1-h, something)

Specifically, for k ≥ 0:
grad[i] = -h[i+k] (if i < n-k) + (1-h[i-k]) (if i ≥ k)

This is like:
grad = -shift_right(h, k) + shift_left(1-h, k) (with appropriate zero-padding)

Using FFT, we can compute all gradients for all k simultaneously in O(n log n) per "batch"... but actually, each k gives a different shift, so it's not a single convolution.

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, active_idx, n):
    """Compute subgradient of max overlap at active constraints."""
    dx = 2.0 / n
    n_half = n - 1
    g = np.zeros(n)
    for idx in active_idx:
        k = idx - n_half
        if k >= 0:
            g[:n-k] -= h[k:] * dx
            g[k:] += (1.0 - h[:n-k]) * dx
        else:
            m = -k
            g[m:] -= h[:n-m] * dx
            g[:n-m] += (1.0 - h[m:]) * dx
    return g / len(active_idx)

def fw_momentum(h_init, n, target, time_limit, K=50, beta=0.85, max_iter=500000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    best_iter = 0
    no_improve = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        if t % 500 == 0 and time.time() - start > time_limit:
            break
        overlaps = compute_overlaps_array(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = compute_subgrad(h, top_idx, n)
        # FW vertex: minimize g @ h_v s.t. h_v in feasible set
        # The minimizer of g @ h_v over {h_v: 0<=h_v<=1, sum h_v = target}
        # is: set h_v[i] = 1 for the smallest g[i], 0 for the largest, 
        # and one fractional value
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * (v - h)
        gamma = 2.0 / (t + 2)
        h_new = h + gamma * momentum
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_iter = t
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 100000:
            break
    return best_h, best_c5

def lp_round(h_init, n, target, time_limit, K_max=400, max_iter=5000):
    """LP-based optimization with adaptive K."""
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit - 5:
            break
        if it > 50 and stuck > 100:
            break
        
        overlaps = compute_overlaps_array(h)
        n_half = n - 1
        
        # Adaptive K
        progress = min(1.0, it / max(1, int(max_iter * 0.3)))
        K = int(100 + (K_max - 100) * progress)
        
        top_idx = np.argsort(overlaps)[::-1][:K]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target
        ]
        
        for idx in top_idx:
            k = idx - n_half
            O_k = overlaps[idx]
            grad = np.zeros(n)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] -= h[k:] * dx
                    grad[k:] += (1.0 - h[:n-k]) * dx
            else:
                m = -k
                if n - m > 0:
                    grad[m:] -= h[:n-m] * dx
                    grad[:n-m] += (1.0 - h[m:]) * dx
            # Linearized: O_k + grad @ (h_var - h) <= t
            # O_k - grad @ h + grad @ h_var <= t
            offset = O_k - grad @ h
            constraints.append(offset + grad @ h_var <= t_var)
        
        # Add regularization to keep close to current h
        reg = 1e-6 * cp.sum_squares(h_var - h)
        
        prob = cp.Problem(cp.Minimize(t_var + reg), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-9, max_iters=100000)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        # Damping
        if it < 20:
            damp = 0.5
        elif it < 50:
            damp = 0.7
        else:
            damp = 0.9
        h = (1.0 - damp) * h + damp * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5
