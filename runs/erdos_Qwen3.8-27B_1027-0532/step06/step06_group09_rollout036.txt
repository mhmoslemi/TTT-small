    #             corr_k gains h[i-k] [if i-k valid and i-k valid as first index]

def delta_for_swap(h, i, j, n):
    n_corr = 2 * n - 1
    mid = n - 1
    k_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    delta = np.zeros(n_corr)
    
    # First factor effect for i (1->0): -(1-h[i+k]) for k where i is valid first index
    # i valid first index for lag k: k in [-i, n-1-i]
    mask_i1 = (k_vals >= -i) & (k_vals <= n-1-i)
    # i+k must be in [0, n-1]
    ik = i + k_vals
    mask_i1 &= (ik >= 0) & (ik < n)
    delta[mask_i1] -= (1.0 - h[ik[mask_i1]])
    
    # Second factor effect for i (1->0): +h[i-k] for k where i-k is valid first index
    # i-k valid first index for lag k: k in [i-(n-1), i]
    mask_i2 = (k_vals >= i-(n-1)) & (k_vals <= i)
    ikm = i - k_vals
    mask_i2 &= (ikm >= 0) & (ikm < n)
    delta[mask_i2] += h[ikm[mask_i2]]
    
    # First factor effect for j (0->1): +(1-h[j+k])
    mask_j1 = (k_vals >= -j) & (k_vals <= n-1-j)
    jk = j + k_vals
    mask_j1 &= (jk >= 0) & (jk < n)
    delta[mask_j1] += (1.0 - h[jk[mask_j1]])
    
    # Second factor effect for j (0->1): -h[j-k]
    mask_j2 = (k_vals >= j-(n-1)) & (k_vals <= j)
    jkm = j - k_vals
    mask_j2 &= (jkm >= 0) & (jkm < n)
    delta[mask_j2] -= h[jkm[mask_j2]]
    
    return delta
