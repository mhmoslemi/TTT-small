    best_h = h.copy()

        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()

        active = np.where(corr >= c_max - delta_tol)[0]
        
        if len(active) == 0:
            break
        
        # Compute average gradient over active set
        grad_avg = np.zeros(n)
        for si in active:
            k = int(si) - mid
            # gradient of f_k = sum h[i]*(1-h[i+k])
            # d(f_k)/d(h[j]) = (1-h[j+k])*[j+k<n] - h[j-k]*[j>=k]  (for k>0)

        grad_norm = np.linalg.norm(grad_avg)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5


def active_gradient_method(h_init, n, target, time_limit, max_iters=20000, delta_tol_init=1e-3):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    delta_tol = delta_tol_init
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        c_max = np.max(corr)
        active = np.where(corr >= c_max - delta_tol)[0]

        if len(active) == 0:
            break

        g_avg = np.zeros(n)
        for si in active:
            k = int(si) - mid
            ak = abs(k)
            if ak == 0:
                g_avg += (1.0 - 2.0 * h)
            else:
                mask1 = idxs < (n - ak)
                mask2 = idxs >= ak
                g_avg[mask1] += (1.0 - h[idxs[mask1] + ak])
                g_avg[mask2] -= h[idxs[mask2] - ak])

        g_avg /= float(len(active))
        g_avg -= np.mean(g_avg)

        grad_norm = np.linalg.norm(g_avg)
        if grad_norm < 1e-15:
            break

        g_avg /= grad_norm

        pos_mask = g_avg < -1e-10
        neg_mask = g_avg > 1e-10

        eps_max = 0.1
        if pos_mask.any():
            lim = np.min((1.0 - h[pos_mask]) / (-g_avg[pos_mask]))
            if lim < eps_max:
                eps_max = lim
        if neg_mask.any():
            lim = np.min(h[neg_mask] / g_avg[neg_mask])
            if lim < eps_max:
                eps_max = lim

        if eps_max < 1e-14:
            break

        improved = False
        for lr in [eps_max * 0.9, eps_max * 0.7, eps_max * 0.5, eps_max * 0.3,
                   eps_max * 0.2, eps_max * 0.1, eps_max * 0.05]:
            if lr < 1e-14:
                continue
            h_try = h - lr * g_avg
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            delta_tol *= 0.5
            if delta_tol < 1e-7:
                break

    return best_h, best_c5


def interpolated_cd(h, n, target, time_limit, n_interp=200):
    """Try interpolating to lower resolution, optimize, then interpolate back."""
    start = time.time()
    if n <= n_interp:
        return h, compute_c5(h)

    x_src = np.linspace(0.0, 1.0, n)
    x_dst = np.linspace(0.0, 1.0, n_interp)
    h_low = np.interp(x_dst, x_src, h)
    h_low = project_feasible(h_low, n_interp / 2.0)

    h_opt, c5_low = fast_pair_cd(h_low, n_interp, n_interp / 2.0,
                                  time_limit=min(60, time_limit * 0.4),
                                  max_sweeps=3000, n_cand=30, n_bisect=12)

    x_back = np.linspace(0.0, 1.0, n)
    h_back = np.interp(x_back, np.linspace(0.0, 1.0, n_interp), h_opt)
    h_back = project_feasible(h_back, target)

    c5_back = compute_c5(h_back)
    if c5_back < compute_c5(h):
        return h_back, c5_back
    return h, compute_c5(h)


def newton_active_set(h_init, n, target, n_active, max_iters=200, time_limit=60.0):
    """Newton method to equalize active correlations."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        sorted_idx = np.argsort(corr)[::-1]
        active_shifts = sorted_idx[:n_active]
        c_max = corr[sorted_idx[0]]

        m = len(active_shifts)
        F = np.zeros(m + 1)
        J = np.zeros((m + 1, n + 1))

        for idx, si in enumerate(active_shifts):
            F[idx] = corr[int(si)] - c_max
            k = int(si) - mid
            ak = abs(k)
            for j in range(n):
                val = 0.0
                if ak == 0:
                    val = 1.0 - 2.0 * h[j]
                else:
                    if j + ak < n:
                        val += (1.0 - h[j + ak])
                    if j >= ak:
                        val -= h[j - ak]
                J[idx, j] = val
            J[idx, n] = -1.0

        F[m] = np.sum(h) - target
        J[m, :] = 1.0

        sol, residuals, rank, sv = lstsq(J, -F, lapack_driver='gelsy')
        dh = sol[:n]
        dt = sol[n]

        scale = 1.0
        for _ in range(30):
            h_try = h + scale * dh
            h_try = project_feasible(h_try, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            c5_try = float(np.max(corr_try)) * dx
            if c5_try < c5_val:
                h = h_try
                break
            scale *= 0.5

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP for initial polish
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target, [2000, 10000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active gradient method (new approach)
    if tleft() > 150:
        ag_time = min(120, tleft() * 0.2)
        h_ag, c5_ag = active_gradient_method(best_h, n, target,
                                             time_limit=ag_time,
                                             max_iters=20000, delta_tol_init=2e-3)
        if c5_ag < best_score:
            best_score = c5_ag
            best_h = h_ag

    # Phase 3: Pair CD with large candidate set
    if tleft() > 200:
        cd_time = min(200, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=8000, n_cand=60, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Newton active set method
    if tleft() > 100:
        for na in [3, 4, 5, 6]:
            if tleft() < 30:
                break
            h_newton, c5_newton = newton_active_set(best_h, n, target,
                                                    n_active=na,
                                                    max_iters=100,
                                                    time_limit=min(30, tleft() * 0.1))
            if c5_newton < best_score:
                best_score = c5_newton
                best_h = h_newton

    # Phase 5: High-precision SLSQP
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [20000, 80000], 1500, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Another round of active gradient + pair CD
    if tleft() > 150:
        ag_time = min(80, tleft() * 0.2)
        h_ag, c5_ag = active_gradient_method(best_h, n, target,
                                             time_limit=ag_time,
                                             max_iters=15000, delta_tol_init=1e-4)
        if c5_ag < best_score:
            best_score = c5_ag
            best_h = h_ag

    if tleft() > 100:
        cd_time = min(100, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=6000, n_cand=80, n_bisect=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Very high alpha SLSQP for final precision
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [50000, 200000], 1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Interpolated optimization (try lower resolution)
    if tleft() > 120:
        h_interp, c5_interp = interpolated_cd(best_h, n, target,
                                              time_limit=min(80, tleft() * 0.4),
                                              n_interp=300)
        if c5_interp < best_score:
            best_score = c5_interp
            best_h = h_interp
            if tleft() > 60:
                h_cd2, c5_cd2 = fast_pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                             time_limit=min(40, tleft() * 0.3),
                                             max_sweeps=3000, n_cand=40, n_bisect=14)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2

    # Phase 9: Multi-restart with structured perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 50:
                break
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)

            h_try, c5_try = fast_pair_cd(h_pert, n_use, target_use,
                                         time_limit=min(40, tleft() * 0.2),
                                         max_sweeps=2000, n_cand=40, n_bisect=14)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try

    # Phase 10: Final active gradient + Newton polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        ag_time = min(30, tleft() * 0.3)
        h_ag, c5_ag = active_gradient_method(best_h, n_use, target_use,
                                             time_limit=ag_time,
                                             max_iters=10000, delta_tol_init=5e-4)
        if c5_ag < best_score:
            best_score = c5_ag
            best_h = h_ag

        if tleft() > 20:
            h_newton, c5_newton = newton_active_set(best_h, n_use, target_use,
                                                    n_active=4,
                                                    max_iters=50,
                                                    time_limit=min(20, tleft() * 0.3))
            if c5_newton < best_score:
                best_score = c5_newton
                best_h = h_newton

    # Phase 11: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(30, tleft() * 0.5)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=60, n_bisect=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use, [100000], 500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
