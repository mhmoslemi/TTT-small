    n = len(h)
    n_corr = 2 * n - 1
    # G has shape (n_corr, n)
    # For shift k (0-indexed in corr, actual shift = k - (n-1)):
    # G[k, j] = (1-h[j+shift]) if j+shift in [0,n) else 0
    #           - h[j-shift] if j-shift in [0,n) else 0
    # But we need to be careful with the definition.
    # corr[k] = sum_i h[i] * (1-h[i+shift]) where shift = k-(n-1)
    #         = sum_i h[i] - sum_i h[i]*h[i+shift]
    # dcorr[k]/dh[j] = d/dh[j] [sum_i h[i](1-h[i+shift])]
    # For term i: h[i](1-h[i+shift])
    #   If j = i: contributes (1-h[i+shift]) = (1-h[j+shift])
    #   If j = i+shift: contributes -h[i] = -h[j-shift]
    # So G[k,j] = (1-h[j+shift]) - h[j-shift] with boundary conditions
    
    shift = np.arange(n_corr) - (n - 1)  # actual shifts
    G = np.zeros((n_corr, n))
    for k in range(n_corr):
        s = shift[k]
        # (1-h[j+s]) term: valid when 0 <= j+s < n, i.e., -s <= j < n-s
        j_min = max(0, -s)
        j_max = min(n, n - s)
        if j_max > j_min:
            G[k, j_min:j_max] += (1.0 - h[j_min+s:j_max+s])
        # -h[j-s] term: valid when 0 <= j-s < n, i.e., s <= j < n+s
        j_min2 = max(0, s)
        j_max2 = min(n, n + s)
        if j_max2 > j_min2:
            G[k, j_min2:j_max2] -= h[j_min2-s:j_max2-s]
    return G

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m_pos = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m_pos - ks
            idx2 = m_pos + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def binary_local_search(h_init, n, target, time_limit, n_trials=100):
    start = time.time()
    dx = 2.0 / n
    h = np.where(h_init > 0.5, 1.0, 0.0)
    ones_needed = int(target)
    n_ones = int(np.sum(h > 0.5))
    if n_ones < ones_needed:
        frac = h - 0.5
        add_idx = np.argsort(-frac)[:ones_needed - n_ones]
        h[add_idx] = 1.0
    elif n_ones > ones_needed:
        frac = h - 0.5
        rem_idx = np.argsort(frac)[:n_ones - ones_needed]
        h[rem_idx] = 0.0
    h = project_feasible(h, int(target))
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(99)

    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < n_trials or len(zeros) < n_trials:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        improved = False

        picks = min(n_trials, min(len(ones), len(zeros)))
        o_idx = rng.choice(len(ones), size=picks, replace=False)
        z_idx = rng.choice(len(zeros), size=picks, replace=False)

        for oi in o_idx:
            if time.time() - start > time_limit:
                break
            i = int(ones[oi])
            for zi in z_idx:
                j = int(zeros[zi])
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c5_try = compute_c5(h)
                if c5_try < cur_max * dx - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
            h[i] = 1.0
            h[j] = 0.0

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [100, 500, 2000], 2000, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 200:
        tl = min(350, int(tleft() * 0.35))
        hp, sc = lp_multi_pair_cd(best_h, n, target, time_limit=tl, m=20, max_sweeps=20000,
                                  alpha_w=50.0, n_top=80)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 2000, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 150:
        tl = min(250, int(tleft() * 0.3))
        hp, sc = lp_multi_pair_cd(best_h, n, target, time_limit=tl, m=30, max_sweeps=20000,
                                  alpha_w=80.0, n_top=100)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 80:
        tl = min(120, int(tleft() * 0.3))
        hp, sc = fast_pair_cd(best_h, n, target, time_limit=tl, n_cand=30, max_sweeps=50000)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for attempt in range(4):
            if tleft() < 70:
                break
            noise_scale = 0.03 * (attempt + 1)
            noise = rng.normal(0, noise_scale, n)
            h_pert = project_feasible(best_h + noise, target)
            hp, sc = slsqp_round(h_pert, n, target, [500, 5000, 50000], 1000, 1e-15)
            if tleft() > 50:
                hp2, sc2 = lp_multi_pair_cd(hp, n, target, time_limit=min(25, int(tleft() * 0.3)),
                                           m=20, max_sweeps=5000, alpha_w=50.0, n_top=60)
                if sc2 < sc:
                    sc = sc2
                    hp = hp2
            if sc < best_c5:
                best_c5 = sc
                best_h = hp

    if tleft() > 40:
        hp, sc = slsqp_round(best_h, n, target, [200000, 500000, 1000000], 1000, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 30:
        hp, sc = lp_multi_pair_cd(best_h, n, target, time_limit=min(20, int(tleft() * 0.5)),
                                  m=25, max_sweeps=5000, alpha_w=100.0, n_top=60)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    if tleft() > 30:
        hp, sc = binary_local_search(best_h, n, target, time_limit=min(20, int(tleft() * 0.5)),
                                     n_trials=80)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
