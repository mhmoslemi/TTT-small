import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, target):
    """Project to binary with exactly target ones."""
    n = len(h)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    
    grad = np.zeros(n)
    omh = 1.0 - h
    
    for s in range(1, n):
        w = weights[n - 1 + s]
        if w < 1e-10:
            continue
        grad[s:] += w * omh[:n - s]
        grad[:n - s] -= w * h[s:]
    
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-10:
            continue
        grad[:n - p] += w * omh[p:]
        grad[p:] -= w * h[:n - p]
    
    return grad

def local_search_binary(h, target, start_time, time_limit, top_k=5, max_cand=60):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    while time.time() - start_time < time_limit:
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        # Get top-k worst
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Generate candidate pairs focused on worst shifts
        cand_i = []
        cand_j = []
        for idx in top_idx[:3]:
            t = int(idx) - (n - 1)
            if t > 0:
                # corr = sum_{m=t}^{n-1} h[m]*(1-h[m-t])
                # Ones contributing: m where h[m]=1 and h[m-t]=0
                for m in range(t, n, max(1, (n-t)//max_cand)):
                    if h[m] >= 0.5 and h[m-t] < 0.5:
                        cand_i.append(m)
                        cand_j.append(m-t)
            elif t < 0:
                d = -t
                # corr = sum_{m=0}^{n-1-d} h[m]*(1-h[m+d])
                for m in range(0, n-d, max(1, (n-d)//max_cand)):
                    if h[m] >= 0.5 and h[m+d] < 0.5:
                        cand_i.append(m)
                        cand_j.append(m+d)
            # Also add some random candidates
            ri = ones[np.random.randint(len(ones))]
            rj = zeros[np.random.randint(len(zeros))]
            cand_i.append(ri)
            cand_j.append(rj)
        
        # Deduplicate and limit
        pairs = set()
        for a, b in zip(cand_i, cand_j):
            if h[a] >= 0.5 and h[b] < 0.5 and a != b:
                pairs.add((a, b))
        # Add random pairs for diversity
        for _ in range(20):
            a = ones[np.random.randint(len(ones))]
            b = zeros[np.random.randint(len(zeros))]
            if a != b:
                pairs.add((a, b))
        
        pair_list = list(pairs)[:max_cand]
        
        best_improve = 0.0
        best_pair = None
        best_corr_new = None
        
        for (i, j) in pair_list:
            if h[i] < 0.5 or h[j] >= 0.5:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c_new = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(c_new.max())
            improve = max_val - new_max
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, j)
                best_corr_new = c_new
            h[i] = 1.0
            h[j] = 0.0
        
        if best_pair is not None and best_improve > 1e-10:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            corr = best_corr_new
            cur_c5 = float(corr.max()) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def adam_continuous(h0, target, start_time, time_budget, alphas, lr, max_steps_per_alpha=500):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        warmup = min(50, max_steps_per_alpha // 5)
        
        while t < max_steps_per_alpha:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj(h, alpha), smooth_grad(h, alpha)
            
            if t > warmup:
                if val < float(smooth_obj(best_h, alpha)) - 1e-9:
                    stall = 0
                else:
                    stall += 1
                if stall > 200:
                    break
            
            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            actual_lr = lr * min(1.0, t / warmup)
            h = h - actual_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = np.clip(h, 0.0, 1.0)
            # Project sum
            h = project_box_sum(h, target)
        
        # Check binary projection
        h_bin = to_binary(h, int(target))
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        c5_cont = compute_c5(h)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h.copy()
    
    return best_h, best_c5
