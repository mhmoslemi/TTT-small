def slp_step(h_current, n):
    target = n / 2.0
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    
    mid = n - 1
    for k in range(-(n-1), n):
        if k == 0:
            # sum h_i * (1-h_i) = sum h_i - sum h_i^2 ≈ sum h_i - sum(2*h_i*h_cur_i - h_cur_i^2)
            # = sum h_i - 2*sum(h_i*h_cur_i) + sum(h_cur_i^2)
            # Linear in h: sum h_i - 2*sum(h_i*h_cur_i) + const
            linearized = cp.sum(h) - 2*cp.sum(h * h_current) + cp.sum(h_current**2)
        else:
            if k > 0:
                # sum_{i=0}^{n-1-k} h_i * (1-h_{i+k})
                # = sum h_i - sum h_i * h_{i+k}
                # Linearize h_i*h_{i+k} ≈ h_i*h_cur_{i+k} + h_{i+k}*h_cur_i - h_cur_i*h_cur_{i+k}
                idx1 = np.arange(n-k)

                prod_linear = h[idx1] * h_current[idx2] + h[idx2] * h_current[idx1] - h_current[idx1]*h_current[idx2]
                linearized = cp.sum(h[idx1]) - cp.sum(prod_linear)
            else:
                # k < 0, let k' = -k
                k_prime = -k
                # sum_{i=k'}^{n-1} h_i * (1-h_{i-k'})

                linearized = cp.sum(h[idx1]) - cp.sum(prod_linear)
        
        constraints.append(linearized * dx <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or CBC, GLPK

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx2 = np.where(h > 1e-14)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_at_lag(h, k_star, n):
    mid = n - 1
    grad = np.zeros(n)
    if k_star >= mid:
        k = k_star - mid
        if k == 0:
            grad = (1.0 - h) - h
        else:
            for p in range(n - k):
                grad[p] += (1.0 - h[p + k])
            for p in range(k, n):
                grad[p] -= h[p - k]
    else:
        k = mid - k_star
        for p in range(k, n):
            grad[p] += (1.0 - h[p - k])
        for p in range(n - k):
            grad[p] -= h[p + k]
    return grad

def compute_smooth_gradient(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    mid = n - 1
    for m in range(len(corr)):
        if weights[m] < 1e-8:
            continue
        w = weights[m]
        if m >= mid:
            k = m - mid
            if k == 0:
                grad += w * ((1.0 - h) - h)
            else:
                for p in range(n - k):
                    grad[p] += w * (1.0 - h[p + k])
                for p in range(k, n):
                    grad[p] -= w * h[p - k]

            for p in range(k, n):
                grad[p] += w * (1.0 - h[p - k])
            for p in range(n - k):
                grad[p] -= w * h[p + k]
    return grad

    target = n / 2.0
    dx = 2.0 / n
    h_cur = h_init.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        mid = n - 1
        for k in range(-(n-1), n):
            if k == 0:
                lin = cp.sum(h_var) - cp.sum(h_var * h_cur) + cp.sum(h_cur * h_cur) - cp.sum(h_cur)
                # Actually: sum h_i(1-h_i) = sum h_i - sum h_i^2
                # Linearize h_i^2 ≈ 2*h_cur_i*h_i - h_cur_i^2
                # sum h_i - sum(2*h_cur_i*h_i - h_cur_i^2) = sum h_i - 2*sum(h_cur_i*h_i) + sum(h_cur_i^2)
                lin = cp.sum(h_var) - 2.0 * cp.sum(h_cur * h_var) + cp.dot(h_cur, h_cur)
            else:
                if k > 0:
                    idx1 = np.arange(n - k)
                    idx2 = np.arange(k, n)
                    # sum_{i=0}^{n-1-k} h_i*(1-h_{i+k}) = sum h_idx1 - sum h_i*h_{i+k}
                    # Linearize h_i*h_{i+k} ≈ h_i*h_cur_{i+k} + h_{i+k}*h_cur_i - h_cur_i*h_cur_{i+k}
                    prod_lin = (h_var[idx1] * h_cur[idx2] + h_var[idx2] * h_cur[idx1]
                               - np.dot(h_cur[idx1], h_cur[idx2]))
                    lin = cp.sum(h_var[idx1]) - prod_lin
                else:
                    kp = -k
                    idx1 = np.arange(kp, n)
                    idx2 = np.arange(n - kp)
                    prod_lin = (h_var[idx1] * h_cur[idx2] + h_var[idx2] * h_cur[idx1]
                               - np.dot(h_cur[idx1], h_cur[idx2]))
                    lin = cp.sum(h_var[idx1]) - prod_lin
            constraints.append(lin * dx <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS, verbose=False)
