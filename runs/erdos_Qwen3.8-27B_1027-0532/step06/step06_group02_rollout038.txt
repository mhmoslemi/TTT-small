def compute_gradient_topk(h, corr_full, alpha, top_k=64):
    """Compute approximate gradient of smooth max using top-k lags."""
    n = len(h)
    offset = n - 1
    # Compute softmax weights
    shifted = alpha * (corr_full - corr_full.max())
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Get top-k indices
    top_indices = np.argsort(weights)[::-1][:top_k]
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k in top_indices:
        w_k = weights[k]
        if w_k < 1e-10:
            continue
        lag = int(k) - offset
        if lag == 0:
            # (1-h[i]) - h[i] = 1 - 2h[i]
            grad += w_k * (1.0 - 2.0 * h)
        else:
            # (1-h[i+lag]) - h[i-lag]
            # First term: (1-h) shifted by -lag, i.e., (1-h[i+lag])
            # Second term: h shifted by +lag, i.e., h[i-lag]
            # For (1-h[i+lag]): valid i where 0 <= i+lag < n, i.e., -lag <= i < n-lag
            lo1 = max(0, -lag)
            hi1 = min(n, n - lag)
            if lo1 < hi1:
                grad[lo1:hi1] += w_k * one_minus_h[lo1+lag:hi1+lag]
            # For -h[i-lag]: valid i where 0 <= i-lag < n, i.e., lag <= i < n+lag
            lo2 = max(0, lag)
            hi2 = min(n, n + lag)

            break
        lr = lr_base / (1.0 + 0.1 * alpha)
        while time.time() - start_time < time_budget and (time.time() - start_time) < time_budget:
            if time.time() - start_time > time_budget:
                break
            t_step += 1
            corr = full_corr_fast(h)
            grad = compute_grad_topk(h, corr, alpha, top_k)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = h - step
            h_new = project_box_sum(h_new, target)
            h = h_new
            if t_step % 200 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    sc = compute_c5(h)
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    return best_h, best_score

def discrete_sa_2opt(h0, target, time_budget, start_time, rng, t_init=0.02, t_final=0.0001):
    n = len(h0)
    h = h0.astype(float).copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)
    start_phase = time.time()

    while time.time() - start_phase < time_budget:
        if stall > max_stall:
            break
        elapsed = time.time() - start_phase
        ratio = min(1.0, elapsed / time_budget)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if rng.random() < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_swap(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                stall += 1
                continue
            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                stall += 1
                continue
            d1 = compute_delta_swap(h, i1, j1)
            h[i1] = 0.0
            h[j1] = 1.0
            d2 = compute_delta_swap(h, i2, j2)
            h[i2] = 0.0
            h[j2] = 1.0
            delta = d1 + d2
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i1] = 1.0
                h[j1] = 0.0
                h[i2] = 1.0
                h[j2] = 0.0
                stall += 1
    return best_h, best_c5

def focused_discrete_sa(h0, target, time_budget, start_time, rng, t_init=0.015, t_final=0.00005):
    n = len(h0)
    h = h0.astype(float).copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 60000
    positions = np.arange(n)
    start_phase = time.time()

    while time.time() - start_phase < time_budget:
        if stall > max_stall:
            break
        elapsed = time.time() - start_phase
        ratio = min(1.0, elapsed / time_budget)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if rng.random() < 0.5:
            top_k = 8
            top_indices = np.argsort(full)[::-1][:top_k]
            moved = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    delta = compute_delta_swap(h, i, j)
                    new_max = float((full + delta).max())
                    c5_new = new_max * dx
                    delta_c5 = c5_new - current_c5
                    if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        current_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
                            stall = 0
                        else:
                            stall += 1
                    else:
                        stall += 1
                    moved = True
                    break
            if not moved:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_swap(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def make_periodic(n, rng, period=4):
    n_ones_per_period = period // 2
    patterns = [
        [1] * n_ones_per_period + [0] * (period - n_ones_per_period),
        [1] * 1 + [0] * 1 + [1] * (n_ones_per_period - 1) + [0] * (period - n_ones_per_period - 1) if period >= 4 else [1, 0],
    ]
    pattern = patterns[0]
    n_full = (n // period) * period
    h = np.tile(pattern, n // period)
    if n > n_full:
        extra = n - n_full
        h = np.concatenate([h, pattern[:extra]])
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        for idx in ones_idx[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        for idx in zeros_idx[:target - n_ones]:
            h[idx] = 1
    return h

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy().astype(float)
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    count = 0
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        count += 1
    return h_pert

def perturb_block(h, rng, max_block_len=10):
    n = len(h)
    h_pert = h.copy().astype(float)
    blen = rng.randint(3, min(max_block_len, n // 10) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a + blen] = h[a:a + blen][::-1]
    return h_pert

def iterated_local_search(h0, target, time_budget, start_time, rng):
    n = len(h0)
    h = h0.astype(float).copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    start_phase = time.time()

    while time.time() - start_phase < time_budget:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = True
        max_inner = 500
        inner_count = 0
        while improved and inner_count < max_inner:
            if time.time() - start_phase > time_budget:
                return best_h, best_c5
            improved = False
            inner_count += 1
            max_val = float(full.max())
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star == 0:
                break
            positions = np.arange(n)
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            top_i = ones[np.argsort(A[ones])[:min(15, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(15, len(zeros))]]
            found = False
            for i in top_i:
                for j in top_j:
                    delta = compute_delta_swap(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        improved = True
                        found = True
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        break
                if found:
                    break
        if not improved or time.time() - start_phase > time_budget:
            n_pert = rng.randint(3, 8)
            h = perturb_diverse(best_h, rng, n_pert)
            full = full_corr_fast(h)
    sc = compute_c5(best_h)
    if sc < best_c5:
        best_c5 = sc
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 8.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0
    int_target = int(n / 2.0)

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Adam continuous optimization from parent
    if tleft() > 150:
        budget1 = min(250, tleft() * 0.28)
        alpha_sched = [(3, 0.08), (10, 0.12), (30, 0.15), (100, 0.2), (500, 0.25), (3000, 0.2)]
        h1, s1 = adam_optimize(best_h, target, budget1, start_time, alpha_sched, top_k=192, lr_base=0.008)
        if s1 < best_score:
            best_score = s1
            best_h = h1.copy()

    # Phase 2: Binary projection + discrete ILS
    if tleft() > 80:
        h_bin = project_binary(best_h, int_target)
        s_bin = compute_c5(h_bin)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
        budget2 = min(120, tleft() * 0.15)
        h2, s2 = iterated_local_search(h_bin, int_target, budget2, start_time, rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    # Phase 3: Multi-start continuous optimization
    n_starts = 5
    if tleft() > 200:
        per_start = min(60, (tleft() * 0.3) / n_starts)
        for s in range(n_starts):
            if tleft() < 30:
                break
            if s == 0:
                h_start = best_h + rng.randn(n) * 0.03
            elif s == 1:
                h_start = project_binary(best_h, int_target).astype(float) + rng.randn(n) * 0.05
            elif s == 2:
                h_start = make_periodic(n, rng, period=6)
                h_start = project_box_sum(h_start, target) + rng.randn(n) * 0.02
            elif s == 3:
                h_start = perturb_diverse(best_h, rng, 15)
            else:
                h_start = perturb_block(best_h, rng, 15)
            h_start = project_box_sum(h_start, target)
            alpha_sched = [(5, 0.15), (50, 0.25), (500, 0.6)]
            h3, s3 = adam_optimize(h_start, target, per_start, start_time, alpha_sched, top_k=128, lr_base=0.01)
            if s3 < best_score:
                best_score = s3
                best_h = h3.copy()

    # Phase 4: Refined Adam from current best
    if tleft() > 100:
        budget4 = min(250, tleft() * 0.25)
        alpha_sched = [(10, 0.1), (100, 0.2), (1000, 0.35), (10000, 0.35)]
        h4, s4 = adam_optimize(best_h, target, budget4, start_time, alpha_sched, top_k=256, lr_base=0.005)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()

    # Phase 5: Focused binary SA
    if tleft() > 60:
        h_bin5 = project_binary(best_h, int_target)
        budget5 = min(120, tleft() * 0.15)
        h5, s5 = focused_discrete_sa(h_bin5, int_target, budget5, start_time, rng, t_init=0.012, t_final=0.00002)
        if s5 < best_score:
            best_score = s5
            best_h = h5.copy()

    # Phase 6: Another Adam pass with very high alpha (basically min-max)
    if tleft() > 60:
        budget6 = min(120, tleft() * 0.2)
        alpha_sched = [(1000, 0.3), (5000, 0.35), (20000, 0.35)]
        h6, s6 = adam_optimize(best_h, target, budget6, start_time, alpha_sched, top_k=64, lr_base=0.003)
        if s6 < best_score:
            best_score = s6
            best_h = h6.copy()

    # Phase 7: Final discrete polish
    if tleft() > 30:
        h_bin7 = project_binary(best_h, int_target)
        budget7 = max(10, tleft() * 0.25)
        h7, s7 = iterated_local_search(h_bin7, int_target, budget7, start_time, rng)
        if s7 < best_score:
            best_score = s7
            best_h = h7.copy()

    # Phase 8: Final Adam polish
    if tleft() > 20:
        budget8 = max(8, tleft() * 0.6)
        alpha_sched = [(5000, 0.4), (50000, 0.6)]
        h8, s8 = adam_optimize(best_h, target, budget8, start_time, alpha_sched, top_k=64, lr_base=0.002)
        if s8 < best_score:
            best_score = s8
            best_h = h8.copy()

    # Compare continuous vs binary
    h_bin_final = project_binary(best_h, int_target)
    s_bin_final = compute_c5(h_bin_final)
    if s_bin_final < best_score:
        best_score = s_bin_final
        best_h = h_bin_final.copy()

    # Ensure feasibility
    best_h = project_box_sum(best_h, target)
    final_score = compute_c5(best_h)
    if final_score > best_score + 1e-10:
        h_bin_check = project_binary(best_h, int_target)
        s_check = compute_c5(h_bin_check)
        if s_check < final_score:
            best_h = h_bin_check
            final_score = s_check

    return best_h, final_score, n
