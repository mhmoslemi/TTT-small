import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    mid = n - 1

    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')

            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Weighted multi-shift gradient
        c_max = float(np.max(corr))
        # Weight: exp(-beta * (c_max - corr) / dx) but use corr directly

- A "kink" point where the slope changes sign

Alternative: vectorize over pairs. For a fixed set of candidate high indices H and low indices L, we have |H|*|L| pairs. For each pair, we need to find the optimal δ.

If we vectorize the binary search over all pairs simultaneously:
- Start with δ_lo and δ_hi for each pair
- In each iteration, compute δ_mid for all pairs
- For each pair, compute f(δ_mid) = max(corr + δ_mid * g_pair)
- The "active" shift is argmax(corr + δ_mid * g_pair)
- The slope at δ_mid is g_pair[active_shift]
- Update δ_lo or δ_hi based on the sign of the slope

```python
def vectorized_pair_search(corr, change, high_set, low_set, h, n):
    """Find the best (i, j, delta) pair that minimizes max correlation."""
    H = high_set.astype(int)
    L = low_set.astype(int)
    n_h = len(H)
    n_l = len(L)
    n_corr = len(corr)
    
    # Build the change difference matrix: g[i,j] = change[L[j]] - change[H[i]]
    # Shape: (n_h, n_l, n_corr)
    g = change[np.newaxis, L, :] - change[H[:, np.newaxis], :]
    # g has shape (n_h, n_l, n_corr)
    
    # Delta bounds
    delta_lo = np.maximum(h[np.newaxis, H] - 1.0, -h[np.newaxis, L])  # (n_h, n_l)
    delta_hi = np.minimum(h[np.newaxis, H], 1.0 - h[np.newaxis, L])  # (n_h, n_l)
    
    # Mask for valid pairs
    valid = (delta_hi - delta_lo) > 1e-12
    
    if not valid.any():
        return -1, -1, 0.0, float(np.max(corr))
    
    # Binary search over delta for all valid pairs simultaneously
    a = delta_lo.copy()
    b = delta_hi.copy()
    
    for _ in range(16):
        mid = 0.5 * (a + b)
        # vals has shape (n_h, n_l, n_corr)
        # corr has shape (n_corr,)
        vals = corr[np.newaxis, np.newaxis, :] + mid[:, :, np.newaxis] * g
        f_vals = np.max(vals, axis=2)  # (n_h, n_l)
        active = np.argmax(vals, axis=2)  # (n_h, n_l)
        # Slope at mid: g[:, :, active]
        # We need g[i,j,active[i,j]]
        slope = g[np.arange(n_h)[:, np.newaxis], np.arange(n_l)[np.newaxis, :], active]
        
        update = slope > 1e-12
        a = np.where(valid & update, mid, a)
        b = np.where(valid & ~update, mid, b)
    
    delta_star = 0.5 * (a + b)
    # Final evaluation
    vals = corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * g
    f_final = np.max(vals, axis=2)  # (n_h, n_l)
    
    # Find the best
    f_final = np.where(valid, f_final, np.inf)
    idx = np.unravel_index(np.argmin(f_final), f_final.shape)
    
    if f_final.flat[idx] >= float(np.max(corr)):
        return -1, -1, 0.0, float(np.max(corr))
    
    return int(H[idx[0]]), int(L[idx[1]]), float(delta_star[idx]), float(f_final[idx])
```

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def vect_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=60, n_shifts=3,
                 beta_weights=5.0, max_no_improve=500, n_iters_binsearch=16):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Compute gradient using top-m shifts with softmax weights
        c_max = float(np.max(corr))
        # Select top m shifts
        top_m = min(n_shifts, n_corr)
        top_indices = np.argsort(corr)[-top_m:]
        
        # Compute weights
        top_corr = corr[top_indices]
        top_range = c_max - np.min(top_corr)
        if top_range > 1e-10:
            w = np.exp(-beta_weights * (c_max - top_corr) / top_range)
        else:
            w = np.ones(top_m) / top_m
        w = w / np.sum(w)
        
        grad = np.zeros(n)
        for t_idx, s_idx in enumerate(top_indices):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[t_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[t_idx] * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidates
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        # Vectorized pair search
        H = high_set.astype(int)
        L = low_set.astype(int)
        n_h = len(H)
        n_l = len(L)
        
        # g[i,j] = change[L[j]] - change[H[i]], shape (n_h, n_l, n_corr)
        g_change = change[L][:, np.newaxis, :] - change[H][: np.newaxis, :]
        # Wait, this indexing is wrong. Let me fix it.
        # change[H] has shape (n_h, n_corr)
        # change[L] has shape (n_l, n_corr)
        # We want g[i,j] = change[L[j], :] - change[H[i], :]
        # = change[L][j, :] - change[H][i, :]
        # Broadcast: change[L][:, np.newaxis, :] - change[H][:, np.newaxis, :]
        # This gives (n_l, 1, n_corr) - (1, n_h, n_corr) -> NO, that's wrong

        # = (1, n_l, n_corr) - (n_h, 1, n_corr) -> (n_h, n_l, n_corr) ✓
        
        g_change = change[L][np.newaxis, :, :] - change[H][:, np.newaxis, :]
        
        # Delta bounds
        h_H = h[H]  # (n_h,)
        h_L = h[L]  # (n_l,)
        delta_lo = np.maximum(h_H[:, np.newaxis] - 1.0, -h_L[np.newaxis, :])  # (n_h, n_l)
        delta_hi = np.minimum(h_H[:, np.newaxis], 1.0 - h_L[np.newaxis, :])  # (n_h, n_l)
        
        valid = (delta_hi - delta_lo) > 1e-12
        
        if not valid.any():
            continue
        
        # Binary search
        a = delta_lo.copy()
        b = delta_hi.copy()
        
        for _ in range(n_iters_binsearch):
            mid_d = 0.5 * (a + b)
            # vals shape: (n_h, n_l, n_corr)
            vals = corr[np.newaxis, np.newaxis, :] + mid_d[:, :, np.newaxis] * g_change
            f_vals = np.max(vals, axis=2)
            active = np.argmax(vals, axis=2)
            # slope[i,j] = g_change[i, j, active[i,j]]
            # Using fancy indexing
            slope = g_change[np.arange(n_h)[:, np.newaxis], np.arange(n_l)[np.newaxis, :], active]
            
            going_up = slope > 1e-12
            a_new = np.where(valid & going_up, mid_d, a)
            b_new = np.where(valid & ~going_up, mid_d, b)
            a = a_new
            b = b_new
        
        delta_star = 0.5 * (a + b)
        vals_final = corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * g_change
        f_final = np.max(vals_final, axis=2)
        f_final = np.where(valid, f_final, np.inf)
        
        flat_idx = np.argmin(f_final)
        f_min = f_final.flat[flat_idx]
        
        if f_min >= float(np.max(corr)) - 1e-14:
            continue
        
        ii, jj = np.unravel_index(flat_idx, f_final.shape)
        best_i = int(H[ii])
        best_j = int(L[jj])
        best_delta = float(delta_star[ii, jj])
        
        # Apply update
        h[best_i] -= best_delta
        h[best_j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        
        # Update change rows
        for p in [best_i, best_j]:
            change[p, :] = 0.0
            pos = (p + shifts >= 0) & (p + shifts < n)
            neg = (p - shifts >= 0) & (p - shifts < n)
            change[p, pos] += (1.0 - h[p + shifts[pos]])
            change[p, neg] -= h[p - shifts[neg]]
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})

    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)

def build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change, shifts

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

            break
        
        # Weighted gradient from top shifts
        top_m = min(n_shifts, n_corr)
        top_indices = np.argsort(corr)[-top_m:]
        top_corr = corr[top_indices]
        c_max = top_corr[-1]

        else:
            w = np.ones(top_m) / top_m
        w = w / np.sum(w)
        
        grad = np.zeros(n)
        for t_i, s_i in enumerate(top_indices):

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[t_i] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[t_i] * h[idxs[v2] + k]

        # Progressive candidate set
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)

        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        L = low_set
        n_h = len(H)
        n_l = len(L)
        
        # Vectorized pair search
        g_ch = change[L][np.newaxis, :, :] - change[H][:, np.newaxis, :]
        
        h_H = h[H]
        h_L = h[L]
        d_lo = np.maximum(h_H[:, np.newaxis] - 1.0, -h_L[np.newaxis, :])
        d_hi = np.minimum(h_H[:, np.newaxis], 1.0 - h_L[np.newaxis, :])
        valid = (d_hi - d_lo) > 1e-12
        
        if not valid.any():
            continue
        
        a = d_lo.copy()
        b = d_hi.copy()
        
        for _ in range(n_bin):
            md = 0.5 * (a + b)
            vals = corr[None, None, :] + md[:, :, None] * g_ch
            f_v = np.max(vals, axis=2)
            act = np.argmax(vals, axis=2)
            sl = g_ch[np.arange(n_h)[:, None], np.arange(n_l)[None, :], act]
            going_up = sl > 1e-12
            a = np.where(valid & going_up, md, a)
            b = np.where(valid & ~going_up, md, b)
        
        d_star = 0.5 * (a + b)
        vals_f = corr[None, None, :] + d_star[:, :, None] * g_ch
        f_final = np.max(vals_f, axis=2)
        f_final = np.where(valid, f_final, np.inf)
        
        best_pos = np.argmin(f_final)
        f_min = f_final.flat[best_pos]
        
        c_max_curr = float(np.max(corr))
        if f_min >= c_max_curr - 1e-14:

        bi = int(H[ii])
        bj = int(L[jj])
        bd = float(d_star[ii, jj])
        
        h[bi] -= bd
        h[bj] += bd
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        update_change_row(change, bi, h, n, shifts)
        update_change_row(change, bj, h, n, shifts)
    
    return best_h, best_c5

    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change, shifts


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]


def vect_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=25, n_cand_end=70, n_shifts=3,
                 beta_w=5.0, max_no_improve=700, n_bin=16):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change, shifts = build_change(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        top_m = min(n_shifts, n_corr)
        top_indices = np.argsort(corr)[-top_m:]
        top_corr = corr[top_indices]
        c_max_val = float(top_corr[-1])
        rng = c_max_val - float(top_corr[0])
        if rng > 1e-10:
            w = np.exp(-beta_w * (c_max_val - top_corr) / rng)
        else:
            w = np.ones(top_m) / top_m
        w = w / np.sum(w)

        grad = np.zeros(n)
        for t_i in range(top_m):
            k = int(top_indices[t_i]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[t_i] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[t_i] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        H = high_set
        L = low_set
        n_h = len(H)
        n_l = len(L)

        g_ch = change[L][np.newaxis, :, :] - change[H][:, np.newaxis, :]

        h_H = h[H]
        h_L = h[L]
        d_lo = np.maximum(h_H[:, np.newaxis] - 1.0, -h_L[np.newaxis, :])
        d_hi = np.minimum(h_H[:, np.newaxis], 1.0 - h_L[np.newaxis, :])
        valid = (d_hi - d_lo) > 1e-12

        if not valid.any():
            continue

        a = d_lo.copy()
        b = d_hi.copy()

        for _ in range(n_bin):
            md = 0.5 * (a + b)
            vals = corr[None, None, :] + md[:, :, None] * g_ch
            act = np.argmax(vals, axis=2)
            sl = g_ch[np.arange(n_h)[:, None], np.arange(n_l)[None, :], act]
            going_up = sl > 1e-12
            a = np.where(valid & going_up, md, a)
            b = np.where(valid & ~going_up, md, b)

        d_star = 0.5 * (a + b)
        vals_f = corr[None, None, :] + d_star[:, :, None] * g_ch
        f_final = np.max(vals_f, axis=2)
        f_final = np.where(valid, f_final, np.inf)

        best_pos = int(np.argmin(f_final))
        f_min = f_final.flat[best_pos]

        if f_min >= c_max_val - 1e-14:
            continue

        ii, jj = np.unravel_index(best_pos, f_final.shape)
        bi = int(H[ii])
        bj = int(L[jj])
        bd = float(d_star[ii, jj])

        h[bi] -= bd
        h[bj] += bd
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        update_change_row(change, bi, h, n, shifts)
        update_change_row(change, bj, h, n, shifts)

    return best_h, best_c5


def stochastic_restart(h_init, n, target, time_limit, seed_val, noise_scale):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h_pert = h_init + rng.randn(n) * noise_scale
    h_pert = project_feasible(h_pert, target)

    if time.time() - start > time_limit * 0.3:
        return h_pert, compute_c5(h_pert)

    remaining = time_limit - (time.time() - start)
    if remaining > 30:
        h_pol = slsqp_round(h_pert, n, target,
                            [200, 1000, 5000],
                            min(800, int(remaining * 3)))
        sc = compute_c5(h_pol)
        if sc < compute_c5(h_pert):
            return h_pol, sc
    return h_pert, compute_c5(h_pert)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    n_best = n0

    # Phase 1: Vectorized pair CD at n=400
    if tleft() > 280:
        h_cd, c5_cd = vect_pair_cd(best_h, n0, target0,
                                   time_limit=min(250, tleft() * 0.75),
                                   max_sweeps=50000,
                                   n_cand_start=25, n_cand_end=70,
                                   n_shifts=3, beta_w=5.0,
                                   max_no_improve=800, n_bin=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: SLSQP polish
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n_best, target0,
                            [100, 500, 2000, 8000, 30000],
                            500)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Pair CD with different parameters
    if tleft() > 200:
        h_cd2, c5_cd2 = vect_pair_cd(best_h, n_best, target0,
                                     time_limit=min(170, tleft() * 0.75),
                                     max_sweeps=50000,
                                     n_cand_start=30, n_cand_end=80,
                                     n_shifts=4, beta_w=8.0,
                                     max_no_improve=1000, n_bin=18)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 4: Interpolate to n=800
    n1 = 800
    target1 = n1 / 2.0
    h800 = None
    score800 = float('inf')

    if tleft() > 300:
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h_cd1, c5_cd1 = vect_pair_cd(h1, n1, target1,
                                         time_limit=min(220, tleft() * 0.65),
                                         max_sweeps=15000,
                                         n_cand_start=20, n_cand_end=50,
                                         n_shifts=3, beta_w=5.0,
                                         max_no_improve=500, n_bin=16)
            if c5_cd1 < sc1:
                sc1 = c5_cd1
                h1 = h_cd1

        if tleft() > 80:
            h_pol1 = slsqp_round(h1, n1, target1,
                                 [300, 1500, 6000],
                                 250)
            sc1 = compute_c5(h_pol1)
            h1 = h_pol1

        if tleft() > 100:
            h_cd1b, c5_cd1b = vect_pair_cd(h1, n1, target1,
                                           time_limit=min(80, tleft() * 0.6),
                                           max_sweeps=10000,
                                           n_cand_start=25, n_cand_end=55,
                                           n_shifts=4, beta_w=6.0,
                                           max_no_improve=400, n_bin=16)
            if c5_cd1b < sc1:
                sc1 = c5_cd1b
                h1 = h_cd1b

        h800 = h1
        score800 = sc1

    # Phase 5: Random restarts at n=400
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        for trial in range(4):
            if tleft() < 40:
                break
            noise = 0.05 + 0.04 * trial
            h_r, sc_r = stochastic_restart(best_h, n_best, target0,
                                           time_limit=min(35, tleft() * 0.5),
                                           seed_val=1000 + trial * 7,
                                           noise_scale=noise)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    # Phase 6: If n=800 is better, try more refinement there
    if h800 is not None and score800 < best_score and tleft() > 120:
        if tleft() > 100:
            h_cd1c, c5_cd1c = vect_pair_cd(h800, n1, target1,
                                           time_limit=min(80, tleft() * 0.6),
                                           max_sweeps=10000,
                                           n_cand_start=30, n_cand_end=60,
                                           n_shifts=5, beta_w=7.0,
                                           max_no_improve=400, n_bin=18)
            if c5_cd1c < score800:
                score800 = c5_cd1c
                h800 = h_cd1c
        if tleft() > 50:
            h_pol2 = slsqp_round(h800, n1, target1,
                                 [5000, 20000],
                                 200)
            sc2 = compute_c5(h_pol2)
            if sc2 < score800:
                score800 = sc2
                h800 = h_pol2

    # Final selection: choose the best (h, n) pair
    final_h = best_h
    final_score = best_score
    final_n = n_best

    if h800 is not None and score800 < final_score:
        final_h = h800
        final_score = score800
        final_n = n1

    final_h = project_feasible(final_h, final_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, final_n
