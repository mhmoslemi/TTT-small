def fourier_perturb(h, n, rng, num_freqs=10, amplitude=0.1):
    """Perturb h by adding low-frequency Fourier components."""
    x = np.linspace(0, 2, n, endpoint=False)
    perturbation = np.zeros(n)
    for _ in range(num_freqs):
        freq = rng.randint(1, 6)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.01, amplitude)
        perturbation += amp * np.cos(2*np.pi*freq*x/2.0 + phase)
    h_new = h + perturbation
    return project_feasible(h_new, n/2.0)

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_iteration(h, n, target, active_k_indices, reg, dx):
    half_n = n - 1
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in sorted(active_k_indices):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    h_new = None
    try:
        prob.solve(solver=cp.SCS, eps=1e-11, max_iters=100000, verbose=False)
        if h_var.value is not None:
            h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    except Exception:
        pass
    if h_new is None:
        try:
            prob.solve(solver=cp.PDDP, verbose=False)
            if h_var.value is not None:
                h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        except Exception:
            pass
    if h_new is None:
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000)
            if h_var.value is not None:
                h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        except Exception:
            pass
    return h_new


def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0,
                   active_tol, max_active):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    current_tol = active_tol

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_current = float(np.max(corr)) * dx
        active_mask = (corr * dx) > (t_current - current_tol)
        active_k_indices = np.where(active_mask)[0]
        if len(active_k_indices) > max_active:
            active_k_indices = np.argsort(corr)[-max_active:]
        elif len(active_k_indices) < 5:
            active_k_indices = np.argsort(corr)[-10:]

        h_new_lp = solve_lp_iteration(h, n, target, active_k_indices, current_reg, dx)
        if h_new_lp is None:
            break

        h = (1.0 - damping) * h + damping * h_new_lp
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.96)
        else:
            damping = min(0.99, damping + 0.02)

        if it % 20 == 0 and current_reg > 1e-10:
            current_reg = max(1e-10, current_reg * 0.75)
        if it % 40 == 0:
            current_tol = max(1e-5, current_tol * 0.85)

    return best_h, best_c5


def power_mean_obj(h, n, target, p, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = np.maximum(corr * dx, 0.0)
    return float(np.sum(overlaps**p)**(1.0/p))


def power_mean_cons_fun(h, n, target):
    return np.array([np.sum(h) - target])


def power_mean_cons_jac(h, n, target):
    return np.array([np.ones(n)])


def power_mean_optimize(h_start, n, target, p_values, maxiter, ftol, time_limit):
    start = time.time()
    dx = 2.0 / n
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    for p in p_values:
        if time.time() - start > time_limit:
            break
        _global_n = n
        _global_target = target

        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            overlaps = np.maximum(corr * dx, 0.0)
            return float(np.sum(overlaps**p)**(1.0/p))

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(_global_n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    return best_h, best_score


def coord_descent_max(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            C = slopes
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = C * mid + intercepts
                max_val = np.max(vals)
                tol = 1e-12 * max(1.0, abs(max_val))
                active = np.abs(vals - max_val) < tol
                if not np.any(active):
                    active = np.zeros(len(C), dtype=bool)
                    active[np.argmax(vals)] = True
                if np.all(C[active] > 1e-14):
                    hi = mid
                elif np.all(C[active] < -1e-14):
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        active_k = new_active

    return best_h, best_c5


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break

    return best_h, best_c5


def fourier_perturb(h, n, rng, num_freqs, amplitude):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    perturbation = np.zeros(n)
    for _ in range(num_freqs):
        freq = rng.randint(1, 8)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.01, amplitude)
        perturbation += amp * np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
    h_new = h + perturbation
    return project_feasible(h_new, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
