import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

    dx = 2.0 / n
    g = 1.0 - h_old
    h_var = cp.Variable(n)

        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1

        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def outer_approximation(h_init, n, target, time_limit, max_iters=500):
    """
    Outer approximation (Dinkelbach-type) for the bilinear program.
    """
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    damping = 0.8
    reg = 1e-3
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        h_new = solve_bilinear_lp(h, n, target, reg)
        if h_new is None:
            break
        
        # Damped update
        h_trial = (1.0 - damping) * h + damping * h_new
        h_trial = project_feasible(h_trial, target)
        
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            stall_count = 0
            damping = min(1.0, damping * 1.05)  # slightly increase
        else:
            stall_count += 1
            if stall_count > 20:
                damping = max(0.3, damping * 0.9)
            h = h_trial  # still update the linearization point
        
        # Decay regularization
        if it % 10 == 0:
            reg = max(1e-6, reg * 0.7)
    
    return best_h, best_c5

def smooth_maxmin_lbfgsb(h_init, n, target, time_limit, alphas=None):
    """
    Maximize min_k R(k) where R(k) = autocorrelation at lag k.
    Equivalently, minimize max_k (1 - R(k)) = max_k I(k).
    Uses smooth log-sum-exp approximation with L-BFGS-B.
    """
    if alphas is None:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000]
    
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        time_left = time_limit - (time.time() - start)
        if time_left < 5:
            break
        
        # Define smooth objective
        def obj(h_flat):
            hp = np.clip(h_flat, 0, 1)
            # Compute overlap integrals
            corr = np.correlate(hp, 1.0 - hp)
            # Smooth max
            c_max = np.max(corr)
            if c_max > 0:
                shifted = np.clip(alpha * (corr - c_max), -500, 0)
                val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            else:
                val = c_max
            return val * dx
        
        def obj_grad(h_flat):
            hp = np.clip(h_flat, 0, 1)
            corr = np.correlate(hp, 1.0 - hp)
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500, 0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            
            # Gradient of smooth max of corr with respect to h
            # d/dh_i corr[k] = (1-h[i+k])*dx - h[i-k]*dx (for valid indices)
            grad_corr = np.zeros(n)
            for k_idx in range(2*n-1):
                k = k_idx - (n - 1)
                w = weights[k_idx]
                if w < 1e-10:
                    continue
                i_lo = max(0, -k)
                i_hi = min(n, n - k)
                # d corr[k_idx] / d h[i] = (1-h[i+k]) - h[i-k] for valid i
                # (contribution from h[i] term: 1-h[i+k])
                # (contribution from (1-h[i]) term in corr): -h[i-k]
                for i in range(i_lo, i_hi):
                    grad_corr[i] += w * dx * (1.0 - hp[i+k])
                # Now the (1-h) part: (1-h[i]) contributes -h[i-k]
                # Actually, corr[k_idx] = sum_i h[i]*(1-h[i+k])
                # d/dh[j] corr[k_idx] = (1-h[j+k]) if j is in the h part
                #                     + (-h[j-k]) if j is in the (1-h) part
                # For the (1-h) part, j-indexed: (1-h[j]) appears when i+k = j, i.e., i = j-k
                # So d/dh[j] has contribution -h[j-k] if j-k is valid (0 <= j-k < n) AND
                # the term h[j-k]*(1-h[j]) exists, which requires 0<=j-k<n and 0<=j<n
                i_lo2 = max(0, -k)  # i = j - k must be in [0, n)
                # j must satisfy: j-k in [i_lo, i_hi) and j in [0, n)
                j_lo = max(0, i_lo + k)
                j_hi = min(n, i_hi + k)
                for j in range(j_lo, j_hi):
                    grad_corr[j] += w * dx * (-hp[j-k])
            
            return grad_corr * dx
        
        # Use L-BFGS-B with equality constraint via penalty
        # Actually, let's eliminate one variable: fix h[0] and express sum constraint
        # Or better: use a penalty for the sum constraint
        
        def penalized_obj(h_flat):
            base = obj(h_flat)
            penalty = 1000.0 * (np.sum(h_flat) - target) ** 2 / n
            return base + penalty
        
        def penalized_grad(h_flat):
            base_g = obj_grad(h_flat)
            penalty_g = 2000.0 * (np.sum(h_flat) - target) / n
            return base_g + penalty_g * np.ones(n)
        
        # Multiple restart attempts
        for attempt in range(3):
            if time.time() - start > time_limit:
                break
            t0 = time.time()
            try:
                res = minimize(penalized_obj, h, jac=penalized_grad,
                             method='L-BFGS-B',
                             bounds=[(0,1)]*n,
                             options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12,
                                      'maxfun': 5000})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
            except:
                pass
        
        # Simple coordinate descent refinement after each alpha
        for i in range(n):
            if time.time() - start > time_limit:
                break
            # Try values around current
            for delta in [0.01, -0.01, 0.005, -0.005, 0.001, -0.001]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + delta, 0, 1)
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    break
    
    return best_h, best_c5

def coord_descent_fast(h_init, n, target, n_active, time_limit, rng):
    """Fast coordinate descent focusing on active constraints."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for pass_num in range(10000):
        if time.time() - start > time_limit:
            break
        
        # Update active set periodically
        if pass_num % 10 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            active_k = np.argsort(corr)[-n_active:]
        
        order = rng.permutation(n)
        for i in order:
            # Compute the contribution of h[i] to each active constraint
            # For active_k[idx] = k_idx, k = k_idx - half_n
            # Contribution: (1-h[i+k])*dx - h[i-k]*dx
            # Total objective = max over active k of (corr[k_idx] + contribution_i - h[i]*contribution_i)
            # Actually: corr[k_idx] = sum_j h[j]*(1-h[j+k])*dx
            # d corr[k_idx] / d h[i] = (1-h[i+k])*dx - h[i-k]*dx (if valid)
            
            slopes = np.zeros(len(active_k))
            intercepts = np.zeros(len(active_k))
            
            for idx in range(len(active_k)):
                k_idx = active_k[idx]
                k = k_idx - half_n
                s = 0.0
                if 0 <= i + k < n:
                    s += (1.0 - h[i+k])
                if 0 <= i - k < n:
                    s -= h[i-k]
                s *= dx
                slopes[idx] = s
                intercepts[idx] = corr[k_idx] - s * h[i]
            
            # Minimize max_k (slopes[k]*h[i] + intercepts[k])
            # This is a 1D minimax: find h[i] in [0,1] that minimizes the max
            # The optimal is where the largest and smallest slopes meet
            # Or at a boundary [0, 1]
            
            # Solve: minimize max_k (s_k * t + b_k) for t in [0,1]
            # The function is piecewise linear in t
            # Optimal t is where the upper envelope has a minimum
            
            # Simple approach: ternary search or binary search on the derivative
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = slopes * mid + intercepts
                max_val = np.max(vals)
                active_mask = np.abs(vals - max_val) < 1e-12 * max(1.0, abs(max_val))
                if not np.any(active_mask):
                    active_mask = np.zeros(len(slopes), dtype=bool)
                    active_mask[np.argmax(vals)] = True
                avg_slope = np.mean(slopes[active_mask])
                if avg_slope > 1e-14:
                    hi = mid  # decreasing, go left
                elif avg_slope < -1e-14:
                    lo = mid  # increasing, go right
                else:
                    break
            h[i] = 0.5 * (lo + hi)
        
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def subgradient_refine(h_init, n, target, time_limit, max_steps=5000):
    """Subgradient method with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        
        # Subgradient of I(k) with respect to h
        grad = np.zeros(n)
        mask1 = (idxs >= -k) & (idxs + k < n)
        mask2 = (idxs >= 0) & (idxs - k < n) & (idxs - k >= 0)
        # d I(k)/d h[i] = (1-h[i+k])*dx - h[i-k]*dx
        for mask, sign in [(mask1, 1.0), (mask2, -1.0)]:
            if not np.any(mask):
                continue
            if sign > 0:
                idxs_valid = idxs[mask]
                j = idxs_valid + k
                grad[idxs_valid] += (1.0 - h[j]) * dx
            else:
                idxs_valid = idxs[mask]
                j = idxs_valid - k
                grad[idxs_valid] -= h[j] * dx
        
        # Line search
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            h_try = project_feasible(h - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve(h_lin, n, target, reg=0.0):
    """Solve the linearized problem."""
    dx = 2.0 / n
    g = 1.0 - h_lin
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    half = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - half
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi <= i_lo:
            continue
        coeff = np.zeros(n)
        coeff[i_lo:i_hi] = g[i_lo + k:i_hi + k] * dx
        if np.any(coeff > 1e-15):
            constraints.append(coeff @ h_var <= t_var)
    
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h_lin)
    else:
        objective = t_var

        if gap < -1e-6:
            gap = 0.0  # LP is conservative

        if gap < 0.001:
            step = 1.0
        elif gap < 0.01:
            step = 0.8
        else:
            step = damping

        h_trial = project_feasible(h_trial, target)

            h = h_trial
            stall += 1
            if stall > 30:

        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve_reduced(h_lin, n, target, k_indices, reg=0.0):
    dx = 2.0 / n
    g = 1.0 - h_lin
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    half = n - 1
    for k_idx in k_indices:
        k = k_idx - half
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi <= i_lo:
            continue
        coeff = np.zeros(n)
        coeff[i_lo:i_hi] = g[i_lo + k:i_hi + k] * dx
        if np.any(coeff > 1e-15):
            constraints.append(coeff @ h_var <= t_var)
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h_lin)
    else:
        objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def coord_descent(h_init, n, target, n_active, time_limit, rng):

def smooth_obj_alpha(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = 500.0 * (np.sum(h) - target) ** 2 / n
    return val * dx + penalty

def smooth_grad_alpha(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    half = n - 1
    grad = np.zeros(n)
    active_idx = np.where(weights > 1e-10)[0]
    for k_idx in active_idx:
        k = k_idx - half
        w = weights[k_idx]
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        for i in range(i_lo, i_hi):
            grad[i] += w * dx * (1.0 - h[i + k])
        j_lo = max(0, i_lo + k)
        j_hi = min(n, i_hi + k)
        for j in range(j_lo, j_hi):
            grad[j] += w * dx * (-h[j - k])
    penalty_grad = 1000.0 * (np.sum(h) - target) / n
    grad += penalty_grad
    return grad * dx

def slsqp_smooth(h_init, n, target, alphas, maxiter, ftol, time_limit):

        return smooth_grad_alpha(np.clip(hf, 0, 1), alphas[-1], n, target)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(obj, h, jac=jfun, method='SLSQP', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})

def subgradient_refine(h_init, n, target, time_limit, max_steps=3000):

        k_star = int(np.argmax(corr)) - half
        k = k_star
        grad = np.zeros(n)
        mask1 = (idxs >= -k) & (idxs + k < n)
        mask2 = (idxs >= 0) & (idxs - k < n)
        if np.any(mask1):
            j = idxs[mask1] + k
            grad[mask1] += (1.0 - h[j]) * dx
        if np.any(mask2):
            j = idxs[mask2] - k
            grad[mask2] -= h[j] * dx
        improved = False

                best_c5 = sc

                stall = 0
                break

                break

    h_final = project_feasible(best_h, tgt_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
