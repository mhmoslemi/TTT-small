```python
    for k_idx, w in enumerate(weights):
        if w < 1e-15:
            continue
        k = k_idx - mid
        if k >= 0:

            for i in range(n - k):
                grad[i+k] += w * (1 - h[i])
                grad[i] -= w * h[i+k]
        else:
            kk = -k
            for i in range(n - kk):
                grad[i] += w * (1 - h[i+kk])
                grad[i+kk] -= w * h[i]
    
    return grad * (2.0 / n)
```

```python
def compute_g(h, i, j, n):
    """Compute g = A_j - A_i, the change in corr per unit transfer from i to j"""
    n_corr = 2*n - 1
    mid = n - 1
    g = np.zeros(n_corr)
    
    # For k >= 0 (lags 0 to n-1), index mid+k
    ks = np.arange(0, n)
    # g[mid+k] = [k<=j]*(1-h[j-k]) - [k<=n-1-j]*h[j+k] - [k<=i]*(1-h[i-k]) + [k<=n-1-i]*h[i+k]
    
    mask_j1 = ks <= j
    mask_j2 = ks <= (n-1-j)
    mask_i1 = ks <= i
    mask_i2 = ks <= (n-1-i)
    
    val = np.zeros(n)
    val[mask_j1] += (1.0 - h[j - ks[mask_j1]])
    val[mask_j2] -= h[j + ks[mask_j2]]
    val[mask_i1] -= (1.0 - h[i - ks[mask_i1]])
    val[mask_i2] += h[i + ks[mask_i2]]
    g[mid:mid+n] = val
    
    # For k < 0 (lags -1 to -(n-1)), index mid+k = mid-kk where kk = 1..n-1
    kks = np.arange(1, n)
    # g[mid-kk] = [kk<=n-1-j]*(1-h[j+kk]) - [kk<=j]*h[j+kk] - [kk<=n-1-i]*(1-h[i+kk]) + [kk<=i]*h[i+kk]
    
    mask_j1 = kks <= (n-1-j)
    mask_j2 = kks <= j
    mask_i1 = kks <= (n-1-i)
    mask_i2 = kks <= i
    
    val = np.zeros(n-1)
    val[mask_j1] += (1.0 - h[j + kks[mask_j1]])
    val[mask_j2] -= h[j + kks[mask_j2]]
    val[mask_i1] -= (1.0 - h[i + kks[mask_i1]])
    val[mask_i2] += h[i + kks[mask_i2]]
    g[:mid] = val  # indices 0 to mid-1 correspond to lags -(n-1) to -1
    
    return g
```

Actually, here's another idea: **proximal gradient method**. The function f(h) = max_k corr_k(h) * dx is the max of bilinear functions. We can write:
f(h) = max_k (a_k^T h - h^T B_k h) * dx

where a_k and B_k depend on k. This is a DC (difference of convex) function in a sense.

The proximal gradient step would be:
h_{t+1} = prox_g( h_t - lr * grad_f(h_t) )

where g is the indicator of the constraint set {h ∈ [0,1]^n : ∑h = n/2}.

2. h_new = h - lr * subgrad
3. h_new = project_feasible(h_new, target)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lbfags_optimize(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n

        return smooth_obj(x, alpha)

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def compute_g_vec(h, i, j, n):

    n_corr = 2 * n - 1
    mid = n - 1

    if m2.any():

    m3 = ks <= i
    if m3.any():
        val[m3] -= (1.0 - h[i - ks[m3]])
    m4 = ks <= (n - 1 - i)
    if m4.any():
        val[m4] += h[i + ks[m4]]
    g[mid:] = val
    
    # Negative lags k = -1, ..., -(n-1) at indices 0, 1, ..., n-2
    kks = np.arange(1, n)
    val2 = np.zeros(n - 1)
    
    m1 = kks <= (n - 1 - j)
    if m1.any():
        val2[m1] += (1.0 - h[j + kks[m1]])
    m2 = kks <= j
    if m2.any():
        val2[m2] -= h[j + kks[m2]]
    m3 = kks <= (n - 1 - i)
    if m3.any():
        val2[m3] -= (1.0 - h[i + kks[m3]])
    m4 = kks <= i
    if m4.any():
        val2[m4] += h[i + kks[m4]]
    g[:mid] = val2
    
    return g

def find_optimal_delta(corr, g, delta_max):
    """Find delta that minimizes max(corr + delta*g) on [0, delta_max]"""
    if delta_max < 1e-12:
        return 0.0
    
    lo = 0.0

        elif g_at_max < -1e-14:

            n_cand=25, patience=300):

    h = h.copy()

    best_h = h.copy()
```

corr_orig = np.correlate([0.2, 0.5, 0.8], [0.8, 0.5, 0.2], 'full')

I'll compute this by hand:
corr[0]: overlap of a[0] with v[0]: a[0]*v[0] = 0.2*0.8 = 0.16
corr[1]: a[0]*v[1] + a[1]*v[0] = 0.2*0.5 + 0.5*0.8 = 0.1 + 0.4 = 0.5
corr[2]: a[0]*v[2] + a[1]*v[1] + a[2]*v[0] = 0.2*0.2 + 0.5*0.5 + 0.8*0.8 = 0.04 + 0.25 + 0.64 = 0.93
corr[3]: a[1]*v[2] + a[2]*v[1] = 0.5*0.2 + 0.8*0.5 = 0.1 + 0.4 = 0.5
corr[4]: a[2]*v[2] = 0.8*0.2 = 0.16

So corr_orig = [0.16, 0.5, 0.93, 0.5, 0.16]
