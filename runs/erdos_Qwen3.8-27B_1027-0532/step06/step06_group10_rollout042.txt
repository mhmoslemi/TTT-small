def project_feasible(h, target_sum):

        return h
    if diff > 0:
        # Need to increase some elements
        # Prioritize elements that are farthest from 1
        order = np.argsort(1.0 - h)
        idx = order[:int(abs(diff) * 2 + 1)]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    else:

            for s_idx in top_k_shifts:
                k = int(s_idx) - mid
                v1m = (idxs - k >= 0) & (idxs - k < n)
                v2m = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1m):
                    grad_multi[v1m] += (1.0 - h[idxs[v1m] - k])
                if np.any(v2m):
                    grad_multi[v2m] -= h[idxs[v2m] + k]
            grad_multi /= len(top_k_shifts)
            grad_multi -= np.mean(grad_multi)
            sorted_multi = np.argsort(grad_multi)
            lm = sorted_multi[:min(n_cand, 20)]
            hm = sorted_multi[-min(n_cand, 20):]
            for i in hm:
                for j in lm:
                    if i == j:
                        continue
                    d_hi = min(h[i], 1.0 - h[j])
                    if d_hi < 1e-12:
                        continue
                    g = np.zeros(n_corr)
                    v_ip = (i + shifts >= 0) & (i + shifts < n)
                    g[v_ip] += h[i + shifts[v_ip]]
                    v_im = (i - shifts >= 0) & (i - shifts < n)
                    g[v_im] += h[i - shifts[v_im]]
                    v_jp = (j + shifts >= 0) & (j + shifts < n)
                    g[v_jp] -= h[j + shifts[v_jp]]
                    v_jm = (j - shifts >= 0) & (j - shifts < n)
                    g[v_jm] -= h[j - shifts[v_jm]]
                    a_d = 0.0
                    b_d = d_hi
                    for _ in range(18):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-13
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-13:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_approx:
                        best_approx = approx_max
                        best_delta = delta_star
                        best_i = i
                        best_j = j
            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
    return best_h, best_c5

def perturb_and_optimize(h0, n, target, time_limit, start_time, n_perturb, sigma, cd_time_each):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    rng = np.random.default_rng(777)
    for p in range(n_perturb):
        if time.time() - start_time > time_limit:
            break
        h_pert = h0 + sigma * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, target)
        h_cd, c5_cd = greedy_cd(h_pert, n, target, time_limit=cd_time_each,
                                 start_time=start_time, max_sweeps=800, n_cand=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    return best_h, best_c5

def active_set_lp(h0, n, target, time_limit, n_active, max_rounds, start_time):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad_k[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var / dx)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        improved = False
        if c5_lp < c5_val - 1e-10:
            h = h_lp
            improved = True
        else:
            for mix_frac in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix_frac * h + (1.0 - mix_frac) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def subgradient_descent(h0, n, target, time_limit, start_time, max_steps=20000, n_top=8):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    v = np.zeros(n)
    no_improve = 0
    lr = 0.05

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
            lr = max(0.01, lr * 1.05)
        else:
            no_improve += 1
            if no_improve > 30:
                lr = lr * 0.7
            if no_improve > 500:
                break

        sorted_idx = np.argsort(corr)[::-1][:n_top]
        alpha_w = 20.0
        c_max = corr[sorted_idx[0]]
        w = np.exp(alpha_w * (corr[sorted_idx] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, sorted_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) * (step + 1))) / 2.0
        beta = (step - 1) / t_new if step > 1 else 0.0

        h_try = project_feasible(h + beta * v - lr * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_now:
            v = beta * v + grad
            h = h_try
        else:
            h_try2 = project_feasible(h - lr * 0.1 * grad, target)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < c5_now:
                v = 0.5 * v + grad
                h = h_try2
            else:
                v = 0.8 * v
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP with moderate alpha
    if tleft() > 120:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [50, 200, 1000, 5000, 20000, 80000],
                                3000, 1e-14, min(100.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Aggressive CD
    if tleft() > 200:
        cd_time = min(250.0, tleft() * 0.35)
        h_cd, c5_cd = greedy_cd(best_h, n, target, time_limit=cd_time,
                                 start_time=start_time, max_sweeps=30000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with high alpha
    if tleft() > 120:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [500, 2000, 10000, 50000, 200000],
                                2500, 1e-15, min(100.0, tleft() * 0.3), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: Subgradient with momentum
    if tleft() > 150:
        h_sg, c5_sg = subgradient_descent(best_h, n, target,
                                          time_limit=min(120.0, tleft() * 0.3),
                                          start_time=start_time, max_steps=15000, n_top=10)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Perturbation-based escape
    if tleft() > 200:
        for sigma in [0.08, 0.12, 0.04, 0.15, 0.03]:
            if tleft() < 30:
                break
            h_pb, s_pb = perturb_and_optimize(best_h, n, target,
                                               time_limit=min(40.0, tleft() * 0.25),
                                               start_time=start_time,
                                               n_perturb=5, sigma=sigma,
                                               cd_time_each=6.0)
            if s_pb < best_score:
                best_score = s_pb
                best_h = h_pb

    # Phase 6: CD with large candidates
    if tleft() > 150:
        h_cd, c5_cd = greedy_cd(best_h, n, target, time_limit=min(150.0, tleft() * 0.35),
                                 start_time=start_time, max_sweeps=25000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: LP refinement
    if tleft() > 100:
        h_lp, c5_lp = active_set_lp(best_h, n, target,
                                     time_limit=min(100.0, tleft() * 0.3),
                                     n_active=12, max_rounds=35, start_time=start_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 8: CD after LP
    if tleft() > 120:
        h_cd, c5_cd = greedy_cd(best_h, n, target, time_limit=min(120.0, tleft() * 0.35),
                                 start_time=start_time, max_sweeps=20000, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Multi-resolution attempt
    if tleft() > 250:
        for n_try in [800, 1200]:
            if tleft() < 120:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_sr, s_sr = slsqp_round(h_interp, n_try, target_try,
                                      [50, 200, 1000, 5000, 20000],
                                      2000, 1e-14,
                                      min(100.0, tleft() * 0.3), start_time)
            if s_sr < best_score:
                best_score = s_sr
                best_h = h_sr
            if tleft() < 80:
                break
            n_use = len(best_h)
            target_use = float(n_use) / 2.0
            h_cd2, c5_cd2 = greedy_cd(best_h, n_use, target_use,
                                       time_limit=min(80.0, tleft() * 0.25),
                                       start_time=start_time, max_sweeps=10000, n_cand=25)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 10: Final SLSQP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_round(best_h, n_use, target_use,
                                         [2000, 10000, 50000, 200000, 800000],
                                         2000, 1e-15, min(40.0, tleft() * 0.6), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 11: Final CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = greedy_cd(best_h, n_use, target_use,
                                 time_limit=min(15.0, tleft() * 0.7),
                                 start_time=start_time, max_sweeps=8000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
