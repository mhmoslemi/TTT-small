def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr

    result[mask] -= h[idx[mask]]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0

        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def nesterov_refine(h0, target, start_time, time_limit, alpha_schedule):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(40.0, time_limit - (time.time() - start_time))
        n_iter = max(100, int(phase_time * 150))
        lr = 1.0 / (alpha * 0.1 + 1.0)
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            grad = smooth_gradient(h, alpha, corr)
            
            # Nesterov step
            v = momentum * v - lr * grad
            h_new = project_box_sum(h + v, target)
            
            # Backtracking line search
            h_trial = h_new
            for _ in range(5):
                sc = compute_c5(h_trial)
                if sc < best_score or True:  # accept if improved or first try
                    break
                lr *= 0.5
                v = v * 0.5
                h_trial = project_box_sum(h + v, target)
            
            h = h_trial
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    
    return best_h, best_score

def block_optimize(h, rng, time_limit, start_time, k_arr, n_blocks, block_len):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    
    for _ in range(n_blocks):
        if time.time() - start_time > time_limit:
            break
        a = rng.randint(0, n - block_len + 1)
        b = a + block_len - 1
        block = h[a:b+1].copy()
        n_ones_in_block = int(round(block.sum()))
        
        # Enumerate all assignments with same number of 1s
        # For block_len=10 and n_ones=5, C(10,5)=252 candidates
        from itertools import combinations
        positions_in_block = list(range(block_len))
        best_local = None
        best_local_delta = 0
        current_corr = full.copy()
        
        # Get the current contribution of the block to the correlation
        # For each candidate, compute the delta
        for combo in combinations(positions_in_block, n_ones_in_block):
            new_block = np.zeros(block_len)
            new_block[list(combo)] = 1.0
            if np.array_equal(new_block, block):
                continue
            # Compute delta
            delta = np.zeros(2 * n - 1)
            for pos in range(block_len):
                old_val = block[pos]
                new_val = new_block[pos]
                if old_val == new_val:
                    continue
                global_pos = a + pos
                if old_val == 1.0:
                    # 1 -> 0
                    idx = global_pos + k_arr
                    mask = (idx >= 0) & (idx < n)
                    delta[mask] += h[idx[mask]]
                    idx = global_pos - k_arr
                    mask = (idx >= 0) & (idx < n)
                    delta[mask] += h[idx[mask]]
                else:
                    # 0 -> 1
                    idx = global_pos - k_arr
                    mask = (idx >= 0) & (idx < n)
                    delta[mask] -= h[idx[mask]]
                    idx = global_pos + k_arr
                    mask = (idx >= 0) & (idx < n)
                    delta[mask] -= h[idx[mask]]
            new_max = float((full + delta).max())
            if new_max < float(full.max()) - 1e-12:
                if best_local is None or new_max < best_local_delta:
                    best_local = new_block
                    best_local_delta = new_max
        
        if best_local is not None:
            h[a:b+1] = best_local
            # Recompute full correlation (simpler and safer)
            full = full_corr_fast(h)
            c5_new = float(full.max()) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5
