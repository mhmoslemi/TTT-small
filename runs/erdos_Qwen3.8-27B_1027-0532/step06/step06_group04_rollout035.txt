```python
def lse_gradient(h, alpha, dx, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    # Numerical stability
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # weights has length 2n-1, indexed from -(n-1) to n-1
    # weight at shift k is weights[k + n - 1]
    
    grad = np.zeros(n)
    for m in range(n):
        # First term: sum_j weights[m-j+n-1] * (1-h[j]) for valid indices
        s1 = 0.0
        s2 = 0.0
        for j in range(n):
            idx1 = m - j + n - 1  # index into weights for shift m-j
            if 0 <= idx1 < 2*n-1:
                s1 += weights[idx1] * (1.0 - h[j])
            idx2 = j - m + n - 1  # index into weights for shift j-m
            if 0 <= idx2 < 2*n-1:
                s2 += weights[idx2] * h[j]
        grad[m] = dx * (s1 - s2)
    
    # Add penalty gradient
    grad += 2.0 * penalty * (np.sum(h) - target)
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for exact feasibility
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(h, alpha, dx, target, penalty):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    f = (max_s + np.log(sum_exp)) / alpha
    weights = exp_s / sum_exp
    
    # Gradient computation (vectorized)
    m_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    idx1 = m_idx - j_idx + (n - 1)
    idx2 = j_idx - m_idx + (n - 1)
    mask1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
    mask2 = (idx2 >= 0) & (idx2 < 2 * n - 1)
    W1 = np.where(mask1, weights[np.clip(idx1, 0, 2*n-2)], 0.0)
    W2 = np.where(mask2, weights[np.clip(idx2, 0, 2*n-2)], 0.0)
    first = W1 @ (1.0 - h)
    second = W2 @ h
    grad = dx * (first - second) + 2.0 * penalty * (np.sum(h) - target)
    return f, grad

def lbfgs_round(h_init, n, target, alphas, penalty):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            f, _ = lse_objective_and_grad(x, alpha, dx, target, penalty)
            return f
        
        def grad(x):
            _, g = lse_objective_and_grad(x, alpha, dx, target, penalty)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score
```

```python
def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True

            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break

        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 5e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.98, damping + 0.03)
        if it % 60 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_lbfgs_with_proj(h_init, n, target, alpha, penalty, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    def obj(x):
        return lse_obj_grad(x, alpha, dx, target, penalty)[0]

    def grad_fn(x):
        return lse_obj_grad(x, alpha, dx, target, penalty)[1]

    try:
        res = minimize(obj, h_cur, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h_cur, cur_score


def multi_alpha_lbfgs(h_init, n, target, time_limit):
    start = time.time()
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    alphas = [15, 40, 100, 250, 600, 1500, 4000, 10000, 25000]
    penalty = 5000.0
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxit = min(500, max(50, int(remaining * 3)))
        h_cur, sc = smooth_lbfgs_with_proj(h_cur, n, target, alpha, penalty, maxit)
        if sc < cur_score:
            cur_score = sc
            h_best = h_cur.copy()
        else:
            h_cur = project_feasible(h_cur, target)
    return h_cur, min(cur_score, compute_c5(h_cur))
```
