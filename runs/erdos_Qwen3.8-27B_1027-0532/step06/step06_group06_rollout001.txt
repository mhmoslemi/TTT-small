- For continuous h, it's a non-linear program (since C[k] is quadratic in h)

Actually, for binary h, C[k] is linear in h (since h² = h for binary). So the problem becomes:
min t
s.t. C[k] ≤ t for all k
     Σ h[i] = n/2
     h[i] ∈ {0,1}

This is an integer linear program! We could potentially solve it with a MILP solver, but n=606 is likely too large.

However, we can use the LP relaxation:
min t
s.t. C[k] ≤ t for all k
     Σ h[i] = n/2
     0 ≤ h[i] ≤ 1

This is a linear program! (C[k] is linear in h when we treat h[i]h[j] as... wait, no. C[k] = Σ_i h[i](1-h[i+k]) = Σ_i h[i] - Σ_i h[i]h[i+k]. The second term is NOT linear.

So it's not a standard LP. But for the smoothed objective, we can use the approach we already have.

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum != target_sum:
        if current_sum > target_sum:
            ones = np.where(h_r == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:current_sum - target_sum]:
                h_r[ones[idx]] = 0
        else:
            zeros = np.where(h_r == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:target_sum - current_sum]:
                h_r[zeros[idx]] = 1
    return h_r.astype(float)

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def soft_max_gradient(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    one_minus_h = 1.0 - h
    t = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - t + n - 1
        idx2 = t - j + n - 1
        v1 = (idx1 >= 0) & (idx1 < 2*n-1)
        v2 = (idx2 >= 0) & (idx2 < 2*n-1)
        grad[j] = np.dot(one_minus_h, weights[idx1[v1]]) - np.dot(h, weights[idx2[v2]])
    grad *= dx
    if penalty > 0:

    return grad

    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_obj = soft_max_objective(h_cur, beta, target, penalty)
    
    for _ in range(3):
        if time.time() - t0 > time_budget:
            break
        try:
            res = minimize(soft_max_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=soft_max_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})

            obj_new = soft_max_objective(h_new, beta, target, penalty)
            if obj_new < best_obj:
                h_cur = h_new
                best_obj = obj_new
        except Exception:
            pass
    return h_cur

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, t0, time_budget):

def interpolate_to(h_src, n_dst, target_dst):

            tl = min(25, tleft() * 0.25)
            h_opt = lbfgs_optimize(h_r, beta_r, target, penalty=1000.0,
                                   maxiter=200, time_budget=tl, t0=t0)
            h_r = h_opt
            sc = compute_c5(h_r)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_r.copy()

    # Phase 7: Try different grid sizes
    n_options = [n + 2, n - 2, n + 4, n - 4, n + 6, n - 6]
    n_options = [x for x in n_options if x >= 100 and x % 2 == 0 and x != n]
    for n_try in n_options:
        if tleft() < 40:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        tl = min(25, tleft() * 0.15)
        h_opt = lbfgs_optimize(h_interp, 50, target_try, penalty=500.0,
                               maxiter=200, time_budget=tl, t0=t0)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()
            n = n_try
            target = target_try
            target_int = target_try_int
        if tleft() < 25:
            continue
        h_bt = project_binary(h_opt if sc < best_c5 else best_h,
                              target_int if sc < best_c5 else target_int)
        if len(h_bt) != target_int * 2:
            h_bt = project_binary(best_h, target_int)
        tl2 = min(20, tleft() * 0.12)
        h_ls3, c5_ls3 = precise_binary_search(h_bt, int(len(h_bt) / 2),
                                               t0=t0, time_budget=tl2)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3.copy()
            n = len(best_h)
            target = float(n / 2)
            target_int = int(n / 2)

    # Phase 8: Final precision binary polish
    if tleft() > 30:
        h_bf = project_binary(best_h, int(len(best_h) / 2))
        n_use = len(h_bf)
        tl = min(50, tleft() - 8)
        h_pf, c5_pf = precise_binary_search(h_bf, n_use // 2, t0=t0, time_budget=tl)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_pf.copy()

    # Phase 9: Final continuous polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_fc = project_box_sum(best_h.copy(), target_use)
        for beta_f in [500, 1500, 5000]:
            if tleft() < 8:
                break
            tl = min(18, tleft() * 0.4)
            h_pol = lbfgs_optimize(h_fc, beta_f, target_use, penalty=3000.0,
                                   maxiter=150, time_budget=tl, t0=t0)
            h_fc = h_pol
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
