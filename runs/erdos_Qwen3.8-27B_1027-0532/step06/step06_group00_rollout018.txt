def smooth_obj_with_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weights are underflowed, use argmax
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = w / w_sum
    
    obj = float(c_max + np.log(w_sum) / alpha)
    
    # Gradient
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    
    # Only process significant weights
    threshold = 1e-8
    significant = w > threshold
    for k_idx in np.where(significant)[0]:
        ki = int(k_idx) - mid
        wi = w[k_idx]
        if ki >= 0:
            # k > 0
            v1 = slice(0, n - ki)
            v2 = slice(ki, n)
            grad[v1] += wi * (1.0 - x[v2])
            grad[v2] -= wi * x[v1]
        elif ki < 0:
            kp = -ki
            v1 = slice(0, n - kp)
            v2 = slice(kp, n)
            grad[v2] += wi * (1.0 - x[v1])
            grad[v1] -= wi * x[v2]
        else:
            grad += wi * (1.0 - 2.0 * x)
    
    return obj, grad
