    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()

        else:

        if len(idx2) > 0:

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        k_lo = -j
        k_hi = n - 1 - j
        if k_hi >= k_lo:

def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:

            break

            res = minimize(smooth_obj, h_cur, args=(alpha,), jac=lambda h, a: smooth_grad(h, a),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

Actually, I realize I should think more carefully about what "focal" optimization means. The idea is:

At the current point, identify the top-k shifts (those with the highest correlation). Then minimize the sum (or max) of just these k shifts, while keeping the other shifts from getting too large.

A simple way to do this: minimize the average of the top-k shifts, but add a penalty if any other shift exceeds the current max - epsilon.

Or even simpler: just minimize the sum of the top-k shifts using SLSQP, and check if the overall max decreased.

Let me implement this:

```python
def focal_obj(h, k_list):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr[k_list])) * dx / len(k_list)

def focal_grad(h, k_list):
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    grad = np.zeros(n)
    for idx in k_list:
        k = int(idx) - mid
        if k >= 0:
            # d corr[k+mid] / d h[j] = (1-h[j+k])*I[0<=j+k<n] - h[j-k]*I[0<=j-k<n]
            j_range1 = np.arange(n - k)
            grad[j_range1] += (1.0 - h[j_range1 + k])
            j_range2 = np.arange(k, n)
            grad[j_range2] -= h[j_range2 - k]
        else:
            kk = -k
            j_range1 = np.arange(kk, n)
            grad[j_range1] += (1.0 - h[j_range1 - kk])
            j_range2 = np.arange(n - kk)
            grad[j_range2] -= h[j_range2 + kk]
    grad *= dx / len(k_list)
    return grad
```

A simple approach:
1. For each j, compute the derivative of C5 with respect to h[j] (using the smooth gradient)
2. If the derivative is negative, pushing h[j] up (toward 1) helps
3. If the derivative is positive, pushing h[j] down (toward 0) helps
4. Push in the helpful direction until h[j] hits 0 or 1 or C5 starts increasing

This is essentially a projected gradient method with a "hard" projection to {0,1} at the end.

Actually, a simpler binarization: for each j, if h[j] > 0.5, try setting h[j] = 1 (and adjust another variable to maintain the constraint). If h[j] < 0.5, try setting h[j] = 0. Accept the change if it doesn't increase C5.

This is similar to the pair CD but with a binary target.

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        k_lo = -j
        k_hi = n - 1 - j
        if k_hi >= k_lo:
            grad[j] += np.dot(weights[k_lo + mid : k_hi + mid + 1], one_minus_h[j : j + k_hi - k_lo + 1])
        k_lo2 = j - (n - 1)
        k_hi2 = j
        if k_hi2 >= k_lo2:
            grad[j] -= np.dot(weights[k_lo2 + mid : k_hi2 + mid + 1], h[(j - k_hi2) : (j - k_lo2) + 1])
    grad *= dx
    return grad

def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

def focal_obj(h, k_list):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr[k_list])) * dx / len(k_list)

def focal_grad(h, k_list):
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    grad = np.zeros(n)
    for idx in k_list:
        k = int(idx) - mid
        if k >= 0:
            j1 = np.arange(n - k)
            grad[j1] += (1.0 - h[j1 + k])
            j2 = np.arange(k, n)
            grad[j2] -= h[j2 - k]
        else:
            kk = -k
            j1 = np.arange(kk, n)
            grad[j1] += (1.0 - h[j1 - kk])
            j2 = np.arange(n - kk)
            grad[j2] -= h[j2 + kk]
    grad *= dx / len(k_list)
    return grad
