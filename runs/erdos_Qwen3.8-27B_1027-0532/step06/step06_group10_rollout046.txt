    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[idxs[v1]] += (1.0 - h[idxs[v1] - k_star])
        grad[idxs[v2]] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = -h[i]
                b_d = 1.0 - h[j]
                if a_d >= b_d:
                    continue
                for _ in range(30):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def multi_top_cd(h0, n, target, time_limit, max_sweeps=3000, n_top=5, n_cand=25):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit / 2:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        top_k = np.argsort(corr)[::-1][:n_top]
        w_top = np.exp(20.0 * (corr[top_k] - corr[top_k[0]]))
        w_top = w_top / w_top.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            np.add.at(grad, idxs[v1], wi * (1.0 - h[idxs[v1] - k]))
            np.add.at(grad, idxs[v2], -wi * h[idxs[v2] + k])
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        k_star = int(top_k[0]) - mid
        n_corr = 2 * n - 1
        shifts = np.arange(-(n - 1), n)
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = -h[i]
                b_d = 1.0 - h[j]
                if a_d >= b_d:
                    continue
                for _ in range(28):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def randomized_restart(h0, n, target, seed, n_restarts=3, time_per_restart=30.0):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_restarts):
        pert = rng.uniform(-0.05, 0.05, n)
        h_pert = project_feasible(h0 + pert, target)
        alphas = [100, 500, 2000]
        h_new, c5_new = lbfgs_round(h_pert, target, alphas, 200, time_per_restart, time.time())
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    def tleft():
        return max(0.0, deadline - time.time())

    if tleft() < 30:
        return best_h, best_score, n

    # Phase 1: L-BFGS-B with increasing alpha at base resolution
    if tleft() > 200:
        alphas = [30, 100, 300, 1000, 3000, 10000]
        h_p, s_p = lbfgs_round(best_h, target, alphas, 500, tleft() * 0.35, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Multi-top CD
    if tleft() > 250:
        cd_time = min(200.0, tleft() * 0.3)
        h_cd, c5_cd = multi_top_cd(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=10000, n_top=4, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Pair CD
    if tleft() > 200:
        cd_time = min(150.0, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                               max_sweeps=8000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: L-BFGS-B refinement with higher alpha
    if tleft() > 150:
        alphas = [500, 2000, 8000, 30000, 100000]
        h_p, s_p = lbfgs_round(best_h, target, alphas, 400, tleft() * 0.3, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 5: Randomized restarts
    if tleft() > 120:
        h_rr, s_rr = randomized_restart(best_h, n, target, seed=seed,
                                         n_restarts=3, time_per_restart=min(30.0, tleft() * 0.08))
        if s_rr < best_score:
            best_score = s_rr
            best_h = h_rr

    # Phase 6: Another CD round after restarts
    if tleft() > 150:
        cd_time = min(120.0, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                               max_sweeps=10000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: L-BFGS-B polish
    if tleft() > 100:
        alphas = [2000, 10000, 50000, 200000]
        h_p, s_p = lbfgs_round(best_h, target, alphas, 300, tleft() * 0.25, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 8: Try higher resolution
    if tleft() > 250:
        for n_try in [800, 1200]:
            if tleft() < 150:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            alphas_high = [50, 200, 800, 3000, 10000, 50000]
            h_hires, s_hires = lbfgs_round(h_interp, target_try, alphas_high,
                                            300, min(100.0, tleft() * 0.35), start_time)
            if tleft() < 60:
                break
            h_cd2, c5_cd2 = pair_cd(h_hires if s_hires < best_score else best_h,
                                     n_try if s_hires < best_score else n,
                                     (n_try if s_hires < best_score else n) / 2.0,
                                     time_limit=min(60.0, tleft() * 0.25),
                                     max_sweeps=5000, n_cand=25)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
            elif s_hires < best_score:
                best_score = s_hires
                best_h = h_hires

    # Phase 9: Final CD at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                               time_limit=min(50.0, tleft() * 0.4),
                               max_sweeps=5000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final L-BFGS-B
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_final = [10000, 50000, 200000]
        h_final, s_final = lbfgs_round(best_h, n_use, target_use, alphas_final,
                                        200, min(30.0, tleft() * 0.6), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 11: Very short final CD
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                               time_limit=min(10.0, tleft() * 0.7),
                               max_sweeps=3000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
