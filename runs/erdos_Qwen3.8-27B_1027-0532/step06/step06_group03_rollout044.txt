def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

            h = np.clip(h, 0.0, 1.0)

    c = np.correlate(h, h, mode='full')
    c_min = c.min()
    log_w = np.clip(t * (c - c_min), 0.0, 500.0)
    mw = log_w.max()
    p = np.exp(log_w - mw)
    p /= p.sum()
    n_fft = 1
    while n_fft < 3 * n:
        n_fft *= 2
    P = np.fft.fft(p, n_fft)
    H = np.fft.fft(h, n_fft)
    conv = np.real(np.fft.ifft(P * H))
    grad = 2.0 * conv[n - 1:2 * n - 1] * dx
    return grad


def neg_automin_obj(h, t):
    return -automin_obj(h, t)


def neg_automin_grad(h, t):
    return -automin_grad(h, t)


def nesterov_projected(h_start, n, target, time_limit, t_init, t_final, max_iter):
    start_t = time.time()
    h = h_start.copy()
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h)
    beta = 0.9
    max_lr = 0.5
    min_lr = 1e-8
    n_steps = max_iter
    for step in range(n_steps):
        if time.time() - start_t > time_limit:
            break
        progress = min(1.0, step / max(1, n_steps * 0.3))
        t_cur = t_init * (t_final / t_init) ** progress
        grad_y = neg_automin_grad(y, t_cur)
        grad_y -= np.mean(grad_y)
        y_new = project_feasible(h - (step / (step + 2.0)) * 0.0 + y, target)
        y = y - 0.0
        beta_cur = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = project_feasible(h + (beta_cur - 1.0) / (beta_cur + 1.0) * (y - h), target)
        grad_y = neg_automin_grad(y, t_cur)
        grad_y -= np.mean(grad_y)
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-12:
            break
        grad_y = grad_y / gnorm * gnorm
        lr = min(max_lr, 0.1 / (1.0 + 0.01 * step))
        f_y = neg_automin_obj(y, t_cur)
        for _bt in range(30):
            h_try = project_feasible(y - lr * grad_y, target)
            f_try = neg_automin_obj(h_try, t_cur)
            if f_try <= f_y - 1e-7 * lr * gnorm:
                break
            lr *= 0.5
            if lr < min_lr:
                break
        else:
            break
        h_new = h_try
        y = project_feasible(h_new + (beta_cur - 1.0) / (beta_cur + 1.0) * (h_new - h), target)
        h = h_new
        c5_now = compute_c5(h)
        if c5_now < best_val - 1e-12:
            best_val = c5_now
            best_h = h.copy()
    return best_h, best_val


def slsqp_smooth(h_start, n, target, alphas, maxiter, time_limit):
    start_t = time.time()
    n_corr = 2 * n - 1
    dx = 2.0 / n
    mid = n - 1

    def obj_fn(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(-alpha_cur[0] * (corr - c_max), -500.0, 0.0)
        return float(c_max - np.log(np.sum(np.exp(shifted))) / alpha_cur[0]) * dx

    def obj_grad_fn(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(-alpha_cur[0] * (corr - c_max), -500.0, 0.0)
        exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()
        grad = zeros_buf[0].copy()
        idxs = np.arange(n)
        shifts_arr = np.arange(-(n - 1), n)
        for s_idx in range(n_corr):
            wi = weights[s_idx]
            if wi < 1e-15:
                continue
            k = s_idx - mid
            if k >= 0:
                mask1 = idxs >= k
                grad[idxs[mask1]] += wi * (1.0 - h[idxs[mask1] - k])
                mask2 = idxs < n - k
                grad[idxs[mask2]] -= wi * h[idxs[mask2] + k]
            else:
                kk = -k
                mask1 = idxs < n - kk
                grad[idxs[mask1]] += wi * (1.0 - h[idxs[mask1] + kk])
                mask2 = idxs >= kk
                grad[idxs[mask2]] -= wi * h[idxs[mask2] - kk]
        grad *= dx
        return grad

    alpha_cur = [0]
    zeros_buf = [np.zeros(n)]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        alpha_cur[0] = alpha
        try:
            res = minimize(obj_fn, h_cur, jac=obj_grad_fn,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_trial = project_feasible(res.x, target)
            sc = compute_c5(h_trial)
            if sc < compute_c5(h_cur):
                h_cur = h_trial
        except Exception:
            pass
    return h_cur


def lbfgs_automin(h_start, n, target, t_vals, time_limit, maxiter=500):
    start_t = time.time()
    h_cur = h_start.copy()

    def obj_and_grad(h):
        t_cur = t_vals[0]
        val = -automin_obj(h, t_cur)
        g = -automin_grad(h, t_cur)
        g -= np.mean(g)
        return val, g

    t_vals = list(t_vals)
    for tv in t_vals:
        if time.time() - start_t > time_limit:
            break
        t_vals[0] = tv
        h_cur = project_feasible(h_cur, target)
        try:
            res = minimize(obj_and_grad, h_cur, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12,
                                    'maxls': 50})
            h_trial = project_feasible(res.x, target)
            if compute_c5(h_trial) <= compute_c5(h_cur) + 1e-12:
                h_cur = h_trial
        except Exception:
            pass
    return h_cur


def spectral_flatten_step(h, n_iter=5):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    for _ in range(n_iter):
        H = np.fft.fft(h_cur, n)
        magnitudes = np.abs(H)
        target_mag = np.sqrt(np.mean(magnitudes ** 2))
        phase = np.angle(H)
        H_new = target_mag * np.exp(1j * phase)
        h_new = np.real(np.fft.ifft(H_new))
        h_new = project_feasible(h_new, target)
        if compute_c5(h_new) <= compute_c5(h_cur) + 0.01:
            h_cur = h_new
    return h_cur


def pair_cd_refine(h_start, n, target, time_limit, max_sweeps, n_cand, top_k):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_alpha = 80.0
        w_top = np.exp(soft_alpha * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            if k >= 0:
                v1 = idxs >= k
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                v2 = idxs < n - k
                grad[v2] -= wi * h[idxs[v2] + k]
            else:
                kk = -k
                v1 = idxs < n - kk
                grad[v1] += wi * (1.0 - h[idxs[v1] + kk])
                v2 = idxs >= kk
                grad[v2] -= wi * h[idxs[v2] - kk]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global_start = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - global_start)

    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 200:
        h_nb = lbfgs_automin(best_h, n, target,
                             t_vals=[50, 200, 1000, 5000],
                             time_limit=min(100, tleft() * 0.4),
                             maxiter=800)
        sc = compute_c5(h_nb)
        if sc < best_score:
            best_score = sc
            best_h = h_nb

    if tleft() > 150:
        h_ns = nesterov_projected(best_h, n, target,
                                  time_limit=min(100, tleft() * 0.4),
                                  t_init=50, t_final=5000,
                                  max_iter=20000)
        if h_ns is not None:
            sc = compute_c5(h_ns)
            if sc < best_score:
                best_score = sc
                best_h = h_ns

    if tleft() > 200:
        h_cd, c5_cd = pair_cd_refine(best_h, n, target,
                                     time_limit=min(120, tleft() * 0.4),
                                     max_sweeps=8000, n_cand=50, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        h_sl = slsqp_smooth(best_h, n, target,
                            alphas=[200, 1000, 5000, 20000],
                            maxiter=2000,
                            time_limit=min(100, tleft() * 0.4))
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    if tleft() > 100:
        h_cd, c5_cd = pair_cd_refine(best_h, n, target,
                                     time_limit=min(60, tleft() * 0.4),
                                     max_sweeps=6000, n_cand=65, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfgs_automin(h_interp, n_try, target_try,
                                    t_vals=[50, 300, 2000],
                                    time_limit=min(50, tleft() * 0.4),
                                    maxiter=500)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try
                if tleft() > 50:
                    h_cd2, c5_cd2 = pair_cd_refine(best_h, n, target,
                                                   time_limit=min(30, tleft() * 0.3),
                                                   max_sweeps=4000, n_cand=40, top_k=3)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2

    if tleft() > 100:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfgs_automin(h_pert, len(best_h), target,
                                  t_vals=[100, 1000, 5000],
                                  time_limit=min(35, tleft() * 0.3),
                                  maxiter=500)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 80:
        h_sl = slsqp_smooth(best_h, len(best_h), target,
                            alphas=[5000, 20000, 100000],
                            maxiter=1500,
                            time_limit=min(50, tleft() * 0.5))
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    if tleft() > 50:
        h_cd, c5_cd = pair_cd_refine(best_h, len(best_h), target,
                                     time_limit=min(35, tleft() * 0.5),
                                     max_sweeps=5000, n_cand=55, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_opt = lbfgs_automin(best_h, n_use, target_use,
                                    t_vals=[10000, 50000],
                                    time_limit=min(20, tleft() * 0.6),
                                    maxiter=500)
        sc = compute_c5(h_final_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_final_opt

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
