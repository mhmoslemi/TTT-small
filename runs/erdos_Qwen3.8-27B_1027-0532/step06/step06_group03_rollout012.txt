    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    mid = n - 1

        # k-mid ranges from -(n-1) to n-1
        # j + (k-mid) must be in [0, n-1]
        # So k-mid in [-j, n-1-j], i.e., k in [mid-j, mid+n-1-j]
        k_lo = max(0, mid - j)
        k_hi = min(2*n-2, mid + n - 1 - j)
        if k_hi >= k_lo:
            ks = np.arange(k_lo, k_hi+1)
            idx = j + ks - mid
            grad_full[j] += np.dot(weights[ks], 1.0 - h[idx])

        # j-(k-mid) = j - k + mid must be in [0, n-1]
        # So k - mid in [j-(n-1), j], i.e., k in [mid+j-(n-1), mid+j]
        k_lo2 = max(0, mid + j - n + 1)
        k_hi2 = min(2*n-2, mid + j)
        if k_hi2 >= k_lo2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            idx2 = j - ks2 + mid
            grad_full[j] -= np.dot(weights[ks2], h[idx2])
    
    grad_full *= dx
    half = n // 2
    grad_v = grad_full[:half] + grad_full[n-1:n-1-half:-1]

Let me also think about the "pair CD" from the previous solution. It's a coordinate descent method that picks pairs (i, j) to transfer mass from h[i] to h[j] (or vice versa) in a way that minimizes the maximum correlation. This is a very targeted approach.

The key idea of pair CD:
- For each pair (i, j), consider h[i] -= δ, h[j] += δ for some δ
- The correlation changes linearly in δ (since corr is bilinear in h)
- Find the δ that minimizes the max of the resulting correlation
- This is a 1D optimization problem

The previous implementation does this over a subset of candidate pairs (top-k by gradient magnitude). This is efficient and effective.

def project_box_sum(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}."""
    lo = -5.0
    hi = 5.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v - lam, 0.0, 1.0)

                v_mom = v

            v_try = project_box_sum(v_y - step * grad_tan, target)

                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                
                g = np.zeros(n_corr)
