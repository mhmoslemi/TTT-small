    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def build_constraint_indices(n):
    half_n = n - 1
    constraint_info = []
    for k_idx in range(2 * n - 1):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            src_lo = lo - k
            src_hi = hi - k
            constraint_info.append((lo, hi, src_lo, src_hi))
    return constraint_info

def full_lp_solve(h_old, n, target, reg, solver_name, eps, max_iters, constraint_info):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h_old
    for (lo, hi, src_lo, src_hi) in constraint_info:
        coeff = np.zeros(n)
        coeff[lo:hi] = g[src_lo:src_hi] * dx
        constraints.append(cp.multiply(coeff, h_var).sum() <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_old)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        if solver_name == 'SCS':
            prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters, verbose=False)
        elif solver_name == 'ECOS':
            prob.solve(solver=cp.ECOS, max_iters=max_iters)
        else:
            prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    stall_count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        h_new = full_lp_solve(h, n, target, current_reg, solver_name, eps, 200000, constraint_info)
        if h_new is None:
            if it > 5:
                break
            h_new = full_lp_solve(h, n, target, current_reg, 'ECOS', 1e-12, 100000, constraint_info)
            if h_new is None:
                break
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0

        if it % 20 == 0 and current_reg > 1e-10:

    return best_h, best_c5


def coord_descent_full(h_init, target, n, max_passes, time_limit, rng):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_k = 2 * n - 1
    half_n = n - 1
    k_values = np.arange(n_k) - half_n
    lam = 0.0
    lam_rate = 0.005

    corr = np.correlate(h, 1.0 - h, mode='full')

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.zeros(n_k)
            intercepts = np.zeros(n_k)
            for kk in range(n_k):
                k = k_values[kk]
                A = 0.0
                j = i + k
                if 0 <= j < n:
                    A += (1.0 - h[j])
                j2 = i - k
                if 0 <= j2 < n:
                    A -= h[j2]
                A *= dx
                slopes[kk] = A
                intercepts[kk] = corr[kk] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -0.5, 0.5)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5


def pair_exchange_search(h_init, target, n, max_steps, time_limit, rng):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.5, 0.5) * 0.3
        h_try = h.copy()
        h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h_try[j] = np.clip(h[j] - delta, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)

            pass

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
