                        beta=0.9, lr_init=0.01, n_restarts=5):

    n_total_steps = 50000  # we'll break early based on time
    restart_interval = n_total_steps // n_restarts
    
    for step in range(n_total_steps):
        if time.time() - start > time_limit:
            break
        
        # Adaptive alpha
        progress = step / n_total_steps
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute smooth max and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        mid = n - 1
        for m in range(len(corr)):
            if w[m] < 1e-12:
                continue
            s = m - mid
            if s >= 0:
                valid1 = np.arange(0, n - s)
                grad[valid1] += w[m] * (1.0 - h[valid1 + s])
                valid2 = np.arange(s, n)
                grad[valid2] -= w[m] * h[valid2 - s]
            else:
                t = -s
                valid1 = np.arange(t, n)
                grad[valid1] += w[m] * (1.0 - h[valid1 - t])
                valid2 = np.arange(0, n - t)
                grad[valid2] -= w[m] * h[valid2 + t]

        v = beta * v + grad
        lr = lr_init / (1.0 + step * 1e-5)
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:

            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, max_steps, n_shifts, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 40.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved and no_improve > 800:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_perturb_restart(h, n, target, n_keep, rng, scale):
    h_fft = np.fft.rfft(h)
    n_fft = len(h_fft)
    n_keep_fft = min(n_keep + 1, n_fft)
    h_fft_pert = h_fft.copy()
    pert = rng.normal(0, scale, n_keep_fft)
    pert[0] = 0.0
    h_fft_pert[:n_keep_fft] += pert
    if n_keep_fft < n_fft:
        h_fft_pert[n_keep_fft:] = 0.0
    h_new = np.fft.irfft(h_fft_pert, n=n)
    h_new = project_feasible(h_new, target)
    return h_new


def structured_start(n, target, pattern_type, rng):
    if pattern_type == 0:
        period = max(4, n // 60)
        h = np.zeros(n)
        for i in range(n):
            phase = (i % period) / period
            h[i] = 0.5 + 0.5 * np.cos(2.0 * np.pi * phase)
    elif pattern_type == 1:
        block = max(3, n // 100)
        h = np.zeros(n)
        for i in range(n):
            seg = (i // block) % 3
            h[i] = 1.0 if seg < 1 else 0.0
    else:
        h = np.full(n, 0.5)
        noise = rng.normal(0, 0.2, n)
        h += noise
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 150:
        hmom, sc = momentum_smooth_max(best_h, n, target, time_limit=min(120, tleft() * 0.4),
                                       alpha_start=100, alpha_end=50000,
                                       beta=0.9, lr_init=0.008, n_restarts=4, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = hmom

    if tleft() > 200:
        cd_time = min(280, tleft() * 0.35)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 100:
        hsg, c5sg = multi_shift_subgrad(best_h, n, target, time_limit=min(60, tleft() * 0.4),
                                        max_steps=50000, n_shifts=5, base_lr=0.015)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 150:
        cd_time = min(200, tleft() * 0.4)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 200:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, n_new, target_new, [100, 500, 2000, 10000], 2000, 1e-15)
        if tleft() > 120:
            hmom2, sc2 = momentum_smooth_max(hp, n_new, target_new,
                                             time_limit=min(80, tleft() * 0.4),
                                             alpha_start=200, alpha_end=100000,
                                             beta=0.9, lr_init=0.005, n_restarts=3, rng=rng)
            if sc2 < sc:
                sc = sc2
                hp = hmom2
        if tleft() > 80:
            hcd2, c5cd2 = fast_pair_cd(hp, n_new, target_new,
                                       time_limit=min(70, tleft() * 0.5),
                                       n_cand=25, max_sweeps=50000)
            if c5cd2 < sc:
                sc = c5cd2
                hp = hcd2
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use,
                                 time_limit=min(120, tleft() * 0.5),
                                 n_cand=35, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fourier = fourier_perturb_restart(best_h, n_use, target_use,
                                            n_keep=8, rng=rng, scale=0.15)
        hp, sc = slsqp_round(h_fourier, n_use, target_use, [500, 2000, 10000], 1500, 1e-15)
        if tleft() > 60:
            hcd, c5cd = fast_pair_cd(hp, n_use, target_use,
                                     time_limit=min(50, tleft() * 0.5),
                                     n_cand=25, max_sweeps=30000)
            if c5cd < sc:
                sc = c5cd
                hp = hcd
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use,
                                 time_limit=min(80, tleft() * 0.5),
                                 n_cand=30, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hmom, sc = momentum_smooth_max(best_h, n_use, target_use,
                                       time_limit=min(50, tleft() * 0.5),
                                       alpha_start=5000, alpha_end=500000,
                                       beta=0.85, lr_init=0.003, n_restarts=3, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = hmom

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [50000, 200000, 500000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use,
                                 time_limit=min(25, tleft() * 0.6),
                                 n_cand=30, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [500000, 2000000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
