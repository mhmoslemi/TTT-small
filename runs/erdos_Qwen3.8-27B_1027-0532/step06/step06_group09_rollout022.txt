    h = np.clip(h, 0.0, 1.0)

    mid = n - 1

combined_grad[p] = sum_{m=1}^{M} w_m * sens[p, k_m]

where w_m are weights (e.g., softmax of the top M correlation values).

This gives pairs that reduce the weighted sum of the worst shifts, which is a better heuristic.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0, 1, n_dst)
    x_old = np.linspace(0, 1, n_src)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, target_dst)

def compute_sensitivity(h, n):
    mid = n - 1
    idxs = np.arange(n)
    sens = np.zeros((n, 2 * n - 1))
    for k in range(-(n - 1), n):
        ki = k + mid
        vp = (idxs + k >= 0) & (idxs + k < n)
        vm = (idxs - k >= 0) & (idxs - k < n)
        sens[vp, ki] += (1.0 - h[idxs[vp] + k])
        sens[vm, ki] -= h[idxs[vm] - k]
    return sens

def lse_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -400, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    val = (cmax + np.log(w_sum) / alpha) * dx
    
    # Gradient: sum_k w_k * sens[:, k]
    sens = compute_sensitivity(h, n)
    grad = sens.T @ w * dx
    return val, grad

def lbfags_polish(h0, n, target, alpha, maxiter=2000):
    _alpha = alpha
    _n = n
    def obj(x):
        v, _ = lse_objective_and_grad(x, _alpha)
        return v
    def jac(x):
        _, g = lse_objective_and_grad(x, _alpha)
        return g
    bounds = [(0.0, 1.0)] * _n
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except:
        return h0

def pair_cd(h0, n, target, time_limit, n_cand=40, patience=200, m_top=1, seed=42):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    
    for sweep in range(500000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        sens = compute_sensitivity(h, n)
        
        # Select shifts to target
        if m_top == 1:
            k_star = int(np.argmax(corr))
            d_corr = sens[:, k_star]
        else:
            top_m = np.argsort(corr)[-m_top:]
            # Weight by softmax
            top_corr = corr[top_m]
            w_m = np.exp(10 * (top_corr - top_corr.max()))
            w_m /= w_m.sum()
            d_corr = sum(w * sens[:, int(k)] for w, k in zip(w_m, top_m))
        
        sorted_d = np.argsort(d_corr)
        low_cands = sorted_d[:n_cand]
        high_cands = sorted_d[-n_cand:]
        
        best_improvement = 0
        best_pair = None
        best_delta = 0
        
        for i in high_cands:
            if h[i] < 1e-10:
                continue
            for j in low_cands:
                if i == j or h[j] > 1.0 - 1e-10:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                effect = sens[int(j)] - sens[int(i)]
                if m_top == 1:
                    if effect[k_star] >= 0:
                        continue
                else:
                    # Check weighted effect
                    if np.dot(effect[top_m], w_m) >= 0:
                        continue
                
                # Binary search
                lo = 0.0
                hi = delta_max
                for _ in range(18):
                    mid_d = 0.5 * (lo + hi)
                    new_vals = corr + mid_d * effect
                    f_mid = float(np.max(new_vals))
                    active = new_vals >= f_mid - 1e-14
                    subgrad = float(np.max(effect[active])) if active.any() else 0.0
                    if subgrad > 1e-14:
                        hi = mid_d
                    else:
                        lo = mid_d
                
                opt_delta = 0.5 * (lo + hi)
                if opt_delta < 1e-12:
                    continue
                new_c5 = float(np.max(corr + opt_delta * effect)) * dx
                improvement = c5_val - new_c5
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (int(i), int(j))
                    best_delta = opt_delta
        
        if best_pair is not None:
            i, j = best_pair
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0, 1, n_dst)
    x_old = np.linspace(0, 1, n_src)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, target_dst)

        sens[vp, ki] -= h[idxs[vp] + k]

    val = (cmax + np.log(w_sum) / alpha) * dx
    sens = compute_sensitivity(h, n)
    grad = (sens.T @ w) * dx
    return val, grad

    v, _ = lse_objective_grad(x, alpha)

    _, g = lse_objective_grad(x, alpha)
