def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max(corr, alpha):

    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    exp_shifted = np.exp(shifted)
    return c_max + np.log(exp_shifted.sum()) / alpha

def smooth_max_grad(corr, h, alpha, n):

    grad *= dx

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_optimize(h0, n, target, time_limit, max_steps=10000, top_k=20, alpha_sm=50.0):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    # Nesterov-like momentum
    v = np.zeros(n)
    t_cnt = 1.0
    x_avg = h.copy()
    
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Weighted subgradient
        c_max = np.max(corr)
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[sorted_idx]
        w_raw = np.exp(alpha_sm * (top_vals - c_max))
        w = w_raw / w_raw.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(w, sorted_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        
        # Nesterov acceleration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_cnt * t_cnt)) / 2.0
        beta = (t_cnt - 1.0) / t_new
        t_cnt = t_new
        lr = 2.0 / (1.0 + t_cnt)
        
        improved = False
        for lr_t in [lr, lr*0.5, lr*0.25, lr*0.1]:
            h_try = project_feasible(h + beta * v - lr_t * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                x_avg = x_avg + (h_try - h) / t_cnt  # Nesterov averaging
                v = grad
                h = h_try
                improved = True
                break
        
        if not improved:
            v = 0.5 * v
            if step > 500:
                break
    
    return best_h, best_c5

def cd_fast(h0, n, target, time_limit, max_sweeps=10000, n_cand=20):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            break
        
        # Find the worst shift
        k_star = int(np.argmax(corr)) - mid
        
        # Compute subgradient for this shift
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Select candidate coordinates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_new_c5 = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-12:
                    continue
                
                # Compute the effect on all correlations
                # corr_new[s] = corr[s] + delta * (effect of moving mass from j to i)
                # For shift s (corresponding to lag k = s - mid):
                # The change is: delta * [(1-h[i+k]) - h[i-k]] - delta * [(1-h[j+k]) - h[j-k]]
                # (with appropriate clipping)
                
                # Actually, let's just compute the new correlation directly for the active shifts
                # But that's expensive. Instead, use the precomputed gradient.
                
                # Simpler: just do a few trial values of delta
                best_local_delta = 0.0
                best_local_c5 = c5_val
                # Geometric search
                delta_trials = [d_max * f for f in [0.1, 0.3, 0.5, 0.7, 1.0]]
                for d_trial in delta_trials:
                    if d_trial > d_max:
                        continue
                    h_trial = h.copy()
                    h_trial[i] -= d_trial
                    h_trial[j] += d_trial
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_local_c5:
                        best_local_c5 = c5_trial
                        best_local_delta = d_trial
                
                if best_local_c5 < best_new_c5:
                    best_new_c5 = best_local_c5
                    best_delta = best_local_delta
                    best_i = i
                    best_j = j
        
        if best_i >= 0 and best_delta > 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active=30, max_rounds=50):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        
        # Build LP: minimize t
        # s.t. corr[s] + grad_s @ (h_new - h) <= t for s in active
        #      0 <= h_new <= 1
        #      sum(h_new) = target
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for s_idx in active:
            k = int(s_idx) - mid
            gs = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            gs[v1] += (1.0 - h[idxs[v1] - k])
            gs[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + gs @ (h_var - h)
            constraints.append(lin_val <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, maxiters=5000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            # Try a blend
            for mix in [0.5, 0.3, 0.1]:
                h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-12:
                    h = h_mix
                    break
            else:
                break
    
    return best_h, best_c5

def slsqp_refine(h0, n, target, time_limit, alphas=None, maxiter=2000):
    if alphas is None:
        alphas = [100, 500, 2000, 10000, 50000]
    
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    idxs = np.arange(n)
    mid = n - 1
    target_sum = target
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
            return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)
        
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros_like(w)
                w[int(np.argmax(corr))] = 1.0
            else:
                w = w / ws
            g = np.zeros(n)
            active = np.where(w > 1e-8)[0]
            for s in active:
                k = int(s) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += w[s] * (1.0 - x[idxs[v1] - k])
                g[v2] -= w[s] * x[idxs[v2] + k]
            g *= dx
            g -= np.mean(g)
            return g
        
        def con_fun(x):
            return np.sum(x) - target_sum
        
        def con_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def perturb(h, n, target, magnitude=0.05, seed=0):
    rng = np.random.RandomState(seed)
    h_new = h + magnitude * (rng.random(n) - 0.5)
    return project_feasible(h_new, target)
