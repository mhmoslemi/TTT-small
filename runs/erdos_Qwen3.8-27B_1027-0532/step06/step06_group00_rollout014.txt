            velocity *= 0.5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_active(h, shifts, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx, k in enumerate(shifts):
        w = weights[idx]
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            ka = -k
            grad[ka:] += w * (1.0 - h[:n - ka])
            grad[:n - ka] -= w * h[ka:]
    grad -= np.mean(grad)
    return grad


def momentum_descent(h_init, n, target, time_limit,

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    return project_feasible(h_out, target_dst)

    # Phase 9: If time permits, try n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_mom3, c5_mom3 = momentum_descent(h2, n2, target2,
                                               time_limit=min(120, tleft() * 0.4),
                                               lr_init=0.03, momentum=0.82,
                                               p_start=20, p_end=3,
                                               alpha_decay_rate=0.9998)
            if c5_mom3 < best_score:
                best_score = c5_mom3
                best_h = h_mom3
                h_work = h_mom3
                n_work = n2
                target_work = target2
            if tleft() > 80:
                h_cd3, c5_cd3 = pair_cd(h_work, n_work, target_work,
                                        time_limit=min(100, tleft() * 0.5),
                                        n_cand_start=20, n_cand_end=80,
                                        max_no_improve=400)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3
                    h_work = h_cd3

    # Phase 10: Final polish
    if tleft() > 60:
        h_pol3 = slsqp_polish(h_work, n_work, target_work,
                              [5000, 20000, 80000], 400)
        sc3 = compute_c5(h_pol3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    # Phase 11: Final pair CD
    if tleft() > 50:
        h_cd_final, c5_final = pair_cd(best_h, n_work, target_work,
                                       time_limit=min(40, tleft() * 0.6),
                                       n_cand_start=50, n_cand_end=120,
                                       max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Final projection and evaluation
    n_final = len(best_h)
    final_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, n_final
