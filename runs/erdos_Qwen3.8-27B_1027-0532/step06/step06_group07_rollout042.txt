```python
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    
    if f_star_guess is None:
        f_star_guess = best_c5 - 0.001
    
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active shifts
        active_mask = corr >= np.max(corr) - 1e-10
        active_shifts = np.where(active_mask)[0]
        
        # Compute subgradient (average of gradients of active functions)
        grad = np.zeros(n)
        for s in active_shifts:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active_shifts)
        
        # Polyak step size
        grad_centered = grad - np.mean(grad)
        grad_norm_sq = np.sum(grad_centered**2)
        if grad_norm_sq < 1e-14:
            break
        
        gap = c5_val - f_star_guess
        if gap <= 0:
            gap = 1e-4  # minimum gap
        alpha = gap / (grad_norm_sq + 1e-10)
        alpha = min(alpha, 0.1)  # cap to avoid overshooting
        
        # Take step and project
        h_new = project_feasible(h - alpha * grad_centered, target)
        
        # Backtracking if needed
        c5_new = compute_c5(h_new)
        if c5_new > c5_val + 1e-12:
            # Try smaller step
            for mult in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - alpha * mult * grad_centered, target)
                if compute_c5(h_try) < c5_val:
                    h = h_try
                    break
            else:
                break
        else:
            h = h_new
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(H, alpha):
    h = H  # already 1D
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(H):
    return np.sum(H) - _global_target

def cons_jac(H):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def softmax_subgrad(h_init, n, target, time_limit, alpha_start=100.0, alpha_end=50000.0, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    
    # Adaptive alpha: geometric increase
    n_steps = max_steps
```

```python
def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

            if time.time() - start > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()

            scaled = alpha * (corr - np.max(corr))
            scaled = np.clip(scaled, -500.0, 0.0)
            weights = np.exp(scaled)
            w_sum = np.sum(weights)
            if w_sum < 1e-30:
                weights = np.zeros(n_corr)
                weights[np.argmax(corr)] = 1.0
                w_sum = 1.0
            else:
                weights /= w_sum

            grad = np.zeros(n)
            active = np.where(weights > 1e-10)[0]
            for s in active:
                k = int(s) - mid
                w = weights[int(s)]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]

            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step_size = 0.5 / grad_norm
            improved = False
            for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
                h_try = project_feasible(h - step_size * mult * grad, target)
                f_try = compute_c5(h_try)
                if f_try < c5_val - 1e-15:
                    h = h_try
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand=35, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 40:
                break

    return best_h, best_c5


def multi_seed_pair_cd(h_init, n, target, time_limit, n_seeds=5, n_cand=30, n_bisect=13):
    start = time.time()
    rng = np.random.RandomState(12345)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    time_per_seed = time_limit / n_seeds

    for s in range(n_seeds):
        if time.time() - start > time_limit:
            break
        h_try = best_h + rng.randn(n) * 0.05
        h_try = project_feasible(h_try, target)
        t_seg = min(time_per_seed, tleft() - (s + 1) * time_per_seed / n_seeds)
        t_seg = max(t_seg, 10.0)
        h_res, c5_res = pair_cd(h_try, n, target, time_limit=t_seg,
                                max_sweeps=2000, n_cand=n_cand, n_bisect=n_bisect)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _run_start, _run_budget
    _run_start = time.time()
    _run_budget = budget_s

    h_init = np.array(initial_h_values, dtype=float)
    n_base = len(h_init)
    target_base = n_base / 2.0

    best_h = project_feasible(h_init, target_base)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at base resolution
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n_base, target_base, [500, 2000, 8000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Softmax gradient descent at base resolution
    if tleft() > 150:
        gd_time = min(120, tleft() * 0.25)
        h_gd, c5_gd = softmax_gd(best_h, n_base, target_base, time_limit=gd_time,
                                 alpha_seq=np.geomspace(50, 50000, 12), max_inner=2000)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd

    # Phase 3: Pair CD at base resolution
    if tleft() > 150:
        cd_time = min(120, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n_base, target_base, time_limit=cd_time,
                              max_sweeps=5000, n_cand=40, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP polish
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n_base, target_base, [5000, 20000, 80000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Try n=800
    if tleft() > 250:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cur = project_feasible(h_interp, target_try)
        sc_cur = compute_c5(h_cur)

        if tleft() > 120:
            gd_time = min(100, tleft() * 0.2)
            h_gd, c5_gd = softmax_gd(h_cur, n_try, target_try, time_limit=gd_time,
                                     alpha_seq=np.geomspace(50, 80000, 10), max_inner=1500)
            if c5_gd < sc_cur:
                h_cur = h_gd
                sc_cur = c5_gd

        if tleft() > 100:
            cd_time = min(80, tleft() * 0.15)
            h_cd, c5_cd = pair_cd(h_cur, n_try, target_try, time_limit=cd_time,
                                  max_sweeps=3000, n_cand=30, n_bisect=14)
            if c5_cd < sc_cur:
                h_cur = h_cd
                sc_cur = c5_cd

        if sc_cur < best_score:
            best_score = sc_cur
            best_h = h_cur.copy()

    # Phase 6: Try n=1000
    if tleft() > 250:
        n_try = 1000
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cur = project_feasible(h_interp, target_try)
        sc_cur = compute_c5(h_cur)

        if tleft() > 120:
            gd_time = min(120, tleft() * 0.2)
            h_gd, c5_gd = softmax_gd(h_cur, n_try, target_try, time_limit=gd_time,
                                     alpha_seq=np.geomspace(50, 100000, 10), max_inner=1500)
            if c5_gd < sc_cur:
                h_cur = h_gd
                sc_cur = c5_gd

        if tleft() > 100:
            cd_time = min(80, tleft() * 0.15)
            h_cd, c5_cd = pair_cd(h_cur, n_try, target_try, time_limit=cd_time,
                                  max_sweeps=2000, n_cand=25, n_bisect=14)
            if c5_cd < sc_cur:
                h_cur = h_cd
                sc_cur = c5_cd

        if sc_cur < best_score:
            best_score = sc_cur
            best_h = h_cur.copy()

    # Phase 7: Try n=1200
    if tleft() > 200:
        n_try = 1200
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cur = project_feasible(h_interp, target_try)
        sc_cur = compute_c5(h_cur)

        if tleft() > 100:
            gd_time = min(100, tleft() * 0.2)
            h_gd, c5_gd = softmax_gd(h_cur, n_try, target_try, time_limit=gd_time,
                                     alpha_seq=np.geomspace(50, 100000, 8), max_inner=1200)
            if c5_gd < sc_cur:
                h_cur = h_gd
                sc_cur = c5_gd

        if tleft() > 80:
            cd_time = min(60, tleft() * 0.15)
            h_cd, c5_cd = pair_cd(h_cur, n_try, target_try, time_limit=cd_time,
                                  max_sweeps=1500, n_cand=20, n_bisect=13)
            if c5_cd < sc_cur:
                h_cur = h_cd
                sc_cur = c5_cd

            best_h = h_cur.copy()

    if tleft() > 80:

        h_mr, c5_mr = multi_seed_pair_cd(best_h, n_use, target_use,
                                         time_limit=mr_time, n_seeds=4, n_cand=30, n_bisect=13)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        gd_time = min(30, tleft() * 0.4)
        h_gd, c5_gd = softmax_gd(best_h, n_use, target_use, time_limit=gd_time,

            best_h = h_gd

    # Phase 10: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(25, tleft() * 0.5)
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use, time_limit=cd_time,
                              max_sweeps=2000, n_cand=35, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
