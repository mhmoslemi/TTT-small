def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff < 0:
        # need to reduce
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = h[idx]
            if can_reduce > remaining:
                h[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                h[idx] = 0.0
    else:
        # need to increase
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - h[idx]
            if can_inc > remaining:
                h[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                h[idx] = 1.0
    return h

def compute_lse_gradient(h, corr, alpha, shifts):
    """Compute gradient of log-sum-exp approximation."""
    n = len(h)
    max_corr = corr.max()
    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    W = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = W[k]
        if wk < 1e-12:
            continue
        s = shifts[k]
        if s >= 0:
            # f_k = sum_i h[i](1-h[i+s]) for i in [0, n-s)
            # df_k/dh[j] = (1-h[j+s]) - h[j-s] (with boundary conditions)
            # For j in [0, n-s): contributes (1-h[j+s])
            # For j in [s, n): contributes -h[j-s]
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        else:

            # f_k = sum_i h[i](1-h[i-abs_s]) for i in [abs_s, n)

            grad[abs_s:] += wk * (1.0 - h[:n-abs_s])
            grad[:n-abs_s] += wk * (-h[abs_s:])
    return grad

    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r >= 1.0)[0]
    zeros = np.where(h_r <= 0.0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0

def compute_delta_swap(h, i, j):
    """Compute change in correlation when swapping h[i] and h[j] (set i to 0, j to 1)."""
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    
    return result

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff < 0:
        order = np.argsort(h)[::-1]
        remaining = -diff
        h = h.copy()
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = h[idx]
            if can_reduce > remaining:
                h[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                h[idx] = 0.0
    else:
        order = np.argsort(h)
        remaining = diff
        h = h.copy()
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - h[idx]
            if can_inc > remaining:
                h[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                h[idx] = 1.0
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r >= 0.5)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r = h_r.copy()
            h_r[ones[order]] = 0.0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r < 0.5)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r = h_r.copy()
            h_r[zeros[order]] = 1.0
    return h_r

def compute_swap_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    
    return result

def lse_grad_descent(h_init, n_iters, alpha_schedule, lr_schedule, time_limit, start_time, momentum=0.9):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    stall = 0
    max_stall = 5000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        alpha = alpha_schedule(ratio)
        lr = lr_schedule(ratio)
        
        # Compute LSE gradient
        max_corr = corr.max()
        shifted = alpha * (corr - max_corr)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        exp_sum = exp_vals.sum()
        W = exp_vals / exp_sum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = W[k]
            if wk < 1e-12:
                continue
            s = shifts[k]
            if s >= 0:
                grad[:n-s] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:n-s])
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - h[:n-abs_s])
                grad[:n-abs_s] += wk * (-h[abs_s:])
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        h_new = project_box_sum(h + v, target)
        
        # Check improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        
        if c5_new < best_c5 - 1e-13:
            h = h_new
            corr = corr_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        elif c5_new < compute_c5(h) + 1e-10:
            h = h_new
            corr = corr_new
            stall += 1
        else:
            v *= 0.5
            stall += 1
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff < 0:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = h[idx]
            if can_reduce > remaining:
                h[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                h[idx] = 0.0
    else:
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - h[idx]
            if can_inc > remaining:
                h[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                h[idx] = 1.0
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r >= 0.5)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r < 0.5)[0]
        if need <= len(zeros):

def swap_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def paired_swap_delta(h, i1, i2, j1, j2):

        idx = pos + k

def lse_grad_descent(h_init, n_iters, alpha_start, alpha_end, lr_start, lr_end, time_limit, start_time, use_momentum=True):
    n = len(h_init)

    target = n / 2.0

    v = np.zeros(n)

        if time.time() - start_time > time_limit:
            break

            break
        ratio = it / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        lr = lr_start * (lr_end / lr_start) ** (ratio * 0.6)

        max_corr = corr.max()
        shifted = alpha * (corr - max_corr)
        shifted = np.clip(shifted, -500.0, 0.0)

        if exp_sum < 1e-300:

            stall = 0

        # 5c: LSE from best
        if tl() > 35:
            hs = project_binary(best_cont, int(n_init // 2))
            if compute_c5(hs) > compute_c5(best_bin):
                hs = best_bin.copy()
            alloc = min(40.0, tl() * 0.10)
            if alloc > 12:
                ni = int(min(400000, alloc * 8000))
                hl, cl = lse_grad_descent(hs, ni, 80.0, 3000.0, 0.025, 0.001, alloc, start_time)
                if cl < best_cont_score:
                    best_cont_score = cl
                    best_cont = hl.copy()

    # Phase 6: Multi-resolution
    for nd in [n_init + 4, n_init - 4, n_init + 8, n_init - 8]:
        if tl() < 50:
            break
        if nd < 100 or nd % 2 != 0:
            continue
        xt = np.linspace(0.0, 1.0, n_init)
        xn = np.linspace(0.0, 1.0, nd)
        hi = np.interp(xn, xt, best_cont)
        hi = project_box_sum(hi, float(nd) / 2.0)
        alloc = min(25.0, tl() * 0.08)
        if alloc < 10:
            continue
        ni = int(min(300000, alloc * 8000))
        hn, cn = lse_grad_descent(hi, ni, 80.0, 2000.0, 0.03, 0.001, alloc, start_time)
        if cn < best_cont_score:
            best_cont_score = cn
            best_cont = hn.copy()

    # Phase 7: Final SA polish
    if tl() > 30:
        alloc = min(50.0, tl() - 15)
        hr = project_binary(best_cont, int(n_init // 2))
        rs = compute_c5(hr)
        if rs < best_bin_score:
            best_bin_score = rs
            best_bin = hr.copy()
        nsa = int(min(5000000, alloc * 60000))
        hs, cs = sa_with_restarts(best_bin, nsa, 0.004, 0.000003, rng, alloc, start_time, n_active=6, use_paired=False)
        if cs < best_bin_score:
            best_bin_score = cs
            best_bin = hs.copy()

    # Final selection
    fc = project_box_sum(best_cont, target)
    fcs = compute_c5(fc)
    fb = project_binary(best_bin, int(n_init // 2))
    fbs = compute_c5(fb)

    if fbs < fcs:
        return fb, fbs, n_init
    else:
        return fc, fcs, n_init
