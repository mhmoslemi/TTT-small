    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    # Find top lags
    top_idx = np.argsort(corr)[-n_top:]
    top_corr = corr[top_idx]
    shifted = np.clip(alpha * (top_corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    # Compute gradient using only top lags
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for ii in range(len(top_idx)):

        w = weights[ii]
        if w < 1e-12: continue

            grad[:n-p] -= w * h[p:]
    return value, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def fast_smooth_obj_grad(h, alpha, n_top=30):
    """Compute smooth objective and gradient using only top n_top lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def fast_smooth_obj_grad(h, alpha, n_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    use_top = min(n_top, len(corr))
    top_idx = np.argsort(corr)[-use_top:]
    top_corr = corr[top_idx]
    shifted = np.clip(alpha * (top_corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for ii in range(use_top):
        s = int(top_idx[ii]) - (n - 1)
        w = float(weights[ii])
        if w < 1e-14:
            continue
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += w * one_minus_h[s:]
            if s > 0:
                grad[s:] -= w * h[:ns]
        else:
            p = -s
            np2 = n - p
            if np2 > 0:
                grad[p:] += w * one_minus_h[:np2]
                grad[:np2] -= w * h[p:]
    return value, grad

def full_smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = float(weights[n - 1 + s])
        if w < 1e-14:
            continue
        ns = n - s
        if ns > 0:
            grad[:ns] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:ns]
    for p in range(1, n):
        w = float(weights[n - 1 - p])
        if w < 1e-14:
            continue
        np2 = n - p
        if np2 > 0:
            grad[p:] += w * one_minus_h[:np2]
            grad[:np2] -= w * h[p:]
    return value, grad

def adam_anneal(h0, target_sum, start_time, time_budget, alpha_init, alpha_final,
                lr, beta1, beta2, eps, n_top=50, checkpoint_every=30):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_actual = compute_c5(h)
    log_ratio = np.log(alpha_final / alpha_init)
    total_iters_estimate = int(time_budget * 30)
    iters_per_alpha = max(1, total_iters_estimate // 5)

    while time.time() - start_time < time_budget:
        t += 1
        progress = min(1.0, t / max(total_iters_estimate, 1))
        alpha = alpha_init * np.exp(log_ratio * progress)
        if alpha < 80:
            val, grad = full_smooth_obj_grad(h, alpha)
        else:
            val, grad = fast_smooth_obj_grad(h, alpha, n_top)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_box_sum(h_new, target_sum)
        if t % checkpoint_every == 0:
            actual = compute_c5(h)
            if actual < best_actual - 1e-10:
                best_actual = actual
                best_h = h.copy()
    actual = compute_c5(h)
    if actual < best_actual:
        best_actual = actual
        best_h = h.copy()
    return best_h, best_actual

def subgrad_refine(h0, target_sum, start_time, time_budget, init_lr=0.005, min_lr=1e-6):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = init_lr
    dx = 2.0 / n

    while time.time() - start_time < time_budget:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        f_curr = float(corr[max_idx]) * dx

        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        if k_star >= 0:
            ns = n - k_star
            if ns > 0:
                grad[:ns] += one_minus_h[k_star:]
            if k_star > 0:
                grad[k_star:] -= h[:ns]
        else:
            p = -k_star
            np2 = n - p
            if np2 > 0:
                grad[p:] += one_minus_h[:np2]
                grad[:np2] -= h[p:]

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-10:
            break

        grad_normed = grad / g_norm
        accepted = False
        for _ in range(12):
            h_try = project_box_sum(h - lr * grad_normed, target_sum)
            f_try = compute_c5(h_try)
            dist = lr * float(np.dot(grad_normed, h - h_try))
            if f_try + 0.1 * dist < f_curr + 1e-12:
                h = h_try
                accepted = True
                if f_try < best_score:
                    best_score = f_try
                    best_h = h.copy()
                break
            lr *= 0.5
            if lr < min_lr:
                break
        if not accepted:
            break
    return best_h, best_score

    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
