    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(60):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 3 == 0:
            top_k = 8
            top_indices = np.argsort(full)[::-1][:top_k]
            best_move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.5]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 0.5]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move_found = True
                    break
            if not best_move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        elif it % 3 == 1:
            blen = rng.randint(4, min(16, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_trial = h.copy()
            h_trial[a:a+blen] = h[a:a+blen][::-1]
            full_trial = full_corr_fast(h_trial)
            c5_trial = float(full_trial.max()) * dx
            delta_c5 = c5_trial - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_trial
                full = full_trial
                current_c5 = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            continue
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def smooth_gradient_directed(h, alpha, corr_full, top_n=20):
    n = len(h)
    max_c = corr_full.max()
    top_idx = np.argsort(corr_full)[::-1][:top_n]
    w = np.zeros(len(corr_full))
    for idx in top_idx:
        w[idx] = np.exp(alpha * (corr_full[idx] - max_c))
    w /= w.sum() + 1e-30
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    grad = 1.0 - 2.0 * conv
    return grad

def continuous_refine_advanced(h0, target, start_time, time_limit):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alphas = [5, 10, 25, 50, 100, 250, 500, 1000, 3000, 8000, 20000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(40.0, time_limit - (time.time() - start_time))
        if phase_time < 5:
            break
        lr = 3.0 / (alpha + 5.0)
        lr_min = lr * 0.01
        momentum = 0.8
        v = np.zeros(n)
        n_iter = int(phase_time * 500)
        n_iter = max(n_iter, 100)
        stall_count = 0
        max_stall = 200

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            if stall_count > max_stall:
                break
            corr = full_corr_fast(h)
            grad = smooth_gradient_directed(h, alpha, corr, top_n=max(10, alpha // 5))
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 30 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                    lr = max(lr_min, lr * 0.8)
    return best_h, best_score

def continuous_refine_weighted(h0, target, start_time, time_limit):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alphas = [8, 20, 40, 80, 160, 400, 800, 2000, 5000, 12000, 30000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(35.0, time_limit - (time.time() - start_time))
        if phase_time < 4:
            break
        lr = 4.0 / (alpha * np.log(alpha + 2))
        lr = min(lr, 0.5)
        lr_min = lr * 0.005
        momentum = 0.75
        v = np.zeros(n)
        n_iter = int(phase_time * 400)
        n_iter = max(n_iter, 80)
        prev_sc = None
        stall = 0

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            if stall > 300:
                break
            corr = full_corr_fast(h)
            sc = float(corr.max()) * (2.0 / n)
            if prev_sc is not None and sc > prev_sc:
                h = h - 0.5 * (h - (h - v))
                v *= 0.3
                lr = max(lr_min, lr * 0.5)
                stall += 1
            prev_sc = sc
            grad = smooth_gradient_directed(h, alpha, corr, top_n=max(5, n // (alpha + 1)))
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 20 == 0 or it == n_iter - 1:
                sc2 = compute_c5(h)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h.copy()
                    stall = 0
    return best_h, best_score

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_box_sum(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick binary polish with greedy
    allowed = min(40, tleft() * 0.04)
    h_bin = project_binary(best_h, int(n_init / 2))
    h_g, c5_g = fast_greedy(h_bin, max_iters=50000, rng=rng,
                            time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Binary SA
    allowed = min(120, tleft() * 0.12)
    n_sa = int(min(3000000, allowed * 50000))
    h_sa, c5_sa = sa_focused(best_h, n_sa, 0.02, 0.0001, rng,
                             time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Aggressive continuous refinement (main workhorse)
    allowed = min(300, tleft() * 0.3)
    h_ref, sc = continuous_refine_advanced(best_h, target_init, start_time, allowed)
    if sc < best_score:
        best_score = sc
        best_h = h_ref.copy()

    # Phase 4: Second continuous refinement with different params
    if tleft() > 80:
        allowed = min(200, tleft() * 0.2)
        h_ref2, sc2 = continuous_refine_weighted(best_h, target_init, start_time, allowed)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 5: SLSQP on the continuous solution
    if tleft() > 80:
        allowed = min(tleft() - 40, 200)
        alphas_slsqp = [20, 50, 100, 300, 800, 2000, 6000]
        h_sls, sc_sls = slsqp_refine(best_h, n_init, target_init, alphas_slsqp,
                                      2000, 1e-14, time_limit=allowed, start_time=start_time)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls.copy()

    # Phase 6: Binary SA from best continuous (round and refine)
    if tleft() > 60:
        h_bin2 = project_binary(best_h, int(n_init / 2))
        allowed = min(60, tleft() * 0.1)
        n_sa2 = int(min(1500000, allowed * 40000))
        h_sa2, c5_sa2 = sa_focused(h_bin2, n_sa2, 0.015, 0.00005, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 7: Continuous refinement from the binary result
    if tleft() > 80:
        allowed = min(tleft() - 40, 250)
        h_ref3, sc3 = continuous_refine_advanced(best_h, target_init, start_time, allowed)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    # Phase 8: Try different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init + 12, n_init - 8]:
        if tleft() < 60:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = float(n_try) / 2.0
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, int(n_try / 2))
        allowed = min(30, tleft() * 0.06)
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=20000, rng=rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr_try)
        if tleft() > 40:
            allowed2 = min(40, tleft() * 0.08)
            h_r3, sc_r3 = continuous_refine_advanced(h_g3, target_try, start_time, allowed2)
            if sc_r3 < c5_g3:
                c5_g3 = sc_r3
                h_g3 = h_r3
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        allowed = max(10.0, tleft() - 15)
        h_final_sls, sc_final = slsqp_refine(best_h, len(best_h), float(len(best_h)) / 2.0,
                                              [500, 2000, 8000, 20000], 3000, 1e-15,
                                              time_limit=allowed, start_time=start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls.copy()

    # Phase 10: Final gradient burst
    if tleft() > 30:
        allowed = max(5.0, tleft() - 10)
        h_final_grad, sc_final2 = continuous_refine_advanced(best_h, float(len(best_h)) / 2.0,
                                                              start_time, allowed)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_h = h_final_grad.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
