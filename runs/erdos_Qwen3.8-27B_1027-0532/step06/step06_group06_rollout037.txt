def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):

            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1

Let me think about the binary local search more carefully. The previous program's greedy_binary:
1. Find the worst shift (max correlation)
2. Find candidates: 1s that overlap with 0s at that shift, and 0s that overlap with 1s
3. Try all pairs from the top candidates
4. Accept the best improving swap

This is a 1-exchange heuristic. To improve it, I can:
1. Try 2-exchanges and 3-exchanges (swap 2 or 3 pairs simultaneously)
2. Use "no-improvement threshold" to break ties
3. Apply random kicks when stuck (ILS)

For 2-exchanges: swap pairs (i1,j1) and (i2,j2) simultaneously. The change in correlation at shift k is the sum of changes from each pair (approximately, ignoring interactions). This gives O(n_i^2 * n_j^2) candidates, which is too many. Instead, I can:
- First find the best single swap
- Then look for a second swap that, combined with the first, gives better improvement
- This is O(n_i * n_j * n_i * n_j) in the worst case, but with pruning it's manageable

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(-h[ones])[:need]  # remove the smallest h values
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]  # add the largest h values
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def lse_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def lse_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    one_minus_h = 1.0 - h
    j_arr = np.arange(N)
    m_arr = np.arange(2 * N - 1)
    
    idx_f = j_arr[:, None] + N - 1 - m_arr[None, :]
    idx_s = j_arr[:, None] + m_arr[None, :] - N + 1
    
    mask_f = (idx_f >= 0) & (idx_f < N)
    mask_s = (idx_s >= 0) & (idx_s < N)
    
    idx_f_c = np.clip(idx_f, 0, N - 1)
    idx_s_c = np.clip(idx_s, 0, N - 1)
    
    w = weights.astype(float)[None, :]
    term1 = np.sum(w * one_minus_h[idx_f_c] * mask_f, axis=1)
    term2 = np.sum(w * h[idx_s_c] * mask_s, axis=1)
    
    grad = (term1 - term2) * dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def run_continuous(h_start, target, time_limit, start_time, beta_schedule=None):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000]
    
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        penalty = 500.0
        maxiter = 300
        try:
            res = minimize(lse_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=lse_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'iprint': -1, 'maxls': 50})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur or lse_objective(h_candidate, beta, target, penalty) < lse_objective(h_cur, beta, target, penalty):
                h_cur = h_candidate
        except Exception:
            pass
    
    return project_box_sum(h_cur, target)

def binary_local_search(h_start, target_sum, time_limit, start_time, rng, scan_width=30):
    n = len(h_start)

    corr = np.correlate(h, 1.0 - h, mode='full')

    best_h = h.copy()

    max_stall = 800

        max_val = float(np.max(corr))
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates at the worst shift
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
        j_cand = [j for j in zeros if 0 <= j - k_star < n and h[j - k_star] == 1]
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            # Try other high-correlation shifts
            top_thresh = max_val * 0.85
            top_indices = np.where(corr >= top_thresh)[0]
            found = False
            for t_idx in top_indices:
                if t_idx == max_idx:
                    continue
                k = t_idx - (n - 1)
                i_c = [i for i in ones if 0 <= i + k < n and h[i + k] == 0]
                j_c = [j for j in zeros if 0 <= j - k < n and h[j - k] == 1]
                if len(i_c) > 0 and len(j_c) > 0:
                    i_cand = i_c
                    j_cand = j_c
                    max_idx = t_idx
                    k_star = k
                    found = True
                    break
            if not found:
                stall += 1
                continue

        n_j = min(len(j_cand), scan_width)

        top_thresh = max_val * 0.75
        top_indices = np.where(corr >= top_thresh)[0]
        if len(top_indices) > 30:
            top_indices = top_indices[:30]

        best_j = -1

                    best_i = i
                    best_j = j

            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')

                    h[j] = 0
                    corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
