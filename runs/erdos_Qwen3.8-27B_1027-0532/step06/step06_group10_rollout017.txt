import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_val(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

        self.alpha = alpha

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_budget, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    con = SumEq(target)
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break

                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)

def greedy_cd_multi(h0, n, target, time_budget, n_cand=30, max_sweeps=30000, n_top_shifts=5):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        sorted_corr = np.argsort(corr)[::-1]
        top_shifts = sorted_corr[:n_top_shifts]
        c_max = corr[top_shifts[0]]

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w_s = np.exp(30.0 * (corr[s_idx] - c_max))
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w_s * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w_s * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_delta = 0.0
        best_c5_trial = c5_val
        best_i = -1
        best_j = -1

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                h_trial = h.copy()
                h_trial[i] -= delta_star
                h_trial[j] += delta_star
                h_trial = np.clip(h_trial, 0.0, 1.0)
                h_trial = project_feasible(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5_trial - 1e-14:
                    best_c5_trial = c5_trial
                    best_delta = delta_star
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def lp_refine(h0, n, target, time_budget, n_active=30, max_rounds=50):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for rnd in range(max_rounds):
        if time.time() - start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            improved = False
            for mix in [0.7, 0.5, 0.3]:
                h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_smooth(h0, n, target, time_budget, max_steps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    for step in range(max_steps):
        if time.time() - start > time_budget:
            break
        alpha = 100.0 * (1.0 + step * 0.02)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        grad = smooth_obj_grad(h, alpha, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_try = project_feasible(h + beta * v - lr_trial * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                v = v + grad
                v = v / (np.linalg.norm(v) + 1e-15) * min(np.linalg.norm(v), 1.0)
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
            if step > 200:
                break
    return best_h, best_c5


def random_perturb(h, n, target, n_trials=200, sigma=0.05):

            best_score = s_final
            best_h = h_final

    # Phase 12: Final greedy polish
    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = float(n_cur) / 2.0
        t_alloc = min(15.0, tleft() * 0.7)
        h_cd, c5_cd = greedy_cd_multi(best_h, n_cur, target_cur,
                                       time_budget=t_alloc, n_cand=35,
                                       max_sweeps=8000, n_top_shifts=7)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
