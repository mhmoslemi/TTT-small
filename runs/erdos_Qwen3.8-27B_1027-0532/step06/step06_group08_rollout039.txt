**Key insight**: The previous approach does binary swaps (t=1). But for the continuous optimum, we might want fractional swaps (0 < t < 1). Let me implement a version that tries fractional swaps.

For a fractional swap (h[i] -= t, h[j] += t where 0 ≤ t ≤ 1):
The change in correlation is t * delta (where delta is the same as the full swap delta).
So new_corr = corr + t * delta.
We want to minimize max(corr + t * delta) over t ∈ [0, 1].

This is a 1D optimization! We can solve it analytically:
- The function f(t) = max_m (corr[m] + t * delta[m]) is piecewise linear in t.
- The minimum occurs either at t=0, t=1, or at a crossing point of two lines.

We can find the optimal t by:
1. Computing all crossing points: for each pair (m, m'), find t where corr[m] + t*delta[m] = corr[m'] + t*delta[m']
2. Evaluating f(t) at all critical points and at t=0, t=1
3. Taking the minimum

This is O(M²) where M = 2n-1, which for n=600 is about 720² = 518400 pairs. That's feasible per iteration if we limit the number of iterations.

Actually, we can do better. The minimum of max of linear functions over [0,1] can be found by:
1. At t=0: max is corr.max()
2. At t=1: max is (corr + delta).max()
3. In between: the optimal t is where the "maximizing" lines cross

A simpler approach: binary search on t, or just evaluate at a few candidate points.

Actually, the cleanest approach: 
- The function f(t) = max_m (a_m + b_m * t) where a_m = corr[m], b_m = delta[m]
- This is convex in t (max of linear functions)
- We can find the minimum by checking where the derivative changes sign
- The derivative is b_m* where m* achieves the max
- At the optimum, either the max is achieved by multiple m's (and their b_m* average to 0), or the minimum is at a boundary

For practical purposes, let me just use a ternary search or evaluate at a grid of t values.

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                candidates_i = ones[A[ones] <= 1.0]
                if len(candidates_i) == 0:
                    candidates_i = ones
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                candidates_j = zeros[B[zeros] >= 1.0]
                if len(candidates_j) == 0:
                    candidates_j = zeros
                i = candidates_i[rng.randint(len(candidates_i))]
                j = candidates_j[rng.randint(len(candidates_j))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((corr + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def continuous_refine(h, max_iters, n_trials, time_limit, start_time, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 400:
            break

        max_val = float(corr.max())
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improvement = 0.0
        best_i, best_j, best_t = -1, -1, 0.0

        top_k_idx = np.argsort(corr)[-4:]
        for k_pick in top_k_idx:
            k_star = int(k_pick) - (n - 1)
            if k_star == 0:
                continue

            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += (1.0 - h[idx[valid]])
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += (1.0 - h[idx[valid]])

            B = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += (1.0 - h[idx[valid]])
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]

            top_i = ones[np.argsort(-A[ones])[:min(25, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(25, len(zeros))]]

            for i in top_i:
                for j in top_j:
                    delta = compute_delta(h, i, j)
                    t, new_max = optimal_t(corr, delta)
                    if 0 < t < 1:
                        improvement = max_val - new_max
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_i, best_j, best_t = i, j, t

        if best_improvement < 1e-12:
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                t, new_max = optimal_t(corr, delta)
                improvement = max_val - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j, best_t = i, j, t

        if best_improvement > 1e-12 and best_i >= 0:
            h[best_i] -= best_t
            h[best_j] += best_t
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(corr.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def projected_grad_descent(h, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 4000:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += (-h[abs_s:])
        grad /= topk

        h = h - lr * grad
        h = project_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5
