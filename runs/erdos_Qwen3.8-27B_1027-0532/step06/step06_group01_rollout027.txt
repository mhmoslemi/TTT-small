import numpy as np
from scipy.optimize import minimize
import time

# Global target for SLSQP constraint
_S = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        idx2 = np.where(h < 1.0)[0] if r > 0 else np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += r / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def slq_con_fun(x):
    return np.sum(x) - _S

def slq_con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol):
    global _S
    _S = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_con_fun, 'jac': slq_con_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    h_cur = h0.copy()
    for a in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(a,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except:
            pass
    return best_h, best_c5

    best_h = h.copy()

    idxs = np.arange(n)
    mid = n - 1
    
    for step in range(max_steps):

        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Softmax weights over top shifts
        top_idx = np.argsort(corr)[-n_active:]
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)  # Center to maintain sum constraint
        
        # Line search
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.025]:

            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        
        if not improved and step > 2000:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_S = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        idx2 = np.where(h < 1.0)[0] if r > 0 else np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += r / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)


def slq_con_fun(x):
    return np.sum(x) - _S


def slq_con_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h0, n, target, alphas, maxiter, ftol):
    global _S
    _S = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slq_con_fun, 'jac': slq_con_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    h_cur = h0.copy()
    for a in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(a,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def pair_cd(h0, n, target, time_limit, n_cand=25, max_sweeps=50000, seed=42):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    rng = np.random.RandomState(seed)
    perturb_count = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                if perturb_count < 5:
                    perturb_count += 1
                    mask = rng.rand(n) < 0.3
                    h[mask] += rng.uniform(-0.02, 0.02, mask.sum())
                    h = project_feasible(h, target)
                    no_improve = 0
                else:
                    break

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_g = np.argsort(-grad)
        high_set = sorted_g[:n_cand]
        low_set = sorted_g[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)

        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_imp = 0.0
        best_i = -1
        best_j = -1
        best_d = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            pi = profiles[ci]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                dmax = min(hi_val, 1.0 - hj_val)
                if dmax < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = pi - profiles[cj]

                lo = 0.0
                hi_d = dmax
                for _ in range(10):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid

                d_star = 0.5 * (lo + hi_d)
                if d_star < 1e-10:
                    continue

                new_max = float(np.max(corr + d_star * g))
                imp = cur_max - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_i = int(i)
                    best_j = int(j)
                    best_d = d_star

        if best_i >= 0 and best_imp > 1e-13:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

        if best_move is not None and best_imp > 1e-13:

def proj_subgrad(h0, n, target, time_limit, max_steps=100000, n_active=5, base_lr=0.03):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-n_active:]
        alpha_w = 40.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 3000:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
