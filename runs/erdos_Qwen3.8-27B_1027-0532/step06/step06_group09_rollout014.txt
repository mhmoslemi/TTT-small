import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_slp_data(h_star):
    """Compute the linearization data for SLP.
    Returns (grads, b) where the linearized constraint for lag k is:
    grads[k] @ h + b[k] <= t
    """
    n = len(h_star)
    n_corr = 2 * n - 1
    mid = n - 1
    grads = np.zeros((n_corr, n))
    b = np.zeros(n_corr)
    
    for k_idx in range(n_corr):
        k = k_idx - mid
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_start >= i_end:
            b[k_idx] = 0.0
            continue
        i_range = np.arange(i_start, i_end)
        j_plus = i_range + k
        j_minus = i_range
        vals_k = np.sum(h_star[j_plus] * (1.0 - h_star[j_minus]))
        grads[k_idx, j_plus] += (1.0 - h_star[j_minus])
        grads[k_idx, j_minus] += (-h_star[j_plus])
        b[k_idx] = vals_k - grads[k_idx] @ h_star
    
    return grads, b

def slp_solve(h_star, n, target, max_iters=100, time_limit=60.0, tol=1e-10):
    """Successive Linear Programming to minimize max_k c[k]."""
    start = time.time()
    h = h_star.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute linearization
        grads, b = compute_slp_data(h)
        
        # Solve LP
        h_cvx = cp.Variable(n)
        t = cp.Variable()
        objective = cp.Minimize(t)
        constraints = [
            cp.sum(h_cvx) == target,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        for k_idx in range(2 * n - 1):
            constraints.append(grads[k_idx] @ h_cvx + b[k_idx] <= t)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.CLARABEL, verbose=False)
                except Exception:
                    break
        
        if h_cvx.value is None:
            break
        h_new = np.clip(h_cvx.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Line search
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try interpolation
            improved = False
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h + alpha * (h_new - h), target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                if abs(c5_new - best_c5) < tol:
                    break
                # Stuck, try a small perturbation
                h = project_feasible(h + 0.01 * np.random.randn(n), target)
        
        if it > 0 and abs(compute_c5(h) - best_c5) < 1e-8:
            break
    
    return best_h, best_c5

def pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, patience=100):
    """Coordinate descent with pair updates."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        k_star = int(np.argmax(corr)) - mid
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] += -h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_trial = c5_val

        for i in high_set:
            if i in low_set:
                continue
            ip = (i + shifts >= 0) & (i + shifts < n)
            j_p = np.zeros(n_corr)
            j_p[ip] = h[i + shifts[ip]]
            im = (i - shifts >= 0) & (i - shifts < n)
            j_m = np.zeros(n_corr)
            j_m[im] = h[i - shifts[im]]
            
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                jp = (j + shifts >= 0) & (j + shifts < n)
                jp_arr = np.zeros(n_corr)
                jp_arr[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jm_arr = np.zeros(n_corr)
                jm_arr[jm] = h[j - shifts[jm]]
                
                g = j_p + j_m - jp_arr - jm_arr
                delta_lo2 = max(delta_lo, -1e-10)
                delta_hi2 = min(delta_hi, 1e-10)

                    delta_hi2 = max(delta_hi2, delta_lo2 + 1e-10)

                    else:

                    continue

                    best_trial = approx
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

        return h_new

        return h_start

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
        else:
            idx = np.where(h > 1e-13)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] += per
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_slp_data(h_star):
    n = len(h_star)
    n_corr = 2 * n - 1
    mid = n - 1
    grads = np.zeros((n_corr, n))
    b = np.zeros(n_corr)
    for k_idx in range(n_corr):
        k = k_idx - mid
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_start >= i_end:
            b[k_idx] = 0.0
            continue
        i_range = np.arange(i_start, i_end)
        j_plus = i_range + k
        j_minus = i_range
        vals_k = np.sum(h_star[j_plus] * (1.0 - h_star[j_minus]))
        grads[k_idx, j_plus] += (1.0 - h_star[j_minus])
        grads[k_idx, j_minus] += (-h_star[j_plus])
        b[k_idx] = vals_k - np.dot(grads[k_idx], h_star)
    return grads, b

def slp_solve(h_star, n, target, max_iters=100, time_limit=60.0, tol=1e-12):
    start = time.time()
    h = h_star.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        grads, b = compute_slp_data(h)
        h_cvx = cp.Variable(n)
        t = cp.Variable()
        objective = cp.Minimize(t)
        constraints = [
            cp.sum(h_cvx) == target,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        for k_idx in range(2 * n - 1):
            constraints.append(grads[k_idx] @ h_cvx + b[k_idx] <= t)
        prob = cp.Problem(objective, constraints)
        solved = False
        for solver in [cp.GLOP, cp.CLARABEL, cp.CBC]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_cvx.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.clip(h_cvx.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stale = 0
        else:
            improved = False
            for alpha in [0.7, 0.4, 0.2, 0.1]:
                h_try = project_feasible(h + alpha * (h_new - h), target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                stale += 1
                if stale > 15:
                    break
                h = project_feasible(h + 0.005 * np.random.RandomState(it).randn(n), target)
        if it > 0 and abs(compute_c5(h) - best_c5) < tol * 50:
            stale += 3
            if stale > 15:
                break
    return best_h, best_c5

def pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, patience=150):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        k_star = int(np.argmax(corr)) - mid
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] += -h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_trial = c5_val
        for i in high_set:
            ip = (i + shifts >= 0) & (i + shifts < n)
            i_p = np.zeros(n_corr)
            i_p[ip] = h[i + shifts[ip]]
            im = (i - shifts >= 0) & (i - shifts < n)
            i_m = np.zeros(n_corr)
            i_m[im] = h[i - shifts[im]]
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                jp = (j + shifts >= 0) & (j + shifts < n)
                j_p = np.zeros(n_corr)
                j_p[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                j_m = np.zeros(n_corr)
                j_m[jm] = h[j - shifts[jm]]
                g = i_p + i_m - j_p - j_m
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, a_d + 1e-10)
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def game_theoretic_descent(h_init, n, target, time_limit, max_steps=5000, eta_base=1.0, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve_count = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 400:
                break
        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for _ in range(8):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = lr * 1.1
                improved = True
                break
            lr *= 0.5
        if not improved:
            best_lr *= 0.9
            if best_lr < 0.001:
                break
        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.7 * lam + 0.3 * lam_new
    return best_h, best_c5

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    def smooth_obj(h):
        corr = compute_corr(h)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def slp_active_set(h_star, n, target, max_iters=80, time_limit=60.0, n_active_ref=50):
    start = time.time()
    h = h_star.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    active = np.where(corr >= corr.max() - 0.1 * corr.max() * (2.0/n))[0]
    if len(active) < n_active_ref:
        active = np.argsort(corr)[-n_active_ref:]
    stale = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c_max = np.max(corr) * (2.0/n)
        active = np.where(corr >= c_max / (2.0/n) - 0.05)[0]
        if len(active) < 5:
            active = np.argsort(corr)[-n_active_ref:]
        grads_all, b_all = compute_slp_data(h)
        h_cvx = cp.Variable(n)
        t = cp.Variable()
        objective = cp.Minimize(t)
        constraints = [
            cp.sum(h_cvx) == target,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        for k_idx in active:
            constraints.append(grads_all[k_idx] @ h_cvx + b_all[k_idx] <= t)
        prob = cp.Problem(objective, constraints)
        solved = False
        for solver in [cp.GLOP, cp.CLARABEL, cp.CBC]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_cvx.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.clip(h_cvx.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stale = 0
        else:
            improved = False
            for alpha in [0.6, 0.3, 0.15]:
                h_try = project_feasible(h + alpha * (h_new - h), target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                stale += 1
                if stale > 12:
                    break
    return best_h, best_c5
