def fast_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    curr_c5 = best_c5
    positions = np.arange(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for it in range(n_iters):
        if (it & 1023) == 0 and time.time() - start_time > time_limit:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Pick a shift to target (biased toward worst)
        # Use top-5 shifts with probability proportional to their values
        top5 = np.argsort(corr)[-5:]
        k_idx = top5[rng.randint(5)]
        k = shifts[k_idx]
        
        if k == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
        else:
            # Find candidate i where h[i+k]+h[i-k] is large (1s)
            # And candidate j where h[j+k]+h[j-k] is small (0s)
            # Pre-compute influence for this k
            infl = np.zeros(n)
            if k > 0:
                infl[:n-k] += h[k:]
                infl[k:] += h[:n-k]
            else:
                infl[-k:] += h[:k]
                infl[:k] += h[-k:]
            
            # i should be a 1 with high influence, j should be a 0 with low influence
            ones_mask = (h == 1.0)
            zeros_mask = (h == 0.0)
            if ones_mask.sum() == 0 or zeros_mask.sum() == 0:
                break
            
            i_candidates = np.where(ones_mask)[0]
            j_candidates = np.where(zeros_mask)[0]
            
            # Bias toward high influence for i, low for j
            i_wts = infl[i_candidates]
            j_wts = -infl[j_candidates]
            
            # Simple: pick randomly from top candidates
            i = i_candidates[rng.randint(len(i_candidates))]
            j = j_candidates[rng.randint(len(j_candidates))]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        idx = i + shifts
        mask = (idx >= 0) & (idx < n)
        delta[mask] += h[idx[mask]]
        idx = i - shifts

        delta[mask] += h[idx[mask]]
        idx = j - shifts

        delta[mask] -= h[idx[mask]]
        idx = j + shifts
        mask = (idx >= 0) & (idx < n)
        delta[mask] -= h[idx[mask]]
        
        new_corr = corr + delta
        new_max = float(new_corr.max())
        delta_c5 = (new_max - float(corr.max())) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0

def targeted_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(2*n-1) - (n-1)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    curr_max = float(corr.max())

        if (it & 511) == 0 and time.time() - start_time > time_limit:
            break

        t = t_init * (t_final / t_init) ** ratio

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            if y[idx] > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= y[idx]
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r > 0.5)[0]
    zeros = np.where(h_r < 0.5)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_max_value(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    top_idx = np.argsort(p)[-topk:]
    top_p = p[top_idx]
    top_p = top_p / top_p.sum()
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m, pm in zip(top_idx, top_p):
        if pm < 1e-12:
            continue
        s = shifts[m]
        if s >= 0:
            grad[:n - s] += pm * (1.0 - h[s:])
            grad[s:] += -pm * h[:n - s]
        else:
            abs_s = -s
            grad[abs_s:] += pm * (1.0 - h[:n - abs_s])
            grad[:n - abs_s] += -pm * h[abs_s:]
    return grad

def fast_greedy_targeted(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(2 * n - 1) - (n - 1)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0

    for it in range(max_iters):
        if (it & 255) == 0 and time.time() - start_time > time_limit:
            break
        if stall > 300:
            break
        max_val = float(corr.max())
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top5 = np.argsort(corr)[-5:]
        found = False
        for k_idx in top5:
            k_star = int(shifts[k_idx])
            if k_star == 0:
                continue
            ak = abs(k_star)
            infl = np.zeros(n)
            infl[:n - ak] += h[ak:]
            infl[ak:] += h[:n - ak]
            n_ci = min(20, len(ones))
            n_cj = min(20, len(zeros))
            top_i = ones[np.argsort(infl[ones])[-n_ci:]]
            bot_j = zeros[np.argsort(infl[zeros])[:n_cj]]
            for i in top_i:
                for j in bot_j:
                    idx_i = i + shifts
                    mask_i = (idx_i >= 0) & (idx_i < n)
                    idx_j = j - shifts
                    mask_j = (idx_j >= 0) & (idx_j < n)
                    delta = np.zeros(2 * n - 1)
                    delta[mask_i] += h[idx_i[mask_i]]
                    idx2 = i - shifts
                    mask2 = (idx2 >= 0) & (idx2 < n)
                    delta[mask2] += h[idx2[mask2]]
                    idx3 = j + shifts
                    mask3 = (idx3 >= 0) & (idx3 < n)
                    delta[mask3] -= h[idx3[mask3]]
                    idx4 = j - shifts
                    mask4 = (idx4 >= 0) & (idx4 < n)
                    delta[mask4] -= h[idx4[mask4]]
                    new_max = float((corr + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        corr += delta
                        stall = 0
                        new_c5 = new_max * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def targeted_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(2 * n - 1) - (n - 1)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    curr_max = float(corr.max())
    positions = np.arange(n)

    for it in range(n_iters):
        if (it & 511) == 0 and time.time() - start_time > time_limit:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        mode = rng.random()
        if mode < 0.55:
            worst_idx = int(np.argmax(corr))
            k_star = int(shifts[worst_idx])
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                ak = abs(k_star)
                infl = np.zeros(n)
                infl[:n - ak] += h[ak:]
                infl[ak:] += h[:n - ak]
                n_cand = min(40, len(ones))
                top_i = ones[np.argsort(infl[ones])[-n_cand:]]
                i = top_i[rng.randint(len(top_i))]
                n_cand_j = min(40, len(zeros))
                bot_j = zeros[np.argsort(infl[zeros])[:n_cand_j]]
                j = bot_j[rng.randint(len(bot_j))]
        elif mode < 0.80:
            sorted_idx = np.argsort(corr)
            pick = sorted_idx[-(1 + rng.randint(3))]
            k_star = int(shifts[pick])
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                ak = abs(k_star)
                infl = np.zeros(n)
                infl[:n - ak] += h[ak:]
                infl[ak:] += h[:n - ak]
                n_cand = min(30, len(ones))
                top_i = ones[np.argsort(infl[ones])[-n_cand:]]
                i = top_i[rng.randint(len(top_i))]
                n_cand_j = min(30, len(zeros))
                bot_j = zeros[np.argsort(infl[zeros])[:n_cand_j]]
                j = bot_j[rng.randint(len(bot_j))]
        elif mode < 0.92:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            n_rand = rng.randint(2, 5)
            i_list = ones[rng.randint(0, len(ones), n_rand)]
            j_list = zeros[rng.randint(0, len(zeros), n_rand)]
            i = i_list[0]
            j = j_list[0]

        delta = np.zeros(2 * n - 1)
        idx = i + shifts
        mask = (idx >= 0) & (idx < n)
        delta[mask] += h[idx[mask]]
        idx2 = i - shifts
        mask2 = (idx2 >= 0) & (idx2 < n)
        delta[mask2] += h[idx2[mask2]]
        idx3 = j - shifts
        mask3 = (idx3 >= 0) & (idx3 < n)
        delta[mask3] -= h[idx3[mask3]]
        idx4 = j + shifts
        mask4 = (idx4 >= 0) & (idx4 < n)
        delta[mask4] -= h[idx4[mask4]]

        new_corr = corr + delta
        new_max = float(new_corr.max())
        delta_c5 = (new_max - curr_max) * dx

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            curr_max = new_max
            curr_c5 = new_max * dx
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    return best_h, best_c5

def projected_grad_lse(h, n_iters, alpha_seq, lr, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0

    for alpha in alpha_seq:
        if time.time() - start_time > time_limit:
            break
        stall_alpha = 0
        iters_for_alpha = n_iters // len(alpha_seq)
        for it in range(iters_for_alpha):
            if time.time() - start_time > time_limit:
                break
            if stall_alpha > 500:
                break
            topk = min(20, 2 * n - 1)
            grad = lse_grad(h, alpha, topk)
            h_new = h - lr * grad
            h_new = project_box_sum(h_new, target)
            corr = np.correlate(h_new, 1.0 - h_new, mode='full')
            curr_c5 = float(corr.max()) * dx
            if curr_c5 < best_c5 - 1e-14:
                best_c5 = curr_c5
                best_h = h_new.copy()
                h = h_new
                stall_alpha = 0
            else:
                h = h_new
                stall_alpha += 1
    return best_h, best_c5

def sa_with_restarts(h, total_time, start_time, rng, n_restarts=5):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_per_restart = total_time / n_restarts

    for r in range(n_restarts):
        if time.time() - start_time > total_time:
            break
        remaining = total_time - (time.time() - start_time)
        if remaining < 5:
            break
        t_alloc = min(time_per_restart, remaining * 0.7)
        t_init = 0.015 * (1 + r * 0.3)
        t_final = 0.00003
        n_iters = int(max(10000, t_alloc * 80000))
        h_cur, c5_cur = targeted_sa(best_h, n_iters, t_init, t_final, rng,
                                     t_alloc, start_time)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        if r < n_restarts - 1:
            h_pert = best_h.copy()
            n_pert = rng.randint(3, 8 + r * 2)
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                if i in ones:
                    ones = np.delete(ones, np.where(ones == i)[0][0])
                if j in zeros:
                    zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                if i in zeros:
                    zeros = np.append(zeros, i)
                if j in ones:
                    ones = np.append(ones, j)
            best_h_for_next = h_pert
            best_h = h_pert
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alpha, maxiter, ftol):
    corr = np.correlate(h_start, 1.0 - h_start, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    obj_val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    return obj_val

def slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():

        if allowed > 30:

        if n_try % 2 == 0 and n_try >= 200:
            target_try = n_try // 2
            h_interp = interpolate_to(best_cont_h, n_try, float(target_try))
            h_bin_try = project_binary(h_interp, target_try)
            allowed = min(30.0, tleft() * 0.10)
            if allowed > 10:
                h_g3, c5_g3 = fast_greedy_targeted(h_bin_try, max_iters=20000, rng=rng,

                                                        alpha_seq=[50, 150, 400],
                                                        lr=0.015, time_limit=allowed2 * 0.5,
                                                        start_time=start_time)
                    if c5_pg3 < best_cont_score:
                        best_cont_score = c5_pg3
                        best_cont_h = h_pg3.copy()

    # Phase 7: Final combined optimization

        # Try SA from best binary one more time with lower temp

    # Phase 8: Final SLSQP
    if tleft() > 20:
        n_use = len(best_cont_h)

                                          alphas=[200, 800, 3000],
