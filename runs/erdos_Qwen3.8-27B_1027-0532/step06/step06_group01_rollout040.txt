def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Bisection for the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        trial = np.clip(h - mid, 0.0, 1.0)
        if np.sum(trial) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h = h.copy()
            h[idx] += residual / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def multi_shift_grad(h, n_top=3):
    """Compute gradient using top n_top active shifts with softmax weights."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(corr)
    alpha_w = 50.0
    top_idx = np.argsort(corr)[-n_top:]
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w /= w.sum()
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for wi, si in zip(w, top_idx):
        k = int(si) - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    return grad, cur_max
