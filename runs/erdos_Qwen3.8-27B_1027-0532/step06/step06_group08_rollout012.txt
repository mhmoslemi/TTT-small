            valid = (idx >= 0) & (idx < n)
            B[valid] += (1.0 - h[idx[valid]])
            
            # We want A[i] + B[j] < 0 for the swap to help at this shift
            # So we want small A[i] (from ones) and small B[j] (from zeros)
            top_i = ones[np.argsort(A[ones])[:min(20, len(ones))]]
            top_j = zeros[np.argsort(B[zeros])[:min(20, len(zeros))]]
            
            for i in top_i[:8]:
                for j in top_j[:8]:
                    i = int(i)
                    j = int(j)
                    delta = compute_delta(h, i, j)
                    new_max = float((corr + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        corr = corr + delta
                        stall = 0
                        new_c5 = float(corr.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found_improve = True
                        break
                if found_improve:
                    break
            if found_improve:
                break
        
        # Try 2-swap if enabled and no 1-swap found
        if not found_improve and use_2swap:
            # Try a random 2-swap
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(20):
                    i1 = int(ones[rng.randint(len(ones))])
                    j1 = int(zeros[rng.randint(len(zeros))])
                    remaining_ones = np.setdiff1d(ones, [i1])
                    remaining_zeros = np.setdiff1d(zeros, [j1])
                    i2 = int(remaining_ones[rng.randint(len(remaining_ones))])
                    j2 = int(remaining_zeros[rng.randint(len(remaining_zeros))])
                    
                    delta = compute_delta(h, i1, j1) + compute_delta(h, i2, j2)
                    new_max = float((corr + delta).max())
                    if new_max < max_val:
                        h[i1] = 0.0
                        h[j1] = 1.0
                        h[i2] = 0.0
                        h[j2] = 1.0
                        corr = corr + delta
                        stall = 0
                        new_c5 = float(corr.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found_improve = True
                        break
        
        if not found_improve:
            stall += 1
    
    return best_h, best_c5

    """Simulated annealing with 1-swap and 2-swap moves."""
