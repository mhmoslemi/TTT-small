import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

def find_boundary_indices(h):
    n = len(h)
    bounds = []
    for i in range(n - 1):
        if h[i] != h[i + 1]:
            bounds.append(i)
            bounds.append(i + 1)
    arr = np.array(bounds, dtype=int)
    unique_arr = np.unique(arr)
    return unique_arr

def sa_with_boundary_bias(h, target_sum, start_time, time_limit, rng, t_init=0.005, t_final=0.000005, top_k_worst=30):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 5000
    it = 0
    max_iters = 100000000

    while time.time() - start_time < start_time + time_limit and stall < max_stall:
        it += 1
        ratio = min(1.0, it / (max_iters * 0.5))
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.random()
        h_new = h.copy()
        moved = False

        if move_type < 0.45:
            worst_idx = np.argsort(corr)[-top_k_worst:]
            k_val = int(worst_idx[rng.randint(len(worst_idx))]) - (n - 1)
            i_pool = [i for i in ones if 0 <= i + k_val < n and h[i + k_val] == 0]
            j_pool = [j for j in zeros if 0 <= j + k_val < n and h[j + k_val] == 1]
            if not i_pool:
                i_pool = [i for i in ones if 0 <= i + k_val < n]
            if not j_pool:
                j_pool = [j for j in zeros if 0 <= j + k_val < n]
            if len(i_pool) > 0 and len(j_pool) > 0:
                h_new[i_pool[rng.randint(len(i_pool))]] = 0.0
                h_new[j_pool[rng.randint(len(j_pool))]] = 1.0
                moved = True
        elif move_type < 0.75:
            boundary = find_boundary_indices(h)
            if len(boundary) > 0:
                idx = boundary[rng.randint(len(boundary))]
                if h[idx] == 1:
                    j = zeros[rng.randint(len(zeros))]
                    h_new[idx] = 0.0
                    h_new[j] = 1.0
                    moved = True
                else:
                    i = ones[rng.randint(len(ones))]
                    h_new[i] = 0.0
                    h_new[idx] = 1.0
                    moved = True
        elif move_type < 0.90:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True

        if not moved:
            stall += 1
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-15))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if it % 5000 == 0 and time.time() - start_time > start_time + time_limit * 0.5:
            t = max(t * 0.5, t_final)

    return best_h, best_c5

def greedy_focus(h, target_sum, start_time, time_limit, top_k=20, scan_range=40):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 200

    for it in range(100000):
        if time.time() - start_time > start_time + time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-top_k:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        hot_score = np.zeros(n)
        for t_idx in top_idx:
            k = int(t_idx) - (n - 1)
            if k >= 0:
                lo = max(0, k)
                hi = min(n, n - k)
                hot_score[lo:hi] += 1.0
            else:
                lo = 0
                hi = min(n, n + k)
                hot_score[lo:hi] += 1.0

        n_scan = min(len(ones), scan_range)
        n_scan_j = min(len(zeros), scan_range)
        hot_ones = ones[np.argsort(-hot_score[ones])[:n_scan]]
        hot_zeros = zeros[np.argsort(-hot_score[zeros])[:n_scan_j]]

        best_improve = 1e-10
        best_i = -1
        best_j = -1

        for i in hot_ones:
            for j in hot_zeros:
                if i == j:
                    continue
                ok = True
                new_max = 0.0
                for t_idx in top_idx:
                    k = int(t_idx) - (n - 1)
                    delta = 0.0
                    if 0 <= i - k < n:
                        delta -= (1.0 - h[i - k])
                    if 0 <= j - k < n:
                        delta += (1.0 - h[j - k])
                    if 0 <= i + k < n:
                        delta += h[i + k]
                    if 0 <= j + k < n:
                        delta -= h[j + k]
                    nv = corr[t_idx] + delta
                    if nv > new_max:
                        new_max = nv
                    if nv >= max_val - 1e-10:
                        ok = False
                        break
                if ok:
                    imp = max_val - new_max
                    if imp > best_improve:
                        best_improve = imp
                        best_i = i
                        best_j = j

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def ils_search(h_start, target_sum, start_time, time_limit, rng, n_iterations=None):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    if n_iterations is None:
        elapsed_check = 0
        n_iterations = 50
    else:
        elapsed_check = 0

    for iteration in range(n_iterations):
        if time.time() - start_time > start_time + time_limit:
            break

        phase_time = max(5.0, (start_time + time_limit - time.time()) / max(1, (n_iterations - iteration) * 0.8))
        h, c5 = sa_with_boundary_bias(h, target_sum, time.time(), phase_time, rng,
                                       t_init=0.003, t_final=0.000002, top_k_worst=25)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        permute_size = rng.randint(3, 12)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        if len(ones) >= permute_size and len(zeros) >= permute_size:
            flip_ones = rng.choice(len(ones), permute_size, replace=False)
            flip_zeros = rng.choice(len(zeros), permute_size, replace=False)
            h_pert[ones[flip_ones]] = 0.0
            h_pert[zeros[flip_zeros]] = 1.0
        h = h_pert.copy()

    return best_h, best_c5

def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_slsqp_n = 0
_slsqp_target = 0.0

def _slsqp_cons_fun(h):
    return float(np.sum(h)) - _slsqp_target

def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def adam_continuous(h0, target_sum, start_time, time_budget, alphas, lr=0.0015, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > start_time + time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')

        while True:
            if time.time() - start_time > start_time + time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 400:
                break

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
        else:
            h = best_h.copy()

    return best_h, best_actual

def multi_start_sa(h_base, target_sum, start_time, time_limit, rng, n_starts=5):
    n = len(h_base)
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h)
    per_start = time_limit / n_starts

    for s in range(n_starts):
        if time.time() - start_time > start_time + time_limit:
            break
        pert_scale = rng.randint(5, 20)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        if len(ones) >= pert_scale and len(zeros) >= pert_scale:
            fi = rng.choice(len(ones), pert_scale, replace=False)
            fj = rng.choice(len(zeros), pert_scale, replace=False)
            h_pert[ones[fi]] = 0.0
            h_pert[zeros[fj]] = 1.0
        remaining = start_time + time_limit - time.time()
        t_alloc = min(per_start, remaining * 0.9)
        if t_alloc < 10:
            break
        h_sa, c5_sa = sa_with_boundary_bias(h_pert, target_sum, time.time(), t_alloc, rng,
                                             t_init=0.008, t_final=0.00001, top_k_worst=30)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

                                            t_init=0.004, t_final=0.000003, top_k_worst=30)

            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Greedy refinement

        gt2 = min(80, tleft() * 0.15)
        h_g2, c5_g2 = greedy_focus(best_h, target, start_time, gt2, top_k=20, scan_range=50)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        ms_t = min(300, tleft() * 0.4)
        h_ms, c5_ms = multi_start_sa(best_h, target, start_time, ms_t, rng, n_starts=4)

            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 5: ILS
    if tleft() > 200:
        ils_t = min(250, tleft() * 0.45)
        h_ils, c5_ils = ils_search(best_h, target, start_time, ils_t, rng, n_iterations=None)

            best_score = c5_ils
            best_h = h_ils.copy()

    # Phase 6: Final greedy
    if tleft() > 60:
        gt3 = min(40, tleft() * 0.3)
        h_g3, c5_g3 = greedy_focus(best_h, target, start_time, gt3, top_k=25, scan_range=40)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    # Phase 7: Continuous polish with Adam
    if tleft() > 80:
        adam_t = min(150, tleft() * 0.5)
        alphas = [200, 500, 1500, 5000, 15000, 50000]
        h_a, s_a = adam_continuous(best_h, float(target), start_time, adam_t,
                                     alphas=alphas, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8)
        h_a_bin = project_binary(h_a, target)
        s_a_bin = compute_c5(h_a_bin)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()
        elif s_a_bin < best_score:
            best_score = s_a_bin
            best_h = h_a_bin.copy()

    # Phase 8: SLSQP polish
    if tleft() > 50:
        sl_t = min(80, tleft() * 0.5)
        alphas_sl = [5000, 20000, 80000]
        h_ref, sc_ref = slsqp_refine(best_h, n, float(target), alphas_sl, 2000, sl_t, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 9: Final SA on best solution
    if tleft() > 30:
        final_t = min(50, tleft() * 0.7)
        h_f, c5_f = sa_with_boundary_bias(best_h, target, start_time, final_t, rng,
                                           t_init=0.002, t_final=0.000001, top_k_worst=40)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    # Final feasibility
    h_final = project_box_sum(best_h, float(target))
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
