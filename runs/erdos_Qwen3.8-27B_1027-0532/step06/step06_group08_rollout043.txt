        total = y.sum()
        if total > 1e-10:
            y = y + (diff / total) * y
    return np.clip(y, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r
    if current > target_sum:
        need = current - target_sum
        ones = np.where(h_r == 1.0)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = target_sum - current
        zeros = np.where(h_r == 0.0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0
    return h_r

def compute_gradient(h, w):
    n = len(h)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    nz = np.where(w > 1e-13)[0]
    for m in nz:
        wm = w[m]
        s = m - (n - 1)
        if s >= 0:
            term1[:n - s] += wm * (1.0 - h[s:])
            term2[s:] += wm * h[:n - s]
        else:
            a = -s
            term1[a:] += wm * (1.0 - h[:n - a])
            term2[:n - a] += wm * h[a:]
    return term1 - term2

def smooth_phase(h_init, start_time, time_budget, alpha_start, alpha_end, lr_start, lr_end, n_max_iters, decay_alpha_every=200):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = project_box_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0

    for it in range(n_max_iters):
        if time.time() - start_time > time_budget:
            break
        if stall > decay_alpha_every * 5:
            break

        ratio = it / max(n_max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio
        lr = lr_start * (lr_end / lr_start) ** ratio

        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c_max = c.max()
        shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
        exp_w = np.exp(shifted)
        w = exp_w / exp_w.sum()

        grad = compute_gradient(h, w) * dx
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)

        curr_c5 = compute_c5(h_new)
        if curr_c5 < best_c5 - 1e-15:
            best_c5 = curr_c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new

    return best_h, best_c5

def binary_sa_fast(h_init, n_iters, t_init, t_final, rng, start_time, time_budget):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    curr_c5 = best_c5
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_budget:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        if r < 0.4:
            max_k = int(np.argmax(corr))
            k_star = max_k - (n - 1)
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            elif k_star > 0:
                ci = ones[ones <= n - 1 - k_star]
                if len(ci) == 0:
                    ci = ones
                cj = zeros[k_star:]
                if len(cj) == 0:
                    cj = zeros
                i = ci[rng.randint(len(ci))]
                j = cj[rng.randint(len(cj))]
            else:
                a = -k_star
                ci = ones[ones >= a]
                if len(ci) == 0:
                    ci = ones
                cj = zeros[:n - a]
                if len(cj) == 0:
                    cj = zeros
                i = ci[rng.randint(len(ci))]
                j = cj[rng.randint(len(cj))]
        elif r < 0.7:
            top_idx = np.argsort(corr)[-5:]
            k_pick = top_idx[rng.randint(len(top_idx))]
            k_star = int(k_pick) - (n - 1)
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            elif k_star > 0:
                ci = ones[ones <= n - 1 - k_star]
                if len(ci) == 0:
                    ci = ones
                cj = zeros[k_star:]
                if len(cj) == 0:
                    cj = zeros
                i = ci[rng.randint(len(ci))]
                j = cj[rng.randint(len(cj))]
            else:
                a = -k_star
                ci = ones[ones >= a]
                if len(ci) == 0:
                    ci = ones
                cj = zeros[:n - a]
                if len(cj) == 0:
                    cj = zeros
                i = ci[rng.randint(len(ci))]
                j = cj[rng.randint(len(cj))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(new_corr.max()) * dx
        delta = new_c5 - curr_c5

        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            curr_c5 = new_c5
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    return np.interp(x_new, x_old, h_src)

def slsqp_polish(h_init, n, target, alphas, maxiter, start_time, time_budget):
    bounds = [(0.0, 1.0)] * n

    def objective(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    h_cur = h_init.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        try:
            res = minimize(lambda x: objective(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_raw = np.array(initial_h_values, dtype=float)
    n0 = len(h_raw)
    target0 = float(n0) / 2.0

    h_cont = project_box_sum(h_raw, target0)
    h_bin = project_binary(h_cont, int(target0))
    best_cont = h_cont.copy()
    best_cont_score = compute_c5(h_cont)
    best_bin = h_bin.copy()
    best_bin_score = compute_c5(h_bin)
    best_h = best_cont.copy()
    best_score = best_cont_score

    # Phase 1: Continuous PG - coarse
    alloc = min(180.0, tleft() * 0.15)
    if alloc > 20:
        n_it = int(min(50000, alloc * 800))
        h1, s1 = smooth_phase(h_cont, start_time, start_time + alloc,
                              10.0, 80.0, 0.05, 0.01, n_it)
        if s1 < best_cont_score:
            best_cont_score = s1
            best_cont = h1
            best_h = h1
            best_score = s1

    # Phase 2: Continuous PG - medium
    alloc = min(250.0, tleft() * 0.20)
    if alloc > 30:
        n_it = int(min(100000, alloc * 600))
        h2, s2 = smooth_phase(best_cont, start_time, start_time + alloc,
                              80.0, 400.0, 0.03, 0.005, n_it)
        if s2 < best_cont_score:
            best_cont_score = s2
            best_cont = h2
            best_h = h2
            best_score = s2

    # Phase 3: Binary SA
    alloc = min(150.0, tleft() * 0.12)
    if alloc > 20:
        h_b = project_binary(best_cont, int(len(best_cont) / 2.0))
        n_it = int(min(5000000, alloc * 30000))
        h3, s3 = binary_sa_fast(h_b, n_it, 0.015, 0.00005, rng,
                                 start_time, start_time + alloc)
        if s3 < best_bin_score:
            best_bin_score = s3
            best_bin = h3

    # Phase 4: Continuous PG - fine
    alloc = min(250.0, tleft() * 0.20)
    if alloc > 30:
        # Start from best of continuous and binary (interp)
        start_h = best_cont.copy()
        if best_bin_score < best_cont_score:
            start_h = np.interp(np.linspace(0, 1, len(best_cont)),
                                np.linspace(0, 1, len(best_bin)), best_bin)
            start_h = project_box_sum(start_h, len(start_h) / 2.0)
        n_it = int(min(150000, alloc * 800))
        h4, s4 = smooth_phase(start_h, start_time, start_time + alloc,
                              200.0, 2000.0, 0.015, 0.002, n_it)
        if s4 < best_cont_score:
            best_cont_score = s4
            best_cont = h4
            best_h = h4
            best_score = s4

    # Phase 5: SLSQP polish
    alloc = min(120.0, tleft() * 0.10)
    if alloc > 30:
        h5, s5 = slsqp_polish(best_cont, len(best_cont), float(len(best_cont)) / 2.0,
                               [100, 500, 2000], 3000, start_time, start_time + alloc)
        if s5 < best_cont_score:
            best_cont_score = s5
            best_cont = h5
            best_h = h5
            best_score = s5

    # Phase 6: Multi-resolution
    for n_try in [n0 + 4, n0 - 4, n0 + 8, n0 + 12]:
        if tleft() < 60:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        tgt = n_try / 2.0
        h_interp = project_box_sum(interpolate_h(best_cont, n_try), tgt)
        alloc1 = min(30.0, tleft() * 0.06)
        if alloc1 < 15:
            continue
        n_it = int(min(30000, alloc1 * 1000))
        h_m, s_m = smooth_phase(h_interp, start_time, start_time + alloc1,
                                100.0, 800.0, 0.02, 0.003, n_it)
        alloc2 = min(25.0, tleft() * 0.05)
        if alloc2 > 10:
            n_it2 = int(min(20000, alloc2 * 1000))
            h_m2, s_m2 = smooth_phase(h_m, start_time, start_time + alloc2,
                                       500.0, 3000.0, 0.008, 0.001, n_it2)
            if s_m2 < s_m:
                h_m, s_m = h_m2, s_m2
        if s_m < best_cont_score:
            best_cont_score = s_m
            best_cont = h_m
            best_h = h_m
            best_score = s_m

    # Phase 7: Final binary SA burst from best continuous
    if tleft() > 40:
        alloc = min(60.0, tleft() * 0.08)
        h_b2 = project_binary(best_cont, int(len(best_cont) / 2.0))
        n_it = int(min(2000000, alloc * 25000))
        h7, s7 = binary_sa_fast(h_b2, n_it, 0.01, 0.00003, rng,
                                 start_time, start_time + alloc)
        if s7 < best_bin_score:
            best_bin_score = s7
            best_bin = h7

    # Phase 8: Final ultra-fine continuous
    if tleft() > 30:
        alloc = min(80.0, tleft() * 0.10)
        n_it = int(min(100000, alloc * 1000))
        h8, s8 = smooth_phase(best_cont, start_time, start_time + alloc,
                              500.0, 5000.0, 0.005, 0.0005, n_it)
        if s8 < best_cont_score:
            best_cont_score = s8
            best_cont = h8
            best_h = h8
            best_score = s8

    # Phase 9: Final SLSQP
    if tleft() > 25:
        alloc = min(40.0, tleft() * 0.06)
        h9, s9 = slsqp_polish(best_cont, len(best_cont), float(len(best_cont)) / 2.0,
                               [500, 2000, 5000], 5000, start_time, start_time + alloc)
        if s9 < best_cont_score:
            best_cont_score = s9
            best_cont = h9
            best_h = h9
            best_score = s9

    # Final selection
    final_h = project_box_sum(best_cont, float(len(best_cont)) / 2.0)
    final_score = compute_c5(final_h)

    h_bin_final = project_binary(best_cont, int(len(best_cont) / 2.0))
    bin_final_score = compute_c5(h_bin_final)

    if bin_final_score < final_score:
        final_h = h_bin_final
        final_score = bin_final_score

    if best_bin_score < final_score:
        final_h = best_bin
        final_score = best_bin_score

    n_final = len(final_h)
    return final_h, final_score, n_final
