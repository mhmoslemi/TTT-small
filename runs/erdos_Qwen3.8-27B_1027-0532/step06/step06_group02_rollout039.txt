For each lag k (from -(n-1) to n-1):
- Valid pairs: i such that 0 ≤ i < n and 0 ≤ i+k < n
- dC(k)/dh[j] for valid j:
  - If j is a valid "i" (i.e., 0 ≤ j < n and 0 ≤ j+k < n): contribution is (1-h[j+k])
  - If j is a valid "i+k" (i.e., 0 ≤ j-k < n and 0 ≤ j < n): contribution is -h[j-k]
  - Note: both can be true (if 0≤j<n, 0≤j+k<n, AND 0≤j-k<n)

So:
dC(k)/dh[j] = (1-h[j+k]) * 1_{0≤j<n, 0≤j+k<n} - h[j-k] * 1_{0≤j<n, 0≤j-k<n}

Since j is always in [0,n) (that's our variable), the conditions simplify to:
dC(k)/dh[j] = (1-h[j+k]) * 1_{0≤j+k<n} - h[j-k] * 1_{0≤j-k<n}

For k ≥ 0:
- 0 ≤ j+k < n → 0 ≤ j < n-k
- 0 ≤ j-k < n → k ≤ j < n+k → j ≥ k (since j < n always)

So for k ≥ 0:
- First term: j = 0, 1, ..., n-k-1: grad[j] += w * (1-h[j+k])
- Second term: j = k, k+1, ..., n-1: grad[j] -= w * h[j-k]

For k < 0 (let m = -k > 0):
- 0 ≤ j+k < n → 0 ≤ j-m < n → m ≤ j < n+m → j ≥ m (since j < n)
- 0 ≤ j-k < n → 0 ≤ j+m < n → 0 ≤ j < n-m

So for k = -m < 0:
- First term: j = m, m+1, ..., n-1: grad[j] += w * (1-h[j-m])
- Second term: j = 0, 1, ..., n-m-1: grad[j] -= w * h[j+m]

```python
def fast_gradient(h, w, n):
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-18:
            continue
        if k == 0:
            # First term: all j: grad[j] += wk * (1-h[j])
            # Second term: all j: grad[j] -= wk * h[j]
            # Total: grad[j] += wk * (1-h[j]-h[j]) = wk*(1-2h[j])
            grad += wk * (1.0 - 2.0*h)
        elif k > 0:
            # First: j=0..n-k-1: grad[j] += wk*(1-h[j+k])
            grad[:n-k] += wk * (1.0 - h[k:])
            # Second: j=k..n-1: grad[j] -= wk*h[j-k]
            grad[k:] -= wk * h[:-k]
        else:
            m = -k
            # First: j=m..n-1: grad[j] += wk*(1-h[j-m])
            grad[m:] += wk * (1.0 - h[:-m])
            # Second: j=0..n-m-1: grad[j] -= wk*h[j+m]
            grad[:n-m] -= wk * h[m:]
    grad *= 2.0 / n
    return grad
```

Let me verify: for k > 0, first term: j ranges 0 to n-k-1, and h[j+k] ranges h[k] to h[n-1]. So `h[k:]` has length n-k. ✓
Second term: j ranges k to n-1, and h[j-k] ranges h[0] to h[n-k-1]. So `h[:-k]` has length n-k. ✓

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def fast_gradient(h, w, n):
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-18:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:-k]
        else:
            m = -k
            grad[m:] += wk * (1.0 - h[:-m])
            grad[:n-m] -= wk * h[m:]
    grad *= 2.0 / n
    return grad

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def smoothed_objective(h, alpha, n):
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    val = c_max + (1.0/alpha) * np.log(np.sum(np.exp(shifted)))
    return val * (2.0/n)

def smoothed_gradient(h, alpha, n):
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = fast_gradient(h, w, n)
    return grad

def lbfgsb_smooth(h0, n, target, alphas, max_iter_per_alpha, ftol, start_time, time_limit):
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        def obj(h_flat):
            return smoothed_objective(h_flat, alpha, n)
        
        def grad(h_flat):
            return smoothed_gradient(h_flat, alpha, n)
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)]*n,
                          options={'maxiter': max_iter_per_alpha, 'ftol': ftol,
                                   'gtol': 1e-8, 'maxls': 50})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    
    return best_h, best_score

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def sa_focused(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr, target_lags=5):
    n = len(h0)
    h = h0.copy().astype(float)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Target the worst lags
        if it % 3 == 0:
            top_indices = np.argsort(corr)[::-1][:target_lags]
            best_move = None
            best_new_c5 = current_c5
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                # Find positions that contribute to this lag's overlap
                # For lag k_star: C = sum h[i]*(1-h[i+k_star])
                # A 1 at position i contributes (1-h[i+k_star]) to the overlap
                # We want to move a 1 from a high-contribution position to a 0
                # that has high "absorption" (h[j-k_star] or h[j+k_star] is 1)
                
                pos = np.arange(n)
                if k_star > 0:
                    # i valid: 0 <= i < n-k_star; h[i+k_star] must be defined
                    valid_i = np.arange(0, n - k_star)
                else:
                    valid_i = np.arange(-k_star, n)

                contrib = np.zeros(n)
                idx = valid_i + k_star
                contrib[valid_i] = 1.0 - h[idx]
                
                # Where should we put the new 0 (new 1)?
                # A 0 at position j becomes 1. The change in overlap at lag k_star:
                # We lose the contribution of h[i] = 1: remove contrib[i]
                # We add: h[j] becomes 1, so it contributes (1-h[j+k_star]) to lag k_star
                # But also it affects other lags
                # Simplified: pick i with high contrib (large overlap contribution)
                # and j where placing a 1 would minimize new overlap
                
                # For the move i->j at lag k_star:
                # delta at lag k_star = -(1-h[i+k_star]) + (1-h[j+k_star]) [if j is in valid range]
                # - h[i-k_star] + h[j-k_star] [if in valid range for the -h[i-k] part]
                
                # Just pick randomly from top contributors
                i_cands = ones[np.isin(ones, valid_i)]
                if len(i_cands) > 0:
                    # Pick i with highest contribution
                    i_order = np.argsort(-contrib[i_cands])[:10]
                    i = i_cands[i_order[rng.randint(min(5, len(i_order)))]]
                else:
                    i = ones[rng.randint(len(ones))]
                
                # Pick j: prefer positions where h[j+k_star] = 1 and h[j-k_star] = 1
                # (so placing a 1 there doesn't add much overlap)
                j_score = np.zeros(n)
                idx_j = pos + k_star
                valid_j = (idx_j >= 0) & (idx_j < n)
                j_score[valid_j] = -h[idx_j[valid_j]]  # prefer h[j+k_star]=1
                idx_j2 = pos - k_star
                valid_j2 = (idx_j2 >= 0) & (idx_j2 < n)
                j_score[valid_j2] += -h[idx_j2[valid_j2]]
                j_score[ones] = -999  # don't pick from ones
                
                j_cand_list = zeros[j_score[zeros] >= -2.0]
                if len(j_cand_list) > 0:
                    # Weighted random selection
                    w_sel = j_score[j_cand_list] + 2.0
                    w_sel = np.maximum(w_sel, 0.1)
                    w_sel /= w_sel.sum()
                    j = j_cand_list[rng.choice(len(j_cand_list), p=w_sel)]
                else:
                    j = zeros[rng.randint(len(zeros))]
                
                delta = compute_delta(h, i, j, k_arr)
                new_corr = corr + delta
                new_max = float(np.max(new_corr))
                new_c5 = new_max * dx
                
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_move = (i, j, delta, new_corr)
            
            if best_move is not None:
                i, j, delta, new_corr = best_move
                delta_c5 = best_new_c5 - current_c5
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    current_c5 = best_new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Random move
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            new_c5 = new_max * dx
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0

                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def two_swap_sa(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    """SA with 2-swap moves (exchange two 1s with two 0s)."""
    n = len(h0)
    h = h0.copy().astype(float)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    pos = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                stall += 1
                continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                stall += 1
                continue
        
        # Compute delta for 2-swap
        delta = (compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr))
        # But we need to be careful: if the swaps interact
        # For simplicity, just compute the new correlation directly if delta looks bad
        new_corr = corr + delta
        new_c5 = float(np.max(new_corr)) * dx
        
        if new_c5 < 0:  # sanity check
            continue
            
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            # Recompute correlation to be safe (in case of interaction)
            corr = full_corr_fast(h)
            current_c5 = float(np.max(corr)) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def perturb_random(h, rng, n_swaps):
    n = len(h)

        j = zeros[rng.randint(len(zeros))]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def fast_gradient(h, w, n):
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-18:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:-k]
        else:
            m = -k
            grad[m:] += wk * (1.0 - h[:-m])
            grad[:n-m] -= wk * h[m:]
    grad *= 2.0 / n
    return grad

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:

    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def smoothed_grad(h, alpha, n):
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

def lbfgsb_phase(h0, n, target, alphas, max_iter, ftol, start_time, deadline):
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:

            break
        try:

                          h,

                          method='L-BFGS-B',

                                   'gtol': 1e-6, 'maxls': 40})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_score

        if time.time() >= deadline:
            break
        if stall > max_stall:
            break
        ratio = float(it) / float(max(n_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found_improve = False
        if it % 3 == 0:
            top_idx = np.argsort(corr)[::-1][:n_top]
            best_delta_c5 = 0.0
            best_i = -1
            best_j = -1
            best_new_corr = None
            for ti in top_idx:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                if k_star > 0:
                    valid_i = np.arange(0, n - k_star)
                else:
                    valid_i = np.arange(-k_star, n)
                contrib = np.zeros(n)
                idx2 = valid_i + k_star
                contrib[valid_i] = 1.0 - h[idx2]
                i_cands = ones[np.isin(ones, valid_i)]
                if len(i_cands) == 0:
                    continue
                top_n_i = min(8, len(i_cands))
                i_order = np.argsort(-contrib[i_cands])[:top_n_i]
                j_score = np.zeros(n)
                idx_j = pos + k_star
                vj = (idx_j >= 0) & (idx_j < n)
                j_score[vj] += h[idx_j[vj]]
                idx_j2 = pos - k_star
                vj2 = (idx_j2 >= 0) & (idx_j2 < n)
                j_score[vj2] += h[idx_j2[vj2]]
                j_score[ones] = -100.0
                j_list = zeros[j_score[zeros] >= -100.0]
                if len(j_list) == 0:
                    j_list = zeros
                top_n_j = min(8, len(j_list))
                j_order = np.argsort(-j_score[j_list])[:top_n_j]
                for ii in i_order:
                    i = i_cands[ii]
                    for jj in j_order:
                        j = j_list[jj]
                        delta = compute_delta(h, i, j, k_arr)
                        new_c = corr + delta
                        nc5 = float(np.max(new_c)) * dx
                        dc5 = nc5 - current_c5
                        if dc5 < best_delta_c5:
                            best_delta_c5 = dc5
                            best_i = i
                            best_j = j
                            best_new_corr = new_c
            if best_i >= 0:
                i, j = best_i, best_j
                dc5 = best_delta_c5
                if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = best_new_corr
                    current_c5 = best_c5 if dc5 >= 0 else current_c5 + dc5
                    current_c5 = float(np.max(corr)) * dx
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                    found_improve = True
        if not found_improve:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_corr = corr + delta
            nc5 = float(np.max(new_corr)) * dx
            dc5 = nc5 - current_c5
            if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                current_c5 = nc5
                if nc5 < best_c5:
                    best_c5 = nc5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def sa_block(h0, n_iters, t_init, t_final, rng, start_time, deadline, k_arr,
            block_frac=0.25, max_block=12):
    n = len(h0)
    h = h0.copy().astype(float)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 40000

    for it in range(n_iters):
        if time.time() >= deadline:
            break
        if stall > max_stall:
            break
        ratio = float(it) / float(max(n_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        use_block = rng.random() < block_frac
        if use_block:
            blen = rng.randint(2, min(max_block, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            if abs(h_new.sum() - h.sum()) > 1e-10:
                stall += 1
                continue
            corr_new = full_corr_fast(h_new)
            c5_new = float(np.max(corr_new)) * dx
            dc5 = c5_new - current_c5
            if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                h = h_new
                corr = corr_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_corr = corr + delta
            nc5 = float(np.max(new_corr)) * dx
            dc5 = nc5 - current_c5
            if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                current_c5 = nc5
                if nc5 < best_c5:
                    best_c5 = nc5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def two_swap_sa(h0, n_iters, t_init, t_final, rng, start_time, deadline, k_arr):
    n = len(h0)
    h = h0.copy().astype(float)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000

    for it in range(n_iters):
        if time.time() >= deadline:
            break
        if stall > max_stall:
            break
        ratio = float(it) / float(max(n_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            stall += 1
            continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            stall += 1
            continue

        delta = compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
        new_corr = corr + delta
        if np.min(new_corr) < -1e-6:
            stall += 1
            continue
        nc5 = float(np.max(new_corr)) * dx
        dc5 = nc5 - current_c5
        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            corr = full_corr_fast(h)
            current_c5 = float(np.max(corr)) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def perturb_random(h, rng, n_swaps):
    n = len(h)
    h_p = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h_p == 1.0)[0]
        zeros = np.where(h_p == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_p[i] = 0.0
        h_p[j] = 1.0
    return h_p

def perturb_block(h, rng, max_len):
    n = len(h)
    h_p = h.copy()
    blen = rng.randint(3, min(max_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_p[a:a+blen] = h[a:a+blen][::-1]
    return h_p

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def coarse_to_fine(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    x_new = np.linspace(0.0, 1.0, n_fine)
    x_old = np.linspace(0.0, 1.0, n_coarse)
    h_fine = np.interp(x_new, x_old, h_coarse)
    return project_binary(h_fine, target_fine)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    def deadline_from(allowed):
        return start_time + (time.time() - start_time) + allowed

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuous refinement
    if tleft() > 250:
        alphas = [3, 8, 20, 50, 120, 300, 800, 2000]
        allowed = min(150, tleft() * 0.12)
        h_ref, sc = lbfgsb_phase(best_h, n_init, float(target_init),
                                alphas, 300, 1e-12, start_time, deadline_from(allowed))
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 2: Focused SA on binary
    if tleft() > 200:
        h_bin = project_binary(best_h, target_init)
        allowed = min(180, tleft() * 0.18)
        n_sa = int(min(5000000, allowed * 30000))
        h_sa, c5_sa = sa_lag_focused(h_bin, n_sa, 0.02, 0.00003, rng,
                                      start_time, deadline_from(allowed), k_arr, n_top=5)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: L-BFGS-B again from refined binary
    if tleft() > 200:
        alphas2 = [10, 30, 80, 200, 500, 1500, 5000]
        allowed = min(100, tleft() * 0.1)
        h_ref2, sc2 = lbfgsb_phase(best_h, n_init, float(target_init),
                                    alphas2, 400, 1e-12, start_time, deadline_from(allowed))
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 4: Block SA
    if tleft() > 150:
        h_bin2 = project_binary(best_h, target_init)
        allowed = min(120, tleft() * 0.14)
        n_sa2 = int(min(3000000, allowed * 25000))
        h_sa2, c5_sa2 = sa_block(h_bin2, n_sa2, 0.015, 0.00002, rng,
                                  start_time, deadline_from(allowed), k_arr,
                                  block_frac=0.3, max_block=12)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Focus SA again
    if tleft() > 120:
        allowed = min(100, tleft() * 0.12)
        n_sa3 = int(min(3000000, allowed * 35000))
        h_sa3, c5_sa3 = sa_lag_focused(best_h, n_sa3, 0.012, 0.00002, rng,
                                        start_time, deadline_from(allowed), k_arr, n_top=6)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 6: Multi-start with diverse perturbations
    for restart in range(12):
        if tleft() < 50:
            break
        r = rng.random()
        if r < 0.3:
            h_pert = perturb_random(best_h, rng, rng.randint(3, 7))
        elif r < 0.6:
            h_pert = perturb_block(best_h, rng, 10)
        elif r < 0.8:
            h_pert = perturb_random(best_h, rng, 2)
            h_pert = perturb_block(h_pert, rng, 5)
        else:
            h_pert = perturb_random(best_h, rng, rng.randint(5, 12))

        allowed = min(35, tleft() * 0.04)
        n_sa_r = int(min(800000, allowed * 30000))
        h_sr, c5_sr = sa_lag_focused(h_pert, n_sa_r, 0.01, 0.00002, rng,
                                      start_time, deadline_from(allowed), k_arr, n_top=4)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Two-swap SA
    if tleft() > 80:
        allowed = min(60, tleft() * 0.06)
        n_ts = int(min(300000, allowed * 5000))
        h_ts, c5_ts = two_swap_sa(best_h, n_ts, 0.012, 0.00003, rng,
                                   start_time, deadline_from(allowed), k_arr)
        if c5_ts < best_score:
            best_score = c5_ts
            best_h = h_ts.copy()

    # Phase 8: Coarse-to-fine
    if tleft() > 150:
        n_coarse = 120
        target_coarse = n_coarse // 2
        k_arr_c = np.arange(2 * n_coarse - 1) - (n_coarse - 1)
        h_coarse = interpolate_to(best_h, n_coarse, float(target_coarse))
        h_coarse_bin = project_binary(h_coarse, target_coarse)

        allowed_c = min(30, tleft() * 0.03)
        n_sac = int(min(300000, allowed_c * 25000))
        h_csa, c5_csa = sa_lag_focused(h_coarse_bin, n_sac, 0.015, 0.00005, rng,
                                        start_time, deadline_from(allowed_c), k_arr_c, n_top=5)

        h_fine = coarse_to_fine(h_csa, n_init, target_init)
        allowed_f = min(40, tleft() * 0.04)
        n_saf = int(min(500000, allowed_f * 30000))
        h_fsa, c5_fsa = sa_lag_focused(h_fine, n_saf, 0.01, 0.00002, rng,
                                        start_time, deadline_from(allowed_f), k_arr, n_top=5)
        if c5_fsa < best_score:
            best_score = c5_fsa
            best_h = h_fsa.copy()

    # Phase 9: L-BFGS-B polish
    if tleft() > 60:
        alphas3 = [50, 150, 500, 1500, 5000, 15000]
        allowed = min(50, tleft() * 0.06)
        h_ref3, sc3 = lbfgsb_phase(best_h, n_init, float(target_init),
                                    alphas3, 500, 1e-12, start_time, deadline_from(allowed))
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    # Phase 10: Different resolution attempts
    for n_try in [n_init + 8, n_init - 8, n_init + 16, n_init + 4]:
        if tleft() < 40:
            break
        if n_try < 80 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed_r = min(25, tleft() * 0.03)
        n_sa_r2 = int(min(300000, allowed_r * 30000))
        h_sr2, c5_sr2 = sa_lag_focused(h_bin_try, n_sa_r2, 0.01, 0.00003, rng,
                                        start_time, deadline_from(allowed_r), k_arr_try, n_top=4)
        if c5_sr2 < best_score:
            best_score = c5_sr2
            best_h = h_sr2.copy()

    # Phase 11: Final L-BFGS-B
    if tleft() > 40:
        alphas_f = [200, 800, 3000, 10000, 50000]
        allowed = min(30, tleft() * 0.04)
        h_rf, scf = lbfgsb_phase(best_h, n_init, float(target_init),
                                  alphas_f, 800, 1e-12, start_time, deadline_from(allowed))
        if scf < best_score:
            best_score = scf
            best_h = h_rf.copy()

    # Phase 12: Final SA burst
    if tleft() > 30:
        h_bin_f = project_binary(best_h, target_init)
        allowed_f = max(5.0, tleft() - 10)
        n_sa_f = int(min(2000000, allowed_f * 35000))
        h_sf, c5_sf = sa_lag_focused(h_bin_f, n_sa_f, 0.008, 0.00001, rng,
                                      start_time, start_time + (time.time() - start_time) + allowed_f,
                                      k_arr, n_top=8)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
