        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def neighbor_sums_fast(h, k):
    n = len(h)
    NS = np.zeros(n)
    pos = np.arange(n)
    idx = pos + k
    v = (idx >= 0) & (idx < n)
    NS[v] += h[idx[v]]
    idx = pos - k
    v = (idx >= 0) & (idx < n)
    NS[v] += h[idx[v]]
    return NS

def swap_delta_full(h, i, j):
    n = len(h)
    k_range = np.arange(2 * n - 1) - (n - 1)
    contrib_i = np.zeros(2 * n - 1)
    idx = i - k_range
    v = (idx >= 0) & (idx < n)
    contrib_i[v] += h[idx[v]]
    idx = i + k_range
    v2 = (idx >= 0) & (idx < n)
    contrib_i[v2] += h[idx[v2]]
    contrib_j = np.zeros(2 * n - 1)
    idx = j - k_range
    v = (idx >= 0) & (idx < n)
    contrib_j[v] += h[idx[v]]
    idx = j + k_range
    v2 = (idx >= 0) & (idx < n)
    contrib_j[v2] += h[idx[v2]]
    return contrib_i - contrib_j

def greedy_binary(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = full_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    shifts = np.arange(2 * n - 1) - (n - 1)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 400:
            break

        max_val = float(corr.max())
        top5_idx = np.argsort(corr)[-5:]
        k_star = shifts[top5_idx[-1]]

        if k_star == 0:
            stall += 1
            continue

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        NS = neighbor_sums_fast(h, k_star)

        top_i = ones[np.argsort(-NS[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(NS[zeros])[:min(30, len(zeros))]]

        found = False
        for i in top_i:
            if NS[i] <= NS[top_j[0]]:
                break
            for j in top_j:
                if NS[i] <= NS[j]:
                    break
                delta = swap_delta_full(h, i, j)
                new_corr = corr + delta
                if float(new_corr.max()) < max_val - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    stall = 0
                    c5_new = float(corr.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if NS[i] <= NS[j]:
                    continue
                delta = swap_delta_full(h, i, j)
                new_corr = corr + delta
                if float(new_corr.max()) < max_val - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    stall = 0
                    c5_new = float(corr.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def sa_binary(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = full_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    shifts = np.arange(2 * n - 1) - (n - 1)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 80000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        mode = rng.random()
        if mode < 0.6:
            k_star = shifts[np.argmax(corr)]
            if k_star != 0:
                NS = neighbor_sums_fast(h, k_star)
                ci = ones[NS[ones] > 0.5]
                if len(ci) == 0:
                    ci = ones
                i = ci[rng.randint(len(ci))]
                cz = zeros[NS[zeros] < 0.5]
                if len(cz) == 0:
                    cz = zeros
                j = cz[rng.randint(len(cz))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = swap_delta_full(h, i, j)
        new_corr = corr + delta
        new_max = float(new_corr.max())
        dc5 = (new_max - float(corr.max())) * dx

        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            c5_new = new_max * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def continuous_pairwise(h, n_iters, rng, time_limit, start_time, eps_init=0.05, eps_min=0.002):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    corr = full_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 20000:
            break

        ratio = it / max(n_iters - 1, 1)
        eps = eps_init * (eps_min / eps_init) ** ratio

        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-5:]
        max_corr_val = corr[top_idx[-1]]

        weights = np.exp(-(max_corr_val - corr[top_idx]) * 30.0)
        if weights.sum() < 1e-10:
            break
        weights = weights / weights.sum()

        wNS = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            k = shifts[ki]
            if k == 0:
                continue
            NS = neighbor_sums_fast(h, k)
            wNS += wi * NS

        cand_i = np.where(h > eps * 0.5)[0]
        cand_j = np.where(h < 1.0 - eps * 0.5)[0]
        if len(cand_i) == 0 or len(cand_j) == 0:
            break

        top_i = cand_i[np.argsort(-wNS[cand_i])[:25]]
        top_j = cand_j[np.argsort(wNS[cand_j])[:25]]

        found = False
        for i in top_i:
            if wNS[i] <= wNS[top_j[0]]:
                break
            for j in top_j:
                if i == j:
                    continue
                if wNS[i] <= wNS[j]:
                    break
                delta = swap_delta_full(h, i, j) * eps
                new_corr = corr + delta
                new_max = float(new_corr.max())
                if new_max < max_val - 1e-13:
                    h[i] -= eps
                    h[j] += eps
                    h = project_box_sum(h, target)
                    corr = full_corr(h)
                    stall = 0
                    c5_new = float(corr.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True
                    break
                else:
                    eps_trial = eps * 0.3
                    if eps_trial > eps_min:
                        delta_s = swap_delta_full(h, i, j) * eps_trial
                        new_corr_s = corr + delta_s
                        if float(new_corr_s.max()) < max_val - 1e-13:
                            h[i] -= eps_trial
                            h[j] += eps_trial
                            h = project_box_sum(h, target)
                            corr = full_corr(h)
                            stall = 0
                            c5_new = float(corr.max()) * dx
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h.copy()
                            found = True
                            break
        if found:
            continue

        if rng.random() < 0.3:
            i = cand_i[rng.randint(len(cand_i))]
            j = cand_j[rng.randint(len(cand_j))]
            if i != j:
                delta = swap_delta_full(h, i, j) * eps
                new_corr = corr + delta
                if float(new_corr.max()) < max_val - 1e-13:
                    h[i] -= eps
                    h[j] += eps
                    h = project_box_sum(h, target)
                    corr = full_corr(h)
                    stall = 0
                    c5_new = float(corr.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True

        if not found:
            stall += 1

    return best_h, best_c5

def proj_grad_topk(h, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = full_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 8000:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                s = -s
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] += (-h[s:])
        grad /= topk

        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = full_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def proj_grad_softmax(h, n_iters, alpha, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = full_corr(h)
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 6000:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        max_corr = float(corr.max())
        shifted_w = alpha * (corr - max_corr)
        shifted_w = np.clip(shifted_w, -400.0, 0.0)
        exp_w = np.exp(shifted_w)
        W = exp_w / exp_w.sum()

        grad = np.full(n, 1.0)
        for k in range(2 * n - 1):
            wk = W[k]
            if wk < 1e-12:
                continue
            s = shifts[k]
            if s >= 0:
                grad[:n - s] -= wk * h[s:]
                grad[s:] -= wk * h[:n - s]
            else:
                s = -s
                grad[s:] -= wk * h[:n - s]
                grad[:n - s] -= wk * h[s:]

        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = full_corr(h)
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def slsqp_smooth(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def smooth_obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = float(np.max(corr))
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

            def cons_fun(x):
                return np.sum(x) - target

            def cons_jac(x):
                return np.ones(n)

            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def spectral_smoothing(h, n_keep, target):
    n = len(h)
    spectrum = np.fft.fft(h)
    if n_keep < n // 2:
        mask = np.zeros(n, dtype=bool)
        mask[:n_keep] = True
        mask[-n_keep:] = True
        mask[n // 2] = True
        spectrum[~mask] = 0
    h_smooth = np.fft.ifft(spectrum).real
    return project_box_sum(h_smooth, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_cont_h = project_box_sum(h_init, target_init)
    best_cont_score = compute_c5(best_cont_h)
    best_bin_h = project_binary(h_init, int(n_init / 2))
    best_bin_score = compute_c5(best_bin_h)

    # Phase 1: Binary greedy
    allowed = min(40.0, tleft() * 0.08)
    if allowed > 10:
        h_g, c5_g = greedy_binary(best_bin_h, max_iters=100000, rng=rng,
                                   time_limit=allowed, start_time=start_time)
        if c5_g < best_bin_score:
            best_bin_score = c5_g
            best_bin_h = h_g.copy()

    # Phase 2: Binary SA
    allowed = min(120.0, tleft() * 0.18)
    if allowed > 20:
        n_sa = int(min(2000000, allowed * 15000))
        h_sa, c5_sa = sa_binary(best_bin_h, n_sa, 0.015, 0.00003, rng,
                                 time_limit=allowed, start_time=start_time)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_bin_h = h_sa.copy()

    # Phase 3: Continuous gradient from binary
    allowed = min(100.0, tleft() * 0.15)
    if allowed > 20:
        n_iters = int(min(200000, allowed * 2000))
        h_pg, c5_pg = proj_grad_topk(best_bin_h, n_iters, topk=5,
                                      lr_init=0.04, lr_end=0.001,
                                      time_limit=allowed, start_time=start_time)
        if c5_pg < best_cont_score:
            best_cont_score = c5_pg
            best_cont_h = h_pg.copy()

    # Phase 4: Continuous pairwise moves
    allowed = min(200.0, tleft() * 0.25)
    if allowed > 30:
        n_iters = int(min(50000, allowed * 500))
        h_cp, c5_cp = continuous_pairwise(best_cont_h, n_iters, rng,
                                           time_limit=allowed, start_time=start_time,
                                           eps_init=0.04, eps_min=0.003)
        if c5_cp < best_cont_score:
            best_cont_score = c5_cp
            best_cont_h = h_cp.copy()

    # Phase 5: Softmax gradient
    allowed = min(100.0, tleft() * 0.12)
    if allowed > 20:
        n_iters = int(min(100000, allowed * 1000))
        h_pgs, c5_pgs = proj_grad_softmax(best_cont_h, n_iters, alpha=50.0,
                                           lr_init=0.03, lr_end=0.0005,
                                           time_limit=allowed, start_time=start_time)
        if c5_pgs < best_cont_score:
            best_cont_score = c5_pgs
            best_cont_h = h_pgs.copy()

    # Phase 6: More pairwise (finer)
    allowed = min(150.0, tleft() * 0.18)
    if allowed > 40:
        n_iters = int(min(80000, allowed * 800))
        h_cp2, c5_cp2 = continuous_pairwise(best_cont_h, n_iters, rng,
                                             time_limit=allowed, start_time=start_time,
                                             eps_init=0.02, eps_min=0.001)
        if c5_cp2 < best_cont_score:
            best_cont_score = c5_cp2
            best_cont_h = h_cp2.copy()

    # Phase 7: SLSQP polish
    allowed = min(tleft() - 20, 100.0)
    if allowed > 30:
        h_ref, sc_ref = slsqp_smooth(best_cont_h, target_init,
                                      alphas=[50, 200, 1000, 3000],
                                      maxiter=3000, ftol=1e-14,
                                      time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 8: Iterative refinement
    for iteration in range(4):
        if tleft() < 100:
            break

        # Perturb and SA
        allowed = min(30.0, tleft() * 0.08)
        if allowed > 15:
            n_pert = rng.randint(3, 8)
            h_pert = best_bin_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) < 2 or len(zeros) < 2:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            n_sa2 = int(min(500000, allowed * 10000))
            h_sa2, c5_sa2 = sa_binary(h_pert, n_sa2, 0.01, 0.00002, rng,
                                       time_limit=allowed, start_time=start_time)
            if c5_sa2 < best_bin_score:
                best_bin_score = c5_sa2
                best_bin_h = h_sa2.copy()

        if tleft() < 80:
            break

        # Gradient + pairwise from best binary
        allowed = min(50.0, tleft() * 0.10)
        if allowed > 20:
            n_iters = int(min(100000, allowed * 2000))
            h_pg2, c5_pg2 = proj_grad_topk(best_bin_h, n_iters, topk=3,
                                            lr_init=0.03, lr_end=0.0003,
                                            time_limit=allowed * 0.5, start_time=start_time)
            if c5_pg2 < best_cont_score:
                best_cont_score = c5_pg2
                best_cont_h = h_pg2.copy()

        if tleft() < 60:
            break

        allowed = min(40.0, tleft() * 0.08)
        if allowed > 15:
            n_iters2 = int(min(30000, allowed * 600))
            h_cp3, c5_cp3 = continuous_pairwise(best_cont_h, n_iters2, rng,
                                                 time_limit=allowed, start_time=start_time,
                                                 eps_init=0.015, eps_min=0.001)
            if c5_cp3 < best_cont_score:
                best_cont_score = c5_cp3
                best_cont_h = h_cp3.copy()

    # Phase 9: Try different resolutions
    for n_try in [n_init + (n_init // 150), n_init - (n_init // 150)]:
        if tleft() < 80:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_cont_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, int(target_try))

        allowed = min(15.0, tleft() * 0.06)
        if allowed < 8:
            continue
        h_g3, c5_g3 = greedy_binary(h_bin_try, max_iters=30000, rng=rng,
                                     time_limit=allowed, start_time=start_time)

        allowed2 = min(30.0, tleft() * 0.08)
        if allowed2 < 10:
            continue
        n_iters3 = int(min(50000, allowed2 * 1500))
        h_pg3, c5_pg3 = proj_grad_topk(h_g3, n_iters3, topk=4,
                                        lr_init=0.03, lr_end=0.0005,
                                        time_limit=allowed2, start_time=start_time)

        allowed3 = min(25.0, tleft() * 0.06)
        if allowed3 < 10 and c5_pg3 < best_cont_score:
            c5_pg3 = None
        if c5_pg3 is not None and allowed3 >= 10:
            n_iters4 = int(min(20000, allowed3 * 500))
            h_cp4, c5_cp4 = continuous_pairwise(h_pg3, n_iters4, rng,
                                                 time_limit=allowed3, start_time=start_time,
                                                 eps_init=0.02, eps_min=0.001)
            if c5_cp4 < best_cont_score:
                best_cont_score = c5_cp4
                best_cont_h = h_cp4.copy()
        elif c5_pg3 is not None and c5_pg3 < best_cont_score:
            best_cont_score = c5_pg3
            best_cont_h = h_pg3.copy()

    # Phase 10: Spectral smoothing attempt
    if tleft() > 40:
        n_s = len(best_cont_h)
        for frac in [0.3, 0.5, 0.7]:
            n_keep = int(n_s * frac)
            h_smooth = spectral_smoothing(best_cont_h, n_keep, float(n_s) / 2.0)
            sc_smooth = compute_c5(h_smooth)
            if sc_smooth < best_cont_score:
                best_cont_score = sc_smooth
                best_cont_h = h_smooth.copy()

    # Phase 11: Final SLSQP burst
    if tleft() > 30:
        n_use = len(best_cont_h)
        h_ref3, sc_ref3 = slsqp_smooth(best_cont_h, float(n_use) / 2.0,
                                        alphas=[200, 1000, 5000],
                                        maxiter=5000, ftol=1e-15,
                                        time_limit=max(10.0, tleft() - 10),
                                        start_time=start_time)
        if sc_ref3 < best_cont_score:
            best_cont_score = sc_ref3
            best_cont_h = h_ref3.copy()

    # Phase 12: Final gradient
    if tleft() > 15:
        n_use = len(best_cont_h)
        n_iters_f = int(min(50000, max(10000, (tleft() - 5) * 3000)))
        h_pgf, c5_pgf = proj_grad_topk(best_cont_h, n_iters_f, topk=3,
                                        lr_init=0.01, lr_end=0.0001,
                                        time_limit=max(5.0, tleft() - 5),
                                        start_time=start_time)
        if c5_pgf < best_cont_score:
            best_cont_score = c5_pgf
            best_cont_h = h_pgf.copy()

    # Final selection
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    n_bin = len(best_bin_h)
    if n_bin == n_final:
        h_bin_final = project_binary(best_bin_h, int(n_bin) / 2)
        bin_score = compute_c5(h_bin_final)
    else:
        h_bin_interp = interpolate_to(best_bin_h, n_final, target_final)
        h_bin_final = project_binary(h_bin_interp, int(n_final) // 2)
        bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
