```python
def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Numerical stability
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    p = np.exp(shifted)
    p = p / p.sum()
    
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    grad = np.zeros(n)
    for k in range(n_corr):
        if p[k] < 1e-15:
            continue
        s = k - mid
        # ∂corr[k]/∂h[i] = (1-h[i+s]) for i+s in [0,n) minus h[i-s] for i-s in [0,n)
        # For the first term: i ranges over [0, n-1-s] if s>=0, or [0, n-1] if s<0
        #   Actually: i+s valid means 0 <= i+s < n, so i in [-s, n-1-s] intersected with [0, n-1]
        # For the second term: i-s valid means 0 <= i-s < n, so i in [s, s+n-1] intersected with [0, n-1]
        
        v1 = (idxs + s >= 0) & (idxs + s < n)
        grad[v1] += p[k] * (1.0 - h[idxs[v1] + s])
        
        v2 = (idxs - s >= 0) & (idxs - s < n)
        grad[v2] -= p[k] * h[idxs[v2] - s]
    
    grad *= dx
```

I need to vectorize the gradient computation!

The gradient is:
∇f[i] = dx * Σ_k p_k * [(1-h[i+s_k]) * 1_{valid} - h[i-s_k] * 1_{valid}]

Let me define:
A[i] = Σ_k p_k * h[i+s_k] * 1_{i+s_k valid}
B[i] = Σ_k p_k * h[i-s_k] * 1_{i-s_k valid}
C = Σ_k p_k = 1

Then: ∇f[i] = dx * (C - A[i] - B[i]) = dx * (1 - A[i] - B[i])

Now, A[i] = Σ_k p_k * h[i+s_k] where s_k = k-(n-1) ranges from -(n-1) to (n-1).

Let me reindex: let t = s_k, so t ranges from -(n-1) to (n-1), and p_t = p[t+(n-1)].

A[i] = Σ_t p_t * h[i+t] * 1_{0≤i+t<n}

This is a cross-correlation of h with p (reversed)! Specifically:
A[i] = (h ⊛ p)[i] where p is the "kernel" with p[t] = p_abs[t] for t in [-n+1, n-1].

We can compute this using np.correlate or a manual sum. For n=800, the direct sum is O(n * 2n) = O(n^2) = 640K, which is fast in numpy.

Actually, let me just use a simple approach: create a matrix M where M[i, k] = h[i+s_k] (with zero padding) and compute A = M @ p. This is a matrix-vector product of size n × n_corr times n_corr × 1, which is O(n * n_corr) = O(n^2).

```python
# Create shift matrix
idxs = np.arange(n)
shifts = np.arange(-(n-1), n)  # s_k values
M_plus = np.zeros((n, n_corr))
for k in range(n_corr):
    s = shifts[k]
    valid = (idxs + s >= 0) & (idxs + s < n)
    M_plus[valid, k] = h[idxs[valid] + s]
A = M_plus @ p  # (n,)
```

But creating M_plus is O(n * n_corr) = O(n^2) which is fine, but the Python loop over k is slow (n_corr = 1599 iterations).

Better: vectorize the whole thing.

```python
idxs_col = idxs[:, np.newaxis]  # (n, 1)
shifts_row = shifts[np.newaxis, :]  # (1, n_corr)
valid_plus = (idxs_col + shifts_row >= 0) & (idxs_col + shifts_row < n)  # (n, n_corr)
# M_plus[i, k] = h[i + s_k] if valid, else 0
# We can't directly index h with (i+s_k) for all i,k at once without advanced indexing
M_plus = np.where(valid_plus, h[idxs_col + shifts_row], 0.0)
A = M_plus @ p
```

The advanced indexing `h[idxs_col + shifts_row]` creates an (n, n_corr) array of indices, then gathers from h. For n=800, n_corr=1599, that's 1.28M elements. Memory: ~10MB. Computation: O(n*n_corr). This should be fast.

```python
valid_minus = (idxs_col - shifts_row >= 0) & (idxs_col - shifts_row < n)
M_minus = np.where(valid_minus, h[idxs_col - shifts_row], 0.0)
B = M_minus @ p
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(corr)) / len(corr)
        p_sum = 1.0
    else:
        p = p / p_sum
    obj = (c_max + np.log(p_sum) / alpha) * dx
    
    # Vectorized gradient
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    idxs_col = idxs[:, np.newaxis]
    shifts_row = shifts[np.newaxis, :]
    
    valid_plus = (idxs_col + shifts_row >= 0) & (idxs_col + shifts_row < n)
    valid_minus = (idxs_col - shifts_row >= 0) & (idxs_col - shifts_row < n)
    
    A = np.where(valid_plus, h[idxs_col + shifts_row], 0.0) @ p
    B = np.where(valid_minus, h[idxs_col - shifts_row], 0.0) @ p
    
    grad = (1.0 - A - B) * dx
    return obj, grad

    def obj_and_grad(x):
        f, g = smooth_obj_and_grad(x, alpha)
        # Add soft penalty for sum constraint (helps L-BFGS-B respect it)
        s = np.sum(x) - target
        penalty = 500.0 * s * s
        pen_grad = 1000.0 * s * np.ones(n)
        return f + penalty, g + pen_grad

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return project_feasible(h_start, target)
```

```python
    grad = (1.0 - A - B) * dx
    return obj, grad


def lbfags_polish(h_start, n, target, alpha, maxiter=1000):
    def obj_and_grad(x):
        f, g = smooth_obj_and_grad(x, alpha)
        s = np.sum(x) - target
        penalty = 2000.0 * s * s
        pen_grad = 4000.0 * s * np.ones(n)
        return f + penalty, g + pen_grad
    try:
        res = minimize(obj_and_grad, h_start, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start, target)


def vectorized_pair_cd(h_start, n, target, time_limit, max_sweeps=5000,
                       n_cand=30, patience=150, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star = int(np.argmax(corr))
        s_star = k_star - mid
        grad = np.zeros(n)
        v1 = (idxs + s_star >= 0) & (idxs + s_star < n)
        grad[v1] += (1.0 - h[idxs[v1] + s_star])
        v2 = (idxs - s_star >= 0) & (idxs - s_star < n)
        grad[v2] -= h[idxs[v2] - s_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        H_cnt = len(high_set)
        L_cnt = len(low_set)

        hp_arr = np.zeros((H_cnt, n_corr))
        hm_arr = np.zeros((H_cnt, n_corr))
        for ii in range(H_cnt):
            i = int(high_set[ii])
            vp = (i + shifts >= 0) & (i + shifts < n)
            hp_arr[ii, vp] = h[i + shifts[vp]]
            vm = (i - shifts >= 0) & (i - shifts < n)
            hm_arr[ii, vm] = h[i - shifts[vm]]

        jp_arr = np.zeros((L_cnt, n_corr))
        jm_arr = np.zeros((L_cnt, n_corr))
        for jj in range(L_cnt):
            j = int(low_set[jj])
            vp = (j + shifts >= 0) & (j + shifts < n)
            jp_arr[jj, vp] = h[j + shifts[vp]]
            vm = (j - shifts >= 0) & (j - shifts < n)
            jm_arr[jj, vm] = h[j - shifts[vm]]

        G = hp_arr[:, np.newaxis, :] - jp_arr[np.newaxis, :, :] \
            + hm_arr[:, np.newaxis, :] - jm_arr[np.newaxis, :, :]

        h_hi = h[high_set]
        h_lo = h[low_set]
        delta_hi = np.minimum(h_hi[:, np.newaxis], 1.0 - h_lo[np.newaxis, :])
        delta_lo = np.zeros_like(delta_hi)

        for _ in range(n_bisect):
            mid_d = 0.5 * (delta_lo + delta_hi)
            vals = corr[np.newaxis, np.newaxis, :] + mid_d[:, :, np.newaxis] * G
            f_max = np.max(vals, axis=2)
            active = vals >= f_max[:, :, np.newaxis] - 1e-12
            g_prime = np.where(active, G, -1e30).max(axis=2)
            delta_hi = np.where(g_prime > 1e-10, mid_d, delta_hi)
            delta_lo = np.where(g_prime <= 1e-10, mid_d, delta_lo)

        delta_star = 0.5 * (delta_lo + delta_hi)
        final_vals = corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * G
        final_max = np.max(final_vals, axis=2) * dx
        final_max[delta_hi < 1e-10] = np.inf

        best_flat = int(np.argmin(final_max))
        bi = best_flat // L_cnt
        bj = best_flat % L_cnt
        best_delta = float(delta_star[bi, bj])

        if final_max[bi, bj] < c5_val - 1e-15 and abs(best_delta) > 1e-12:
            i = int(high_set[bi])
            j = int(low_set[bj])
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            no_improve += 10
            if no_improve > patience:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    try:
        cs = CubicSpline(x_old, h_src)
        h_out = cs(x_new)
    except Exception:
        h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Polish at original resolution
    if tleft() > 80:
        for alpha in [3000, 15000, 60000]:
            if tleft() < 25:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 2: Upscale to 800 and optimize
    n2 = 800
    target2 = n2 / 2.0
    if tleft() > 200:
        h_up = interpolate_to(best_h, n2, target2)
        for alpha in [3000, 10000, 40000, 150000]:
            if tleft() < 50:
                break
            h_try = lbfags_polish(h_up, n2, target2, alpha, maxiter=700)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
                h_up = h_try
        if tleft() > 150:
            cd_time = min(200, tleft() * 0.35)
            h_cd, c5_cd = vectorized_pair_cd(h_up, n2, target2,
                                             time_limit=cd_time,
                                             max_sweeps=4000, n_cand=35,
                                             patience=120, n_bisect=14)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 3: L-BFGS-B after CD at 800
    if tleft() > 80:
        for alpha in [80000, 300000, 800000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n2, target2, alpha, maxiter=500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Second pair CD round at 800
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n2, target2,
                                         time_limit=cd_time,
                                         max_sweeps=3000, n_cand=40,
                                         patience=100, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Try n=1200
    n3 = 1200
    target3 = n3 / 2.0
    if tleft() > 280:
        h_up3 = interpolate_to(best_h, n3, target3)
        for alpha in [5000, 25000, 80000]:
            if tleft() < 70:
                break
            h_try = lbfags_polish(h_up3, n3, target3, alpha, maxiter=500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
                h_up3 = h_try
        if tleft() > 130:
            cd_time = min(160, tleft() * 0.35)
            h_cd, c5_cd = vectorized_pair_cd(h_up3, n3, target3,
                                             time_limit=cd_time,
                                             max_sweeps=2000, n_cand=30,
                                             patience=100, n_bisect=14)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 6: Try n=1600 if time allows
    n4 = 1600
    target4 = n4 / 2.0
    if tleft() > 350:
        h_up4 = interpolate_to(best_h, n4, target4)
        for alpha in [5000, 30000]:
            if tleft() < 80:
                break
            h_try = lbfags_polish(h_up4, n4, target4, alpha, maxiter=400)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
                h_up4 = h_try
        if tleft() > 150:
            cd_time = min(120, tleft() * 0.3)
            h_cd, c5_cd = vectorized_pair_cd(h_up4, n4, target4,
                                             time_limit=cd_time,
                                             max_sweeps=1500, n_cand=25,
                                             patience=80, n_bisect=12)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 7: Final polish at best resolution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if tleft() > 40:
        h_try = lbfags_polish(best_h, n_final, target_final, 500000, maxiter=300)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score

    return h_final, final_score, len(h_final)
```
