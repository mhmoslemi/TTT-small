```python
import numpy as np
from scipy.optimize import minimize
import time

_CON_TARGET = 0.0

def _con_fun(x):
    return np.sum(x) - _CON_TARGET

def _con_jac(x):
    return np.ones(len(x))

_SLSQP_ALPHA = 1000.0
_SLSQP_DX = 1.0 / 300.0

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(_SLSQP_ALPHA * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / _SLSQP_ALPHA) * _SLSQP_DX

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_phase(h_start, n, target, alphas, maxiter, ftol):
    global _CON_TARGET, _SLSQP_ALPHA, _SLSQP_DX
    _CON_TARGET = target
    _SLSQP_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    h_cur = h_start.copy()
    for alpha in alphas:
        _SLSQP_ALPHA = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def compute_full_g(h, i, j, n):
    n_corr = 2 * n - 1
    ks = np.arange(-(n - 1), n)
    g = np.zeros(n_corr)
    idx_ik = i - ks
    idx_iak = i + ks
    idx_jk = j - ks
    idx_jak = j + ks
    v1 = (idx_ik >= 0) & (idx_ik < n)
    v2 = (idx_iak >= 0) & (idx_iak < n)
    v3 = (idx_jk >= 0) & (idx_jk < n)
    v4 = (idx_jak >= 0) & (idx_jak < n)
    if np.any(v1):
        g[v1] += h[idx_ik[v1]]
    if np.any(v2):
        g[v2] += h[idx_iak[v2]]
    if np.any(v3):
        g[v3] -= h[idx_jk[v3]]
    if np.any(v4):
        g[v4] -= h[idx_jak[v4]]
    return g

def find_optimal_delta(corr, g, delta_max):
    lo = 0.0
    hi = delta_max
    for _ in range(18):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * g
        cur_max = float(np.max(vals))
        active = vals >= (cur_max - 1e-14)
        if not np.any(active):
            break
        max_g_active = float(np.max(g[active]))
        min_g_active = float(np.min(g[active]))
        if max_g_active > 1e-12:
            hi = mid
        elif min_g_active < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def focused_pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=80000, m_focus=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1000:
                break

        top_idx = np.argsort(corr)[-m_focus:]
        focus_ks = np.array([int(ki) - mid for ki in top_idx])

        grad = np.zeros(n)
        for t in range(m_focus):
            k = focus_ks[t]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)

        profiles = np.zeros((n_c, m_focus))
        for ci in range(n_c):
            m = int(all_cands[ci])
            for t in range(m_focus):
                k = focus_ks[t]
                val = 0.0
                if 0 <= m - k < n:
                    val += h[m - k]
                if 0 <= m + k < n:
                    val += h[m + k]
                profiles[ci, t] = val

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        top_pairs = []

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g_f = profiles[ci] - profiles[cj]
                max_g_f = float(np.max(g_f))
                if max_g_f > 2.0:
                    continue
                top_pairs.append((float(np.max(g_f)), int(i), int(j), delta_max))

        top_pairs.sort(key=lambda x: x[0])

        for _, i_val, j_val, delta_max in top_pairs[:5]:
            g = compute_full_g(h, i_val, j_val, n)
            delta_star = find_optimal_delta(corr, g, delta_max)
            if delta_star < 1e-10:
                continue
            new_max = float(np.max(corr + delta_star * g))
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_i = i_val
                best_j = j_val
                best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def perturb_restart(h, n, target, rng, scale=0.05):
    noise = rng.normal(0, scale, size=n)
    h_new = h + noise
    h_new = project_feasible(h_new, target)
    return h_new

def multi_restart_pair_cd(h_init, n, target, time_limit, n_restarts=3, n_cand=30, m_focus=5):
    start = time.time()
    rng = np.random.default_rng(123)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    time_per_restart = time_limit / (n_restarts + 1)

    for r in range(n_restarts + 1):
        if time.time() - start > time_limit - 10:
            break
        remaining = time_limit - (time.time() - start)
        t_alloc = min(time_per_restart, remaining)
        if t_alloc < 30:
            break

        hcd, c5cd = focused_pair_cd(h, n, target, time_limit=t_alloc,
                                     n_cand=n_cand, m_focus=m_focus)
        if c5cd < best_c5:
            best_c5 = c5cd
            best_h = hcd.copy()
        h = best_h.copy()

        if r < n_restarts:
            scale = 0.03 + 0.02 * r
            h = perturb_restart(h, n, target, rng, scale=scale)

    return best_h, best_c5
```
