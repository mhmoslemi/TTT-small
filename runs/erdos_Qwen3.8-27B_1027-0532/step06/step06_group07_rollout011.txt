        if residual > 0:

    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    tabu = {}
    sweep_num = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        sweep_num += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 100:
            break

        # Use multi-active gradient
        eps_active = 1e-4 + 0.001 * (1.0 - min(sweep_num / max_sweeps, 1.0))
        active_mask = corr >= np.max(corr) - eps_active / dx
        k_star = int(np.argmax(corr)) - mid

        grad = np.zeros(n)
        active_shifts = np.where(active_mask)[0]
        for s_idx in active_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active_shifts), 1)
        grad -= np.mean(grad)

        # Adaptive candidate size
        cur_n_cand = int(n_cand * (1.0 - 0.3 * min(sweep_num / 2000.0, 1.0)))
        cur_n_cand = max(15, cur_n_cand)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_n_cand]
        high_set = sorted_idx[-cur_n_cand:]

        # Tabu check
        if swep_num > 20 and len(tabu) > tabu_size:
            # Remove oldest entries
            while len(tabu) > tabu_size:
                oldest_key = min(tabu, key=tabu.get)
                del tabu[oldest_key]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            if hi < 1e-10:
                continue
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                pair_key = (min(int(i), int(j)), max(int(i), int(j)))
                if pair_key in tabu and sweep_num - tabu[pair_key] < 20:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active2 = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active2]) if active2.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pair_key = (min(int(best_i), int(best_j)), max(int(best_i), int(best_j)))
            tabu[pair_key] = sweep_num
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5


def subspace_optimize(h, n, target, time_limit, max_rounds=500, subspace_sizes=(3, 4, 5)):
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if no_improve > 30:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx_n(n)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient to select subspace
        grad, _, _ = compute_active_gradient(h, n, epsilon=1e-4)
        abs_grad = np.abs(grad)
        
        # Select top coordinates by |gradient|, but ensure mix of high and low
        top = np.argsort(-abs_grad)
        
        # Try different subspace sizes
        found_improvement = False
        for m in subspace_sizes:
            if time.time() - start > time_limit:
                break
            S = top[:m]
            # Ensure we have both "source" (high h, positive grad) and "sink" (low h, negative grad)
            S_high = S[h[S] > 0.3]
            S_low = S[h[S] < 0.7]
            if len(S_high) == 0 or len(S_low) == 0:
                continue
            
            frac_sum = target - np.sum(h[np.setdiff1d(idxs, S)])
            lo_frac = max(0.0, frac_sum - (len(S) - len(S_high)) * 1.0)
            hi_frac = min(float(len(S_high)), frac_sum)
            if lo_frac >= hi_frac:
                continue

            x0 = h[S].copy()
            bounds_s = [(0.0, 1.0)] * m

            def subspace_obj(x, S_local=S):
                h_full = h.copy()
                h_full[S_local] = x
                c = np.correlate(h_full, 1.0 - h_full, mode='full')
                return float(np.max(c))

            def subspace_constraint(x):
                return np.sum(x) - frac_sum

            def subspace_jac(x):
                return np.ones(len(S))

            try:
                res = minimize(subspace_obj, x0, method='SLSQP',
                               bounds=bounds_s,
                               constraints=[{'type': 'eq', 'fun': subspace_constraint, 'jac': subspace_jac}],
                               options={'maxiter': 50, 'ftol': 1e-14})
                if res.fun < c5_val / dx_n(n) - 1e-10:
                    h_new = h.copy()
                    h_new[S] = project_feasible_1d(res.x, frac_sum)
                    h_new = project_feasible(h_new, target)
                    sc = compute_c5(h_new)
                    if sc < best_c5 - 1e-12:
                        best_c5 = sc
                        best_h = h_new.copy()
                        h = h_new
                        found_improvement = True
                        break
            except Exception:
                pass

        if not found_improvement and no_improve > 15:
            # Try random subspace
            rng_s = np.random.RandomState(rnd + 12345)
            m = int(rng_s.choice([3, 4, 5]))
            S = rng_s.choice(n, size=m, replace=False)
            S_high = S[h[S] > 0.2]
            S_low = S[h[S] < 0.8]
            if len(S_high) == 0 or len(S_low) == 0:
                continue
            frac_sum = target - np.sum(h[np.setdiff1d(idxs, S)])
            lo_frac = max(0.0, frac_sum - (len(S) - len(S_high)) * 1.0)
            hi_frac = min(float(len(S_high)), frac_sum)
            if lo_frac >= hi_frac:
                continue
            x0 = h[S].copy()
            bounds_s = [(0.0, 1.0)] * m
            try:
                def subspace_obj2(x, S_local=S):
                    h_full = h.copy()
                    h_full[S_local] = x
                    c = np.correlate(h_full, 1.0 - h_full, mode='full')
                    return float(np.max(c))

                res = minimize(subspace_obj2, x0, method='SLSQP',
                               bounds=bounds_s,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - frac_sum, 'jac': lambda x: np.ones(len(S))}],
                               options={'maxiter': 80, 'ftol': 1e-14})
                if res.fun < c5_val / dx_n(n) - 1e-10:
                    h_new = h.copy()
                    h_new[S] = project_feasible_1d(res.x, frac_sum)
                    h_new = project_feasible(h_new, target)
                    sc = compute_c5(h_new)
                    if sc < best_c5 - 1e-12:
                        best_c5 = sc
                        best_h = h_new.copy()
                        h = h_new
                        found_improvement = True
            except Exception:
                pass
        if not found_improvement:
            no_improve += 1

    return best_h, best_c5


def dx_n(n):
    return 2.0 / n


def project_feasible_1d(x, target_sum):
    x = np.clip(x.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    m = len(x)
    if diff > 0:
        idx = np.where(x < 1.0 - 1e-8)[0]
    else:
        idx = np.where(x > 1e-8)[0]
    if len(idx) == 0:
        return x
    x = x.copy()
    x[idx] += diff / len(idx)
    return np.clip(x, 0.0, 1.0)


def momentum_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=13, momentum=0.3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    momentum_vec = np.zeros(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 80:
            break

        # Compute gradient from top active shifts
        eps_act = 2e-4
        active = corr >= np.max(corr) - eps_act
        grad = np.zeros(n)
        active_shifts = np.where(active)[0]
        for s_idx in active_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active_shifts), 1)
        grad -= np.mean(grad)

        # Momentum update
        momentum_vec = momentum * momentum_vec + (1 - momentum) * grad
        combined = grad + 0.5 * momentum_vec

        sorted_idx = np.argsort(combined)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            if hi < 1e-10:
                continue
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active2 = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active2]) if active2.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 30:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, seed_val):
    rng = np.random.RandomState(seed_val)
    perturbations = []
    # Sine perturbation
    freq = rng.choice([1, 2, 3, 4, 5])
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.08)
    t = np.linspace(0, 2 * np.pi * freq, n)
    perturbations.append(h + amp * np.sin(t + phase))
    # Gaussian perturbation
    perturbations.append(h + rng.randn(n) * 0.05)
    # Block perturbation
    h_block = h.copy()
    n_blocks = int(rng.choice([3, 4, 5, 6]))
    block_size = n // n_blocks
    for b in range(n_blocks):
        s = b * block_size
        e = s + block_size if b < n_blocks - 1 else n
        h_block[s:e] += rng.uniform(-0.05, 0.05)
    perturbations.append(h_block)
    results = []
    for p in perturbations:
        results.append(project_feasible(p, target))
    return results


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-up
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 8000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Multi-active pair CD with tabu (main workhorse)
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd_multi(best_h, n, target, time_limit=cd_time,
                                         max_sweeps=10000, n_cand=50, n_bisect=14, tabu_size=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Momentum pair CD
    if tleft() > 150:
        cd_time = min(150, tleft() * 0.2)
        h_mom, c5_mom = momentum_pair_cd(best_h, n, target, time_limit=cd_time,
                                         max_sweeps=8000, n_cand=45, n_bisect=14, momentum=0.4)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 4: Subspace optimization
    if tleft() > 120:
        sub_time = min(120, tleft() * 0.15)
        h_sub, c5_sub = subspace_optimize(best_h, n, target, time_limit=sub_time,
                                          max_rounds=300, subspace_sizes=(3, 4, 5))
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 5: SLSQP polish with high alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [10000, 50000, 150000], 1200, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Another round of pair CD (tighter)
    if tleft() > 120:
        cd_time = min(120, tleft() * 0.2)
        h_cd, c5_cd = fast_pair_cd_multi(best_h, n, target, time_limit=cd_time,
                                         max_sweeps=8000, n_cand=35, n_bisect=16, tabu_size=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Subspace again after more refinement
    if tleft() > 100:
        sub_time = min(100, tleft() * 0.12)
        h_sub, c5_sub = subspace_optimize(best_h, n, target, time_limit=sub_time,
                                          max_rounds=250, subspace_sizes=(3, 4, 5, 6))
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 8: Structured multi-start
    n_starts = 4
    for si in range(n_starts):
        if tleft() < 80:
            break
        candidates = structured_perturb(best_h, n, target, seed + si * 100)
        for cand in candidates:
            if tleft() < 60:
                break
            cd_t = min(50, tleft() * 0.1)
            h_try, c5_try = fast_pair_cd_multi(cand, n, target, time_limit=cd_t,
                                               max_sweeps=3000, n_cand=40, n_bisect=13, tabu_size=15)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try
            # Quick subspace
            if tleft() > 30:
                h_try2, c5_try2 = subspace_optimize(h_try, n, target, time_limit=min(20, tleft() * 0.1),
                                                    max_rounds=100, subspace_sizes=(3, 4))
                if c5_try2 < best_score:
                    best_score = c5_try2
                    best_h = h_try2

    # Phase 9: Try different resolution
    for n_try in [500, 800]:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, int(n_try), target_try)
        h_hires = slsqp_round(h_interp, int(n_try), target_try, [1000, 5000], 500, 1e-15)
        sc = compute_c5(h_hires)
        if tleft() > 60 and sc < best_score + 0.002:
            cd_t = min(50, tleft() * 0.25)
            h_hires2, sc2 = fast_pair_cd_multi(h_hires, int(n_try), int(target_try),
                                               time_limit=cd_t, max_sweeps=3000, n_cand=40, n_bisect=14, tabu_size=20)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_hires2
                n = int(n_try)
                target = int(target_try)

    # Phase 10: Final tight pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.35)
        h_cd, c5_cd = fast_pair_cd_multi(best_h, n_use, target_use,
                                         time_limit=cd_time, max_sweeps=5000, n_cand=30, n_bisect=18, tabu_size=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final subspace polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sub, c5_sub = subspace_optimize(best_h, n_use, target_use,
                                          time_limit=min(35, tleft() * 0.4),
                                          max_rounds=150, subspace_sizes=(3, 4, 5))
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 12: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use, [50000, 200000], 600, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
