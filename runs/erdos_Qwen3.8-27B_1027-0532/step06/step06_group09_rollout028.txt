    dx = 2.0 / n
    idxs = np.arange(n)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    # Build linearized constraints
    A_rows = []
    b_vals = []

        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)
        i_vals = np.arange(i_start, i_end+1)
        i_k = i_vals + k
        
        if len(i_vals) == 0:
            continue
        
        fk0 = np.sum(h0[i_vals] * (1.0 - h0[i_k]))
        grad = np.zeros(n)
        grad[i_vals] += (1.0 - h0[i_k])
        grad[i_k] += (-h0[i_vals])

        A_rows.append(A_row)
        b_vals.append(b_val)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # A @ [h; t] >= b
    # Add to constraints
    z = cp.hstack([h_var, t])
    constraints.append(A @ z >= b)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return h0.copy()

        h_new = project_feasible(h_var.value, target)
        return h_new

def interpolate_to(h_src, n_dst, target_dst):

        if c5_b2 < best_score:
            best_score = c5_b2
            best_h = h_b2

    # Phase 6: Multi-start pair CD
    if tleft() > 200:
        ms_t = min(200, tleft() * 0.2)
        h_ms, c5_ms = multi_pair_cd(best_h, n, target,
                                     time_limit=ms_t, n_starts=4, seed=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 7: Game descent again
    if tleft() > 100:
        gt2_t = min(60, tleft() * 0.1)
        h_gt2, c5_gt2 = game_descent(best_h, n, target, time_limit=gt2_t,
                                      max_steps=5000, eta_base=2.0, m_top=12)
        if c5_gt2 < best_score:
            best_score = c5_gt2
            best_h = h_gt2

    # Phase 8: L-BFGS-B fine polish
    if tleft() > 80:
        for alpha in [50000, 200000, 500000]:
            if tleft() < 25:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 9: Final pair CD
    if tleft() > 100:
        cd2_t = min(150, tleft() * 0.2)
        h_cd2, c5_cd2 = pair_cd(best_h, n, target, time_limit=cd2_t,
                                 max_sweeps=6000, n_cand_start=40,
                                 n_cand_max=120, patience=150)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 10: Higher resolution attempt
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, target_try, 1000, maxiter=800)
            sc_h = compute_c5(h_hires)
            if sc_h < best_score + 0.001:
                h_hires, c5_h = pair_cd(h_hires, n_try, target_try,
                                         time_limit=min(80, tleft() * 0.3),
                                         max_sweeps=3000, n_cand_start=30,
                                         n_cand_max=70, patience=100)
                sc_h = compute_c5(h_hires)
                if sc_h < best_score:
                    best_score = sc_h
                    best_h = h_hires
                    n = n_try
                    target = target_try
                    h_hires, c5_h2 = game_descent(h_hires, n, target,
                                                   time_limit=min(40, tleft() * 0.15),
                                                   max_steps=3000, eta_base=1.5, m_top=10)
                    if c5_h2 < best_score:
                        best_score = c5_h2
                        best_h = h_hires

    # Phase 11: Final Benders polish
    if tleft() > 60:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_fin, c5_fin = benders_descent(best_h, n_final, target_final,
                                         time_limit=min(40, tleft() * 0.5),
                                         max_iters=100, alpha_start=0.5)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Phase 12: Ultra-fine L-BFGS-B
    if tleft() > 30:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_fin = lbfags_polish(best_h, n_final, target_final, 1000000, maxiter=3000)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
