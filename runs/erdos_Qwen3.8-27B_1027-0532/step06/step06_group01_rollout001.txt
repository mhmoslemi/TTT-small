import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

_SLQ_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha, top_k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    idx = np.argsort(corr)[-top_k:]
    c_vals = corr[idx]
    c_max = np.max(c_vals)
    shifted = np.clip(alpha * (c_vals - c_max), -500.0, 0.0)
    rest = np.delete(corr, idx)
    c_rest = np.max(rest)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    if c_rest > c_max:
        obj = c_rest + (obj - c_max) * 0.1
    return float(obj)

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, top_k=None):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    if top_k is None:
        top_k = min(50, 2 * n - 1)
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, top_k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

    start = time.time()
    dx = 2.0 / n
    rng = np.random.default_rng(42)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        T = T_start * (T_end / T_start) ** (sweep / max(1.0, max_sweeps))
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800 and T < T_end * 2:
                break
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        sorted_by_grad = np.argsort(-grad)
        if rng.random() < random_frac and len(sorted_by_grad) > 2 * n_cand:
            high_set = sorted_by_grad[:n_cand // 2]
            extra_high = rng.choice(sorted_by_grad[n_cand // 2:n_cand // 2 + n_cand], size=n_cand // 2, replace=False)
            high_set = np.concatenate([high_set, extra_high])
            low_extra = rng.choice(sorted_by_grad[-n_cand // 2 - n_cand:-n_cand // 2], size=n_cand // 2, replace=False)
            low_set = np.concatenate([sorted_by_grad[-n_cand // 2:], low_extra])
        else:
            high_set = sorted_by_grad[:n_cand]
            low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_cost_diff = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                cost_diff = new_max - cur_max
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
                    best_cost_diff = cost_diff
                elif T > T_end and cost_diff < 0.05:
                    prob = np.exp(cost_diff * dx / T)
                    if rng.random() < prob:
                        best_improve = improve
                        best_i = int(i)
                        best_j = int(j)
                        best_delta = delta_star
                        best_cost_diff = cost_diff
        if best_i >= 0 and (best_improve > 1e-13 or best_cost_diff < 0):
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        elif T > T_start * 0.5 and sweep % 500 == 0:
            n_perturb = max(5, n // 50)
            pert_idx = rng.choice(n, size=n_perturb, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.normal(0, 0.05, n_perturb)
            h_pert = project_feasible(h_pert, target)
            if compute_c5(h_pert) < best_c5 * 1.05:
                h = h_pert
    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, block_size=3, max_sweeps=30000):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.default_rng(123)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:block_size]
        low_set = sorted_by_grad[-block_size:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        g_block = np.zeros(n_corr)
        for i in high_set:
            pos = int(np.where(all_cands == i)[0][0])
            g_block -= profiles[pos]
        for j in low_set:
            pos = int(np.where(all_cands == j)[0][0])
            g_block += profiles[pos]
        max_delta = min(min(h[high_set]), min(1.0 - h[low_set])) / block_size
        if max_delta < 1e-12:
            continue
        lo = 0.0
        hi_d = max_delta
        for _ in range(20):
            mid = 0.5 * (lo + hi_d)
            vals = corr + mid * g_block
            f_mid = float(np.max(vals))
            active = vals >= (f_mid - 1e-14)
            if not np.any(active):
                break
            f_prime = float(np.max(g_block[active]))
            if f_prime > 1e-14:
                hi_d = mid
            else:
                lo = mid
        delta_star = 0.5 * (lo + hi_d)
        if delta_star > 1e-10:
            new_max = float(np.max(corr + delta_star * g_block))
            if new_max < cur_max - 1e-14:
                for i in high_set:
                    h[i] -= delta_star
                for j in low_set:
                    h[j] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_start_optimize(h_init, n, target, time_limit, n_starts=3):
    start = time.time()
    rng = np.random.default_rng(99)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    time_per_start = time_limit / n_starts
    for s in range(n_starts):
        if time.time() - start > time_limit - 10:
            break
        if s == 0:
            h_try = h_init.copy()
        else:
            h_try = h_init.copy()
            n_pert = max(n // 10, 20)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_try[pert_idx] += rng.uniform(-0.1, 0.1, n_pert)
            h_try = project_feasible(h_try, target)
        t_rem = min(time_per_start, time_limit - (time.time() - start) - 5)
        if t_rem < 20:
            break
        h_cd, c5_cd = fast_pair_cd(h_try, n, target, time_limit=t_rem * 0.6, n_cand=20, max_sweeps=30000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5
