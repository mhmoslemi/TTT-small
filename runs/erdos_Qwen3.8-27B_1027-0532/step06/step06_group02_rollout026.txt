                    j = zeros[rng.randint(len(zeros))]
                    delta = compute_delta(h, i, j, k_arr)
                    new_max = float((full + delta).max())
                    c5_new = new_max * dx
                    d_c5 = c5_new - current_c5
                    if d_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-d_c5 / t)):
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        current_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
                            stall = 0
                        else:
                            stall += 1
                    else:
                        stall += 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                d_c5 = c5_new - current_c5
                if d_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-d_c5 / t)):
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
    return best_h, best_c5

def greedy_local_search(h0, start_time, time_limit, k_arr, n_top=5, candidate_size=30):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    full = full_corr_fast(h)
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    max_stall = 200

    while time.time() - start_time < time_limit:
        if stall > max_stall:
            break
        max_val = float(full.max())
        top_indices = np.argsort(full)[::-1][:n_top]
        found = False
        best_move_val = max_val
        best_move = None
        best_delta = None
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for ti in top_indices:
            k_star = int(ti) - (n - 1)
            if k_star == 0:
                continue
            A_arr = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A_arr[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A_arr[valid] += (1.0 - h[idx[valid]])
            B_arr = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B_arr[valid] += (1.0 - h[idx[valid]])
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B_arr[valid] -= h[idx[valid]]
            top_i = ones[np.argsort(-A_arr[ones])[:candidate_size]]
            top_j = zeros[np.argsort(B_arr[zeros])[:candidate_size]]
            for i in top_i:
                for j in top_j:
                    if i == j:
                        continue
                    delta = compute_delta(h, i, j, k_arr)
                    new_max = float((full + delta).max())
                    if new_max < best_move_val:
                        best_move_val = new_max
                        best_move = (i, j)
                        best_delta = delta
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            full += best_delta
            stall = 0
            if best_move_val * dx < best_c5:
                best_c5 = best_move_val * dx
                best_h = h.copy()
        else:
            stall += 1
    return best_h, best_c5

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)
    k_arr = np.arange(2 * n - 1) - (n - 1)

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(90, tleft() * 0.06)
    h_g, c5_g = greedy_local_search(best_h, start_time, start_time + allowed, k_arr, n_top=5, candidate_size=40)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Informed SA (main driver)
    allowed = min(350, tleft() * 0.30)
    n_sa = int(min(6000000, allowed * 50000))
    h_sa, c5_sa = informed_sa(best_h, n_sa, 0.03, 0.00002, rng,
                             time_limit=allowed, start_time=start_time, k_arr=k_arr,
                             top_k=5, block_frac=0.15, max_block_len=12, two_opt_frac=0.1)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Greedy polish
    allowed = min(60, tleft() * 0.05)
    h_g2, c5_g2 = greedy_local_search(best_h, start_time, start_time + allowed, k_arr, n_top=8, candidate_size=50)
    if c5_g2 < best_score:
        best_score = c5_g2
        best_h = h_g2.copy()

    # Phase 4: Continuous gradient refinement
    if tleft() > 150:
        target_f = float(n) / 2.0
        alphas = [8, 20, 50, 150, 400, 1200, 4000, 12000]
        allowed = min(250, tleft() * 0.25)
        iters = int(min(5000, allowed * 25))
        h_ref, sc = smooth_gradient_descent(best_h, target_f, start_time, allowed, alphas, max(iters, 200))
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: Binary from continuous + SA
    if tleft() > 100:
        h_bin = project_binary(best_h, target)
        allowed = min(120, tleft() * 0.12)
        n_sa2 = int(min(3000000, allowed * 50000))
        h_sa2, c5_sa2 = informed_sa(h_bin, n_sa2, 0.02, 0.00001, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                    top_k=6, block_frac=0.12, max_block_len=10, two_opt_frac=0.12)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Multi-start SA
    for restart in range(6):
        if tleft() < 60:
            break
        if restart % 3 == 0:
            h_p = perturb_diverse(best_h, rng, rng.randint(8, 18))
        elif restart % 3 == 1:
            h_p = perturb_block(best_h, rng, 18)
        else:
            h_p = perturb_diverse(best_h, rng, 4)
            h_p = perturb_block(h_p, rng, 10)
        allowed = min(70, tleft() * 0.06)
        n_sa_r = int(min(2000000, allowed * 45000))
        h_sr, c5_sr = informed_sa(h_p, n_sa_r, 0.015, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                  top_k=5, block_frac=0.1, max_block_len=8, two_opt_frac=0.1)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()
        allowed2 = min(20, tleft() * 0.03)
        h_gr, c5_gr = greedy_local_search(h_sr, start_time, start_time + allowed2, k_arr, n_top=5, candidate_size=30)
        if c5_gr < best_score:
            best_score = c5_gr
            best_h = h_gr.copy()

    # Phase 7: Another continuous pass
    if tleft() > 80:
        target_f = float(n) / 2.0
        alphas2 = [50, 200, 800, 3000, 10000, 30000]
        allowed = min(150, tleft() * 0.2)
        iters2 = int(min(4000, allowed * 25))
        h_ref2, sc2 = smooth_gradient_descent(best_h, target_f, start_time, allowed, alphas2, max(iters2, 200))
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 8: Final binary SA burst
    if tleft() > 40:
        h_bin_f = project_binary(best_h, target)
        allowed = min(80, tleft() * 0.25)
        n_sa_f = int(min(4000000, max(100000, allowed * 50000)))
        h_sf, c5_f = informed_sa(h_bin_f, n_sa_f, 0.008, 0.000005, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                 top_k=8, block_frac=0.08, max_block_len=8, two_opt_frac=0.15)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sf.copy()

    # Phase 9: Final greedy
    if tleft() > 20:
        allowed = min(30, tleft() * 0.3)
        h_gf, c5_gf = greedy_local_search(best_h, start_time, start_time + allowed, k_arr, n_top=10, candidate_size=60)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gf.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    if np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10) and abs(best_h.sum() - target_final) < n * 0.51:
        h_final = np.clip(best_h, 0.0, 1.0)
        h_final = project_box_sum(h_final, target_final)
    else:
        h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score <= best_score + 1e-10:
        return h_final, final_score, n_final
    return h_final, final_score, n_final
