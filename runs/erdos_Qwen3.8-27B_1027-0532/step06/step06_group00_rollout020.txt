def swap_delta_corr(h, i, j, n):
    """Compute Δcorr[k] = D(i,k) - D(j,k) for swapping h[i]=1→0 and h[j]=0→1.
    Returns array of length 2n-1."""
    shifts = np.arange(-(n-1), n)
    # D(i, k) = h[i+k] (if valid) + h[i-k] (if valid)
    Di = np.zeros(2*n-1)
    pos_i = np.where((i + shifts >= 0) & (i + shifts < n))[0]
    Di[pos_i] += h[i + shifts[pos_i]]
    neg_i = np.where((i - shifts >= 0) & (i - shifts < n))[0]
    Di[neg_i] += h[i - shifts[neg_i]]
    
    Dj = np.zeros(2*n-1)
    pos_j = np.where((j + shifts >= 0) & (j + shifts < n))[0]
    Dj[pos_j] += h[j + shifts[pos_j]]
    neg_j = np.where((j - shifts >= 0) & (j - shifts < n))[0]
    Dj[neg_j] += h[j - shifts[neg_j]]
    
    return Di - Dj

def compute_D_full(h, n, shifts):
    """Compute D(p,k) for all p in [0,n) and k in shifts.
    D[p, k_idx] = h[p+k] (if valid) + h[p-k] (if valid)"""
    n_corr = 2*n - 1
    D = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        D[p, pos] += h[p + shifts[pos]]
        neg = (p - shifts >= 0) & (p - shifts < n)
        D[p, neg] += h[p - shifts[neg]]
    return D

def update_D(D, p, old_val, new_val, n, shifts):
    """Update D after changing h[p] from old_val to new_val.
    D(q,k) changes when q+k=p or q-k=p, i.e., k=p-q or k=q-p."""
    n_corr = 2*n - 1
    delta = new_val - old_val
    # For each q, the affected k values are k = p-q (from h[p+k]) and k = q-p (from h[p-k])
    # h[p+k] contributes to D[q,k] for k = p-q (if 0 ≤ p-q ≤ n-1 in shift index)
    # h[p-k] contributes to D[q,k] for k = q-p (if 0 ≤ q-p ≤ n-1 in shift index)
    
    # k = p - q means shift index = (p-q) + (n-1)
    # This is valid when |p-q| ≤ n-1, which is always true since p,q ∈ [0,n)
    
    # For h[p+k] term: D[q, k] gets h[p+k] when p+k = p, i.e., k=0? No...
    # Let me re-derive. D[q,k_idx] where k = k_idx - (n-1)
    # D[q,k] = h[q+k] (if valid) + h[q-k] (if valid)
    # h[p] appears in D[q,k] when:
    #   q+k = p, i.e., k = p-q, shift_idx = p-q+(n-1)
    #   q-k = p, i.e., k = q-p, shift_idx = q-p+(n-1)
    
    for q in range(n):
        k1 = p - q  # from h[q+k] = h[p]
        k2 = q - p  # from h[q-k] = h[p]
        s1 = k1 + (n - 1)
        s2 = k2 + (n - 1)
        if 0 <= s1 < 2*n-1:
            D[q, s1] += delta
        if 0 <= s2 < 2*n-1:
            D[q, s2] += delta

Now let me put it all together. The key advantage of this approach:
- Precompute D once: O(n²) 
- Each swap: compute D(i,·) - D(j,·) in O(1) (just subtract two rows), then find max of corr + (Di - Dj) in O(n)
- Update D after swap: O(n)
- Total per sweep: O(n_cand² * n) for candidate evaluation + O(n) for update
