```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
```

```python
def update_f(f, h, p, delta, n):
    # Update row p of f when h[p] changes by delta
    # f[p,s] = -1 + h[p+k_s]*valid + h[p-k_s]*valid
    # When h[p] changes by delta, f[q,s] changes if p = q+k_s or p = q-k_s
    # p = q+k_s → q = p-k_s = p+s-(n-1)
    # p = q-k_s → q = p+k_s = p+n-1-s
    # So for each s, f[p+s-(n-1), s] += delta (if row valid) and f[p+n-1-s, s] += delta (if row valid)
    s = np.arange(2*n-1)
    row1 = p + s - (n-1)
    mask1 = (row1 >= 0) & (row1 < n)
    f[row1[mask1], s[mask1]] += delta
    row2 = p + n - 1 - s
    mask2 = (row2 >= 0) & (row2 < n)
    f[row2[mask2], s[mask2]] += delta
```

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        g[-k:] += (1.0 - h[:n + k])
        g[:n + k] -= h[-k:]
    return g

def precompute_f(h, n):
    f = -np.ones((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    for p in range(n):
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += h[idx1[mask1]]
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    return f

def update_f_row(f, delta, p, n):
    s = np.arange(2 * n - 1)
    row1 = p + s - (n - 1)
    mask1 = (row1 >= 0) & (row1 < n)
    f[row1[mask1], s[mask1]] += delta
    row2 = p + n - 1 - s
    mask2 = (row2 >= 0) & (row2 < n)
    f[row2[mask2], s[mask2]] += delta

def smooth_obj_pen(h, alpha, n, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sm = np.sum(h) - n / 2.0
    return float(lse * dx + penalty * sm * sm)

def smooth_grad_pen(h, alpha, n, penalty, n_top=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if n_top is not None and n_top < 2 * n - 1:
        top_idx = np.argsort(corr)[-n_top:]
        c_top = corr[top_idx]
        shifted_top = np.clip(alpha * (c_top - np.max(c_top)), -500.0, 0.0)
        wsum = np.sum(np.exp(shifted_top))
        if wsum < 1e-300:
            w = np.zeros(n_top)
            w[np.argmax(c_top)] = 1.0
        else:
            w = np.exp(shifted_top) / wsum
        grad = np.zeros(n)
        mid = n - 1
        for ii in range(n_top):
            if w[ii] < 1e-12:
                continue
            k = int(top_idx[ii]) - mid
            grad += w[ii] * get_grad_shift(h, k, n)
    else:
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(n)
        mid = n - 1
        for s_idx in range(2 * n - 1):
            if w[s_idx] < 1e-12:
                continue
            k = s_idx - mid
            grad += w[s_idx] * get_grad_shift(h, k, n)
    grad *= dx
    sm = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * sm
    return grad

def lbfgs_optimize(h_start, n, target, alphas, penalty, time_limit, n_top=50):
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x):
            return smooth_obj_pen(x, alpha, n, penalty)
        def grad(x):
            return smooth_grad_pen(x, alpha, n, penalty, n_top)
        try:
            res_x, res_f, res_info = fmin_l_bfgs_b(obj, h, fprime=grad,
                                                     pgtol=1e-8, maxiter=500,
                                                     iprint=0, pgtol=1e-8)
            h_new = project_feasible(res_x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_score

def adam_projected(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                   lr_init, beta1, beta2, eps, alpha_init, alpha_end):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    lr = lr_init
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 500:
            break
        frac = step / max(1, max_steps)
        alpha_cur = alpha_init * (alpha_end / alpha_init) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
            lr = lr * 1.05
        else:
            no_improve += 1
            lr = lr * 0.95
        n_top = int(n_top_start + (n_top_end - n_top_start) * min(frac * 2, 1.0))
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        c_top = corr[top_idx]
        c_tmax = np.max(c_top)
        shifted = np.clip(alpha_cur * (c_top - c_tmax), -500.0, 0.0)
        wsum = np.sum(np.exp(shifted))
        if wsum < 1e-300:
            w = np.zeros(n_top)
            w[0] = 1.0
        else:
            w = np.exp(shifted) / wsum
        grad = np.zeros(n)
        for ii in range(n_top):
            if w[ii] < 1e-12:
                continue
            k = int(top_idx[ii]) - mid
            grad += w[ii] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def pair_cd_improved(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    f_arr = precompute_f(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad_star = get_grad_shift(h, k_star, n)
        grad_star -= np.mean(grad_star)
        sorted_idx = np.argsort(grad_star)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = f_arr[i] - f_arr[j]
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                a_d = 0.0
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial - 1e-15:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_f_row(f_arr, -delta, i, n)
            update_f_row(f_arr, delta, j, n)
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter, ftol):
    global _G_n_glob, _G_target_glob, _G_alpha_glob
    _G_n_glob = n
    _G_target_glob = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha_glob = alpha
        def _obj(h_vec):
            corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(_G_alpha_glob * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / _G_alpha_glob)
        def _con_fun(h_vec):
            return np.sum(h_vec) - _G_target_glob
        def _con_jac(h_vec):
            return np.ones(_G_n_glob)
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def multi_restart(h_best, n, target, time_limit, seed, n_restarts=6):

    rng = np.random.RandomState(seed)
    best_h = h_best.copy()

    for r in range(n_restarts):
        if time.time() - start > time_limit:

        remaining = time_limit - (time.time() - start)
        if remaining < 120:
            break
        scale = float(rng.choice([0.02, 0.04, 0.07, 0.1, 0.15]))
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        t_alloc = min(remaining * 0.7, 180.0)
        h_ad, sc_ad = adam_projected(h_pert, n, target,
                                     time_limit=t_alloc * 0.5, max_steps=5000,
                                     n_top_start=5, n_top_end=20,

                                     eps=1e-8, alpha_init=100, alpha_end=50000)
        sc_ad = compute_c5(h_ad)
        h_cd, sc_cd = pair_cd_improved(h_ad, n, target,
