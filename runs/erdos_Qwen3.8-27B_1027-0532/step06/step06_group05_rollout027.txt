    h = np.asarray(h, dtype=float)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-14:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

        m = np.zeros(n)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-14:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

def adam_optimize(h0, target_sum, start_time, time_budget, alphas, lr_base, beta1, beta2, eps, max_stall_per_alpha=400):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    best_smooth = float('inf')
    
    for alpha_idx, alpha in enumerate(alphas):
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = best_smooth
        # Learning rate: scale with alpha (higher alpha → smoother → can use bigger lr)
        lr = lr_base * (1.0 + 0.1 * alpha_idx)
        
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-11:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > max_stall_per_alpha:
                break
            
            # Adaptive learning rate: decrease if we're oscillating
            effective_lr = lr * max(0.3, 1.0 - stall / max_stall_per_alpha)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            
            # Gradient clipping
            g_norm = np.sqrt(np.sum(grad * grad))
            if g_norm > 10.0:
                grad = grad / g_norm * 10.0
                m = beta1 * m + (1.0 - beta1) * grad
                m_hat = m / (1.0 - beta1 ** t)
            
            h_new = h - effective_lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
        
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
            best_smooth = local_best
    
    return best_h, best_actual

def binary_ls(h_start, target_sum, max_iters, time_limit, start_time, rng, n_top=8, n_cand=25):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-n_top:]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score candidates
        i_score = np.zeros(len(ones), dtype=np.float64)
        j_score = np.zeros(len(zeros), dtype=np.float64)
        
        for t_idx in top_idx:
            k = int(t_idx) - (n - 1)
            if k == 0:
                continue
            w = corr[t_idx] / max_val  # weight by how close to max
            for ii in range(len(ones)):
                i = ones[ii]
                d = 0.0
                if 0 <= i + k < n:
                    d += h[i + k]
                if 0 <= i - k < n:
                    d += h[i - k]
                i_score[ii] += d * w
            for jj in range(len(zeros)):
                j = zeros[jj]
                d = 0.0
                if 0 <= j + k < n:
                    d += h[j + k]
                if 0 <= j - k < n:
                    d += h[j - k]
                j_score[jj] += d * w
        
        best_i_list = [ones[x] for x in np.argsort(i_score)[:n_cand]]
        best_j_list = [zeros[x] for x in np.argsort(j_score)[-n_cand:]]
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for i in best_i_list:
            for j in best_j_list:
                new_max = max_val
                reject = False
                for t_idx in top_idx:
                    k = int(t_idx) - (n - 1)
                    if k == 0:
                        continue
                    d = 0.0
                    if 0 <= i + k < n:
                        d += h[i + k]
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d -= h[j + k]
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > new_max:
                        new_max = v
                        if new_max >= max_val - best_improve:
                            reject = True
                            break
                if not reject and new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 5000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.5:
            # Targeted move: focus on worst lag
            worst_idx = int(np.argmax(corr))
            k = worst_idx - (n - 1)
            if k != 0:
                ip = [i for i in ones if 0 <= i + k < n or 0 <= i - k < n]
                jp = [j for j in zeros if 0 <= j + k < n or 0 <= j - k < n]
                if len(ip) > 0 and len(jp) > 0:
                    h_new[ip[rng.randint(len(ip))]] = 0.0
                    h_new[jp[rng.randint(len(jp))]] = 1.0
                    moved = True
        elif r < 0.85:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                idx_i = rng.choice(len(ones), 2, replace=False)
                idx_j = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[idx_i[0]]] = 0.0
                h_new[ones[idx_i[1]]] = 0.0
                h_new[zeros[idx_j[0]]] = 1.0
                h_new[zeros[idx_j[1]]] = 1.0
                moved = True
        
        if not moved:
            continue
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

        )
        if s_a < best_score:

                h_pert, target, start_time, time.time() - start_time + rt,

            )

        h_sa, c5_sa = sa_binary(h_bin3, int(target), 100000, 0.005, 0.00001, rng, sa_budget, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 6: Final polish
    if tleft() > 60:
        h_bin4 = project_binary(best_h, int(target))
        h_b4, c5_b4 = binary_ls(h_bin4, int(target), 30000, min(60, tleft() * 0.5), start_time, rng, n_top=12, n_cand=35)
        if c5_b4 < best_score:
            best_score = c5_b4
            best_h = h_b4.copy()
    
    if tleft() > 30:
        h_a2, s_a2 = adam_optimize(
            best_h, target, start_time, time.time() - start_time + max(10, tleft() - 10),
            alphas=[50000, 100000, 200000],
            lr_base=0.001, beta1=0.9, beta2=0.999, eps=1e-8,
            max_stall_per_alpha=200
        )
        if s_a2 < best_score:
            best_score = s_a2
            best_h = h_a2.copy()
    
    # Final check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    
    return h_final, final_score, n
