import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    val = smooth_val + penalty * (sum_h - target) ** 2
    grad += 2.0 * penalty * (sum_h - target) * np.ones(n)
    
    return val, grad

    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

            if sc < best_score:

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    Z = np.sum(np.exp(shifted))
    smooth_val = (c_max + np.log(Z) / alpha) * dx
    p = np.exp(shifted) / Z
    half = n - 1
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx] * dx
        if w < 1e-16:
            continue
        lag = k_idx - half
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n - lp])
            grad[:n - lp] -= w * h[lp:]
    sum_h = np.sum(h)
    val = smooth_val + penalty * (sum_h - target) ** 2
    grad += 2.0 * penalty * (sum_h - target)
    return val, grad

def optimize_lbfgsb(h_init, target, alphas, maxiter, time_limit, penalty=5000.0):
    start = time.time()
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        alpha_val = alpha
        pen_val = penalty
        tgt_val = target
        def fun(x):
            return smooth_obj_and_grad(x, alpha_val, tgt_val, pen_val)[0]
        def jac(x):
            return smooth_obj_and_grad(x, alpha_val, tgt_val, pen_val)[1]
        try:
            res = minimize(fun, h_cur, jac=jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_sinusoidal(n, freq, amp, phase):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(freq * np.pi * x + phase)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)

def make_bimodal(n, center1, center2, width):
    x = np.linspace(0, 2, n)
    h = 0.3 + 0.7 * np.exp(-((x - center1) / width) ** 2) + 0.7 * np.exp(-((x - center2) / width) ** 2)
    return project_feasible(np.clip(h, 0, 1), n / 2.0)

def coord_descent_refine(h, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - (n - 1)
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            C = slopes
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = C * mid + intercepts
                max_val = np.max(vals)
                tol = 1e-12 * max(1.0, abs(max_val))
                active = np.abs(vals - max_val) < tol
                if not np.any(active):
                    active = np.zeros(n_active, dtype=bool)
                    active[np.argmax(vals)] = True
                if np.all(C[active] > 1e-14):
                    hi = mid
                elif np.all(C[active] < -1e-14):
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_base = len(h_init)
    target_base = n_base / 2.0
    best_h = project_feasible(h_init.copy(), target_base)
    best_score = compute_c5(best_h)
    best_n = n_base

    # Phase 1: L-BFGS-B at n=400 with progressive alpha
    if tleft() > 100:
        alphas1 = [50, 200, 1000, 5000, 20000]
        h1, s1 = optimize_lbfgsb(best_h, target_base, alphas1, 300, tleft() * 0.3, penalty=5000.0)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Coordinate descent refinement
    if tleft() > 60:
        h2, s2 = coord_descent_refine(best_h, best_h.size / 2.0, n_active=50,
                                      max_passes=5000, time_limit=min(40, tleft() * 0.2), rng=rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Try n=600
    if tleft() > 180:
        n600 = 600
        t600 = n600 / 2.0
        h600 = interpolate_to(best_h, n600, t600)
        alphas3 = [50, 200, 1000, 5000, 20000]
        h600_o, s600 = optimize_lbfgsb(h600, t600, alphas3, 300, tleft() * 0.3, penalty=5000.0)
        if s600 < best_score:
            best_score = s600
            best_h = h600_o
            best_n = n600
    if tleft() > 60 and best_n == 600:
        h2, s2 = coord_descent_refine(best_h, best_n / 2.0, n_active=50,
                                      max_passes=5000, time_limit=min(40, tleft() * 0.2), rng=rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 4: Try n=800
    if tleft() > 200:
        n800 = 800
        t800 = n800 / 2.0
        h800 = interpolate_to(best_h, n800, t800)
        alphas4 = [50, 200, 1000, 5000, 20000, 80000]
        h800_o, s800 = optimize_lbfgsb(h800, t800, alphas4, 250, tleft() * 0.35, penalty=5000.0)
        if s800 < best_score:
            best_score = s800
            best_h = h800_o
            best_n = n800
    if tleft() > 60 and best_n == 800:
        h2, s2 = coord_descent_refine(best_h, best_n / 2.0, n_active=60,
                                      max_passes=5000, time_limit=min(40, tleft() * 0.2), rng=rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 5: Multiple restarts with different initializations
    n_use = best_n
    t_use = n_use / 2.0
    init_list = [best_h.copy()]
    init_list.append(make_sinusoidal(n_use, 2, 0.3, 0.0))
    init_list.append(make_sinusoidal(n_use, 2, 0.3, np.pi / 2))
    init_list.append(make_sinusoidal(n_use, 4, 0.2, 0.0))
    init_list.append(make_bimodal(n_use, 0.6, 1.4, 0.3))
    init_list.append(0.5 * np.ones(n_use) + 0.15 * np.sin(np.pi * np.linspace(0, 2, n_use)))
    init_list[-1] = project_feasible(np.clip(init_list[-1], 0, 1), t_use)
    for init_h in init_list:
        if tleft() < 80:
            break
        h_r = init_h + rng.randn(n_use) * 0.05
        h_r = project_feasible(h_r, t_use)
        alphas_r = [200, 1000, 5000, 20000]
        h_r_o, s_r = optimize_lbfgsb(h_r, t_use, alphas_r, 200, min(50, tleft() * 0.2), penalty=5000.0)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r_o

    # Phase 6: High-alpha polish at best resolution
    if tleft() > 80:
        n_use = len(best_h)
        t_use = n_use / 2.0
        alphas6 = [5000, 20000, 80000, 200000]
        h6, s6 = optimize_lbfgsb(best_h, t_use, alphas6, 500, tleft() * 0.3, penalty=10000.0)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final coordinate descent
    if tleft() > 40:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h7, s7 = coord_descent_refine(best_h, t_use, n_active=80,
                                      max_passes=10000, time_limit=min(30, tleft() * 0.4), rng=rng)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Try going back to n=400 with refined solution (sometimes lower n is better)
    if tleft() > 60 and len(best_h) > 400:
        h400 = interpolate_to(best_h, n_base, target_base)
        alphas8 = [1000, 5000, 20000, 80000]
        h400_o, s400 = optimize_lbfgsb(h400, target_base, alphas8, 300, tleft() * 0.3, penalty=10000.0)
        if s400 < best_score:
            best_score = s400
            best_h = h400_o

    # Phase 9: Final L-BFGS-B with very high alpha
    if tleft() > 30:
        n_use = len(best_h)
        t_use = n_use / 2.0
        alphas9 = [50000, 200000, 500000]
        h9, s9 = optimize_lbfgsb(best_h, t_use, alphas9, 500, tleft() * 0.5, penalty=50000.0)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Phase 10: Perturbation + quick optimization
    for noise_level in [0.01, 0.03, 0.08]:
        if tleft() < 40:
            break
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_p = best_h + rng.randn(n_use) * noise_level
        h_p = project_feasible(h_p, t_use)
        alphas_p = [2000, 10000, 50000]
        h_p_o, s_p = optimize_lbfgsb(h_p, t_use, alphas_p, 200, min(25, tleft() * 0.3), penalty=10000.0)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p_o

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
