def logsumexp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def logsumexp_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    # gradient of log-sum-exp with respect to h
    # d/dh_i of corr[k] for each k
    # corr[k] = sum_j h[j] * (1 - h[j+k])
    # This is bilinear, so gradient is:
