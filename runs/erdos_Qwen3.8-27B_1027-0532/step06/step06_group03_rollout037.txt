            g[j] += np.dot(weights[ks + mid], 1.0 - h_full[j - ks])
        k_lo2 = max(-j, -(n - 1))
        k_hi2 = min(n - 1 - j, n - 1)
        if k_hi2 >= k_lo2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            g[j] -= np.dot(weights[ks2 + mid], h_full[j + ks2])
    g *= dx
    return g

def smooth_obj_hp(h_red, alpha, n):
    h_full = expand_hp(h_red, n)
    dx = 2.0 / n
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_grad_hp(h_red, alpha, n):
    h_full = expand_hp(h_red, n)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    g_full = grad_full_compute(w, h_full, n)
    m = n // 2
    return g_full[:m] - g_full[m:]

def smooth_obj_rf(h_red, alpha, n):

def smooth_obj_full(h_full, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_grad_full(h_full, alpha, n):
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    return grad_full_compute(w, h_full, n)

def _con_effun(h, target):
    return np.sum(h) - target

def _con_ejac(h, target):
    return np.ones(len(h))

def lbfgsb_run(h_red, n, sym_type, alpha_list, maxiter, time_limit):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    start_t = time.time()
    h_cur = h_red.copy()
    for alpha in alpha_list:
        if time.time() - start_t > time_limit:
            break
        try:
            if sym_type == 'hp':
                res = minimize(smooth_obj_hp, h_cur, args=(alpha, n),
                              jac=smooth_grad_hp, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
            else:
                res = minimize(smooth_obj_rf, h_cur, args=(alpha, n),
                              jac=smooth_grad_rf, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_cur = h_new
        except Exception:
            pass
    return h_cur

def slsqp_full_run(h_start, n, target, alpha_list, maxiter, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_effun, 'jac': _con_ejac, 'args': (target,)}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alpha_list:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj_full, h_cur, args=(alpha, n),
                          jac=smooth_grad_full, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd_sym(h_red, n, sym_type, time_limit, max_sweeps, n_cand, top_k):
    m = n // 2
    start_t = time.time()
    h_cur = h_red.copy()
    best_h = h_cur.copy()
    if sym_type == 'hp':
        best_c5 = compute_c5(expand_hp(h_cur, n))
    else:
        best_c5 = compute_c5(expand_rf(h_cur, n))
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        if sym_type == 'hp':
            h_full = expand_hp(h_cur, n)
        else:
            h_full = expand_rf(h_cur, n)
        c5 = compute_c5(h_full)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        w_top = np.exp(50.0 * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h_full[idxs[v1] - k])
            grad[v2] -= wi * h_full[idxs[v2] + k]
        if sym_type == 'hp':
            g_red = grad[:m] - grad[m:]
        else:
            g_red = grad[:m] - grad[m:n][::-1]
        sorted_idx = np.argsort(g_red)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h_cur[i] - 1.0, -h_cur[j])
                delta_hi = min(h_cur[i], 1.0 - h_cur[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h_full[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h_full[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h_full[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h_full[j - shifts[v_jm]]
                if sym_type == 'hp':
                    g[m:] = -g[m:][::-1] * 0 + g[:m] * 0
                    g2 = np.zeros(n_corr)
                    vi_ip = (i + shifts >= 0) & (i + shifts < m)
                    g2[vi_ip] += h_cur[i + shifts[vi_ip]]
                    vi_im = (i - shifts >= 0) & (i - shifts < m)
                    g2[vi_im] += h_cur[i - shifts[vi_im]]
                    vi2p = ((i + shifts) - n >= 0) & ((i + shifts) - n < m)
                    g2[vi2p] -= h_cur[(i + shifts) - n]
                    vi2m = ((i - shifts) - n >= 0) & ((i - shifts) - n < m)
                    g2[vi2m] -= h_cur[(i - shifts) - n]
                    vj_ip = (j + shifts >= 0) & (j + shifts < m)
                    g2[vj_ip] -= h_cur[j + shifts[vj_ip]]
                    vj_im = (j - shifts >= 0) & (j - shifts < m)
                    g2[vj_im] -= h_cur[j - shifts[vj_im]]
                    vj2p = ((j + shifts) - n >= 0) & ((j + shifts) - n < m)
                    g2[vj2p] += h_cur[(j + shifts) - n]
                    vj2m = ((j - shifts) - n >= 0) & ((j - shifts) - n < m)
                    g2[vj2m] += h_cur[(j - shifts) - n]
                    g = g2
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * (2.0 / n)
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h_cur[best_i] -= best_delta
            h_cur[best_j] += best_delta
            h_cur = np.clip(h_cur, 0.0, 1.0)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    m = n // 2

    h_init_proj = project_feasible(h_init, target)
    best_h = h_init_proj.copy()
    best_score = compute_c5(best_h)

    h_red_hp = 0.5 * (h_init_proj[:m] + (1.0 - h_init_proj[m:]))
    h_red_hp = np.clip(h_red_hp, 0.0, 1.0)
    h_red_rf = 0.5 * (h_init_proj[:m] + (1.0 - h_init_proj[m:][::-1]))
    h_red_rf = np.clip(h_red_rf, 0.0, 1.0)

    alpha_schedules = [
        [30, 80, 200, 500, 1500],
        [50, 150, 400, 1000, 3000],
        [100, 300, 800, 2000, 5000],
    ]

    rng = np.random.RandomState(seed)

    # Phase 1: Half-period symmetric L-BFGS-B
    if tleft() > 300:
        h_r = lbfgsb_run(h_red_hp, n, 'hp', alpha_schedules[0], 500, min(60, tleft() * 0.15))
        h_full = expand_hp(h_r, n)
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = h_full
            best_h = project_feasible(best_h, target)

    # Phase 2: Reflection symmetric L-BFGS-B
    if tleft() > 250:
        h_r = lbfgsb_run(h_red_rf, n, 'rf', alpha_schedules[0], 500, min(60, tleft() * 0.15))
        h_full = expand_rf(h_r, n)
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = h_full
            best_h = project_feasible(best_h, target)

    # Phase 3: Multi-restart half-period with different alphas
    if tleft() > 200:
        for restart in range(5):
            if tleft() < 30:
                break
            noise = 0.05
            h_pert = h_red_hp + rng.randn(m) * noise
            h_pert = np.clip(h_pert, 0.0, 1.0)
            sched = alpha_schedules[restart % len(alpha_schedules)]
            h_r = lbfgsb_run(h_pert, n, 'hp', sched, 400, min(25, tleft() * 0.15))
            h_full = expand_hp(h_r, n)
            sc = compute_c5(h_full)
            if sc < best_score:
                best_score = sc
                best_h = h_full
                best_h = project_feasible(best_h, target)
                h_red_hp = h_r.copy()

    # Phase 4: Multi-restart reflection with different alphas
    if tleft() > 150:
        for restart in range(4):
            if tleft() < 25:
                break
            noise = 0.05
            h_pert = h_red_rf + rng.randn(m) * noise
            h_pert = np.clip(h_pert, 0.0, 1.0)
            sched = alpha_schedules[(restart + 1) % len(alpha_schedules)]
            h_r = lbfgsb_run(h_pert, n, 'rf', sched, 400, min(20, tleft() * 0.15))
            h_full = expand_rf(h_r, n)
            sc = compute_c5(h_full)
            if sc < best_score:
                best_score = sc
                best_h = h_full
                best_h = project_feasible(best_h, target)
                h_red_rf = h_r.copy()

    # Phase 5: Pair CD on best symmetric solution (half-period)
    if tleft() > 150:
        h_sym = 0.5 * (best_h[:m] + (1.0 - best_h[m:]))
        h_sym = np.clip(h_sym, 0.0, 1.0)
        h_cd, c5_cd = pair_cd_sym(h_sym, n, 'hp', time_limit=min(60, tleft() * 0.3),
                                   max_sweeps=5000, n_cand=40, top_k=3)
        h_full = expand_hp(h_cd, n)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = project_feasible(h_full, target)

    # Phase 6: Pair CD on best symmetric solution (reflection)
    if tleft() > 100:
        h_sym = 0.5 * (best_h[:m] + (1.0 - best_h[m:][::-1]))
        h_sym = np.clip(h_sym, 0.0, 1.0)
        h_cd, c5_cd = pair_cd_sym(h_sym, n, 'rf', time_limit=min(50, tleft() * 0.3),
                                   max_sweeps=4000, n_cand=40, top_k=3)
        h_full = expand_rf(h_cd, n)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = project_feasible(h_full, target)

    # Phase 7: Non-symmetric SLSQP polish
    if tleft() > 120:
        h_sls = slsqp_full_run(best_h, n, target, [500, 2000, 5000], 800, min(60, tleft() * 0.4))
        sc = compute_c5(h_sls)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 8: More aggressive multi-restart at high alpha
    if tleft() > 150:
        for restart in range(4):
            if tleft() < 30:
                break
            noise = 0.03
            h_pert = best_h + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_sls = slsqp_full_run(h_pert, n, target, [2000, 5000, 10000], 600, min(30, tleft() * 0.2))
            sc = compute_c5(h_sls)
            if sc < best_score:
                best_score = sc
                best_h = h_sls

    # Phase 9: Pair CD on best (non-symmetric)
    if tleft() > 100:
        h_sym_hp = 0.5 * (best_h[:m] + (1.0 - best_h[m:]))
        h_sym_hp = np.clip(h_sym_hp, 0.0, 1.0)
        h_cd, c5_cd = pair_cd_sym(h_sym_hp, n, 'hp', time_limit=min(50, tleft() * 0.3),
                                   max_sweeps=5000, n_cand=50, top_k=5)
        h_full = expand_hp(h_cd, n)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = project_feasible(h_full, target)

    # Phase 10: Higher resolution attempt
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            m_try = n_try // 2
            x_new = np.linspace(0.0, 1.0, n_try)
            x_old = np.linspace(0.0, 1.0, n)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)
            h_red_hires = 0.5 * (h_interp[:m_try] + (1.0 - h_interp[m_try:]))
            h_red_hires = np.clip(h_red_hires, 0.0, 1.0)
            h_r = lbfgsb_run(h_red_hires, n_try, 'hp', [50, 200, 800, 3000], 500, min(50, tleft() * 0.3))
            h_full = expand_hp(h_r, n_try)
            sc = compute_c5(h_full)
            if sc < best_score:
                best_score = sc
                best_h = project_feasible(h_full, target_try)
                n = n_try
                m = m_try
                target = target_try

    # Phase 11: Final L-BFGS-B polish at best resolution
    if tleft() > 60:
        m_cur = len(best_h) // 2
        h_sym = 0.5 * (best_h[:m_cur] + (1.0 - best_h[m_cur:]))
        h_sym = np.clip(h_sym, 0.0, 1.0)
        h_r = lbfgsb_run(h_sym, len(best_h), 'hp', [5000, 10000, 20000], 500, min(40, tleft() * 0.5))
        h_full = expand_hp(h_r, len(best_h))
        sc = compute_c5(h_full)
        if sc < best_score:
            best_score = sc
            best_h = project_feasible(h_full, len(best_h) / 2.0)

    # Phase 12: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_full_run(best_h, n_use, target_use, [5000, 20000], 500, min(20, tleft() * 0.6))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final
            best_score = sc

    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
