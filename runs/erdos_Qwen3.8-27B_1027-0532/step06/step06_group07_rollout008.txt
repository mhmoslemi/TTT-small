def block_cd(h_init, alpha, target, B, n_passes, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_blocks = n // B
    
    for p in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        for b in range(n_blocks):
            if time.time() - start_time > time_limit:
                break
            i_start = b * B
            i_end = i_start + B
            block = h[i_start:i_end].copy()
            others_sum = np.sum(h) - np.sum(block)
            block_target = target - others_sum
            
            if block_target < 0 or block_target > B:
                continue
            
            # SLSQP on the block
            def block_obj(x):
                h_trial = h.copy()
                h_trial[i_start:i_end] = x
                return smooth_obj(h_trial, alpha)
            
            def block_obj_grad(x):
                h_trial = h.copy()
                h_trial[i_start:i_end] = x
                g_full = smooth_grad(h_trial, alpha)
                return g_full[i_start:i_end]
            
            bounds = [(0.0, 1.0)] * B
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - block_target,
                     'jac': lambda x: np.ones(B)}]
            
            res = minimize(block_obj, block, args=(), jac=block_obj_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-15})
            
            h[i_start:i_end] = project_feasible(res.x, block_target)
            
        # Check improvement
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    p = exp_vals / Z
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(p > 1e-14)[0]
    for pos in active:
        pk = p[pos]
        k = pos - mid
        if k >= 0:
            grad[k:] += pk * dx * (1.0 - h[:n - k])
            grad[:n - k] -= pk * dx * h[k:]
        else:
            grad[:n + k] += pk * dx * (1.0 - h[-k:])
            grad[-k:] -= pk * dx * h[:n + k]
    return grad

def projected_gradient_opt(h0, alpha, target, max_iter, time_limit, start_time):

    h = h0.copy()
    v = np.zeros(n)
    beta = 0.85

    lr = 1.0 / (alpha * 3.0)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        g = smooth_grad(h, alpha)
        g -= np.mean(g)
        v = beta * v + g
        f_cur = smooth_obj(h, alpha)
        improved = False
        lr_try = lr
        for bt in range(25):
            h_try = project_feasible(h - lr_try * v, target)
            f_try = smooth_obj(h_try, alpha)
            if f_try < f_cur - 1e-16:
                h = h_try
                lr = lr_try
                improved = True
                break
            lr_try *= 0.5
        if not improved:
            no_improve += 1
            if no_improve > 20:

            v *= 0.5

        no_improve = 0
        c5_now = compute_c5(h)
        if c5_now < best_c5:

_global_block_alpha = 0.0
_global_i0 = 0
_global_i1 = 0
_global_B = 0

def _b_obj(x):

    ht[_global_i0:_global_i1] = x
    return smooth_obj(ht, _global_block_alpha)

def _b_grad(x):

    ht[_global_i0:_global_i1] = x
    g = smooth_grad(ht, _global_block_alpha)
    return g[_global_i0:_global_i1]

def _b_con(x):
    return np.sum(x) - _global_block_target

def _b_jac(x):
    return np.ones(_global_B)

def block_cd(h_init, alpha, target, B, max_passes, time_limit, start_time):
    global _global_block_h, _global_block_alpha, _global_i0, _global_i1, _global_B, _global_block_target
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_blocks = n // B
    _global_block_alpha = alpha
    _global_B = B
    bounds = [(0.0, 1.0)] * B
    cons = [{'type': 'eq', 'fun': _b_con, 'jac': _b_jac}]

    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        for b in range(n_blocks):
            if time.time() - start_time > time_limit:
                break
            i0 = b * B
            i1 = i0 + B
            if i1 > n:
                break
            _global_i0 = i0
            _global_i1 = i1
            others_sum = np.sum(h[:i0]) + np.sum(h[i1:])
            bt = target - others_sum
            if bt < -1e-6 or bt > B + 1e-6:
                continue
            bt = max(0.0, min(float(B), bt))
            _global_block_target = bt
            _global_block_h = h
            block = h[i0:i1].copy()
            try:
                res = minimize(_b_obj, block, jac=_b_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 80, 'ftol': 1e-16})
