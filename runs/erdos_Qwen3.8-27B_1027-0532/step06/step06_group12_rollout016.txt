Instead of linearizing f_k(h) as a linear function, use a quadratic approximation:

H_k[j,j+k] = H_k[j+k,j] = -dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgradient(h, active_indices, n):
    dx = 2.0 / n
    n_half = n - 1
    g = np.zeros(n)
    for idx in active_indices:
        k = idx - n_half
        abs_k = abs(k)
        if abs_k := 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n - k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n - m]) * dx
            g[:n - m] += (-h[m:]) * dx
    return g / len(active_indices)


def fw_phase(h_init, n, target, time_limit, max_iter=500000, K=60):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = compute_subgradient(h, top_idx, n)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        best_gamma = 2.0 / (t + 2)
        best_val = compute_c5(h)
        for gamma in [best_gamma, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_trial = (1.0 - gamma) * h + gamma * v
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_val:
                best_val = c5_trial
                best_gamma = gamma
        h_new = (1.0 - best_gamma) * h + best_gamma * v
        h = project_feasible(h_new, target)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            best_t = t
        if t > 20000 and t - best_t > 100000:
            break
    return best_h, best_c5


def lp_phase(h_init, n, target, time_limit, n_top=500, damping=0.95, solver_name='CLARABEL'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if stuck > 100:
            break
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Parameter()
        t_var.value = float(np.max(overlaps))
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
                constraints.append(coeff @ h_var <= t_var)
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = (1.0 - h[k:]) * dx
                coeff[k:] += (-h[:n - k]) * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n - m]) * dx
                coeff[:n - m] += (-h[m:]) * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        for sol in [solver_name, 'SCS', 'ECOS', 'CLARABEL']:
            try:
                if sol == 'SCS':
                    prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
                elif sol == 'ECOS':
                    prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
                else:
                    prob.solve(solver=cp.CLARABEL, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def sm_objective(h_flat, alpha):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h_flat, 1.0 - h_flat, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def cons_eq_fun(h_flat):
    return np.sum(h_flat) - _g_target


def cons_eq_jac(h_flat):
    return np.ones(_g_n)


def slsqp_phase(h_init, n, target, time_limit, alphas, maxiter=3000):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(sm_objective, h, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5


def grad_descent_phase(h_init, n, target, time_limit, alphas, steps_per_alpha=500):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_half = n - 1
    total_steps = steps_per_alpha * len(alphas)
    done = 0
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for step in range(steps_per_alpha):
            if time.time() - start > time_limit:
                break
            progress = done / total_steps
            done += 1
            lr = 0.003 * (1.0 + np.cos(np.pi * progress)) / 2.0 + 0.0003
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-30:
                break
            weights = weights / wsum
            g = np.zeros(n)
            for p in range(len(weights)):
                w = weights[p]
                if w < 1e-12:
                    continue
                k = p - n_half
                abs_k = abs(k)
                if abs_k >= n:
                    continue
                if abs_k == 0:
                    g += w * (1.0 - 2.0 * h) * dx
                elif k > 0:
                    g[:n - k] += w * (1.0 - h[k:]) * dx
                    g[k:] += w * (-h[:n - k]) * dx
                else:
                    m = -k
                    g[m:] += w * (1.0 - h[:n - m]) * dx
                    g[:n - m] += w * (-h[m:]) * dx
            h_new = h - lr * g
            h = project_feasible(h_new, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5


def random_binary_init(n, target, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:int(target)]] = 1.0
    return h


def structured_init(n, target, rng, shift):
    idx = np.arange(n)
    h = np.zeros(n)
    block = int(target // 5)
    offsets = [0, block, 2 * block, 3 * block, 4 * block]
    for off in offsets:
        start_pos = (off + shift) % n
        for j in range(block):
            h[(start_pos + j) % n] = 1.0
    return project_feasible(h, target)
