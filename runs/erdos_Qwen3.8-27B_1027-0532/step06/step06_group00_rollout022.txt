For block CD:
- Pick a block of B consecutive indices
- The contribution of this block to the objective is a quadratic function of the block variables (plus linear terms from interactions with the rest)
- Solve: min max_k [quadratic_k(block) + const_k] s.t. sum(block) = target_block, 0 ≤ block ≤ 1
- This is a min-max of quadratics, which we can approximate by:
  min t s.t. t ≥ quadratic_k(block) + const_k for all k, sum(block) = target, 0 ≤ block ≤ 1
- This is a QCQP with B+1 variables. For B=5-10, this is very fast.

Actually, the "quadratic_k" for a block is:
For shift k, the contribution from the block to c_k is:
c_k_block = sum_{i in block} h[i](1-h[i+k])
= sum_{i in block} h[i] - sum_{i in block} h[i]h[i+k]

If i+k is also in the block: h[i]h[i+k] is a product of two block variables (bilinear)
If i+k is not in the block: h[i]h[i+k] = h[i] * (fixed value) (linear in block)

So c_k_block is truly quadratic (bilinear) in the block variables. The QCQP is:
min t
s.t. t ≥ sum_{i∈B} h_i - sum_{i∈B: i+k∈B} h_i h_{i+k} - sum_{i∈B: i+k∉B} h_i * h_{i+k}^{fixed} + const_k]
     for all k (or just active k)
     sum_{i∈B} h_i = target_B
     0 ≤ h_i ≤ 1 for i ∈ B

The constraint t ≥ ... involves quadratic terms h_i h_j, making it a non-convex QCQP (since h_i h_j is indefinite).

However, we can use a convex relaxation: replace h_i h_j with a conservative bound. For 0 ≤ h_i, h_j ≤ 1:
h_i h_j ≥ min(h_i, h_j) * max(h_i, h_j) ... no that's circular.

Actually, for the purpose of MINIMIZING t, we want to MAXIMIZE the lower bound on c_k. Since c_k_block contains the term -h_i h_j (negative), maximizing c_k_block means minimizing h_i h_j. And h_i h_j ≥ 0, so -h_i h_j ≤ 0.

                # i.e., h[global_i] appears as the "1-h" part when the other index is global_i-k

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        h = project_feasible(h, target)

    return best_h, best_c5


def fictitious_play(h_init, n, target, time_limit, max_iter=3000, decay=0.95):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1

    lam = np.zeros(2 * n - 1)
    lam[int(np.argmax(np.correlate(h, 1.0 - h, mode='full')))] = 1.0

    no_improve = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = np.max(corr)
        active = np.where(corr >= c_max - 0.05 * c_max)[0]
        eps_fp = 1.0 / len(active)
        for a in active:
            lam[a] = decay * lam[a] + eps_fp
        lam /= np.sum(lam)

        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if lam[k_idx] < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += lam[k_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= lam[k_idx] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Solve the inner problem: minimize sum_k lam_k * c_k(h)
        # c_k = sum_i h[i](1-h[i+k]) -> this is quadratic
        # Use SLSQP on the weighted sum (which is smoother)
        def weighted_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(lam, c))

        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(weighted_obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < c5_val + 1e-10:
                h = h_new
        except Exception:
            pass

    return best_h, best_c5


def spectral_restart(h, n, target, keep_frac=0.5):
    fft_h = np.fft.rfft(h)
    n_freq = len(fft_h)
    cutoff = int(n_freq * keep_frac)
    fft_h[cutoff:] = 0.0
    h_smooth = np.fft.irfft(fft_h, n=n)
    h_smooth = project_feasible(h_smooth, target)
    return h_smooth


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, max_steps=20000, base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs_p = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs_p] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at original resolution
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0,
                            [30, 100, 500, 2000, 8000, 30000],
                            600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Upscale to n=800
    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 3: SLSQP at n=800 with long alpha schedule
    if tleft() > 200:
        h_pol1 = slsqp_round(best_h, n1, target1,
                             [100, 500, 2000, 8000, 30000, 100000],
                             800, 1e-14)
        sc = compute_c5(h_pol1)
        if sc < best_score:
            best_score = sc
            best_h = h_pol1
            n0 = n1
            target0 = target1

    # Phase 4: Block LP coordinate descent (novel)
    if tleft() > 180:
        h_bl, c5_bl = block_lp_cd(best_h, n0, target0,
                                  time_limit=min(150, tleft() * 0.6),
                                  block_size=6, max_iter=6000,
                                  eps_rel=0.02, max_no_improve=200)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl

    # Phase 5: Pair CD with expanded candidates
    if tleft() > 150:
        h_cd, c5_cd = fast_pair_cd(best_h, n0, target0,
                                   time_limit=min(120, tleft() * 0.6),
                                   max_sweeps=25000,
                                   n_cand_start=25, n_cand_end=90,
                                   max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Fictitious play (novel)
    if tleft() > 150:
        h_fp, c5_fp = fictitious_play(best_h, n0, target0,
                                      time_limit=min(120, tleft() * 0.6),
                                      max_iter=2000, decay=0.93)
        if c5_fp < best_score:
            best_score = c5_fp
            best_h = h_fp

    # Phase 7: SLSQP polish
    if tleft() > 80:
        h_pol2 = slsqp_round(best_h, n0, target0,
                             [5000, 20000, 80000],
                             1000, 1e-14)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 8: Spectral restart + re-optimize
    if tleft() > 200:
        for kf in [0.4, 0.55, 0.7]:
            if tleft() < 80:
                break
            h_sr = spectral_restart(best_h, n0, target0, keep_frac=kf)
            h_sr_pol = slsqp_round(h_sr, n0, target0,
                                   [200, 1000, 5000, 20000],
                                   400, 1e-14)
            sc_sr = compute_c5(h_sr_pol)
            if sc_sr < best_score:
                best_score = sc_sr
                best_h = h_sr_pol

    # Phase 9: Block LP again (different params)
    if tleft() > 120:
        h_bl2, c5_bl2 = block_lp_cd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.6),
                                    block_size=4, max_iter=5000,
                                    eps_rel=0.01, max_no_improve=150)
        if c5_bl2 < best_score:
            best_score = c5_bl2
            best_h = h_bl2

    # Phase 10: Upscale to n=1200
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 250:
            h2_pol = slsqp_round(h2, n2, target2,
                                 [200, 1000, 5000, 20000],
                                 500, 1e-14)
            sc2 = compute_c5(h2_pol)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2_pol
                n0 = n2
                target0 = target2

        if tleft() > 150:
            h2_cd, c5_2cd = fast_pair_cd(best_h, n0, target0,
                                         time_limit=min(120, tleft() * 0.6),
                                         max_sweeps=15000,
                                         n_cand_start=20, n_cand_end=60,
                                         max_no_improve=300)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd

        if tleft() > 80:
            h2_pol2 = slsqp_round(best_h, n0, target0,
                                  [10000, 50000],
                                  600, 1e-14)
            sc2b = compute_c5(h2_pol2)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h2_pol2

    # Phase 11: Stochastic refinement
    if tleft() > 80:
        h_st, c5_st = stochastic_refine(best_h, n0, target0,
                                        time_limit=min(50, tleft() * 0.6),
                                        max_steps=15000, base_scale=0.03)
        if c5_st < best_score:
            best_score = c5_st
            best_h = h_st

    # Phase 12: Final pair CD
    if tleft() > 60:
        h_final_cd, c5_final_cd = fast_pair_cd(best_h, n0, target0,
                                               time_limit=min(45, tleft() * 0.6),
                                               max_sweeps=10000,
                                               n_cand_start=30, n_cand_end=100,
                                               max_no_improve=200)
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    # Phase 13: Final SLSQP polish
    if tleft() > 40:
        h_final = slsqp_round(best_h, n0, target0,
                              [20000, 100000],
                              800, 1e-15)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 14: Random restarts with perturbation
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.05, 0.1, 0.2]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_pert_pol = slsqp_round(h_pert, n0, target0,
                                     [100, 1000, 5000, 20000, 80000],
                                     500, 1e-14)
            sc_pert = compute_c5(h_pert_pol)
            if sc_pert < best_score:
                best_score = sc_pert
                best_h = h_pert_pol

    # Final projection and evaluation
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
