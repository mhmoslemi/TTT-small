            h = project_feasible(h, target)
            sweeps_done += 1

    return best_h, best_c5


def progressive_pair_cd(h, n, target, time_limit):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    stages = [
        (30, 50, 300),
        (50, 80, 250),
        (80, 100, 200),
        (100, 120, 150),
    ]

    total_patience = 400
    for n_cand, max_ncand, patience in stages:
        if time.time() - start > time_limit * 0.9:
            break
        phase_time = (time_limit * 0.9) - (time.time() - start)
        if phase_time < 10:
            break
        h_try, c5_try = fast_pair_cd(best_h, n, target,
                                     time_limit=phase_time,
                                     max_sweeps=100000,
                                     n_cand=n_cand,
                                     patience=patience,
                                     n_bisect=14)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try

    if time.time() - start < time_limit * 0.95:
        remaining = time_limit * 0.95 - (time.time() - start)
        if remaining > 20:
            h_try, c5_try = fast_pair_cd(best_h, n, target,
                                         time_limit=remaining,
                                         max_sweeps=100000,
                                         n_cand=60,
                                         patience=100,
                                         n_bisect=16)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try

    return best_h, best_c5


def multi_restart_pair_cd(h, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    n_restarts = max(2, int(time_limit / 80))
    per_restart = time_limit / n_restarts - 5

    for r in range(n_restarts):
        if time.time() - start > time_limit - 5:
            break
        if r == 0:
            h_cur = best_h.copy()
        else:
            scale = 0.02 + 0.03 * r
            h_pert = best_h + rng.randn(n) * scale
            h_cur = project_feasible(h_pert, target)

        h_try, c5_try = fast_pair_cd(h_cur, n, target,
                                     time_limit=per_restart,
                                     max_sweeps=50000,
                                     n_cand=50,
                                     patience=150,
                                     n_bisect=14)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try

    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def game_theoretic_descent(h_init, n, target, time_limit, max_steps=10000,
                           eta_base=1.5, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve_count = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 300:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / (lam_top.sum() + 1e-30)

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(8):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.1, 0.5)
                improved = True
                break
            lr *= 0.5

        if not improved:
            best_lr *= 0.9
            if best_lr < 0.001:
                break

        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / (lam_new.sum() + 1e-30)
        lam = 0.7 * lam + 0.3 * lam_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    _global_n = 400
    _global_target = 200.0
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    _global_n = n
    _global_target = target

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast pair CD (main workhorse)
    if tleft() > 200:
        p1_time = min(300, tleft() * 0.25)
        h_try, sc_try = progressive_pair_cd(best_h, n, target, time_limit=p1_time)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Phase 2: L-BFGS-B polish
    if tleft() > 100:
        for alpha in [500, 2000, 10000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 3: More pair CD with different parameters
    if tleft() > 150:
        p3_time = min(250, tleft() * 0.3)
        h_try, sc_try = fast_pair_cd(best_h, n, target,
                                     time_limit=p3_time,
                                     max_sweeps=100000,
                                     n_cand=80,
                                     patience=200,
                                     n_bisect=16)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Phase 4: Game-theoretic descent for different landscape
    if tleft() > 80:
        gt_time = min(60, tleft() * 0.15)
        h_try, sc_try = game_theoretic_descent(best_h, n, target,
                                               time_limit=gt_time,
                                               max_steps=8000,
                                               eta_base=2.0, m_top=12)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try
        # Follow up with pair CD
        if tleft() > 80:
            h_try2, sc_try2 = fast_pair_cd(best_h, n, target,
                                           time_limit=min(80, tleft() * 0.3),
                                           max_sweeps=50000,
                                           n_cand=60, patience=150,
                                           n_bisect=14)
            if sc_try2 < best_score:
                best_score = sc_try2
                best_h = h_try2

    # Phase 5: SLSQP polish
    if tleft() > 80:
        h_try = slsqp_round(best_h, n, target,
                            [5000, 20000], 800, 1e-15)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Phase 6: Multi-restart with perturbations
    if tleft() > 120:
        mr_time = min(120, tleft() * 0.25)
        h_try, sc_try = multi_restart_pair_cd(best_h, n, target,
                                              time_limit=mr_time, seed=seed)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Phase 7: Final intensive pair CD
    if tleft() > 80:
        p7_time = min(150, tleft() * 0.4)
        h_try, sc_try = fast_pair_cd(best_h, n, target,
                                     time_limit=p7_time,
                                     max_sweeps=200000,
                                     n_cand=100,
                                     patience=150,
                                     n_bisect=16)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Phase 8: Final L-BFGS-B + SLSQP
    if tleft() > 40:
        for alpha in [50000, 200000]:
            if tleft() < 20:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=2000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 9: Ultimate pair CD squeeze
    if tleft() > 30:
        h_try, sc_try = fast_pair_cd(best_h, n, target,
                                     time_limit=max(10, tleft() - 5),
                                     max_sweeps=200000,
                                     n_cand=70,
                                     patience=80,
                                     n_bisect=18)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
