def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

def solve_1d_max_line(slopes, intercepts, lam):

            try:
                prob.solve(solver=cp.ECOS, max_iter=10000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            damping = min(0.98, damping + 0.02)
        else:
            damping = max(0.4, damping * 0.92)

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = max(float(t_var.value) if t_var.value is not None else best_c5, best_c5)
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 2e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 120:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: CD with 150 active constraints
    if tleft() > 200:
        cd_time = min(180, tleft() * 0.2)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=150,
                                    max_passes=30000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: LP refinement (pure LP, no regularization)
    if tleft() > 180:
        lp_time = min(160, tleft() * 0.2)
        h_lp, c5_lp = lp_refine(best_h, target, n_active=100,
                                max_iters=5000, time_limit=lp_time, damping_init=0.8)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: CD again with more iterations
    if tleft() > 150:
        cd_time = min(130, tleft() * 0.18)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=200,
                                    max_passes=30000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP moderate
    if tleft() > 150:
        h_ref, sc_ref = slsqp_round(best_h, n, target, [200, 1000, 5000], 2000, 1e-14)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 5: LP with tighter settings
    if tleft() > 150:
        lp_time = min(130, tleft() * 0.18)
        h_lp, c5_lp = lp_refine(best_h, target, n_active=130,
                                max_iters=5000, time_limit=lp_time, damping_init=0.9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: Try higher resolution n=600
    if tleft() > 350:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)

        if tleft() > 300:
            cd_time = min(120, tleft() * 0.3)
            h_cd6, c5_cd6 = coord_descent(h_interp, target_try, n_active=100,
                                          max_passes=20000, time_limit=cd_time, rng=rng)
            if c5_cd6 < best_score:
                best_score = c5_cd6
                best_h = h_cd6
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp

        if tleft() > 200:
            cur_n = len(best_h)
            cur_target = cur_n / 2.0
            lp_time = min(100, tleft() * 0.3)
            h_lp6, c5_lp6 = lp_refine(best_h, cur_target, n_active=100,
                                      max_iters=3000, time_limit=lp_time, damping_init=0.85)
            if c5_lp6 < best_score:
                best_score = c5_lp6
                best_h = h_lp6

    # Phase 7: SLSQP high alpha
    if tleft() > 120:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc_ref = slsqp_round(best_h, n_use, target_use,
                                    [5000, 20000, 80000], 1500, 1e-15)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 8: Final CD
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(80, tleft() * 0.4)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=200,
                                    max_passes=30000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Subgradient refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=8000, time_limit=min(45, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: Perturbation restarts
    for noise in [0.01, 0.03, 0.08, 0.15]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc_ref = slsqp_round(h_pert, n_use, target_use,
                                    [500, 3000, 15000], 1200, 1e-14)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref
        h_cd, c5_cd = coord_descent(h_pert, target_use, n_active=100,
                                    max_passes=5000, time_limit=30, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final LP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(30, tleft() * 0.7)
        h_lp, c5_lp = lp_refine(best_h, target_use, n_active=150,
                                max_iters=3000, time_limit=lp_time, damping_init=0.95)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc_ref = slsqp_round(best_h, n_use, target_use,
                                    [50000, 200000, 500000], 600, 1e-16)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 13: Final micro CD
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=250,
                                    max_passes=10000, time_limit=min(10, tleft() * 0.7), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
