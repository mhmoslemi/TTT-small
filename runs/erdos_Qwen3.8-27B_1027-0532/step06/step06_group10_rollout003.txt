    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    t_fw = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute smooth gradient (using log-sum-exp weighting)
        c_max = np.max(corr)
        shifted = np.clip(alpha_smooth * (corr - c_max), -50.0, 0.0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(corr)
            w[int(np.argmax(corr))] = 1.0
        else:
            w = w / ws
        
        # Compute weighted gradient
        grad = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[s] * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)  # Remove mean to handle equality constraint
        
        # Linear minimization oracle
        sorted_idx = np.argsort(grad)
        y = np.zeros(n)
        y[sorted_idx[:target]] = 1.0
        
        # Step size (2.5/(t+2) for strong convexity)
        t_fw += 1
        gamma = 2.0 / (t_fw + 2.0)
        
        h_new = (1 - gamma) * h + gamma * y
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new >= c5_val - 1e-14:
            # Try exact line search
            best_gamma = 0
            best_f = c5_val
            for g_try in [0.5, 0.3, 0.2, 0.1, 0.05]:
                h_try = (1 - g_try) * h + g_try * y

                if f_try < best_f:

                    best_gamma = g_try
            if best_gamma > 0:

            # else: no improvement, break

        else:

+t*(1-h[j+k]) - t*h[j-k]

    dx = 2.0 / n

    best_h = h.copy()

    for a_start, a_end in alpha_schedule:
        n_iters = a_end - a_start
        for it in range(n_iters):

            y = np.zeros(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo = -1.5
    hi = 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(proj > 1e-12)[0]
        if len(idx) > 0:
            proj[idx] += residual / len(idx)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_max_val(corr, alpha):
    cm = np.max(corr)
    sh = np.clip(alpha * (corr - cm), -50.0, 0.0)
    return cm + np.log(np.sum(np.exp(sh))) / alpha

def smooth_max_w(corr, alpha):
    cm = np.max(corr)
    sh = np.clip(alpha * (corr - cm), -50.0, 0.0)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    return w

def smax_obj_fn(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(smooth_max_val(corr, alpha) * dx)

def smax_grad_fn(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_w(corr, alpha)
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    act = np.where(w > 1e-8)[0]
    for s in act:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, tlimit, t0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - t0 > tlimit:
            break
        a_val = alpha
        n_val = n
        def _obj(x):
            return smax_obj_fn(x, a_val)
        def _grad(x):
            return smax_grad_fn(x, a_val, n_val)
        def _con(x):
            return np.sum(x) - target
        def _cjac(x):
            return np.ones(n_val)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _cjac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def frank_wolfe(h0, n, target, time_limit, max_iters=30000):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    t_fw = 0
    stuck_count = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        alpha = 20.0 + 30.0 * (it / max(1.0, float(max_iters)))
        alpha = min(alpha, 2000.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 3000:
            break
        w = smooth_max_w(corr, alpha)
        grad = np.zeros(n)
        act = np.where(w > 1e-8)[0]
        for s in act:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[s] * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        y = np.zeros(n)
        y[sorted_idx[:int(target)]] = 1.0
        t_fw += 1
        gamma = 2.0 / (t_fw + 2.0)
        h_new = project_feasible((1.0 - gamma) * h + gamma * y, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_val - 1e-14:
            h = h_new
        else:
            improved = False
            for g_try in [gamma * 0.5, gamma * 0.25, gamma * 0.1]:
                h_try = project_feasible((1.0 - g_try) * h + g_try * y, target)
                f_try = compute_c5(h_try)
                if f_try < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved and it > 200:
                stuck_count += 500
    return best_h, best_c5

def fista_optimize(h0, n, target, time_limit, max_steps=50000):
    t0 = time.time()
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    best_h = x.copy()
    best_c5 = compute_c5(x)
    t_nest = 1.0
    alpha_base = 50.0
    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        alpha = alpha_base * (1.0 + step * 0.01)
        alpha = min(alpha, 50000.0)
        grad_y = smax_grad_fn(y, alpha, n)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c5_y = float(np.max(corr_y)) * dx
        H_est = max(2.0, alpha * 0.5)
        lr = 1.0 / H_est
        x_new = project_feasible(y - lr * grad_y, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        y = x_new + beta * (x_new - x)
        y = project_feasible(y, target)
        x = x_new
        c5_x = compute_c5(x)
        if c5_x < best_c5 - 1e-14:
            best_c5 = c5_x
            best_h = x.copy()
    return best_h, best_c5

def coord_descent(h0, n, target, time_limit, max_sweeps=50000, n_cand=20):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        w_full = smooth_max_w(corr, 50.0)
        act = np.where(w_full > 1e-6)[0]
        for s in act:
            k2 = int(s) - mid
            if k2 == k_star:
                continue
            v1b = (idxs - k2 >= 0) & (idxs - k2 < n)
            v2b = (idxs + k2 >= 0) & (idxs + k2 < n)
            grad[v1b] += w_full[s] * (1.0 - h[idxs[v1b] - k2])
            grad[v2b] -= w_full[s] * h[idxs[v2b] + k2]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(28):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_round(h0, n, target, time_limit, n_active=15, max_rounds=40):
    t0 = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var * dx)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP with increasing alpha (aggressive)
    if tleft() > 200:
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [50, 150, 500, 2000, 8000, 30000, 100000],
                                   4000, 1e-15, tleft() * 0.35, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Frank-Wolfe (new method)
    if tleft() > 200:
        fw_time = min(150.0, tleft() * 0.25)
        h_fw, c5_fw = frank_wolfe(best_h, n, target, time_limit=fw_time, max_iters=40000)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 3: SLSQP refinement after FW
    if tleft() > 200:
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [200, 800, 3000, 12000, 50000, 200000],
                                   4000, 1e-15, tleft() * 0.25, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: Coordinate descent with exact line search
    if tleft() > 200:
        cd_time = min(200.0, tleft() * 0.25)
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     time_limit=cd_time, max_sweeps=50000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if tleft() > 150:
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [500, 2000, 10000, 50000, 200000],
                                   3000, 1e-15, tleft() * 0.3, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 6: LP refinement
    if tleft() > 120:
        h_lp, c5_lp = lp_round(best_h, n, target,
                                time_limit=min(100.0, tleft() * 0.3),
                                n_active=20, max_rounds=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Another CD round
    if tleft() > 150:
        cd_time = min(120.0, tleft() * 0.25)
        h_cd, c5_cd = coord_descent(best_h, n, target,
                                     time_limit=cd_time, max_sweeps=30000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Try n=800
    if tleft() > 250:
        n_try = 800
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, s_hires = slsqp_optimize(h_interp, n_try, target_try,
                                           [50, 200, 800, 3000, 10000, 40000],
                                           3000, 1e-14,
                                           min(120.0, tleft() * 0.35), start_time)
        if s_hires < best_score:
            best_score = s_hires
            best_h = h_hires
        if tleft() > 100:
            h_cd2, c5_cd2 = coord_descent(best_h, len(best_h), float(len(best_h)) / 2.0,
                                           time_limit=min(80.0, tleft() * 0.35),
                                           max_sweeps=20000, n_cand=20)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
        if tleft() > 60:
            h_hires2, s_hires2 = slsqp_optimize(best_h, len(best_h), float(len(best_h)) / 2.0,
                                                 [2000, 10000, 50000],
                                                 2000, 1e-14,
                                                 min(50.0, tleft() * 0.4), start_time)
            if s_hires2 < best_score:
                best_score = s_hires2
                best_h = h_hires2

    # Phase 9: Fista at best resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ne, c5_ne = fista_optimize(best_h, n_use, target_use,
                                      time_limit=min(80.0, tleft() * 0.35),
                                      max_steps=30000)
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne

    # Phase 10: Final SLSQP + CD polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_optimize(best_h, n_use, target_use,
                                           [2000, 10000, 50000, 200000],
                                           2000, 1e-15,
                                           min(35.0, tleft() * 0.5), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final
        if tleft() > 20:
            h_cd, c5_cd = coord_descent(best_h, n_use, target_use,
                                         time_limit=min(15.0, tleft() * 0.6),
                                         max_sweeps=10000, n_cand=20)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
