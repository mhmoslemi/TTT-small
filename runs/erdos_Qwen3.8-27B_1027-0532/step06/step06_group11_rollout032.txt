    import cvxpy as cp
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        # Get top-m active shifts
        top_idx = np.argsort(corr)[-n_active:]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        mid = n - 1
        for s_idx in top_idx:
            k = int(s_idx) - mid
            # Linearize S_k = sum_i h[i]*(1-h[i+k])
            # around current h
            linearized = 0
            for i in range(n):
                if 0 <= i + k < n:
                    # f = h[i]*(1-h[i+k])
                    # df/dh[i] = 1 - h[i+k]
                    # df/dh[i+k] = -h[i]
                    # f(h) ≈ f(h_old) + (1-h_old[i+k])*(h[i]-h_old[i]) - h_old[i]*(h[i+k]-h_old[i+k])
                    val = h[i] * (1 - h[i+k])
                    lin = val + (1 - h[i+k]) * (h_var[i] - h[i]) - h[i] * (h_var[i+k] - h[i+k])
                    linearized += lin
            constraints.append(cp.sum(linearized) * dx <= t_var)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, mip_gap=0.01)
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # No improvement, break
            h = h_new  # still update for next linearization
            if it > 5:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def active_set_lp(h_start, n, target, time_limit, n_active=8, max_iters=200):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        # Get top active shifts (including the max and nearby)
        sorted_idx = np.argsort(corr)[::-1]
        top_idx = sorted_idx[:n_active]
        
        # Build LP with McCormick relaxation
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        # w variables for each active shift
        w_vars = {}
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        obj_terms = []
        
        for s_idx in top_idx:
            k = int(s_idx) - mid
            # Determine valid range
            if k >= 0:
                valid_i = np.arange(0, n - k)
            else:
                valid_i = np.arange(-k, n)
            
            w = cp.Variable(len(valid_i))
            w_vars[s_idx] = (w, valid_i)
            
            # S_k = sum_{i in valid_i} h[i]*(1-h[i+k]) * dx
            #     = sum h[i]*dx - sum w[i]*dx
            # where w[i] = h[i]*h[i+k]
            
            # McCormick: w <= h[i], w <= h[i+k], w >= h[i]+h[i+k]-1, w >= 0
            for j, i in enumerate(valid_i):
                ip = i + k
                w[j] <= h_var[i]
                w[j] <= h_var[ip]
                w[j] >= h_var[i] + h_var[ip] - 1
                w[j] >= 0
            
            # S_k constraint: (sum valid h[i] - sum w[j]) * dx <= t
            s_k_expr = (cp.sum(h_var[valid_i]) - cp.sum(w)) * dx
            constraints.append(s_k_expr <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Nudge towards LP solution even if true C5 doesn't improve
            h = 0.7 * h + 0.3 * h_new
            h = project_feasible(h, target)
            if it > 20:
                break
    
    return best_h, best_c5

def penalty_optimize(h_start, n, target, time_limit, max_steps, n_active, base_lr):

def slsqp_logsumexp(h_start, n, target, alphas, maxiter_per_alpha, ftol):

    _n = n
    _target = target

        return np.ones(_n)

            max_C = np.max(C)
            shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
            return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def _slsqp_obj(h):
    n = _G_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def top1_slsqp(h_start, n, target, time_limit, maxiter):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    _G_alpha = 1000000.0
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1

    def top1_obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        max_C = np.max(C)
        active = np.argmax(C)
        return float(C[active])

    def top1_obj_jac(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        active = int(np.argmax(C))
        k = active - mid
        g = get_grad_shift(h, k, n)
        g *= dx
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]

    while time.time() - start < time_limit:
        try:
            res = minimize(top1_obj, h, jac=top1_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': min(maxiter, 500), 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                break
        except Exception:
            break
    return best_h, best_c5

def active_set_lp(h_start, n, target, time_limit, n_active, max_iters):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        sorted_idx = np.argsort(corr)[::-1]
        top_idx = sorted_idx[:n_active]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]

        for s_idx in top_idx:
            k = int(s_idx) - mid
            if k >= 0:
                valid_i = np.arange(0, n - k)
            else:
                valid_i = np.arange(-k, n)
            nv = len(valid_i)
            if nv == 0:
                continue
            w = cp.Variable(nv)
            for j in range(nv):
                i = int(valid_i[j])
                ip = i + k
                w[j] <= h_var[i]
                w[j] <= h_var[ip]
                w[j] >= 0
            s_k_expr = (cp.sum(h_var[valid_i]) - cp.sum(w)) * dx
            constraints.append(s_k_expr <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = 0.5 * h + 0.5 * h_new
            h = project_feasible(h, target)
            if it > 10:
                break
    return best_h, best_c5

def penalty_subgrad(h_start, n, target, time_limit, max_steps, n_top, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    t_target = 0.38

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 600:
            break
        corr = compute_corr(h)
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        sorted_idx = np.argsort(corr)[::-1]
        top_idx = sorted_idx[:n_top]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            s_k = corr[s_idx] * dx
            if s_k > t_target:
                weight = (s_k - t_target) ** 2
                grad += weight * get_grad_shift(h, k, n)

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 3
            t_target = max(0.30, t_target - 0.0005)
    return best_h, best_c5

def adam_topk(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end, lr, beta1, beta2):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 1000:
            break
        corr = compute_corr(h)
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.5), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_starts(n, target):

    candidates = []

                h_try = project_feasible(h_try, target)
                candidates.append(h_try)
    return candidates
