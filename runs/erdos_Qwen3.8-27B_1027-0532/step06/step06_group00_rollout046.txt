import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except:
            pass
    return h_cur

def nesterov_subgrad(h_init, n, target, time_limit, lr=0.005, momentum=0.9,
                      n_active=5, max_iter=200000, patience=10000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    no_improve = 0
    lr_cur = lr
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        # Find top n_active shifts
        top_idx = np.argsort(corr)[-n_active:]
        weights = corr[top_idx] / np.sum(corr[top_idx])
        
        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            k = int(s_idx) - mid
            m1 = idxs[(idxs + k >= 0) & (idxs + k < n)]
            grad[m1] += w * (1.0 - h[m1 + k])
            m2 = idxs[(idxs - k >= 0) & (idxs - k < n)]
            grad[m2] -= w * h[m2 - k]
        
        grad -= np.mean(grad)
        
        # Nesterov step
        vel = momentum * vel - lr_cur * grad
        h_new = project_feasible(h + vel, target)
        h = h_new
        
        # Adaptive learning rate
        if no_improve > patience // 4:
            lr_cur *= 0.9
    
    return best_h, best_c5

def change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change

def update_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_lo=15, n_cand_hi=60, patience=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    chg = change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        # Gradient for candidate selection
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        nc = int(n_cand_lo + (n_cand_hi - n_cand_lo) * progress)
        nc = min(nc, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                
                g = chg[j] - chg[i]
                a, b = d_lo, d_hi
                for _ in range(12):
                    m = 0.5 * (a + b)
                    vals = corr + m * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-12:
                        b = m
                    else:
                        a = m
                
                d_star = 0.5 * (a + b)
                if abs(d_star) < 1e-12:
                    continue
                
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_row(chg, best_i, h, n, shifts)
            update_row(chg, best_j, h, n, shifts)
    
    return best_h, best_c5

def interpolate(h_src, n_dst, target):
    x_new = np.linspace(0, 1, n_dst)
    x_old = np.linspace(0, 1, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target)

            h_try[block] = res.x

                best_c5 = sc

                no_improve = 0

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_chg_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    chg = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        chg[p, pos] += (1.0 - h[p + shifts[pos]])
        chg[p, neg] -= h[p - shifts[neg]]
    return chg


def update_chg_row(chg, p, h, n, shifts):
    chg[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    chg[p, pos] += (1.0 - h[p + shifts[pos]])
    chg[p, neg] -= h[p - shifts[neg]]
    return chg


def nesterov_subgrad(h_init, n, target, time_limit, lr=0.003, momentum=0.92,
                     n_active=5, patience=8000):
    start = time.time()
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    no_improve = 0
    lr_cur = lr

    for it in range(500000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum < 1e-15:
            break
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            k = int(s_idx) - mid
            m1 = idxs[(idxs + k >= 0) & (idxs + k < n)]
            grad[m1] += w * (1.0 - h[m1 + k])
            m2 = idxs[(idxs - k >= 0) & (idxs - k < n)]
            grad[m2] -= w * h[m2 - k]

        grad -= np.mean(grad)
        vel = momentum * vel - lr_cur * grad
        h = project_feasible(h + vel, target)

        if no_improve > patience // 5 and no_improve < patience // 5 + 100:
            lr_cur *= 0.95

    return best_h, best_c5 * (2.0 / n)

            if approx < best_trial:
                best_trial = approx
                best_i = i
                best_j = j
                best_delta = d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_chg_row(chg, best_i, h, n, shifts)
            update_chg_row(chg, best_j, h, n, shifts)
        else:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def generate_structured_starts(n, target):
    starts = []
    x = np.linspace(0, 2, n)

    h1 = 0.5 + 0.5 * np.sin(np.pi * x)
    starts.append(project_feasible(h1, target))

    h2 = np.where((x % 2.0) < 0.5, 1.0, 0.0)
    starts.append(project_feasible(h2, target))

    rng = np.random.RandomState(777)
    h3 = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(4 * np.pi * x)
                 + rng.randn(n) * 0.1, 0, 1)
    starts.append(project_feasible(h3, target))

    h4 = np.zeros(n)
    h4[::4] = 1.0
    h4[1::4] = 0.5
    starts.append(project_feasible(h4, target))

    return starts


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 30:
        best_h = slsqp_optimize(best_h, n0, target0, [100, 500, 2000], 500)
        best_score = compute_c5(best_h)

    if tleft() > 60:
        h_ns, c5_ns = nesterov_subgrad(best_h, n0, target0,
                                       time_limit=min(40, tleft() * 0.5),
                                       lr=0.003, momentum=0.92, n_active=5)
        if c5_ns < best_score:
            best_score = c5_ns
            best_h = h_ns

    if tleft() > 100:
        h_cd, c5_cd = fast_pair_cd(best_h, n0, target0,
                                   time_limit=min(80, tleft() * 0.6),
                                   nc=35, top_k=12, patience=200)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_ma, c5_ma = multi_active_pair_cd(best_h, n0, target0,
                                           time_limit=min(50, tleft() * 0.6),
                                           nc=30, top_k=10, n_shifts=3, patience=200)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    best_n = n0
    best_target = target0

    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 40:
            h1 = slsqp_optimize(h1, n1, target1, [200, 1000], 400)
            sc1 = compute_c5(h1)

        if tleft() > 60:
            h1_ns, c5_1ns = nesterov_subgrad(h1, n1, target1,
                                             time_limit=min(40, tleft() * 0.3),
                                             lr=0.002, momentum=0.92, n_active=5)
            if c5_1ns < sc1:
                sc1 = c5_1ns
                h1 = h1_ns

        if tleft() > 100:
            h1_cd, c5_1cd = fast_pair_cd(h1, n1, target1,
                                         time_limit=min(80, tleft() * 0.5),
                                         nc=35, top_k=12, patience=200)
            if c5_1cd < sc1:
                sc1 = c5_1cd
                h1 = h1_cd

        if tleft() > 60:
            h1_ma, c5_1ma = multi_active_pair_cd(h1, n1, target1,
                                                 time_limit=min(50, tleft() * 0.5),
                                                 nc=30, top_k=10, n_shifts=3, patience=200)
            if c5_1ma < sc1:
                sc1 = c5_1ma
                h1 = h1_ma

        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            best_n = n1
            best_target = target1

    if tleft() > 80:
        best_h = slsqp_optimize(best_h, best_n, best_target, [500, 2000, 8000], 800)
        best_score = compute_c5(best_h)

    if tleft() > 100:
        h_cd2, c5_cd2 = fast_pair_cd(best_h, best_n, best_target,
                                     time_limit=min(80, tleft() * 0.6),
                                     nc=50, top_k=15, patience=300)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 80:
        h_ma2, c5_ma2 = multi_active_pair_cd(best_h, best_n, best_target,
                                             time_limit=min(60, tleft() * 0.5),
                                             nc=40, top_k=12, n_shifts=4, patience=250)
        if c5_ma2 < best_score:
            best_score = c5_ma2
            best_h = h_ma2

    if tleft() > 200:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 40:
            h2 = slsqp_optimize(h2, n2, target2, [500, 2000], 500)
            sc2 = compute_c5(h2)

        if tleft() > 50:

    if tleft() > 30:
        best_h = slsqp_optimize(best_h, best_n, best_target, [5000, 20000], 600)
        best_score = compute_c5(best_h)

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
