    # h is the current binary vector
    # returns delta for all lags k from -(n-1) to (n-1)
    # represented as array of length 2n-1, index n-1+k corresponds to lag k
    delta = np.zeros(2*n - 1)
    one_minus_h = 1.0 - h
    
    # Term 1: -(1-h[a+k]) for k where 0 <= a+k < n
    k_min1 = max(-(n-1), -a)
    k_max1 = min(n-1, n-1-a)
    if k_min1 <= k_max1:
        idx = a + k_min1  # start index in h
        delta[n-1+k_min1 : n-1+k_max1+1] += one_minus_h[idx : idx + (k_max1-k_min1+1)] * (-1)
    
    # Term 2: h[a-k] for k where 0 <= a-k < n
    k_min2 = max(-(n-1), a-(n-1))
    k_max2 = min(n-1, a)
    if k_min2 <= k_max2:
        idx = a - k_min2  # when k=k_min2, a-k = a-k_min2
        # h[a-k] for k from k_min2 to k_max2 is h[a-k_min2], h[a-k_min2-1], ..., h[a-k_max2]
        # which is h[idx], h[idx-1], ..., h[idx - (k_max2-k_min2)]
        # i.e., h[idx : idx - (k_max2-k_min2) - 1 : -1] reversed
        idx_start = a - k_min2

    n = len(h)

    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = min(current_sum - target_sum, len(ones))
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = min(target_sum - current_sum, len(zeros))
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1

def compute_swap_delta(h, a, b, n):
    """Compute change in correlation vector for swap a:1->0, b:0->1."""
    omh = 1.0 - h
    delta = np.zeros(2 * n - 1)
    
    # Term 1: -(1-h[a+k]) for k where 0 <= a+k < n
    klo = max(-(n-1), -a)
    khi = min(n-1, n-1-a)
    if klo <= khi:
        s = n - 1 + klo
        ln = khi - klo + 1
        delta[s:s+ln] -= omh[a+klo : a+khi+1]
    
    # Term 2: h[a-k] for k where 0 <= a-k < n
    klo = max(-(n-1), a-n+1)
    khi = min(n-1, a)
    if klo <= khi:
        s = n - 1 + klo
        ln = khi - klo + 1
        delta[s:s+ln] += h[a-khi : a-klo+1][::-1]
    
    # Term 3: (1-h[b+k]) for k where 0 <= b+k < n
    klo = max(-(n-1), -b)
    khi = min(n-1, n-1-b)
    if klo <= khi:
        s = n - 1 + klo
        ln = khi - klo + 1
        delta[s:s+ln] += omh[b+klo : b+khi+1]
    
    # Term 4: -h[b-k] for k where 0 <= b-k < n
    klo = max(-(n-1), b-n+1)
    khi = min(n-1, b)
    if klo <= khi:
        s = n - 1 + klo
        ln = khi - klo + 1
        delta[s:s+ln] -= h[b-khi : b-klo+1][::-1]
    
    # Corrections for k = b-a and k = a-b
    delta[n-1 + (b-a)] += 1.0
    delta[n-1 + (a-b)] += 1.0
    
    # Fix k=0 (index n-1): always 0 for binary h
    delta[n-1] = 0.0
    
    return delta
