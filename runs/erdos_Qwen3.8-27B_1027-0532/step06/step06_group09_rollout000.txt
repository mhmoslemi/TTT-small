```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_func(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    sum_h = np.sum(h)
    penalty = 1e6 * (sum_h - target) ** 2
    return val + penalty

def _smooth_wrapped(h_flat, alpha, n, target):
    h = h_flat.reshape(-1)
    return smooth_obj_func(h, alpha, n, target)

def gradient_focus(h, corr, n, top_m, eta):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    c_max = np.max(corr)
    top_idx = np.argsort(corr)[-top_m:]
    weights = np.exp(eta * (corr[top_idx] - c_max) / (c_max + 1e-12))
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for w, s_idx in zip(weights, top_idx):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def focal_subgradient(h_init, n, target, time_limit, max_steps=20000,
                      eta=2.0, top_m=10, base_lr=0.04):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.7
    lr = base_lr
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break
        grad = gradient_focus(h, corr, n, top_m, eta)
        v = mu * v + grad
        h_try = project_feasible(h - lr * v, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_now - 1e-15:
            h = h_try
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 0.002:
                v = np.zeros(n)
                lr = base_lr
    return best_h, best_c5

def block_coordinate_descent(h_init, n, target, time_limit, max_sweeps=5000,
                             patience=200, block_sizes=None):
    start = time.time()
    dx = 2.0 / n
    if block_sizes is None:
        block_sizes = [1, 2, 3, 5, 8, 12, 20]
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        bs = block_sizes[sweep % len(block_sizes)]
        sorted_idx = np.argsort(grad)
        n_cand = min(40, n // 2)
        low_cand = sorted_idx[:n_cand]
        high_cand = sorted_idx[-n_cand:]
        best_improvement = 0.0
        best_move = None
        for _ in range(3):
            i = int(np.random.choice(high_cand))
            j = int(np.random.choice(low_cand))
            if i == j:
                continue
            i_lo = max(0, i - bs // 2)
            i_hi = min(n, i_lo + bs)
            j_lo = max(0, j - bs // 2)
            j_hi = min(n, j_lo + bs)
            i_block = np.arange(i_lo, i_hi)
            j_block = np.arange(j_lo, j_hi)
            delta_lo = max(h[i_block].min() - 1.0, -(1.0 - h[j_block].max()), -0.5)
            delta_hi = min(h[i_block].max(), 1.0 - h[j_block].min(), 0.5)
            if delta_hi - delta_lo < 1e-12:
                continue
            g = np.zeros(n_corr)
            h_i_sp = np.zeros(n_corr)
            h_i_sm = np.zeros(n_corr)
            for idx in i_block:
                ip = (idx + shifts >= 0) & (idx + shifts < n)
                h_i_sp[ip] += h[idx + shifts[ip]]
                im = (idx - shifts >= 0) & (idx - shifts < n)
                h_i_sm[im] += h[idx - shifts[im]]
            h_j_sp = np.zeros(n_corr)
            h_j_sm = np.zeros(n_corr)
            for idx in j_block:
                jp = (idx + shifts >= 0) & (idx + shifts < n)
                h_j_sp[jp] += h[idx + shifts[jp]]
                jm = (idx - shifts >= 0) & (idx - shifts < n)
                h_j_sm[jm] += h[idx - shifts[jm]]
            g = h_i_sp + h_i_sm - h_j_sp - h_j_sm
            a_d = max(delta_lo, -1e-10)
            b_d = min(delta_hi, 1.0)
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < c5_val - 1e-14:
                improvement = c5_val - approx_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (i_block, j_block, delta_star)
        if best_move is not None:
            i_block, j_block, delta_star = best_move
            h[i_block] -= delta_star
            h[j_block] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    h_flat = h_start.reshape(-1)
    try:
        res = minimize(_smooth_wrapped, h_flat, args=(alpha, n, target),
                       method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def slsqp_polish(h_start, n, target, alpha, maxiter=500):
    global _slq_n, _slq_target
    _slq_n = n
    _slq_target = target
    h_flat = h_start.reshape(-1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slq_obj, h_flat, args=(alpha,),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _slq_cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def _slq_obj(h_flat, alpha):
    n = _slq_n
    target = _slq_target
    h = h_flat.reshape(-1)
    return smooth_obj_func(h, alpha, n, target)

def _slq_cons(h_flat):
    return np.sum(h_flat) - _slq_target

_slq_n = 0
_slq_target = 0.0

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_search(best_h, n, target, time_limit, seed=123,
                         n_restarts=8, polish_alphas=None):
    start = time.time()
    if polish_alphas is None:
        polish_alphas = [500, 2000, 8000]
    rng = np.random.RandomState(seed)
    best = best_h.copy()
    best_c5 = compute_c5(best)
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        scale = 0.05 + 0.1 * r / max(1, n_restarts - 1)
        h_pert = best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        for alpha in polish_alphas:
            if time.time() - start > time_limit:
                break
            h_try = lbfags_polish(h_pert, n, target, alpha, maxiter=500)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best = h_try
                h_pert = h_try
    return best, best_c5
```
