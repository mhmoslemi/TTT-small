For a swap of h[i]=1 → 0 and h[j]=0 → 1, the change in corr at shift k is:

where:

For δ₁(i,k): h[i] changes from 1 to 0.
- In sum_m h[m](1-h[m+k]): the term m=i changes from 1*(1-h[i+k]) to 0, so Δ = -(1-h[i+k]) [if i+k valid]
- The term m=i-k changes from h[i-k]*(1-1) to h[i-k]*(1-0), so Δ = +h[i-k] [if i-k valid]

For δ₂(j,k): h[j] changes from 0 to 1.
- In sum_m h[m](1-h[m+k]): the term m=j changes from 0 to 1*(1-h[j+k]), so Δ = +(1-h[j+k]) [if j+k valid]
- The term m=j-k changes from h[j-k]*(1-0) to h[j-k]*(1-1), so Δ = -h[j-k] [if j-k valid]

    current_sum = int(h_r.sum())

        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):

            h_r[ones[order]] = 0
    else:

        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum

            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:

            extra = final_sum - target_sum

        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

    """Get the indices of the top T correlations and their shifts"""
    top_idx = np.argsort(full)[::-1][:T]
    return top_idx, offsets[top_idx], full[top_idx]

    """Compute delta in correlation at specific shifts for swap i->0, j->1"""
    d = np.zeros(len(shifts))

            val += h[i - k]
        if 0 <= j + k < n:
            val += (1.0 - h[j + k])
        if 0 <= j - k < n:
            val += -h[j - k]
        d[t] = val
    return d

**Projection-based approach**:

6. Project: h_new = project_box_sum(h_new, n/2)

def subgradient_refine(h0, target, start_time, time_limit, n_inner=5):
    """Projected subgradient with Nesterov momentum and multiple subgradients"""
    n = len(h0)
    h = h0.copy()

    full = full_corr_fast(h)

    best_h = h.copy()

        if time.time() - start_time > time_limit:
            break
        h = h0.copy()
        h = project_box_sum(h, target)
        full = full_corr_fast(h)
        v = np.zeros(n)
        lr = 1.0 / np.sqrt(n)
        beta = 0.5
        T = 5  # number of top shifts to use
        
        n_steps = int(500000 / n)  # target ~500k gradient evaluations
        n_steps = max(n_steps, 1000)
        
        for step in range(n_steps):
            if time.time() - start_time > time_limit:
                break
            
            # Get top T subgradients
            top_idx = np.argsort(full)[::-1][:T]
            grad = np.zeros(n)
            for ti in top_idx:
                k = offsets[ti]
                # ∂corr_k/∂h[i] = (1-h[i+k]) for i+k valid
                #                 - h[i-k] for i-k valid
                idx1 = np.arange(n) + k
                v1 = (idx1 >= 0) & (idx1 < n)
                grad[v1] += (1.0 - h[idx1[v1]])
                idx2 = np.arange(n) - k
                v2 = (idx2 >= 0) & (idx2 < n)
                grad[v2] += -h[idx2[v2]]
            
            # Nesterov update
            v = beta * v - lr * grad
            h_new = project_box_sum(h + v, target)
            h = h_new
            
            # Update full correlation periodically
            if step % 50 == 0 or step == n_steps - 1:
                full = full_corr_fast(h)
                score = float(full.max()) * dx
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            
            # Decay learning rate
            lr *= 0.999
            
            # Increase momentum
            beta = min(0.9, 0.5 + step / (2 * n_steps))
    
    return best_h, best_score
