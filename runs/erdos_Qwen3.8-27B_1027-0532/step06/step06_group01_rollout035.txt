```python
def project_feasible(h, target_sum):

    h = np.clip(h.astype(float), 0.0, 1.0)

    lo, hi = -1.0, 1.0

        else:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Water-filling projection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_try) < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        if np.any(mask):
            h_proj[mask] += residual / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def projected_gradient(h_init, n, target, alpha_schedule, time_limit,
                       method='adam', lr=0.05, max_steps=200000):
    """Projected gradient with Adam or Nesterov momentum."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    
    # Precompute validity masks (these don't change)
    pos1 = idxs[:, None].astype(float) - shifts[None, :].astype(float)
    valid1 = (pos1 >= 0) & (pos1 < n)
    pos2 = idxs[:, None].astype(float) + shifts[None, :].astype(float)
    valid2 = (pos2 >= 0) & (pos2 < n)
    
    # Integer positions for indexing
    pos1_int = (idxs[:, None] - shifts[None, :]).astype(int)
    pos2_int = (idxs[:, None] + shifts[None, :]).astype(int)
    
    if method == 'adam':
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        t_step = 0
    else:  # nesterov
        vel = np.zeros(n)
        momentum = 0.9
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        # Get current alpha
        frac = min(step / max(max_steps - 1, 1), 1.0)
        alpha = alpha_schedule[frac] if hasattr(alpha_schedule, '__call__') else alpha_schedule[min(int(frac * len(alpha_schedule)), len(alpha_schedule)-1)]
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        
        # Softmax weights
        shifted = alpha * (corr - cmax)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / wsum
        
        # Compute gradient
        # grad[j] = sum_m weights[m] * [(1-h[pos1[j,m]]) * valid1[j,m] - h[pos2[j,m]] * valid2[j,m]]
        # Using precomputed positions:
        temp1 = np.zeros((n, n_corr))
        temp1[valid1] = 1.0 - h[pos1_int[valid1]]
        temp2 = np.zeros((n, n_corr))
        temp2[valid2] = h[pos2_int[valid2]]
        
        grad = (temp1 @ weights) - (temp2 @ weights)
        grad *= dx
        
        if method == 'adam':
            t_step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        else:
            vel = momentum * vel - lr * grad
            h_new = h + vel
        
        h_new = project_feasible(h_new, target)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Early stopping
        if step > 1000 and step % 1000 == 0:
            # Check if we're making progress
            pass
    
    return best_h, best_c5
```

```python
def compute_grad_fft(h, weights, n):
    """Compute gradient using FFT. O(n log n) instead of O(n^2)."""
    n_corr = 2 * n - 1
    L = 1 << (4 * n - 1).bit_length()  # next power of 2 >= 4n
    
    # Signal A = 1-h (length n, zero-padded to L)
    A = np.zeros(L)
    A[:n] = 1.0 - h
    
    # Signal B: B[i] = weights[i - (n-1)] for i = n-1, ..., 3n-3
    B = np.zeros(L)
    B[n-1:3*n-2] = weights  # weights has length 2n-1 = 3n-2-(n-1)
    
    # Cross-correlation: (A ⋆ B)[k] = IFFT(FFT(A) * conj(FFT(B)))[k]
    FA = np.fft.rfft(A)
    FB = np.fft.rfft(B)
    cross = np.fft.irfft(FA * np.conj(FB), L)
    
    # grad_first[j] = cross[n-1-j] for j=0,...,n-1
    grad_first = cross[n-1::-1]  # cross[n-1], cross[n-2], ..., cross[0]
    
    # Second term: replace 1-h with h
    A2 = np.zeros(L)
    A2[:n] = h
    FA2 = np.fft.rfft(A2)
    cross2 = np.fft.irfft(FA2 * np.conj(FB), L)
    grad_second = cross2[n-1::-1]
    
    grad = grad_first - grad_second
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(h_proj)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        if np.any(mask):
            h_proj[mask] += residual / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
def projected_gradient_adam(h_init, n, target, alpha, time_limit, lr=0.1, max_steps=100000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    
    # Precompute valid positions
    pos1 = idxs[:, None] - shifts[None, :]
    valid1 = (pos1 >= 0) & (pos1 < n)
    pos2 = idxs[:, None] + shifts[None, :]
    valid2 = (pos2 >= 0) & (pos2 < n)
    
    r1, c1 = np.where(valid1)
    r2, c2 = np.where(valid2)
    lookup1 = pos1[r1, c1]
    lookup2 = pos2[r2, c2]
    
    # Preallocate
    M1 = np.zeros((n, n_corr))
    M2 = np.zeros((n, n_corr))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        
        # Softmax weights
        shifted = alpha * (corr - cmax)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= ws
        
        # Gradient
        M1[r1, c1] = 1.0 - h[lookup1]
        M2[r2, c2] = h[lookup2]
        grad = (M1 @ weights) - (M2 @ weights)
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve > 20000:
            break
    
    return best_h, best_c5
```

```python
def pair_cd_polish(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Find the worst shifts (top few)
        top_k = 3
        top_indices = np.argsort(corr)[-top_k:]
        
        # Compute gradient from top shifts
        weights_top = np.zeros(n_corr)
        for idx in top_indices:
            weights_top[idx] = 1.0 / top_k  # uniform weight
        
        # Actually, use softmax over top shifts
        top_corrs = corr[top_indices]
        w_exp = np.exp(100 * (top_corrs - top_corrs.max()))
        w_exp /= w_exp.sum()
        for idx, w in zip(top_indices, w_exp):
            weights_top[idx] = w
        
        grad = np.zeros(n)
        for m in top_indices:
            wm = weights_top[m]
            if wm < 1e-10:
                continue
            s = int(m) - (n - 1)
            v1 = (idxs - s >= 0) & (idxs - s < n)
            v2 = (idxs + s >= 0) & (idxs + s < n)
            if np.any(v1):
                grad[v1] += wm * (1.0 - h[idxs[v1] - s])
            if np.any(v2):
                grad[v2] -= wm * h[idxs[v2] + s)
        
        # Find candidates
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        # Precompute profiles
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m_idx = int(all_cands[ci])
            ks = np.arange(-(n-1), n)
            idx_a = m_idx - ks
            idx_b = m_idx + ks
            va = (idx_a >= 0) & (idx_a < n)
            vb = (idx_b >= 0) & (idx_b < n)
            if np.any(va):
                profiles[ci, va] += h[idx_a[va]]
            if np.any(vb):
                profiles[ci, vb] += h[idx_b[vb]]
        
        # Try all pairs
        best_improve = 0.0
        best_i, best_j, best_delta = -1, -1, 0.0
        
        for i in high_set:
            i_val = float(h[i])
            if i_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                j_val = float(h[j])
                delta_max = min(i_val, 1.0 - j_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Bisection for optimal delta
                lo_d, hi_d = 0.0, delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo_d + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo_d = mid_d
                
                delta_star = 0.5 * (lo_d + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i, best_j, best_delta = int(i), int(j), delta_star
        
        if best_i >= 0 and best_improve > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5
```

The output is indexed from 0 to 4, corresponding to k = -2, -1, 0, 1, 2.

corr_out[m] = sum_n h[n + (m - 2)] * (1-h[n]) for valid n (0 <= n < 3 and 0 <= n+m-2 < 3)

For m=0 (k=-2): n+m-2 = n-2, valid when 0<=n-2<3, i.e., 2<=n<5, but n<3, so n=2.
corr[0] = h[2] * (1-h[2])

For m=1 (k=-1): n+m-2 = n-1, valid when 0<=n-1<3, i.e., 1<=n<4, but n<3, so n=1,2.
corr[1] = h[1]*(1-h[1]) + h[2]*(1-h[2])

For m=2 (k=0): n+m-2 = n, valid when 0<=n<3, n=0,1,2.
corr[2] = h[0]*(1-h[0])
