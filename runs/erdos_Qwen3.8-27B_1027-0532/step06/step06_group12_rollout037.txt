import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    """Project onto {sum(h) = target_sum, 0 <= h <= 1}"""
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the water-filling
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    projected = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    return projected

def smooth_objective_and_grad(h, alpha):
    """Compute log-sum-exp smoothing of max overlap and its gradient."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    # Shift for numerical stability
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = (max_ov + log_sum / alpha)
    
    # Gradient: p_k = exp(alpha * (g_k - max)) / sum(exp(alpha * (g_j - max)))
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of g_k with respect to h
    # g_k = dx * correlate(h, 1-h)[k + n - 1]
    # For the full correlation, we need the gradient
    # Let's compute it efficiently
    
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    active = np.where(p > 1e-6)[0]
    n_half = n - 1
    for idx in active:
        k = idx - n_half
        abs_k = abs(k)
        w = p[idx] * dx
        if abs_k == 0:
            # g_0 = dx * sum(h[i] * (1-h[i]))
            # d/dh[i] = dx * (1 - 2*h[i])
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            # g_k = dx * sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # d/dh[i] for i in [0, n-1-k]: dx * (1-h[i+k])
            # d/dh[i] for i in [k, n-1]: dx * (-h[i-k])
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            m = -k
            # g_{-m} = dx * sum_{i=m}^{n-1} h[i] * (1-h[i-m])
            # d/dh[i] for i in [m, n-1]: dx * (1-h[i-m])
            # d/dh[i] for i in [0, n-1-m]: dx * (-h[i+m])
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] += w * (-h[m:])
    
    return val, grad

def proximal_gradient(h_init, n, target, alphas, time_limit, max_iter_per_alpha=5000):
    """Proximal gradient with Polyak step size for smoothed objective."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        # Get lower bound (use current best C5 as proxy)
        f_star = best_c5 * 0.99  # Lower bound approximation
        
        for it in range(max_iter_per_alpha):
            if time.time() - start > time_limit:
                break
            
            val, grad = smooth_objective_and_grad(h, alpha)
            grad_norm = np.linalg.norm(grad)
            
            if grad_norm < 1e-12:
                break
            
            # Polyak step size
            step = (val - f_star) / (grad_norm ** 2)
            step = min(step, 1.0 / alpha)  # Cap step size
            
            # Gradient descent step
            y = h - step * grad
            h_new = project_feasible(y, target)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new
            
            # Check convergence
            if it > 100 and abs(val - f_star) < 1e-10:
                break
    
    return best_h, best_c5

**Alternative: Using the KKT conditions directly**

At the optimum, the KKT conditions say:
- There exist multipliers λ_k ≥ 0 for each lag constraint g_k(h) ≤ t
- There exist μ_i ≥ 0 for h_i ≥ 0 and ν_i ≥ 0 for h_i ≤ 1
- There exists a ν_0 for the equality constraint sum(h) = 200
- Stationarity: ∇_h L = 0 where L = t + Σ λ_k(g_k(h) - t) - Σ μ_i h_i + Σ ν_i(h_i - 1) + ν_0(sum(h) - 200)

At the optimum:
- For active lags (g_k(h) = t): λ_k > 0
- For h_i = 0: μ_i > 0, ν_i = 0
- For h_i = 1: μ_i = 0, ν_i > 0
- For 0 < h_i < 1: μ_i = 0, ν_i = 0

The stationarity condition becomes:
For 0 < h_i < 1: Σ_{k: g_k = t} λ_k * ∂g_k/∂h_i + ν_0 = 0

This is a system of equations. If we know which lags are active and which coordinates are in (0,1), we can solve for the multipliers and the values of h_i.

1. Guess the active set of lags A and the free set of coordinates F
2. Solve the KKT system
3. Check complementarity conditions
4. Update the active/free sets
5. Iterate

    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Identify active lags (those at or near the max)
    for outer in range(50):
        if time.time() - start > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        t_val = np.max(overlaps)
        
        # Active lags: those within tolerance of max
        tol = 1e-4
        active = np.where(overlaps > t_val - tol)[0]
        n_half = n - 1
        active_lags = active - n_half
        
        # Free coordinates: those in (0.01, 0.99)
        free = np.where((h > 0.01) & (h < 0.99))[0]
        
        if len(active) == 0 or len(free) == 0:
            break
        
        # Set up KKT system
        # For free i: Σ_{k in active} λ_k * ∂g_k/∂h_i = -ν_0
        # For all active k: g_k(h) = t
        # sum(h) = target
        # 0 <= h <= 1
        # λ_k >= 0
        
        # This is a large system. Let's use a simpler approach:
        # Minimize t subject to g_k(h) <= t for active k, using the current h as warm start
        
        # Actually, let's just do a local QP
        # Linearize g_k around current h:
        # g_k(h + δ) ≈ g_k(h) + ∇g_k(h)^T δ
        # Minimize t s.t. t >= g_k(h) + ∇g_k(h)^T δ for active k
        #                sum(δ) = 0 (to maintain sum constraint)
        #                -h_i <= δ_i <= 1 - h_i
        
        # This is an LP!
        m = len(active)
        p = len(free)
        # Variables: δ_free (free coordinate changes), t
        # We fix non-free coordinates
        
        # Build the LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        n_half = n - 1
        dx = 2.0 / n
        for idx in active:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - 2.0 * h) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n-k] = (1.0 - h[k:]) * dx
                coeff[k:] = (-h[:n-k]) * dx
            else:
                m2 = -k
                coeff = np.zeros(n)
                coeff[m2:] = (1.0 - h[:n-m2]) * dx
                coeff[:n-m2] = (-h[m2:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        
        # Add regularization to stay close to current
        reg = 1e-4 * cp.sum_squares(h_var - h)
        objective = t_var + reg
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
        except:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0, 1)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:

            best_h = h_new.copy()
            h = h_new
        else:
            # Try with tighter tolerance
            tol *= 0.5
            if tol < 1e-8:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    # Water-filling projection
    for _ in range(80):
        projected = np.clip(h, 0.0, 1.0)
        residual = target_sum - np.sum(projected)
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            idx = np.where(projected < 1.0 - 1e-14)[0]
        else:
            idx = np.where(projected > 1e-14)[0]
        if len(idx) == 0:
            break
        h[idx] += residual / len(idx)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            # Distribute proportionally to available space
            if residual > 0:
                space = 1.0 - h[idx]
            else:
                space = h[idx]
            total_space = np.sum(space)
            if total_space > 0:
                h[idx] += residual * space / total_space
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(h, alpha, target, mu):
    """Log-sum-exp smoothed objective with sum penalty."""
    n = len(h)
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - max_ov), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    val = max_ov + log_sum / alpha
    
    # Sum penalty
    sum_h = np.sum(h)
    val += mu * (sum_h - target) ** 2
    
    # Gradient of LSE part
    p = exp_shifted / np.sum(exp_shifted)
    n_half = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Only process significant weights for efficiency
    thresh = 1.0 / (n * 10)
    active = np.where(p > thresh)[0]
    
    for idx in active:
        k = idx - n_half
        w = p[idx] * dx
        abs_k = abs(k)
        if abs_k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] += w * (-h[m:])
    
    # Gradient of sum penalty
    grad += 2.0 * mu * (sum_h - target)
    
    return val, grad

def lse_lbfgs(h_init, n, target, alphas, time_limit, mu_start=1.0, mu_factor=10.0):
    """Run L-BFGS-B with increasing alpha and mu."""
    start_time = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    mu = mu_start
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        # Define objective and gradient for scipy
        def obj(x):
            v, _ = lse_objective_and_grad(x, alpha, target, mu)
            return v
        
        def jac(x):
            _, g = lse_objective_and_grad(x, alpha, target, mu)
            return g
        
        bounds = [(0.0, 1.0)] * n
        
        try:
            result = minimize(obj, h, jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(result.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        
        mu *= mu_factor
    
    return best_h, best_c5

def active_lp(h, n, target, n_top, time_limit, reg=1e-6):
    """LP with top-n_top active constraints."""
    start_time = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h_current = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(200):
        if time.time() - start_time > time_limit:
            break
        
        overlaps = compute_overlaps(h_current)
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - 2.0 * h_current) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n-k] = (1.0 - h_current[k:]) * dx
                coeff[k:] = (-h_current[:n-k]) * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h_current[:n-m]) * dx
                coeff[:n-m] = (-h_current[m:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h_current)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
        except:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_current = h_new
        else:
            # Decay regularization and continue
            reg *= 0.5
            if reg < 1e-10:
                break
    
    return best_h, best_c5

def coordinate_descent(h_init, n, target, time_limit, max_rounds=1000):
    """Pairwise coordinate descent: try (increase j, decrease j') moves."""
    start_time = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Precompute which pairs to try based on gradient
    for round_num in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        n_half = n - 1
        # Find the worst lag
        worst_idx = np.argmax(overlaps)
        k = worst_idx - n_half
        
        # Compute gradient of the worst overlap
        dx = 2.0 / n
        grad = np.zeros(n)
        if abs(k) == 0:
            grad = (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[:n-k] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:n-k]) * dx
        else:
            m = -k
            grad[m:] += (1.0 - h[:n-m]) * dx
            grad[:n-m] += (-h[m:]) * dx
        
        # Best increase: coordinate with most negative gradient (decreasing h there helps)
        # Best decrease: coordinate with most positive gradient (increasing h there hurts)
        # We want to decrease h where grad is positive, increase where grad is negative
        
        # Try: increase h[argmin grad] and decrease h[argmax grad]
        j_inc = np.argmin(grad)  # Most negative grad → increasing helps
        j_dec = np.argmax(grad)  # Most positive grad → decreasing helps
        
        if j_inc == j_dec:
            break
        
        # Line search
        best_delta = 0
        best_val = best_c5
        for trial in range(10):
            delta = (trial + 1) * 0.01
            if h[j_inc] + delta > 1.0 or h[j_dec] - delta < 0.0:
                break
            h_trial = h.copy()
            h_trial[j_inc] += delta
            h_trial[j_dec] -= delta
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_val:
                best_val = c5_trial
                best_delta = delta
                best_h_trial = h_trial.copy()
        
        if best_delta > 0:
            h = best_h_trial
            if best_val < best_c5 - 1e-14:
                best_c5 = best_val
                best_h = h.copy()
        else:
            # Try other pairs
            # Second worst
            j_inc_candidates = np.argsort(grad)[:5]
            j_dec_candidates = np.argsort(grad)[::-1][:5]
            improved = False
            for ji in j_inc_candidates:
                for jd in j_dec_candidates:
                    if ji == jd:
                        continue
                    delta = 0.005
                    if h[ji] + delta > 1.0 or h[jd] - delta < 0.0:
                        continue
                    h_trial = h.copy()
                    h_trial[ji] += delta
                    h_trial[jd] -= delta
                    h_trial = project_feasible(h_trial, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-14:
                        h = h_trial
                        best_c5 = c5_trial
                        best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved and round_num > 50:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        residual = target_sum - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += residual / len(idx)
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            if residual > 0:
                space = 1.0 - h[idx]
            else:
                space = h[idx]
            total = np.sum(space)
            if total > 1e-14:
                h[idx] += residual * space / total
    return np.clip(h, 0.0, 1.0)


def lse_val_grad(h, alpha, target, mu):
    n = len(h)
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - max_ov), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = max_ov + np.log(sum_exp) / alpha
    sum_h = np.sum(h)
    val += mu * (sum_h - target) ** 2
    p = exp_s / sum_exp
    n_half = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    thresh = 1.0 / (2.0 * n)
    active = np.where(p > thresh)[0]
    for idx in active:
        k = idx - n_half
        w = p[idx] * dx
        abs_k = abs(k)
        if abs_k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n - k])
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] += w * (-h[m:])
    grad += 2.0 * mu * (sum_h - target)
    return val, grad


def lse_obj_func(x, alpha, target, mu):
    v, _ = lse_val_grad(x, alpha, target, mu)
    return v


def lse_jac_func(x, alpha, target, mu):
    _, g = lse_val_grad(x, alpha, target, mu)
    return g


def run_lbfgs_lse(h_init, n, target, alphas, time_limit, mu_start=1.0, mu_factor=5.0):
    start_t = time.time()
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = mu_start
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(lse_obj_func, h, args=(alpha, target, mu),
                           jac=lse_jac_func, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 80})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        mu *= mu_factor
    return best_h, best_c5


def run_active_lp(h_init, n, target, n_top, time_limit, reg_init=1e-4):
    start_t = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    reg = reg_init
    for it in range(300):
        if time.time() - start_t > time_limit:
            break
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - 2.0 * h) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = (1.0 - h[k:]) * dx
                coeff[k:] = (-h[:n - k]) * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n - m]) * dx
                coeff[:n - m] = (-h[m:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-11, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            reg *= 0.5
            if reg < 1e-12:
                break
    return best_h, best_c5


def run_coord_descent(h_init, n, target, time_limit, max_rounds=20000):
    start_t = time.time()
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_half = n - 1
    stuck = 0
    for rnd in range(max_rounds):
        if time.time() - start_t > time_limit:
            break
        if stuck > 200:
            break
        overlaps = compute_overlaps(h)
        worst_idx = np.argmax(overlaps)
        k = worst_idx - n_half
        abs_k = abs(k)
        if abs_k == 0:
            grad = (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad = np.zeros(n)
            grad[:n - k] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:n - k]) * dx
        else:
            m = -k
            grad = np.zeros(n)
            grad[m:] += (1.0 - h[:n - m]) * dx
            grad[:n - m] += (-h[m:]) * dx
        inc_candidates = np.argsort(grad)[:8]
        dec_candidates = np.argsort(grad)[::-1][:8]
        improved = False
        for ji in inc_candidates:
            for jd in dec_candidates:
                if ji == jd:
                    continue
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    if h[ji] + delta > 1.0 or h[jd] - delta < 0.0:
                        continue
                    h_trial = h.copy()
                    h_trial[ji] += delta
                    h_trial[jd] -= delta
                    h_trial = project_feasible(h_trial, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-14:
                        h = h_trial
                        best_c5 = c5_trial
                        best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5
