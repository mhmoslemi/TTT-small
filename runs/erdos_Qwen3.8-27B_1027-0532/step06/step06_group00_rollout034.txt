import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return float(np.sum(x) - _g_target)

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                    n_cand_start=30, n_cand_end=100, 
                    n_top_j=15, max_no_improve=500):

    Improved pair CD with smarter candidate selection.
    For each i, only try the top n_top_j candidates for j (ranked by change[j][k*]).
    """
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        # Multi-shift gradient
        top_k = min(5, n_corr)
        top_shifts = np.argsort(corr)[-top_k:]
        weights = corr[top_shifts]
        w_sum = weights.sum()
        if w_sum < 1e-12:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        # Pre-rank low_set by change[j][k_star] (we want most negative = most beneficial)
        low_change_kstar = change[low_set, k_star_idx]
        low_rank_order = np.argsort(low_change_kstar)  # most negative first
        # Also rank by weighted sum over top shifts
        low_change_weighted = np.zeros(len(low_set))
        for s_idx in top_shifts:
            low_change_weighted += corr[s_idx] / max(w_sum, 1e-15) * change[low_set, s_idx]
        low_rank_w = np.argsort(low_change_weighted)
        
        # Combine: top n_top_j from each ranking
        j_candidates = list(set(
            [int(low_set[low_rank_order[x]]) for x in range(min(n_top_j, len(low_rank_order)))] +
            [int(low_set[low_rank_w[x]]) for x in range(min(n_top_j, len(low_rank_w)))]
        ))
        j_candidates = j_candidates[:2 * n_top_j]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            g_i = change[i]
            for j in j_candidates:
                if j == i:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - g_i

                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Fix sum
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            # No improvement found, try a perturbation
            if consecutive_no_improve % 200 == 0 and consecutive_no_improve > 50:
                rng = np.random.RandomState(consecutive_no_improve)

                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return float(np.sum(x) - _g_target)

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def cd_smart(h_init, n, target, time_limit, max_sweeps=100000,
             ncand_start=30, ncand_end=100, ntopj_start=10, ntopj_end=30,
             max_no_improve=600, pert_density=0.04):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    rows = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))

        top_k = min(5, n_corr)
        top_shifts = np.argsort(corr)[-top_k:]
        w = corr[top_shifts].copy()
        w_sum = w.sum()
        if w_sum > 1e-12:
            w = w / w_sum
        else:
            w = np.ones(top_k) / top_k

        grad = np.zeros(n)
        for si, wi in zip(top_shifts, w):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if v1.any():
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if v2.any():
                grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        cur_ncand = int(ncand_start + (ncand_end - ncand_start) * progress)
        cur_ncand = min(cur_ncand, n // 3)
        cur_ntopj = int(ntopj_start + (ntopj_end - ntopj_start) * progress)
        cur_ntopj = min(cur_ntopj, cur_ncand)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_ncand]
        high_set = sorted_idx[-cur_ncand:]

        low_ck = change[low_set, k_star_idx]
        rank1 = np.argsort(low_ck)
        low_cw = np.zeros(len(low_set))
        for si in top_shifts:
            low_cw += w[list(top_shifts).index(si)] * change[low_set, si]
        rank2 = np.argsort(low_cw)

        j_set = list(set(
            [int(low_set[rank1[x]]) for x in range(min(cur_ntopj, len(rank1)))] +
            [int(low_set[rank2[x]]) for x in range(min(cur_ntopj, len(rank2)))]
        ))
        j_set = j_set[:2 * cur_ntopj]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            g_i = change[i]
            for j in j_set:
                if j == i:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - g_i
                a_d = d_lo
                b_d = d_hi
                for _ in range(14):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = np.max(g[active])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-13:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial_c5:
                    best_trial_c5 = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star

        if best_i >= 0:
            d = best_delta
            corr += d * (change[best_j] - change[best_i])
            h[best_i] -= d
            h[best_j] += d
            if h[best_i] < 0.0:
                h[best_i] = 0.0
            if h[best_j] > 1.0:
                h[best_j] = 1.0
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            s1 = (best_i - rows) + (n - 1)
            s2 = (rows - best_i) + (n - 1)
            s3 = (best_j - rows) + (n - 1)
            s4 = (rows - best_j) + (n - 1)
            change[rows, s1] += d
            change[rows, s2] += d
            change[rows, s3] -= d
            change[rows, s4] -= d
        else:
            if consecutive_no_improve > 30 and consecutive_no_improve % 150 == 0:
                rng = np.random.RandomState(consecutive_no_improve * 7 + 13)
                n_pert = max(3, int(n * pert_density))
                pidx = rng.choice(n, size=n_pert, replace=False)
                h[pidx] += rng.randn(n_pert) * 0.06
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full').copy()

    return best_h, best_c5

def stochastic_refine(h_init, n, target, time_limit, max_steps=20000, base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(99999)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break
        scale = base_scale * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        pidx = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[pidx] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 150:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

                               max_sweeps=50000,

                               max_no_improve=400)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 100:
        h_sl, c5_sl = stochastic_refine(best_h, n1, target1,
                                        time_limit=min(60, tleft() * 0.4),
                                        max_steps=15000, base_scale=0.03)
        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for ns in [0.04, 0.08, 0.12]:
            if tleft() < 60:
                break
            h_p = best_h + rng.randn(n1) * ns
            h_p = project_feasible(h_p, target1)
            h_r = slsqp_round(h_p, n1, target1, [200, 2000, 10000, 40000], 500, 1e-14)
            sc = compute_c5(h_r)
            if sc < best_score:
                best_score = sc
                best_h = h_r
                if tleft() > 60:
                    h_cdr, c5r = cd_smart(best_h, n1, target1,
                                          time_limit=min(50, tleft() * 0.5),
                                          max_sweeps=50000,
                                          ncand_start=40, ncand_end=90,
                                          ntopj_start=12, ntopj_end=28,
                                          max_no_improve=300)
                    if c5r < best_score:
                        best_score = c5r
                        best_h = h_cdr

    if tleft() > 80:
        h_cd3, c5_cd3 = cd_smart(best_h, n1, target1,
                                 time_limit=min(60, tleft() * 0.6),
                                 max_sweeps=50000,
                                 ncand_start=60, ncand_end=120,
                                 ntopj_start=20, ntopj_end=40,
                                 max_no_improve=300, pert_density=0.02)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 50:
        h_fpol = slsqp_round(best_h, n1, target1, [20000, 80000, 200000], 600, 1e-14)
        sc = compute_c5(h_fpol)
        if sc < best_score:
            best_score = sc
            best_h = h_fpol

    if tleft() > 40:
        h_cdf, c5f = cd_smart(best_h, n1, target1,
                              time_limit=min(30, tleft() * 0.7),
                              max_sweeps=30000,
                              ncand_start=70, ncand_end=120,
                              ntopj_start=25, ntopj_end=45,
                              max_no_improve=200, pert_density=0.02)
        if c5f < best_score:
            best_score = c5f
            best_h = h_cdf

    final_h = project_feasible(best_h, target1)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
