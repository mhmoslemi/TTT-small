    dx = 2.0 / n
    corr = fft_correlate_full(h, 1.0 - h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def greedy_cd(h0, n, target, time_limit, max_sweeps=10000, n_cand=25, bisect_iters=35):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = fft_correlate_full(h, 1.0 - h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(bisect_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

        if best_move is not None:

            h = np.clip(h, 0.0, 1.0)

def perturbation_restart(h0, n, target, time_limit, start_time, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_disturb = max(5, n // 20)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        h_try = h0.copy()
        perm = rng.permutation(n)
        disturb_idx = perm[:n_disturb]
        noise = rng.uniform(-0.05, 0.05, n_disturb)
        h_try[disturb_idx] += noise
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 + 0.01:
            h_p, s_p = slsqp_phase(h_try, n, target,
                                    [100, 500, 2000, 8000],
                                    1500, 1e-14,
                                    time_limit * 0.3, start_time)
            if s_p < best_c5:
                best_c5 = s_p
                best_h = h_p.copy()
            if time.time() - start_time > time_limit:
                break
            h_cd, c5_cd = greedy_cd(h_p, n, target,
                                     time_limit=min(60.0, (time_limit - (time.time() - start_time))),
                                     max_sweeps=5000, n_cand=30)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active=20, max_rounds=40):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = fft_correlate_full(h, 1.0 - h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            for mix in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    break
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    def tleft():
        return deadline - time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)

    if tleft() > 200:

                                [50, 200, 800, 3000, 10000, 30000],
                                3000, 1e-14, tleft() * 0.35, start_time)

                                2500, 1e-15, tleft() * 0.3, start_time)
        if s_p < best_score:

            best_h = h_pr

    if tleft() > 100:

                                 time_limit=min(100.0, tleft() * 0.3),
                                 n_active=20, max_rounds=35)

            best_h = h_lp

    if tleft() > 120:

        h_cd, c5_cd = greedy_cd(best_h, n, target,
                                 time_limit=cd_time, max_sweeps=25000, n_cand=50, bisect_iters=45)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_tc, s_tc = trust_constr_phase(best_h, n, target,
                                         [2000, 10000, 50000, 200000],
                                         tleft() * 0.3, start_time)
        if s_tc < best_score:
            best_score = s_tc
            best_h = h_tc

    if tleft() > 80:
        h_p, s_p = slsqp_phase(best_h, n, target,
                                [2000, 10000, 50000, 200000],
                                2000, 1e-15, tleft() * 0.35, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    if tleft() > 60:

                                 time_limit=min(60.0, tleft() * 0.4),
                                 max_sweeps=20000, n_cand=60, bisect_iters=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

        for n_try in [800, 1200]:
            if tleft() < 100:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)

            h_cd2, c5_cd2 = greedy_cd(best_h, len(best_h), float(len(best_h)) / 2.0,
                                       time_limit=min(40.0, tleft() * 0.35),
                                       max_sweeps=10000, n_cand=30, bisect_iters=40)
            if c5_cd2 < best_score:

    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
