        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-10:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            if y[idx] > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= y[idx]
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def greedy_ls(h, n, max_iters, time_limit, start_time, rng, topk=25):
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    s_values = np.arange(-(n - 1), n, dtype=np.int64)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break

        max_val = float(corr.max())
        top3 = np.argsort(corr)[-3:]
        found = False
        for k_idx in top3:
            k_star = int(k_idx) - (n - 1)
            if k_star == 0:
                continue
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                return best_h, best_c5
            score_i = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            score_i[valid] -= h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            score_i[valid] -= h[idx[valid]]
            score_j = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            score_j[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            score_j[valid] += h[idx[valid]]
            ci = ones[np.argsort(-score_i[ones])[:min(topk, len(ones))]]
            cj = zeros[np.argsort(-score_j[zeros])[:min(topk, len(zeros))]]
            A = np.zeros((len(ci), 2 * n - 1))
            for p, ii in enumerate(ci):
                idx_p = ii + s_values
                mask = (idx_p >= 0) & (idx_p < n)
                A[p, mask] += h[idx_p[mask]]
                idx_m = ii - s_values
                mask = (idx_m >= 0) & (idx_m < n)
                A[p, mask] += h[idx_m[mask]]
            B = np.zeros((len(cj), 2 * n - 1))
            for q, jj in enumerate(cj):
                idx_p = jj + s_values
                mask = (idx_p >= 0) & (idx_p < n)
                B[q, mask] += h[idx_p[mask]]
                idx_m = jj - s_values
                mask = (idx_m >= 0) & (idx_m < n)
                B[q, mask] += h[idx_m[mask]]
            C = corr[None, None, :] + A[:, None, :] - B[None, :, :]
            for p, ii in enumerate(ci):
                for q, jj in enumerate(cj):
                    s_ij = ii - jj
                    if -(n - 1) <= s_ij <= n - 1:
                        C[p, q, s_ij + n - 1] += 2.0
                    s_ji = jj - ii
                    if -(n - 1) <= s_ji <= n - 1:
                        C[p, q, s_ji + n - 1] += 1.0
            new_maxes = C.max(axis=2)
            flat_idx = int(new_maxes.argmin())
            p_best, q_best = np.unravel_index(flat_idx, new_maxes.shape)
            if new_maxes[p_best, q_best] < max_val - 1e-12:
                ii = ci[p_best]
                jj = cj[q_best]
                delta = compute_delta(h, ii, jj, n)
                h[ii] = 0.0
                h[jj] = 1.0
                corr += delta
                stall = 0
                new_c5 = float(corr.max()) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                found = True
                break
        if not found:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            found_r = False
            for _ in range(15):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, ii, jj, n)
                if float((corr + delta).max()) < max_val - 1e-12:
                    h[ii] = 0.0
                    h[jj] = 1.0
                    corr += delta
                    stall = 0
                    new_c5 = float(corr.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found_r = True
                    break
            if not found_r:
                stall += 1
    return best_h, best_c5

def ils_search(h, n, max_rounds, time_limit, start_time, rng):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        h_gs, c5_gs = greedy_ls(h, n, 3000, time_limit, start_time, rng, topk=25)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()
        h = h_gs.copy()
        n_pert = rng.randint(8, 18)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            ii = ones[rng.randint(len(ones))]
            jj = zeros[rng.randint(len(zeros))]
            h[ii] = 0.0
            h[jj] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    return best_h, best_c5

def sa_search(h, n, max_iters, t_init, t_final, time_limit, start_time, rng,
             focused_frac=0.5, topk_candidates=15):
    h = h.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 80000:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        mode = rng.random()
        if mode < focused_frac:
            top3_idx = np.argsort(corr)[-3:]
            k_pick = int(top3_idx[rng.randint(len(top3_idx))])
            k_star = k_pick - (n - 1)
            if k_star != 0:
                score_i = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                score_i[valid] -= h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                score_i[valid] -= h[idx[valid]]
                ci = ones[np.argsort(-score_i[ones])[:min(topk_candidates, len(ones))]]
                ii = ci[rng.randint(len(ci))]
                score_j = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                score_j[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                score_j[valid] += h[idx[valid]]
                cj = zeros[np.argsort(-score_j[zeros])[:min(topk_candidates, len(zeros))]]
                jj = cj[rng.randint(len(cj))]
            else:
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
        else:
            ii = ones[rng.randint(len(ones))]
            jj = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, ii, jj, n)
        corr_new = corr + delta
        new_max = float(corr_new.max())
        c5_new = new_max * dx
        dc5 = c5_new - current_c5
        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[ii] = 0.0
            h[jj] = 1.0
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def dual_ascent(h, n, max_iters, time_limit, start_time, alpha_init=20.0, alpha_growth=1.05,
                lr_init=0.02, lr_end=0.001):
    h = h.copy().astype(float)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    shifts = np.arange(2 * n - 1) - (n - 1)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 5000:
            break
        alpha = alpha_init * (alpha_growth ** it)
        ratio = it / max(max_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        max_corr = float(corr.max())
        shifted = np.clip(alpha * (corr - max_corr), -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = W[k]
            if wk < 1e-10:
                continue
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] -= wk * h[:n - s]
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - h[:n - abs_s])
                grad[:n - abs_s] -= wk * h[abs_s:]
        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def topk_grad(h, n, max_iters, time_limit, start_time, topk=5, lr_init=0.04, lr_end=0.0005):
    h = h.copy().astype(float)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)
    shifts = np.arange(2 * n - 1) - (n - 1)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 5000:
            break
        ratio = it / max(max_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += (-h[abs_s:])
        grad /= topk
        h = h - lr * grad
        h = project_box_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = _slsqp_step(h_cur, n, target, alpha, maxiter, ftol)
            h_new = project_box_sum(res, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def _slsqp_step(h_cur, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    c_max0 = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full')))

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.log(np.sum(np.exp(alpha * (corr - c_max0))))) / alpha + c_max0

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    result = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return result.x

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)
    best_cont_h = project_feasible(h_init, float(n) / 2.0)
    best_cont_score = compute_c5(best_cont_h)

    # Phase 1: Quick greedy
    allowed = min(40.0, tleft() * 0.08)
    if allowed > 10:
        h_g, c5_g = greedy_ls(best_h, n, 5000, allowed, start_time, rng, topk=25)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: ILS (main phase)
    allowed = min(500.0, tleft() * 0.40)
    if allowed > 50:
        h_ils, c5_ils = ils_search(best_h, n, 100, allowed, start_time, rng)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

    # Phase 3: SA with focused moves
    allowed = min(300.0, tleft() * 0.25)
    if allowed > 30:
        n_sa = int(min(8000000, allowed * 10000))
        h_sa, c5_sa = sa_search(best_h, n, n_sa, 0.015, 0.00003, allowed, start_time, rng,
                                focused_frac=0.6, topk_candidates=12)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Continuous dual ascent
    allowed = min(180.0, tleft() * 0.15)
    if allowed > 30:
        n_iters = int(min(2000000, allowed * 20000))
        h_da, c5_da = dual_ascent(best_h, n, n_iters, allowed, start_time,
                                  alpha_init=20.0, alpha_growth=1.04,
                                  lr_init=0.015, lr_end=0.0003)
        if c5_da < best_cont_score:
            best_cont_score = c5_da
            best_cont_h = h_da.copy()

    # Phase 5: Top-k gradient on continuous
    allowed = min(120.0, tleft() * 0.12)
    if allowed > 20:
        n_iters = int(min(2000000, allowed * 25000))
        h_tk, c5_tk = topk_grad(best_cont_h, n, n_iters, allowed, start_time,
                                topk=4, lr_init=0.03, lr_end=0.0003)
        if c5_tk < best_cont_score:
            best_cont_score = c5_tk
            best_cont_h = h_tk.copy()

    # Phase 6: SLSQP polish
    allowed = min(80.0, tleft() * 0.08)
    if allowed > 20:
        h_ref, sc_ref = slsqp_refine(best_cont_h, n, float(n) / 2.0,
                                     alphas=[50, 200, 1000],
                                     maxiter=3000, ftol=1e-14,
                                     time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 7: SA from continuous-rounded solution
    allowed = min(100.0, tleft() * 0.10)
    if allowed > 20:
        h_rounded = project_binary(best_cont_h, target)
        n_sa = int(min(3000000, allowed * 8000))
        h_sa2, c5_sa2 = sa_search(h_rounded, n, n_sa, 0.01, 0.00002, allowed, start_time, rng,
                                  focused_frac=0.7, topk_candidates=10)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 8: Another ILS round from best
    allowed = min(80.0, tleft() * 0.08)
    if allowed > 30:
        h_ils2, c5_ils2 = ils_search(best_h, n, 30, allowed, start_time, rng)
        if c5_ils2 < best_score:
            best_score = c5_ils2
            best_h = h_ils2.copy()

    # Phase 9: SA with different parameters
    allowed = min(60.0, tleft() * 0.06)
    if allowed > 20:
        n_sa = int(min(2000000, allowed * 8000))
        h_sa3, c5_sa3 = sa_search(best_h, n, n_sa, 0.005, 0.00001, allowed, start_time, rng,
                                  focused_frac=0.4, topk_candidates=20)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Final SLSQP if time allows
    if tleft() > 20:
        h_ref2, sc_ref2 = slsqp_refine(best_cont_h, n, float(n) / 2.0,
                                       alphas=[1000, 5000],
                                       maxiter=5000, ftol=1e-15,
                                       time_limit=max(10.0, tleft() - 5),
                                       start_time=start_time)
        if sc_ref2 < best_cont_score:
            best_cont_score = sc_ref2
            best_cont_h = h_ref2.copy()

    # Choose best between binary and continuous
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
