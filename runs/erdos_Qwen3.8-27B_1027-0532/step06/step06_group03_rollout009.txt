def active_set_lp(h, n, target, plateau_size=5, max_iters=100, time_limit=60):
    start_t = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        # Find plateau: shifts within 0.01*dx of the max
        threshold = c_max - 0.01 * dx  # Hmm, this is in correlation units
        # Actually let's use a relative threshold
        threshold = c_max * 0.999  # top 0.1% of max
        
        plateau = np.where(corr >= threshold)[0]
        if len(plateau) == 0:
            plateau = np.where(corr >= c_max - 1e-10)[0]
        plateau = plateau[:plateau_size]
        
        if len(plateau) <= 1:
            # Only one active shift: just use gradient descent
            break
        
        # Build LP: minimize t
        # subject to: for each k in plateau:
        #   c_k + sum_i [(1-h[i+k])δ_i - h[i]δ_{i+k}]dx ≤ t (linearized)
        #   sum δ = 0
        #   -h[i] ≤ δ[i] ≤ 1-h[i]
        
        n_vars = n + 1  # δ_0..δ_{n-1}, t
        c_vec = np.zeros(n_vars)
        c_vec[-1] = 1.0  # minimize t
        
        A_ub = []  # A_ub @ x ≤ b_ub
        b_ub = []
        
        for k_idx in plateau:
            k = k_idx - mid
            c_k = corr[k_idx]
            # Linearized: c_k + sum_i [(1-h[i+k])δ_i - h[i]δ_{i+k}]dx ≤ t
            # => sum_i [(1-h[i+k])δ_i - h[i]δ_{i+k}]dx - t ≤ -c_k
            row = np.zeros(n_vars)
            for i in range(n):
                # (1-h[i+k])δ_i term
                if 0 <= i+k < n:
                    row[i] += (1.0 - h[i+k]) * dx
                # -h[i]δ_{i+k} term
                if 0 <= i+k < n:
                    row[i+k] -= h[i] * dx
            row[-1] = -1.0  # -t
            A_ub.append(row)
            b_ub.append(-c_k)
        
        # sum δ = 0 => this is an equality constraint
        # For scipy linprog, equality: A_eq @ x = b_eq
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        # Bounds: -h[i] ≤ δ[i] ≤ 1-h[i], t ≥ 0
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
