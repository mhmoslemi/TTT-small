import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(x, alpha):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def sum_constraint_jac(x):
    return np.ones(len(x))

def slsqp_round(h0, alphas, maxiter, ftol, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_val, h, args=(alpha,), jac=smooth_obj_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lp_cutting_plane(h0, n_active, max_rounds, deadline):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(np.asarray(h_var.value).flatten(), 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            for mix_val in [0.7, 0.5, 0.3]:
                h_mix = project_feasible(mix_val * h + (1.0 - mix_val) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    break
            else:
                break
    return best_h, best_c5

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def nesterov_smooth(h0, max_steps, deadline):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        alpha = 50.0 * (1.0 + step * 0.05)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx_of(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        grad = smooth_obj_grad(h, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = project_feasible(h + beta * v - lr_trial * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                v = 0.9 * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
            if step > 500:
                break
    return best_h, best_c5

def dx_of(h):
    return 2.0 / len(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: LP cutting plane with many constraints
    if tleft() > 200:
        h_lp, c5_lp = lp_cutting_plane(best_h, n_active=60, max_rounds=40,
                                        deadline=start_time + min(180.0, tleft() * 0.5))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: SLSQP with fine alpha schedule
    if tleft() > 200:
        h_p, s_p = slsqp_round(best_h,
                                alphas=[100, 400, 1500, 6000, 25000],
                                maxiter=2000, ftol=1e-14,
                                deadline=start_time + min(200.0, tleft() * 0.45))
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 3: Binary projection check
    h_bin, c5_bin = try_binary_projection(best_h)
    if c5_bin < best_score:
        best_score = c5_bin
        best_h = h_bin

    # Phase 4: Coordinate descent
    if tleft() > 200:
        h_cd, c5_cd = coord_descent_fast(best_h, n_cand=25, max_sweeps=20000,
                                          deadline=start_time + min(200.0, tleft() * 0.4))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Second LP round with even more constraints
    if tleft() > 180:
        h_lp, c5_lp = lp_cutting_plane(best_h, n_active=100, max_rounds=35,
                                        deadline=start_time + min(180.0, tleft() * 0.5))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: SLSQP at very high alpha
    if tleft() > 150:
        h_p, s_p = slsqp_round(best_h,
                                alphas=[10000, 50000, 200000, 500000],
                                maxiter=1500, ftol=1e-15,
                                deadline=start_time + min(150.0, tleft() * 0.5))
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 7: Subgradient with averaging
    if tleft() > 120:
        h_sg, c5_sg = subgradient_averaged(best_h, max_steps=20000,
                                            deadline=start_time + min(120.0, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: CD again
    if tleft() > 120:
        h_cd, c5_cd = coord_descent_fast(best_h, n_cand=30, max_sweeps=15000,
                                          deadline=start_time + min(120.0, tleft() * 0.4))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Binary local search
    if tleft() > 150:
        h_bin2, c5_bin2 = try_binary_projection(best_h)
        if c5_bin2 < best_score:
            best_score = c5_bin2
            best_h = h_bin2
        if tleft() > 80:
            h_bls, c5_bls = binary_local_search(best_h, max_swaps=5000,
                                                  deadline=start_time + min(80.0, tleft() * 0.5))
            if c5_bls < best_score:
                best_score = c5_bls
                best_h = h_bls

    # Phase 10: LP final round
    if tleft() > 100:
        h_lp, c5_lp = lp_cutting_plane(best_h, n_active=80, max_rounds=30,
                                        deadline=start_time + min(100.0, tleft() * 0.5))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Nesterov
    if tleft() > 80:
        h_ne, c5_ne = nesterov_smooth(best_h, max_steps=10000,
                                       deadline=start_time + min(80.0, tleft() * 0.5))
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne

    # Phase 12: Higher resolution attempt
    if tleft() > 200:
        for n_try in [800]:
            if tleft() < 180:
                break
            h_interp = interpolate_to(best_h, n_try)
            h_lp2, c5_lp2 = lp_cutting_plane(h_interp, n_active=50, max_rounds=20,
                                              deadline=start_time + min(100.0, tleft() * 0.4))
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2
            if tleft() < 80:
                break
            h_p2, s_p2 = slsqp_round(h_lp2 if c5_lp2 < best_score else h_interp,
                                      alphas=[200, 1000, 5000, 20000],
                                      maxiter=1500, ftol=1e-14,
                                      deadline=start_time + min(80.0, tleft() * 0.4))
            if s_p2 < best_score:
                best_score = s_p2
                best_h = h_p2
            if tleft() < 60:
                break
            h_cd2, c5_cd2 = coord_descent_fast(best_h, n_cand=20, max_sweeps=8000,
                                                deadline=start_time + min(60.0, tleft() * 0.4))
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 13: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        h_final, s_final = slsqp_round(best_h,
                                        alphas=[50000, 200000, 500000],
                                        maxiter=1500, ftol=1e-15,
                                        deadline=start_time + min(35.0, tleft() * 0.7))
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 14: Final CD
    if tleft() > 20:
        h_cd, c5_cd = coord_descent_fast(best_h, n_cand=25, max_sweeps=5000,
                                          deadline=start_time + min(15.0, tleft() * 0.7))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    h_out = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
