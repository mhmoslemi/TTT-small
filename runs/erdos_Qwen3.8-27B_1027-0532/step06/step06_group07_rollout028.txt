def active_set_step(h, n, dx, target, m_active=5, epsilon=1e-4):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    mid = n - 1
    idxs = np.arange(n)
    
    # Find active shifts
    sorted_corr = np.sort(corr)[::-1]
    threshold = sorted_corr[0] - corr * 0  # we'll use relative threshold
    # Actually, let's find shifts within epsilon of the max in terms of C5
    max_val = np.max(corr)
    active_mask = corr >= max_val - epsilon / dx
    active_shifts = np.where(active_mask)[0] - mid
    
    if len(active_shifts) > m_active:
        # Keep only top m_active
        top_indices = np.argsort(corr)[::-1][:m_active]
        active_shifts = top_indices - mid
    
    # Compute combined gradient
    g = np.zeros(n)
    for k in active_shifts:
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += (1.0 - h[idxs[v1] - k])
        g[v2] -= h[idxs[v2] + k]
    
    g /= len(active_shifts)
    g -= np.mean(g)  # project onto sum constraint null space
    
    # Find optimal step size
    # For each active shift k, f_k(h - αg) ≈ f_k(h) - α * (g · ∇f_k)
    # We want to minimize max_k [f_k(h) - α * d_k]
    # where d_k = g · ∇f_k(h)
    
    c_vals = corr[active_shifts + mid]  # current values
    d_vals = np.zeros(len(active_shifts))
    for i, k in enumerate(active_shifts):
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad_k = np.zeros(n)
        grad_k[v1] += (1.0 - h[idxs[v1] - k])
        grad_k[v2] -= h[idxs[v2] + k]
        d_vals[i] = np.dot(g, grad_k)
    
    # The optimal α: minimize max_i (c_i - α * d_i)
    # This happens where the lines intersect
    # c_i - α * d_i = c_j - α * d_j => α = (c_i - c_j) / (d_j - d_i)
    
    best_alpha = 0.0
    best_max_val = np.max(c_vals)
    
    # Candidate alphas: 0, and all intersection points
    alphas = [0.0]
    for i in range(len(c_vals)):
        for j in range(i+1, len(c_vals)):
            if abs(d_vals[j] - d_vals[i]) > 1e-12:
                a = (c_vals[i] - c_vals[j]) / (d_vals[j] - d_vals[i])
                if a > 0:
                    alphas.append(a)
    
    # Also consider the boundary where h hits 0 or 1
    if np.min(g) < 0:
        max_alpha = np.min(h[np.where(g < 0)[0]]) / (-np.min(g))
        if max_alpha > 0:
            alphas.append(max_alpha)
    if np.max(g) > 0:
        max_alpha = np.min(1.0 - h[np.where(g > 0)[0]]) / np.max(g)
        if max_alpha > 0:
            alphas.append(max_alpha)
    
    for a in alphas:
        if a <= 0:
            continue
        new_max = np.max(c_vals - a * d_vals)
        if new_max < best_max_val - 1e-14:
            best_max_val = new_max
            best_alpha = a
    
    if best_alpha < 1e-10:
        return h, c5, None
    
    h_new = h - best_alpha * g
    h_new = project_feasible(h_new, target)
    
    corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
    c5_new = np.max(corr_new) * dx
    
    return h_new, c5_new, best_alpha

        return h

        return h

        if len(idx2) > 0:

        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 60:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = h[i]
            if hi_val < 1e-10:
                continue
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi_val, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                d_i = np.zeros(n)
                d_i[i] = -1.0
                d_i[j] = 1.0
                d_i -= np.mean(d_i)

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    h_trial = h + mid_d * d_i
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                    corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                    if float(np.max(corr_trial)) < float(np.max(corr)) - 1e-12:
                        a_d = mid_d
                    else:
                        b_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                h_trial = project_feasible(h + delta_star * d_i, target)
                corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                approx = float(np.max(corr_trial)) * dx
                if approx < best_trial - 1e-15:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            d = np.zeros(n)
            d[best_i] = -1.0
            d[best_j] = 1.0
            h = project_feasible(h + best_delta * d, target)
        elif no_improve > 25:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP warm start
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 8000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active-set optimization (main workhorse)
    if tleft() > 300:
        as_time = min(350, tleft() * 0.4)
        h_as, c5_as = active_set_optimize(best_h, n, target,
                                          time_limit=as_time,
                                          m_active=5, eps_frac=0.003,
                                          max_iters=3000)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 3: Pair CD for fine-tuning
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.25)
        h_cd, c5_cd = pairwise_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   n_cand=40, n_bisect=14,
                                   max_sweeps=8000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Active-set with wider active set
    if tleft() > 200:
        as_time = min(200, tleft() * 0.3)
        h_as2, c5_as2 = active_set_optimize(best_h, n, target,
                                             time_limit=as_time,
                                             m_active=8, eps_frac=0.01,
                                             max_iters=2000)
        if c5_as2 < best_score:
            best_score = c5_as2
            best_h = h_as2

    # Phase 5: Null-space shaking
    if tleft() > 150:
        shake_time = min(150, tleft() * 0.25)
        h_sh, c5_sh = null_space_shake_and_optimize(best_h, n, target,
                                                     time_limit=shake_time,
                                                     m_active=5, eps_frac=0.005,
                                                     n_shakes=30, shake_scale=0.04)
        if c5_sh < best_score:
            best_score = c5_sh
            best_h = h_sh

    # Phase 6: Another active-set round
    if tleft() > 100:
        as_time = min(100, tleft() * 0.3)
        h_as3, c5_as3 = active_set_optimize(best_h, n, target,
                                             time_limit=as_time,
                                             m_active=4, eps_frac=0.002,
                                             max_iters=2000)
        if c5_as3 < best_score:
            best_score = c5_as3
            best_h = h_as3

    # Phase 7: Structured perturbation exploration
    if tleft() > 200:
        sp_time = min(200, tleft() * 0.3)
        h_sp, c5_sp = structured_perturbation_optimize(best_h, n, target,
                                                        time_limit=sp_time,
                                                        m_active=5, eps_frac=0.005)
        if c5_sp < best_score:
            best_score = c5_sp
            best_h = h_sp

    # Phase 8: Higher resolution attempt
    if tleft() > 250:
        for n_try in [800, 1000]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                   [300, 1000], 500, 1e-14)
            sc = compute_c5(h_hires)
            if tleft() > 100 and sc < best_score + 0.002:
                h_hires2, sc2 = active_set_optimize(h_hires, n_try, target_try,
                                                     time_limit=min(80, tleft() * 0.3),
                                                     m_active=5, eps_frac=0.003,
                                                     max_iters=500)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 9: Final pair CD
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(80, tleft() * 0.4)
        h_cd2, c5_cd2 = pairwise_cd(best_h, n_use, target_use,
                                     time_limit=cd_time,
                                     n_cand=50, n_bisect=15,
                                     max_sweeps=5000)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 10: Final active-set
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_as, c5_final = active_set_optimize(best_h, n_use, target_use,
                                                    time_limit=min(50, tleft() * 0.5),
                                                    m_active=3, eps_frac=0.001,
                                                    max_iters=500)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_as

    # Final SLSQP polish if time allows
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slsq = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000], 500, 1e-16)
        sc = compute_c5(h_slsq)
        if sc < best_score:
            best_score = sc
            best_h = h_slsq

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
