def compute_delta_correct(h, i, j, n):
    """Compute change in full correlation when h[i]:1->0 and h[j]:0->1."""
    result = np.zeros(2*n-1)
    shifts = np.arange(2*n-1) - (n-1)
    
    # For each shift, compute the delta
    # Only shifts where at least one affected term is valid need computation
    
    # Affected shifts: s where 0≤i+s<n or 0≤j+s<n or (s≠0 and 0≤i-s<n) or (s≠0 and 0≤j-s<n)
    # This is: s ∈ [-i, n-i-1] ∪ [-j, n-j-1] ∪ [i-n+1, i-1] ∪ [j-n+1, j-1] (for s≠0 parts)
    
    # For simplicity, just iterate over all shifts and compute (O(n) per shift, O(n^2) total)
    # Actually no, for each shift it's O(1) work (4 terms + correction), so O(n) total per move.
    
    for idx, s in enumerate(shifts):
        d = 0.0
        # k=i, first factor
        if 0 <= i + s < n:
            d += -(1.0 - h[i + s])
        # k=j, first factor
        if 0 <= j + s < n:
            if s == 0:
                pass  # delta = 0
            elif j + s == i:
                d += 1.0
            else:
                d += (1.0 - h[j + s])
        # k=i-s, second factor
        if s != 0 and 0 <= i - s < n:
            if i - s == j:
                d += 1.0
            else:
                d += h[i - s]
        # k=j-s, second factor
        if s != 0 and 0 <= j - s < n:
            d += -h[j - s]
        # Correction
        if s == j - i and j != i:
            d += 1.0
        elif s == i - j and j != i:
            d -= 1.0
        
        result[idx] = d
    
    return result
