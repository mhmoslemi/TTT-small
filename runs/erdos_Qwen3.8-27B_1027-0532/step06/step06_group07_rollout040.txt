    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = (float(c_max) + np.log(sum_exp) / alpha) * dx + mu * (np.sum(h) - target) ** 2
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(-(n - 1), n):
        w = weights[k + mid]
        if w < 1e-15:
            continue
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - target)
    return obj, grad


def lbfhb_round(h_init, n, target, alphas, mu, maxiter_per_alpha, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_and_grad, h_cur,
                           args=(alpha, mu, target, dx),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': ftol, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _global_target,
             'jac': lambda x: np.ones(_global_n)}]
    h_cur = h_start.copy()
    dx = 2.0 / n

    def obj_fn(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    for alpha in alphas:
        try:
            res = minimize(obj_fn, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5


def subgradient_method(h_init, n, target, time_limit, max_steps=5000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    step_size = 0.01
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step_size = min(step_size * 1.1, 0.1)
        else:
            no_improve += 1
            if no_improve > 50:
                step_size *= 0.9
            if no_improve > 200:
                break

        active_thresh = np.max(corr) - 0.02 / dx
        active = np.where(corr >= active_thresh)[0]
        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        h_try = project_feasible(h - step_size * grad, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
        else:
            step_size *= 0.8
            if step_size < 1e-6:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def level_set_method(h_init, n, target, time_limit, T_init, T_target, max_steps=3000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)

    T_lo = T_target
    T_hi = T_init
    n_bisect_T = 6

    for T_iter in range(n_bisect_T):
        if time.time() - start > time_limit:
            break
        T_mid = 0.5 * (T_lo + T_hi)
        T_corr = T_mid / dx

        h_cur = h.copy()
        lr = 0.03
        for step in range(max_steps):
            if time.time() - start > time_limit:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c5_now = float(np.max(corr)) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_cur.copy()

            violations = np.maximum(corr - T_corr, 0.0)
            if violations.max() < 1e-9:
                break

            grad = np.zeros(n)
            active_idx = np.where(corr > T_corr + 1e-10)[0]
            for s_idx in active_idx:
                k = int(s_idx) - mid
                w = (corr[int(s_idx)] - T_corr)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]

            grad *= dx
            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

            improved = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old * 0.99:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-7:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_cur.copy()

        if c5_test <= T_mid + 1e-6:
            T_lo = T_mid
            h = h_cur.copy()
        else:
            T_hi = T_mid

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    # Phase 1: L-BFGS-B with smooth objective (fast, many iterations)
    if tleft() > 80:
        h_l = lbfhb_round(best_h, n, target,
                          [200, 800, 3000, 10000], 100.0,
                          maxiter_per_alpha=300, ftol=1e-16)
        sc = compute_c5(h_l)
        if sc < best_score:
            best_score = sc
            best_h = h_l

    # Phase 2: Pair CD (main local optimizer)
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=8000, n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish with equal constraint
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target,
                            [5000, 20000, 80000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: L-BFGS-B with very high alpha
    if tleft() > 60:
        h_l = lbfhb_round(best_h, n, target,
                          [50000, 100000, 200000], 1000.0,
                          maxiter_per_alpha=200, ftol=1e-16)
        sc = compute_c5(h_l)
        if sc < best_score:
            best_score = sc
            best_h = h_l

    # Phase 5: Pair CD with larger candidate set
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.2)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=6000, n_cand=70, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Subgradient method
    if tleft() > 80:
        sg_time = min(100, tleft() * 0.12)
        h_sg, c5_sg = subgradient_method(best_h, n, target, time_limit=sg_time,
                                         max_steps=10000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
        # Pair CD after subgradient
        if tleft() > 80:
            cd_t = min(100, tleft() * 0.15)
            h_cd, c5_cd = pair_cd(best_h, n, target,
                                  time_limit=cd_t,
                                  max_sweeps=5000, n_cand=50, n_bisect=14)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 7: Level set method
    if tleft() > 120:
        ls_time = min(120, tleft() * 0.15)
        T_init = best_score + 0.003
        T_target = best_score - 0.003
        h_ls, c5_ls = level_set_method(best_h, n, target,
                                       time_limit=ls_time,
                                       T_init=T_init, T_target=T_target,
                                       max_steps=2000)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls
        if tleft() > 60:
            cd_t = min(80, tleft() * 0.12)
            h_cd, c5_cd = pair_cd(best_h, n, target,
                                  time_limit=cd_t,
                                  max_sweeps=4000, n_cand=50, n_bisect=14)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Multi-resolution exploration
    if tleft() > 200:
        for n_try in [500, 700, 800]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfhb_round(h_interp, n_try, target_try,
                                  [500, 2000, 8000], 50.0,
                                  maxiter_per_alpha=200, ftol=1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 100 and sc < best_score + 0.002:
                cd_t = min(100, tleft() * 0.25)
                h_hires2, sc2 = pair_cd(h_hires, n_try, target_try,
                                        time_limit=cd_t,
                                        max_sweeps=4000, n_cand=45, n_bisect=14)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 9: Try different resolutions with SLSQP
    if tleft() > 150:
        for n_try in [400, 500]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [2000, 10000, 50000], 500, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 60 and sc < best_score:
                cd_t = min(60, tleft() * 0.2)
                h_hires2, sc2 = pair_cd(h_hires, n_try, target_try,
                                        time_limit=cd_t,
                                        max_sweeps=3000, n_cand=50, n_bisect=14)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 10: Multi-restart perturbation + pair CD
    for noise_scale in [0.02, 0.05, 0.1, 0.2]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise_scale
        h_pert = project_feasible(h_pert, target)
        n_cur = len(best_h)
        cd_t = min(50, tleft() * 0.12)
        h_try, c5_try = pair_cd(h_pert, n_cur, target,
                                time_limit=cd_t,
                                max_sweeps=3000, n_cand=40, n_bisect=13)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 11: Final L-BFGS-B polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = lbfhb_round(best_h, n_use, target_use,
                              [50000, 100000, 200000], 5000.0,
                              maxiter_per_alpha=150, ftol=1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 12: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                              time_limit=cd_time,
                              max_sweeps=5000, n_cand=60, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 13: Final SLSQP
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [50000, 200000, 500000], 500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
