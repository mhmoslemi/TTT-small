import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, k, n):
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

def compute_c5_grad(h, n, beta_temp):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp(beta_temp * (corr - max_corr))
    weights /= np.sum(weights)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[k_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[k_idx] * h[idxs[v2] + k]
    return grad, max_corr

    h_prev = h.copy()

    best_c5 = compute_c5(h)
    no_improve = 0
    y = h.copy()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break
        
        t = float(it)
        progress = min(t / 5000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        momentum = mom_start * (1.0 - 0.3 * progress)
        
        y = h + momentum * (h - h_prev)
        y = project_feasible(y, target)
        
        grad, max_corr = compute_c5_grad(y, n, beta)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-15:

def smooth_slsqp(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    dx = 2.0 / n
    def obj_func(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(x.shape[0])
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
