For the greedy step, we need to compute the delta for each candidate swap. The delta for swapping position i (from 1 to 0) and position j (from 0 to 1) at the active lag k is:
delta[k] = -h[i+k] - h[i-k] + h[j+k] + h[j-k] (with boundary checks)

But for the full delta across all lags, we use the `compute_delta` function from the previous code.

Actually, for the greedy step, we only need to check if the new max is lower than the current max. We can do this efficiently by:
1. Computing the full correlation vector
2. For each candidate (i,j), computing the delta vector
3. Checking if (full + delta).max() < full.max()

The delta vector has non-zero entries only at lags related to i and j, so it's sparse. But computing it fully is O(n) which is fine.

def project_box_sum(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target} using Euclidean projection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    """Round h to binary while maintaining sum = target_sum."""
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        # Flip the 'most zero-like' ones to zero
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Final fix
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        extra = final_sum - target_sum
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:extra]] = 0
    elif final_sum < target_sum:
        extra = target_sum - final_sum
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):

    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        
        # Score candidates based on their effect on the peak lag
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += (1.0 - h[idx[valid]])
        
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += (1.0 - h[idx[valid]])
        
        top_i = ones[np.argsort(A[ones])[::-1][:min(30, len(ones))]]
        top_j = zeros[np.argsort(B[zeros])[::-1][:min(30, len(zeros))]]
        
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta_swap(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            # Try random swaps
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta_swap(h, i, j, k_arr)

def gradient_descent(h0, target, start_time, time_limit, alphas, lr_base=0.01):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        
        # Number of iterations for this alpha
        n_iter = int(remaining * 50 / len(alphas))  # distribute time
        n_iter = max(n_iter, 20)
        
        momentum = 0.8
        v = np.zeros(n)
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            grad = gradient_smooth(h, corr, alpha)
            
            # Adaptive learning rate
            lr = lr_base / (1.0 + 0.1 * it)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            
            if it % 20 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    return best_h, best_score
