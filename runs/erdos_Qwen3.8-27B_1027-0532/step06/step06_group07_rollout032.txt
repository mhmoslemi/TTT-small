from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def active_set_gradient(h, active_shifts, n):
    """Compute gradient of sum of c(r) over active shifts r."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    for r in active_shifts:
        # d c(r) / d h[j] = (1-h[j+r])*1[0<=j+r<n] - h[j-r]*1[0<=j-r<n]
        v_pos = (idxs + r >= 0) & (idxs + r < n)
        v_neg = (idxs - r >= 0) & (idxs - r < n)
        grad[v_pos] += (1.0 - h[idxs[v_pos] + r])
        grad[v_neg] -= h[idxs[v_neg] - r]
    grad -= np.mean(grad)
    return grad

def slsqp_smooth_obj(h, alpha, active_shifts_cache):
    n = len(h)
    corr = get_correlations(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(_global_n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha, None),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def active_set_descent(h_init, n, target, time_limit, tol=0.01, max_iters=5000):
    """Subgradient descent on the active set sum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = 0.5  # initial learning rate (in correlation units)
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = get_correlations(h)
        c_max = float(np.max(corr))
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Identify active set
        active_mask = corr >= c_max - tol / dx
        active_shifts = np.where(active_mask)[0]
        
        if len(active_shifts) == 0:
            break
        
        # Compute gradient of sum over active set
        active_r = (active_shifts - (n-1)).astype(int)
        grad = active_set_gradient(h, active_r, n)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-14:
            break
        
        # Backtracking line search

        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:

            h_try = project_feasible(h - lr_try * grad, target)

                h = h_try
                lr = lr_try
                improved = True
                break
        
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def slsqp_obj(h, alpha):
    corr = get_correlations(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_cons(h):
    return np.sum(h) - _global_target


def slsqp_cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        n_high = len(high_set)
        n_low = len(low_set)
        r_vals = np.arange(n_corr) - mid

        high_g = np.zeros((n_high, n_corr))
        for ii in range(n_high):
            i = high_set[ii]
            v_pos = (i + r_vals >= 0) & (i + r_vals < n)
            v_neg = (i - r_vals >= 0) & (i - r_vals < n)
            d = np.zeros(n_corr)
            d[v_pos] = (1.0 - h[i + r_vals[v_pos]])
            d[v_neg] -= h[i - r_vals[v_neg]]
            high_g[ii] = -d

        low_g = np.zeros((n_low, n_corr))
        for jj in range(n_low):
            j = low_set[jj]
            v_pos = (j + r_vals >= 0) & (j + r_vals < n)
            v_neg = (j - r_vals >= 0) & (j - r_vals < n)
            d = np.zeros(n_corr)
            d[v_pos] = (1.0 - h[j + r_vals[v_pos]])
            d[v_neg] -= h[j - r_vals[v_neg]]
            low_g[jj] = d

        best_trial_c5 = c5_val
        best_ii = -1
        best_jj = -1
        best_delta = 0.0

        for ii in range(n_high):
            i = high_set[ii]
            dm_row = np.minimum(h[i], 1.0 - h[low_set])
            valid = dm_row > 1e-12
            if not valid.any():
                continue
            g_batch = high_g[ii][None, :] + low_g[valid]
            dm_batch = dm_row[valid]
            nv = len(dm_batch)
            a_d = np.zeros(nv)
            b_d = dm_batch.copy()
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr[None, :] + mid_d[:, None] * g_batch
                f_mid = vals.max(axis=1)
                active = vals >= f_mid[:, None] - 1e-14
                fp = np.where(active, g_batch, -np.inf).max(axis=1)
                mask_up = fp > 1e-14
                mask_dn = fp < -1e-14
                b_d[mask_up] = mid_d[mask_up]
                a_d[mask_dn] = mid_d[mask_dn]
            delta_star = 0.5 * (a_d + b_d)
            approx = (corr[None, :] + delta_star[:, None] * g_batch).max(axis=1) * dx
            bi = np.argmin(approx)
            if approx[bi] < best_trial_c5 - 1e-15:
                best_trial_c5 = approx[bi]
                best_ii = ii
                best_jj = np.where(valid)[0][bi]
                best_delta = delta_star[bi]

        if best_ii >= 0:
            i = high_set[best_ii]
            j = low_set[best_jj]
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        elif no_improve > 30:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
