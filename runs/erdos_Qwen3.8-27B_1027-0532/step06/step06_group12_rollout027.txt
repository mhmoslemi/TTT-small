    dx = 2.0 / n
    n_half = n - 1

    h[i] = 0.0

    # For δ ≠ 0: linear term a[m]*t
    # For δ = 0: quadratic term t - t² (coefficient of t is 1-dx, coefficient of t² is -dx)

    #   So a[m=n_half] gets +dx, and c[m=n_half] = -dx

    c = np.zeros(2*n - 1)

            a[m] = dx
            c[m] = -dx

            if 0 <= i - delta < n:
                coeff += -h[i - delta]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lse_value(overlaps, alpha):
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    shifted = np.clip(shifted, -500, 0)
    return max_ov + np.log(np.sum(np.exp(shifted))) / alpha

def project_feasible(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_lse(h, alpha, top_k=None):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n), overlaps
    weights = exp_vals / total
    
    if top_k is not None:
        top_idx = np.argsort(weights)[::-1][:top_k]
        mask = np.zeros(2*n-1, dtype=bool)
        mask[top_idx] = True
        weights = weights * mask
        wsum = np.sum(weights)
        if wsum > 0:
            weights = weights / wsum
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-18:
            continue
        k = m - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            m2 = -k
            grad[m2:] += w * (1.0 - h[:n-m2])
            grad[:n-m2] += w * (-h[m2:])
    
    return grad, overlaps

def proj_grad_lse(h_init, n, target, alphas, time_limit, max_iter_per_alpha=5000, top_k=None):

    best_h = h.copy()
    best_c5 = compute_c5(h)

        f_cur = lse_value(compute_overlaps(h_cur), alpha)
        for t in range(max_iter_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5
            grad, overlaps = compute_grad_lse(h_cur, alpha, top_k=top_k)
            grad_norm_sq = np.dot(grad, grad)
            if grad_norm_sq < 1e-24:
                break
            
            lr = 2.0 / (alpha * grad_norm_sq + 1e-6)
            lr = min(lr, 5.0)
            
            improved = False
