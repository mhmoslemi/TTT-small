def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

        else:

            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=_g_bounds,
                           constraints=_g_cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur

_g_n = 0
_g_target = 0.0
_g_bounds = None
_g_cons = None
_g_deadline = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_wrap(h_start, n, target, alphas, maxiter=2000, ftol=1e-15, deadline=0.0):
    global _g_n, _g_target, _g_bounds, _g_cons, _g_deadline
    _g_n = n
    _g_target = target
    _g_bounds = [(0.0, 1.0)] * n
    _g_cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    _g_deadline = deadline
    return slsqp_smooth(h_start, n, target, alphas, maxiter, ftol)

def alternating_lp_full(h_init, n, target, time_limit, max_iter=2000,
                        damping=0.85, reg_start=1e-3, reg_end=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    n_full = 2 * n - 1
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for lag_idx in range(n_full):
            k = lag_idx - n_half
            if k == 0:
                coeff = g * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = g[k:] * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = g[:n - m] * dx
            constraints.append(coeff @ h_var <= t_var)
        progress = min(1.0, it / max(1, max_iter * 0.5))
        reg = reg_start * (reg_end / reg_start) ** progress
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def time_remaining(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_deadline
    start_time = time.time()
    _g_deadline = start_time + budget_s - 35.0
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FW with weighted subgradient
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        fw_time = min(250, tl * 0.25)
        h_fw, c5_fw = fw_weighted(best_h, n, target, fw_time,
                                   max_iter=300000, temp_start=0.005, temp_end=0.0003)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 2: LSE gradient descent
    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        gd_time = min(200, tl * 0.25)
        h_gd, c5_gd = lse_gradient_descent(best_h, n, target, gd_time,
                                            max_iter=20000, alpha_start=50.0,
                                            alpha_end=100000.0, lr_start=0.3, lr_end=0.03)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd

    # Phase 3: Alternating LP (full)
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        lp_time = min(200, tl * 0.3)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target, lp_time,
                                           max_iter=1000, damping=0.8,
                                           reg_start=1e-3, reg_end=1e-10)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP with escalating smoothing
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        h_slq = slsqp_wrap(best_h, n, target,
                           [100, 500, 2000, 10000, 50000, 200000, 1000000],
                           maxiter=2000, ftol=1e-15, deadline=_g_deadline)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: Binary local search
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        bls_time = min(120, tl * 0.5)
        h_bls, c5_bls = binary_local_search(best_h, n, target, bls_time, rng,
                                             max_rounds=300, n_trials_per_round=150)
        if c5_bls < best_score:
            best_score = c5_bls
            best_h = h_bls

    # Phase 6: Second FW round
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        fw_time = min(100, tl * 0.3)
        h_fw2, c5_fw2 = fw_weighted(best_h, n, target, fw_time,
                                     max_iter=200000, temp_start=0.002, temp_end=0.0001)
        if c5_fw2 < best_score:
            best_score = c5_fw2
            best_h = h_fw2

    # Phase 7: Another LSE GD round
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        gd_time = min(100, tl * 0.3)
        h_gd2, c5_gd2 = lse_gradient_descent(best_h, n, target, gd_time,
                                              max_iter=10000, alpha_start=1000.0,
                                              alpha_end=500000.0, lr_start=0.15, lr_end=0.02)
        if c5_gd2 < best_score:
            best_score = c5_gd2
            best_h = h_gd2

    # Phase 8: More binary search
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        bls_time = min(80, tl * 0.5)
        h_bls2, c5_bls2 = binary_local_search(best_h, n, target, bls_time, rng,
                                               max_rounds=200, n_trials_per_round=200)
        if c5_bls2 < best_score:
            best_score = c5_bls2
            best_h = h_bls2

    # Phase 9: Final SLSQP polish
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        h_final_slq = slsqp_wrap(best_h, n, target,
                                  [10000, 100000, 500000, 2000000],
                                  maxiter=3000, ftol=1e-16, deadline=_g_deadline)
        sc = compute_c5(h_final_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_final_slq

    # Phase 10: Final binary polish
    tl = time_remaining(start_time, budget_s)
    if tl > 40:
        h_bls3, c5_bls3 = binary_local_search(best_h, n, target, min(30, tl * 0.5),
                                               rng, max_rounds=100, n_trials_per_round=300)
        if c5_bls3 < best_score:
            best_score = c5_bls3
            best_h = h_bls3

    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
