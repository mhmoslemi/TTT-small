import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_box_simplex(y, s):
    """Project y onto {x : 0<=x_i<=1, sum(x_i)=s}"""
    if y.sum() == s:
        return np.clip(y, 0.0, 1.0)
    lo = y.min() - 0.1
    hi = y.max() + 0.1
    for _ in range(200):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > s:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Fix any numerical issues
    diff = s - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_subgrad(h, lags, dx, n):
    """Compute the average subgradient over the given lags"""
    g = np.zeros(n)
    for k in lags:
        if k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n-k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n-k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n-m]) * dx
            g[:n-m] += (-h[m:]) * dx
    g /= len(lags)
    return g

def solve_alternating_lp(h, n, target, reg=0.0):
    """Solve the alternating LP: min t (+ reg*||h'-h||^2) s.t. ..."""
    dx = 2.0 / n
    n_half = n - 1
    g = 1.0 - h
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    for idx in range(2 * n - 1):
        k = idx - n_half
        if k == 0:
            coeff = g * dx
        elif k > 0:
            coeff = np.zeros(n)
            coeff[:n-k] = g[k:]
            coeff *= dx
        else:
            m = -k
            coeff = np.zeros(n)
            coeff[m:] = g[:n-m]
            coeff *= dx
        constraints.append(coeff @ h_var <= t_var)
    
    if reg > 0:
        objective = t_var + reg * cp.sum_squares(h_var - h)
    else:
        objective = t_var
    
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, verbose=False)
        except Exception:
            return None
    
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

**Key question: Does the alternating LP converge?**

The alternating LP is a form of **Dinkelbach's algorithm** or **generalized proximal point method** for the bilinear objective. Convergence is not guaranteed in general, but in practice it works well when:
- The damping factor is appropriate
- The regularization is small
- We don't get stuck at a "saddle point" of the bilinear form

One issue: the alternating LP can oscillate between two points h1 and h2 where:
- LP from h1 gives h2
- LP from h2 gives h1

To prevent this, we can:
- Use damping (h_new = (1-λ)h + λh_lp)
- Use "贵的" regularization
- Track the best solution and restore if oscillation is detected

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_box_simplex(y, s):
    lo = y.min() - 0.1
    hi = y.max() + 0.1
    for _ in range(200):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > s:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    diff = s - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if mask.sum() > 0:
            result = result.copy()
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_subgrad(h, lags, dx, n):
    g = np.zeros(n)
    for k in lags:
        if k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n-k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n-k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n-m]) * dx
            g[:n-m] += (-h[m:]) * dx
    g /= len(lags)
    return g

def get_coeff(g, k, n, dx):
    c = np.zeros(n)
    if k == 0:
        c[:] = g
    elif k > 0:
        c[k:] = g[:n-k]
    else:
        m = -k
        c[:n-m] = g[m:]
    return c * dx

def time_left(start, budget):
    max_t = budget - 30
    return max(0.0, max_t - (time.time() - start))

def alternating_lp_method(h_init, n, target, time_limit, 
                          n_lags=None, reg_init=1e-4, reg_end=1e-8,
                          damping=0.8, line_search=True):
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    n_lags = n_lags or (2 * n - 1)
    
    h = h_init.copy()
    h = project_box_simplex(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    max_iter = 5000
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 50 and stuck > 200:
            break
        
        progress = min(1.0, it / max(1, max_iter * 0.5))
        reg = reg_init * (reg_end / reg_init) ** progress
        g = 1.0 - h
        
        # Select lags to include in LP
        overlaps = compute_overlaps(h)
        if n_lags < 2 * n - 1:
            top_idx = np.argsort(overlaps)[::-1][:n_lags]
            lags = top_idx - n_half
        else:
            lags = np.arange(-(n-1), n)
        
        # Build and solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            coeff = get_coeff(g, k, n, dx)
            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        h_prime = None
        for solver in [cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    h_prime = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
                    break
            except Exception:
                continue
        
        if h_prime is None:
            break
        
        h_prime = project_box_simplex(h_prime, target)
        
        if line_search:
            best_trial_c5 = best_c5
            best_trial_h = best_h.copy()
            for lam in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                h_trial = (1.0 - lam) * h + lam * h_prime
                h_trial = project_box_simplex(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_trial_c5:
                    best_trial_c5 = c5_trial
                    best_trial_h = h_trial.copy()
            h = best_trial_h
        else:
            h = project_box_simplex((1.0 - damping) * h + damping * h_prime, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

    return best_h, best_c5


def block_perturb(h, n, target, rng, block_size, n_blocks):
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        h_new[start:start + block_size] = 1.0 - h_new[start:start + block_size]
    return project_box_simplex(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    tl = time_left(start_time, budget_s)
    if tl > 200:
        h1, c5_1 = alternating_lp(best_h, n, target, min(180, tl * 0.35),
                                  n_top=None, reg_init=5e-5, reg_end=1e-9)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1

    tl = time_left(start_time, budget_s)
    if tl > 100:
        h2, c5_2 = alternating_lp(best_h, n, target, min(90, tl * 0.4),
                                  n_top=300, reg_init=1e-5, reg_end=1e-10)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2

    tl = time_left(start_time, budget_s)
    if tl > 80:
        h3, c5_3 = subgradient_method(best_h, n, target, min(70, tl * 0.5),
                                      K=80, alpha_init=0.005)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3

    tl = time_left(start_time, budget_s)
    if tl > 60:
        h4, c5_4 = local_search(best_h, n, target, min(50, tl * 0.5), rng)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4

    tl = time_left(start_time, budget_s)
    if tl > 120:
        h5, c5_5 = alternating_lp(best_h, n, target, min(100, tl * 0.45),
                                  n_top=None, reg_init=1e-6, reg_end=1e-11)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5

    tl = time_left(start_time, budget_s)
    if tl > 60:
        h6, c5_6 = subgradient_method(best_h, n, target, min(50, tl * 0.5),
                                      K=100, alpha_init=0.003)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6

    tl = time_left(start_time, budget_s)
    if tl > 40:
        h7, c5_7 = local_search(best_h, n, target, min(30, tl * 0.5), rng)
        if c5_7 < best_score:
            best_score = c5_7
            best_h = h7

    tl = time_left(start_time, budget_s)
    if tl > 100:
        for attempt in range(3):
            tl = time_left(start_time, budget_s)
            if tl < 30:
                break
            h_pert = block_perturb(best_h, n, target, rng, block_size=5, n_blocks=1)
            h8, c5_8 = alternating_lp(h_pert, n, target, min(25, tl * 0.25),
                                      n_top=250, reg_init=1e-4, reg_end=1e-8)
            if c5_8 < best_score:
                best_score = c5_8
                best_h = h8

    tl = time_left(start_time, budget_s)
    if tl > 40:
        h9, c5_9 = alternating_lp(best_h, n, target, min(30, tl * 0.7),
                                  n_top=None, reg_init=1e-7, reg_end=1e-12)
        if c5_9 < best_score:
            best_score = c5_9
            best_h = h9

    tl = time_left(start_time, budget_s)
    if tl > 20:
        h10, c5_10 = local_search(best_h, n, target, min(15, tl * 0.7), rng)
        if c5_10 < best_score:
            best_score = c5_10
            best_h = h10

    h_final = project_box_simplex(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
