def compute_gradient_lse(h, alpha, corr_full):
    n = len(h)
    dx = 2.0 / n
    
    max_c = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - max_c))
    w = exp_c / exp_c.sum()
    
    # s[j] = sum_{m=n-1-j}^{2n-2-j} w[m]
    # This is a sliding window of size n over w
    w_cumsum = np.concatenate([[0], np.cumsum(w)])
    s = np.zeros(n)
    for j in range(n):
        lo = n - 1 - j
        hi = 2 * n - 2 - j + 1  # exclusive
        s[j] = w_cumsum[hi] - w_cumsum[lo]
    
    # t[j] = np.correlate(h, w, mode='full')[n-1+j]
    corr_t = np.correlate(h, w, mode='full')  # length n + (2n-1) - 1 = 3n - 2
    t = corr_t[n-1 : n-1+n]  # t[j] = corr_t[n-1+j]
    
    # u[j] = np.correlate(h, w[::-1], mode='full')[j+n-1]
    w_rev = w[::-1]
    corr_u = np.correlate(h, w_rev, mode='full')  # length 3n - 2
    u = corr_u[n-1 : n-1+n]  # u[j] = corr_u[n-1+j]
    
    grad = dx * (s - t - u)
    return grad
