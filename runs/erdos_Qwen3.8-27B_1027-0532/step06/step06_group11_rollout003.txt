        else:

    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

        self.target = target

    corr = np.correlate(h, 1.0 - h, mode='full')

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def smooth_max_and_weights(c, alpha):
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    exponents = np.exp(shifted)
    weights = exponents / exponents.sum()
    smooth_val = c_max + np.log(exponents.sum()) / alpha
    return smooth_val, weights

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

class PenaltyLbfgs:
    def __init__(self, n, target, alpha, lam):
        self.n = n
        self.target = target
        self.alpha = alpha
        self.lam = lam
        self.dx = 2.0 / n
        self.mid = n - 1
    
    def objective(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * self.dx
        sm, _ = smooth_max_and_weights(c, self.alpha)
        penalty = self.lam * (np.sum(h) - self.target) ** 2
        return sm + penalty
    
    def gradient(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * self.dx
        _, weights = smooth_max_and_weights(c, self.alpha)
        grad = np.zeros(self.n)
        sig = np.where(weights > 1e-8)[0]
        for s_idx in sig:

            grad += weights[s_idx] * get_grad_shift(h, k, self.n)
        grad *= self.dx
        constraint_violation = np.sum(h) - self.target
        grad += 2.0 * self.lam * constraint_violation
        return grad

def lbfgs_round(h_start, n, target, alphas, lam_schedule, maxiter_per_alpha, time_limit):
    start = time.time()
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

                best_h = h_new
            h = h_new
        except Exception:
            pass

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]

def get_grad_shift(h, k, n):

    g = np.zeros(n)
