        if len(idx2) > 0:

    return h

def grad_gk(h, k, dx, n):
    """Gradient of g_k with respect to h.
    g_k = sum_{i: valid} h[i]*(1-h[i+k])*dx
    """
    grad = np.zeros(n)
    if k >= 0:

        if nk > 0:
            grad[:nk] += (1.0 - h[k:]) * dx
            grad[k:] -= h[:nk] * dx

def topk_gradient(h, dx, n, top_k=5):
    """Compute weighted gradient using top-k shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[-top_k:]
    # Soft weights
    alpha = 50.0
    w = np.exp(alpha * (corr[sorted_idx] - c_max))
    w /= w.sum()
    
    mid = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, sorted_idx):
        k = int(idx) - mid
        grad += wi * grad_gk(h, k, dx, n)
    return grad, c_max * dx

def active_set_optimize(h_init, n, target, time_limit, top_k=5, 
                         max_steps=100000, base_lr=0.05, mu=0.8):
    """Projected gradient with momentum, active-set tracking."""
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_t = time.time()
    step = 0
    while time.time() - start_t < time_limit and step < max_steps:
        # Compute gradient
        grad, c5_cur = topk_gradient(h, dx, n, top_k)
        # Center the gradient (to maintain the sum constraint approximately)
        grad -= np.mean(grad)
        
        # Momentum
        v = mu * v + grad
        
        # Backtracking line search
        lr = base_lr
        h_cur_c5 = compute_c5(h)
        for _ in range(30):
            h_try = project_feasible(h - lr * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-14:
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                break

        else:

            lr = max(lr, 1e-10)
            h_try = project_feasible(h - lr * v, target)
            h = h_try
        
        # Decay learning rate
        if step % 1000 == 999:
            base_lr *= 0.95
        
        step += 1
    
    return best_h, best_c5

def periodic_objective(f, p):
    """Compute C5 for a periodic pattern with period p."""
    # f has length p, sum(f) = p/2
    # Compute c_k for k = 0, ..., p-1
    vals = []
    for k in range(p):
        s = 0.0
        for j in range(p):
            s += f[j] * (1.0 - f[(j + k) % p])
        vals.append(s)
    return (2.0 / p) * max(vals)

def periodic_gradient(f, p):
    """Gradient of periodic_objective with respect to f."""
    # This is the max of linear functions, so gradient is the gradient of the max
    vals = np.zeros(p)
    for k in range(p):
        s = 0.0
        for j in range(p):
            s += f[j] * (1.0 - f[(j + k) % p])
        vals[k] = s
    
    k_max = np.argmax(vals)
    # Gradient of vals[k_max] with respect to f[j]
    grad = np.zeros(p)
    for j in range(p):
        # ∂/∂f[j] sum_l f[l]*(1-f[(l+k_max)%p])
        # = (1-f[(j+k_max)%p]) - f[(j-k_max)%p]  (mod p)
        ip = (j + k_max) % p
        im = (j - k_max) % p
        grad[j] = (1.0 - f[ip]) - f[im]
    
    return (2.0 / p) * grad, (2.0 / p) * vals[k_max]

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def grad_gk(h, k, dx, n):
    grad = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            grad[:nk] += (1.0 - h[k:]) * dx
            grad[k:] -= h[:nk] * dx
    else:
        k2 = -k
        nk = n - k2
        if nk > 0:
            grad[k2:] += (1.0 - h[:nk]) * dx

def topk_gradient(h, dx, n, top_k=5, soft_alpha=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[-top_k:]
    w = np.exp(soft_alpha * (corr[sorted_idx] - c_max))
    w /= w.sum()
    mid = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, sorted_idx):
        k = int(idx) - mid
        grad += wi * grad_gk(h, k, dx, n)
    return grad, c_max * dx


def active_set_optimize(h_init, n, target, time_limit, top_k=5,

def smooth_obj_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def smooth_grad_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad


def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n

    def con_fun(h):
        return np.sum(h) - _global_target

    def con_jac(h):
        return np.ones(_global_n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()

    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj_logsumexp, h_cur, args=(alpha,),
                          jac=lambda x, a: smooth_grad_logsumexp(x, a),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_alpha = 50.0
        w_top = np.exp(soft_alpha * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(25):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def periodic_c5_value(f, p):
    vals = np.zeros(p)
    for k in range(p):
        s = 0.0
        for j in range(p):
            s += f[j] * (1.0 - f[(j + k) % p])
        vals[k] = s
    return (2.0 / p) * np.max(vals)


def optimize_periodic(p, time_limit=4.0):
    f0 = np.full(p, 0.5)
    best_f = f0.copy()
    best_val = periodic_c5_value(f0, p)
    rng = np.random.RandomState(42)
    bounds = [(0.0, 1.0)] * p
    cons = [{'type': 'eq', 'fun': lambda f: np.sum(f) - p / 2.0,
             'jac': lambda f: np.ones(p)}]

    def obj(f):
        return periodic_c5_value(f, p)

    for trial in range(15):
        if trial == 0:
            f_start = f0.copy()
        else:
            f_start = rng.dirichlet(np.ones(p)) * p / 2.0
            f_start = np.clip(f_start, 0.0, 1.0)
            adj = (p / 2.0 - np.sum(f_start)) / p
            f_start = np.clip(f_start + adj, 0.0, 1.0)

        try:
            res = minimize(obj, f_start, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.fun < best_val:
                best_val = res.fun
                best_f = res.x.copy()
        except Exception:
            pass

    return best_f, best_val


def make_periodic_h(f, p, n):
    reps = n // p
    h = np.tile(f, reps)
    rem = n % p
    if rem > 0:
        h = np.concatenate([h, f[:rem]])
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation(h_init, n, target, max_tries, time_limit, seed=999):
    start_t = time.time()
    rng_local = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8, 13]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.02, 0.05, 0.1, 0.15]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Periodic structure exploration
    if tleft() > 60:
        for p in [3, 4, 5, 6, 7, 8, 10, 12]:
            if tleft() < 10:
                break
            f_opt, val = optimize_periodic(p, time_limit=3.0)
            for n_try in [n0]:
                if n_try % p == 0:
                    h_per = make_periodic_h(f_opt, p, n_try)
                    h_per = project_feasible(h_per, n_try / 2.0)
                    sc = compute_c5(h_per)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_per

    # Phase 2: Active-set gradient at n=400 (multiple configs)
    if tleft() > 200:
        for tk in [3, 5, 8]:
            if tleft() < 60:
                break
            h_as, c5_as = active_set_optimize(best_h, n0, target0,
                                              time_limit=min(50, tleft() * 0.2),
                                              top_k=tk, max_steps=30000,
                                              base_lr=0.05, mu=0.8,
                                              soft_alpha=50.0)
            if c5_as < best_score:
                best_score = c5_as
                best_h = h_as

    # Phase 3: Pair CD at n=400
    if tleft() > 150:
        h_cd, c5_cd = pair_coordinate_descent(best_h, n0, target0,
                                              time_limit=min(60, tleft() * 0.25),
                                              max_sweeps=5000, n_cand=50, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP at n=400
    if tleft() > 120:
        h_s = slsqp_smooth(best_h, n0, target0,
                           [50, 200, 1000, 5000],
                           2000, 1e-15, min(60, tleft() * 0.3))
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score = sc
            best_h = h_s

    # Phase 5: Pair CD again at n=400
    if tleft() > 80:
        h_cd, c5_cd = pair_coordinate_descent(best_h, n0, target0,
                                              time_limit=min(40, tleft() * 0.3),
                                              max_sweeps=3000, n_cand=65, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Go to n=800
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 250:
            h_as1, c5_as1 = active_set_optimize(h1, n1, target1,
                                                time_limit=min(120, tleft() * 0.25),
                                                top_k=5, max_steps=40000,
                                                base_lr=0.03, mu=0.85,
                                                soft_alpha=50.0)
            if c5_as1 < best_score:
                best_score = c5_as1
                best_h = h_as1

        if tleft() > 150:
            h_as1b, c5_as1b = active_set_optimize(best_h, n1, target1,
                                                  time_limit=min(60, tleft() * 0.2),
                                                  top_k=3, max_steps=30000,
                                                  base_lr=0.02, mu=0.9,
                                                  soft_alpha=80.0)
            if c5_as1b < best_score:
                best_score = c5_as1b
                best_h = h_as1b

        if tleft() > 80:
            h_cd1, c5_cd1 = pair_coordinate_descent(best_h, n1, target1,
                                                    time_limit=min(50, tleft() * 0.3),
                                                    max_sweeps=3000, n_cand=60, top_k=3)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    # Phase 7: Try n=1600 if time permits
    if tleft() > 250:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 180:
            h_as2, c5_as2 = active_set_optimize(h2, n2, target2,
                                                time_limit=min(100, tleft() * 0.3),
                                                top_k=3, max_steps=25000,
                                                base_lr=0.02, mu=0.85,
                                                soft_alpha=50.0)
            if c5_as2 < best_score:
                best_score = c5_as2
                best_h = h_as2

    # Phase 8: Go back to best resolution and final SLSQP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        if n_use <= 500:
            h_final = slsqp_smooth(best_h, n_use, target_use,
                                   [5000, 20000, 100000],
                                   1500, 1e-16, min(40, tleft() * 0.5))
            sc = compute_c5(h_final)
            if sc < best_score:
                best_score = sc
                best_h = h_final

    # Phase 9: Final active-set + pair CD polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_as, c5_as = active_set_optimize(best_h, n_use, target_use,
                                           time_limit=min(25, tleft() * 0.4),
                                           top_k=5, max_steps=20000,
                                           base_lr=0.01, mu=0.9,
                                           soft_alpha=100.0)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = pair_coordinate_descent(best_h, n_use, target_use,
                                              time_limit=min(25, tleft() * 0.5),
                                              max_sweeps=3000, n_cand=70, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Local perturbation for final nudge
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert, c5_pert = local_perturbation(best_h, n_use, target_use,
                                             max_tries=2000,
                                             time_limit=min(15, tleft() * 0.5),
                                             seed=12345)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
