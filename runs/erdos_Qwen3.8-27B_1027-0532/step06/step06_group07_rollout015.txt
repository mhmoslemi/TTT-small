def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:

        if len(idx2) > 0:

            h = np.clip(h, 0.0, 1.0)

                          lr_init=0.01, lr_decay=0.999, seed_val=42):

                g[vmn] -= (delta / block_size) * h[m - shifts[vmn]]

            approx = np.max(corr + g) * dx
            if approx < best_trial - 1e-15:
                best_trial = approx
                best_move = (src_block, dst_block, delta / block_size)

        if best_move is not None:
            src_block, dst_block, delta = best_move
            h[src_block] -= delta
            h[dst_block] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def multi_restart_cd(h_init, n, target, time_limit, n_restarts=5, seed_val=777):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    time_per_restart = time_limit / n_restarts
    for r in range(n_restarts):
        if time.time() - start > time_limit - 5:
            break
        noise = rng.randn(n) * 0.05
        h_pert = project_feasible(h_init + noise, target)
        remain = min(time_per_restart, time_limit - (time.time() - start))
        if remain < 10:
            break
        h_try, c5_try = pair_cd_multi(h_pert, n, target, time_limit=remain,
                                      max_sweeps=3000, n_cand=30, n_bisect=12, n_active=2)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 8000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Fast subgradient (main workhorse - much faster than SLSQP)
    if tleft() > 250:
        sg_time = min(250, tleft() * 0.35)
        h_sg, c5_sg = fast_subgradient(best_h, n, target, time_limit=sg_time,
                                       seed_val=42, top_k_init=1, lr_init=0.06, mom_init=0.3)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 3: Multi-active pair CD
    if tleft() > 250:
        cd_time = min(250, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_multi(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=5000, n_cand=40, n_bisect=13, n_active=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Another subgradient round with different params
    if tleft() > 200:
        sg_time = min(200, tleft() * 0.3)
        h_sg, c5_sg = fast_subgradient(best_h, n, target, time_limit=sg_time,
                                       seed_val=123, top_k_init=2, lr_init=0.04, mom_init=0.5)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Block coordinate descent
    if tleft() > 150:
        bc_time = min(150, tleft() * 0.25)
        h_bc, c5_bc = block_cd(best_h, n, target, time_limit=bc_time,
                               block_size=3, n_trials=25, n_bisect=12)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 6: SLSQP polish
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Final pair CD
    if tleft() > 120:
        cd_time = min(120, tleft() * 0.4)
        h_cd, c5_cd = pair_cd_multi(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=4000, n_cand=50, n_bisect=14, n_active=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Multi-restart exploration
    if tleft() > 100:
        mr_time = min(100, tleft() * 0.3)
        h_mr, c5_mr = multi_restart_cd(best_h, n, target, time_limit=mr_time,
                                       n_restarts=4, seed_val=seed)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 9: Final subgradient polish
    if tleft() > 60:
        sg_time = min(60, tleft() * 0.5)
        h_sg, c5_sg = fast_subgradient(best_h, n, target, time_limit=sg_time,
                                       seed_val=999, top_k_init=1, lr_init=0.03, mom_init=0.6)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: Final pair CD
    if tleft() > 40:
        cd_time = min(40, tleft() * 0.7)
        h_cd, c5_cd = pair_cd_multi(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=3000, n_cand=45, n_bisect=14, n_active=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
