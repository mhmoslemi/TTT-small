        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k in range(-mid, mid + 1):
        idx_k = k + mid
        w = weights[idx_k] / wsum
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def multi_active_cd(h, n, target, time_limit, max_sweeps=8000, n_cand=60, n_bisect=18, top_m=3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 100:
            break

        sorted_corr = np.argsort(corr)[::-1]
        active_indices = sorted_corr[:top_m]
        grad = np.zeros(n)
        for am in active_indices:
            k = int(am) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 50:
                break

    return best_h, best_c5


def block_pair_cd(h, n, target, time_limit, max_sweeps=4000, n_bisect=16, block_size=3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        block_starts_low = [i for i in range(0, n - block_size + 1) if np.mean(grad[i:i+block_size]) < 0]
        block_starts_high = [i for i in range(0, n - block_size + 1) if np.mean(grad[i:i+block_size]) > 0]
        if not block_starts_low or not block_starts_high:
            if no_improve > 40:
                break
            continue

        block_starts_low.sort(key=lambda i: np.mean(grad[i:i+block_size]))
        block_starts_high.sort(key=lambda i: -np.mean(grad[i:i+block_size]))
        low_bs = block_starts_low[:20]
        high_bs = block_starts_high[:20]

        best_trial_c5 = c5_val
        best_bi = -1
        best_bj = -1
        best_delta = 0.0

        for bi in high_bs:
            hi_avg = np.mean(h[bi:bi+block_size])
            if hi_avg < 1e-10:
                continue
            for bj in low_bs:
                if not (bj + block_size <= bi or bi + block_size <= bj):
                    continue
                lo_avg = np.mean(h[bj:bj+block_size])
                delta_max = min(hi_avg, 1.0 - lo_avg) * 0.8
                if delta_max < 1e-10:
                    continue
                g = np.zeros(n_corr)
                for d in range(block_size):
                    i = bi + d
                    j = bj + d
                    v_ip = (i + shifts >= 0) & (i + shifts < n)
                    v_im = (i - shifts >= 0) & (i - shifts < n)
                    v_jp = (j + shifts >= 0) & (j + shifts < n)
                    v_jm = (j - shifts >= 0) & (j - shifts < n)
                    g[v_ip] -= (1.0 - h[i + shifts[v_ip]]) / block_size
                    g[v_im] += h[i - shifts[v_im]] / block_size
                    g[v_jp] += (1.0 - h[j + shifts[v_jp]]) / block_size
                    g[v_jm] -= h[j - shifts[v_jm]] / block_size

                a_d = 0.0
                b_d = delta_max
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_bi = bi
                    best_bj = bj
                    best_delta = delta_star

        if best_bi >= 0:
            h[best_bi:best_bi+block_size] -= best_delta
            h[best_bj:best_bj+block_size] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def bisection_target_solver(h_init, n, target, time_limit, T_lo_init, T_hi_init, max_inner_steps=500):
    start_time = time.time()
    dx = 2.0 / n
    h_best = h_init.copy()
    best_c5 = compute_c5(h_best)
    idxs = np.arange(n)
    mid = n - 1

    for T_iter in range(12):
        if time.time() - start_time > time_limit:
            break
        T_mid = 0.5 * (T_lo_init + T_hi_init)
        T_corr = T_mid / dx

        h_cur = h_best.copy()
        lr = 0.02
        momentum = np.zeros(n)
        beta = 0.7

        for step in range(max_inner_steps):
            if time.time() - start_time > time_limit:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c5_now = float(np.max(corr)) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                h_best = h_cur.copy()

            violations = np.maximum(corr - T_corr, 0.0)
            if violations.max() < 1e-9:
                break

            grad = np.zeros(n)
            active_idx = np.where(violations > 1e-10)[0]
            for s_idx in active_idx:
                k = int(s_idx) - mid
                w = 2.0 * violations[int(s_idx)]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]

            grad *= dx
            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-13:
                break

            momentum = beta * momentum + (1 - beta) * grad
            improved = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * momentum, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old * 0.999:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-7:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            h_best = h_cur.copy()

        if c5_test <= T_mid + 1e-5:
            T_lo_init = T_mid
            h_best = h_cur.copy()
        else:
            T_hi_init = T_mid

    return h_best, best_c5


def structured_perturbation_cd(h_init, n, target, time_limit, seed_val=12345):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h_best = h_init.copy()
    best_c5 = compute_c5(h_best)
    idxs = np.arange(n)

    n_freqs = 8
    for trial in range(n_freqs):
        if time.time() - start > time_limit:
            break
        freq = rng.uniform(1.0, 8.0)
        amp = rng.uniform(0.02, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        x = idxs / float(n)
        pert = amp * np.sin(2 * np.pi * freq * x + phase)
        h_try = h_init + pert
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5:
            best_c5 = sc
            h_best = h_try.copy()

    if time.time() - start > time_limit:
        return h_best, best_c5

    cd_time = max(10, time_limit - (time.time() - start))
    h_cd, c5_cd = multi_active_cd(h_best, n, target, time_limit=cd_time,
                                   max_sweeps=3000, n_cand=50, n_bisect=16, top_m=3)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        h_best = h_cd

    return h_best, best_c5


def random_repair_cd(h_init, n, target, time_limit, seed_val=777):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h_best = h_init.copy()
    best_c5 = compute_c5(h_best)

    n_rounds = 4
    time_per_round = time_limit / n_rounds
    for _ in range(n_rounds):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.randint(n // 10, n // 3))
        scale = rng.choice([0.03, 0.06, 0.1, 0.15])
        idxs_pert = rng.choice(n, size=n_pert, replace=False)
        h_try = h_init.copy()
        h_try[idxs_pert] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        cd_t = max(5, time_per_round * 0.8 - (time.time() - start))
        if cd_t < 3:
            break
        h_cd, c5_cd = multi_active_cd(h_try, n, target, time_limit=cd_t,
                                       max_sweeps=2000, n_cand=40, n_bisect=14, top_m=2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            h_best = h_cd

    return h_best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target,
                            [400, 1500, 5000],
                            800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Multi-active pair CD (main workhorse, first pass)
    if tleft() > 200:
        cd_time = min(280, tleft() * 0.3)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_cand=65, n_bisect=18, top_m=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Bisection target solver
    if tleft() > 150:
        pen_time = min(180, tleft() * 0.25)
        T_hi = best_score + 0.003
        T_lo = best_score - 0.015
        h_pen, c5_pen = bisection_target_solver(best_h, n, target,
                                                 time_limit=pen_time,
                                                 T_lo_init=T_lo,
                                                 T_hi_init=T_hi,
                                                 max_inner_steps=400)
        if c5_pen < best_score:
            best_score = c5_pen
            best_h = h_pen

    # Phase 4: Block pair CD
    if tleft() > 120:
        cd_time = min(120, tleft() * 0.2)
        h_blk, c5_blk = block_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=3000, n_bisect=16, block_size=3)
        if c5_blk < best_score:
            best_score = c5_blk
            best_h = h_blk

    # Phase 5: SLSQP with higher alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target,
                            [5000, 20000, 60000],
                            1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Second multi-active CD round
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=6000, n_cand=75, n_bisect=20, top_m=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Structured perturbation + CD
    if tleft() > 100:
        sp_time = min(100, tleft() * 0.2)
        h_sp, c5_sp = structured_perturbation_cd(best_h, n, target,
                                                  time_limit=sp_time, seed_val=seed)
        if c5_sp < best_score:
            best_score = c5_sp
            best_h = h_sp

    # Phase 8: Random repair CD
    if tleft() > 100:
        rr_time = min(100, tleft() * 0.2)
        h_rr, c5_rr = random_repair_cd(best_h, n, target,
                                        time_limit=rr_time, seed_val=seed + 1)
        if c5_rr < best_score:
            best_score = c5_rr
            best_h = h_rr

    # Phase 9: Very high alpha SLSQP
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target,
                            [50000, 150000, 400000],
                            800, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Final targeted CD
    if tleft() > 60:
        cd_time = min(80, tleft() * 0.5)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=4000, n_cand=80, n_bisect=20, top_m=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP
    if tleft() > 25:
        h_final = slsqp_round(best_h, n, target,
                              [100000, 300000],
                              500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
