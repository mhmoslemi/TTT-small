```python
def smooth_gradient(h, alpha):
    """Compute gradient of (1/alpha) * log(sum(exp(alpha * overlap_k)))"""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Gradient
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        d = k - n_half  # lag
        w = weights[k] * dx
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            # h[j] as first factor: j in [0, n-1-d]
            grad[:n-d] += w * (1.0 - h[d:])
            # h[j] as second factor: j in [d, n-1]
            grad[d:] -= w * h[:n-d]
        else:
            m = -d
            # h[j] as h[s+m]: j in [m, n-1]
            grad[m:] += w * (1.0 - h[:n-m])
            # h[j] as h[s]: j in [0, n-1-m]
            grad[:n-m] -= w * h[m:]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_exact(h, S):
    """Project h onto {x : 0 <= x <= 1, sum(x) = S} using bisection"""
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum exactly
    diff = S - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-14)[0]
        else:
            idx = np.where(x > 1e-14)[0]
        if len(idx) > 0:
            x = x.copy()
            x[idx] += diff / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj_and_grad(h, alpha):
    """Compute smoothed objective and its gradient"""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    obj = max_ov + np.log(Z) / alpha
    p = exp_shifted / Z  # softmax weights
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-16:
            continue
        d = k - n_half
        w = p[k] * dx
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
        else:
            m = -d
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return obj, grad

def fista_optimize(h0, n, S, alpha, max_iter, time_limit, start_time):
    """FISTA for smoothed objective with exact projection"""
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5 / (alpha * n * 0.01 + 1.0)  # initial step size
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        obj_y, grad_y = smooth_obj_and_grad(y, alpha)
        
        # Gradient step
        h_new = y - step * grad_y
        h_new = project_exact(h_new, S)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + (t_k - 1.0) / t_new * (h_new - h)
        y = np.clip(y, -0.5, 1.5)  # keep y somewhat bounded
        h = h_new
        t_k = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def lbfgsb_optimize(h0, n, S, alpha, max_iter, time_limit, start_time):
    """L-BFGS-B with penalty for sum constraint"""
    bounds = [(0.0, 1.0)] * n
    
    def objfun(h):
        obj, grad = smooth_obj_and_grad(h, alpha)
        penalty = 100.0 * (np.sum(h) - S) ** 2
        grad_pen = 200.0 * (np.sum(h) - S)
        return obj + penalty, grad + grad_pen * np.ones(n)
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    try:
        res = minimize(objfun, h, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_exact(res.x, S)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c5
```

This is a multi-objective optimization. The trade-off is between making h concentrated (minimizing k=0 overlap) and making h spread out (minimizing large-k overlaps).

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_exact(h, S):
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-14)[0]
        else:
            idx = np.where(x > 1e-14)[0]
        if len(idx) > 0:
            x = x.copy()
            x[idx] += diff / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x


def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    obj = max_ov + np.log(Z) / alpha
    p = exp_shifted / Z
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n_half
        w = pk * dx
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
        else:
            m = -d
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return obj, grad


def fista_optimize(h0, n, S, alpha, max_iter, time_limit, start_time):
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 1.0 / (alpha * 0.02 + 2.0)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        obj_y, grad_y = smooth_obj_grad(y, alpha)
        h_new = project_exact(y - step * grad_y, S)
        obj_new = compute_c5(h_new)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + (t_k - 1.0) / t_new * (h_new - h)
        y = np.clip(y, -0.1, 1.1)
        h = h_new
        t_k = t_new
        if obj_new < best_c5 - 1e-10:
            best_c5 = obj_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.95
                t_k = 1.0
                y = h.copy()
                no_improve = 0
                if it > 200 and step < 1e-4:
                    break
    return best_h, best_c5


def lbfgsb_optimize(h0, n, S, alpha, max_iter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    pen_scale = 50.0
    state = {'h': h0.copy()}

    def objfun(h):
        obj, grad = smooth_obj_grad(h, alpha)
        residual = np.sum(h) - S
        penalty = pen_scale * residual * residual
        grad_pen = 2.0 * pen_scale * residual * np.ones(n)
        return obj + penalty, grad + grad_pen

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    try:
        res = minimize(objfun, h0, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': max_iter * 3})
        h_new = project_exact(res.x, S)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass
    if time.time() - start_time > time_limit * 0.7:
        return best_h, best_c5
    try:
        res2 = minimize(objfun, best_h, method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': max_iter, 'ftol': 1e-18,
                                 'gtol': 1e-14, 'maxfun': max_iter * 3})
        h_new2 = project_exact(res2.x, S)
        c5_new2 = compute_c5(h_new2)
        if c5_new2 < best_c5:
            best_c5 = c5_new2
            best_h = h_new2
    except Exception:
        pass
    return best_h, best_c5


def random_binary(n, S, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:int(S)]] = 1.0
    return h


def structured_init(n, S, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        block = int(S // 4)
        offsets = [0, block, 2 * block, 3 * block]
        shift = rng.randint(0, n)
        for off in offsets:
            start = (off + shift) % n
            for j in range(block):
                h[(start + j) % n] = 1.0
    elif mode == 1:
        block = int(S // 8)
        for j in range(int(S)):
            h[(j * 5 + rng.randint(0, 3)) % n] = 1.0
    else:
        freq = rng.randint(3, 20)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2.0, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        h = np.clip(h, 0, 1)
    return project_exact(h, S)


def perturb_h(h, n, S, rng, n_flips):
    h_new = h.copy()
    for _ in range(n_flips):
        start = rng.randint(0, max(1, n - 10))
        length = rng.randint(3, 11)
        h_new[start:start + length] = 1.0 - h_new[start:start + length]
    return project_exact(h_new, S)


def slsqp_polish(h0, n, S, alpha, maxiter, time_limit, start_time):
    global _slsqp_n, _slsqp_S
    _slsqp_n = n
    _slsqp_S = S
    bounds = [(0.0, 1.0)] * n

    def cons_fun(h):
        return np.sum(h) - _slsqp_S

    def cons_jac(h):
        return np.ones(_slsqp_n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    try:
        obj, grad = smooth_obj_grad(h0, alpha)
        res = minimize(smooth_obj_grad, h0, args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'iprint': 0})
        h_new = project_exact(res.x, S)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    except Exception:
        pass
    return best_h, best_c5
```
