        constraints = [

            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        corr = np.correlate(h, 1.0 - h, mode='full')

        if n_active == 'all':

                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        h = project_feasible(h, target)
        c5_val = compute_c5(h)

    dx = 2.0 / n
    g = 1.0 - h
    half_n = n - 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    for k_idx in active_k_indices:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)

            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)
    
    if h_var.value is None:
        return h, None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), t_var.value

def run_lp_round(h_init, target, n, n_active, time_limit, scs_eps, damping_init, reg_init):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    damping = damping_init
    reg = reg_init
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        sorted_active = sorted(active_k.tolist())
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * (2.0 / n)
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=200000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        
        if not solved and not (h_var.value is not None):
            try:
                prob.solve(solver=cp.ECOS, max_iter=200000)
                solved = h_var.value is not None
            except Exception:
                pass
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Update active set
        top_k = np.argsort(corr)[-n_active:]
        t_val = t_var.value if t_var.value is not None else best_c5
        dx = 2.0 / n
        # Check if any non-active k is now a problem
        for k_idx in range(1, 2 * n - 1):
            if k_idx not in set(active_k.tolist()) and k_idx not in set(top_k.tolist()):
                if corr[k_idx] * dx > t_val + 1e-4:
                    break
        
        active_k = top_k
        
        # Adaptive damping
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.5, damping * 0.995)
        else:
            damping = min(0.99, damping + 0.005)
        
        # Reduce regularization
        if it % 50 == 0 and reg > 1e-8:
            reg = max(1e-8, reg * 0.9)
    
    return best_h, best_c5
