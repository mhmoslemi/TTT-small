Let me now think about the Frank-Wolfe algorithm:

```
FW algorithm for min f(h) s.t. h ∈ P = {h: 0≤h≤1, sum(h) = n/2}

h_0 = initial point
for t = 1, 2, ...:
    g = FWoracle(∇f(h_{t-1}))  # solves min_z <∇f, z> s.t. z ∈ P
    γ_t = 2/(t+2)  # or use line search
    h_t = (1-γ_t) h_{t-1} + γ_t g
```

The FW oracle (linear minimization over the polytope):
min_z <∇f, z> s.t. 0≤z≤1, sum(z) = n/2

Solution: Sort ∇f in ascending order. Set z_i = 1 for the smallest n/2 values of ∇f, and z_i = 0 for the rest. (Tie-breaking: if ∇f_{(n/2)} = ∇f_{(n/2+1)}, any convex combination works.)

This is O(n log n) per iteration (for the sort), which is very fast.

The step size: Use the standard 2/(t+2) or the exact line search:
γ* = argmin_{γ∈[0,1]} f(h + γ(g-h))

The line search is expensive (requires evaluating f at each trial point), but we can use a backtracking approach or a quadratic approximation.

Actually, for our problem, we can do a "strong" FW variant where we use:
h_t = h_{t-1} + γ_t * (g - h_{t-1})

with γ_t chosen by line search. The line search can be done by trying a few values of γ.

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    return val

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    p /= p.sum()

    h_p = np.zeros(N)

    B = np.fft.irfft(np.fft.rfft(a_B_p) * np.fft.rfft(h_p))[:n]
    
    # C[j] = sum_i p[mid + j - i] * h[i] using FFT Toeplitz
    # M_C first row: [p[mid], p[mid-1], ..., p[0]]
    # M_C first col: [p[mid], p[mid+1], ..., p[mid+n-1]]
    a_C = np.concatenate([p[mid::-1], p[mid+n-1:mid:-1]])
    a_C_p = np.zeros(N)
    a_C_p[:len(a_C)] = a_C
    C = np.fft.irfft(np.fft.rfft(a_C_p) * np.fft.rfft(h_p))[:n]
    
    grad = (A - B - C) * dx
    return grad

def fw_oracle(grad, n):
    """Solve min <grad, z> s.t. 0<=z<=1, sum(z)=n/2"""

    z = np.zeros(n)

    count = int(target)
    z[sorted_idx[:count]] = 1.0
    # Handle fractional if target is not integer (shouldn't happen for even n)
    return z

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    p /= p.sum()
    mid = n - 1
    idxs = np.arange(n)
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    A = p_cum[mid - idxs + n] - p_cum[mid - idxs]
    N = 1
    while N < 2 * n - 1:
        N *= 2
    h_p = np.zeros(N)
    h_p[:n] = h
    a_B = np.concatenate([p[mid:], p[:mid]])
    a_B_p = np.zeros(N)
    a_B_p[:len(a_B)] = a_B
    B = np.fft.irfft(np.fft.rfft(a_B_p) * np.fft.rfft(h_p))[:n]
    a_C = np.concatenate([p[mid::-1], p[mid + n - 1:mid:-1]])
    a_C_p = np.zeros(N)
    a_C_p[:len(a_C)] = a_C
    C = np.fft.irfft(np.fft.rfft(a_C_p) * np.fft.rfft(h_p))[:n]
    return (A - B - C) * dx


def fw_oracle(grad, n):
    target = n // 2
    sorted_idx = np.argsort(grad)
    z = np.zeros(n)
    z[sorted_idx[:target]] = 1.0
    return z


def frank_wolfe(h_init, n, target, alphas, iters_per_alpha, time_limit):
    start_t = time.time()
    h = h_init.copy()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        for t in range(iters_per_alpha):
            if time.time() - start_t > time_limit:
                break
            grad = smooth_grad(h, alpha)
            z = fw_oracle(grad, n)
            gamma = 2.0 / (t + 2)
            h_new = (1.0 - gamma) * h + gamma * z
            h = project_feasible(h_new, target)
    return h


def fw_shrink(h_init, n, target, alphas, iters_per_alpha, time_limit):
    start_t = time.time()
    h = h_init.copy()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        for t in range(iters_per_alpha):
            if time.time() - start_t > time_limit:
                break
            grad = smooth_grad(h, alpha)
            z = fw_oracle(grad, n)
            gamma = 2.0 / (t + 2)
            mu = 1.0 / (t + 1) * 0.1
            h_new = (1.0 - gamma - mu) * h + gamma * z
            h_new = np.clip(h_new, 0.0, 1.0)
            h = project_feasible(h_new, target)
    return h


def con_fun(x):
    return np.sum(x) - _global_target


def con_jac(x):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
