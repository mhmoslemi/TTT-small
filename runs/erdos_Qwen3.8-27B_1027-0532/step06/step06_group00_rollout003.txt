def fast_pair_cd_active(h_init, n, target, time_limit, max_sweeps=50000,
                        n_cand_start=20, n_cand_end=70, max_no_improve=300,
                        top_m=50):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    best_h = h.copy()

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts_arr >= 0) & (p + shifts_arr < n)
        neg_mask = (p - shifts_arr >= 0) & (p - shifts_arr < n)
        change[p, pos_mask] += (1.0 - h[p + shifts_arr[pos_mask]])
        change[p, neg_mask] -= h[p - shifts_arr[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        # Find gradient from top shifts
        top_m_actual = min(top_m, n_corr)
        top_idxs = np.argsort(corr)[-top_m_actual:]
        top_sum = np.sum(corr[top_idxs])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)

            grad[v1] += w * (1.0 - h[idxs[v1] - k]])
            grad[v2] -= w * h[idxs[v2] + k]]

        current_ncand = min(current_ncand, n // 2)

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1

        top_shifts_idx = np.argsort(corr)[-top_m_actual:]
        
        n_h = len(high_set)
        n_l = len(low_set)

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:

                # Fast bisection using only top shifts

                        f_prime = 0.0
                    if f_prime > 1e-12:

                    else:
                        a_d = mid_d

                if abs(delta_star) < 1e-12:
                    continue

                # Verify with full correlation
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change matrix rows
            change[best_i, :] = 0.0
            pos_mask = (best_i + shifts_arr >= 0) & (best_i + shifts_arr < n)
            neg_mask = (best_i - shifts_arr >= 0) & (best_i - shifts_arr < n)
            change[best_i, pos_mask] += (1.0 - h[best_i + shifts_arr[pos_mask]])

            change[best_j, pos_mask] += (1.0 - h[best_j + shifts_arr[pos_mask]])
            change[best_j, neg_mask] -= h[best_j - shifts_arr[neg_mask]]

    return best_h, best_c5

import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        change[p, pos_mask] += (1.0 - h[p + shifts_arr[pos_mask]])
        change[p, neg_mask] -= h[p - shifts_arr[neg_mask]]
    return change, shifts_arr

            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5


def cd_multi_shift(h_init, n, target, time_limit, max_sweeps=30000,
                   n_cand=50, n_shifts=5, top_m=60):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change, shifts_arr = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 200:
            break

        top_m_actual = min(top_m, n_corr)
        top_idxs = np.argsort(corr)[-top_m_actual:]
        top_sum = np.sum(corr[top_idxs])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_top = g[top_idxs]
                corr_top = corr[top_idxs]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals_top = corr_top + mid_d * g_top
                    f_mid = np.max(vals_top)
                    active_mask = vals_top >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_top[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, max_steps=20000, base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(99999)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at original resolution
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0,
                            [50, 200, 1000, 5000],
                            600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at original resolution
    if tleft() > 120:
        h_cd, c5_cd = cd_active(best_h, n0, target0,
                                time_limit=min(80, tleft() * 0.5),
                                max_sweeps=20000,
                                n_cand_start=20, n_cand_end=60,
                                max_no_improve=300, top_m=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish again
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0,
                            [2000, 10000, 50000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Interpolate to n=800 and optimize
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            n0 = n1
            target0 = target1

        if tleft() > 250:
            h_cd1, c5_cd1 = cd_active(best_h, n0, target0,
                                      time_limit=min(180, tleft() * 0.55),
                                      max_sweeps=40000,
                                      n_cand_start=20, n_cand_end=70,
                                      max_no_improve=400, top_m=60)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 100:
            h_pol1 = slsqp_round(best_h, n0, target0,
                                 [5000, 20000, 100000],
                                 1500, 1e-14)
            sc1 = compute_c5(h_pol1)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

        if tleft() > 150:
            h_cd1b, c5_cd1b = cd_active(best_h, n0, target0,
                                        time_limit=min(100, tleft() * 0.5),
                                        max_sweeps=30000,
                                        n_cand_start=30, n_cand_end=80,
                                        max_no_improve=400, top_m=80)
            if c5_cd1b < best_score:
                best_score = c5_cd1b
                best_h = h_cd1b

    # Phase 5: Try n=600 as alternative path
    if tleft() > 250:
        n_alt = 600
        target_alt = n_alt / 2.0
        h_alt = interpolate_to(best_h, n_alt, target_alt)
        sc_alt = compute_c5(h_alt)

        if tleft() > 200:
            h_cd_alt, c5_cd_alt = cd_active(h_alt, n_alt, target_alt,
                                            time_limit=min(120, tleft() * 0.5),
                                            max_sweeps=30000,
                                            n_cand_start=20, n_cand_end=60,
                                            max_no_improve=300, top_m=50)
            if c5_cd_alt < sc_alt:
                sc_alt = c5_cd_alt
                h_alt = h_cd_alt

        if tleft() > 80:
            h_pol_alt = slsqp_round(h_alt, n_alt, target_alt,
                                    [5000, 20000, 100000],
                                    1000, 1e-14)
            sc_alt2 = compute_c5(h_pol_alt)
            if sc_alt2 < sc_alt:
                sc_alt = sc_alt2
                h_alt = h_pol_alt

        if sc_alt < best_score:
            best_score = sc_alt
            best_h = h_alt
            n0 = n_alt
            target0 = target_alt

    # Phase 6: Interpolate to n=1000 for final refinement
    if tleft() > 300:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n0 = n2
            target0 = target2

        if tleft() > 250:
            h_cd2, c5_cd2 = cd_active(best_h, n0, target0,
                                      time_limit=min(180, tleft() * 0.55),
                                      max_sweeps=35000,
                                      n_cand_start=20, n_cand_end=65,
                                      max_no_improve=350, top_m=60)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        if tleft() > 100:
            h_pol2 = slsqp_round(best_h, n0, target0,
                                 [2000, 10000, 50000],
                                 1200, 1e-14)
            sc2 = compute_c5(h_pol2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

        if tleft() > 120:
            h_cd2b, c5_cd2b = cd_active(best_h, n0, target0,
                                        time_limit=min(80, tleft() * 0.5),
                                        max_sweeps=25000,
                                        n_cand_start=30, n_cand_end=75,
                                        max_no_improve=300, top_m=80)
            if c5_cd2b < best_score:
                best_score = c5_cd2b
                best_h = h_cd2b

    # Phase 7: Multi-start from perturbed versions
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_round(h_pert, n0, target0,
                                [500, 5000, 20000],
                                600, 1e-14)
            sc = compute_c5(h_ref)
            if tleft() > 60 and sc < best_score:
                h_cd_ref, c5_ref = cd_active(h_ref, n0, target0,
                                             time_limit=min(40, tleft() * 0.4),
                                             max_sweeps=10000,
                                             n_cand_start=25, n_cand_end=70,
                                             max_no_improve=200, top_m=60)
                if c5_ref < sc:
                    sc = c5_ref
                    h_ref = h_cd_ref
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref

    # Phase 8: Stochastic refinement
    if tleft() > 80:
        h_st, c5_st = stochastic_refine(best_h, n0, target0,
                                        time_limit=min(50, tleft() * 0.5),
                                        max_steps=15000, base_scale=0.03)
        if c5_st < best_score:
            best_score = c5_st
            best_h = h_st

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        h_final = slsqp_round(best_h, n0, target0,
                              [10000, 50000, 200000],
                              800, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 10: Final CD
    if tleft() > 40:
        h_cd_final, c5_final = cd_active(best_h, n0, target0,
                                         time_limit=min(30, tleft() * 0.6),
                                         max_sweeps=10000,
                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=150, top_m=80)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
