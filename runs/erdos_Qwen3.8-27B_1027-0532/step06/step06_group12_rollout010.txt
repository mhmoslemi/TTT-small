    if abs(diff) < 1e-12:
        return h
    if diff > 0:

        total_space = space.sum()
        if total_space < 1e-10:
            return h

        if total < 1e-10:
            return h
        h = h + diff * h / total
    h = np.clip(h, 0.0, 1.0)

    if abs(residual) > 1e-10:

            idx = np.where(h < 1.0 - 1e-8)[0]

            idx = np.where(h > 1e-8)[0]

            h = np.clip(h, 0.0, 1.0)

    coef[valid1] += (1.0 - h[idx1[valid1]]) * dx
    coef[valid2] -= h[idx2[valid2]] * dx
    return coef

    if delta_hi - delta_lo < 1e-12:
        return 0.0

    f_hi = np.max(overlaps + delta_hi * slopes)

            return delta_hi

    hi = delta_hi

def find_optimal_delta(overlaps, slopes, delta_lo, delta_hi):
    if delta_hi - delta_lo < 1e-12:
        return 0.0
    f_lo = np.max(overlaps + delta_lo * slopes)
    f_hi = np.max(overlaps + delta_hi * slopes)
    if f_lo <= f_hi:
        vals_lo = overlaps + delta_lo * slopes
        active = np.argwhere(np.abs(vals_lo - f_lo) < 1e-9).flatten()
        if active.size == 0:
            active = np.array([np.argmax(vals_lo)])
        if slopes[active].min() >= -1e-10:
            return delta_lo
    if f_hi <= f_lo:
        vals_hi = overlaps + delta_hi * slopes
        active = np.argwhere(np.abs(vals_hi - f_hi) < 1e-9).flatten()
        if active.size == 0:
            active = np.array([np.argmax(vals_hi)])
        if slopes[active].max() <= 1e-10:
            return delta_hi
    lo = delta_lo
    hi = delta_hi
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = overlaps + mid * slopes
        f_max = vals.max()
        active = np.argwhere(np.abs(vals - f_max) < 1e-9).flatten()
        if active.size == 0:
            active = np.array([np.argmax(vals)])
        s_min = slopes[active].min()
        s_max = slopes[active].max()
        if s_min > 1e-10:
            hi = mid
        elif s_max < -1e-10:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def pairwise_cd(h_init, n, target, time_limit, start_time, seed=42, guided=True):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    overlaps = compute_overlaps(h)
    recompute_every = 300
    n_iter = 0
    no_improve = 0
    while True:
        if time.time() - start_time > time_limit:
            break
        n_iter += 1
        if n_iter % recompute_every == 0:
            overlaps = compute_overlaps(h)
        if guided and n_iter > 100:
            max_ov = overlaps.max()
            thresh = max_ov - 1e-5 * max(1.0, max_ov)
            active = np.where(overlaps > thresh)[0]
            if active.size > 0 and rng.random() < 0.6:
                m_star = active[rng.randint(active.size)]
                lo = max(0, m_star - n_half)
                hi = min(n - 1, m_star)
                if lo <= hi:
                    j_range = np.arange(lo, hi + 1)
                    shifted = j_range - m_star + n_half
                    scores = h[j_range] * (1.0 - h[shifted])
                    if scores.max() > 1e-8:
                        best_idx = np.argmax(scores)
                        i = int(j_range[best_idx])
                        j = int(shifted[best_idx])
                    else:
                        i = rng.randint(n)
                        j = rng.randint(n)
                else:
                    i = rng.randint(n)
                    j = rng.randint(n)
            else:
                i = rng.randint(n)
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        ci = compute_single_coef(h, n, i)
        cj = compute_single_coef(h, n, j)
        slopes = cj - ci
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-10:
            continue
        delta = find_optimal_delta(overlaps, slopes, delta_lo, delta_hi)
        if abs(delta) < 1e-12:
            continue
        h[i] = np.clip(h[i] - delta, 0.0, 1.0)
        h[j] = np.clip(h[j] + delta, 0.0, 1.0)
        sum_diff = target - h.sum()
        if abs(sum_diff) > 1e-8:
            h = project_feasible(h, target)
            overlaps = compute_overlaps(h)
        else:
            overlaps += delta * slopes
        c5 = overlaps.max()
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80000 and n_iter > 10000:
            break
    return best_h, best_c5

def binary_swap_search(h_init, n, target, time_limit, start_time, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h_bin[sorted_idx[:int(target)]] = 1.0
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    overlaps = compute_overlaps(h_bin)
    n_half = n - 1
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0
    it = 0
    while True:
        if time.time() - start_time > time_limit:
            break
        it += 1
        if it % 500 == 0:
            overlaps = compute_overlaps(h_bin)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        ci = compute_single_coef(h_bin, n, i)
        cj = compute_single_coef(h_bin, n, j)
        slopes = cj - ci
        new_overlaps = overlaps + slopes
        new_c5 = float(new_overlaps.max())
        if new_c5 < best_c5 - 1e-12:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            overlaps = new_overlaps
            best_c5 = new_c5
            best_h = h_bin.copy()
            no_improve = 0
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            no_improve += 1
        if no_improve > 30000:
            break
    return best_h, best_c5

_g_n = 0
_g_target = 0.0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _g_target

def cons_jac(h):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter=3000, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def structured_init(n, target, rng, shift):
    h = np.zeros(n)
    block = int(target // 4)
    offsets = [0, block, 2 * block, 3 * block]
    for off in offsets:
        start = (off + shift) % n
        for j in range(block):
            h[(start + j) % n] = 1.0
    return project_feasible(h, target)

def time_remaining(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Pairwise coordinate descent from initial (long run)
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_cd1, c5_cd1 = pairwise_cd(best_h, n, target, min(250, tl * 0.3), start_time, seed=seed, guided=True)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    # Phase 2: Binary swap search
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_bin, c5_bin = binary_swap_search(best_h, n, target, min(120, tl * 0.2), start_time, seed=seed + 1)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin

    # Phase 3: Pairwise CD from binary result (fractional refinement)
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_cd2, c5_cd2 = pairwise_cd(best_h, n, target, min(200, tl * 0.25), start_time, seed=seed + 2, guided=True)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 4: SLSQP soft-max polish
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        h_slq = slsqp_optimize(best_h, n, target,
                                [50, 200, 1000, 5000, 20000, 100000, 500000],
                                maxiter=4000, ftol=1e-15)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: More pairwise CD after SLSQP
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_cd3, c5_cd3 = pairwise_cd(best_h, n, target, min(150, tl * 0.2), start_time, seed=seed + 3, guided=True)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 6: Another binary swap from current best
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        h_bin2, c5_bin2 = binary_swap_search(best_h, n, target, min(80, tl * 0.15), start_time, seed=seed + 4)
        if c5_bin2 < best_score:
            best_score = c5_bin2
            best_h = h_bin2

    # Phase 7: Pairwise CD unguided (different dynamics)
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_cd4, c5_cd4 = pairwise_cd(best_h, n, target, min(100, tl * 0.15), start_time, seed=seed + 5, guided=False)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Phase 8: SLSQP with very high alpha (near-exact max)
    tl = time_remaining(start_time, budget_s)
    if tl > 120:
        h_slq2 = slsqp_optimize(best_h, n, target,
                                 [1000, 10000, 100000, 1000000, 5000000],
                                 maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_slq2)
        if sc < best_score:
            best_score = sc
            best_h = h_slq2

    # Phase 9: Final pairwise CD polish
    tl = time_remaining(start_time, budget_s)
    if tl > 50:
        h_cd5, c5_cd5 = pairwise_cd(best_h, n, target, min(40, tl * 0.5), start_time, seed=seed + 6, guided=True)
        if c5_cd5 < best_score:
            best_score = c5_cd5
            best_h = h_cd5

    # Phase 10: Final SLSQP if time permits
    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        h_final_try = slsqp_optimize(best_h, n, target,
                                      [50000, 500000, 5000000, 20000000],
                                      maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = project_feasible(best_h.copy(), target)
        final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
