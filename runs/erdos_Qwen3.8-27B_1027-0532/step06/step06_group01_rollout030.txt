import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        residual = target_sum - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            break
        h[mask] += residual / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def grad_for_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m1 = np.arange(0, n - k)
        if len(m1) > 0:
            grad[m1] += (1.0 - h[m1 + k])
        m2 = np.arange(k, n)
        if len(m2) > 0:
            grad[m2] -= h[m2 - k]
    else:
        kp = -k
        m1 = np.arange(kp, n)
        if len(m1) > 0:
            grad[m1] += (1.0 - h[m1 - kp])
        m2 = np.arange(0, n - kp)
        if len(m2) > 0:
            grad[m2] -= h[m2 + kp]
    return grad

def smooth_obj(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp_round(h_start, n, target, alphas, maxiter, ftol, time_limit):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    start = time.time()

def lp_guided_descent(h_init, n, target, time_limit, 
                      n_active=10, tol_frac=0.001, step_bound_init=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    step_bound = step_bound_init
    no_improve = 0
    max_no_improve = 50
    
    it = 0
    while time.time() - start < time_limit:
        it += 1
        if no_improve > max_no_improve:
            break
        
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Identify active set
        threshold = cur_max * (1.0 - tol_frac)
        active_idx = np.where(corr >= threshold)[0]
        if len(active_idx) > n_active:
            active_idx = active_idx[np.argsort(-corr[active_idx])[:n_active]]
        n_act = len(active_idx)
        if n_act == 0:
            active_idx = np.array([np.argmax(corr)])
            n_act = 1
        
        # Compute gradients
        gradients = np.zeros((n_act, n))
        for ai, si in enumerate(active_idx):
            k = int(si) - mid
            gradients[ai] = grad_for_shift(h, k)
        
        # Solve LP
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((n_act, n_vars))
        b_ub = np.zeros(n_act)
        for ai in range(n_act):
            A_ub[ai, :n] = gradients[ai]
            A_ub[ai, -1] = -1.0
            b_ub[ai] = -corr[active_idx[ai]]
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        lb = np.full(n_vars, -step_bound)
        ub = np.full(n_vars, step_bound)
        ub[-1] = 1.0  # t upper bound (generous)
        lb[-1] = 0.0
        bounds = list(zip(lb, ub))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            # Fallback: use negative gradient of argmax
            k_star = int(np.argmax(corr)) - mid
            g = grad_for_shift(h, k_star)
            d = -g / (np.linalg.norm(g) + 1e-15) * step_bound
        else:
            d = res.x[:n].copy()
        
        # Ensure sum(d) = 0 (the LP should enforce this, but just in case)
        d -= np.mean(d)
        d = np.clip(d, -step_bound, step_bound)
        
        # Line search
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            h_try = project_feasible(h + frac * d, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        
        if not improved:
            # Try smaller step
            step_bound *= 0.7
            if step_bound < 1e-6:
                break
            no_improve += 1
        else:
            # Reset or increase step bound
            step_bound = min(step_bound * 1.1, step_bound_init)
    
    return best_h, best_c5

def pair_cd_improved(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0

        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        top_k = 5
        top_idx = np.argsort(-corr)[:top_k]
        combined_grad = np.zeros(n)

        weights = weights / weights.sum()
        for wi, si in zip(weights, top_idx):
            k = int(si) - mid
            combined_grad += wi * grad_for_shift(h, k)

        sorted_by_grad = np.argsort(-combined_grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]

                    continue
                
                # Optimal delta ≈ (cur_max - second_best) / (-max_g)
                # But simpler: just use bisection with fewer iterations
                lo = 0.0
                hi_d = delta_max
                for _ in range(10):  # reduced from 15

                    vals = corr + mid_d * g

                    # Check if we can do better

                    if not np.any(active):

                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-12:
                        hi_d = mid_d
                    else:
                        lo = mid_d

    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        residual = target_sum - np.sum(h)
        if abs(residual) < 1e-15:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not np.any(mask):
            break
        h[mask] += residual / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def grad_for_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m1 = np.arange(0, n - k)
        if len(m1) > 0:
            grad[m1] += (1.0 - h[m1 + k])
        m2 = np.arange(k, n)

            grad[m2] -= h[m2 - k]

        if len(m1) > 0:
            grad[m1] += (1.0 - h[m1 - kp])
        m2 = np.arange(0, n - kp)
        if len(m2) > 0:
            grad[m2] -= h[m2 + kp]
    return grad
