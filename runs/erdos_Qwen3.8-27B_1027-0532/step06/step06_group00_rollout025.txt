import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

# Smooth objective for SLSQP
_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    # Use log-sum-exp for smooth max
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def block_cd(h_init, n, target, time_limit, max_sweeps=20000, block_size=3):
    """Block coordinate descent: modify 'block_size' variables at once"""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        # Compute gradient
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        # Select block: lowest and highest gradient elements
        sorted_idx = np.argsort(grad)
        low_candidates = sorted_idx[:block_size * 3]
        high_candidates = sorted_idx[-(block_size * 3):]
        
        # Try all combinations of block_size from low and high
        # For efficiency, try a few combinations
        best_trial = c5_val
        best_move = None
        
        # Try moving 1 element from high to 1 from low (pair)
        for i in high_candidates[:block_size * 2]:
            for j in low_candidates[:block_size * 2]:
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                # Simple bisection for delta
                a_d, b_d = 0.0, delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    h_try = h.copy()
                    h_try[i] -= mid_d
                    h_try[j] += mid_d
                    corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                    if np.max(corr_try) < np.max(corr):
                        a_d = mid_d
                    else:
                        b_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue
                h_try = h.copy()
                h_try[i] -= delta_star
                h_try[j] += delta_star
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                trial_val = float(np.max(corr_try)) * dx
                if trial_val < best_trial:
                    best_trial = trial_val
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    for step in range(max_steps):

        grad = np.zeros(n)

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Pick random pair from candidates
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        # Random delta in feasible range
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        # Acceptance criterion
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < 0:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-delta_c / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h_old_i = h[i]
            h_old_j = h[j]
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx_global
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def multi_shift_pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
                        n_cand=50, n_shifts=4):
    dx = 2.0 / n
    start = time.time()
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-5.0 * (c_max - corr[top_shifts]) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5

def sa_pair_cd(h_init, n, target, time_limit, max_steps=40000,
               t_start=0.05, t_end=0.0005, n_cand=60):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(42)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 600:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def restart_refine(h_best, n, target, rng, n_restarts, time_limit, start_time):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        scale = 0.05 * (r + 1)
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, dx_global
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP on initial grid
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000],
                                800, 1e-14,
                                time_limit=tleft() * 0.6, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD on n=600
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd, c5_cd = fast_pair_cd_v2(h1, n1, target1,
                                      time_limit=min(200, tleft() * 0.6),
                                      max_sweeps=50000,
                                      n_cand_start=30, n_cand_end=90,
                                      max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            n0 = n1
            target0 = target1

    # Phase 3: Multi-shift CD on current grid
    if tleft() > 100:
        h_ms, c5_ms = multi_shift_pair_cd(best_h, n0, target0,
                                          time_limit=min(80, tleft() * 0.5),
                                          max_sweeps=15000, n_cand=50, n_shifts=4)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 4: SLSQP refinement
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [2000, 10000],
                                1200, 1e-14,
                                time_limit=tleft() * 0.6, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: CD on n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2, c5_cd2 = fast_pair_cd_v2(h2, n2, target2,
                                        time_limit=min(200, tleft() * 0.6),
                                        max_sweeps=40000,
                                        n_cand_start=35, n_cand_end=100,
                                        max_no_improve=500)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n0 = n2
            target0 = target2

    # Phase 6: SA for escape
    if tleft() > 100:
        h_sa, c5_sa = sa_pair_cd(best_h, n0, target0,
                                 time_limit=min(70, tleft() * 0.5),
                                 max_steps=30000,
                                 t_start=0.04, t_end=0.0003,
                                 n_cand=60)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 7: SLSQP with high alpha
    if tleft() > 70:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [5000, 20000, 80000],
                                1500, 1e-14,
                                time_limit=tleft() * 0.6, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Rapid CD
    if tleft() > 100:
        h_cd8, c5_cd8 = fast_pair_cd_v2(best_h, n0, target0,
                                        time_limit=min(80, tleft() * 0.6),
                                        max_sweeps=20000,
                                        n_cand_start=50, n_cand_end=100,
                                        max_no_improve=300)
        if c5_cd8 < best_score:
            best_score = c5_cd8
            best_h = h_cd8

    # Phase 9: Try n=1000
    if tleft() > 250:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        h_cd3, c5_cd3 = fast_pair_cd_v2(h3, n3, target3,
                                        time_limit=min(180, tleft() * 0.6),
                                        max_sweeps=25000,
                                        n_cand_start=30, n_cand_end=80,
                                        max_no_improve=400)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            n0 = n3
            target0 = target3

        if tleft() > 80:
            h_pol3, sc3 = slsqp_round(best_h, n0, target0,
                                      [3000, 10000],
                                      1000, 1e-14,
                                      time_limit=tleft() * 0.6, start_time=start_time)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 10: Final multi-shift + CD
    if tleft() > 100:
        h_ms2, c5_ms2 = multi_shift_pair_cd(best_h, n0, target0,
                                            time_limit=min(60, tleft() * 0.5),
                                            max_sweeps=10000, n_cand=50, n_shifts=4)
        if c5_ms2 < best_score:
            best_score = c5_ms2
            best_h = h_ms2

    # Phase 11: Final SLSQP
    if tleft() > 40:
        h_fp, sc_fp = slsqp_round(best_h, n0, target0,
                                  [10000, 50000],
                                  600, 1e-14,
                                  time_limit=tleft() * 0.7, start_time=start_time)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    # Final rapid CD
    if tleft() > 30:
        h_cf, c5_cf = fast_pair_cd_v2(best_h, n0, target0,
                                      time_limit=min(20, tleft() * 0.7),
                                      max_sweeps=8000,
                                      n_cand_start=60, n_cand_end=100,
                                      max_no_improve=150)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
