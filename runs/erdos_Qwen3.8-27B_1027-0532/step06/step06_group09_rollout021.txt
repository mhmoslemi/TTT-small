def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_push(h, target, strength=0.1):
    """Push values toward 0 or 1 to exploit binary structure."""
    h = h.copy()
    # Push values in (0.3, 0.7) toward the nearest binary value
    mid_mask = (h > 0.3) & (h < 0.7)
    if np.any(mid_mask):
        h[mid_mask] = np.where(h[mid_mask] > 0.5, 
                               h[mid_mask] + strength * (1 - h[mid_mask]),
                               h[mid_mask] - strength * h[mid_mask])
    return project_feasible(h, target)

def multi_alpha_polish(h_start, n, target, alphas, maxiter_per=1000):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_obj(x, alpha), h_start,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def slsqp_polish(h_start, n, target, alpha, maxiter=500):
    def obj_fn(x, a=alpha):
        return smooth_obj(x, a)
    def cons_fn(x, t=target):
        return np.sum(x) - t
    def cons_jac_fn(x, n=n):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac_fn}]
    try:
        res = minimize(obj_fn, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def game_theoretic_descent(h_init, n, target, time_limit, max_steps=5000,
                           eta_base=1.0, m_top=10, momentum=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve_count = 0
    v = np.zeros(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 500:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        if momentum > 0:
            v = momentum * v + (1.0 - momentum) * grad
            step_grad = v
        else:
            step_grad = grad

        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(10):
            h_try = project_feasible(h - lr * step_grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.15, 0.5)
                improved = True
                break
            lr *= 0.5

            best_lr *= 0.85
            if best_lr < 0.0005:
                break

        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.7 * lam + 0.3 * lam_new

        if step % 300 == 0 and no_improve_count > 150:

                no_improve_count = 0

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

                delta_lo = max(h[i] - 1.0, -h[j])

                jsm_arr[jm] = h[j - shifts[jm]]

                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e10)
                if b_d - a_d < 1e-10:
                    b_d = a_d + 1e-10

                jpm_arr = np.zeros(n_corr)
                jpm_arr[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm_arr = np.zeros(n_corr)
                jsm_arr[jm] = h[j - shifts[jm]]

                g = h_i_sp + h_i_sm - jpm_arr - jsm_arr

                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e10)
                if b_d - a_d < 1e-10:
                    b_d = a_d + 1e-10

                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_alpha_polish(h_start, n, target, alphas, maxiter_per=1500):
    h_cur = h_start.copy()
    for alpha in alphas:
        h_new = lbfags_polish(h_cur, n, target, alpha, maxiter=maxiter_per)
        if compute_c5(h_new) <= compute_c5(h_cur) + 1e-12:
            h_cur = h_new
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick game-theoretic with momentum
    if tleft() > 100:
        h_gt, c5_gt = game_theoretic_descent(best_h, n, target,
                                             time_limit=min(60, tleft() * 0.1),
                                             max_steps=5000,
                                             eta_base=2.5, m_top=12,
                                             momentum=0.6)
        if c5_gt < best_score:
            best_score = c5_gt
            best_h = h_gt

    # Phase 2: L-BFGS-B moderate alpha
    if tleft() > 80:
        h_pol = lbfags_polish(best_h, n, target, alpha=1000, maxiter=2000)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Main pair CD (single-lag, large candidates)
    if tleft() > 300:
        cd_time = min(450, tleft() * 0.38)

                                       time_limit=cd_time,
                                       max_sweeps=20000,
                                       n_cand_start=50,

                                       patience=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Game-theoretic with higher momentum

        sc = compute_c5(h_pol)

    # Phase 8: Try higher resolution

            target_try = n_try / 2.0
