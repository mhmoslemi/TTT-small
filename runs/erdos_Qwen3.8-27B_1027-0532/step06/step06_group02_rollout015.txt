import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):

                best_h = replicas[r]['h'].copy()
        
        # Exchange attempt (Metropolis criterion)
        if n_replicas > 1:
            ra = rng.randint(n_replicas - 1)

            ta = t_levels[ra]

    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def best_move_sa(h_init, n_iters, t_init, t_final, rng, time_limit, start_time,
                 k_arr, n_cand=6, top_k=5, explore_frac=0.25, use_adaptive=True):
    n = len(h_init)
    h = h_init.copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_d = float('inf')
        best_i = -1
        best_j = -1
        best_delta = None

        n_try = n_cand
        if use_adaptive and it % 2000 == 0:
            n_try = max(4, n_cand - it // 100000)

        for _ in range(n_try):
            if rng.random() < explore_frac:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                top_idx = np.argsort(full)[::-1][:top_k]
                move_found = False
                for ti in top_idx:
                    k_star = int(ti) - (n - 1)
                    if k_star == 0:
                        continue
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    i_c = ones[A[ones] >= 1.0]
                    j_c = zeros[B[zeros] <= 1.0]
                    if len(i_c) > 0 and len(j_c) > 0:
                        i = i_c[rng.randint(len(i_c))]
                        j = j_c[rng.randint(len(j_c))]
                        move_found = True
                        break
                if not move_found:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]

            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            d = (new_max - float(full.max())) * dx
            if d < best_d:
                best_d = d
                best_i = i
                best_j = j
                best_delta = delta

        if best_d < 0 or (t > 1e-15 and rng.random() < np.exp(-best_d / t)):
            h[best_i] = 0.0
            h[best_j] = 1.0
            full += best_delta
            current_c5 += best_d
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return best_h, best_c5

def parallel_tempering(h_init, n_replicas, t_levels, n_steps, rng, time_limit, start_time, k_arr):
    n = len(h_init)
    dx = 2.0 / n
    replicas = []
    for r in range(n_replicas):
        h_r = h_init.copy()
        full_r = full_corr_fast(h_r)
        c5_r = float(full_r.max()) * dx
        replicas.append([h_r, full_r, c5_r])

    best_h = h_init.copy()
    best_c5 = float(full_corr_fast(h_init).max()) * dx

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        r = rng.randint(n_replicas)
        t = t_levels[r]
        h_r = replicas[r][0]
        ones = np.where(h_r == 1.0)[0]
        zeros = np.where(h_r == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue

        best_d = float('inf')
        best_i, best_j, best_delta = -1, -1, None
        for _ in range(3):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h_r, i, j, k_arr)
            new_max = float((replicas[r][1] + delta).max())
            d = (new_max - float(replicas[r][1].max())) * dx
            if d < best_d:
                best_d = d
                best_i, best_j, best_delta = i, j, delta

        if best_d < 0 or (t > 1e-15 and rng.random() < np.exp(-best_d / t)):
            h_r[best_i] = 0.0
            h_r[best_j] = 1.0
            replicas[r][1] += best_delta
            replicas[r][2] += best_d

        if replicas[r][2] < best_c5:
            best_c5 = replicas[r][2]
            best_h = replicas[r][0].copy()

        if n_replicas > 1 and step % 5 == 0:
            ra = rng.randint(n_replicas - 1)
            rb = ra + 1
            ta, tb = t_levels[ra], t_levels[rb]
            if ta > 0 and tb > 0:
                delta_e = (replicas[rb][2] - replicas[ra][2]) * (1.0/tb - 1.0/ta)
                if delta_e < 0 or (rng.random() < np.exp(-min(delta_e, 500.0))):
                    replicas[ra][0], replicas[rb][0] = replicas[rb][0].copy(), replicas[ra][0].copy()
                    replicas[ra][1], replicas[rb][1] = replicas[rb][1].copy(), replicas[ra][1].copy()
                    replicas[ra][2], replicas[rb][2] = replicas[rb][2], replicas[ra][2]

    return best_h, best_c5

def dual_swap_sa(h_init, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h_init)
    h = h_init.copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if rng.random() < 0.3:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                continue
            delta = compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
            new_max = float((full + delta).max())
            d = (new_max - float(full.max())) * dx
            if d < 0 or (t > 1e-15 and rng.random() < np.exp(-d / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 += d
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            d = (new_max - float(full.max())) * dx
            if d < 0 or (t > 1e-15 and rng.random() < np.exp(-d / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 += d
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Best-move SA (long, slow)
    allowed = min(250, tleft() * 0.22)
    n_sa = int(allowed * 15000)
    h1, s1 = best_move_sa(best_h, n_sa, 0.025, 0.00005, rng,
                          time_limit=allowed, start_time=start_time, k_arr=k_arr,
                          n_cand=8, top_k=6, explore_frac=0.2)
    if s1 < best_score:
        best_score = s1
        best_h = h1.copy()

    # Phase 2: Parallel tempering
    if tleft() > 200:
        t_levels = [0.02, 0.01, 0.005, 0.002, 0.0008]
        allowed = min(250, tleft() * 0.2)
        n_steps = int(max(5000, allowed * 8000))
        h2, s2 = parallel_tempering(best_h, 5, t_levels, n_steps, rng,
                                     time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    # Phase 3: Dual-swap SA
    if tleft() > 150:
        allowed = min(200, tleft() * 0.18)
        n_ds = int(allowed * 20000)
        h3, s3 = dual_swap_sa(best_h, n_ds, 0.02, 0.0001, rng,
                               time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if s3 < best_score:
            best_score = s3
            best_h = h3.copy()

    # Phase 4: Iterated best-move SA with perturbations
    for restart in range(20):
        if tleft() < 50:
            break
        n_pert = rng.randint(2, 10)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        allowed = min(40, tleft() * 0.035)
        n_sa_r = int(max(20000, allowed * 12000))
        h_r, s_r = best_move_sa(h_pert, n_sa_r, 0.012, 0.00003, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                 n_cand=6, top_k=5, explore_frac=0.35)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r.copy()

    # Phase 5: Another parallel tempering round with lower temps
    if tleft() > 120:
        t_levels2 = [0.012, 0.006, 0.003, 0.001, 0.0004]
        allowed = min(180, tleft() * 0.18)
        n_steps2 = int(max(5000, allowed * 8000))
        h5, s5 = parallel_tempering(best_h, 5, t_levels2, n_steps2, rng,
                                     time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if s5 < best_score:
            best_score = s5
            best_h = h5.copy()

    # Phase 6: Dual-swap refinement
    if tleft() > 80:
        allowed = min(100, tleft() * 0.12)
        n_ds2 = int(max(50000, allowed * 15000))
        h6, s6 = dual_swap_sa(best_h, n_ds2, 0.01, 0.00003, rng,
                               time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if s6 < best_score:
            best_score = s6
            best_h = h6.copy()

    # Phase 7: Final long best-move SA
    if tleft() > 40:
        allowed = max(20, tleft() - 15)
        n_sa_f = int(max(100000, allowed * 20000))
        h_f, s_f = best_move_sa(best_h, n_sa_f, 0.01, 0.000005, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                 n_cand=10, top_k=8, explore_frac=0.15)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f.copy()

    # Final projection and verification
    h_final = project_binary(best_h, target_init)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_init
