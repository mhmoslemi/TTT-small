    w = np.exp(shifted)
    ws = w.sum()
    val = (c_max + np.log(ws) / alpha) * dx if ws > 0 else c_max * dx

        grad[v2] -= w[s] * h[idxs[v2] + k]

    grad -= np.mean(grad)

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            v, _ = smooth_obj_and_grad(x, alpha, n)
            return v
        def _grad(x):
            _, g = smooth_obj_and_grad(x, alpha, n)
            return g
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

def greedy_cd(h0, n, target, time_limit, max_sweeps=10000, n_cand=30, anneal=True):

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_obj(x, alpha, n):
    return smooth_obj_val(x, alpha, n)

def slsqp_grad(x, alpha, n):
    return smooth_obj_grad(x, alpha, n)

def slsqp_con(x, target):
    return np.sum(x) - target

def slsqp_con_jac(x, target):
    return np.ones_like(x)

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
        try:
            res = minimize(slsqp_obj, h, args=(alpha, n),
                          jac=slsqp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lp_refine_round(h0, n, target, time_limit, n_active=15, max_rounds=30):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1][:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in sorted_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k = np.zeros(n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            h_mix = project_feasible(0.5 * h + 0.5 * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return max(0.0, deadline - time.time())

    # Phase 1: SLSQP from initial
    if tleft() > 250:
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [100, 500, 2000, 8000],
                                   2500, 1e-14, min(180.0, tleft() * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Multi-shift CD (considers top 3-5 shifts)
    if tleft() > 250:
        h_cd, c5_cd = multi_shift_cd(best_h, n, target,
                                      time_limit=min(180.0, tleft() * 0.4),
                                      max_sweeps=12000, n_top=3, n_cand=22)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Perturbation search (key new strategy)
    if tleft() > 300:
        h_pert, c5_pert = perturbation_search(best_h, n, target,
                                               time_limit=min(200.0, tleft() * 0.3),
                                               start_time=start_time, seed=seed + 100)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 4: Fourier parameterization (different basin)
    if tleft() > 180:
        M_four = min(40, n // 10)
        h_four, c5_four = fourier_optimize(n, M_four,
                                            min(120.0, tleft() * 0.25),
                                            start_time, seed + 200)
        if tleft() > 60:
            h_four_cd, c5_four_cd = greedy_cd(h_four, n, target,
                                               time_limit=min(50.0, tleft() * 0.2),
                                               max_sweeps=5000, n_cand=20, use_anneal=False)
            if c5_four_cd < best_score:
                best_score = c5_four_cd
                best_h = h_four_cd
        elif c5_four < best_score:
            best_score = c5_four
            best_h = h_four

    # Phase 5: SLSQP refinement
    if tleft() > 200:
        h_p, s_p = slsqp_optimize(best_h, n, target,
                                   [500, 2000, 8000, 30000],
                                   2500, 1e-14, min(140.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 6: Single-shift CD polish (narrow focus on the max)
    if tleft() > 200:
        h_cd, c5_cd = greedy_cd(best_h, n, target,
                                 time_limit=min(150.0, tleft() * 0.35),
                                 max_sweeps=15000, n_cand=30, use_anneal=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: LP refinement
    if tleft() > 120:
        h_lp, c5_lp = lp_refine_round(best_h, n, target,
                                       time_limit=min(90.0, tleft() * 0.3),
                                       n_active=18, max_rounds=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 8: Higher resolution
    if tleft() > 250:
        for n_try in [800, 1200]:
            if tleft() < 180:
                break
            target_try = float(n_try) / 2.0
            x_old = np.linspace(0.0, 1.0, n)
            x_new = np.linspace(0.0, 1.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)
            h_hi, s_hi = slsqp_optimize(h_interp, n_try, target_try,
                                         [100, 500, 2000, 8000],
                                         2000, 1e-13,
                                         min(100.0, tleft() * 0.35), start_time)
            if s_hi < best_score:
                best_score = s_hi
                best_h = h_hi
            if tleft() < 80:
                break
            n_cur = len(best_h)
            t_cur = float(n_cur) / 2.0
            h_cd2, c5_cd2 = greedy_cd(best_h, n_cur, t_cur,
                                       time_limit=min(50.0, tleft() * 0.25),
                                       max_sweeps=8000, n_cand=25, use_anneal=False)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 9: Second perturbation round on best
    if tleft() > 200:
        h_pert2, c5_pert2 = perturbation_search(best_h, len(best_h), float(len(best_h)) / 2.0,
                                                 time_limit=min(120.0, tleft() * 0.3),
                                                 start_time=start_time, seed=seed + 500)
        if c5_pert2 < best_score:
            best_score = c5_pert2
            best_h = h_pert2

    # Phase 10: Final CD + SLSQP
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = greedy_cd(best_h, n_use, target_use,
                                 time_limit=min(50.0, tleft() * 0.3),
                                 max_sweeps=8000, n_cand=30, use_anneal=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
        if tleft() > 30:
            h_p, s_p = slsqp_optimize(best_h, n_use, target_use,
                                       [5000, 20000, 100000],
                                       1500, 1e-15, min(25.0, tleft() * 0.4), start_time)
            if s_p < best_score:
                best_score = s_p
                best_h = h_p

    # Phase 11: Final polish CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = greedy_cd(best_h, n_use, target_use,
                                 time_limit=min(15.0, tleft() * 0.7),
                                 max_sweeps=5000, n_cand=35, use_anneal=False)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
