import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, weights, n, dx):
    """Compute gradient of weighted sum of overlaps.
    weights has length 2n-1, corresponding to np.correlate output indices.
    """
    n_half = n - 1
    grad = np.zeros(n)
    
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - n_half
        if k >= 0:
            # Positive lag k: C[n-1+k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            if k == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            else:
                # j in [0, k-1]: (1-h[j+k])*dx
                grad[:k] += w * dx * (1.0 - h[k:2*k])
                # j in [k, n-k-1]: (1-h[j+k]-h[j-k])*dx
                if n - k > k:
                    mid_len = n - 2 * k
                    grad[k:n-k] += w * dx * (1.0 - h[2*k:n] - h[:mid_len])
                # j in [n-k, n-1]: -h[j-k]*dx
                grad[n-k:] += w * dx * (-h[n-2*k:n-k])
        else:
            m = -k
            # Negative lag m: C[n-1-m] = sum_{i=m}^{n-1} h[i](1-h[i-m])
            # j in [0, m-1]: -h[j+m]*dx
            grad[:m] += w * dx * (-h[m:2*m])
            # j in [m, n-m-1]: (1-h[j-m]-h[j+m])*dx
            if n - m > m:
                mid_len = n - 2 * m
                grad[m:n-m] += w * dx * (1.0 - h[:mid_len] - h[2*m:n])
            # j in [n-m, n-1]: (1-h[j-m])*dx
            grad[n-m:] += w * dx * (1.0 - h[n-2*m:n-m])
    
    return grad
