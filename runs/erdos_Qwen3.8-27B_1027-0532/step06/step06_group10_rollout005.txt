import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        res = minimize(
            lambda x: smooth_obj_val(x, alpha),
            h,
            jac=lambda x: smooth_obj_grad(x, alpha, n),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

        # Find best pair: decrease combined_grad[i], increase combined_grad[j]
        # i has highest combined_grad (decreasing it helps most)
        # j has lowest combined_grad (increasing it helps most)
        sorted_idx = np.argsort(combined_grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Try all pairs and find the best step
        best_delta = 0.0
        best_new_c5 = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-12:
                    continue
                
                # Direction: h[i] -= delta, h[j] += delta
                # Change in corr[s]: delta * (influence)
                # influence of h[i] on corr[s]: (1-h[i-k_s]) - h[i+k_s] (for shift k_s)
                # influence of h[j] on corr[s]: (1-h[j-k_s]) - h[j+k_s]
                # Net change: -delta * infl_i[s] + delta * infl_j[s] = delta * (infl_j[s] - infl_i[s])
                
                # Compute infl vectors for i and j
                infl_i = np.zeros(2*n - 1)
                infl_j = np.zeros(2*n - 1)
                
                for s in range(2*n - 1):
                    k_s = s - mid
                    # h[i] contribution: (1-h[i-k_s]) if 0<=i-k_s<n, minus h[i+k_s] if 0<=i+k_s<n
                    ci = 0.0
                    if 0 <= i - k_s < n:
                        ci += (1.0 - h[i - k_s])
