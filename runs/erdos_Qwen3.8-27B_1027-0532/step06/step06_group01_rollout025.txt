def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

    n = len(h)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    _target = target
    bounds = [(0.0, 1.0)] * n
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def con_fun(x):
        return np.sum(x) - _target
    
    def con_jac(x):
        return np.ones(len(x))
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def find_optimal_deltas(corr, g_array, delta_max_array, n_iter=15):
    """Find optimal delta for each row of g_array.
    corr: shape (n_corr,)
    g_array: shape (n_pairs, n_corr)
    delta_max_array: shape (n_pairs,)
    Returns: shape (n_pairs,)
    """
    n_pairs = g_array.shape[0]
    lo = np.zeros(n_pairs)
    hi = delta_max_array.copy()
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = corr[None, :] + mid[:, None] * g_array
        f_mid = vals.max(axis=1)
        active = vals >= (f_mid[:, None] - 1e-14)
        f_prime = np.where(active, g_array, -np.inf).max(axis=1)
        inc = f_prime > 1e-14
        hi[inc] = mid[inc]
        lo[~inc] = mid[~inc]
    return 0.5 * (lo + hi)

    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = h_init.copy()

    profiles = compute_all_profiles(h)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    ms = np.arange(n)
    n1 = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:

            if no_improve > 500:

        k_star = int(np.argmax(corr)) - n1

        v2 = (ms + k_star >= 0) & (ms + k_star < n)

_SLQ_ALPHA = 0.0
_SLQ_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _slq_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_SLQ_ALPHA * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _SLQ_ALPHA)

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _SLQ_ALPHA = alpha
        try:
            res = minimize(_slq_obj, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def compute_all_profiles(h):
    n = len(h)
    n_corr = 2 * n - 1
    ms = np.arange(n)
    ks = np.arange(n_corr) - (n - 1)
    idx1 = ms[:, None] - ks[None, :]
    idx2 = ms[:, None] + ks[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return h1 + h2

def update_profiles(profiles, i, j, delta, n):

    profiles[:, ms - i + n1] -= delta
    profiles[:, i - ms + n1] -= delta
    profiles[:, ms - j + n1] += delta
    profiles[:, j - ms + n1] += delta

def find_optimal_deltas(corr, g_array, delta_max_arr, n_iter=12):
    n_pairs = g_array.shape[0]
    lo = np.zeros(n_pairs)
    hi = delta_max_arr.copy()
    for _ in range(n_iter):

        vals = corr[None, :] + mid[:, None] * g_array

        active = vals >= (f_mid[:, None] - 1e-14)

        f_prime = g_active.max(axis=1)
        inc = f_prime > 1e-14
        hi[inc] = mid[inc]
        lo[~inc] = mid[~inc]
    return 0.5 * (lo + hi)
