```python
def trust_region_step(h, n, target, n_active=20, max_trust=0.5, min_trust=1e-4):
    """One trust region step for the max-over-shifts problem."""
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr))
    
    # Select active shifts
    sorted_idx = np.argsort(corr)[::-1][:n_active]
    active_shifts = [int(s) - mid for s in sorted_idx]
    
    # Compute gradients for active shifts
    grads = []
    for k in active_shifts:
        g = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += (1.0 - h[idxs[v1] - k])
        g[v2] -= h[idxs[v2] + k]
        grads.append(g)
    
    # Solve trust region subproblem:
    # min t + mu/2 ||p||^2
    # s.t. corr[s] + grads[j]^T p <= t for j in active
    #      h + p in [0,1]^n, sum(h+p) = target
    #      ||p|| <= Δ

    # s.t. grads[j]^T p <= t - corr[s_j] for j in active
    #      sum(p) = 0  (to maintain integral constraint)
    #      -h <= p <= 1-h
    
    # Actually, let's use cvxpy for this
    
    p_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        p_var >= -h,
        p_var <= 1 - h,
        cp.sum(p_var) == 0
```

```python
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):

        else:

        if not mask.any():

        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / wsum
    idxs = np.arange(n)
    mid = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    def _obj(x, alpha):

        return smooth_max_value(corr, alpha) * dx

        return smooth_max_grad(x, alpha, n)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:

                best_h = h_new.copy()
                h = h_new
```

```python
def subspace_step(h, n, target, n_active=15, trust_radius=0.3, time_limit_s=5.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * dx
    
    sorted_idx = np.argsort(corr)[::-1][:n_active]
    
    p_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        p_var >= -h,
        p_var <= 1 - h,
        cp.sum(p_var) == 0,
        cp.norm(p_var, 2) <= trust_radius
    ]
    
    for s_idx in sorted_idx:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad_k = np.zeros(n)
        grad_k[v1] += (1.0 - h[idxs[v1] - k])
        grad_k[v2] -= h[idxs[v2] + k]
        constraints.append(grad_k @ p_var <= t_var - corr[s_idx])
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            prob.solve(solver=cp.CVXOPT)
        except Exception:
            return h, c5_val, trust_radius
    
    if p_var.value is None:
        return h, c5_val, trust_radius
    
    p = p_var.value
    h_new = project_feasible(h + p, target)
    c5_new = compute_c5(h_new)
    
    # Adaptive trust region
    if c5_new < c5_val - 1e-10:
        trust_radius = min(trust_radius * 1.3, 0.5)
    else:
        trust_radius = trust_radius * 0.5
    
    return h_new, c5_new, trust_radius
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / wsum
    return w

def slsqp_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, alpha) * (2.0 / n)

def slsqp_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = smooth_max_weights(corr, alpha)
    idxs = np.arange(n)
    mid = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def eq_con_fun(x, target):
    return np.sum(x) - target

def eq_con_jac(x, target):
    return np.ones(len(x))

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (target,)}]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_obj, h, args=(alpha, n),
                           jac=slsqp_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def fast_cd(h0, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        best_g = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(30):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
                    best_g = g
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def subspace_round(h0, n, target, time_limit, max_rounds, n_active):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trust = 0.4
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        dx = 2.0 / n
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1][:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            h_var >= 0, h_var <= 1, cp.sum(h_var) == target
        ]
        for s_idx in sorted_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            lin_coeff = np.zeros(n)
            lin_coeff[v1] += (1.0 - h[idxs[v1] - k])
            lin_coeff[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + lin_coeff @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            h_mix = project_feasible(0.5 * h + 0.5 * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def randomized_restart(h, n, target, seed, scale):
    rng = np.random.RandomState(seed)
    h_pert = h + rng.randn(n) * scale
    return project_feasible(h_pert, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with aggressive alpha ramp
    tl = deadline - time.time()
    if tl > 180:
        h_p, s_p = slsqp_round(best_h, n, target,
                               [100, 500, 2000, 10000, 50000, 200000],
                               5000, 1e-16, tl * 0.45, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: CD with 30 candidates
    tl = deadline - time.time()
    if tl > 200:
        cd_time = min(250.0, tl * 0.35)
        h_cd, c5_cd = fast_cd(best_h, n, target,
                               time_limit=cd_time, max_sweeps=20000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement with very high alpha
    tl = deadline - time.time()
    if tl > 150:
        h_p, s_p = slsqp_round(best_h, n, target,
                               [500, 2000, 10000, 50000, 200000, 1000000],
                               4000, 1e-16, tl * 0.35, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: Subspace / LP refinement
    tl = deadline - time.time()
    if tl > 100:
        h_lp, c5_lp = subspace_round(best_h, n, target,
                                      time_limit=min(100.0, tl * 0.35),
                                      max_rounds=50, n_active=20)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: CD with larger candidate set
    tl = deadline - time.time()
    if tl > 150:
        cd_time = min(180.0, tl * 0.35)
        h_cd, c5_cd = fast_cd(best_h, n, target,
                               time_limit=cd_time, max_sweeps=15000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Multi-resolution to n=800
    tl = deadline - time.time()
    if tl > 220:
        n_try = 800
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, s_hires = slsqp_round(h_interp, n_try, target_try,
                                        [50, 200, 1000, 5000, 20000, 100000, 500000],
                                        3000, 1e-15,
                                        min(130.0, tl * 0.4), start_time)
        if s_hires < best_score:
            best_score = s_hires
            best_h = h_hires
            n = n_try
            target = target_try
        else:
            tl2 = deadline - time.time()
            if tl2 > 80:
                h_cd2, c5_cd2 = fast_cd(h_hires, n_try, target_try,
                                         time_limit=min(70.0, tl2 * 0.3),
                                         max_sweeps=10000, n_cand=25)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2
                    n = n_try
                    target = target_try

    # Phase 7: SLSQP + CD at current best resolution
    tl = deadline - time.time()
    if tl > 120:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [1000, 5000, 20000, 100000, 500000, 1000000],
                                3000, 1e-16, min(60.0, tl * 0.3), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p
        tl = deadline - time.time()
        if tl > 50:
            h_cd, c5_cd = fast_cd(best_h, n, target,
                                   time_limit=min(40.0, tl * 0.4),
                                   max_sweeps=10000, n_cand=35)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Randomized restarts
    tl = deadline - time.time()
    if tl > 80:
        for restart_seed in [seed + 100, seed + 200, seed + 300]:
            if deadline - time.time() < 30:
                break
            h_pert = randomized_restart(best_h, n, target, restart_seed, 0.02)
            h_p, s_p = slsqp_round(h_pert, n, target,
                                    [200, 2000, 20000, 200000],
                                    2000, 1e-15,
                                    min(25.0, (deadline - time.time()) * 0.3), start_time)
            if s_p < best_score:
                best_score = s_p
                best_h = h_p

    # Phase 9: Final subspace
    tl = deadline - time.time()
    if tl > 40:
        h_lp, c5_lp = subspace_round(best_h, n, target,
                                      time_limit=min(30.0, tl * 0.4),
                                      max_rounds=25, n_active=25)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Final SLSQP + CD polish
    tl = deadline - time.time()
    if tl > 20:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [5000, 50000, 500000, 1000000],
                                2000, 1e-16,
                                min(15.0, tl * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p
    tl = deadline - time.time()
    if tl > 10:
        h_cd, c5_cd = fast_cd(best_h, n, target,
                               time_limit=min(8.0, tl * 0.7),
                               max_sweeps=5000, n_cand=35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)
    return h_out, final_score, n_final
```
