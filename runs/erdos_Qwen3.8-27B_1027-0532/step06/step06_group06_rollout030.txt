- If we flip h[i] from 1 to 0:

  - Net change in corr(k): -(1-h[i+k]) + h[i-k] = h[i+k] - 1 + h[i-k] (for binary h)
    = (h[i+k] + h[i-k] - 1)

- If we flip h[j] from 0 to 1:

  - Net change in corr(k): (1-h[j+k]) - h[j-k]

This is very simple! For binary h, the change in corr(k) from swapping positions i and j is:

Let me just be conservative and estimate 5 gradient evaluations per step * 500 steps * 10 betas = 25000 gradient evaluations. Each takes ~200K ops. Total: 5B ops. At 1G ops/s, about 5 seconds. Very fast!

So we can afford to be much more aggressive:

- 15-20 beta values
- Multiple restarts

This should give much better convergence than the previous approach.

Now for the binary search, the key improvement is the fast swap evaluation:
Δcorr(k) = h[i+k] + h[i-k] - h[j+k] - h[j-k] (for binary h)

For a given swap (i, j), the new max is:
new_max = max_k (corr[k] + Δcorr(k))

We can compute this for all k in O(n) time. But we can be smarter:
- Only k values where corr[k] + Δcorr(k) > current_max matter
- Δcorr(k) is bounded by 2 (since h values are 0 or 1)
- So only k where corr[k] ≥ current_max - 2 can potentially exceed the current max

This means we only need to check O(1) or O(few) k values per swap!

For the greedy search:
- Find the k* that achieves the maximum
- Find all i with h[i]=1 that can be "moved" (h[i+k*]=0)
- Find all j with h[j]=0 that can "receive" (h[j+k*]=1)
- For each pair, evaluate the new max (checking only critical k values)
- Pick the best swap

This is O(n²) per iteration in the worst case, but with the O(1) evaluation per pair, it's much faster than the previous O(n) per pair.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    if abs(h.sum() - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = -3.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def fft_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    OH = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(OH), N)[:2 * n - 1]

def smooth_max_val(corr, beta):
    c_max = corr.max()
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    return c_max + np.log(np.sum(np.exp(shifted))) / beta

def smooth_weights_fn(corr, beta):
    c_max = corr.max()
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    w = np.exp(shifted)
    return w / w.sum()

def obj_fftc(h, beta, target, penalty):
    corr = fft_corr(h)
    val = smooth_max_val(corr, beta)
    d = h.sum() - target
    return val + penalty * d * d

def grad_fftc(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    W = smooth_weights_fn(corr, beta)
    N = 1
    while N < 3 * n:
        N *= 2
    h_pad = np.zeros(N)
    h_pad[:n] = h
    W_pad = np.zeros(N)
    W_pad[:2 * n - 1] = W
    cross_corr = np.fft.irfft(np.fft.rfft(h_pad, N) * np.conj(np.fft.rfft(W_pad, N)), N)
    conv = np.fft.irfft(np.fft.rfft(h_pad, N) * np.fft.rfft(W_pad, N), N)
    A = cross_corr[n - 1::-1][:n]
    B = conv[n - 1:2 * n - 1]
    prefix_W = np.zeros(2 * n)
    prefix_W[1:] = np.cumsum(W)
    C1 = np.zeros(n)
    for j in range(n):
        lo_i = max(0, n - 1 - j)
        hi_i = 2 * n - 2 - j
        C1[j] = prefix_W[hi_i + 1] - prefix_W[lo_i]
    grad = dx * (C1 - A - B)
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - target)
    return grad

def lbfgs_opt(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for _ in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_fftc, h_cur, args=(beta, target, penalty),
                           jac=grad_fftc, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'iprint': -1, 'maxls': 50})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

            i_cand = [i for i in ones if 0 <= i + k_star < n]
            j_cand = [j for j in zeros if 0 <= j + k_star < n]
            if len(i_cand) == 0 or len(j_cand) == 0:

                j_cand = list(zeros[:scan_w])
                if len(i_cand) == 0 or len(j_cand) == 0:
                    break
        n_i = min(len(i_cand), scan_w)

        h_f = h.astype(float)
