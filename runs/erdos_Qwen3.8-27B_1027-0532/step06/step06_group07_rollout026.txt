```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, alpha, mu, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    lse = c_max + np.log(np.sum(np.exp(alpha * (corr - c_max)))) / alpha
    return lse + mu * (np.sum(h) - target) ** 2

def lse_gradient(h, alpha, mu, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(alpha * (corr - c_max))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    mid = n - 1
    
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-18:
            continue
        k = m_idx - mid
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:-k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] -= w * h[kk:]
    
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - target)
    
    return grad

def optimize_lse(h_init, n, target, alphas, maxiter, mu_schedule):
    h_cur = h_init.copy()
    for i, alpha in enumerate(alphas):
        mu = mu_schedule[i] if i < len(mu_schedule) else mu_schedule[-1]
        res = minimize(
            lse_objective, h_cur,
            args=(alpha, mu, target),
            jac=lambda x, a, m, t: lse_gradient(x, a, m, t),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12}
        )
        h_new = project_feasible(res.x, target)
        if compute_c5(h_new) < compute_c5(h_cur):
            h_cur = h_new
    return h_cur

def pair_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):

    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
```

```python
                best_t = t

        if best_t == 0.0:
            break

        h_cur = project_feasible(h_cur + best_t * delta, target)

        spread = np.max(corr[active]) - np.min(corr[active])
        if spread < tol * c_max:
            break

    return h_cur


def spectral_perturb(h, n, target, rng, scale):
    h_ft = np.fft.rfft(h)
    n_freq = len(h_ft)
    mag = np.abs(h_ft)
    phase = np.angle(h_ft)
    phase += rng.randn(n_freq) * scale
    mag *= (1.0 + rng.randn(n_freq) * scale * 0.5)
    mag = np.maximum(mag, 0.0)
    h_new = np.fft.irfft(mag * np.exp(1j * phase), n=n)
    return project_feasible(h_new, target)


def random_perturb(h, n, target, rng, scale):
    h_new = h + rng.randn(n) * scale
    return project_feasible(h_new, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_restart_optimize(h_init, n, target, rng, time_limit, n_restarts=5):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    start = time.time()

    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break

        scale = 0.03 + 0.07 * r
        if r % 2 == 0:
            h_pert = spectral_perturb(best_h, n, target, rng, scale * 0.5)
        else:
            h_pert = random_perturb(best_h, n, target, rng, scale)

        alphas = [300, 1000, 5000, 20000]
        mu = 1e8
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        per_alpha = max(50, int(min(300, remaining / (len(alphas) * 2))))
        h_opt = optimize_lbfgs(h_pert, n, target, alphas, per_alpha, mu)
        sc = compute_c5(h_opt)

        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: L-BFGS-B optimization with increasing alpha
    if tleft() > 150:
        alpha_list = [100, 300, 1000, 3000, 10000]
        h_pol = optimize_lbfgs(best_h, n, target, alpha_list, 400, 1e8)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active set flattening
    if tleft() > 100:
        for na in [5, 10, 20, 40, 80]:
            if tleft() < 20:
                break
            h_flat = flatten_active(best_h, n, target, na, max_iter=200)
            sc = compute_c5(h_flat)
            if sc < best_score:
                best_score = sc
                best_h = h_flat

    # Phase 3: Pair coordinate descent (main workhorse)
    if tleft() > 250:
        cd_time = min(350, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=10000, n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Another L-BFGS-B with higher alpha
    if tleft() > 150:
        alpha_list2 = [10000, 50000, 200000]
        h_pol2 = optimize_lbfgs(best_h, n, target, alpha_list2, 600, 1e9)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 5: Active set flattening again
    if tleft() > 80:
        for na in [10, 30, 60]:
            if tleft() < 15:
                break
            h_flat = flatten_active(best_h, n, target, na, max_iter=300)
            sc = compute_c5(h_flat)
            if sc < best_score:
                best_score = sc
                best_h = h_flat

    # Phase 6: Multi-restart with spectral/random perturbations
    if tleft() > 200:
        mr_time = min(200, tleft() * 0.2)
        h_mr, c5_mr = multi_restart_optimize(best_h, n, target, rng,
                                             time_limit=mr_time, n_restarts=6)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 7: Final pair CD polish
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=12000, n_cand=55, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Try higher resolution
    if tleft() > 300:
        for n_try in [700, 800, 900]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            alpha_list3 = [200, 1000, 5000, 20000]
            h_hires = optimize_lbfgs(h_interp, n_try, target_try, alpha_list3, 400, 1e8)
            sc = compute_c5(h_hires)
            if sc < best_score + 0.003 and tleft() > 80:
                cd_t = min(60, tleft() * 0.2)
                h_hires2, sc2 = pair_cd(h_hires, n_try, target_try,
                                        time_limit=cd_t,
                                        max_sweeps=4000, n_cand=40, n_bisect=14)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2

    # Phase 9: Final L-BFGS-B polish at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alpha_list4 = [50000, 200000, 500000]
        h_final = optimize_lbfgs(best_h, n_use, target_use, alpha_list4, 500, 1e9)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 10: Very short final CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                              time_limit=min(25, tleft()),
                              max_sweeps=3000, n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
