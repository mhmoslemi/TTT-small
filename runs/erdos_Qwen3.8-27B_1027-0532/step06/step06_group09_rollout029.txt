def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

            h_temp = h.copy()
            h_temp[free_idx] = np.clip(x, 0.0, 1.0)
            corr = np.correlate(h_temp, 1.0 - h_temp, mode='full')
            c_max = np.max(corr)
            w = alpha * np.exp(np.clip(alpha * (corr - c_max), -500, 0))
            w = w / (w.sum() + 1e-30)
            g_full = compute_gradient_weighted(h_temp, w, n, n - 1)
            return g_full[free_idx]

        x0 = h[free_idx]
        bounds = [(0.0, 1.0)] * n_free
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_free,
                 'jac': lambda x: np.ones(n_free)}]

        try:
            res = minimize(obj_sub, x0, jac=obj_sub_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_new = h.copy()
            h_new[free_idx] = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                # Try unfreezing: nudge a boundary variable
                at_zero = np.where(h <= tol)[0]
                at_one = np.where(h >= 1.0 - tol)[0]
                unfrozen = False
                if len(at_zero) > 0:
                    i = at_zero[0]
                    h[i] = tol * 0.5
                    unfrozen = True
                elif len(at_one) > 0:
                    i = at_one[0]
                    h[i] = 1.0 - tol * 0.5
                    unfrozen = True
                h = project_feasible(h, target)
                if not unfrozen and rnd > 50:
                    break
        except Exception:
            pass

    return best_h, best_c5

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    _gn = n
    _gt = target

    def cons_fun(x):
        return np.sum(x) - _gt

    def cons_jac(x):
        return np.ones(_gn)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_local(h_init, n, target, time_limit, seed=123):
    start = time.time()
    rng_local = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    while time.time() - start < time_limit:
        # Try a structured perturbation
        scale = float(rng_local.choice([0.02, 0.04, 0.06, 0.1, 0.15]))
        n_pert = int(rng_local.choice([3, 5, 8, 12, 20, 30, 50]))
        n_pert = min(n_pert, n)
        pidx = rng_local.choice(n, size=n_pert, replace=False)
        h_try = h_cur.copy()
        h_try[pidx] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)

        # Quick EM polish from this start
        h_pol, c5_pol = cooling_em(h_try, n, target,
                                    time_limit=min(5, time_limit - (time.time() - start)),
                                    eta_start=50.0, eta_end=200.0,
                                    m_top_start=10, m_top_end=30,
                                    max_steps=2000, base_lr=0.05)

        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_cur = h_pol

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Cooling EM (main workhorse, 250s)
    if tleft() > 250:
        em_time = min(250, tleft() * 0.2)
        h_em, c5_em = cooling_em(best_h, n, target,
                                  time_limit=em_time,
                                  eta_start=3.0, eta_end=300.0,
                                  m_top_start=3, m_top_end=50,
                                  max_steps=30000, base_lr=0.1)
        if c5_em < best_score:
            best_score = c5_em
            best_h = h_em

    # Phase 2: Aggressive pair CD (200s)
    if tleft() > 200:
        cd_time = min(200, tleft() * 0.2)
        h_cd, c5_cd = aggressive_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          n_cand_start=40, n_cand_max=120,
                                          patience=250, max_sweeps=15000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Subspace optimization (150s)
    if tleft() > 150:
        sub_time = min(150, tleft() * 0.15)
        h_sub, c5_sub = subspace_optimize(best_h, n, target,
                                           time_limit=sub_time, tol=0.03,
                                           max_rounds=300)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 4: Another cooling EM round with different params (150s)
    if tleft() > 150:
        em2_time = min(150, tleft() * 0.15)
        h_em2, c5_em2 = cooling_em(best_h, n, target,
                                    time_limit=em2_time,
                                    eta_start=20.0, eta_end=500.0,
                                    m_top_start=10, m_top_end=40,
                                    max_steps=20000, base_lr=0.05)
        if c5_em2 < best_score:
            best_score = c5_em2
            best_h = h_em2

    # Phase 5: L-BFGS-B polish (100s)
    if tleft() > 100:
        for alpha in [1000, 5000, 20000, 80000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 6: Higher resolution attempt (200s)
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            # Quick EM at higher res
            h_hr, c5_hr = cooling_em(h_interp, n_try, target_try,
                                      time_limit=min(80, tleft() * 0.3),
                                      eta_start=5.0, eta_end=200.0,
                                      m_top_start=5, m_top_end=40,
                                      max_steps=10000, base_lr=0.06)
            if c5_hr < best_score:
                # Quick pair CD at higher res
                h_hr2, c5_hr2 = aggressive_pair_cd(h_hr, n_try, target_try,
                                                    time_limit=min(40, tleft() * 0.2),
                                                    n_cand_start=30, n_cand_max=60,
                                                    patience=100, max_sweeps=5000)
                if c5_hr2 < best_score:
                    best_score = c5_hr2
                    best_h = h_hr2
                    n = n_try
                    target = target_try
                else:
                    best_score = c5_hr
                    best_h = h_hr
                    n = n_try
                    target = target_try

    # Phase 7: Final pair CD on best (100s)
    if tleft() > 100:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_cd2, c5_cd2 = aggressive_pair_cd(best_h, n_use, t_use,
                                            time_limit=min(80, tleft() * 0.3),
                                            n_cand_start=50, n_cand_max=100,
                                            patience=200, max_sweeps=10000)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Multi-restart local search (80s)
    if tleft() > 80:
        mr_time = min(80, tleft() * 0.2)
        h_mr, c5_mr = multi_restart_local(best_h, len(best_h), len(best_h) / 2.0,
                                           time_limit=mr_time, seed=seed)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 9: Final SLSQP polish (50s)
    if tleft() > 50:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_fin = slsqp_round(best_h, n_use, t_use,
                            [5000, 20000, 80000], 500, 1e-15)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility
    n_use = len(best_h)
    t_use = n_use / 2.0
    h_final = project_feasible(best_h, t_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
