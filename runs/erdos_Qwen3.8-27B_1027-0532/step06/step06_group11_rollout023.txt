def proximal_projection(h, k, target_corr, n, max_iter=100):
    """Project h to reduce corr_k to target_corr, with minimal L2 change."""
    dx = 2.0 / n
    mid = n - 1
    h = h.copy()
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        current = corr[k + mid] * dx
        if current <= target_corr:
            return h
        # Compute gradient of corr_{k} with respect to h
        # d/dh_i [correlation at shift k] 
        # corr_{k} = sum_i h_i * (1 - h_{i+k}) (with appropriate indexing)
        # d corr_k / dh_i = (1 - h_{i+k}) for the h_i term
        #                     - h_{i-k} for the (1-h_{i}) term (when h appears as the second term)
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)

        h_new = h - step_size * grad
        h_new = project_feasible(h_new, n/2)
        h = h_new
    return h

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def constraint_fun(h):
        return np.sum(h) - target
    
    def constraint_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            max_C = np.max(C)
            shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
            return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)
        
        def obj_grad(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            max_C = np.max(C)
            shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
            weights = np.exp(shifted)
            wsum = weights.sum()
            grad = np.zeros(n)
            mid = n - 1
            for s in range(len(C)):
                if weights[s] / wsum < 1e-6:
                    continue
                k = s - mid
                grad += (weights[s] / wsum) * get_grad_shift(h, k, n)
            grad *= dx
            return grad
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5
```

def compute_profiles(h, n):
    """Precompute profile[i] for each position i.
    profile[i][s] = h[i-k] + h[i+k] where k = s-(n-1), with 0 for out of bounds."""
    profiles = np.zeros((n, 2*n-1))
    k_vals = np.arange(-(n-1), n)  # k values for s = 0, ..., 2n-2

        mask2 = (ik2 >= 0) & (ik2 < n)
        profiles[i, mask1] += h[ik[mask1]]
        profiles[i, mask2] += h[ik2[mask2]]
    return profiles

Actually, minimizing Σ_k corr_k = Σ_k Σ_i h[i](1-h[i+k]) = Σ_i h[i]·(number of valid j where j=i+k for some k) - Σ_k Σ_i h[i]h[i+k]

The first term is Σ_i h[i]·(n-1) (approximately, for interior points) = (n-1)·Σ_i h[i] = (n-1)·n/2 (by the constraint). So it's constant!

The second term is Σ_k Σ_i h[i]h[i+k] = Σ_k (autocorrelation of h at lag k). This is the energy of the Fourier transform of h (by Parseval). So minimizing Σ_k corr_k is equivalent to maximizing Σ_k autocorr_k = (1/n)·Σ|ĥ(ω)|² (up to constants).

Wait, that means minimizing the average overlap is equivalent to maximizing the L2 norm of h in Fourier space, which means making h as "concentrated" as possible. This would push h towards a step function (concentrated in frequency).

But we want to minimize the MAX, not the average. So this doesn't directly help.

Let me try another angle. The key constraint is that the maximum correlation should be small. If I think of the correlations as a "spectrum", I want to flatten it. This is like a "zero-padding" or "equalization" problem.

In signal processing, one way to flatten a spectrum is to use an IIR filter or to use a specific modulation. In our case, the "filter" is the function h itself.

I think the right approach is still the iterative optimization I've been planning. Let me just make it more efficient and add the binary local search as a new element.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_alpha = 0.0
_G_dx = 0.0
_G_target = 0.0
_G_n = 0

def _slsqp_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * _G_dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def _slsqp_grad(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * _G_dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    mid = _G_n - 1
    grad = np.zeros(_G_n)
    for s in range(len(C)):
        w = weights[s] / wsum
        if w < 1e-8:
            continue
        k = s - mid
        grad += w * get_grad_shift(h, k, _G_n)
    grad *= _G_dx
    return grad

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_alpha, _G_dx, _G_target, _G_n
    _G_n = n
    _G_target = target
    _G_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def compute_profiles(h, n):
    profiles = np.zeros((n, 2 * n - 1))
    k_vals = np.arange(-(n - 1), n)
    for i in range(n):
        ik = i - k_vals
        mask1 = (ik >= 0) & (ik < n)
        if mask1.any():
            profiles[i, mask1] += h[ik[mask1]]
        ik2 = i + k_vals
        mask2 = (ik2 >= 0) & (ik2 < n)
        if mask2.any():
            profiles[i, mask2] += h[ik2[mask2]]
    return profiles

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        profiles = compute_profiles(h, n)
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = profiles[i] - profiles[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def binary_local_search(h_bin, n, target, time_limit, max_steps, rng):
    start = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = get_grad_shift(h, k_star, n)
        ones_grad = grad[ones]
        zeros_grad = grad[zeros]
        i_best = ones[np.argmin(ones_grad)]
        j_best = zeros[np.argmax(zeros_grad)]
        # Try this swap
        h[i_best] = 0.0
        h[j_best] = 1.0
        new_c5 = compute_c5(h)
        # Also try a few random swaps
        best_trial = new_c5
        best_i, best_j = i_best, j_best
        n_rand = min(10, len(ones), len(zeros))
        for _ in range(n_rand):
            io = rng.randint(len(ones))
            iz = rng.randint(len(zeros))
            i_r, j_r = ones[io], zeros[iz]
            h[i_r] = 0.0
            h[j_r] = 1.0
            tc5 = compute_c5(h)
            h[i_r] = 1.0
            h[j_r] = 0.0
            if tc5 < best_trial:
                best_trial = tc5
                best_i, best_j = i_r, j_r
        # Reset and apply best
        h[i_best] = 1.0
        h[j_best] = 0.0
        h[best_i] = 0.0
        h[best_j] = 1.0
        # Update ones/zeros
        ones = np.delete(ones, np.where(ones == best_i)[0][0])
        zeros = np.delete(zeros, np.where(zeros == best_j)[0][0])
        ones = np.append(ones, best_j)
        zeros = np.append(zeros, best_i)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
