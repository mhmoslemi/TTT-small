    """Alternating LP with all constraints."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1

        if time.time() - start > time_limit:
            break

        g = 1.0 - h_fixed

        else:
            l = -k

    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A @ h - t <= 0 for each row
    # [A, -1] @ [h, t] <= 0

    A_eq = np.hstack([np.ones((1, n)), [[0]]])
    b_eq = np.array([target])

    bounds = [(0, 1)] * n + [(0, None)]

def frank_wolfe(h_init, n, target, time_limit, max_iter=200000, K=45):

        top_idx = np.argsort(overlaps)[::-1][:K]

                                      damping_schedule=(1.0, 0.9))
        if c5_rl < best_score:
            best_score = c5_rl
            best_h = h_rl

    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        h_sls2 = slsqp_optimize(best_h, n, target,
                                 [2000, 10000, 50000, 200000],
                                 maxiter=4000, ftol=1e-16)
        sc = compute_c5(h_sls2)
        if sc < best_score:
            best_score = sc
            best_h = h_sls2

    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        lp_time = min(130, tl * 0.4)
        h_lp4, c5_lp4 = alternating_lp(best_h, n, target, lp_time,
                                        max_iter=6000, n_top=None,
                                        damping_schedule=(1.0, 0.98))
        if c5_lp4 < best_score:
            best_score = c5_lp4
            best_h = h_lp4

    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_sls3 = slsqp_optimize(best_h, n, target,
                                 [10000, 50000, 200000, 1000000],
                                 maxiter=5000, ftol=1e-16)
        sc = compute_c5(h_sls3)
        if sc < best_score:
            best_score = sc
            best_h = h_sls3

    tl = time_remaining(start_time, budget_s)
    if tl > 60:
        lp_time = min(50, tl * 0.7)
        h_lp5, c5_lp5 = alternating_lp(best_h, n, target, lp_time,
                                        max_iter=3000, n_top=None,
                                        damping_schedule=(1.0, 0.99))
        if c5_lp5 < best_score:
            best_score = c5_lp5
            best_h = h_lp5

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = project_feasible(best_h, target_use)
        final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
