        else:

    lam = (lo + hi) / 2.0

def project_binary(h, target_sum):

    current_sum = int(h_r.sum())
    if current_sum == target_sum:

    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):

        if w < 1e-13:

        grad[:n - s] += w * one_minus_h[s:]

        if w < 1e-13:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

            base_grad[pp:] += w * one_minus_h[:n - pp]
            base_grad[:n - pp] -= w * h[pp:]
    pen_grad = 2.0 * mu * (float(np.sum(h)) - target) * np.ones(n)
    return base_grad + pen_grad


def lbfgsb_phase(h_start, n, target, time_limit, start_time):
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    p = 20
    mus = [0.1, 1.0, 10.0, 100.0, 1000.0]
    for mu in mus:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                return lnorm_penalty_obj(x, p, mu, target)

            def grad(x):
                return lnorm_penalty_grad(x, p, mu, target)

            bounds = [(0.0, 1.0)] * n
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def binary_hill_climb(h, target_sum, max_iters, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue

        top_threshold = max_val * 0.90
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        n_top = min(len(top_indices), 10)
        top_indices = top_indices[:n_top]

        scan_w = 30
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_w)
        n_j = min(len(j_cand), scan_w)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def make_periodic_start(n, period, duty, offset, rng):
    h = np.zeros(n)
    x = np.arange(n) * (2.0 / n)
    phase = (x + offset) % period
    h[phase < duty * period] = 1.0
    h = h + rng.randn(n) * 0.01
    return project_box_sum(h, n / 2.0)


def make_random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_block_start(n, rng):
    h = np.zeros(n)
    num_blocks = rng.randint(3, 7)
    block_size = n // (num_blocks * 2)
    for k in range(num_blocks):
        start = rng.randint(0, max(1, n - block_size))
        h[start:start + block_size] = 1.0
    h = project_box_sum(h, n / 2.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick binary hill climb from initial
    if tleft() > 120:
        h_bin = project_binary(best_h, int(target))
        h_g, c5_g = binary_hill_climb(h_bin, int(target), max_iters=50000,
                                      time_limit=min(40, tleft() * 0.1),
                                      start_time=start_time)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Adam with log-sum-exp (main continuous optimization)
    if tleft() > 300:
        adam_time = min(400, tleft() - 150)
        h_a, s_a = adam_run(
            best_h, target, start_time, adam_time,
            alphas=[50, 100, 200, 500, 1000, 2500, 5000, 10000, 25000, 50000],
            lr_init=0.003, lr_min=0.0003,
            beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 3: L-BFGS-B with p-norm penalty
    if tleft() > 200:
        lbfgs_time = min(200, tleft() - 100)
        h_lb, s_lb = lbfgsb_phase(best_h, n, target, lbfgs_time, start_time)
        if s_lb < best_score:
            best_score = s_lb
            best_h = h_lb.copy()

    # Phase 4: Multi-restart Adam from diverse starting points
    if tleft() > 250:
        starts = []
        starts.append(best_h + rng.randn(n) * 0.004)
        starts.append(make_periodic_start(n, 1.0, 0.4, 0.0, rng))
        starts.append(make_periodic_start(n, 0.5, 0.5, 0.1, rng))
        starts.append(make_random_binary(n, rng))
        starts.append(project_box_sum(best_h + rng.randn(n) * 0.008, target))
        starts.append(make_block_start(n, rng))
        starts.append(make_periodic_start(n, 2.0/3.0, 0.5, 0.05, rng))

        n_restarts = len(starts)
        time_per = (tleft() - 100) / n_restarts
        time_per = max(30.0, min(time_per, 120.0))

        for ri in range(n_restarts):
            if tleft() < 80:
                break
            h_pert = project_box_sum(starts[ri], target)
            h_r, s_r = adam_run(
                h_pert, target, start_time, time_per,
                alphas=[100, 500, 2000, 8000, 30000],
                lr_init=0.002, lr_min=0.0005,
                beta1=0.9, beta2=0.999, eps=1e-8
            )
            if s_r < best_score:
                best_score = s_r
                best_h = h_r.copy()

    # Phase 5: Binary hill climb from best
    if tleft() > 150:
        h_bin2 = project_binary(best_h, int(target))
        hc_time = min(150, tleft() - 60)
        h_hc, c5_hc = binary_hill_climb(h_bin2, int(target), max_iters=200000,
                                        time_limit=hc_time, start_time=start_time)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc.copy()

    # Phase 6: SLSQP polish
    if tleft() > 80:
        slsqp_time = min(120, tleft() - 40)
        h_ref, sc_ref = slsqp_refine(best_h, n, target,
                                     alphas=[5000, 15000, 50000, 150000],
                                     maxiter=2000, ftol=1e-16,
                                     time_limit=slsqp_time, start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 7: Final high-alpha Adam polish
    if tleft() > 40:
        polish_time = max(15.0, tleft() - 15)
        h_p, s_p = adam_run(
            best_h, target, start_time, polish_time,
            alphas=[50000, 100000, 200000],
            lr_init=0.0008, lr_min=0.0001,
            beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
