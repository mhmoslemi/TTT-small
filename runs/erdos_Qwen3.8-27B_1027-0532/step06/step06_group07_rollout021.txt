def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, corr, n, top_m=10):

        active = np.argsort(corr[-top_m:]) + (2*n - 1 - top_m)
    
    grad = np.zeros(n)

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_objective(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def smooth_objective(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def subgradient_method(h_init, n, target, time_limit, m_active=10, 
                       t0=0.1, decay=0.9995, restart_interval=5000):

        corr = np.correlate(h, 1.0 - h, mode='full')

        c_max = np.max(corr)
        active_thresh = c_max - max(1e-6, 0.005 * c_max)

        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)

        grad -= np.mean(grad)
        
        # Polyak step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break
        gap = c5_now - T_lower
        if gap <= 0:
            break
        t = min(gap / (grad_norm_sq * dx * dx), t0)
        
        # Take step
        h_new = project_feasible(h - t * grad, target)
        
        num_steps += 1
        h_avg = (h_avg * (num_steps - 1) + h_new) / num_steps
        
        # Check averaged iterate
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5 - 1e-14:
            best_c5 = c5_avg
            best_h = h_avg.copy()
        
        h = h_new
    
    # Also try the averaged solution
    c5_avg = compute_c5(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg.copy()
    
    return best_h, best_c5

def interp_to_resolution(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i + np.zeros(n_corr)  # copy
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]
                
                # Bisection
                a_d, b_d = 0.0, delta_max
                for _ in range(n_bisect):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    act_b = vals >= f_m - 1e-14
                    fp = np.max(g[act_b]) if act_b.any() else 0.0
                    if fp > 1e-14:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if d_star < 1e-12:
                    continue
                trial = np.max(corr + d_star * g) * dx
                if trial < best_trial - 1e-15:
                    best_trial = trial
                    best_i = i
                    best_j = j
                    best_delta = d_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, target)
        elif no_improve > 40:
            break
    
    return best_h, best_c5
