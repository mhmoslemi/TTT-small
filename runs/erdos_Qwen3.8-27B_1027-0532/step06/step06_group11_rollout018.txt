    if wsum < 1e-300:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(-h)
    per = diff / len(order)
    for i in order:
        if diff > 0:
            add = min(per, 1.0 - h[i])
            if add > 1e-15:
                h[i] += add
                diff -= add
        else:
            rem = min(-per, h[i])
            if rem > 1e-15:
                h[i] -= rem
                diff += rem
        if abs(diff) < 1e-13:
            break
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def downsample_block(h, factor):
    n = len(h)
    m = n // factor
    h_blocks = h[:m * factor].reshape(m, factor)
    h_small = h_blocks.mean(axis=1)
    return h_small

_G = {}

def _obj_fn(h):
    a = _G['alpha']
    lam = _G['lam']
    tgt = _G['target']
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    mC = np.max(C)
    sh = a * (C - mC)
    val = mC + np.log(np.sum(np.exp(sh))) / a
    val += lam * (np.sum(h) - tgt)
    return float(val)

def _grad_fn(h):
    a = _G['alpha']
    lam = _G['lam']
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    mC = np.max(C)
    lw = a * (C - mC)
    lw -= np.max(lw)
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-300:
        return np.full(n, lam)
    w /= ws
    g = np.zeros(n)
    mid = n - 1
    idxs = np.where(w > 1e-10)[0]
    for k in idxs:
        lag = k - mid
        if lag >= 0:
            L = n - lag
            if L > 0:
                g[:L] += w[k] * (1.0 - h[lag:])
                g[lag:] -= w[k] * h[:L]
        else:
            j = -lag
            L = n - j
            if L > 0:
                g[j:] += w[k] * (1.0 - h[:L])
                g[:L] -= w[k] * h[j:]
    g *= dx
    g += lam
    return g

def _get_fg(h):
    return _obj_fn(h), _grad_fn(h)

def lbfgs_dual(h_start, target, alphas, max_iter, dual_iters, dual_lr, time_budget):
    start = time.time()
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    for alpha in alphas:
        if time.time() - start > time_budget:
            break
        for d in range(dual_iters):
            if time.time() - start > time_budget:
                break
            _G['alpha'] = alpha
            _G['lam'] = lam
            _G['target'] = target
            try:
                res = minimize(_obj_fn, h, jac=_grad_fn, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
                h_cand = res.x
            except Exception:
                h_cand = h
            h_cand = project_feasible(h_cand, target)
            c5_val = compute_c5(h_cand)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_cand.copy()
            h = h_cand
            sum_err = np.sum(h) - target
            lam += dual_lr * sum_err
            lam = float(np.clip(lam, -10.0, 10.0))
            if abs(sum_err) < 1e-10 and d >= 2:
                break
    return best_h, best_c5

_G_sl_n = 0

def _sl_cons_fun(h):
    return np.sum(h) - _G_sl_target

def _sl_cons_jac(h):

def slsqp_opt(h_start, target, alphas, max_iter, time_budget):
    global _G_sl_target, _G_sl_n
    start = time.time()
    n = len(h_start)
    _G_sl_target = target
    _G_sl_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_cons_fun, 'jac': _sl_cons_jac}]
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        if time.time() - start > time_budget:
            break
        _G['alpha'] = alpha
        _G['lam'] = 0.0
        _G['target'] = target
        try:
            res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-15})
            h_cand = project_feasible(res.x, target)
        except Exception:
            h_cand = h
        c5_val = compute_c5(h_cand)

            best_h = h_cand.copy()

    return best_h, best_c5

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        if L > 0:
            g[:L] += (1.0 - h[k:])
            g[k:] -= h[:L]
    else:
        j = -k
        L = n - j
        if L > 0:
            g[j:] += (1.0 - h[:L])
            g[:L] -= h[j:]
    return g

def top_k_subgrad(h_start, n, target, time_limit, max_steps, k_start, k_end, lr_init, lr_end):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 500:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.5), 1.0)
        k_top = int(k_start + (k_end - k_start) * frac)
        k_top = max(k_top, 1)
        top_idx = np.argsort(corr)[-k_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            lag = int(s_idx) - mid
            grad += get_grad_shift(h, lag, n)
        grad /= k_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        lr = lr_init + (lr_end - lr_init) * frac
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, target)
    return best_h, best_c5

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        f_arr = np.zeros((n, 2 * n - 1))
        idx_arr = np.arange(2 * n - 1)
        for p in range(n):
            idx1 = p + mid - idx_arr
            mask1 = (idx1 >= 0) & (idx1 < n)
            f_arr[p, mask1] += h[idx1[mask1]]
            idx2 = p + idx_arr - mid
            mask2 = (idx2 >= 0) & (idx2 < n)
            f_arr[p, mask2] += h[idx2[mask2]]
        best_trial = c5_val
        best_move = None
        for i in high_set:
            hi = h[i]
            if hi < 1e-10:
                continue
            for j in low_set:
                if i == j:
                    continue
                lj = h[j]
                if lj > 1.0 - 1e-10:
                    continue
                dcorr = f_arr[i] - f_arr[j]
                b_d = min(hi, 1.0 - lj)
                if b_d <= 1e-14:
                    continue
                a_d = 0.0
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    test = corr + mid_d * dcorr
                    f_mid = np.max(test)
                    if np.any(test >= f_mid - 1e-14):
                        active = test >= f_mid - 1e-14
                        fp = np.max(dcorr[active])
                    else:
                        fp = 0.0
                    if fp > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue
                trial = float(np.max(corr + delta_star * dcorr)) * dx
                if trial < best_trial - 1e-15:
                    best_trial = trial
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def soft_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = np.max(corr)
        log_w = alpha_w * (corr - c_max)
        log_w -= np.max(log_w)
        w = np.exp(log_w)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        sig = np.where(w > 1e-6)[0]
        if len(sig) == 0:
            break
        grad = np.zeros(n)
        for s_idx in sig:
            lag = int(s_idx) - mid
            grad += w[s_idx] * get_grad_shift(h, lag, n)
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 10
    return best_h, best_c5

def random_restart_optimize(h_init, n, target, rng, max_tries, time_limit, slsqp_alphas, slsqp_iter):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for t in range(max_tries):
        if time.time() - start > time_limit:
            break
        scale = float(rng.choice([0.01, 0.02, 0.04, 0.08, 0.12]))
        h_pert = h_cur + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_res, sc = slsqp_opt(h_pert, target, slsqp_alphas, slsqp_iter,
                              time_budget=min(60, time_limit - (time.time() - start)))
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_res.copy()
            h_cur = h_res.copy()
    return best_h, best_c5

def local_perturb(h_init, n, target, max_tries, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for t in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.02, 0.05, 0.08, 0.12, 0.18]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale

        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5
