def grad_overlap_at_lag(h, k, n):
    """Gradient of overlap_k = dx * corr[k+n-1] with respect to h."""

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            val = 1.0 - h[j - k]
            if j + k < n:
                val -= h[j + k]
            g[j] = val * dx
    else:
        m = -k
        for j in range(0, n - m):
            val = 1.0 - h[j + m]
            if j >= m:
                val -= h[j - m]
            g[j] = val * dx
    return g

    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:K]
    g = np.zeros(n)
    for idx in top_idx:
        k = int(idx) - (n - 1)
        g += grad_overlap_k(h, k, n)
    return g / K

def fw_backtracking(h_init, n, target, time_limit, K=100, max_iter=200000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = topk_subgradient(h, K)
        # Linear minimization: find v in P minimizing g dot v
        # P = {v : 0 <= v <= 1, sum(v) = target}
        # Solution: v[i] = 1 for the 'target' indices with smallest g[i]
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        # Backtracking line search
        c5_cur = compute_c5(h)
        tau = 1.0
        for _ in range(30):
            h_new = (1.0 - tau) * h + tau * v
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                break
            tau *= 0.5
        else:
            tau = 1.0 / (t + 1)
            h_new = (1.0 - tau) * h + tau * v
            h_new = project_feasible(h_new, target)
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            best_t = t
        if t > 5000 and t - best_t > 50000:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_overlap_k(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            val = 1.0 - h[j - k]
            if j + k < n:
                val -= h[j + k]
            g[j] = val * dx
    else:
        m = -k
        for j in range(0, n - m):
            val = 1.0 - h[j + m]
            if j >= m:
                val -= h[j - m]
            g[j] = val * dx
    return g

def topk_subgradient(h, K):
    n = len(h)
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:K]
    g = np.zeros(n)
    for idx in top_idx:
        k = int(idx) - (n - 1)
        g += grad_overlap_k(h, k, n)
    return g / K

def fw_standard(h_init, n, target, time_limit, K=150, max_iter=200000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = topk_subgradient(h, K)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 10000 and t - best_t > 80000:
            break
    return best_h, best_c5

For SLSQP to work well, I need:
- A smooth, differentiable objective
- The gradient of the objective
- Good initial point

The log-sum-exp objective:
f(h) = (1/alpha) * log(sum_k exp(alpha * overlap_k(h)))

Gradient:
∇f(h)_j = sum_k w_k * ∇overlap_k(h)_j
where w_k = exp(alpha * overlap_k) / sum_l exp(alpha * overlap_l)

This is a weighted average of the individual gradients. For large alpha, the weights concentrate on the top overlaps.

Let me implement the projection:
To project y onto {0 <= h <= 1, sum(h) = S} (where S = n/2):
1. First clip: h0[i] = clip(y[i], 0, 1)
2. If sum(h0) = S, done.
3. Otherwise, we need to adjust. This is equivalent to finding a threshold t such that:
   sum_i clip(y[i] - t, 0, 1) = S
4. We can find t by bisection.

Actually, the standard approach is:
- Find t such that sum_i max(0, min(1, y[i] - t)) = S
- This is equivalent to finding t such that the "water-filling" level gives the right sum.

For our case (y is a vector of real numbers, we want to project onto the intersection of the box [0,1]^n and the hyperplane sum = S):
1. Sort y in decreasing order: y_(1) >= y_(2) >= ... >= y_(n)
2. Find the largest k such that y_(k) - (S - sum_{i=1}^{k-1} max(0, y_(i))) / k >= y_(k+1) (or k=n)
3. Set t = (S - sum_{i=1}^{k-1} y_(i)) / k (clipping to [0,1])
4. h[i] = clip(y[i] - t, 0, 1)

But this is O(n log n) per projection, which for n=400 is fine.

                                  initial_step=0.01, momentum=0.9, max_iter=100000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum buffer
    step = initial_step
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        obj = smooth_objective(h, alpha)
        g = smooth_gradient(h, alpha)  # full gradient (or top-K)
        
        # Momentum update
        v = momentum * v - step * g
        h_new = project_feasible(h + v, target)
        
        # Check if function decreased
        obj_new = smooth_objective(h_new, alpha)
        if obj_new > obj + 1e-10:
            # Backtrack
            step *= 0.5
            h_new = project_feasible(h - step * g, target)
            v = -step * g
            obj_new = smooth_objective(h_new, alpha)
        
        h = h_new
        
        # Track best using true C5
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            # Reset momentum on improvement
            # v = np.zeros(n)
        
        # Adaptive momentum (restart when needed)
        if t % 1000 == 0 and t > 0:
            momentum = min(0.9, 0.5 + 0.4 * t / 10000)
    
    return best_h, best_c5

One more consideration: the `smooth_gradient` function. For the log-sum-exp:
f(h) = (1/alpha) * log(sum_k exp(alpha * overlap_k))
∇f(h)_j = sum_k (exp(alpha * overlap_k) / sum_l exp(alpha * overlap_l)) * ∂overlap_k/∂h_j

where w_k = softmax(alpha * overlaps)_k

For large alpha, only the top few overlaps have significant weight. So I can compute:
1. The top-K overlaps (by value)
2. The weights w_k for these K overlaps (normalized)
3. The weighted sum of their gradients

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def smooth_gradient(h, alpha, K=100):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    # Compute softmax weights for top-K
    top_idx = np.argsort(overlaps)[::-1][:K]
    top_vals = overlaps[top_idx]
    shifted = alpha * (top_vals - top_vals[0])  # for numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        k = int(idx) - (n - 1)
        w = weights[i]
        if k == 0:
            g += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if 2 * k <= n - 1:
                if k > 0:
                    g[0:k] += w * (-h[k:2*k]) * dx
                lb = n - 2 * k
                if lb > 0:
                    g[k:k+lb] += w * ((1.0 - h[0:lb]) - h[2*k:n]) * dx
                g[n-k:n] += w * (1.0 - h
