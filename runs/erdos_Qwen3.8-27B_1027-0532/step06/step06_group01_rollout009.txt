            shift = k - (n - 1)
            # f_k(h) ≈ ∑_i [(1-h*[i+k]) h[i] - h*[i] h[i+k]] - f_k(h*)
            # Build the linear expression
            expr = 0

        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except:

                prob.solve(solver=cp.CBC)
            except:
                break
        
        if h.value is None:

    return h_star, compute_c5(h_star)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_linearize_step(h_star, n, target):
    """One step of linearized LP. Returns new h or None."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    f_current = np.correlate(h_star, 1.0 - h_star, mode='full')
    
    # Build A_ub matrix (sparse would be better but let's use dense for n=600)
    # Variables: [h_0, ..., h_{n-1}, t], total n+1
    # Constraints:
    #   For each k in range(n_corr): a_k^T h - t <= f_current[k]
    #   h_i <= 1 (n constraints)
    #   -h_i <= 0 (n constraints)
    # Equality: sum(h) = target
    
    n_vars = n + 1
    n_ub = n_corr + 2 * n
    
    A_ub = np.zeros((n_ub, n_vars))
    b_ub = np.zeros(n_ub)
    
    # Shift constraints
    for k in range(n_corr):
        shift = k - mid
        a = np.zeros(n)
        v1 = (idxs - shift >= 0) & (idxs - shift < n)
        v2 = (idxs + shift >= 0) & (idxs + shift < n)
        if np.any(v1):
            a[v1] += (1.0 - h_star[idxs[v1] - shift])
        if np.any(v2):
            a[v2] -= h_star[idxs[v2] + shift]
        A_ub[k, :n] = a
        A_ub[k, n] = -1.0
        b_ub[k] = f_current[k]
    
    # Upper bounds h_i <= 1
    for i in range(n):
        A_ub[n_corr + i, i] = 1.0
        b_ub[n_corr + i] = 1.0
    
    # Lower bounds h_i >= 0 (as -h_i <= 0)
    for i in range(n):
        A_ub[n_corr + n + i, i] = -1.0
        b_ub[n_corr + n + i] = 0.0
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs')
        if res.status == 0 and res.x is not None:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            return h_new
    except Exception:
        pass
    return None
