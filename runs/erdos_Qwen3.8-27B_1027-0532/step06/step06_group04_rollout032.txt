    """Solve the LP: min t s.t. sum_i h[i]*g[i+k]*dx <= t for k in active, 0<=h<=1, sum h = target"""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in active_k:
        k = k_idx - half_n
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo + k:hi + k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:

        else:

    """Subgradient descent on the max of f_k with projected steps."""
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    half_n = n - 1

    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = 0.5
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        # Get top n_active lags
        top_idx = np.argsort(corr)[-n_active:]

        grad = np.zeros(n)
        for k_idx in top_idx:
            k = k_idx - half_n
            idx_valid1 = (idxs - k >= 0) & (idxs - k < n)  # i-k valid
            idx_valid2 = (idxs + k >= 0) & (idxs + k < n)  # i+k valid
            # d f_k / d h[i] = (1 - h[i+k])*dx - h[i-k]*dx
            if np.any(idx_valid2):

                grad[idx_valid1] -= h_cur[idxs[idx_valid1] - k] * dx

                break

import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_size(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def solve_lp_for_h(g, n, target, active_k_list):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in active_k_list:
        k = k_idx - half_n
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo + k:hi + k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=500000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=500000)
            except Exception:
                return None, None
    if h_var.value is None:
        return None, None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), float(t_var.value)

    half_n = n - 1

        grad = np.zeros(n)

            k = k_idx - half_n
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v2):

            if np.any(v1):
                grad[v1] -= h_cur[idxs[v1] - k] * dx

def smooth_max_slsqp(h_start, n, target, alphas, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

            cmax = np.max(corr)
            shifted = np.clip(alpha * (corr - cmax), -400.0, 0.0)
            return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha) * dx
        def con_fun(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        try:

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
