import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def targeted_cd(h, n, target, time_limit, n_top=10, patience=200):
    """Coordinate descent focusing on the worst lags."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    
    while time.time() - start < time_limit and no_improve < patience:
        corr_dl = compute_corr(h)
        C5_now = np.max(corr_dl)
        
        if C5_now < best_c5 - 1e-14:
            best_c5 = C5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top worst lags
        top_lags = np.argsort(-corr_dl)[:n_top]
        
        improved = False
        for m in top_lags:
            k = int(m) - mid
            
            # Gradient of this specific correlation
            grad = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
            grad -= np.mean(grad)  # maintain sum constraint approximately
            
            # Try to reduce this correlation: move h in -grad direction
            # But we need to be careful not to increase other correlations
            # Use line search
            best_trial = C5_now
            best_step = 0
            lr = 0.1
            for _ in range(10):
                h_try = project_feasible(h - lr * grad, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_trial - 1e-14:
                    best_trial = c5_try
                    best_step = lr
                lr *= 0.5
            
            if best_step > 0:
                h = project_feasible(h - best_step * grad, target)
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def game_theoretic(h, n, target, time_limit, max_steps=20000, eta=2.0, m_top=15):
    """Weighted subgradient method with multiplicative weight updates."""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    no_improve = 0
    lr = 0.05
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_now = np.max(corr)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute weighted gradient
        weights = np.exp(eta * (corr - c5_now))
        weights = weights / weights.sum()
        
        # Only use top m for efficiency
        top_m = np.argsort(-weights)[:m_top]
        w_top = weights[top_m]
        w_top = w_top / w_top.sum()
        
        grad = np.zeros(n)
        for m, w in zip(top_m, w_top):
            k = int(m) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)

            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Line search
        best_step = 0
        best_c5_try = c5_now
        lr_try = lr
        for _ in range(8):
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5_try - 1e-15:
                best_c5_try = c5_try
                best_step = lr_try
            lr_try *= 0.5

        lam_new = np.exp(eta * (corr - c5_now))
        lam_new = lam_new / lam_new.sum()
