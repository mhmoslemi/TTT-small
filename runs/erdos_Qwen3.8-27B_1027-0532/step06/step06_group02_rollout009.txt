                    lo = max(0, lo)
                    hi = min(len(arr_cumsum)-2, hi)
                    if lo > hi:
                        return 0.0
                    return arr_cumsum[hi+1] - arr_cumsum[lo]
                
                # For position i:
                # Term A: sum of w[k] for k in [max(-(n-1), -i), min(n-1, n-1-i)]
                #   idx range: [max(0, n-1-i), min(2n-2, 2n-2-i)]
                # Term B: sum of w[k]*h[i+k] for k in valid range
                #   = sum of w[k]*h[i+k] ... this is a weighted sum, not just a range sum
                #   Need a different approach.
                
                # OK let me just do it directly with numpy broadcasting for n=600.
                # It's O(n*(2n-1)) = O(600*1199) ≈ 720K multiplications, very fast.
                
                # grad[i] = sum_k w[k] * (1 - h[i+k] - h[i-k])
                # Create a matrix M where M[i, k_idx] = (1 - h[i+k] - h[i-k]) with OOB = 0 for h terms
                # But that's n x (2n-1) = 600 * 1199 ≈ 720K, fine.
                
                # Actually, let me think about this more carefully.
                # k ranges from -(n-1) to (n-1), indexed by k_idx from 0 to 2n-2
                # k = k_idx - (n-1)
                # h[i+k] = h[i + k_idx - (n-1)]
                # Valid when 0 <= i + k_idx - (n-1) < n, i.e., n-1-i <= k_idx <= 2n-2-i
                # h[i-k] = h[i - k_idx + (n-1)]
                # Valid when 0 <= i - k_idx + (n-1) < n, i.e., i <= k_idx <= i + n - 1
                
                # For each i, the valid k_idx range for h[i+k]: [n-1-i, 2n-2-i]
                # The valid k_idx range for h[i-k]: [i, i+n-1]
                
                # grad[i] = sum_{k_idx} w[k_idx] * (1 - h[i+k_idx-(n-1)] - h[i-(k_idx-(n-1))])
                #         = sum_{k_idx} w[k_idx] - sum_{k_idx in [n-1-i, 2n-2-i]} w[k_idx]*h[n-1+k_idx-i] 
                #                         - sum_{k_idx in [i, i+n-1]} w[k_idx]*h[n-1-i+k_idx-i]... 
                
                # This is getting confusing. Let me just use a loop-free approach:
                
                # Create index matrix
                i_idx = np.arange(n)[:, None]  # shape (n, 1)
                k_idx = np.arange(2*n-1)[None, :]  # shape (1, 2n-1)
                k_vals = k_idx - (n-1)  # shape (1, 2n-1), values from -(n-1) to n-1
                
                # h[i+k]: index = i_idx + k_vals, valid when 0 <= i+k < n
                plus_idx = i_idx + k_vals  # shape (n, 2n-1)
                plus_valid = (plus_idx >= 0) & (plus_idx < n)
                # h[i-k]: index = i_idx - k_vals, valid when 0 <= i-k < n
                minus_idx = i_idx - k_vals
                minus_valid = (minus_idx >= 0) & (minus_idx < n)
                
                # Get values (use 0 for invalid)
                plus_vals = np.where(plus_valid, h[np.clip(plus_idx, 0, n-1)], 0.0)
                minus_vals = np.where(minus_valid, h[np.clip(minus_idx, 0, n-1)], 0.0)
                
                # grad[i] = sum_k w[k] * (1 - plus_vals[i,k] - minus_vals[i,k]) ... wait
                # No: grad[i] = sum_k w[k] * (1 - h[i+k] - h[i-k])
                # But h[i+k] is 0 when out of range, and "1" only when i+k is in range?
                
                # Let me re-derive:
                # c[k] = sum_j h[j]*(1-h[j+k]) where j and j+k are both in [0,n-1]
                # dc[k]/dh[i] = product of terms involving h[i]:
                #   h[i] appears in c[k] when:
                #     j = i (first factor): term is h[i]*(1-h[i+k]), derivative = (1-h[i+k])
                #     j+k = i (second factor): term is h[j]*(1-h[i]) = h[i-k]*(1-h[i]), derivative = -h[i-k]
                #   So dc[k]/dh[i] = (1-h[i+k]) - h[i-k] (with OOB terms making it 0)
                #   Specifically: (1-h[i+k]) if i+k in [0,n-1] else 0, minus h[i-k] if i-k in [0,n-1] else 0
                
                # So: grad[i] = sum_k w[k] * [(1-h[i+k]) * I(0<=i+k<n) - h[i-k] * I(0<=i-k<n)]
                
                # = sum_k w[k] * I(0<=i+k<n) - sum_{k: 0<=i+k<n} w[k]*h[i+k] - sum_{k: 0<=i-k<n} w[k]*h[i-k]
                
                # First term: sum_k w[k] * I(0<=i+k<n) = sum of w[k] over k in [-i, n-1-i]
                # In k_idx: [-i+(n-1), n-1-i+(n-1)] = [n-1-i, 2n-2-i]
                
                # Second term: cross-correlation type
                # Third term: another cross-correlation type
                
                # For n=600, the matrix approach is 600*1199 ≈ 720K elements, totally fine.
                
                term1 = np.sum(w_arr[None, :] * plus_valid, axis=1)  # sum of w where i+k is valid
                term2 = np.sum(w_arr[None, :] * plus_vals, axis=1)  # sum of w[k]*h[i+k] where valid
                term3 = np.sum(w_arr[None, :] * minus_vals, axis=1)  # sum of w[k]*h[i-k] where valid
                
                grad = term1 - term2 - term3
                
                # Rescale gradient (the log-sum-exp objective has a natural scale)
                # grad = grad / alpha  (chain rule from log-sum-exp)
                # Actually, f = (1/alpha) * log(sum exp(alpha*c[k]))
                # df/dh[i] = sum_k exp(alpha*c[k]) * (1/alpha) * (1/sum exp(alpha*c[k])) * alpha * dc[k]/dh[i]
                #         = sum_k w[k] * dc[k]/dh[i]
                # So the alpha cancels! The gradient is just sum_k w[k] * dc[k]/dh[i]
                
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            # Adaptive learning rate
            lr = 2.0 / (alpha + 1.0)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_box_sum(h_new, target)
            
            if it % 100 == 0:
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h = h_new
            else:
                h = h_new
        
        # After each alpha phase, recompute
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

def sa_binary(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr, 
              block_frac=0.2, max_block_len=10, cluster_frac=0.1):
    """Improved SA with cluster moves."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        r = rng.random()
        
        if r < cluster_frac:
            # Cluster move: swap a block of 1s with a block of 0s of same length
            # Find a consecutive run of 1s and a consecutive run of 0s
            run_len = rng.randint(2, min(6, n // 20) + 1)
            # Find positions where we have run_len consecutive 1s
            ones_runs = []
            run_start = 0
            in_run = False
            for p in range(n):
                if h[p] == 1.0:
                    if not in_run:
                        run_start = p
                        in_run = True
                else:
                    if in_run:
                        if p - run_start >= run_len:
                            ones_runs.append((run_start, p - 1))
                        in_run = False
            if in_run and n - run_start >= run_len:
                ones_runs.append((run_start, n - 1))
            
            zeros_runs = []
            run_start = 0
            in_run = False
            for p in range(n):
                if h[p] == 0.0:
                    if not in_run:
                        run_start = p
                        in_run = True
                else:
                    if in_run:
                        if p - run_start >= run_len:
                            zeros_runs.append((run_start, p - 1))
                        in_run = False
            if in_run and n - run_start >= run_len:
                zeros_runs.append((run_start, n - 1))
            
            if len(ones_runs) > 0 and len(zeros_runs) > 0:
                a_start, a_end = ones_runs[rng.randint(len(ones_runs))]
                b_start, b_end = zeros_runs[rng.randint(len(zeros_runs))]
                h_new = h.copy()
                h_new[a_start:a_start+run_len] = 0.0
                h_new[b_start:b_start+run_len] = 1.0
                full_new = full_corr_fast(h_new)
                c5_new = float(full_new.max()) * dx
                delta_c5 = c5_new - current_c5
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    h = h_new
                    full = full_new
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        elif r < cluster_frac + block_frac:
            # Block reversal
            blen = rng.randint(3, min(max_block_len, n // 10) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            if abs(h_new.sum() - h.sum()) > 1e-10:
                stall += 1
                continue
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def greedy_multi(h, max_iters, rng, time_limit, start_time, k_arr, top_k=20):
    """Greedy that considers multiple top lags, not just the max."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 500
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(full.max())
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Consider top several lags
        top_indices = np.argsort(full)[::-1][:5]
        found = False
        for ti in top_indices:
            k_star = int(ti) - (n - 1)
            if k_star == 0:
                continue
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            
            top_i = ones[np.argsort(A[ones])[:min(top_k, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(top_k, len(zeros))]]
            
            for i in top_i[:10]:
                for j in top_j[:10]:
                    delta = compute_delta(h, i, j, k_arr)
                    new_max = float((full + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        stall = 0
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            stall += 1
    
    return best_h, best_c5

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def diverse_start(n, rng):
    """Create diverse random binary start with exactly n/2 ones."""
    target = n // 2
    h = np.zeros(n)
    # Use a quasi-random distribution
    # Place ones at positions determined by golden ratio sequence
    if rng.random() < 0.5:
        # Golden ratio spacing
        positions = [(int((i * (n / target) + rng.uniform(0, 1))) % n) for i in range(target)]
        positions = sorted(set(positions))
        # If we have duplicates, fill in randomly
        assigned = set(positions)
        h[list(assigned)] = 1.0
        if int(h.sum()) < target:
            remaining = target - int(h.sum())
            free = np.where(h == 0)[0]
            chosen = rng.choice(free, size=min(remaining, len(free)), replace=False)
            h[chosen] = 1.0
    else:
        # Random
        positions = rng.choice(n, size=target, replace=False)
        h[positions] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)
    
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick greedy on initial
    allowed = min(30, tleft() * 0.03)
    if allowed > 5:
        h_g, c5_g = greedy_multi(best_h, max_iters=20000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()
    
    # Phase 2: SA with cluster moves
    allowed = min(150, tleft() * 0.15)
    if allowed > 10:
        n_sa = int(min(3000000, allowed * 50000))
        h_sa, c5_sa = sa_binary(best_h, n_sa, 0.02, 0.0001, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                  block_frac=0.25, max_block_len=12, cluster_frac=0.15)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Phase 3: Adam continuous refinement
    if tleft() > 150:
        target_use = float(n_init) / 2.0
        alphas = [10, 30, 100, 300, 1000]
        n_inner = 500
        allowed = min(200, tleft() * 0.2)
        h_ref, sc = adam_continuous_refine(best_h, target_use, start_time, allowed, alphas, n_inner)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
    
    # Phase 4: Binary projection + SA
    if tleft() > 80:
        h_bin = project_binary(best_h, target_init)
        allowed = min(80, tleft() * 0.15)
        if allowed > 10:
            n_sa2 = int(min(2000000, allowed * 50000))
            h_sa2, c5_sa2 = sa_binary(h_bin, n_sa2, 0.015, 0.00005, rng,
                                       time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                       block_frac=0.2, max_block_len=8, cluster_frac=0.1)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()
    
    # Phase 5: Multi-start with diverse initializations
    for restart in range(5):
        if tleft() < 80:
            break
        
        if restart == 0:
            # Perturb current best
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(8):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
        elif restart == 1:
            # Block reversal perturbation
            h_pert = best_h.copy()
            blen = rng.randint(10, 30)
            a = rng.randint(0, n_init - blen + 1)
            h_pert[a:a+blen] = h_pert[a:a+blen][::-1]
        elif restart == 2:
            # Quasi-random start
            h_pert = diverse_start(n_init, rng)
        elif restart == 3:
            # Perturb with larger cluster swap
            h_pert = best_h.copy()
            # Swap two random blocks
            blen = rng.randint(10, 25)
            a = rng.randint(0, n_init - blen)
            b = rng.randint(0, n_init - blen)
            if abs(a - b) >= blen:
                tmp = h_pert[a:a+blen].copy()
                h_pert[a:a+blen] = h_pert[b:b+blen]
                h_pert[b:b+blen] = tmp
            # Fix sum
            h_pert = project_binary(h_pert, target_init)
        else:
            # Small perturbation
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(4):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
        
        # Greedy from perturbed
        allowed_g = min(20, tleft() * 0.05)
        if allowed_g < 5:
            break
        h_pg, c5_pg = greedy_multi(h_pert, max_iters=10000, rng=rng,
                                    time_limit=allowed_g, start_time=start_time, k_arr=k_arr)
        
        # SA from greedy result
        allowed_s = min(40, tleft() * 0.08)
        if allowed_s < 5:
            break
        n_sa_r = int(min(1500000, allowed_s * 40000))
        h_sr, c5_sr = sa_binary(h_pg, n_sa_r, 0.01, 0.00003, rng,
                                 time_limit=allowed_s, start_time=start_time, k_arr=k_arr,
                                 block_frac=0.2, max_block_len=8, cluster_frac=0.1)
        
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()
    
    # Phase 6: Coarse-to-fine
    if tleft() > 120:
        n_coarse = 80
        target_coarse = n_coarse // 2
        k_arr_c = np.arange(2 * n_coarse - 1) - (n_coarse - 1)
        x_new = np.linspace(0, 1, n_coarse)
        x_old = np.linspace(0, 1, n_init)
        h_coarse = np.interp(x_new, x_old, best_h)
        h_coarse_bin = project_binary(h_coarse, target_coarse)
        
        allowed_g = min(15, tleft() * 0.08)
        h_cg, c5_cg = greedy_multi(h_coarse_bin, max_iters=5000, rng=rng,
                                    time_limit=allowed_g, start_time=start_time, k_arr=k_arr_c)
        allowed_s = min(20, tleft() * 0.08)
        n_sac = int(min(300000, allowed_s * 25000))
        h_cs, c5_cs = sa_binary(h_cg, n_sac, 0.01, 0.0001, rng,
                                 time_limit=allowed_s, start_time=start_time, k_arr=k_arr_c,
                                 block_frac=0.3, max_block_len=6, cluster_frac=0.15)
        h_coarse_best = h_cs if c5_cs < c5_cg else h_cg
        c5_coarse_best = min(c5_cs, c5_cg)
        
        # Embed back
        h_fine = np.zeros(n_init)
        ratio = n_init / n_coarse
        for i in range(n_coarse):
            start = int(i * ratio)
            end = int((i + 1) * ratio)
            if end > n_init:
                end = n_init
            if h_coarse_best[i] > 0.5:
                h_fine[start:end] = 1.0
            else:
                h_fine[start:end] = 0.0
        h_fine = project_binary(h_fine, target_init)
        
        allowed_rg = min(20, tleft() * 0.08)
        h_fg, c5_fg = greedy_multi(h_fine, max_iters=8000, rng=rng,
                                    time_limit=allowed_rg, start_time=start_time, k_arr=k_arr)
        allowed_rs = min(25, tleft() * 0.1)
        n_saf = int(min(500000, allowed_rs * 30000))
        h_fs, c5_fs = sa_binary(h_fg, n_saf, 0.008, 0.00003, rng,
                                 time_limit=allowed_rs, start_time=start_time, k_arr=k_arr,
                                 block_frac=0.2, max_block_len=8, cluster_frac=0.1)
        c5_embed = c5_fs if c5_fs < c5_fg else c5_fg
        h_embed = h_fs if c5_fs < c5_fg else h_fg
