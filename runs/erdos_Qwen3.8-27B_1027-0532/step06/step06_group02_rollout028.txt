import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):

    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)

    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

    position = np.arange(2 * n - 1) - (n - 1)
    
    # Find top K correlation values

    w = np.zeros(2 * n - 1)

    grad = np.zeros(n)

        # where h_{j+k} is the value at position j+k (or 0 if out of bounds)
        
        # For term (1 - h_{j+k}):

        # Valid when j+k is in [0, n-1]
        j_plus = np.arange(n) + k
        valid_plus = (j_plus >= 0) & (j_plus < n)
        # contribution: (1 - h[j_plus]) for valid j_plus
        contrib_plus = np.zeros(n)
        contrib_plus[valid_plus] = 1.0 - h[j_plus[valid_plus]]
        
        # For term -h_{j-k}:
        # j-k ranges over [-k, n-1-k]
        # Valid when j-k is in [0, n-1]
        j_minus = np.arange(n) - k
        valid_minus = (j_minus >= 0) & (j_minus < n)
        contrib_minus = np.zeros(n)
        contrib_minus[valid_minus] = h[j_minus[valid_minus]]
        
        grad += (contrib_plus - contrib_minus) / K
    
    return grad

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def level_set_grad(h, corr, K):
    n = len(h)
    top_k = np.argsort(corr)[::-1][:K]
    grad = np.zeros(n)
    for idx in top_k:
        k = idx - (n - 1)
        # First term: (1 - h[j+k]) for valid j+k
        jp = np.arange(n) + k
        vp = (jp >= 0) & (jp < n)
        ct = np.zeros(n)
        ct[vp] = 1.0 - h[jp[vp]]
        # Second term: -h[j-k] for valid j-k
        jm = np.arange(n) - k
        vm = (jm >= 0) & (jm < n)
        cm = np.zeros(n)
        cm[vm] = h[jm[vm]]
        grad += (ct - cm) / K
    return grad

def adam_levelsset(h0, target, time_limit, start_time, K_schedule, lr_init):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    best_score = float(corr.max()) * dx
    best_h = h.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = lr_init
    
    while time.time() - start_time < time_limit:
        t += 1
        # Determine K for this iteration
        progress = min(1.0, t / 5000)
        K = int(K_schedule[0] + (K_schedule[1] - K_schedule[0]) * progress)
        
        grad = level_set_grad(h, corr, K)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_box_sum(h, target)
        
        # Check score periodically
        if t % 100 == 0:
            corr = full_corr_fast(h)
            sc = float(corr.max()) * dx
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
            # Decay learning rate
            lr *= 0.995
        elif t % 10 == 0:
            corr = full_corr_fast(h)
    
    return best_h, best_score

def binary_sa_multiswap(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):

    dx = 2.0 / n
    h = h0.astype(float).copy()

    stall = 0
    max_stall = 20000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if rng.random() < 0.3:

                j = zeros[rng.randint(len(zeros))]
                if i in used or j in used:

                if i == j:
                    continue

                continue

                for i, j in swaps:
                    h[i] = 0.0

                    best_score = new_score
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            j = zeros[rng.randint(len(zeros))]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:

    b = np.zeros(N)

        v_hat = v / (1.0 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_box_sum(h, target)
        if t % 50 == 0:
            corr = full_corr_fast(h)
            sc = float(corr.max()) * dx
            if sc < best_score - 1e-8:
                best_score = sc
                best_h = h.copy()
                plateau_pts = 0
            else:
                plateau_pts += 1
            if plateau_pts > 30:
                lr *= 0.9
                if lr < 1e-6:
                    break
        elif t % 20 == 0:
            corr = full_corr_fast(h)
    return best_h, best_score

def quick_greedy(h0, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    corr = full_corr_fast(h)
    best_score = float(corr.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(20, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(20, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    new_score = new_max * dx
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                    found = True
                    stall = 0
                    break
            if found:
                break
        if not found:
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    new_score = new_max * dx
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                    found = True
                    stall = 0
                    break
        if not found:
            stall += 1
    return best_h, best_score

def binary_sa_multiswap(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    corr = full_corr_fast(h)
    best_score = float(corr.max()) * dx
    best_h = h.copy()
    current_score = best_score
    stall = 0
    max_stall = 30000
    positions = np.arange(n)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        use_multi = rng.random() < 0.25
        if use_multi:
            n_swaps = rng.randint(2, 4)
            delta_total = np.zeros(2 * n - 1)
            swaps = []
            used = set()
            for _ in range(n_swaps):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i in used or j in used or i == j:
                    continue
                delta = compute_delta(h, i, j, k_arr)
                delta_total += delta
                swaps.append((i, j))
                used.add(i)
                used.add(j)
            if len(swaps) == 0:
                stall += 1
                continue
            new_max = float((corr + delta_total).max())
            new_score = new_max * dx
            dc = new_score - current_score
            if dc < 0 or (temp > 1e-15 and rng.random() < np.exp(-dc / temp)):
                for ii, jj in swaps:
                    h[ii] = 0.0
                    h[jj] = 1.0
                corr = corr + delta_total
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((corr + delta).max())
            new_score = new_max * dx
            dc = new_score - current_score
            if dc < 0 or (temp > 1e-15 and rng.random() < np.exp(-dc / temp)):
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + delta
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_score

def sa_focused(h0, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    corr = full_corr_fast(h)
    best_score = float(corr.max()) * dx
    best_h = h.copy()
    current_score = best_score
    stall = 0
    max_stall = 40000
    positions = np.arange(n)
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 3 == 0:
            top_k = 4
            top_indices = np.argsort(corr)[::-1][:top_k]
            best_move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move_found = True
                    break
            if not best_move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((corr + delta).max())
        new_score = new_max * dx
        dc = new_score - current_score
        if dc < 0 or (temp > 1e-15 and rng.random() < np.exp(-dc / temp)):
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + delta
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_score

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_global_n = 0
_global_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)
    target_f = float(n) / 2.0
    k_arr = np.arange(2 * n - 1) - (n - 1)
    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    # Phase 1: Quick greedy
    t1 = min(25, tleft() * 0.08)
    h_g, c5_g = quick_greedy(best_h, max_iters=50000, rng=rng,
                             time_limit=t1, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Level-set gradient (main workhorse)
    if tleft() > 100:
        t2 = min(200, tleft() * 0.25)
        h_ls, c5_ls = adam_levelset(best_h, target_f, t2, start_time,
                                     K_lo=3, K_hi=25, lr_init=0.008)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls.copy()
        h_bin = project_binary(h_ls, target)
        t2b = min(20, tleft() * 0.04)
        h_gb, c5_gb = quick_greedy(h_bin, max_iters=30000, rng=rng,
                                    time_limit=t2b, start_time=start_time, k_arr=k_arr)
        if c5_gb < best_score:
            best_score = c5_gb
            best_h = h_gb.copy()

    # Phase 3: Focused SA
    if tleft() > 100:
        t3 = min(180, tleft() * 0.2)
        n_iters3 = int(min(4000000, t3 * 50000))
        h_sa, c5_sa = sa_focused(best_h, n_iters3, 0.018, 0.00003, rng,
                                  time_limit=t3, start_time=start_time, k_arr=k_arr)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Another level-set gradient round
    if tleft() > 100:
        t4 = min(180, tleft() * 0.22)
        h_ls2, c5_ls2 = adam_levelset(best_h, target_f, t4, start_time,
                                       K_lo=5, K_hi=40, lr_init=0.004)
        if c5_ls2 < best_score:
            best_score = c5_ls2
            best_h = h_ls2.copy()
        h_bin2 = project_binary(h_ls2, target)
        t4b = min(20, tleft() * 0.04)
        h_gb2, c5_gb2 = quick_greedy(h_bin2, max_iters=30000, rng=rng,
                                      time_limit=t4b, start_time=start_time, k_arr=k_arr)
        if c5_gb2 < best_score:
            best_score = c5_gb2
            best_h = h_gb2.copy()

    # Phase 5: Multi-swap SA
    if tleft() > 80:
        t5 = min(150, tleft() * 0.2)
        n_iters5 = int(min(3000000, t5 * 40000))
        h_ms, c5_ms = binary_sa_multiswap(best_h, n_iters5, 0.015, 0.00002, rng,
                                           time_limit=t5, start_time=start_time, k_arr=k_arr)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 6: Multi-start
    for restart in range(6):
        if tleft() < 70:
            break
        if restart % 2 == 0:
            n_pert = rng.randint(4, 10)
        else:
            n_pert = rng.randint(10, 18)
        h_pert = perturb_diverse(best_h, rng, n_pert)
        t_r = min(50, tleft() * 0.08)
        n_sa_r = int(min(800000, t_r * 40000))
        h_sr, c5_sr = sa_focused(h_pert, n_sa_r, 0.012, 0.00002, rng,
                                  time_limit=t_r, start_time=start_time, k_arr=k_arr)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()
        # Quick greedy from the SA result
        t_rg = min(15, tleft() * 0.02)
        h_gr, c5_gr = quick_greedy(h_sr, max_iters=15000, rng=rng,
                                    time_limit=t_rg, start_time=start_time, k_arr=k_arr)
        if c5_gr < best_score:
            best_score = c5_gr
            best_h = h_gr.copy()

    # Phase 7: SLSQP polish
    if tleft() > 60:
        alphas_slsqp = [30, 100, 500, 2000, 8000]
        t7 = min(120, tleft() * 0.2)
        h_ref, sc_ref = slsqp_refine(best_h, n, target_f, alphas_slsqp,
                                      2000, 1e-14, time_limit=t7, start_time=start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()
        h_bin_final = project_binary(h_ref, target)
        t7b = min(20, tleft() * 0.05)
        h_gbf, c5_gbf = quick_greedy(h_bin_final, max_iters=20000, rng=rng,
                                      time_limit=t7b, start_time=start_time, k_arr=k_arr)
        if c5_gbf < best_score:
            best_score = c5_gbf
            best_h = h_gbf.copy()

    # Phase 8: Final level-set from SLSQP result
    if tleft() > 50:
        t8 = min(60, tleft() * 0.2)
        h_ls3, c5_ls3 = adam_levelset(best_h, target_f, t8, start_time,
                                       K_lo=10, K_hi=60, lr_init=0.003)
        if c5_ls3 < best_score:
            best_score = c5_ls3
            best_h = h_ls3.copy()
        h_bin3 = project_binary(h_ls3, target)
        t8b = min(15, tleft() * 0.03)
        h_gb3, c5_gb3 = quick_greedy(h_bin3, max_iters=15000, rng=rng,
                                      time_limit=t8b, start_time=start_time, k_arr=k_arr)
        if c5_gb3 < best_score:
            best_score = c5_gb3
            best_h = h_gb3.copy()

    # Phase 9: Final SA burst
    if tleft() > 25:
        t9 = max(8.0, tleft() - 12)
        h_bin_f = project_binary(best_h, target)
        n_sa_f = int(min(500000, t9 * 35000))
        h_sa_f, c5_sa_f = sa_focused(h_bin_f, n_sa_f, 0.004, 0.000005, rng,
                                      time_limit=t9, start_time=start_time, k_arr=k_arr)
        if c5_sa_f < best_score:
            best_score = c5_sa_f
            best_h = h_sa_f.copy()

    # Final feasibility
    h_final = project_box_sum(best_h, target_f)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
