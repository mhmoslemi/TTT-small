def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    step = diff / len(idx)
    h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-6)[0]
        else:
            idx2 = np.where(h > 1e-6)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_constraint_matrix(g, n):
    dx = 2.0 / n
    A = np.zeros((2 * n - 1, n))
    half_n = n - 1
    for k_idx in range(2 * n - 1):
        k = k_idx - half_n
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            A[k_idx, i_lo:i_hi] = g[i_lo + k:i_hi + k] * dx
    return A

def solve_lp(h_current, n, target, damping, reg_weight, n_constraints, solver_eps):
    dx = 2.0 / n
    g = 1.0 - h_current
    half_n = n - 1

    A = build_constraint_matrix(g, n)

    corr_vals = A @ h_current
    if n_constraints < len(corr_vals):
        top_idx = np.argsort(corr_vals)[-n_constraints:]
        A_use = A[top_idx, :]
    else:
        A_use = A

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        A_use @ h_var <= t_var
    ]

    if reg_weight > 1e-14:
        objective = t_var + reg_weight * cp.sum_squares(h_var - h_current)
    else:
        objective = t_var

    prob = cp.Problem(cp.Minimize(objective), constraints)
    solved = False
    try:
        prob.solve(solver=cp.SCS, eps=solver_eps, max_iters=500000, verbose=False)
        solved = True
    except Exception:
        pass
    if not solved or h_var.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iter=500000)
            solved = True
        except Exception:
            return None, None

    if h_var.value is None:
        return None, None

    h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    t_val = float(t_var.value) if t_var.value is not None else compute_c5(h_new)
    return h_new, t_val

def alternating_lp_full(h_init, n, target, max_iter, time_limit, start_damping=1.0,
                        reg_weight=0.0, n_constraints=None, solver_eps=1e-10):
    start = time.time()
    if n_constraints is None:
        n_constraints = 2 * n - 1
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = start_damping
    stall_count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        h_new, t_val = solve_lp(h, n, target, 1.0, reg_weight, n_constraints, solver_eps)
        if h_new is None:
            break

        h_trial = (1.0 - damping) * h + damping * h_new
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)

        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            stall_count = 0
            damping = min(1.0, damping + 0.05)
        else:
            h = h_trial
            stall_count += 1
            damping = max(0.5, damping * 0.95)
            if stall_count > 100:
                break

        if it > 0 and it % 50 == 0:
            if reg_weight > 1e-8:
                reg_weight *= 0.5

    return best_h, best_c5

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    lam_rate = 0.01
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active

        if p % 30 == 29:
            lam_rate = max(0.001, lam_rate * 0.93)

    return best_h, best_c5

def smooth_objective(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val)

def smooth_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    half_n = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - half_n
        valid1 = (idxs + k >= 0) & (idxs + k < n)
        grad[valid1] += w * (1.0 - h[idxs[valid1] + k])
        valid2 = (idxs - k >= 0) & (idxs - k < n)
        grad[valid2] -= w * h[idxs[valid2] - k]
    grad *= dx
    return grad

def slsqp_optimize(h_start, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_objective(x, alpha, n, target),
                       h_start, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def lbfgs_smooth(h_start, n, target, alpha_list, maxiter_each, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alpha_list:
        if time.time() - start > time_limit:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(lambda x: smooth_objective(x, alpha, n, target),
                           h_cur, method='L-BFGS-B', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs + k) & ((idxs + k) < n)
        idx_valid2 = (0 <= idxs - k) & ((idxs - k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] + k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] - k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: LP with all constraints, moderate damping
    if tleft() > 200:
        lp_time = min(200, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target, max_iter=5000,
                                          time_limit=lp_time, start_damping=0.9,
                                          reg_weight=1e-4, n_constraints=2*n-1,
                                          solver_eps=1e-9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: LP with all constraints, no regularization, high damping
    if tleft() > 150:
        lp_time = min(180, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target, max_iter=5000,
                                          time_limit=lp_time, start_damping=1.0,
                                          reg_weight=0.0, n_constraints=2*n-1,
                                          solver_eps=1e-10)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Coordinate descent
    if tleft() > 80:
        cd_time = min(80, tleft() * 0.1)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=80,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: LP at higher resolution n=500
    if tleft() > 250:
        n2 = 500
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        sc_int = compute_c5(h_interp)
        if sc_int < best_score:
            best_score = sc_int
            best_h = h_interp
            n = n2
            target = target2

        if tleft() > 200:
            lp_time = min(150, tleft() * 0.25)
            h_lp, c5_lp = alternating_lp_full(best_h, len(best_h), len(best_h)/2.0,
                                              max_iter=5000, time_limit=lp_time,
                                              start_damping=0.9, reg_weight=5e-5,
                                              n_constraints=2*len(best_h)-1,
                                              solver_eps=1e-9)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = len(best_h)
                target = n / 2.0

    # Phase 5: LP at n=600
    if tleft() > 250:
        n3 = 600
        target3 = n3 / 2.0
        h_interp3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h_interp3)
        if tleft() > 200:
            lp_time = min(150, tleft() * 0.25)
            h_lp3, c5_lp3 = alternating_lp_full(h_interp3, n3, target3,
                                                max_iter=5000, time_limit=lp_time,
                                                start_damping=0.9, reg_weight=5e-5,
                                                n_constraints=2*n3-1,
                                                solver_eps=1e-9)
            if c5_lp3 < best_score:
                best_score = c5_lp3
                best_h = h_lp3
                n = n3
                target = target3
            elif sc3 < best_score:
                best_score = sc3
                best_h = h_interp3
                n = n3
                target = target3

    # Phase 6: Coordinate descent (longer)
    if tleft() > 60:
        cd_time = min(60, tleft() * 0.1)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=100,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: L-BFGS with smooth objective
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lb_time = min(100, tleft() * 0.15)
        h_s, c5_s = lbfgs_smooth(best_h, n_use, target_use,
                                 alpha_list=[200, 1000, 5000],
                                 maxiter_each=500, time_limit=lb_time)
        if c5_s < best_score:
            best_score = c5_s
            best_h = h_s

    # Phase 8: Coarse-resolution exploration (try n=200, 250)
    if tleft() > 200:
        for n_coarse in [200, 250]:
            if tleft() < 100:
                break
            target_coarse = n_coarse / 2.0
            h_coarse_init = interpolate_to(best_h, n_coarse, target_coarse)
            # Try random restarts
            for restart in range(3):
                if tleft() < 80:
                    break
                if restart == 0:
                    h_try = h_coarse_init
                else:
                    h_try = project_feasible(h_coarse_init + rng.randn(n_coarse) * 0.1,
                                             target_coarse)
                lp_time = min(40, tleft() * 0.15)
                h_lp_c, c5_lp_c = alternating_lp_full(h_try, n_coarse, target_coarse,
                                                      max_iter=3000, time_limit=lp_time,
                                                      start_damping=0.9, reg_weight=1e-3,
                                                      n_constraints=2*n_coarse-1,
                                                      solver_eps=1e-8)
                if c5_lp_c < best_score:
                    best_score = c5_lp_c
                    best_h = interpolate_to(h_lp_c, n, n / 2.0)

    # Phase 9: Final LP polish on best solution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(60, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_full(best_h, n_use, target_use,
                                          max_iter=5000, time_limit=lp_time,
                                          start_damping=1.0, reg_weight=0.0,
                                          n_constraints=2*n_use-1,
                                          solver_eps=1e-11)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Subgradient micro-refinement
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=5000, time_limit=min(25, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 11: Final coordinate descent
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=120,
                                    max_passes=20000, time_limit=min(25, tleft() * 0.5),
                                    rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final LP micro-polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, c5_lp = alternating_lp_full(best_h, n_use, target_use,
                                          max_iter=3000, time_limit=min(15, tleft() * 0.7),
                                          start_damping=1.0, reg_weight=0.0,
                                          n_constraints=2*n_use-1,
                                          solver_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Final subgradient
    if tleft() > 10:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=3000, time_limit=min(8, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
