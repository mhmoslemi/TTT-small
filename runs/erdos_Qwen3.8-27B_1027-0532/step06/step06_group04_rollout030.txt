```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def smooth_max_obj(h, alpha, target, n):
    """Log-sum-exp smooth approximation of the max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, target, n):
    """Gradient of the smooth max approximation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    dx = 2.0 / n
    grad = np.zeros(n)
    half_n = n - 1
    idxs = np.arange(n)
    
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - half_n
        v1 = (0 <= idxs + k) & ((idxs + k) < n)
        v2 = (0 <= idxs - k) & ((idxs - k) < n)
        grad[v1] += weights[k_idx] * (1.0 - h[idxs[v1] + k]) * dx
        grad[v2] -= weights[k_idx] * h[idxs[v2] - k] * dx
    
    return grad

def projected_gradient(h, target, alpha, n_steps, time_limit, lr0=0.01):
    """Projected gradient descent with backtracking on the smooth objective."""
    start = time.time()
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr0

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        obj_val = smooth_max_obj(h_cur, alpha, target, len(h_cur))
        grad = smooth_grad(h_cur, alpha, target, len(h_cur))
        
        # Backtracking line search
        for _ in range(20):
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            obj_try = smooth_max_obj(h_try, alpha, target, len(h_try))
            if obj_try < obj_val - 1e-10:
                break
            lr *= 0.5
        
        h_cur = h_try
        lr = min(lr * 1.1, 0.1)
        
        score = compute_c5(h_cur)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_cur.copy()
    
    return best_h, best_score

def fourier_perturb(h, rng, n_modes=5, amp=0.1):
    """Generate a perturbation using low-frequency Fourier modes."""
    n = len(h)
    pert = np.zeros(n)
    freqs = rng.randint(1, max(2, n // 2), size=min(n_modes, n // 2 - 1))
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    return pert
```

```python
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()

    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):

            break

        # Update blocks
        n_blocks = n // block_size
        for b in range(n_blocks):
            block = slice(b * block_size, (b+1) * block_size)
            block_idx = np.arange(b * block_size, (b+1) * block_size)

            # The contribution from the block is: sum_{i in block} h[i](1-h[i+k])
            #   = sum_{i in block, i+k valid} h[i] - h[i+k]
            #   = sum_{i in block, i+k valid} h[i]*(1-h[i+k])
```

```python
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def solve_1d_max(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):

        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)
```

Alternative: for each point i, the function f_i(x) = max_j (C_ij * x + b_ij) is a convex piecewise-linear function of x. Its minimum on [0,1] is at:

- x = 1 if f_i is increasing at 1 (i.e., all active constraints at x=1 have C < 0)
- Otherwise, at the point where the slope changes sign

The slope of f_i at x is max of {C_ij : j active at x}. The minimum is where this slope changes from negative to positive.

- Compute f_i(mid) = max_j (C_ij * mid + b_ij)
- Compute the "subgradient" at mid: the set of j that achieve the max
- If all such j have C_ij > 0: f is increasing → hi = mid
- If all such j have C_ij < 0: f is decreasing → lo = mid
- Otherwise: we're at the minimum (or very close)

```python
def solve_1d_max_batch(slopes, intercepts, lam, n_iter=45):
    C = slopes + lam  # (n, m)
    n_pts, m = C.shape
    
    lo = np.zeros(n_pts)
    hi = np.ones(n_pts)
    
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = mid[:, None] * C + intercepts  # (n, m)
        max_val = np.max(vals, axis=1)  # (n,)
        
        # Find active constraints
        diff = max_val[:, None] - vals  # (n, m)
        tol = 1e-12 * np.maximum(1.0, np.abs(max_val))[:, None]
        active = diff < tol  # (n, m)
        
        # Ensure at least one active per row
        no_active = ~np.any(active, axis=1)
        if np.any(no_active):
            amax_idx = np.argmax(vals[no_active], axis=1)
            rows = np.where(no_active)[0]
            active[rows, amax_idx] = True
        
        # Check slopes of active constraints
        # min_C_active = minimum C among active constraints
        # max_C_active = maximum C among active constraints
        # If min_C > 0: all active slopes positive → f increasing → hi = mid
        # If max_C < 0: all active slopes negative → f decreasing → lo = mid
        # Else: at minimum
        
        min_C = np.where(active, C, np.inf).min(axis=1)
        max_C = np.where(active, C, -np.inf).max(axis=1)
        
        increasing = min_C > 1e-14  # all active slopes positive
        decreasing = max_C < -1e-14  # all active slopes negative
        
        hi[increasing] = mid[increasing]
        lo[decreasing] = mid[decreasing]
        # For "at minimum" points, neither lo nor hi changes → they converge
        
    return 0.5 * (lo + hi)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    total_lags = 2 * n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, total_lags - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 3e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 5 * n_active:
            active_k = set(np.argsort(corr)[-5 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.94)
        else:
            damping = min(0.98, damping + 0.04)
        if it % 40 == 39 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.6)

    return best_h, best_c5


def fourier_perturb(h, rng, n_modes, amp):
    n = len(h)
    pert = np.zeros(n)
    max_f = max(2, n // 2 - 1)
    freqs = rng.randint(1, max_f, size=min(n_modes, max_f - 1))
    for f in freqs:
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += amp * np.sin(2.0 * np.pi * f * np.arange(n) / n + phase)
    return pert


def smooth_obj(h, alpha, n, target):
```
