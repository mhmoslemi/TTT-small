import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def frank_wolfe_fast(h_init, n, target, time_limit, K=300, max_iter=500000):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    dx = 2.0 / n
    n_half = n - 1
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n - k] += (1.0 - h[k:]) * dx
                g[k:] += (-h[:n - k]) * dx
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m]) * dx
                g[:n - m] += (-h[m:]) * dx
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 3000 and t - best_t > 80000:
            break
    return best_h, best_c5

def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=300,
                   damping_start=0.9, damping_end=0.85,
                   reg_start=5e-4, reg_end=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 50 and stuck > 100:
            break
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        progress = min(1.0, it / max(1, max_iter * 0.4))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
                constraints.append(coeff @ h_var <= t_var)
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = (1.0 - h[k:]) * dx
                constraints.append(coeff @ h_var <= t_var)
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n - m]) * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, max_iters=80000, eps=1e-10, verbose=False)
            solved = True
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=80000, verbose=False)
                solved = True
            except Exception:
                solved = False
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def _cons_fun(h):
    return np.sum(h) - _gs['target']

def _cons_jac(h):
    return np.ones(_gs['n'])

def slsqp_optimize(h_start, n, target, alphas, maxiter=3000, ftol=1e-14):

    _gs['target'] = target
    bounds = [(0.0, 1.0)] * n

                           method='SLSQP', bounds=bounds, constraints=cons,

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def upsample_downsample(h, n_src, n_dst):
    if n_dst == n_src:
        return h.copy()
    if n_dst > n_src:
        x_src = np.linspace(0.0, 1.0, n_src, endpoint=False)
        x_dst = np.linspace(0.0, 1.0, n_dst, endpoint=False)
        h_interp = np.interp(x_dst, x_src, h, period=1.0)
        return project_feasible(h_interp, n_dst / 2.0)
    else:
        x_dst = np.linspace(0.0, 1.0, n_dst, endpoint=False)
        x_src = np.linspace(0.0, 1.0, n_src, endpoint=False)
        h_interp = np.interp(x_dst, x_src, h, period=1.0)
        return project_feasible(h_interp, n_dst / 2.0)

def time_remaining(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)
