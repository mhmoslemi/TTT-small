        # Actually, the gradient is complex due to the max. Let me just use SLSQP without gradient

Actually, the issue with the log-sum-exp smoothing is that the gradient of max_C is non-smooth. A common trick is to use the softmax approximation: instead of max, use (1/alpha) * log(sum(exp(alpha * C))). This is smooth and its gradient is:
d/dh[p] [(1/alpha) log(sum_s exp(alpha * C[s]))] = sum_s softmax(alpha*C)[s] * dC[s]/dh[p]
= sum_s softmax(alpha*C)[s] * dx * grad_m[p, s]

This is the "soft" part. If we just use this (without the extra max_C), then:
obj(h) = (1/alpha) * log(sum_s exp(alpha * C[s]))
grad(obj)[p] = dx * sum_s softmax(alpha*C)[s] * grad_m[p, s]
= dx * (softmax(alpha*C) @ grad_m[p, :])

Wait, let me be more careful:
obj(h) = (1/alpha) * log(Σ_s exp(alpha * C[s]))

d(obj)/d(h[p]) = (1/alpha) * Σ_s alpha * exp(alpha*C[s]) * dC[s]/d(h[p]) / Σ_s exp(alpha*C[s])
= Σ_s [exp(alpha*C[s]) / Σ_t exp(alpha*C[t])] * dC[s]/d(h[p])
= Σ_s softmax_s * (dx * grad_m[p, s])
= dx * (grad_m[p, :] @ softmax)

where softmax = exp(alpha*C) / sum(exp(alpha*C))

This is a weighted average of the gradient rows, which is easy to compute!

So the objective is:
obj(h) = (1/alpha) * log(Σ_s exp(alpha * C[s]))  [this is a smooth approximation to max C[s]]
grad[p] = dx * (grad_m[p, :] @ softmax)  [weighted sum of gradient rows]

We can vectorize this:
grad_all = dx * (softmax @ grad_m)  # shape (n,)

This is O(n * n_corr) = O(n²) per gradient evaluation, which is the same as computing corr.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    n_free = mask.sum()
    if n_free == 0:
        return h
    h[mask] += diff / n_free
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        n_free2 = mask2.sum()
        if n_free2 > 0:
            h[mask2] += residual / n_free2
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_matrix(h, n):
    n_corr = 2 * n - 1
    p_idx = np.arange(n)[:, None]
    s_idx = np.arange(n_corr)[None, :]
    idx1 = (p_idx + s_idx - (n - 1)).astype(int)
    mask1 = (idx1 >= 0) & (idx1 < n)
    first = np.zeros((n, n_corr))
    first[mask1] = (1.0 - h)[idx1[mask1]]
    idx2 = (p_idx - s_idx + (n - 1)).astype(int)
    mask2 = (idx2 >= 0) & (idx2 < n)
    second = np.zeros((n, n_corr))
    second[mask2] = h[idx2[mask2]]
    return first - second

def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit):
    start = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            aC = alpha * (C - np.max(C))
            return float(np.max(aC) + np.log(np.sum(np.exp(aC)))) / alpha
        
        def jac(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            aC = alpha * (C - np.max(C))
            exp_aC = np.exp(aC)
            wsum = np.sum(exp_aC)
            softmax = exp_aC / wsum
            gm = grad_matrix(h, n)
            return dx * (softmax @ gm)
        
        con_fun = lambda h: np.sum(h) - target
        con_jac = lambda h: np.ones(n)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h_cur, jac=jac, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5
