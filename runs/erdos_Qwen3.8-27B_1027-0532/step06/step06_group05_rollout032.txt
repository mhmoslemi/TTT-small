def compute_overlap_2int(L1, p1, p2, n=606, total_ones=303):
    L2 = total_ones - L1
    # h = 1 on [p1, p1+L1-1] and [p2, p2+L2-1]
    # Compute A(k) for k = 0, 1, ..., n-1
    # A(k) = max(0, L1-k) + max(0, L2-k) + cross(k)
    # cross(k) = length of [max(p1, p2-k), min(p1+L1-1, p2+L2-1-k)]
    
    max_overlap = 0
    for k in range(1, n):  # k=0 gives A(0)=303, overlap=0
        a1 = max(0, L1 - k)
        a2 = max(0, L2 - k)
        lo = max(p1, p2 - k)
        hi = min(p1 + L1 - 1, p2 + L2 - 1 - k)
        cross = max(0, hi - lo + 1) if hi >= lo else 0
        A = a1 + a2 + cross
        overlap = total_ones - A
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * (2.0 / n)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(1, n):
        wp = weights[n - 1 + s]
        wn = weights[n - 1 - s]
        if wp > 1e-13 or wn > 1e-13:

                grad[s:] -= wp * h[:n - s]

    return value, grad

            v_hat = v / (1.0 - 0.999 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = project_box_sum(h_new, target_sum)
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score
