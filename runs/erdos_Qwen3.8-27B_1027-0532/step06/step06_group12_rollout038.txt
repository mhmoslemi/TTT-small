- +h[i-k] = 1 if i-k ∈ S, or 0 if i-k ∉ S
- -h[j-k] = -1 if j-k ∈ S, or 0 if j-k ∉ S

So Δ_k = dx * [-1 + 1_{j+k ∉ S} + 1_{i-k ∈ S} - 1_{j-k ∈ S}]

For this to be negative (improving lag k), we need:
1_{j+k ∉ S} + 1_{i-k ∈ S} - 1_{j-k ∈ S} < 1

Possible cases:
- If j+k ∈ S, i-k ∉ S, j-k ∉ S: Δ = dx * [-1 + 0 + 0 - 0] = -dx ✓
- If j+k ∈ S, i-k ∉ S, j-k ∈ S: Δ = dx * [-1 + 0 + 0 - 1] = -2dx ✓ (even better!)
- If j+k ∈ S, i-k ∈ S, j-k ∉ S: Δ = dx * [-1 + 0 + 1 - 0] = 0
- If j+k ∈ S, i-k ∈ S, j-k ∈ S: Δ = dx * [-1 + 0 + 1 - 1] = -dx ✓

So for the best case, we want: j+k ∈ S (so adding j doesn't create a new "deficit" at lag k), and j-k ∈ S (so adding j removes a "deficit" at lag k from the second position). And ideally i-k ∉ S.

This gives us a clear heuristic for the swap:
- Pick i from the "deficit" positions at the most critical lag k*
- Pick j such that j+k* ∈ S and j-k* ∈ S (if possible)

This is a 2-sat type constraint on j. We can find such j efficiently.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad(h, weights, n, dx, active):
    g = np.zeros(n)
    n_half = n - 1
    one_minus_h = 1.0 - h
    for idx in active:
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            g += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += w * one_minus_h[k:] * dx
            g[k:] += w * (-h[:n - k]) * dx
        else:
            m = -k
            g[m:] += w * one_minus_h[:n - m] * dx
            g[:n - m] += w * (-h[m:]) * dx
    return g

def momentum_descent(h_init, n, target, dx, time_limit,
                     alpha_init=50.0, alpha_max=200000.0,
                     alpha_rate=1.002,
                     lr_init=0.02, lr_decay=0.99995,
                     momentum=0.9, max_iter=200000):
    start = time.time()
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    no_improve = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        cur_max = float(np.max(overlaps))
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50000:
                break
        n_ov = 2 * n - 1
        top_k = min(80, n_ov)
        if alpha > 2000:
            top_idx = np.argsort(overlaps)[::-1][:top_k]
            weights = np.zeros(n_ov)
            top_vals = overlaps[top_idx]
            w_top = top_vals - top_vals[0]
            w_top = np.clip(alpha * w_top, -500.0, 0.0)
            e = np.exp(w_top)
            sw = np.sum(e)
            if sw < 1e-300:
                weights[top_idx[0]] = 1.0
            else:
                weights[top_idx] = e / sw
            active = top_idx
        else:
            m = float(np.max(overlaps))
            logits = alpha * (overlaps - m)
            logits = np.clip(logits, -500.0, 0.0)
            e = np.exp(logits)
            sw = np.sum(e)
            if sw < 1e-300:
                weights = np.zeros(n_ov)
                weights[np.argmax(overlaps)] = 1.0
            else:
                weights = e / sw
            threshold = 1.0 / (50.0 * n_ov)
            active = np.where(weights > threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(overlaps)])
        g = weighted_grad(h, weights, n, dx, active)
        lr = lr_init * (lr_decay ** t)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new
        alpha = min(alpha_max, alpha * alpha_rate)
    return best_h, best_c5

def round_to_binary(h, n, target):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin

def swap_delta(h, i, j, k, n, dx):
    delta = 0.0
    ik = i + k
    ikm = i - k
    jk = j + k
    jkm = j - k
    if 0 <= ik < n:
        delta -= (1.0 - h[ik])
    if 0 <= ikm < n:
        delta += h[ikm]
    if 0 <= jk < n:
        delta += (1.0 - h[jk])
    if 0 <= jkm < n:
        delta -= h[jkm]
    return delta * dx

def binary_local_search(h_bin, n, target, dx, time_limit, max_rounds=200):
    start = time.time()
    h = h_bin.copy()
    n_half = n - 1
    overlaps = compute_overlaps(h)
    current_max = float(np.max(overlaps))
    best_h = h.copy()
    best_c5 = current_max

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        n_crit = 30
        crit_idx = np.argsort(overlaps)[::-1][:n_crit]
        crit_threshold = overlaps[crit_idx[-1]]
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        rng_local = np.random.RandomState(rnd * 7 + 3)
        rng_local.shuffle(zeros)
        best_gain = -1e-9
        best_swap = None
        for ci in crit_idx[:15]:
            k = ci - n_half
            deficit_positions = []
            for pos in ones:
                if 0 <= pos + k < n and h[pos + k] == 0.0:
                    deficit_positions.append(pos)
                if improved:
                    break
            if not deficit_positions:
                continue
            for i in deficit_positions[:10]:
                if improved:
                    break
                for j in zeros[:50]:
                    if j == i:
                        continue
                    delta_ci = swap_delta(h, i, j, k, n, dx)
                    if delta_ci < best_gain - 0.005:
                        worst_other = 0.0
                        ok = True
                        for co in crit_idx[:10]:
                            if co == ci:
                                continue
                            ko = co - n_half
                            do = swap_delta(h, i, j, ko, n, dx)
                            if overlaps[co] + do > overlaps[ci] + delta_ci + 0.002:
                                ok = False
                                break
                            if do > worst_other:
                                worst_other = do
                        if ok:
                            best_gain = delta_ci
                            best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            overlaps = compute_overlaps(h)
            new_max = float(np.max(overlaps))
            if new_max < current_max - 1e-12:
                current_max = new_max
                improved = True
                if current_max < best_c5 - 1e-14:
                    best_c5 = current_max
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
                overlaps = compute_overlaps(h)
        if not improved:
            break
    return best_h, best_c5

def anneal_binary(h_bin, n, target, dx, time_limit, T_init=0.003, T_end=1e-5,
                  max_iter=50000):
    start = time.time()
    h = h_bin.copy()
    n_half = n - 1
    overlaps = compute_overlaps(h)
    current_max = float(np.max(overlaps))
    best_h = h.copy()
    best_c5 = current_max
    rng = np.random.RandomState(12345)
    n_iter_est = max(1000, int(time_limit * 30))
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        T = T_init * (T_end / T_init) ** (t / max(1, n_iter_est))
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        overlaps_new = compute_overlaps(h)
        new_max = float(np.max(overlaps_new))
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            current_max = new_max
            overlaps = overlaps_new
            if current_max < best_c5 - 1e-14:
                best_c5 = current_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def fractional_polish(h_bin, n, target, dx, time_limit):
    start = time.time()
    h = h_bin.astype(float).copy()
    eps = 0.05
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    boundary_ones = ones[:min(40, len(ones))]
    boundary_zeros = zeros[:min(40, len(zeros))]
    h[boundary_ones] = 1.0 - eps
    h[boundary_zeros] = eps
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 5000.0
    lr = 0.003
    n_ov = 2 * n - 1
    for t in range(20000):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        cur_max = float(np.max(overlaps))
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
        top_idx = np.argsort(overlaps)[::-1][:40]
        m = overlaps[top_idx[0]]
        w_top = np.clip(alpha * (overlaps[top_idx] - m), -500.0, 0.0)
        e = np.exp(w_top)
        sw = np.sum(e)
        if sw < 1e-300:
            weights = np.zeros(n_ov)
            weights[top_idx[0]] = 1.0
        else:
            weights = np.zeros(n_ov)
            weights[top_idx] = e / sw
        g = weighted_grad(h, weights, n, dx, top_idx)
        h_new = h - lr * g
        h_new = project_feasible(h_new, target)
        h = h_new
        lr *= 0.999
        alpha = min(500000.0, alpha * 1.005)
    return best_h, best_c5

def multi_start_fw(h_init, n, target, dx, time_limit, seed):
    start = time.time()

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    n_ov = 2 * n - 1

        K = 50
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = weighted_grad(h, np.ones(K) / K, n, dx, top_idx)
        v = np.zeros(n)
        sort_idx = np.argsort(g)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h = project_feasible(h_new, target)
    return best_h, best_c5
