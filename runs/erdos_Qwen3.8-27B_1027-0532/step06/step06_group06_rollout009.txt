```python
def active_set_objective(h, k_active, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top-k + some margin
    sorted_idx = np.argsort(corr)[-k_active:]
    top_vals = corr[sorted_idx]
    # Smooth max of top-k
    beta = 200
    c_ref = top_vals.max()
    shifted = beta * (top_vals - c_ref)
    smooth_max = c_ref + np.log(np.sum(np.exp(shifted))) / beta
    # Also add a penalty for any corr exceeding smooth_max
    excess = np.maximum(corr - c_ref, 0)
    val = smooth_max * dx + 0.1 * np.sum(excess**2) * dx
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val
```

The delta computation for a swap (i, j) can be vectorized:

For the effect on corr[k] at lag k:
- From h[i] 1→0: corr[k] -= (1-h[i+k]) if 0≤i+k<n, minus h[i-k] if 0≤i-k<n
- From h[j] 0→1: corr[k] += (1-h[j+k]) if 0≤j+k<n, minus h[j-k] if 0≤j-k<n

So delta[k] = -(1-h[i+k])*[valid] - h[i-k]*[valid] + (1-h[j+k])*[valid] - h[j-k]*[valid]

I can compute this as:
  delta = -(1-h[i+k])*[valid] - h[i-k]*[valid] + (1-h[j+k])*[valid] - h[j-k]*[valid]

where each term is a vector of length 2n-1 (indexed by lag).

```python
    lags = np.arange(-(n-1), n)  # length 2n-1

    idx1 = i + lags
    valid1 = (idx1 >= 0) & (idx1 < n)

    # Wait, need to be careful with indexing
    
    # Actually let me just use direct computation
    delta = np.zeros(2*n - 1)
    
    # h[i] 1->0:
    # corr[k+k*] where k* = n-1: the term h[i]*(1-h[i+k]) is at index i+k + ... 
    # Let me use the lag convention: lag r means correlation between h[x] and (1-h)[x+r]
    # corr array index = r + n - 1
    # corr[r+n-1] = sum_x h[x] * (1-h[x+r]) for valid x
    
    # Effect of h[i] going 1->0 on corr[r+n-1]:
    # - The term where x=i: h[i]*(1-h[i+r]) = 1*(1-h[i+r]) removed, if 0<=i+r<n
    # - The term where x+r=i, i.e., x=i-r: h[i-r]*(1-h[i]) = h[i-r]*1 removed, if 0<=i-r<n
    # So: delta[r+n-1] -= (1-h[i+r]) if 0<=i+r<n
    #             delta[r+n-1] -= h[i-r] if 0<=i-r<n
    
    # Effect of h[j] going 0->1 on corr[r+n-1]:
    # - The term where x=j: 0*(1-h[j+r]) becomes 1*(1-h[j+r]), if 0<=j+r<n
    # - The term where x+r=j, i.e., x=j-r: h[j-r]*(1-0) becomes h[j-r]*(1-1)=0, if 0<=j-r<n
    # So: delta[r+n-1] += (1-h[j+r]) if 0<=j+r<n
    #             delta[r+n-1] -= h[j-r] if 0<=j-r<n
```

```python
def compute_swap_delta(h, i, j):
    n = len(h)
    r = np.arange(-(n-1), n)  # lags, length 2n-1
    
    # idx for h[i+r]
    idx_ip = i + r
    v1 = (idx_ip >= 0) & (idx_ip < n)
    t1 = np.where(v1, 1.0 - h[np.clip(idx_ip, 0, n-1)], 0.0)
    
    # idx for h[i-r]  
    idx_im = i - r
    v2 = (idx_im >= 0) & (idx_im < n)
    t2 = np.where(v2, h[np.clip(idx_im, 0, n-1)], 0.0)
    
    # idx for h[j+r]
    idx_jp = j + r
    v3 = (idx_jp >= 0) & (idx_jp < n)
    t3 = np.where(v3, 1.0 - h[np.clip(idx_jp, 0, n-1)], 0.0)
    
    # idx for h[j-r]
    idx_jm = j - r
    v4 = (idx_jm >= 0) & (idx_jm < n)
    t4 = np.where(v4, h[np.clip(idx_jm, 0, n-1)], 0.0)
    
    delta = -t1 - t2 + t3 - t4
    return delta
```

```python
def project_box_sum(h, target):

    hi = float(h.max()) + 1.0

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def compute_swap_delta(h, i, j, n, lags):
    idx_ip = i + lags
    v1 = (idx_ip >= 0) & (idx_ip < n)
    t1 = np.where(v1, 1.0 - h[np.clip(idx_ip, 0, n - 1)], 0.0)
    idx_im = i - lags
    v2 = (idx_im >= 0) & (idx_im < n)
    t2 = np.where(v2, h[np.clip(idx_im, 0, n - 1)], 0.0)
    idx_jp = j + lags
    v3 = (idx_jp >= 0) & (idx_jp < n)

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_optimize(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_objective(h_cur, beta, target, penalty)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < compute_c5(h_cur):
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def active_set_objective(h, beta, target, penalty, k_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[-k_active:]

    excess = np.maximum(corr - c_ref, 0.0)
    val = smooth_max + 0.5 * float(np.sum(excess * excess))

    val += penalty * d * d
    return val

    h_cur = h_start.copy()
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(active_set_objective, h_cur,
                          args=(beta, target, penalty, k_active),
                          jac=active_set_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def multi_pair_swap_search(h_start, target_sum, time_limit, start_time, max_rounds=200):
    n = len(h_start)
    target_int = int(target_sum)
    lags = np.arange(-(n - 1), n)
    h = h_start.copy().astype(int)
    hf = h.astype(float)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    current_max = float(corr.max())
    best_c5 = current_max * (2.0 / n)
    best_h = h.copy()
    stall = 0

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        if stall > 50:
            break
        top_T = min(15, len(corr))
        top_indices = np.argsort(corr)[-top_T:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        n_ones = len(ones)
        n_zeros = len(zeros)
        improved = False
        for _ in range(200):
            if time.time() - start_time > time_limit:
                break
            i1 = int(ones[rng.randint(n_ones) if False else np.random.randint(n_ones)])
            i2 = int(ones[np.random.randint(n_ones)])
            j1 = int(zeros[np.random.randint(n_zeros)])
            j2 = int(zeros[np.random.randint(n_zeros)])
            if i1 == i2 or j1 == j2:
                continue
            delta1 = compute_swap_delta(hf, i1, j1, n, lags)
            delta2 = compute_swap_delta(hf, i2, j2, n, lags)
            total_delta = delta1 + delta2
            new_corr = corr + total_delta
            nm = float(new_corr.max())
            if nm < current_max - 1e-12:
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                hf = h.astype(float)
                corr = np.correlate(hf, 1.0 - hf, mode='full')
                current_max = float(corr.max())
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                n_ones = len(ones)
                n_zeros = len(zeros)
                improved = True
                c5_now = current_max * (2.0 / n)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h.astype(float), best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(h_init, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if tleft() > 60:
        time_alloc = min(180, tleft() * 0.25)
        h_ls, c5_ls = fast_binary_local_search(h_bin, target_int, time_alloc, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    if tleft() > 80:
        time_alloc = min(200, tleft() * 0.25)
        max_it = int(min(150000, time_alloc * 3000))
        h_sa, c5_sa = sa_binary_fast(best_h, target_int, max_it, 0.01, 0.00003,
                                      rng, time_alloc, start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    if tleft() > 60:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [50, 100, 200, 500]:
            if tleft() < 30:
                break
            time_alloc = min(40, tleft() * 0.2)
            h_opt = lbfgs_optimize(h_cont, beta, target, 500.0, 200, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    if tleft() > 60:
        h_bin_now = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.2)
        h_ls2, c5_ls2 = fast_binary_local_search(h_bin_now, target_int, time_alloc, start_time)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    if tleft() > 80:
        h_cont = project_box_sum(best_h.copy(), target)
        k_active_list = [5, 10, 20]
        ka = k_active_list[0]
        for beta in [30, 80, 200]:
            if tleft() < 30:
                break
            time_alloc = min(45, tleft() * 0.2)
            h_opt = lbfgs_active_set(h_cont, beta, target, 800.0, ka, 200, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    if tleft() > 40:
        h_bin_now = project_binary(best_h, target_int)
        time_alloc = min(80, tleft() * 0.3)
        h_mp, c5_mp = multi_pair_swap_search(h_bin_now, target_int, time_alloc, start_time)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()

    if tleft() > 60:
        h_bin_now = project_binary(best_h, target_int)
        time_alloc = min(150, tleft() * 0.3)
        max_it = int(min(200000, time_alloc * 3000))
        h_sa2, c5_sa2 = sa_binary_fast(h_bin_now, target_int, max_it, 0.006, 0.00001,
                                        rng, time_alloc, start_time)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    if tleft() > 60:
        for restart in range(3):
            if tleft() < 40:
                break
            noise = rng.normal(0, 0.03, n)
            h_pert = project_box_sum(best_h + noise, target)
            h_bin_pert = project_binary(h_pert, target_int)
            time_alloc = min(30, tleft() * 0.2)
            h_ls3, c5_ls3 = fast_binary_local_search(h_bin_pert, target_int, time_alloc, start_time)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
            if tleft() < 20:
                break
            h_cont_pert = project_box_sum(h_ls3, target)
            time_alloc2 = min(30, tleft() * 0.2)
            h_opt2 = lbfgs_optimize(h_cont_pert, 200, target, 800.0, 150, time_alloc2, start_time)
            sc2 = compute_c5(h_opt2)
            if sc2 < best_c5:
                best_c5 = sc2
                best_h = h_opt2.copy()

    if tleft() > 40:
        h_bin_final = project_binary(best_h, target_int)
        time_alloc = min(60, tleft() * 0.5)
        h_ls4, c5_ls4 = fast_binary_local_search(h_bin_final, target_int, time_alloc, start_time, max_rounds=300)
        if c5_ls4 < best_c5:
            best_c5 = c5_ls4
            best_h = h_ls4.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_final = project_box_sum(best_h.copy(), target_use)
        for beta_f in [200, 500, 1000]:
            if tleft() < 8:
                break
            time_alloc = min(15, tleft() * 0.4)
            h_pol = lbfgs_optimize(h_final, beta_f, target_use, 2000.0, 100, time_alloc, start_time)
            h_final = h_pol
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
```
