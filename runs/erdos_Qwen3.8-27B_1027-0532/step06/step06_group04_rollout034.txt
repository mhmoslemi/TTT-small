import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def solve_lp_subproblem(h, n, target, active_list, scs_eps, max_iters, reg, center):
    dx = 2.0 / n
    half_n = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h
    for k_idx in active_list:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    if reg > 1e-15 and center is not None:
        objective = t_var + reg * cp.sum_squares(h_var - center)
    else:
        objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=max_iters, verbose=False)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), t_var.value
    except Exception:
        pass
    try:
        prob.solve(solver=cp.ECOS, max_iter=max_iters)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), t_var.value
    except Exception:
        pass
    return None, None


def slp_optimize(h_init, n, target, max_iter, time_limit, n_active, start_time,
                 scs_eps=1e-12, reg0=1e-4, damp0=0.5):
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    dx = 2.0 / n
    damping = damp0
    current_reg = reg0

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = sorted(np.argsort(corr)[-n_active:].tolist())
    stall_count = 0

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break

        h_new, t_val = solve_lp_subproblem(h, n, target, active_set, scs_eps, 50000, current_reg, h)
        if h_new is None:
            active_set = sorted(np.argsort(corr)[-max(10, n_active // 2)].tolist())
            continue

        h_trial = (1.0 - damping) * h + damping * h_new
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)

        if c5_trial < best_c5 - 1e-15:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            damping = min(0.99, damping + 0.05)
            stall_count = 0
        else:
            h = h_trial
            damping = max(0.15, damping * 0.85)
            stall_count += 1
            if stall_count > 40:
                damping = 0.5
                current_reg = max(1e-8, current_reg * 0.5)
                stall_count = 0

        if it % 20 == 0 and current_reg > 1e-9:
            current_reg = max(1e-9, current_reg * 0.7)

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_ref = t_val if t_val is not None else best_c5
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_set and corr[k_idx] * dx > t_ref + 2e-5:
                active_set.append(k_idx)
                break
        active_set = sorted(list(top_k | set(active_set)))
        if len(active_set) > 4 * n_active:
            active_set = active_set[-4 * n_active:]

    return best_h, best_c5


def coord_descent_fast(h_init, target, n_active, max_passes, time_limit, start_time, rng):
    n = len(h_init)
    dx = 2.0 / n
    half_n = n - 1
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        active_k = np.argsort(corr)[-n_active:]

        for i in range(n):
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            C = slopes
            lo_b, hi_b = 0.0, 1.0
            for _ in range(45):
                mid = 0.5 * (lo_b + hi_b)
                vals = C * mid + intercepts
                mv = np.max(vals)
                active_mask = np.abs(vals - mv) < 1e-12 * max(1.0, abs(mv))
                if np.all(C[active_mask] > 1e-14):
                    hi_b = mid
                elif np.all(C[active_mask] < -1e-14):
                    lo_b = mid
                else:
                    break
            h[i] = 0.5 * (lo_b + hi_b)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def nesterov_subgrad(h_init, n, target, max_steps, time_limit, start_time, alpha0=200.0):
    dx = 2.0 / n
    half_n = n - 1
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    y = h.copy()
    alpha = alpha0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        s = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        weights = np.exp(s)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            break
        weights = weights / wsum

        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = k_idx - half_n
            idx_pos = np.arange(n) + k
            valid_pos = (idx_pos >= 0) & (idx_pos < n)
            idx_neg = np.arange(n) - k
            valid_neg = (idx_neg >= 0) & (idx_neg < n)
            grad[valid_pos] += w * (1.0 - y[idx_pos[valid_pos]]) * dx
            grad[valid_neg] -= w * y[idx_neg[valid_neg]] * dx

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / t_new
        z = y + theta * (y - h)
        lr = 1.0 / (alpha * n * dx)
        h_new = z - lr * grad
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()

        h = h_new
        y = h_new - lr * grad
        y = np.clip(y, 0.0, 1.0)
        t_k = t_new

        if step % 200 == 199:
            alpha = max(50.0, alpha * 0.7)

    return best_h, best_c5


def block_shift_perturb(h, shift_blocks, rng):
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    n_blocks = shift_blocks
    block_size = n // n_blocks
    perm = rng.permutation(n_blocks)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        new_start = perm[b] * block_size
        new_end = new_start + block_size if perm[b] < n_blocks - 1 else n
        h_new[new_start:new_end] = h[start:end]
    return project_feasible(h_new, target)

            best_h = h_c

    # Phase 12: Final micro-refinement
    if remaining() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ng, s_ng = nesterov_subgrad(best_h, n_use, target_use,
                                      max_steps=2000,
                                      time_limit=min(15, remaining() * 0.7),
                                      start_time=start_time, alpha0=2000.0)
        if s_ng < best_score:
            best_score = s_ng
            best_h = h_ng

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
