                return float(np.max(corr_test)) * dx

            result = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-7})
            gamma = result.x
        else:
            gamma = 2.0 / (step + 2.0)

        h_new = h + gamma * d
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def fw_restart(h_start, n, target, time_limit, max_steps, alpha_temp, n_restarts, rng):
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    for r in range(n_restarts):
        if time.time() - (start if False else 0) > time_limit:
            pass
        if time.time() - fw_restart._t0 > time_limit:
            break
        h_init = best_h + rng.randn(n) * 0.08
        h_init = project_feasible(h_init, target)
        h_new, sc = fw_optimize(h_init, n, target,
                                time_limit=time_limit - (time.time() - fw_restart._t0),
                                max_steps=max_steps,
                                alpha_temp=alpha_temp, use_line_search=True)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    return best_h, best_c5


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    start = time.time()
    if time_limit is None:
        time_limit = 300

    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            max_C = np.max(C)
            shifted = np.clip(alpha * (C - max_C), -500.0, 0.0)
            return float(max_C + np.log(np.sum(np.exp(shifted))) / alpha)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass

    return h_cur, best_c5


def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5


def soft_weighted_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 1000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 10
    return best_h, best_c5


def local_perturbation(h_init, n, target, max_tries, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.01, 0.02, 0.05, 0.08, 0.12, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Frank-Wolfe with line search (new approach)
    if tleft() > 200:
        h_fw, sc = fw_optimize(best_h, n, target,
                               time_limit=min(100, tleft() * 0.3),
                               max_steps=100000,
                               alpha_temp=25.0, use_line_search=True)
        if sc < best_score:
            best_score = sc
            best_h = h_fw

    # Phase 2: SLSQP polish
    if tleft() > 250:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [100, 500, 2000, 10000, 50000, 200000, 500000],
                                   2000, 1e-16,
                                   time_limit=min(180, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Adam with tuned params
    if tleft() > 150:
        h_adam, sc = adam_optimize(best_h, n, target,
                                   time_limit=min(80, tleft() * 0.3),
                                   max_steps=30000, n_top_start=2, n_top_end=10,
                                   lr=0.012, beta1=0.88, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_adam

    # Phase 4: Frank-Wolfe with higher temperature (more exploration)
    if tleft() > 180:
        h_fw, sc = fw_optimize(best_h, n, target,
                               time_limit=min(100, tleft() * 0.35),
                               max_steps=100000,
                               alpha_temp=60.0, use_line_search=True)
        if sc < best_score:
            best_score = sc
            best_h = h_fw

    # Phase 5: SLSQP polish
    if tleft() > 200:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [5000, 20000, 100000, 500000],
                                   1500, 1e-16,
                                   time_limit=min(120, tleft() * 0.45))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Soft weighted subgradient (handles "flat" regions well)
    if tleft() > 120:
        h_sg, sc = soft_weighted_subgrad(best_h, n, target,
                                         time_limit=min(60, tleft() * 0.3),
                                         max_steps=30000, alpha_w=50.0, base_lr=0.015)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 7: Noise + re-optimize for escaping local minima
    rng = np.random.RandomState(seed)
    for noise_scale in [0.015, 0.04, 0.08]:
        if tleft() < 130:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_optimize(h_pert, n, target,
                                   [500, 2000, 10000, 50000, 200000],
                                   800, 1e-15, time_limit=50)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: More Frank-Wolfe with restarts
    if tleft() > 150:
        fw_restart._t0 = time.time()
        h_fw, sc = fw_restart(best_h, n, target,
                              time_limit=min(80, tleft() * 0.4),
                              max_steps=30000, alpha_temp=40.0,
                              n_restarts=3, rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_fw

    # Phase 9: Local perturbation
    if tleft() > 80:
        h_pert, sc = local_perturbation(best_h, n, target,
                                        max_tries=5000,
                                        time_limit=min(30, tleft() * 0.3), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_pert
        if tleft() > 50:
            h_pol, sc = slsqp_optimize(best_h, n, target,
                                       [5000, 20000, 100000, 500000],
                                       800, 1e-16, time_limit=40)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 10: Higher resolution attempt
    if tleft() > 180:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = slsqp_optimize(h_interp, n_try, target_try,
                                     [100, 500, 2000, 10000, 50000, 200000],
                                     1200, 1e-15,
                                     time_limit=min(120, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try

    # Final polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [20000, 100000, 500000, 2000000],
                                     600, 1e-16, time_limit=min(30, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
