def project_binary(h, target_sum):

def continuous_refine(h, n_iters, time_limit, start_time, alpha_init=20.0, alpha_end=500.0, lr_init=0.03, lr_end=0.0005):
    n = len(h)
    target = float(n) / 2.0
    dx = 2.0 / n
    h = project_box_sum(h.astype(float), target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 3000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        alpha = alpha_init * (alpha_end / alpha_init) ** ratio
        
        # Log-sum-exp smooth approximation of max
        c_max = float(corr.max())
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -400.0, 0.0)
        exp_w = np.exp(shifted)
        smooth_val = c_max + float(np.log(exp_w.sum())) / alpha
        W = exp_w / exp_w.sum()
        
        # Gradient of smooth objective w.r.t. h
        # d/dh_i [sum_k W_k * corr_k] where corr_k = sum_i h_i(1-h_{i+k})
        # = sum_k W_k * sum_j [delta_{i,j}(1-h_{j+k}) + h_j * delta_{i,j+k}(-1)]
        # = sum_k W_k * [(1-h_{i+k}) - h_{i-k}]  (for appropriate i)
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = W[k_idx]
            if wk < 1e-8:
                continue
            s = shifts[k_idx]
            if s >= 0:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += wk * (-h[abs_s:])
        
        h = h - lr * grad
        h = project_box_sum(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    positions = np.arange(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def objective(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = float(np.max(corr))
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
            
            def objective_grad(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = float(np.max(corr))
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                exp_w = np.exp(shifted)
                W = exp_w / exp_w.sum()
                grad = np.zeros(n)
                for k_idx in range(2 * n - 1):
                    wk = W[k_idx]
                    if wk < 1e-8:
                        continue
                    s = shifts[k_idx]
                    if s >= 0:
                        grad[:n - s] += wk * (1.0 - x[s:])
                        grad[s:] += wk * (-x[:n - s])
                    else:
                        abs_s = -s
                        grad[abs_s:] += wk * (1.0 - x[:n - abs_s])
                        grad[:n - abs_s] += wk * (-x[abs_s:])
                return grad
            
            def cons_fun(x):
                return np.sum(x) - target
            
            def cons_jac(x):
                return np.ones(n)
            
            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(objective, h_cur, jac=objective_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def project_binary(h, target_sum):
    n = len(h)

    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

def project_box_sum(y, target_sum):

            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce

            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                probs = A[ones] + 0.001
                probs = probs / probs.sum()
                i = ones[rng.choice(len(ones), p=probs)]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        elif r < 0.80:
            top3_idx = np.argsort(full)[-3:]
            k_pick = top3_idx[rng.randint(len(top3_idx))]
            k_star = int(k_pick) - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                probs_i = A[ones] + 0.001
                probs_i = probs_i / probs_i.sum()
                probs_j = 1.0 - B[zeros]
                probs_j = np.clip(probs_j, 0.001, None)
                probs_j = probs_j / probs_j.sum()
                i = ones[rng.choice(len(ones), p=probs_i)]
                j = zeros[rng.choice(len(zeros), p=probs_j)]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta_full(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def ga_optimize(h_init, pop_size, n_gen, rng, time_limit, start_time):
    n = len(h_init)
    target = int(n / 2)

    population = np.zeros((pop_size, n))
    population[0] = h_init

    for i in range(1, pop_size):
        if i <= 5:
            h_temp = h_init.copy()
            n_flip = rng.randint(10, 40) * i
            ones = np.where(h_temp == 1.0)[0]
            zeros = np.where(h_temp == 0.0)[0]
            for _ in range(n_flip):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h_temp[a] = 0.0
                h_temp[b] = 1.0
                ones = np.where(h_temp == 1.0)[0]
                zeros = np.where(h_temp == 0.0)[0]
            population[i] = h_temp
        else:
            h_temp = rng.randint(0, 2, n).astype(float)
            h_temp = project_binary(h_temp, target)
            population[i] = h_temp

    fitness = np.zeros(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_c5(population[i])

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    for gen in range(n_gen):
        if time.time() - start_time > time_limit:
            break

        order = np.argsort(fitness)
        population = population[order]
        fitness = fitness[order]

        n_elite = 5
        new_pop = np.zeros_like(population)
        new_fitness = np.zeros(pop_size)
        new_pop[:n_elite] = population[:n_elite]
        new_fitness[:n_elite] = fitness[:n_elite]

        n_offspring = pop_size - n_elite
        for i in range(n_offspring):
            t1 = rng.randint(0, min(6, pop_size))
            t2 = rng.randint(0, min(6, pop_size))
            p1 = population[t1] if fitness[t1] < fitness[t2] else population[t2]
            t1 = rng.randint(0, min(6, pop_size))
            t2 = rng.randint(0, min(6, pop_size))
            p2 = population[t1] if fitness[t1] < fitness[t2] else population[t2]

            if rng.random() < 0.75:
                mask = rng.rand(n) < 0.5
                child = np.where(mask, p1, p2).astype(float)
                child = project_binary(child, target)
            else:
                child = p1.copy()

            n_mut = rng.randint(1, 3) + (1 if gen > n_gen // 2 else 0)
            for _ in range(n_mut):
                ones = np.where(child == 1.0)[0]
                zeros = np.where(child == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    child[a] = 0.0
                    child[b] = 1.0

            new_pop[n_elite + i] = child
            new_fitness[n_elite + i] = compute_c5(child)

        population = new_pop
        fitness = new_fitness

        gen_best = int(np.argmin(fitness))
        if fitness[gen_best] < best_c5:
            best_c5 = fitness[gen_best]
            best_h = population[gen_best].copy()

    return best_h, best_c5


class SmoothObjective:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
        self.shifts = np.arange(2 * n - 1) - (n - 1)

    def __call__(self, x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(self.alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / self.alpha

    def gradient(self, x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(self.alpha * (corr - c_max), -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        grad = np.zeros(self.n)
        for k_idx in range(2 * self.n - 1):
            wk = W[k_idx]
            if wk < 1e-8:
                continue
            s = self.shifts[k_idx]
            if s >= 0:
                grad[:self.n - s] += wk * (1.0 - x[s:])
                grad[s:] += wk * (-x[:self.n - s])
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - x[:self.n - abs_s])
                grad[:self.n - abs_s] += wk * (-x[abs_s:])
        return grad


def continuous_refine(h, n_iters, time_limit, start_time, alpha_init=20.0, alpha_end=400.0, lr_init=0.03, lr_end=0.0003):
    n = len(h)
    target = float(n) / 2.0
    dx = 2.0 / n
    h = project_box_sum(h.astype(float), target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 5000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        alpha = alpha_init * (alpha_end / alpha_init) ** ratio

        c_max = float(corr.max())
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()

        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = W[k_idx]
            if wk < 1e-8:
                continue
            s = shifts[k_idx]
            if s >= 0:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += wk * (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += wk * (-h[abs_s:])

        h = h - lr * grad
        h = project_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    shifts = np.arange(2 * n - 1) - (n - 1)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            obj = SmoothObjective(alpha, n)

            def cons_fun(x):
                return np.sum(x) - target

            def cons_jac(x):
                return np.ones(n)

            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(obj, h_cur, jac=obj.gradient,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)


def multi_start_sa(h_base, n_starts, n_iters_each, t_init, t_final, rng, time_limit, start_time):
    n = len(h_base)
    target = int(n / 2)
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)

    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        h_pert = h_base.copy()
        n_pert = rng.randint(5, 15) + s * 3
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_pert[a] = 0.0
            h_pert[b] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        h_res, c5_res = sa_optimize(h_pert, n_iters_each, t_init, t_final,
                                     rng, time_limit, start_time)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    best_cont_h = project_box_sum(h_init, float(n_init) / 2.0)
    best_cont_score = compute_c5(best_cont_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: GA for broad binary exploration
    ga_budget = min(300.0, budget_s * 0.25)
    tl = tleft()
    if tl > ga_budget + 200:
        h_ga, c5_ga = ga_optimize(best_h, pop_size=60, n_gen=3000,
                                   rng=rng, time_limit=ga_budget, start_time=start_time)
        if c5_ga < best_score:
            best_score = c5_ga
            best_h = h_ga.copy()

    # Phase 2: Greedy improvement
    tl = tleft()
    greedy_budget = min(50.0, tl * 0.06)
    if greedy_budget > 10:
        h_gr, c5_gr = greedy_improve(best_h, max_iters=30000, rng=rng,
                                      time_limit=greedy_budget, start_time=start_time)
        if c5_gr < best_score:
            best_score = c5_gr
            best_h = h_gr.copy()

    # Phase 3: Multi-start SA
    tl = tleft()
    ms_budget = min(250.0, tl * 0.25)
    if ms_budget > 60:
        n_starts = 5
        n_iters_each = int(min(2000000, ms_budget * 40000 // n_starts))
        h_ms, c5_ms = multi_start_sa(best_h, n_starts, n_iters_each,
                                      t_init=0.012, t_final=0.00001,
                                      rng=rng, time_limit=ms_budget, start_time=start_time)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 4: Greedy after SA
    tl = tleft()
    greedy_budget2 = min(40.0, tl * 0.05)
    if greedy_budget2 > 10:
        h_gr2, c5_gr2 = greedy_improve(best_h, max_iters=20000, rng=rng,
                                         time_limit=greedy_budget2, start_time=start_time)
        if c5_gr2 < best_score:
            best_score = c5_gr2
            best_h = h_gr2.copy()

    # Phase 5: Continuous refinement (log-sum-exp gradient)
    tl = tleft()
    cont_budget = min(200.0, tl * 0.20)
    if cont_budget > 30:
        n_iters = int(min(3000000, cont_budget * 25000))
        h_cr, c5_cr = continuous_refine(best_h, n_iters,
                                         time_limit=cont_budget, start_time=start_time,
                                         alpha_init=15.0, alpha_end=500.0,
                                         lr_init=0.025, lr_end=0.0002)
        if c5_cr < best_cont_score:
            best_cont_score = c5_cr
            best_cont_h = h_cr.copy()

    # Phase 6: SLSQP polish
    tl = tleft()
    slsqp_budget = min(150.0, tl * 0.15)
    if slsqp_budget > 30:
        h_pol, sc_pol = slsqp_polish(best_cont_h, n_init, float(n_init) / 2.0,
                                       alphas=[40, 150, 500, 2000],
                                       maxiter=2000,
                                       time_limit=slsqp_budget, start_time=start_time)
        if sc_pol < best_cont_score:
            best_cont_score = sc_pol
            best_cont_h = h_pol.copy()

    # Phase 7: Multi-resolution
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8]:
        tl = tleft()
        if tl < 100:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_cont_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        gb = min(20.0, tl * 0.05)
        if gb < 10:
            continue
        h_g3, c5_g3 = greedy_improve(h_bin_try, max_iters=8000, rng=rng,
                                       time_limit=gb, start_time=start_time)

        cb = min(35.0, tl * 0.08)
        if cb < 15:
            continue
        n_iters3 = int(min(800000, cb * 18000))
        h_c3, c5_c3 = continuous_refine(h_g3, n_iters3,
                                         time_limit=cb, start_time=start_time,
                                         alpha_init=30.0, alpha_end=600.0,
                                         lr_init=0.02, lr_end=0.0002)
        if c5_c3 < best_cont_score:
            best_cont_score = c5_c3
            best_cont_h = h_c3.copy()

    # Phase 8: Final SLSQP burst
    tl = tleft()
    if tl > 50:
        n_use = len(best_cont_h)
        h_f, sc_f = slsqp_polish(best_cont_h, n_use, float(n_use) / 2.0,
                                   alphas=[500, 2000, 8000],
                                   maxiter=4000,
                                   time_limit=max(20.0, tl - 20),
                                   start_time=start_time)
        if sc_f < best_cont_score:
            best_cont_score = sc_f
            best_cont_h = h_f.copy()

    # Phase 9: Final fine gradient
    tl = tleft()
    if tl > 25:
        n_use = len(best_cont_h)
        n_iters_f = int(min(800000, max(100000, (tl - 10) * 40000)))
        h_gf, c5_gf = continuous_refine(best_cont_h, n_iters_f,
                                         time_limit=max(8.0, tl - 8),
                                         start_time=start_time,
                                         alpha_init=200.0, alpha_end=3000.0,
                                         lr_init=0.008, lr_end=0.00005)
        if c5_gf < best_cont_score:
            best_cont_score = c5_gf
            best_cont_h = h_gf.copy()

    # Final selection
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    if len(best_h) == n_final:
        h_bin_final = project_binary(best_h, int(n_final / 2.0))
    else:
        h_bin_final = project_binary(best_cont_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
