import numpy as np
from scipy.optimize import minimize
import time

_gp = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    alpha = _gp['alpha']
    target = _gp['target']
    mu = _gp['mu']
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    lse = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    pen = mu * (np.sum(x) - target) ** 2
    return float(lse + pen)

def _cons_fun(x):
    return np.sum(x) - _gp['target']

def _cons_jac(x):
    return np.ones(len(x))

def smooth_opt(h_init, n, target, alphas, method, maxiter, time_limit, mu=1e4):
    """Optimize smooth objective using given method."""
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _gp['alpha'] = alpha
        _gp['target'] = target
        _gp['mu'] = mu
        
        if method == 'L-BFGS-B':
            try:
                res = minimize(_obj, h_cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
                h_new = project_feasible(res.x, target)
            except:
                continue
        else:  # SLSQP
            try:
                res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                              options={'maxiter': maxiter, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
            except:
                continue
        
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            h_cur = h_new
    
    return h_cur, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = h.copy()

        else:
            no_improve += 1

            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        cur_ncand = min(cur_ncand, n // 2)

        high_set = sorted_idx[-cur_ncand:]
