        active = sorted_idx[:top_m]
        within = np.where(c5_vals >= current_max - epsilon)[0]
        active = np.unique(np.concatenate([active, within]))

        if len(active) > 60:
            active = sorted_idx[:60]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_list = []
        b_list = []
        for k_idx in active:
            k = int(k_idx) - mid
            f_k = float(c5_vals[k_idx])
            g_k = compute_grad_shift(h, k, n, dx)
            rhs = f_k - g_k @ h
            row = np.zeros(n + 1)
            row[:n] = g_k
            row[n] = -1.0
            A_list.append(row)
            b_list.append(rhs)

        A_ub = np.array(A_list)
        b_ub = np.array(b_list)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
            if not res.success:
                if 'infeas' in res.message.lower():
                    active = sorted_idx[:max(5, top_m // 2)]
                    if len(active) < 3:
                        break
                    continue
                break

            h_lp = res.x[:n]
            direction = h_lp - h
            dir_norm = np.linalg.norm(direction)
            if dir_norm < 1e-12:
                break

            best_alpha = 0.0
            best_trial = current_max
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
                h_try = h + alpha * direction
                h_try = project_feasible(h_try, target)
                trial = compute_c5(h_try)
                if trial < best_trial:
                    best_trial = trial
                    best_alpha = alpha

            if best_alpha > 0:
                h = project_feasible(h + best_alpha * direction, target)
                history.append(best_trial)
                if len(history) > 200:
                    if history[0] - history[-1] < 1e-10:
                        break
            else:
                break
        except Exception:
            break

    return best_h, best_c5

def subgradient_method(h_init, n, target, time_limit, max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    eta = 0.01
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        current_max = float(np.max(c5_vals))

        if current_max < best_c5 - 1e-14:
            best_c5 = current_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        weights = c5vals - current_max
        weights = np.exp(weights / 0.005)
        weights /= np.sum(weights)
        active_weights = weights[weights > 1e-8]
        active_idx = np.where(weights > 1e-8)[0]

        grad = np.zeros(n)
        for k_idx in active_idx:
            k = int(k_idx) - mid
            g_k = compute_grad_shift(h, k, n, dx)
            grad += weights[k_idx] * g_k

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        h_try = project_feasible(h - eta * grad / grad_norm, target)
        trial = compute_c5(h_try)
        if trial < current_max:
            h = h_try
            eta *= 1.01
        else:
            eta *= 0.7
            if eta < 1e-8:
                break

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=8000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp_smooth_max(h_init, n, target, time_limit, alphas=None, maxiter_per_alpha=800):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    start = time.time()
    if alphas is None:
        alphas = [200, 1000, 5000, 20000]
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smoosh_wrapper, h_cur, method='SLSQP',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass

    return best_h, best_c5

_slsqp_n = 0
_slsqp_target = 0.0

def _smoosh_wrapper(h):
    n = _slsqp_n
    target = _slsqp_target
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max = np.max(corr)
    alpha = 1000.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def _smoosh_jac(h):
    n = _slsqp_n
    h = np.clip(h, 0.0, 1.0)
    idxs = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max = np.max(corr)
    alpha = 1000.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[k_idx] * dx * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[k_idx] * dx * h[idxs[v2] + k]
    return grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_start_lp(h_init, n, target, time_limit, n_starts=4, seed_val=12345):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for s in range(n_starts):
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.2]))
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        t_alloc = remaining * 0.8 / (n_starts - s)
        h_try, c5_try = lp_chebyshev(h_pert, n, target, time_limit=min(t_alloc, remaining - 5),
                                      max_iters=3000, top_m=10, epsilon=0.002)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target
    t0 = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP Chebyshev - the main new approach
    if tleft() > 150:
        t1 = min(250, tleft() * 0.3)
        h_lp, c5_lp = lp_chebyshev(best_h, n, target, time_limit=t1,
                                    max_iters=20000, top_m=12, epsilon=0.0012)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Pair CD fine-tuning
    if tleft() > 150:
        t2 = min(200, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=t2,
                              max_sweeps=8000, n_cand=45, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP Chebyshev again with refined params
    if tleft() > 100:
        t3 = min(200, tleft() * 0.25)
        h_lp2, c5_lp2 = lp_chebyshev(best_h, n, target, time_limit=t3,
                                      max_iters=20000, top_m=8, epsilon=0.0008)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 4: Multi-start LP
    if tleft() > 120:
        t4 = min(150, tleft() * 0.2)
        h_ms, c5_ms = multi_start_lp(best_h, n, target, time_limit=t4,
                                      n_starts=4, seed_val=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 5: SLSQP smooth max polish
    if tleft() > 80:
        t5 = min(100, tleft() * 0.15)
        h_slq, c5_slq = slsqp_smooth_max(best_h, n, target, time_limit=t5,
                                          alphas=[2000, 10000, 50000],
                                          maxiter_per_alpha=600)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 6: Higher resolution attempt
    if tleft() > 250:
        for n_try in [1000]:
            if tleft() < 200:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            # LP on higher resolution
            t_lp = min(120, tleft() * 0.3)
            h_hi, c5_hi = lp_chebyshev(h_interp, n_try, target_try,
                                        time_limit=t_lp, max_iters=5000,
                                        top_m=10, epsilon=0.0015)
            if tleft() > 80:
                t_cd = min(80, tleft() * 0.2)
                h_hi2, c5_hi2 = pair_cd(h_hi, n_try, target_try,
                                         time_limit=t_cd, max_sweeps=3000,
                                         n_cand=35, n_bisect=13)
                if c5_hi2 < best_score:
                    best_score = c5_hi2
                    best_h = h_hi2
                    n = n_try
                    target = target_try

    # Phase 7: Final LP Chebyshev with very tight epsilon
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t7 = min(100, tleft() * 0.3)
        h_lp3, c5_lp3 = lp_chebyshev(best_h, n_use, target_use, time_limit=t7,
                                      max_iters=50000, top_m=6, epsilon=0.0005)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 8: Final pair CD
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t8 = min(60, tleft() * 0.3)
        h_cd2, c5_cd2 = pair_cd(best_h, n_use, target_use, time_limit=t8,
                                max_sweeps=5000, n_cand=50, n_bisect=15)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 9: Final LP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t9 = min(30, tleft() * 0.5)
        h_lp4, c5_lp4 = lp_chebyshev(best_h, n_use, target_use, time_limit=t9,
                                      max_iters=20000, top_m=5, epsilon=0.0003)
        if c5_lp4 < best_score:
            best_score = c5_lp4
            best_h = h_lp4

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
