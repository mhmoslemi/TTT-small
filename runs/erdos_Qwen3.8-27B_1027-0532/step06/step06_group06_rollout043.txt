def update_corr_after_swap(corr, h, i, j, n):
    offset = n - 1
    # For position i (going from 1 to 0):
    # k range: max(-(n-1), i-n+1) to min(n-1, i)
    k_lo_i = max(-(n-1), i - (n-1))  # = max(-(n-1), i-n+1)
    k_hi_i = min(n-1, i)
    idx_i = np.arange(k_lo_i, k_hi_i + 1) + offset
    # h[i-k] for k in idx range
    vals_i = h[i - (idx_i - offset)]  # = h[i - k] where k = idx_i - offset
    
    # For position j (going from 0 to 1):
    k_lo_j = max(-(n-1), j - (n-1))
    k_hi_j = min(n-1, j)
    idx_j = np.arange(k_lo_j, k_hi_j + 1) + offset
    vals_j = h[j - (idx_j - offset)]
    
    corr[idx_i] += -1.0 + 2.0 * vals_i
    corr[idx_j] += 1.0 - 2.0 * vals_j
    return corr

    # g[j]: (1-h[j]) 1->0, affects m = j+k
    jk2 = j + k
    if 0 <= jk2 < n:
        delta += -h[jk2]
    return delta

        corr[ks + offset] += -h[j + ks]

def smooth_ergodic_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_ergodic_grad(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_refine(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_ergodic_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_ergodic_grad,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < compute_c5(h_cur):
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

        h_it, c5_it = itls_binary(h_bin, target_int, time_limit=time_alloc,
                                    start_time=start_time, rng=rng)
        if c5_it < best_c5:
            best_c5 = c5_it
            best_h = h_it.copy()

    # Phase 5: Second SA round from best
    if tleft() > 100:
        h_bin_best = project_binary(best_h, target_int)
        time_alloc = min(150, tleft() * 0.3)
        h_sa2, c5_sa2 = fast_sa_binary(h_bin_best, target_int, max_iters=3000000,
                                        t_init=0.004, t_final=0.000002,
                                        rng=rng, time_limit=time_alloc,
                                        start_time=start_time, max_stall=3000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Try different grid sizes
    n_options = [n - 4, n + 4, n + 8, n - 8, n + 12]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]
    for n_try in n_options:
        if tleft() < 40:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try_int)
        h_g_try, c5_try = greedy_binary_fast(h_bin_try, target_try_int,
                                              time_limit=min(20, tleft() * 0.15),
                                              start_time=start_time, max_iters=50000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_g_try.copy()
        if tleft() < 30:
            continue
        h_sa_try, c5_sa_try = fast_sa_binary(h_g_try, target_try_int, max_iters=1000000,
                                              t_init=0.005, t_final=0.00001,
                                              rng=rng, time_limit=min(30, tleft() * 0.15),
                                              start_time=start_time, max_stall=2000)
        if c5_sa_try < best_c5:
            best_c5 = c5_sa_try
            best_h = h_sa_try.copy()

    # Phase 7: Continuous optimization from best binary
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_cont = project_box_sum(best_h.copy(), target_use)
        for beta_f in [100, 300, 800]:
            if tleft() < 15:
                break
            h_pol = lbfgs_refine(h_cont, beta_f, target_use, penalty=500.0,
                                  maxiter=150, time_limit=min(20, tleft() * 0.3),
                                  start_time=start_time)
            sc = compute_c5(h_pol)
            h_cont = h_pol
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    # Phase 8: Final greedy polish on best binary projection
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        h_final, c5_final = greedy_binary_fast(h_bin_final, target_use,
                                                time_limit=max(5, tleft() - 10),
                                                start_time=start_time, max_iters=200000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
