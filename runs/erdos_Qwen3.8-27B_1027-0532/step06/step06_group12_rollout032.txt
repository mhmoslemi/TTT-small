- For k>0: ∑_{i=0}^{n-1-k} h_new[i] * (1 - h_old[i+k]) * dx

- k=0: (1 - h_old[i]) * dx for all i

```python
            best_h = h_final

    # Phase 7: One more LP round if time
    tl = remaining()
    if tl > 60:
        lp_time = min(tl * 0.4, 150)
        h_lp3, c5_lp3 = alternating_lp_full(best_h, n, target, lp_time,
                                             max_outer=5000, n_bisect=15,
                                             damping_start=0.9, damping_end=0.5)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 8: Final flip search
    tl = remaining()
    if tl > 20:
        h_ff, c5_ff = local_flip_search(best_h, n, target, min(tl * 0.5, 40), n_tries=10000)
        if c5_ff < best_score:
            best_score = c5_ff
            best_h = h_ff

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
```
