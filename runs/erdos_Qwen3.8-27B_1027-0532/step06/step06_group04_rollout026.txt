import cvxpy as cp

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_active(h_curr, n, target, active_k, half_n, max_iters=100000):
    dx = 2.0 / n
    g = 1.0 - h_curr
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]

        hi = min(n, k + n)
        if hi > lo:

            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=max_iters, verbose=False)
        if h_var.value is not None:

        if h_var.value is not None:

The Erdős minimum overlap problem: we want to find a set A ⊂ [0,2] of measure 1 such that the maximum over all shifts k of the measure of A ∩ (A+k) is minimized... wait, no.

Actually, let me re-read the problem. We have h: [0,2] → [0,1] with ∫h = 1. We want to minimize:
C₅ = max_k ∫ h(x)(1-h(x+k)) dx

Note that ∫ h(x)(1-h(x+k)) dx = ∫ h(x)dx - ∫ h(x)h(x+k)dx = 1 - ∫h(x)h(x+k)dx

So C₅ = max_k [1 - ∫h(x)h(x+k)dx] = 1 - min_k ∫h(x)h(x+k)dx

So we want to MAXIMIZE min_k ∫h(x)h(x+k)dx.

This is the minimum autocorrelation of h. We want h to have as flat (and high) an autocorrelation as possible.

For h = 1/2 everywhere: ∫h(x)h(x+k)dx = 1/4 for all k (in the continuous case with appropriate boundary handling). So C₅ = 1 - 1/4 = 3/4. That's terrible.

For h = 1 on [0,1/2] ∪ [3/4, 5/4] (measure = 1/2 + 1/2 = 1): The autocorrelation would have high values at certain shifts and low at others. The minimum would determine C₅.

The known result C₅ ≤ 5/13 ≈ 0.3846 comes from a specific construction. Our numerical optimization is doing better (0.3808), which means the optimal h is not a simple step function but has a more complex structure.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_pure(h_curr, n, target, active_k_list, half_n):
    dx = 2.0 / n
    g = 1.0 - h_curr
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in active_k_list:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=200000, verbose=False)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.ECOS, max_iter=200000)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    except Exception:
        pass
    return None


def alternating_lp_tight(h_init, n, target, max_iter, time_limit, n_active, eps_tol):
    start = time.time()
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    t_curr = float(np.max(corr)) * dx
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        h_new = solve_lp_pure(h, n, target, sorted(active_k), half_n)
        if h_new is None:
            break
        h_new = project_feasible(h_new, target)
        c5_val = compute_c5(h_new)

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()

        h = h_new
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_est = float(np.max(corr)) * dx

        new_active = set()
        for k_idx in active_k:
            if corr[k_idx] * dx >= t_est - eps_tol:
                new_active.add(k_idx)
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        active_k = new_active | top_k

        if len(active_k) > 5 * n_active:
            active_k = set(np.argsort(corr)[-5 * n_active:].tolist())

        if abs(c5_val - best_c5) < 1e-15 and it > 50:
            if it > 200:
                break

    return best_h, best_c5


def smooth_obj_fn(h):
    alpha = _g_alpha
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fn(h):
    return np.array([np.sum(h) - _g_target])


def cons_jac_fn(h):
    return np.array([np.ones(_g_n)])


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac_fn}]
    h_cur = h_start.copy()
    cur = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj_fn, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur - 1e-15:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur


def nesterov_proximal(h_init, n, target, max_steps, time_limit, alpha, lr0, momentum0):
    start = time.time()
    dx = 2.0 / n
    half_n = n - 1
    idxs = np.arange(n)
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = momentum0
    lr = lr0
    stall = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        weights = np.exp(alpha * (corr_y - np.max(corr_y)))
        weights /= weights.sum()

        grad = np.zeros(n)
        for k_idx in np.where(weights > 1e-10)[0]:
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = k_idx - half_n
            v1 = (0 <= idxs - k) & ((idxs - k) < n)
            v2 = (0 <= idxs + k) & ((idxs + k) < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad *= dx

        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)

        momentum = 0.5 * (1.0 + momentum)
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)

        sc = compute_c5(h_new)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.1, 0.5)
        else:
            stall += 1
            lr = max(lr * 0.85, 0.001)
            if stall > 100:
                break
        h = h_new

    return best_h, best_c5


def coord_descent_fast(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        order = rng.permutation(n)
        for i in order:
            best_val = h[i]
            best_ci = np.inf
            for c in range(21):
                candidate = c / 20.0
                h[i] = candidate
                corr_i = np.zeros(len(corr))
                for idx in range(len(active_k)):
                    k_idx = active_k[idx]
                    k = k_idx - (n - 1)
                    A = 0.0
                    if 0 <= i + k < n:
                        A += (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        A -= h[i - k]
                    A *= dx
                    corr_i[k_idx] = np.correlate(h, 1.0 - h, mode='full')[k_idx] - A * h[i]
                ci = float(np.max(corr_i)) * dx
                if ci < best_ci:
                    best_ci = ci
                    best_val = candidate
            h[i] = best_val

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
            if p % 50 == 0:
                active_k = np.argsort(corr)[-min(n_active * 2, 200):]

    h = project_feasible(h, target)
    if compute_c5(h) < best_c5:
        best_c5 = compute_c5(h)
        best_h = h.copy()
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_refine(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    half_n = n - 1
    idxs = np.arange(n)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    stall = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        v1 = (0 <= idxs - k) & ((idxs - k) < n)
        v2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h_cur[idxs[v1] - k])
        grad[v2] -= h_cur[idxs[v2] + k]
        grad *= dx

        improved = False
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            h_try = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 120:
                break
    return best_h, best_c5
