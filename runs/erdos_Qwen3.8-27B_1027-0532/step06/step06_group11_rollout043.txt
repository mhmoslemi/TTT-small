import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def correlate_matrix(h, n):
    """Compute all correlation values."""
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

# Global variables for SLSQP
_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_C = np.max(corr)
    shifted = np.clip(_G_alpha * (corr - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def trust_region_qp(h_curr, n, target, active_idx, delta_max):
    """
    Solve a trust region QP subproblem.
    At the current point, the max is achieved at active_idx shifts.
    We approximate the max over these shifts as a quadratic and solve:
    min  0.5 * d^T G d + g^T d + c
    s.t. ||d|| <= delta_max
         h_curr + d in [0,1]^n
         sum(h_curr + d) = target
    """
    dx = 2.0 / n
    corr = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
    c_max = np.max(corr)
    
    # Compute gradient of each active correlation
    mid = n - 1
    grads = []
    for s_idx in active_idx:
        k = int(s_idx) - mid
        g = get_grad_shift(h_curr, k, n) * dx
        g -= np.mean(g)  # To handle the equality constraint
        grads.append(g)
    
    grads = np.array(grads)  # shape: (n_active, n)
    
    # The max is approximated by sum of active gradients (equal weights if all are at max)
    g_max = np.mean(grads, axis=0)
    
    # Hessian approximation: use the fact that each corr_k is quadratic
    # corr_k(h+d) ≈ corr_k(h) + g_k^T d + 0.5 * d^T H_k d
    # H_k has entries related to the second derivative
    # For simplicity, use a diagonal approximation or ignore Hessian
    
    # Simple approach: solve min g_max^T d s.t. ||d|| <= delta_max, 0<=h+d<=1, sum(d)=0
    # This is a linear program in the trust region
    
    # Actually, let's just do a steepest descent step within the trust region
    d = -g_max
    d -= np.mean(d)
    d_norm = np.linalg.norm(d)
    if d_norm > delta_max:
        d *= delta_max / d_norm
    if d_norm < 1e-14:
        return h_curr, c_max
    
    h_new = project_feasible(h_curr + d, target)
    c5_new = compute_c5(h_new)
    
    return h_new, c5_new

def active_set_descent(h_start, n, target, time_limit, max_iters, epsilon, delta_init):
    """
    Active set method: at each step, find the active set and do a step.
    """
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    delta = delta_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        c5_now = float(np.max(corr))
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
            delta = min(delta * 1.5, n / 10.0)
        else:
            no_improve += 1
            delta *= 0.7
        
        # Find active set
        active = np.where(corr >= c5_now - epsilon)[0]
        
        # Compute weighted gradient
        weights = np.zeros(len(active))
        for i, idx in enumerate(active):
            weights[i] = (corr[idx] - (c5_now - epsilon)) / (2 * epsilon + 1e-14)
        weights += 1e-6
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for i, s_idx in enumerate(active):
            k = int(s_idx) - mid
            grad += weights[i] * get_grad_shift(h, k, n)
        grad *= (2.0 / n)
        grad -= np.mean(grad)
        
        # Line search
        best_step = 0.0
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = alpha * delta
            h_try = project_feasible(h + step * (-grad), target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_step = step
                break
        
    return best_h, best_c5

def block_perturbation(h, n, target, rng, n_blocks=5):
    """Perturb random contiguous blocks."""
    h_new = h.copy()
    for _ in range(n_blocks):
        block_len = int(rng.randint(5, n // 4))
        start = int(rng.randint(0, n - block_len))
        amp = float(rng.uniform(0.02, 0.15))
        # Sinusoidal or linear perturbation within block
        t = np.linspace(0, 1, block_len)
        pert = amp * np.sin(np.pi * t)  # Bump-like
        h_new[start:start+block_len] += pert
    return project_feasible(h_new, target)

def sine_perturbation(h, n, target, rng, n_freqs=5):
    """Add sinusoidal perturbations."""
    h_new = h.copy()
    for _ in range(n_freqs):
        freq = int(rng.randint(1, n // 4))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.01, 0.1))
        x = np.arange(n)
        h_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, target)

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def adaptive_subgrad(h_start, n, target, time_limit, max_steps):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.array([(i + 0.5) / n_dst for i in range(n_dst)])
    x_old = np.array([(i + 0.5) / n_src for i in range(n_src)])
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def symmetric_refine(h, n, target):
    """Try to exploit symmetry: average h with its reflection."""
    h_refl = h[::-1].copy()
    h_sym = 0.5 * (h + h_refl)
    h_sym = project_feasible(h_sym, target)
    return h_sym

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

def _slsqp_obj(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_C = np.max(corr)
    shifted = np.clip(_G_alpha * (corr - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

        h_interp2 = interpolate_to(best_h, n_try2, target_try2)
        h_hires2, sc = slsqp_optimize(h_interp2, n_try2, target_try2,
                                      [100, 500, 2000, 10000],
                                      1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_hires2
            n = n_try2
            target = target_try2

    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ad2, sc = adam_optimize(best_h, n_use, target_use,
                                  time_limit=min(50, tleft() * 0.5),
                                  max_steps=10000, n_top_start=5, n_top_end=15,
                                  lr=0.008, beta1=0.85, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_ad2

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pol, sc = slsqp_optimize(best_h, n_use, target_use,
                                   [50000, 200000, 1000000],
                                   800, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
