import numpy as np

_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target

    return np.ones(_global_n)

    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturb_search(h_init, n, target, max_tries, time_limit, seed=777):
    start = time.time()
    rng_local = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 4, 6, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice(scales))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Intensive pair CD at initial resolution
    if tleft() > 300:
        cd_time = min(350, tleft() * 0.35)
        h_cd, c5_cd = pair_cd_intensive(best_h, n, target,
                                        time_limit=cd_time,
                                        max_sweeps=30000,
                                        n_cand_start=30,
                                        n_cand_max=100,
                                        patience=400,
                                        m_top=10,
                                        n_bisect=28,
                                        pert_interval=200,
                                        pert_scale=0.02,
                                        seed=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Nesterov subgradient
    if tleft() > 200:
        sg_time = min(150, tleft() * 0.2)
        h_sg, c5_sg = subgradient_descent(best_h, n, target,
                                          time_limit=sg_time,
                                          max_steps=40000,
                                          mu=0.85, m_top=20,
                                          eta=6.0, base_lr=0.012)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 3: Second round of pair CD (larger candidates)
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_intensive(best_h, n, target,
                                        time_limit=cd_time,
                                        max_sweeps=25000,
                                        n_cand_start=50,
                                        n_cand_max=120,
                                        patience=300,
                                        m_top=12,
                                        n_bisect=30,
                                        pert_interval=150,
                                        pert_scale=0.015,
                                        seed=seed + 1)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: L-BFGS-B polish
    if tleft() > 150:
        for alpha in [2000, 10000, 50000]:
            if tleft() < 40:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=2000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 5: Block CD
    if tleft() > 150:
        bc_time = min(120, tleft() * 0.2)
        h_bc, c5_bc = block_cd(best_h, n, target,
                               time_limit=bc_time,
                               max_sweeps=5000,
                               n_cand=60, patience=200, m_top=10)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 6: Try higher resolution
    n_use = len(best_h)
    if tleft() > 300:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, target_try, 5000, maxiter=1500)
            sc = compute_c5(h_hires)
            if sc < best_score:
                h_cd2, c5_cd2 = pair_cd_intensive(h_hires, n_try, target_try,
                                                   time_limit=min(120, tleft() * 0.3),
                                                   max_sweeps=10000,
                                                   n_cand_start=40,
                                                   n_cand_max=80,
                                                   patience=200,
                                                   m_top=10,
                                                   n_bisect=25,
                                                   pert_interval=100,
                                                   pert_scale=0.02,
                                                   seed=seed + 2)
                if c5_cd2 < sc:
                    sc = c5_cd2
                    h_hires = h_cd2
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n_use = n_try

    # Phase 7: SLSQP final polish
    if tleft() > 80:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        h_pol = slsqp_round(best_h, n_pol, target_pol,
                            [5000, 20000, 100000],
                            1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Local perturbation search
    if tleft() > 60:
        n_l = len(best_h)
        target_l = n_l / 2.0
        h_pert, c5_pert = local_perturb_search(best_h, n_l, target_l,
                                               max_tries=5000,
                                               time_limit=min(40, tleft() * 0.4),
                                               seed=seed + 10)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 9: Final pair CD squeeze
    if tleft() > 60:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h_f, c5_f = pair_cd_intensive(best_h, n_f, target_f,
                                      time_limit=min(50, tleft() * 0.5),
                                      max_sweeps=10000,
                                      n_cand_start=60,
                                      n_cand_max=120,
                                      patience=200,
                                      m_top=12,
                                      n_bisect=30,
                                      pert_interval=300,
                                      pert_scale=0.01,
                                      seed=seed + 20)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
