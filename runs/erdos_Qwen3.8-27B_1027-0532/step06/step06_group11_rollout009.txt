    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]

        idx = np.where(h > 1e-10)[0]

        return h
    h = h.copy()

    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    v1 = np.arange(n) - k
    mask1 = (v1 >= 0) & (v1 < n)
    g[mask1] += (1.0 - h[v1[mask1]])
    v2 = np.arange(n) + k
    mask2 = (v2 >= 0) & (v2 < n)
    g[mask2] -= h[v2[mask2]]
    return g

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def adam_lse(h_start, n, target, time_limit, max_steps,
             alpha_start, alpha_end, top_start, top_end,
             lr, beta1, beta2, eps):

    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    no_improve = 0

        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 3000:
            break
        
        frac = min(step / max(1, int(max_steps * 0.7)), 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')

            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute top-k weighted gradient
        idx_sorted = np.argsort(corr)
        top_idx = idx_sorted[-top_k:]
        
        # Use softmax weights on the top-k
        top_corr = corr[top_idx]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    v1 = np.arange(n) - k
    mask1 = (v1 >= 0) & (v1 < n)
    g[mask1] += (1.0 - h[v1[mask1]])
    v2 = np.arange(n) + k
    mask2 = (v2 >= 0) & (v2 < n)
    g[mask2] -= h[v2[mask2]]
    return g

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                  lr, beta1, beta2, eps):

    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 500 and no_improve > 5000:
            break
        
        frac = min(step / max(1, int(max(1, max_steps * 0.8))), 1.0)
        # Anneal alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        # Anneal top_k
        top_k = max(top_start, int(top_start + (top_end - top_start) * frac))
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Top-k weighted gradient (softmax over top-k)
        idx_sorted = np.argsort(corr)
        top_idx = idx_sorted[-top_k:]
        top_corr = corr[top_idx]
        top_max = np.max(top_corr)
        # Softmax weights (scaled by alpha/10 for stability)
        w = np.exp((alpha / 10.0) * (top_corr - top_max))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(top_k) / top_k
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-10:
                continue
            k = int(top_idx[idx]) - mid
            grad += w[idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        
        t = step + 1
        m_arr = beta1 * m_arr + (1.0 - beta1) * grad
        v_arr = beta2 * v_arr + (1.0 - beta2) * grad * grad
        m_hat = m_arr / (1.0 - beta1 ** t)
        v_hat = v_arr / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

The gradient is: sum_k (exp(alpha*C_k) / sum_j exp(alpha*C_j)) * dC_k/dh

w_k = exp(alpha * C_k) / sum_{j in top-k} exp(alpha * C_j)

The issue is that alpha might be very large (say 50000), making the exponentials overflow or underflow. I handle this by subtracting the max:
w_k = exp(alpha * (C_k - C_max)) / sum_j exp(alpha * (C_j - C_max))
