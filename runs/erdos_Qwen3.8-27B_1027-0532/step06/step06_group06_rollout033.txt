def smooth_gradient(h, beta, target, penalty):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())

    shifted -= shifted.max()
    weights = np.exp(shifted)

    one_minus_h = 1.0 - h

    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_continuous(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_objective(h_cur, beta, target, penalty)
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur, args=(beta, target, penalty),
                          jac=smooth_gradient, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': -1})
            h_cand = project_box_sum(res.x, target)
            sc = compute_c5(h_cand)
            obj_cand = smooth_objective(h_cand, beta, target, penalty)
            if obj_cand < curr_obj:
                h_cur = h_cand
                curr_obj = obj_cand
            else:
                h_proj = project_box_sum(res.x, target)
                if compute_c5(h_proj) < compute_c5(h_cur):
                    h_cur = h_proj
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def golden_ratio_binary(n, target_sum, rng):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    frac_phi = phi - 1.0
    positions = set()
    k = 0
    while len(positions) < target_sum and k < n * 10:
        pos = int((k * frac_phi) % 1.0 * n) % n
        positions.add(pos)
        k += 1
    pos_list = list(positions)
    while len(pos_list) < target_sum:
        candidate = rng.randint(0, n)
        if candidate not in positions:
            pos_list.append(candidate)
            positions.add(candidate)
    for p in pos_list[:target_sum]:
        h[p] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = n // 2

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin0 = project_binary(h_init, target_int)
    c5_bin0 = compute_c5(h_bin0)
    if c5_bin0 < best_c5:
        best_c5 = c5_bin0
        best_h = h_bin0.copy()

    starting_points = [h_bin0.copy()]
    starting_points.append(golden_ratio_binary(n, target_int, rng))
    starting_points.append(np.zeros(n))
    _pos = rng.choice(n, target_int, replace=False)
    starting_points[2][_pos] = 1.0
    h_alt = np.zeros(n)
    _step = 2 + int(0.5)
    for _k in range(target_int):
        h_alt[(_k * _step) % n] = 1.0
    if int(h_alt.sum()) != target_int:
        h_alt = project_binary(h_alt, target_int)
    starting_points.append(h_alt)

    phase1_budget = min(400, tleft() * 0.4)
    per_start = max(30, int(phase1_budget / len(starting_points)))

    for si, sp in enumerate(starting_points):
        if tleft() < 40:
            break
        alloc = min(per_start, tleft() - 30)
        if alloc < 10:
            break
        h_ls, c5_ls = fast_greedy_search(sp, target_int, time_limit=alloc,
                                          start_time=start_time, max_stall=200, batch_size=2000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    if tleft() > 60:
        h_sa, c5_sa = sa_binary_fast(best_h, target_int, max_iters=200000,
                                      t_init=0.006, t_final=0.00001, rng=rng,
                                      time_limit=min(120, tleft()*0.3), start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    if tleft() > 50:
        h_from_bin = project_box_sum(best_h.copy(), target)
        for beta in [20, 50, 100, 200, 400, 800]:
            if tleft() < 20:
                break
            alloc = min(35, tleft() * 0.15)
            h_opt = lbfgs_continuous(h_from_bin, beta, target, penalty=500.0,
                                      maxiter=200, time_limit=alloc, start_time=start_time)
            sc = compute_c5(h_opt)
            h_from_bin = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    if tleft() > 50:
        h_bin_again = project_binary(best_h, target_int)
        alloc = min(100, tleft() * 0.3)
        h_g2, c5_g2 = fast_greedy_search(h_bin_again, target_int, time_limit=alloc,
                                          start_time=start_time, max_stall=400, batch_size=3000)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    if tleft() > 40:
        h_sa2, c5_sa2 = sa_binary_fast(best_h, target_int, max_iters=150000,
                                        t_init=0.003, t_final=0.000005, rng=rng,
                                        time_limit=min(60, tleft()*0.5), start_time=start_time)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    if tleft() > 40:
        for restart in range(4):
            if tleft() < 20:
                break
            pert = rng.normal(0, 0.03, n)
            h_pert = project_box_sum(best_h + pert, target)
            h_bin_pert = project_binary(h_pert, target_int)
            alloc = min(20, tleft() * 0.3)
            h_rp, c5_rp = fast_greedy_search(h_bin_pert, target_int, time_limit=alloc,
                                              start_time=start_time, max_stall=100, batch_size=1500)
            if c5_rp < best_c5:
                best_c5 = c5_rp
                best_h = h_rp.copy()

    if tleft() > 30:
        n_final_use = len(best_h)
        target_final = float(n_final_use / 2)
        h_cont_final = project_box_sum(best_h.copy(), target_final)
        for beta_f in [100, 300, 600, 1200]:
            if tleft() < 10:
                break
            alloc = min(15, tleft() * 0.4)
            h_pol = lbfgs_continuous(h_cont_final, beta_f, target_final,
                                      penalty=1000.0, maxiter=150,
                                      time_limit=alloc, start_time=start_time)
            h_cont_final = h_pol
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    if tleft() > 15:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        alloc = min(20, tleft() - 5)
        h_gf, c5_gf = fast_greedy_search(h_bin_final, target_use, time_limit=alloc,
                                          start_time=start_time, max_stall=50, batch_size=2000)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
