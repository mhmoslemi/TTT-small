            bi = scores.index(best_score)
            best_h = population[bi].copy()
    return best_h, best_score

def _tournament(scores, rng, k=3):
    idx = rng.randint(0, len(scores), size=min(k, len(scores)))
    return idx[np.argmin(scores[idx])]

def _uniform_crossover(p1, p2, rng, n, target):
    mask = rng.rand(n) < 0.5
    child = np.where(mask, p1, p2)
    child = project_binary(child, target)
    return child

def upscale(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    ratio = n_fine / n_coarse
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n_fine)
    interp = np.interp(x_fine, x_coarse, h_coarse)
    return project_binary(interp, target_fine)

def periodic_init(n, target, rng, freq_range=(2, 30)):
    f1 = rng.uniform(freq_range[0], freq_range[1])
    f2 = rng.uniform(freq_range[0], freq_range[1])
    phase1 = rng.uniform(0, 2*np.pi)
    phase2 = rng.uniform(0, 2*np.pi)
    x = np.linspace(0, 2, n)
    signal = np.sin(2*np.pi*f1*x/2 + phase1) + np.sin(2*np.pi*f2*x/2 + phase2)
    threshold = np.median(signal)
    h = (signal >= threshold).astype(float)
    h = project_binary(h, target)
    return h

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def continuous_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = project_box_sum(h0.astype(float), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    positions = np.arange(n)
    k_arr = np.arange(2*n-1) - (n-1)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        lr = 1.0 / (alpha * 0.3 + 1.0)
        momentum = 0.7
        v = np.zeros(n)
        n_iter = max(100, int(30 * 200))
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = corr.max()
            exp_c = np.exp(alpha * (corr - max_c))
            w = exp_c / exp_c.sum()
            grad = np.zeros(n)
            for offset in range(-min(n-1, 100), min(n-1, 100)+1):
                wi = w[offset + n - 1] if (0 <= offset + n - 1 < len(w)) else 0
                if wi < 1e-10:
                    continue
                idx_p = positions + offset
                valid_p = (idx_p >= 0) & (idx_p < n)
                idx_m = positions - offset
                valid_m = (idx_m >= 0) & (idx_m < n)
                term = np.zeros(n)
                if np.any(valid_p):
                    term[valid_p] += (1.0 - h[idx_p[valid_p]])
                if np.any(valid_m):
                    term[valid_m] -= h[idx_m[valid_m]]
                grad += wi * term
            v = momentum * v - lr * grad
            h_new = h + v
            h = project_box_sum(h_new, target)
            if it % 100 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 5.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr_init = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    phase1_time = min(40, tleft() * 0.05)
    h1, s1 = focused_sa(best_h, max(50000, int(phase1_time * 50000)), 0.02, 0.0001,
                         rng, time_limit=phase1_time, start_time=start_time, k_arr=k_arr_init, top_k=5)
    if s1 < best_score:
        best_score = s1
        best_h = h1.copy()

    if tleft() > 20:
        h2, s2 = exhaustive_local_search(best_h, k_arr_init, n_init, max_rounds=80)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    phase2_time = min(200, tleft() * 0.22)
    h3, s3 = sa_advanced(best_h, max(200000, int(phase2_time * 40000)), 0.018, 0.00005,
                          rng, time_limit=phase2_time, start_time=start_time, k_arr=k_arr_init,
                          block_frac=0.12, max_block_len=12, multi_swap_frac=0.1)
    if s3 < best_score:
        best_score = s3
        best_h = h3.copy()

    if tleft() > 30:
        h4, s4 = exhaustive_local_search(best_h, k_arr_init, n_init, max_rounds=100)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()

    # Multi-resolution: try 2x
    n_2x = n_init * 2
    target_2x = n_2x // 2
    k_arr_2x = np.arange(2 * n_2x - 1) - (n_2x - 1)
    if tleft() > 150:
        h_up = upscale(best_h, n_2x, target_2x)
        phase3_time = min(120, tleft() * 0.2)
        h5, s5 = sa_advanced(h_up, max(100000, int(phase3_time * 20000)), 0.015, 0.00003,
                              rng, time_limit=phase3_time, start_time=start_time, k_arr=k_arr_2x,
                              block_frac=0.1, max_block_len=15, multi_swap_frac=0.08)
        if s5 < best_score:
            best_score = s5
            best_h = h5.copy()
        else:
            h5b = upscale(h5, n_init, target_init)
            if compute_c5(h5b) < best_score:
                best_score = compute_c5(h5b)
                best_h = h5b.copy()

    # Multi-resolution: try 3x
    n_3x = n_init * 3
    if n_3x % 2 != 0:
        n_3x += 1
    target_3x = n_3x // 2
    k_arr_3x = np.arange(2 * n_3x - 1) - (n_3x - 1)
    if tleft() > 120:
        h_up3 = upscale(best_h, n_3x, target_3x)
        phase4_time = min(100, tleft() * 0.15)
        h6, s6 = sa_advanced(h_up3, max(50000, int(phase4_time * 15000)), 0.012, 0.00002,
                              rng, time_limit=phase4_time, start_time=start_time, k_arr=k_arr_3x,
                              block_frac=0.08, max_block_len=20, multi_swap_frac=0.05)
        if s6 < best_score:
            best_score = s6
            best_h = h6.copy()
        else:
            h6b = upscale(h6, n_init, target_init)
            if compute_c5(h6b) < best_score:
                best_score = compute_c5(h6b)
                best_h = h6b.copy()

    # Genetic algorithm at original resolution
    if tleft() > 120:
        ga_time = min(150, tleft() * 0.15)
        h7, s7 = ga_memetic(best_h, n_init, target_init, rng,
                            time_limit=ga_time, start_time=start_time, k_arr=k_arr_init,
                            pop_size=15, n_gen=80)
        if s7 < best_score:
            best_score = s7
            best_h = h7.copy()

    # Periodic initialization + optimization
    for pi in range(4):
        if tleft() < 60:
            break
        h_per = periodic_init(n_init, target_init, rng, freq_range=(3, 40))
        pert_time = min(30, tleft() * 0.05)
        h8, s8 = sa_advanced(h_per, max(20000, int(pert_time * 30000)), 0.02, 0.0001,
                              rng, time_limit=pert_time, start_time=start_time, k_arr=k_arr_init,
                              block_frac=0.15, max_block_len=10, multi_swap_frac=0.1)
        if s8 < best_score:
            best_score = s8
            best_h = h8.copy()

    # Focused SA on worst lags from current best
    if tleft() > 60:
        focused_time = min(80, tleft() * 0.1)
        h9, s9 = focused_sa(best_h, max(100000, int(focused_time * 30000)), 0.01, 0.00001,
                             rng, time_limit=focused_time, start_time=start_time, k_arr=k_arr_init, top_k=8)
        if s9 < best_score:
            best_score = s9
            best_h = h9.copy()

    # Multi-start perturbation + SA
    for restart in range(6):
        if tleft() < 50:
            break
        if restart % 2 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(5, 15))
        else:
            h_pert = perturb_diverse(best_h, rng, rng.randint(15, 30))
        restart_time = min(30, tleft() * 0.05)
        h10, s10 = sa_advanced(h_pert, max(30000, int(restart_time * 30000)),
                                0.015, 0.00003, rng,
                                time_limit=restart_time, start_time=start_time, k_arr=k_arr_init,
                                block_frac=0.15, max_block_len=10, multi_swap_frac=0.1)
        if s10 < best_score:
            best_score = s10
            best_h = h10.copy()

    # Continuous refinement
    if tleft() > 80:
        target_use = float(n_init) / 2.0
        cont_time = min(100, tleft() * 0.12)
        h11, s11 = continuous_refine(best_h, target_use, start_time, cont_time,
                                      alphas=[10, 50, 200, 1000, 5000])
        if s11 < best_score:
            best_score = s11
            best_h = h11.copy()

    # Final: project to binary and do one more round of focused optimization
    if tleft() > 30:
        h_bin = project_binary(best_h, target_init)
        final_time = min(40, tleft() * 0.08)
        h12, s12 = focused_sa(h_bin, max(50000, int(final_time * 30000)),
                               0.005, 0.00001, rng,
                               time_limit=final_time, start_time=start_time, k_arr=k_arr_init, top_k=5)
        if s12 < best_score:
            best_score = s12
            best_h = h12.copy()

    # Final exhaustive local search
    if tleft() > 15:
        h13, s13 = exhaustive_local_search(best_h, k_arr_init, n_init, max_rounds=60)
        if s13 < best_score:
            best_score = s13
            best_h = h13.copy()

    # Final continuous polish
    if tleft() > 15:
        target_use = float(n_init) / 2.0
        h14, s14 = continuous_refine(best_h, target_use, start_time, tleft() - 5,
                                      alphas=[2000, 10000, 50000])
        if s14 < best_score:
            best_score = s14
            best_h = h14.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
