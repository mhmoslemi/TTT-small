Actually, let me reconsider the CD approach. The key idea is:
- Find the most "active" shift k* (the one with maximum C_k)
- Compute the gradient of C_k* with respect to h
- Move mass from high-gradient positions to low-gradient positions
- Find the optimal step size by 1D optimization

The step size optimization can be done more efficiently:
- The change in C_k* when moving δ from position j to position i is:
  ΔC_k* = δ · (grad[i] - grad[j]) (linear in δ)
- But the change in OTHER C_k values also matters (we don't want to make them worse)
- The worst-case change is: max_k [C_k + δ · (grad_k[i] - grad_k[j])]
- We want to find δ that minimizes this max

This is a 1D convex optimization (max of linear functions), which can be solved exactly by finding the intersection of the steepest rising line with the current max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-9)[0]
    else:
        idx = np.where(h > 1e-9)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-11:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = alpha * (C - max_C)
    shifted = np.clip(shifted, -500.0, 0.0)
    obj_val = max_C + np.log(np.sum(np.exp(shifted))) / alpha
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return float(obj_val), np.zeros(n)
    weights /= wsum
    mid = n - 1
    sig = weights > 1e-10
    sig_idx = np.where(sig)[0]
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s_idx in sig_idx:
        k = int(s_idx) - mid
        w = weights[s_idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return float(obj_val), grad

def lbfgs_surrogate(h_start, n, target, alphas, maxiter, maxfun):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            return smooth_obj_and_grad(x, alpha)[0]
        def grad(x):
            return smooth_obj_and_grad(x, alpha)[1]
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', jac=grad,
                          bounds=bounds,
                          options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, best_c5

def adam_topk(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,

        weights /= wsum
        sig = weights > 1e-5 / (2 * n)
        sig_idx = np.where(sig)[0]
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            w = weights[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 300:
        h_tmp, sc = lbfgs_surrogate(best_h, n0, target0,
                                     [50, 200, 1000, 5000, 20000],
                                     maxiter=300, maxfun=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_tmp

    if tleft() > 250:
        h_tmp, sc = adam_topk(best_h, n0, target0,
                               time_limit=min(80, tleft() * 0.3),
                               max_steps=50000, n_top_start=3, n_top_end=20,
                               lr=0.012, beta1=0.85, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_tmp

    if tleft() > 200:
        h_tmp, sc = pair_cd(best_h, n0, target0,
                             time_limit=min(80, tleft() * 0.3),
                             max_sweeps=2000, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_tmp

    # Try diverse starts at coarser resolution
    if tleft() > 400:
        n_coarse = 150
        target_coarse = n_coarse / 2.0
        starts = []
        h_interp_c = interpolate_to(best_h, n_coarse, target_coarse)
        starts.append(h_interp_c)
        h_burst = np.zeros(n_coarse)
        h_burst[:n_coarse // 3] = 1.0
        h_burst[2 * n_coarse // 3:] = 1.0
        h_burst = project_feasible(h_burst, target_coarse)
        starts.append(h_burst)
        h_rand = rng.rand(n_coarse) * 0.8 + 0.1
        h_rand = project_feasible(h_rand, target_coarse)
        starts.append(h_rand)
        for h_s in starts:
            if tleft() < 150:
                break
            h_c, sc_c = lbfgs_surrogate(h_s, n_coarse, target_coarse,
                                         [100, 500, 2000, 10000],
                                         maxiter=200, maxfun=2000)
            h_c, sc_c2 = adam_topk(h_c, n_coarse, target_coarse,
                                    time_limit=30, max_steps=20000,
                                    n_top_start=2, n_top_end=10,
                                    lr=0.01, beta1=0.8, beta2=0.999, eps=1e-8)
            sc_c = min(sc_c, sc_c2)
            h_c = h_c if sc_c == sc_c2 else h_c
            if sc_c < best_score:
                h_up = interpolate_to(h_c, n0, target0)
                h_up, sc_up = lbfgs_surrogate(h_up, n0, target0,
                                               [500, 2000, 10000],
                                               maxiter=200, maxfun=2000)
                if sc_up < best_score:
                    best_score = sc_up
                    best_h = h_up

    # Refine at higher resolution
    if tleft() > 350:
        n_hi = 600
        target_hi = n_hi / 2.0
        h_hi = interpolate_to(best_h, n_hi, target_hi)
        h_hi, sc = lbfgs_surrogate(h_hi, n_hi, target_hi,
                                    [100, 500, 2000, 10000, 50000],
                                    maxiter=400, maxfun=4000)
        if sc < best_score:
            best_score = sc
            best_h = h_hi
            n0 = n_hi
            target0 = target_hi
        if tleft() > 200:
            h_hi, sc = adam_topk(best_h, n0, target0,
                                  time_limit=min(80, tleft() * 0.3),
                                  max_steps=40000, n_top_start=3, n_top_end=15,
                                  lr=0.01, beta1=0.85, beta2=0.999, eps=1e-8)
            if sc < best_score:
                best_score = sc
                best_h = h_hi
        if tleft() > 150:
            h_hi, sc = pair_cd(best_h, n0, target0,
                                time_limit=min(60, tleft() * 0.3),
                                max_sweeps=1500, n_cand=15)
            if sc < best_score:
                best_score = sc
                best_h = h_hi

    # Even higher resolution if time permits
    if tleft() > 300:
        n_hi2 = 800
        target_hi2 = n_hi2 / 2.0
        h_hi2 = interpolate_to(best_h, n_hi2, target_hi2)
        h_hi2, sc = lbfgs_surrogate(h_hi2, n_hi2, target_hi2,
                                     [100, 500, 2000, 10000],
                                     maxiter=300, maxfun=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_hi2
            n0 = n_hi2
            target0 = target_hi2
        if tleft() > 150:
            h_hi2, sc = adam_topk(best_h, n0, target0,
                                   time_limit=min(60, tleft() * 0.3),
                                   max_steps=30000, n_top_start=3, n_top_end=12,
                                   lr=0.008, beta1=0.85, beta2=0.999, eps=1e-8)
            if sc < best_score:
                best_score = sc
                best_h = h_hi2

    # Final polish at current resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 100:
        h_fin, sc = lbfgs_surrogate(best_h, n_use, target_use,
                                     [5000, 20000, 100000, 500000],
                                     maxiter=500, maxfun=5000)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    if tleft() > 80:
        h_fin, sc = softmax_subgrad(best_h, n_use, target_use,
                                     time_limit=min(40, tleft() * 0.4),
                                     max_steps=20000, alpha_w=50.0, base_lr=0.015)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    if tleft() > 60:
        h_fin, sc = pair_cd(best_h, n_use, target_use,
                             time_limit=min(40, tleft() * 0.5),
                             max_sweeps=1500, n_cand=15)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Random restarts with quick re-optimization
    if tleft() > 80:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            h_pert, sc = lbfgs_surrogate(h_pert, n_use, target_use,
                                          [1000, 5000, 20000],
                                          maxiter=200, maxfun=2000)
            if sc < best_score:
                best_score = sc
                best_h = h_pert

    if tleft() > 30:
        h_fin, sc = lbfgs_surrogate(best_h, n_use, target_use,
                                     [100000, 500000, 2000000],
                                     maxiter=300, maxfun=3000)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
