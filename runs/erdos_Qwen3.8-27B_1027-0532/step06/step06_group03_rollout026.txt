        v = mu * v - grad
        lr = base_lr / (1.0 + 0.005 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = np.clip(h + lr * lr_mult * v, 0.0, 1.0)
            r_try = np.correlate(reduced_to_full(h_try, n), reduced_to_full(h_try, n), mode='full')
            if r_try.min() > best_min_r - 1e-14:
                h = h_try
                best_min_r = r_try.min()
                best_h = h.copy()
                break
    return best_h


def run(seed=42, budget_s=1300, **kwargs):
