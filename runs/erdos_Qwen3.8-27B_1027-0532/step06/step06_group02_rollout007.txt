import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        ones2 = np.where(h_r == 1)[0]
        extra = final_sum - target_sum
        h_r[ones2[:extra]] = 0
    elif final_sum < target_sum:
        zeros2 = np.where(h_r == 0)[0]
        extra = target_sum - final_sum
        h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_gradient_matrix(h, weights):
    n = len(h)
    m_idx = np.arange(n)
    j_idx = np.arange(n)
    w_index1 = m_idx[:, None] - j_idx[None, :] + (n - 1)
    w_index2 = j_idx[None, :] - m_idx[:, None] + (n - 1)
    w_mat1 = weights[w_index1]
    w_mat2 = weights[w_index2]
    first_sum = (1.0 - h) @ w_mat1
    second_sum = h @ w_mat2
    return first_sum - second_sum

def slsqp_objective(h, alpha):
    corr = full_corr_fast(h)
    max_c = corr.max()
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_objective, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def projected_gradient_descent(h_start, target, alphas, lr_schedule, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        itr_time = min(40.0, time_limit - (time.time() - start_time))
        n_iter = int(itr_time * 500)
        n_iter = max(n_iter, 100)
        lr = lr_schedule(alpha)
        momentum_v = np.zeros(n)
        beta = 0.85
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = corr.max()
            shifted = alpha * (corr - max_c)
            weights = np.exp(shifted)
            weights /= weights.sum()
            grad = compute_gradient_matrix(h, weights)
            momentum_v = beta * momentum_v - lr * grad
            h_new = h + momentum_v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 100 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        if k_star != 0 and rng.random() < 0.4:
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)

            idx = positions + k_star

            if len(i_pool) > 0 and len(j_pool) > 0:

            else:

                j = zeros[rng.randint(len(zeros))]

                stall = 0

            stall += 1
    return best_h, best_max * dx
