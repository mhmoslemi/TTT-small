import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_advanced(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_start=30, n_cand_end=120, n_shifts=3,
                     max_no_improve=500, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]

        shift_weights = shift_weights / sw_sum

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, best_i, h, n, shifts)

def subgradient_descent(h_init, n, target, time_limit, n_steps=20000, step_scale=0.005):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.95

        if step % 500 == 0 and step > 0:
            eta = max(step_scale * max(0.1, 1.0 - step / n_steps), 1e-6)

    return best_h, best_c5

        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=5, max_sweeps=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        sorted_blocks = np.argsort(block_grads)
        low_blocks = sorted_blocks[:5]
        high_blocks = sorted_blocks[-5:]

        best_trial = c5_val
        best_move = None

        for hb in high_blocks:
            for lb in low_blocks:
                if hb == lb:
                    continue
                hi_start = hb * block_size
                hi_end = (hb + 1) * block_size
                lo_start = lb * block_size
                lo_end = (lb + 1) * block_size

                max_move = min(np.sum(h[hi_start:hi_end]),
                               np.sum(1.0 - h[lo_start:lo_end]))
                if max_move < 1e-10:
                    continue

                for frac in [0.25, 0.5, 0.75, 1.0]:
                    delta = frac * max_move / block_size
                    h_try = h.copy()
                    h_try[hi_start:hi_end] -= delta
                    h_try[lo_start:lo_end] += delta
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial - 1e-14:
                        best_trial = sc
                        best_move = (h_try.copy(), sc)

        if best_move is not None:
            h = best_move[0]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_perturbation(h_init, n, target, time_limit, n_restarts=5, base_scale=0.05):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(777)

    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        scale = base_scale * (0.5 + 0.5 * (r / max(n_restarts - 1, 1)))
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [50, 200, 1000, 5000],
                                600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd, c5_cd = pair_cd_advanced(best_h, cur_n, cur_target,
                                       time_limit=min(200, tleft() * 0.5),
                                       max_sweeps=40000,
                                       n_cand_start=30, n_cand_end=120,
                                       n_shifts=3, max_no_improve=600,
                                       anneal_temp=0.02, anneal_decay=0.9995)

            best_h = h_cd

    if tleft() > 100:

                                        n_steps=30000, n_weights=3,
                                        step_scale=0.004)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [2000, 10000],
                                1000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:

        sc1 = compute_c5(h1)
        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd_advanced(h1, n1, target1,

                                             n_cand_start=25, n_cand_end=100,
                                             n_shifts=3, max_no_improve=500,
                                             anneal_temp=0.01, anneal_decay=0.9997)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
                cur_n = n1
                cur_target = target1

    if tleft() > 100:
        h_sg2, c5_sg2 = subgradient_descent(best_h, cur_n, cur_target,
                                            time_limit=min(70, tleft() * 0.5),
                                            n_steps=20000, step_scale=0.003)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    if tleft() > 120:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [5000, 20000],
                                1500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 400:
        n2 = 2000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 300:
            h_cd2, c5_cd2 = pair_cd_advanced(h2, n2, target2,
                                             time_limit=min(250, tleft() * 0.6),

                                             n_cand_start=20, n_cand_end=80,
                                             n_shifts=2, max_no_improve=400,

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
