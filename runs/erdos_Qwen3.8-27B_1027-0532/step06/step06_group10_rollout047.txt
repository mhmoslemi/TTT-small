import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / w_sum
    n_corr = 2 * n - 1
    mid = n - 1
    grad = np.zeros(n)
    for s_idx in range(n_corr):
        w_s = w[s_idx]
        if w_s < 1e-10:
            continue
        k = s_idx - mid
        if k >= 0:
            grad[:n - k] += w_s * (1.0 - h[k:])
            grad[k:] -= w_s * h[:n - k]
        else:
            grad[-k:] += w_s * (1.0 - h[:n + k])
            grad[:-k] -= w_s * h[-k:]
    grad *= dx
    grad -= np.mean(grad)
    return grad, corr

def proj_grad_bb(h0, n, target, time_limit, start_time, alphas=None):
    if alphas is None:
        alphas = [30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.7
    prev_grad = None
    prev_h = None
    no_improve = 0
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        alpha_curr = float(alpha)
        n_steps = 0
        max_steps = 5000
        for step in range(max_steps):
            if time.time() - start_time > time_limit:
                break
            grad, corr = compute_gradient(h, alpha_curr, n)
            c5_now = float(np.max(corr)) * (2.0 / n)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 800:
                break
            if prev_grad is not None and prev_h is not None:
                s_vec = h - prev_h
                y_vec = grad - prev_grad
                sy = float(np.dot(s_vec, y_vec))
                ss = float(np.dot(s_vec, s_vec))
                if sy > 1e-10 and ss > 1e-20:
                    lr = ss / sy
                    lr = max(1e-10, min(lr, 10.0))
                else:
                    lr = 0.01
            else:
                lr = 0.01
            v = momentum * v - lr * grad
            h_try = project_feasible(h + v, target)
            c5_try = compute_c5(h_try)
            if c5_try > c5_now + 1e-12:
                backtrack = 0.5
                for _ in range(10):
                    h_try = project_feasible(h - lr * backtrack * grad, target)
                    c5_try = compute_c5(h_try)
                    if c5_try <= c5_now + 1e-14:
                        break
                    backtrack *= 0.5
                if c5_try > c5_now + 1e-12:
                    v = 0.5 * v
                    momentum = max(0.3, momentum - 0.05)
            else:
                momentum = min(0.9, momentum + 0.01)
            prev_h = h.copy()
            prev_grad = grad.copy()
            h = h_try
        prev_grad = None
        prev_h = None
        v = np.zeros(n)
        momentum = 0.7
    return best_h, best_c5

def smart_cd(h0, n, target, time_limit, start_time, n_cand=15, max_sweeps=20000):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 600:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(24):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def slsqp_round(h0, n, target, time_limit, start_time, alphas=None, maxiter=3000):
    if alphas is None:
        alphas = [50, 200, 800, 3000, 10000, 30000]
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj_fn(x, a=alpha, n_local=n):
            dx = 2.0 / n_local
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -50.0, 0.0)
            return float((c_max + np.log(np.sum(np.exp(shifted))) / a) * dx)
        def grad_fn(x, a=alpha, n_local=n):
            g, _ = compute_gradient(x, a, n_local)
            return g
        def con_fn(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def anneal_restart(h0, n, target, time_limit, start_time, n_restarts=5):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h_pert = h0.copy()
        n_pert = max(5, n // 20)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lp_refine(h0, n, target, time_limit, start_time, n_active=20, max_rounds=50):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            for mix in [0.7, 0.5, 0.3]:
                h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-12:
                    h = h_mix
                    break
            else:
                break
    return best_h, best_c5

            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hr, s_hr = proj_grad_bb(h_interp, n_try, target_try,
                                       time_limit=min(120.0, tleft() * 0.4),
                                       start_time=start_time,
                                       alphas=[30, 100, 500, 2000, 8000, 30000])
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr
            if tleft() < 80:
                break
            h_cd3, c5_cd3 = smart_cd(best_h, len(best_h), float(len(best_h)) / 2.0,
                                      time_limit=min(60.0, tleft() * 0.3),
                                      start_time=start_time, n_cand=15, max_sweeps=10000)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 9: Final SLSQP polish at best resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_round(best_h, n_use, target_use,
                                        time_limit=min(60.0, tleft() * 0.5),
                                        start_time=start_time,
                                        alphas=[1000, 5000, 20000, 100000],
                                        maxiter=2000)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 10: Final CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd4, c5_cd4 = smart_cd(best_h, n_use, target_use,
                                  time_limit=min(30.0, tleft() * 0.6),
                                  start_time=start_time, n_cand=25, max_sweeps=15000)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Phase 11: Final gradient polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_gp, s_gp = proj_grad_bb(best_h, n_use, target_use,
                                   time_limit=min(15.0, tleft() * 0.7),
                                   start_time=start_time,
                                   alphas=[50000, 200000, 500000])
        if s_gp < best_score:
            best_score = s_gp
            best_h = h_gp

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)
    return h_out, final_score, n_final
