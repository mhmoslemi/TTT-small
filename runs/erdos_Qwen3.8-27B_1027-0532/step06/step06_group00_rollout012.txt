def vectorized_pair_cd(h, n, target, corr, change, high_set, low_set, n_bisect=12):
    n_corr = 2 * n - 1
    dx = 2.0 / n
    n_h = len(high_set)
    n_l = len(low_set)
    
    # Build change vectors for all pairs
    # change is (n, n_corr)
    # For pair (i, j): g = change[j] - change[i]
    # g_matrix is (n_h * n_l, n_corr)
    
    i_idx = high_set[np.newaxis, :].repeat(n_l, axis=1)  # (n_l, n_h)
    j_idx = low_set[:, np.newaxis].repeat(n_h, axis=1)  # (n_l, n_h)
    
    # Actually let's do (n_h, n_l)
    i_flat = np.tile(high_set, (n_l, 1)).T.ravel()  # (n_h*n_l,)
    j_flat = np.tile(low_set, n_h).ravel()  # (n_h*n_l,)
    
    # Skip i == j
    valid = i_flat != j_flat
    i_flat = i_flat[valid]
    j_flat = j_flat[valid]
    
    n_pairs = len(i_flat)
    if n_pairs == 0:
        return None, None, None
    
    g = change[j_flat] - change[i_flat]  # (n_pairs, n_corr)
    
    # Delta bounds
    delta_lo = np.maximum(h[i_flat] - 1.0, -h[j_flat])  # (n_pairs,)
    delta_hi = np.minimum(h[i_flat], 1.0 - h[j_flat])  # (n_pairs,)
    
    # Bisection for each pair
    # At each step: vals = corr + mid * g, f = max(vals), subgrad = max(g[active])
    for _ in range(n_bisect):
        mid = 0.5 * (delta_lo + delta_hi)
        vals = corr[np.newaxis, :] + mid[:, np.newaxis] * g  # (n_pairs, n_corr)
        f = np.max(vals, axis=1)  # (n_pairs,)
        active = vals >= (f[:, np.newaxis] - 1e-14)
        # subgrad = max of g over active set for each pair
        g_active = np.where(active, g, -np.inf)
        subgrad = np.max(g_active, axis=1)
        too_high = subgrad > 1e-12
        delta_hi[too_high] = mid[too_high]
        delta_lo[~too_high] = mid[~too_high]
    
    delta_star = 0.5 * (delta_lo + delta_hi)
    
    # Evaluate
    vals_final = corr[np.newaxis, :] + delta_star[:, np.newaxis] * g
    new_max = np.max(vals_final, axis=1) * dx
    
    best_idx = np.argmin(new_max)
    
    if new_max[best_idx] < np.max(corr) * dx - 1e-13:
        best_i = i_flat[best_idx]
        best_j = j_flat[best_idx]
        best_delta = delta_star[best_idx]
        return best_i, best_j, best_delta
    else:
        return None, None, None

import time

_g_target = 0.0

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vec_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=30, n_cand_end=80, max_no_improve=200,
                n_bisect=10):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        # Compute gradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        # Candidate selection
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        # Vectorized pair search
        n_h = len(high_set)
        n_l = len(low_set)
        
        i_flat = np.tile(high_set, (n_l, 1)).T.ravel()
        j_flat = np.tile(low_set, n_h).ravel()
        
        valid = i_flat != j_flat
        if not valid.any():
            consecutive_no_improve += 10
            continue
        i_flat = i_flat[valid]
        j_flat = j_flat[valid]
        n_pairs = len(i_flat)
        
        g = change[j_flat] - change[i_flat]
        
        delta_lo = np.maximum(h[i_flat] - 1.0, -h[j_flat])
        delta_hi = np.minimum(h[i_flat], 1.0 - h[j_flat])
        
        valid_range = delta_hi - delta_lo > 1e-12
        if not valid_range.any():
            consecutive_no_improve += 10
            continue
        
        delta_lo = delta_lo[valid_range]
        delta_hi = delta_hi[valid_range]
        g = g[valid_range]
        i_flat = i_flat[valid_range]
        j_flat = j_flat[valid_range]
        n_pairs = len(i_flat)
        
        # Bisection
        for _ in range(n_bisect):
            mid_d = 0.5 * (delta_lo + delta_hi)
            vals = corr[np.newaxis, :] + mid_d[:, np.newaxis] * g
            f_mid = np.max(vals, axis=1)
            active = vals >= (f_mid[:, np.newaxis] - 1e-14)
            g_active = np.where(active, g, -np.inf)
            subgrad = np.max(g_active, axis=1)
            too_high = subgrad > 1e-12
            delta_hi[too_high] = mid_d[too_high]
            delta_lo[~too_high] = mid_d[~too_high]
        
        delta_star = 0.5 * (delta_lo + delta_hi)
        
        vals_final = corr[np.newaxis, :] + delta_star[:, np.newaxis] * g
        new_max_vals = np.max(vals_final, axis=1)
        
        best_pair_idx = np.argmin(new_max_vals)
        new_best = new_max_vals[best_pair_idx] * dx
        
        if new_best < c5_val - 1e-13:
            best_i = int(i_flat[best_pair_idx])
            best_j = int(j_flat[best_pair_idx])
            best_delta = delta_star[best_pair_idx]
            
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update change matrix rows
            change[best_i, :] = 0.0

            neg_mask = (best_i - shifts >= 0) & (best_i - shifts < n)

        else:

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vec_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=30, n_cand_end=80, max_no_improve=300,
                n_bisect=10, top_m_frac=0.3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweep_count = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_m = max(100, int(n_corr * top_m_frac))

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        sweep_count += 1
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        n_h = len(high_set)
        n_l = len(low_set)

        i_flat = np.tile(high_set, (n_l, 1)).T.ravel()
        j_flat = np.tile(low_set, n_h).ravel()

        valid = i_flat != j_flat
        if not valid.any():
            consecutive_no_improve += 5
            continue
        i_flat = i_flat[valid]
        j_flat = j_flat[valid]

        delta_lo = np.maximum(h[i_flat] - 1.0, -h[j_flat])
        delta_hi = np.minimum(h[i_flat], 1.0 - h[j_flat])
        valid_range = delta_hi - delta_lo > 1e-12
        if not valid_range.any():
            consecutive_no_improve += 5
            continue

        delta_lo = delta_lo[valid_range]
        delta_hi = delta_hi[valid_range]
        i_flat = i_flat[valid_range]
        j_flat = j_flat[valid_range]

        top_idx = np.argsort(corr)[-top_m:]
        g = change[j_flat[:, np.newaxis], top_idx[np.newaxis, :]] - \
            change[i_flat[:, np.newaxis], top_idx[np.newaxis, :]]

        corr_sub = corr[top_idx]

        for _ in range(n_bisect):
            mid_d = 0.5 * (delta_lo + delta_hi)
            vals = corr_sub[np.newaxis, :] + mid_d[:, np.newaxis] * g
            f_mid = np.max(vals, axis=1)
            active = vals >= (f_mid[:, np.newaxis] - 1e-14)
            g_act = np.where(active, g, -np.inf)
            subgrad = np.max(g_act, axis=1)
            too_high = subgrad > 1e-12
            delta_hi[too_high] = mid_d[too_high]
            delta_lo[~too_high] = mid_d[~too_high]

        delta_star = 0.5 * (delta_lo + delta_hi)
        vals_final = corr_sub[np.newaxis, :] + delta_star[:, np.newaxis] * g
        new_max_sub = np.max(vals_final, axis=1)

        best_pair_idx = int(np.argmin(new_max_sub))
        new_c5_approx = new_max_sub[best_pair_idx] * dx

        if new_c5_approx < c5_val - 1e-13:
            best_i = int(i_flat[best_pair_idx])
            best_j = int(j_flat[best_pair_idx])
            best_delta = float(delta_star[best_pair_idx])

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, max_steps=30000, base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 80:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

            h_ref = slsqp_optimize(h_pert, n0, target0,
                                   [100, 500, 2000, 8000], 500, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 80:
        h_cd3, c5_cd3 = vec_pair_cd(best_h, n0, target0,
                                    time_limit=min(60, tleft() * 0.6),
                                    max_sweeps=25000,
                                    n_cand_start=40, n_cand_end=80,
                                    max_no_improve=150, n_bisect=10,
                                    top_m_frac=0.25)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 40:
        h_final = slsqp_optimize(best_h, n0, target0,
                                 [10000, 50000], 600, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
