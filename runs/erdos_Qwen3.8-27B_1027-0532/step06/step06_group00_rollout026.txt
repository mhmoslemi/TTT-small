from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha=1000, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                 n_shifts=3):

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    best_h = h.copy()
    consecutive_no_improve = 0

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)

The gradient:

= (sum_k exp(alpha * c_k) * ∂c_k/∂h[i]) / (sum_j exp(alpha * c_j))
= sum_k softmax_k * ∂c_k/∂h[i]

where softmax_k = exp(alpha * c_k) / sum_j exp(alpha * c_j) = exp(alpha * (c_k - c_max)) / sum_j exp(alpha * (c_j - c_max))

- If i = p: (1 - h[i+k]) [for valid i+k]
- If i = p+k, i.e., p = i-k: -h[i-k] [for valid i-k]

So: ∂c_k/∂h[i] = (1 - h[i+k]) * [0 ≤ i+k < n] - h[i-k] * [0 ≤ i-k < n]

This is the same as the "change" matrix! Specifically:

def slsqp_with_grad(h_start, n, target, alphas, maxiter=500, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_make_obj(alpha), h_cur, jac=_make_grad(alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for k in range(n_corr):
        shift_val = shifts[k]
        p1 = (idxs + shift_val >= 0) & (idxs + shift_val < n)
        p2 = (idxs - shift_val >= 0) & (idxs - shift_val < n)
        change[p1, k] += (1.0 - h[idxs[p1] + shift_val])
        change[p2, k] -= h[idxs[p2] - shift_val]
    return change


def update_change_row(change, p, h, shifts, n):
    change[p, :] = 0.0
    p1 = (p + shifts >= 0) & (p + shifts < n)
    p2 = (p - shifts >= 0) & (p - shifts < n)
    change[p, p1] += (1.0 - h[p + shifts[p1]])
    change[p, p2] -= h[p - shifts[p2]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=30, n_cand_end=120, max_no_improve=500,
                 n_shifts=5, kick_interval=300, kick_scale=0.01):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(777)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if kick_interval > 0 and consecutive_no_improve > kick_interval:
            kick_idx = rng.choice(n, size=max(3, n // 50), replace=False)
            h[kick_idx] += rng.randn(len(kick_idx)) * kick_scale
            h = project_feasible(h, target)
            for ki in kick_idx:
                update_change_row(change, int(ki), h, shifts, n)
            consecutive_no_improve = 0

        top_idx = np.argsort(corr)[-n_shifts:]
        top_vals = corr[top_idx]
        w_sum = np.sum(top_vals)
        if w_sum < 1e-30:
            weights = np.ones(n_shifts) / n_shifts
        else:
            weights = top_vals / w_sum

        grad = np.zeros(n)
        for si in range(len(top_idx)):
            s_idx = int(top_idx[si])
            k = s_idx - mid
            w = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            update_change_row(change, best_i, h, shifts, n)
            update_change_row(change, best_j, h, shifts, n)

    return best_h, best_c5


def projected_gradient_step(h, n, target, alpha, lr):
    val, grad = _obj_grad_alpha(h, alpha)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new, val


def projected_grad_descent(h_init, n, target, time_limit, alpha=5000,
                           base_lr=0.01, max_iters=50000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        lr = base_lr * max(0.1, 1.0 - it / max_iters)
        _, grad = _obj_grad_alpha(h, alpha)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < compute_c5(h) - 1e-14:
            h = h_try
            no_improve = 0
            if sc < best_score - 1e-13:
                best_score = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 500:
                base_lr *= 0.5
                no_improve = 0

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol = slsqp_with_grad(best_h, n0, target0,
                                [200, 1000, 5000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    target_n = 1000
    target_sum = target_n / 2.0
    h_work = interpolate_to(best_h, target_n, target_sum)
    sc_work = compute_c5(h_work)
    if sc_work < best_score:
        best_score = sc_work
        best_h = h_work

    if tleft() > 400:
        h_cd, c5_cd = fast_pair_cd(h_work, target_n, target_sum,
                                   time_limit=min(700, tleft() * 0.55),
                                   max_sweeps=50000,
                                   n_cand_start=30, n_cand_end=120,
                                   max_no_improve=500, n_shifts=5,
                                   kick_interval=300, kick_scale=0.008)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            sc_work = c5_cd

    if tleft() > 120:
        h_pol2 = slsqp_with_grad(best_h, target_n, target_sum,
                                 [2000, 10000, 50000], 800, 1e-15)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 250:
        h_cd2, c5_cd2 = fast_pair_cd(best_h, target_n, target_sum,
                                     time_limit=min(350, tleft() * 0.55),
                                     max_sweeps=40000,
                                     n_cand_start=50, n_cand_end=150,
                                     max_no_improve=600, n_shifts=6,
                                     kick_interval=400, kick_scale=0.005)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_pol3 = slsqp_with_grad(best_h, target_n, target_sum,
                                 [10000, 50000, 200000], 1000, 1e-15)
        sc3 = compute_c5(h_pol3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 150:
        h_cd3, c5_cd3 = fast_pair_cd(best_h, target_n, target_sum,
                                     time_limit=min(200, tleft() * 0.6),
                                     max_sweeps=30000,
                                     n_cand_start=60, n_cand_end=160,
                                     max_no_improve=400, n_shifts=5,
                                     kick_interval=250, kick_scale=0.004)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_pg, c5_pg = projected_grad_descent(best_h, target_n, target_sum,
                                              time_limit=min(60, tleft() * 0.5),
                                              alpha=10000, base_lr=0.005)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    if tleft() > 60:
        h_cd4, c5_cd4 = fast_pair_cd(best_h, target_n, target_sum,
                                     time_limit=min(50, tleft() * 0.6),
                                     max_sweeps=20000,
                                     n_cand_start=80, n_cand_end=150,
                                     max_no_improve=200, n_shifts=4,
                                     kick_interval=0, kick_scale=0.0)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    if tleft() > 40:
        h_final = slsqp_with_grad(best_h, target_n, target_sum,

            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
