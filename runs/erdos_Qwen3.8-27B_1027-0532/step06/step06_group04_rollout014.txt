def solve_1d_max_line(slopes, intercepts):
    C = slopes
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo + hi)
        vals = C * mid + intercepts
        mx = np.max(vals)
        tol = 1e-13 * max(1.0, abs(mx))
        active = np.abs(vals - mx) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5*(lo + hi)

def coord_descent_polish(h_init, n, target, time_limit, n_active_top=200):

    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Use all constraints (or top n_active_top)
        active_k = np.argsort(corr)[-n_active_top:]
        improved = False
        for i in range(n):
            slopes = np.empty(len(active_k))
            intercepts = np.empty(len(active_k))
            for j, k_idx in enumerate(active_k):
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[j] = A
                intercepts[j] = corr[k_idx] - A * h[i]
            new_val = solve_1d_max_line(slopes, intercepts)
            if abs(new_val - h[i]) > 1e-12:
                h[i] = new_val
                improved = True
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def subgradient_step(h, n, target, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr)) - (n-1)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
    v2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad *= dx
    h_new = h - lr * grad
    return project_feasible(h_new, target)

def subgradient_polish(h_init, n, target, time_limit, max_steps=100000):

                stall = 0

import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_A_dense(g, n):
    dx = 2.0 / n
    A = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        if k >= 0:
            hi = n - k
            if hi > 0:
                A[idx, :hi] = g[k:k + hi] * dx
        else:
            lo = -k
            A[idx, lo:] = g[:n - lo] * dx
    return A


def solve_lp_scipy(h, n, target):
    g = 1.0 - h
    A = build_A_dense(g, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status != 0:
        return None
    return res.x[:n].copy()


def solve_lp_cvxpy(h, n, target, eps):
    g = 1.0 - h
    A = build_A_dense(g, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    ones_vec = np.ones(2 * n - 1)
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
        A @ h_var <= t_var * ones_vec
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=500000, verbose=False)
    except Exception:
        return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).flatten(), 0.0, 1.0)

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()

def solve_1d_max_line(slopes, intercepts):

    lo, hi = 0.0, 1.0

        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        mx = np.max(vals)

        else:
            break
    return 0.5 * (lo + hi)


def coord_descent_full(h_init, n, target, time_limit, use_all_k=True):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    ks = np.arange(-(n - 1), n)
    for outer in range(2000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for i in range(n):
            i_plus = i + ks
            i_minus = i - ks
            mask_plus = (i_plus >= 0) & (i_plus < n)
            mask_minus = (i_minus >= 0) & (i_minus < n)
            slopes = np.zeros(2 * n - 1)
            slopes[mask_plus] += (1.0 - h[i_plus[mask_plus]])
            slopes[mask_minus] -= h[i_minus[mask_minus]]
            slopes *= dx
            intercepts = corr - slopes * h[i]
            new_val = solve_1d_max_line(slopes, intercepts)
            if abs(new_val - h[i]) > 1e-13:
                h[i] = new_val
                improved = True
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        elif not improved:
            break
    return best_h, best_c5


def subgradient_polish(h_init, n, target, time_limit, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    stall = 0
    lrs = [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        v2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        improved = False
        for lr in lrs:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_try.copy()
                h = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 300:
                break
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Full LP with scipy linprog
    if tleft() > 400:
        lp_time = min(500, tleft() * 0.45)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time, use_cvxpy=False)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Full coordinate descent
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.2)
        h_cd, c5_cd = coord_descent_full(best_h, n, target, cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP with cvxpy for higher precision
    if tleft() > 300:
        lp_time = min(400, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time, use_cvxpy=True, eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: Coordinate descent again
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.15)
        h_cd, c5_cd = coord_descent_full(best_h, n, target, cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Subgradient polish
    if tleft() > 80:
        sg_time = min(100, tleft() * 0.12)
        h_sg, c5_sg = subgradient_polish(best_h, n, target, sg_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: One more full LP
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time, use_cvxpy=False)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Perturbed restarts
    for noise in [0.005, 0.01, 0.02, 0.04]:
        if tleft() < 200:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_lp, c5_lp = alternating_lp(h_pert, n, target, min(120, tleft() * 0.25), use_cvxpy=False)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
        h_cd, c5_cd = coord_descent_full(h_lp, n, target, min(60, tleft() * 0.15))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Final coordinate descent (long)
    if tleft() > 60:
        cd_time = min(80, tleft() * 0.5)
        h_cd, c5_cd = coord_descent_full(best_h, n, target, cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final subgradient micro-refinement
    if tleft() > 30:
        sg_time = min(25, tleft() * 0.6)
        h_sg, c5_sg = subgradient_polish(best_h, n, target, sg_time)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: Try n=500 if time allows
    if tleft() > 400:
        n500 = 500
        target500 = n500 / 2.0
        h500 = interpolate_to_n(best_h, n500, target500)
        sc500 = compute_c5(h500)
        h500_lp, c5500 = alternating_lp(h500, n500, target500,
                                        min(200, tleft() * 0.3), use_cvxpy=False)
        h500_cd, c5500cd = coord_descent_full(h500_lp, n500, target500,
                                               min(80, tleft() * 0.15))
        if c5500cd < best_score:
            best_score = c5500cd
            best_h = h500_cd
        elif c5500 < best_score:
            best_score = c5500
            best_h = h500_lp
        elif sc500 < best_score:
            best_score = sc500
            best_h = h500

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score
    return best_h, final_score, n_use
