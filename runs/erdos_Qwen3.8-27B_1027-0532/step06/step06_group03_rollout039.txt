def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_symmetric(h, n):
    """Project h onto the symmetry subspace h[i] + h[n-1-i] = 1"""
    half = n // 2
    v = np.zeros(half)
    for i in range(half):
        v[i] = (h[i] + 1.0 - h[n-1-i]) / 2.0
    v = np.clip(v, 0.0, 1.0)
    # Reconstruct full h
    h_sym = np.zeros(n)
    h_sym[:half] = v
    h_sym[half:] = 1.0 - v[::-1]
    # Handle the middle if n is odd (but we assume n is even)
    return h_sym, v

def reconstruct_from_v(v, n):
    """Reconstruct full h from first half v"""
    half = n // 2
    h = np.zeros(n)
    h[:half] = v
    h[half:] = 1.0 - v[::-1]
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_max_grad(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    mid = n - 1

    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:

            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def reconstruct_from_v(v):
    n = 2 * len(v)
    h = np.empty(n)
    h[:len(v)] = v
    h[len(v):] = 1.0 - v[::-1]
    return h

def project_to_symmetric(h):
    n = len(h)
    half = n // 2
    v = (h[:half] + 1.0 - h[n-1-half::-1]) / 2.0
    # Actually: h[n-1-i] for i in range(half) is h[n-1], h[n-2], ..., h[n-half]
    # So v[i] = (h[i] + 1 - h[n-1-i]) / 2
    v = (h[:half] + 1.0 - h[n-1:n-half-1:-1]) / 2.0
    v = np.clip(v, 0.0, 1.0)
    return v

def topk_grad_full(h, alpha, top_k):
    """Compute gradient of smooth max using only top-k shifts"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    # Compute softmax weights on top-k
    top_corr = corr[top_idx]
    shifted = alpha * (top_corr - top_corr.max())
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w /= w.sum()
    
    mid = n - 1
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        s = int(idx) - mid  # actual shift
        # ∂c[s]/∂h[j] = (1-h[j+s]) for valid j+s, minus h[j-s] for valid j-s
        if s >= 0:
            # j+s valid: j from 0 to n-1-s
            grad[:n-s] += wi * (1.0 - h[s:])
            # j-s valid: j from s to n-1
            grad[s:] -= wi * h[:n-s]
        else:
            s_abs = -s
            # j+s = j-s_abs valid: j from s_abs to n-1
            grad[s_abs:] += wi * (1.0 - h[:n-s_abs])
            # j-s = j+s_abs valid: j from 0 to n-1-s_abs
            grad[:n-s_abs] -= wi * h[s_abs:]
    grad *= dx
    return grad

def topk_grad_reduced(h, v, alpha, top_k):
    """Compute gradient of smooth max w.r.t. reduced variables v"""
    grad_full = topk_grad_full(h, alpha, top_k)
    n = len(h)
    half = n // 2
    grad_v = grad_full[:half] - grad_full[n-1:n-half-1:-1]
    return grad_v

    return np.sum(h) - _g_target


def _con_jac_full(h):
    return np.ones(_g_n)


def slsqp_full(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun_full, 'jac': _con_jac_full}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj_full, h_cur, args=(alpha,), jac=_smooth_grad_full,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible_full(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def reduced_obj_grad(v, alpha, top_k):
    h = reconstruct_from_v(v)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    mid = n - 1

    if top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(len(corr))

    top_corr = corr[top_idx]
    tc_max = top_corr.max()
    shifted = np.clip(alpha * (top_corr - tc_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum

    obj = float((tc_max + np.log(w_sum)) * dx)

    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        s = int(idx) - mid
        if s >= 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n - s]
        else:
            sa = -s
            grad[sa:] += wi * (1.0 - h[:n - sa])
            grad[:n - sa] -= wi * h[sa:]
    grad *= dx

    half = n // 2
    grad_v = grad[:half] - grad[n-1:n-half-1:-1]
    return obj, grad_v


def slsqp_reduced(v_init, alpha_list, top_k, maxiter, ftol, time_limit):
    v_cur = v_init.copy()
    start_t = time.time()
    for alpha in alpha_list:
        if time.time() - start_t > time_limit:
            break

        def _fun(v):
            obj, _ = reduced_obj_grad(v, alpha, top_k)
            return obj

        def _jac(v):
            _, g = reduced_obj_grad(v, alpha, top_k)
            return g

        bounds = [(0.0, 1.0)] * len(v_cur)
        try:
            res = minimize(_fun, v_cur, jac=_jac, method='SLSQP',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
            v_new = np.clip(res.x, 0.0, 1.0)
            h_new = reconstruct_from_v(v_new)
            h_cur = reconstruct_from_v(v_cur)
            if compute_c5(h_new) < compute_c5(h_cur):
                v_cur = v_new
        except Exception:
            pass
    return v_cur


def projected_grad_reduced(v_init, alpha, top_k, max_steps, lr_init, mu, time_limit):
    v = v_init.copy()
    v = np.clip(v, 0, 1)
    best_v = v.copy()
    best_c5 = compute_c5(reconstruct_from_v(best_v))
    m = np.zeros_like(v)
    lr = lr_init
    start_t = time.time()

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        obj, grad = reduced_obj_grad(v, alpha, top_k)
        m = mu * m + grad
        v_new = v - lr * m
        v_new = np.clip(v_new, 0.0, 1.0)
        h_new = reconstruct_from_v(v_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_v = v_new.copy()
        v = v_new
        lr = max(lr * 0.9995, 1e-7)
    return best_v, best_c5


def fast_pair_cd_reduced(v_init, alpha, top_k, max_sweeps, n_cand, time_limit):
    v = v_init.copy()
    best_v = v.copy()
    best_c5 = compute_c5(reconstruct_from_v(best_v))
    start_t = time.time()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        h = reconstruct_from_v(v)
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        mid = n - 1

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_v = v.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break

        top_idx = np.argsort(corr)[-top_k:]
        top_corr = corr[top_idx]
        tc_max = top_corr.max()
        shifted = np.clip(50.0 * (top_corr - tc_max), -500.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()

        grad_full = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            s = int(idx) - mid
            if s >= 0:
                grad_full[:n - s] += wi * (1.0 - h[s:])
                grad_full[s:] -= wi * h[:n - s]
            else:
                sa = -s
                grad_full[sa:] += wi * (1.0 - h[:n - sa])
                grad_full[:n - sa] -= wi * h[sa:]

        half = n // 2
        grad_v = grad_full[:half] - grad_full[n-1:n-half-1:-1]

        sorted_idx = np.argsort(grad_v)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_max = min(v[i], 1.0 - v[j])
                if delta_max < 1e-12:
                    continue
                for frac in [0.15, 0.4, 0.65, 0.9]:
                    d = frac * delta_max
                    v_try = v.copy()
                    v_try[i] -= d
                    v_try[j] += d
                    h_try = reconstruct_from_v(v_try)
                    corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                    c_try = float(np.max(corr_try)) * dx
                    if c_try < best_trial - 1e-14:
                        best_trial = c_try
                        best_i, best_j, best_delta = i, j, d

        if best_i >= 0:
            v[best_i] -= best_delta
            v[best_j] += best_delta
            v = np.clip(v, 0.0, 1.0)

    return best_v, best_c5


def fast_pair_cd_full(h_init, top_k, max_sweeps, n_cand, time_limit):
    h = h_init.copy()
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_t = time.time()
    no_improve = 0
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break

        top_idx = np.argsort(corr)[-top_k:]
        top_corr = corr[top_idx]
        tc_max = top_corr.max()
        shifted = np.clip(80.0 * (top_corr - tc_max), -500.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            s = int(idx) - mid
            if s >= 0:
                grad[:n - s] += wi * (1.0 - h[s:])
                grad[s:] -= wi * h[:n - s]
            else:
                sa = -s
                grad[sa:] += wi * (1.0 - h[:n - sa])
                grad[:n - sa] -= wi * h[sa:]

        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                for frac in [0.2, 0.5, 0.8]:
                    d = frac * delta_max
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = project_feasible_full(h_try, target)
                    corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                    c_try = float(np.max(corr_try)) * dx
                    if c_try < best_trial - 1e-14:
                        best_trial = c_try
                        best_i, best_j, best_delta = i, j, d
                        best_h_trial = h_try.copy()

        if best_i >= 0:
            h = best_h_trial.copy()

    return best_h, best_c5


def local_perturbation(h_init, n, target, max_tries, time_limit, seed_val):
    start_t = time.time()
    rng = np.random.RandomState(seed_val)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 13]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.15]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible_full(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible_full(h_init, target)
    best_score = compute_c5(best_h)

    # Project to symmetric subspace
    v_sym = project_to_symmetric(best_h)
    h_sym = reconstruct_from_v(v_sym)
    sym_score = compute_c5(h_sym)
    if sym_score < best_score:
        best_score = sym_score
        best_h = h_sym

    # Phase 1: SLSQP in reduced (symmetric) space
    if tleft() > 200:
        v_sol = slsqp_reduced(v_sym, [30, 100, 500, 2000], top_k=50,
                              maxiter=2000, ftol=1e-15, time_limit=min(120, tleft()))
        h_sol = reconstruct_from_v(v_sol)
        sc = compute_c5(h_sol)
        if sc < best_score:
            best_score = sc
            best_h = h_sol
            v_sym = v_sol

    # Phase 2: Projected gradient in reduced space
    if tleft() > 150:
        for alpha_pg in [200, 800, 3000]:
            if tleft() < 50:
                break
            v_pg, c5_pg = projected_grad_reduced(v_sym, alpha_pg, top_k=30,
                                                  max_steps=30000, lr_init=0.03,
                                                  mu=0.9, time_limit=min(50, tleft() * 0.3))
            if c5_pg < best_score:
                best_score = c5_pg
                v_sym = v_pg
                best_h = reconstruct_from_v(v_sym)

    # Phase 3: Pair CD in reduced space
    if tleft() > 120:
        cd_time = min(180, tleft() * 0.3)
        v_cd, c5_cd = fast_pair_cd_reduced(v_sym, 50, top_k=5,
                                            max_sweeps=8000, n_cand=45, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            v_sym = v_cd
            best_h = reconstruct_from_v(v_sym)

    # Phase 4: SLSQP in full space (no symmetry constraint)
    if tleft() > 150:
        h_sol = slsqp_full(best_h, n, target, [200, 1000, 5000],
                           maxiter_per_alpha=1500, ftol=1e-15,
                           time_limit=min(100, tleft()))
        sc = compute_c5(h_sol)
        if sc < best_score:
            best_score = sc
            best_h = h_sol

    # Phase 5: Higher alpha SLSQP in reduced space
    if tleft() > 100:
        v_sol = slsqp_reduced(project_to_symmetric(best_h), [5000, 20000, 50000],
                              top_k=100, maxiter=2000, ftol=1e-16,
                              time_limit=min(80, tleft()))
        h_sol = reconstruct_from_v(v_sol)
        sc = compute_c5(h_sol)
        if sc < best_score:
            best_score = sc
            best_h = h_sol
            v_sym = v_sol

    # Phase 6: Pair CD in full space
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd_full(best_h, top_k=4, max_sweeps=5000,
                                         n_cand=50, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Local perturbation + re-optimize
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible_full(h_pert, target)
            h_ref = slsqp_full(h_pert, n, target, [100, 1000, 5000],
                               maxiter_per_alpha=800, ftol=1e-15,
                               time_limit=min(35, tleft()))
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: More pair CD (reduced)
    if tleft() > 80:
        v_cur = project_to_symmetric(best_h)
        cd_time = min(100, tleft() * 0.3)
        v_cd, c5_cd = fast_pair_cd_reduced(v_cur, 100, top_k=7,
                                            max_sweeps=6000, n_cand=60, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = reconstruct_from_v(v_cd)

    # Phase 9: Local perturbation search
    if tleft() > 60:
        h_pert, c5_pert = local_perturbation(best_h, n, target,
                                              max_tries=5000,
                                              time_limit=min(30, tleft() * 0.3),
                                              seed_val=12345)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        h_final = slsqp_full(best_h, n, target, [10000, 30000],
                             maxiter_per_alpha=1000, ftol=1e-16,
                             time_limit=min(30, tleft()))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 11: Final pair CD
    if tleft() > 30:
        h_cd, c5_cd = fast_pair_cd_full(best_h, top_k=3, max_sweeps=3000,
                                         n_cand=55, time_limit=min(25, tleft()))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible_full(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
