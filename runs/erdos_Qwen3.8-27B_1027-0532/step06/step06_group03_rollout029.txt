```python
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:

    else:
```

```python
def top_k_smooth_grad(h, alpha, k=10):
    """Compute smooth gradient using only top-k correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    # Get top-k indices
    top_idx = np.argsort(corr)[-k:]
    weights = np.exp(alpha * (corr[top_idx] - c_max))
    weights = weights / weights.sum()
    
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    
    grad = np.zeros(n)
    for wi, m in zip(weights, top_idx):
        s = int(m) - mid  # the shift
        # d(corr[m])/dh[j] = (1-h[j+s]) - h[j-s] (with boundary handling)
        if s == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif s > 0:
            # (1-h[j+s]) for j < n-s
            grad[:n-s] += wi * (1.0 - h[s:])
            # -h[j-s] for j >= s
            grad[s:] -= wi * h[:n-s]
        else:
            s_neg = -s
            # (1-h[j+s]) = (1-h[j-s_neg]) for j >= s_neg
            grad[s_neg:] += wi * (1.0 - h[:n-s_neg])
            # -h[j-s] = -h[j+s_neg] for j < n-s_neg
            grad[:n-s_neg] -= wi * h[s_neg:]
    
    grad *= dx
    return grad
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad_topk(h, alpha, k):
    """Compute smooth gradient using only top-k correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    # Get top-k indices
    if k >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(corr)[-k:]
    
    weights = np.exp(alpha * (corr[top_idx] - c_max))
    weights = weights / weights.sum()
    
    mid = n - 1
    grad = np.zeros(n)
    
    for wi, m_val in zip(weights, top_idx):
        s = int(m_val) - mid
        if s == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        else:
            sp = -s
            grad[sp:] += wi * (1.0 - h[:n-sp])
            grad[:n-sp] -= wi * h[sp:]
    
    grad *= dx
    return grad

def fista_smooth(h_start, n, target, alpha_schedule, max_iter_per_alpha, time_limit):
    """FISTA with smooth surrogate."""
    start_t = time.time()
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    k = min(20, 2 * n - 1)  # top-k for gradient
    
    for alpha in alpha_schedule:
        if time.time() - start_t > time_limit:
            break
        
        h = best_h.copy()
        y = best_h.copy()
        t = 1.0
        # Estimate step size
        grad0 = smooth_grad_topk(y, alpha, k)
        g_norm = np.linalg.norm(grad0)
        if g_norm < 1e-15:
            continue
        # Lipschitz estimate: use alpha * n * dx as upper bound (very rough)
        L = alpha * n / max(g_norm * n, 1.0)
        lr = 1.0 / max(L, 1.0)
        lr = min(lr, 0.1)  # cap step size
        
        for it in range(max_iter_per_alpha):
            if time.time() - start_t > time_limit:
                break
            
            grad = smooth_grad_topk(y, alpha, k)
            
            # Backtracking line search
            z = y - lr * grad
            h_new = project_feasible(z, target)
            c5_new = compute_c5(h_new)
            
            # Try to increase step if too conservative
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Nesterov momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h_new + ((t - 1.0) / t_new) * (h_new - h)
            y = project_feasible(y, target)
            h = h_new
            t = t_new
        
        # Reset momentum at the end of each alpha
        h = best_h.copy()
    
    return best_h, best_c5

def subgradient_avg(h_start, n, target, time_limit, max_iter, top_k, avg_window):
    """Subgradient method with PEGASUS step size and averaging."""
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    lr = 0.01
    avg_buf = []
    
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        
        # Compute subgradient (top-k weighted)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]

        w = 1.0 / len(top_idx)

        grad = np.zeros(n)

            elif s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n-s]
            else:
                sp = -s
                grad[sp:] += w * (1.0 - h[:n-sp])
                grad[:n-sp] -= w * h[sp:]
        
        grad *= dx
        # Center the gradient (since we have a sum constraint)
        grad -= np.mean(grad)
        
        # Step
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)

        if c5_try < compute_c5(h) - 1e-14:
            lr = min(lr * 1.5, 0.5)
            h = h_try
        else:
            lr = lr * 0.5
            h_try = project_feasible(h - lr * grad, target)
            h = h_try
        
        # Averaging
        avg_buf.append(h.copy())
        if len(avg_buf) > avg_window:
            avg_buf.pop(0)
        
        # Check average every 50 iterations
        if it % 50 == 0 and len(avg_buf) >= 10:
            h_avg = np.mean(avg_buf, axis=0)
            h_avg = project_feasible(h_avg, target)
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5 - 1e-14:
                best_c5 = c5_avg
                best_h = h_avg.copy()
                # Reset current to average
                h = h_avg.copy()
                lr = 0.01
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_grad_topk(h, alpha, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if k >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(corr)[-k:]
    weights = np.exp(alpha * (corr[top_idx] - c_max))
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / wsum
    mid = n - 1
    grad = np.zeros(n)
    for wi, m_val in zip(weights, top_idx):
        s = int(m_val) - mid
        if s == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n - s]
        else:
            sp = -s
            grad[sp:] += wi * (1.0 - h[:n - sp])
            grad[:n - sp] -= wi * h[sp:]
    grad *= dx
    return grad


def _con_fun(h):
    return np.sum(h) - _global_target


def _con_jac(h):
    return np.ones(_global_n)


def _smooth_obj_fn(h, alpha, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if k >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(corr)[-k:]
    shifted = alpha * (corr[top_idx] - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def _smooth_grad_fn(h, alpha, k):
    return smooth_grad_topk(h, alpha, k)


def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    k = min(20, 2 * n - 1)
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj_fn, h_cur, args=(alpha, k), jac=_smooth_grad_fn,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def momentum_descent(h_start, n, target, time_limit, max_steps, alpha, mu, base_lr, top_k):
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad_topk(h, alpha, top_k)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        moved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h) - 1e-15:
                h = h_try
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h.copy()
                moved = True
                break
        if not moved:
            h = project_feasible(h - lr * 0.01 * v, target)
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation(h_init, n, target, max_tries, time_limit, rng_seed):

    # Phase 5: Interpolate to n=800 and optimize
    if remaining() > 350:
        n2 = 800
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        h_res = h_interp.copy()
        score_res = compute_c5(h_res)

        # Momentum at n=800
        for alpha_m in [100, 500, 2000, 5000]:
            if remaining() < 60:
                break
            h_m, c5_m = momentum_descent(h_res, n2, target2,
                                         time_limit=min(50, remaining() * 0.25),
                                         max_steps=30000, alpha=alpha_m,
                                         mu=0.85, base_lr=0.06, top_k=20)
            if c5_m < score_res:
                score_res = c5_m
                h_res = h_m

        # Pair CD at n=800
        if remaining() > 100:
            h_cd, c5_cd = pair_cd_fast(h_res, n2, target2,
                                       time_limit=min(80, remaining() * 0.3),
                                       max_sweeps=4000, n_cand=40, top_k=3,
                                       max_pairs=300)
            if c5_cd < score_res:
                score_res = c5_cd
                h_res = h_cd

        # Subgradient at n=800
        if remaining() > 80:
            h_sg, c5_sg = subgradient_avg(h_res, n2, target2,
                                          time_limit=min(50, remaining() * 0.3),
                                          max_iter=20000, top_k=5, avg_window=300)
            if c5_sg < score_res:
                score_res = c5_sg
                h_res = h_sg

        # SLSQP at n=800
        if remaining() > 50:
            h_pol = slsqp_smooth(h_res, n2, target2,
                                 [2000, 8000], 500, 1e-15, min(40, remaining()))
            sc = compute_c5(h_pol)
            if sc < score_res:
                score_res = sc
                h_res = h_pol

        if score_res < best_score:
            best_score = score_res
            best_h = h_res

    # Phase 6: Try n=1600
    if remaining() > 300:
        n3 = 1600
        target3 = n3 / 2.0
        h_interp = interpolate_to(best_h, n3, target3)
        h_res = h_interp.copy()
        score_res = compute_c5(h_res)

        for alpha_m in [100, 500, 2000]:
            if remaining() < 60:
                break
            h_m, c5_m = momentum_descent(h_res, n3, target3,
                                         time_limit=min(60, remaining() * 0.25),
                                         max_steps=40000, alpha=alpha_m,
                                         mu=0.85, base_lr=0.05, top_k=25)
            if c5_m < score_res:
                score_res = c5_m
                h_res = h_m

        if remaining() > 100:
            h_cd, c5_cd = pair_cd_fast(h_res, n3, target3,
                                       time_limit=min(80, remaining() * 0.3),
                                       max_sweeps=3000, n_cand=35, top_k=3,
                                       max_pairs=250)
            if c5_cd < score_res:
                score_res = c5_cd
                h_res = h_cd

        if remaining() > 50:
            h_pol = slsqp_smooth(h_res, n3, target3,
                                 [1000, 5000], 400, 1e-15, min(40, remaining()))
            sc = compute_c5(h_pol)
            if sc < score_res:
                score_res = sc
                h_res = h_pol

        if score_res < best_score:
            best_score = score_res
            best_h = h_res

    # Phase 7: Local perturbation + refinement
    if remaining() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert, c5_pert = local_perturbation(best_h, n_use, target_use,
                                             max_tries=2000, time_limit=min(20, remaining() * 0.3),
                                             rng_seed=12345)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if remaining() > 25:
            h_cd, c5_cd = pair_cd_fast(best_h, n_use, target_use,
                                       time_limit=min(20, remaining() * 0.4),
                                       max_sweeps=3000, n_cand=40, top_k=3,
                                       max_pairs=300)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Final SLSQP
    if remaining() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_smooth(best_h, n_use, target_use,
                               [5000, 20000], 500, 1e-16, min(15, remaining()))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
