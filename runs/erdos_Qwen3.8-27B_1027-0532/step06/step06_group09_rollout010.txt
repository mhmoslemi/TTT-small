def structured_perturbation(h, n, target, rng, mode):
    """Apply a structured perturbation to escape local optima."""
    h_pert = h.copy()
    if mode == 'block_swap':
        block_size = int(rng.randint(5, 20))
        start1 = int(rng.randint(0, n - block_size * 2))
        start2 = start1 + block_size + int(rng.randint(1, n - start1 - block_size))
        h_pert[start1:start1 + block_size], h_pert[start2:start2 + block_size] = \
            h_pert[start2:start2 + block_size].copy(), h_pert[start1:start1 + block_size].copy()
    elif mode == 'inversion':
        start = int(rng.randint(0, n - 30))
        length = int(rng.randint(10, 30))
        h_pert[start:start + length] = h_pert[start:start + length][::-1].copy()
    elif mode == 'gaussian_block':
        block_size = int(rng.randint(10, 40))
        start = int(rng.randint(0, n - block_size))
        h_pert[start:start + block_size] += rng.randn(block_size) * 0.1
    elif mode == 'frequency_perturb':
        freq = rng.choice([1, 2, 3, 4, 5])
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        h_pert += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    else:
        h_pert += rng.randn(n) * 0.05
    return project_feasible(h_pert, target)

            w = corr[s_idx] / np.sum(corr[top_k_idx])
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        h_new = project_feasible(h - lr * grad, target)
        h = h_new
        lr *= decay

    return best_h, best_c5


def two_phase_cd(h, n, target, time_limit):
    """Two-phase pair CD: first reduce max, then balance top lags."""
    start = time.time()
    half_time = time_limit / 2.0

    h1, c5_1 = pair_cd_balanced(h, n, target, half_time,
                                max_sweeps=8000, n_cand_start=30,
                                n_cand_max=80, patience=150,
                                topk_balance=1, mode='max')

    if time.time() - start < time_limit - 10:
        remaining = time_limit - (time.time() - start)
        h2, c5_2 = multi_lag_pair_cd(h1, n, target, remaining,
                                     max_sweeps=5000, n_cand=60,
                                     patience=120, n_top_lags=4)
        if c5_2 < c5_1:
            return h2, c5_2
    return h1, c5_1


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick initial pair CD
    if tleft() > 100:
        h1, c5_1 = pair_cd_balanced(best_h, n, target,
                                    time_limit=min(60, tleft() * 0.1),
                                    max_sweeps=3000, n_cand_start=30,
                                    n_cand_max=60, patience=100,
                                    topk_balance=1, mode='max')
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1

    # Phase 2: Multi-lag pair CD (balances top lags)
    if tleft() > 200:
        h2, c5_2 = multi_lag_pair_cd(best_h, n, target,
                                     time_limit=min(150, tleft() * 0.2),
                                     max_sweeps=5000, n_cand=50,
                                     patience=150, n_top_lags=3)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2

    # Phase 3: L-BFGS-B with high alpha and topk
    if tleft() > 100:
        for alpha in [2000, 10000, 50000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=2000, topk=1)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Multi-lag pair CD again (longer)
    if tleft() > 200:
        h4, c5_4 = two_phase_cd(best_h, n, target,
                                time_limit=min(200, tleft() * 0.25))
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4

    # Phase 5: Structured perturbations + refine
    if tleft() > 150:
        modes = ['block_swap', 'inversion', 'gaussian_block', 'frequency_perturb', 'random']
        for mode in modes:
            if tleft() < 50:
                break
            h_pert = structured_perturbation(best_h, n, target, rng, mode)
            h_ref = pair_cd_balanced(h_pert, n, target,
                                     time_limit=min(30, tleft() * 0.15),
                                     max_sweeps=2000, n_cand_start=30,
                                     n_cand_max=60, patience=80,
                                     topk_balance=1, mode='max')
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 6: Gradient descent on top lags
    if tleft() > 100:
        h6, c5_6 = gradient_descent_topk(best_h, n, target,
                                         time_limit=min(60, tleft() * 0.15),
                                         alpha=5000, topk=3,
                                         max_steps=5000, lr_base=0.015)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6
        # Follow up with pair CD
        if tleft() > 60:
            h6b, c5_6b = pair_cd_balanced(best_h, n, target,
                                          time_limit=min(40, tleft() * 0.15),
                                          max_sweeps=3000, n_cand_start=40,
                                          n_cand_max=80, patience=100,
                                          topk_balance=2, mode='topk')
            if c5_6b < best_score:
                best_score = c5_6b
                best_h = h6b

    # Phase 7: More structured perturbations with different scales
    if tleft() > 120:
        for scale in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfags_polish(h_pert, n, target, 5000, maxiter=1000, topk=1)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if tleft() > 30:
                    h_ref2 = pair_cd_balanced(best_h, n, target,
                                              time_limit=min(20, tleft() * 0.1),
                                              max_sweeps=1500, n_cand_start=30,
                                              n_cand_max=50, patience=60,
                                              topk_balance=1, mode='max')
                    sc2 = compute_c5(h_ref2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_h = h_ref2

    # Phase 8: Final aggressive pair CD with large candidate sets
    if tleft() > 150:
        h8, c5_8 = pair_cd_balanced(best_h, n, target,
                                    time_limit=min(150, tleft() * 0.2),
                                    max_sweeps=10000, n_cand_start=50,
                                    n_cand_max=120, patience=200,
                                    topk_balance=2, mode='topk')
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8

    # Phase 9: Multi-lag balance phase
    if tleft() > 100:
        h9, c5_9 = multi_lag_pair_cd(best_h, n, target,
                                     time_limit=min(80, tleft() * 0.15),
                                     max_sweeps=4000, n_cand=70,
                                     patience=120, n_top_lags=5)
        if c5_9 < best_score:
            best_score = c5_9
            best_h = h9

    # Phase 10: SLSQP final polish
    if tleft() > 60:
        h10 = slsqp_round(best_h, n, target, [10000, 30000], 500, 1e-15, topk=1)
        sc = compute_c5(h10)
        if sc < best_score:
            best_score = sc
            best_h = h10

    # Phase 11: Try higher resolution
    if tleft() > 150:
        for n_try in [500, 600]:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            x_old = np.linspace(0.0, 1.0, len(best_h))
            x_new = np.linspace(0.0, 1.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)
            h_interp = lbfags_polish(h_interp, n_try, target_try, 5000, maxiter=1000, topk=1)
            h_interp = pair_cd_balanced(h_interp, n_try, target_try,
                                        time_limit=min(40, tleft() * 0.2),
                                        max_sweeps=2000, n_cand_start=30,
                                        n_cand_max=60, patience=80,
                                        topk_balance=1, mode='max')
            sc = compute_c5(h_interp)
            if sc < best_score:
                best_score = sc
                best_h = h_interp

    # Phase 12: Final polish cycle
    if tleft() > 40:
        h_final = pair_cd_balanced(best_h, n, target,
                                   time_limit=min(30, tleft() * 0.3),
                                   max_sweeps=3000, n_cand_start=50,
                                   n_cand_max=100, patience=100,
                                   topk_balance=1, mode='max')
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
