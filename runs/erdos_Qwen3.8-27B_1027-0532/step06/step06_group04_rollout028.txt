def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_lp_iteration(h, n, target, t_current, reg):

    dx = 2.0 / n

    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    # Add ALL correlation constraints
    for k_idx in range(half_n + 1, 2 * half_n + 1):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_current + 0.01)  # warm start: allow slight violation
    
    objective = t_var + reg * cp.sum_squares(h_var - h)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=200000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=200000)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    return h_new, t_var.value if t_var.value is not None else None

    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):

        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for i in order:

                    A -= h[i - k]

                    elif np.all(C[active] < -1e-14):

def subgradient_improve(h, n, target, max_steps, time_limit, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    v = np.zeros(n)  # momentum
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        # Nesterov-style momentum
        v = momentum * v + grad
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * v
            h_try = project_feasible(h_try, target)

                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                v = np.zeros(n)
                stall = 0
                if step > 500:
                    break
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        # Adaptive alpha
        progress = step / max(1, max_steps)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        grad = smooth_gradient(h_cur, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Nesterov momentum
        if step == 0:
            v = np.zeros(n)
            mu = 0.0
        if step > 0:
            mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
            beta = (1.0 - mu) / (1.0 + mu)
            y = h_cur + beta * (h_cur - h_prev)
            h_prev = h_cur.copy()
            grad = smooth_gradient(y, alpha)
        
        # Backtracking line search
        improved = False
        for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_try = h_cur - lr_try * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_try.copy()
                h_cur = h_try
                stall = 0
                improved = True
                # Increase lr slightly
                lr = min(1.0, lr * 1.05)
                break
        
        if not improved:
            lr *= 0.5
            stall += 1
            if stall > 200:
                break
    
    return best_h, best_score

def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, all_constraints=False):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        if all_constraints:
            sorted_active = range(half_n + 1, 2 * half_n + 1)
        else:
            corr = np.correlate(h, g, mode='full')
            sorted_active = sorted(np.argsort(corr)[-n_active:].tolist())
        
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=200000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=200000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping)
