import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

            B[valid] += h[idx[valid]]
            idx = positions + kk
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
        
        n_try = min(30, max(10, len(ones) // 20))
        top_i = ones[np.argsort(A[ones])[:n_try]]
        top_j = zeros[np.argsort(-B[zeros])[:n_try]]
        
        found = False
        best_delta = None
        best_score = max_val
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_full = full + delta
                new_max = float(new_full.max())
                if new_max < best_score:
                    best_score = new_max
                    best_delta = (i, j, delta)
        if best_delta is not None:
            i, j, delta = best_delta
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            found = True
        if not found:
            for _ in range(40):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def replica_sa(h_init, n_replicas, t_levels, n_steps, block_frac, max_block_len, rng, time_limit, start_time, k_arr):
    n = len(h_init)
    dx = 2.0 / n
    positions = np.arange(n)
    
    replicas_h = []
    replicas_full = []
    replicas_c5 = []
    for r in range(n_replicas):
        h_r = h_init.copy().astype(float)
        if r > 0:
            n_pert = rng.randint(2, 4 + r)
            ones = np.where(h_r == 1.0)[0]
            zeros = np.where(h_r == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_r[i] = 0.0
                h_r[j] = 1.0
                ones = np.where(h_r == 1.0)[0]
                zeros = np.where(h_r == 0.0)[0]
        full_r = full_corr_fast(h_r)
        c5_r = float(full_r.max()) * dx
        replicas_h.append(h_r)
        replicas_full.append(full_r)
        replicas_c5.append(c5_r)
    
    best_c5 = min(replicas_c5)
    best_idx = int(np.argmin(replicas_c5))
    best_h = replicas_h[best_idx].copy()
    
    swap_interval = max(20, n_steps // 100)
    steps_done = 0
    total = n_steps
    
    while steps_done < total:
        if time.time() - start_time > time_limit:
            break
        steps_done += 1
        r = steps_done % n_replicas  # Round-robin
        
        h_r = replicas_h[r]
        full_r = replicas_full[r]
        t = t_levels[r]
        
        ones = np.where(h_r == 1.0)[0]
        zeros = np.where(h_r == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            continue
        
        use_block = rng.random() < block_frac
        if use_block:
            blen = rng.randint(2, min(max_block_len, n // 15) + 1)
            a = rng.randint(0, n - blen + 1)
            b_idx = a + blen - 1
            h_new = h_r.copy()
            h_new[a:b_idx+1] = h_r[a:b_idx+1][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta = c5_new - replicas_c5[r]
            if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                replicas_h[r] = h_new
                replicas_full[r] = full_new
                replicas_c5[r] = c5_new
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta_corr = compute_delta(h_r, i, j, k_arr)
            full_new = full_r + delta_corr
            c5_new = float(full_new.max()) * dx
            delta = c5_new - replicas_c5[r]
            if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                h_r[i] = 0.0
                h_r[j] = 1.0
                replicas_full[r] = full_new
                replicas_c5[r] = c5_new
        
        if replicas_c5[r] < best_c5:
            best_c5 = replicas_c5[r]
            best_h = replicas_h[r].copy()
        
        # Replica exchange
        if steps_done % swap_interval == 0:
            r1 = rng.randint(0, n_replicas - 1)
            r2 = r1 + 1
            # Metropolis swap criterion
            delta_inv_t = 1.0/t_levels[r2] - 1.0/t_levels[r1]
            delta_f = replicas_c5[r1] - replicas_c5[r2]
            exponent = delta_inv_t * delta_f
            if exponent < 0 or (exponent < 50 and rng.random() < np.exp(-exponent)):
                replicas_h[r1], replicas_h[r2] = replicas_h[r2].copy(), replicas_h[r1].copy()
                replicas_full[r1], replicas_full[r2] = replicas_full[r2].copy(), replicas_full[r1].copy()
                replicas_c5[r1], replicas_c5[r2] = replicas_c5[r2], replicas_c5[r1]
    
    return best_h, best_c5

def adaptive_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr, 
                block_frac=0.25, max_block_len=12, target_acc=0.4):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 40000
    positions = np.arange(n)
    
    # Adaptive temperature
    t = t_init
    acc_count = 0
    window = 500
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        # Anneal temperature
        ratio = it / max(n_iters - 1, 1)
        t_base = t_init * (t_final / t_init) ** ratio
        # Blend with adaptive
        t = 0.7 * t_base + 0.3 * t
        t = max(t, t_final)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        accepted = False
        if rng.random() < block_frac:
            blen = rng.randint(2, min(max_block_len, n // 12) + 1)
            a = rng.randint(0, n - blen + 1)
            b_idx = a + blen - 1
            h_new = h.copy()
            h_new[a:b_idx+1] = h[a:b_idx+1][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta = c5_new - current_c5
            if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                accepted = True
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta_corr = compute_delta(h, i, j, k_arr)
            full_new = full + delta_corr
            c5_new = float(full_new.max()) * dx
            delta = c5_new - current_c5
            if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                h[i] = 0.0
                h[j] = 1.0
                full = full_new
                current_c5 = c5_new
                accepted = True
        
        if accepted:
            acc_count += 1
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Adaptive temperature adjustment
        if (it + 1) % window == 0:
            acc_rate = acc_count / window
            if acc_rate > target_acc + 0.1:
                t *= 0.7
            elif acc_rate < target_acc - 0.1:
                t *= 1.4
            t = np.clip(t, t_final, t_init)
            acc_count = 0
    
    return best_h, best_c5

def continuous_grad(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    N_fft = 1
    while N_fft < 4 * n:
        N_fft <<= 1
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        if remaining < 10:
            break
        phase_time = min(50.0, remaining * 0.9)
        lr = 1.0 / (alpha * 0.5 + 1.0)
        mu = 0.7
        v = np.zeros(n)
        
        n_iter = int(phase_time * 100)
        n_iter = max(n_iter, 20)
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            
            corr = full_corr_fast(h)
            max_c = float(corr.max())
            shifted = np.clip(alpha * (corr - max_c), -300.0, 0.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.ones(2*n-1) / (2*n-1)
            else:
                w = w / w_sum
            
            # Compute gradient: grad[j] = 1 - A[j] - B[j]
            # A[j] = sum_q w[q] * h[j+q] (q = -(n-1) to (n-1))
            # B[j] = sum_q w[q] * h[j-q]
            
            # Using FFT:
            y_pad = np.zeros(N_fft)
            y_pad[:n] = h
            fft_h = np.fft.fft(y_pad)
            
            # A: convolution of w_rev with h, extract middle n elements
            w_rev = w[::-1]
            wr_pad = np.zeros(N_fft)
            wr_pad[:2*n-1] = w_rev
            fft_wr = np.fft.fft(wr_pad)
            A_full = np.fft.ifft(fft_wr * fft_h).real
            A = A_full[(n-1):(n-1)+n]
            
            # B: convolution of w with h, extract middle n elements
            w_pad = np.zeros(N_fft)
            w_pad[:2*n-1] = w
            fft_w = np.fft.fft(w_pad)
            B_full = np.fft.ifft(fft_w * fft_h).real
            B = B_full[(n-1):(n-1)+n]
            
            grad = 1.0 - A - B
            
            v = mu * v - lr * grad
            h = project_box_sum(h + v, target)
            
            if it % 50 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    return best_h, best_score

def diverse_perturb(h, rng, n_swaps, n_blocks):
    n = len(h)
    h_p = h.copy().astype(float)
    ones = np.where(h_p == 1.0)[0]
    zeros = np.where(h_p == 0.0)[0]
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_p[i] = 0.0
        h_p[j] = 1.0
        ones = np.where(h_p == 1.0)[0]
        zeros = np.where(h_p == 0.0)[0]
    for _ in range(n_blocks):
        blen = rng.randint(3, min(20, n // 10) + 1)
        a = rng.randint(0, n - blen + 1)
        h_p[a:a+blen] = h_p[a:a+blen][::-1]
    return h_p

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)
    k_arr = np.arange(2 * n - 1) - (n - 1)
    
    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick greedy
    allowed = min(40, max(10, tleft() * 0.04))
    h_g, c5_g = greedy_improved(best_h, max_iters=50000, rng=rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()
    
    # Phase 2: Replica Exchange SA (main workhorse)
    if tleft() > 200:
        n_reps = 6
        t_levels = np.array([0.04, 0.02, 0.01, 0.005, 0.002, 0.0005])
        allowed = min(400, max(60, tleft() * 0.3))
        n_steps = int(allowed * 8000)
        h_rep, c5_rep = replica_sa(best_h, n_reps, t_levels, n_steps,
                                    block_frac=0.2, max_block_len=12,
                                    rng=rng, time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_rep < best_score:
            best_score = c5_rep
            best_h = h_rep.copy()
    
    # Phase 3: Adaptive SA from current best
    if tleft() > 150:
        allowed = min(200, max(30, tleft() * 0.2))
        n_sa = int(allowed * 60000)
        h_ad, c5_ad = adaptive_sa(best_h, n_sa, 0.02, 0.0001, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                   block_frac=0.25, max_block_len=15, target_acc=0.35)
        if c5_ad < best_score:
            best_score = c5_ad
            best_h = h_ad.copy()
    
    # Phase 4: Multi-start with diverse perturbations
    for restart in range(6):
        if tleft() < 60:
            break
        h_pert = diverse_perturb(best_h, rng, rng.randint(3, 8), rng.randint(1, 3))
        allowed = min(50, max(15, tleft() * 0.06))
        n_sa_r = int(allowed * 50000)
        h_r, c5_r = adaptive_sa(h_pert, n_sa_r, 0.015, 0.00005, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                 block_frac=0.3, max_block_len=10, target_acc=0.4)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Continuous gradient refinement
    if tleft() > 80:
        target_f = float(n) / 2.0
        alphas = [5, 20, 80, 300, 1000, 4000, 15000]
        allowed = min(200, max(30, tleft() * 0.25))
        h_cont, sc = continuous_grad(best_h, target_f, start_time, allowed, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_cont.copy()
    
    # Phase 6: Project to binary and do focused SA
    if tleft() > 60:
        h_bin = project_binary(best_h, target)
        c5_bin = compute_c5(h_bin)
        h_cur = h_bin
        sc_cur = c5_bin
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin.copy()
        
        allowed = min(80, max(20, tleft() * 0.12))
        n_sa_f = int(allowed * 60000)
        h_f, c5_f = adaptive_sa(h_cur, n_sa_f, 0.01, 0.00002, rng,
                                 time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                 block_frac=0.15, max_block_len=8, target_acc=0.3)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()
    
    # Phase 7: Second round of replica SA
    if tleft() > 120:
        n_reps = 4
        t_levels2 = np.array([0.015, 0.006, 0.002, 0.0004])
        allowed = min(150, max(30, tleft() * 0.2))
        n_steps2 = int(allowed * 8000)
        h_rep2, c5_rep2 = replica_sa(best_h, n_reps, t_levels2, n_steps2,
                                      block_frac=0.25, max_block_len=10,
                                      rng=rng, time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_rep2 < best
