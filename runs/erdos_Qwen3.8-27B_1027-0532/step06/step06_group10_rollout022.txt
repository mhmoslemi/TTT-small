import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha, target):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    # Add penalty for constraint violation
    penalty = 100.0 * (np.sum(h) - target) ** 2
    return val + penalty

def smooth_obj_grad(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    # Add penalty gradient
    grad += 200.0 * (np.sum(h) - target)
    # Normalize (remove mean to maintain sum constraint approximately)
    grad -= np.mean(grad)
    return grad

def lbfgsb_optimize(h0, n, target, alphas, maxiter, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            return smooth_obj_val(x, alpha, target)
        def _grad(x):
            return smooth_obj_grad(x, alpha, n, target)
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_value(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha, n, target, penalty_weight):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for s in active:
        m = int(s) - mid
        v1 = (idxs + m >= 0) & (idxs + m < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] + m])
        v2 = (idxs - m >= 0) & (idxs - m < n)
        grad[v2] -= w[s] * h[idxs[v2] - m]
    grad *= dx
    if penalty_weight > 0:
        grad += penalty_weight * (np.sum(h) - target)
    return grad

def lbfgsb_round(h0, n, target, alphas, maxiter, time_limit, start_time, penalty_weights=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    if penalty_weights is None:
        penalty_weights = [alpha * 2.0 for alpha in alphas]
    
    for i, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit:
            break
        pw = penalty_weights[i] if i < len(penalty_weights) else alpha * 2.0
        
        def _obj(x):
            sv = smooth_value(x, alpha) * (2.0 / n)
            pen = 0.5 * pw * (np.sum(x) - target) ** 2
            return sv + pen
        
        def _grad(x):
            return smooth_grad(x, alpha, n, target, pw)
        
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxiter*2})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                # Try a smaller step
                h_mix = project_feasible(0.5 * h + 0.5 * res.x, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < best_c5 - 1e-15:
                    best_c5 = c5_mix
                    best_h = h_mix.copy()
                    h = h_mix
        except Exception:
            pass
    return best_h, best_c5

def slsqp_round(h0, n, target, alphas, maxiter, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def _con(x):
        return np.sum(x) - target
    
    def _con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def _obj(x):
            return smooth_value(x, alpha) * (2.0 / n)
        
        def _grad(x):
            return smooth_grad(x, alpha, n, target, 0.0)
        
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def greedy_cd(h0, n, target, time_limit, max_sweeps=5000, n_cand=20):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        
        k_star = int(np.argmax(corr)) - mid
        # Gradient at k_star
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] + k_star])
        grad[v2] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                # Compute the effect of moving delta from i to j
                g = np.zeros(n_corr)
                # h[i] decreases by delta, h[j] increases by delta
                # Effect on corr[s] = sum_k h[k](1-h[k+m_s])
                for s_idx in range(n_corr):
                    m = s_idx - mid
                    # h[i] appears in corr[s] if i+m valid (as h[k] with k=i) or i-m valid (as h[k+m] with k=i-m)
                    # (1-h[i]) appears if i+m valid (as (1-h[k+m]) with k+m=i, k=i-m) or i-m valid (as (1-h[k+m]) with k+m=i-m... wait)
                    
                    # Let me think again. corr[s] = sum_k h[k] * (1-h[k+m])
                    # h[i] appears when k=i: contributes h[i]*(1-h[i+m]) if i+m valid
                    # h[i] appears when k+m=i: contributes h[i-m]*(1-h[i]) if i-m valid
                    # (1-h[i]) appears when k=i: the (1-h) part, coefficient is h[i]
                    # Wait, I need to think about how changing h[i] affects corr[s].
                    
                    # corr[s] = sum_k h[k] * (1-h[k+m])
                    # If we change h[i] by delta:
                    # Terms where h[k] = h[i] (k=i): delta * (1-h[i+m]) if i+m valid
                    # Terms where h[k+m] = h[i] (k = i-m): h[i-m] * (-delta) if i-m valid
                    # Total change: delta * [(1-h[i+m])*1_{i+m valid} - h[i-m]*1_{i-m valid}]
                    
                    # Similarly for h[j] (increasing by delta):
                    # delta * [(1-h[j+m])*1_{j+m valid} - h[j-m]*1_{j-m valid}]
                    
                    # Net change in corr[s] from moving delta from i to j:
                    # delta * [(1-h[i+m])*1_{i+m valid} - h[i-m]*1_{i-m valid}
                    #          - (1-h[j+m])*1_{j+m valid} + h[j-m]*1_{j-m valid}]
                    
                    ds = 0.0
                    if 0 <= i + m < n:
                        ds += (1.0 - h[i + m])
                    if 0 <= i - m < n:
                        ds -= h[i - m]
                    if 0 <= j + m < n:
                        ds -= (1.0 - h[j + m])
                    if 0 <= j - m < n:
                        ds += h[j - m]
                    g[s_idx] = ds
                
                # Binary search for optimal delta
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_round(h0, n, target, time_limit, n_active=15, max_rounds=30):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for s_idx in active:
            m = int(s_idx) - mid
            # Linearization of corr[s_idx] at current h
            # corr[s] = sum_k h[k](1-h[k+m])
            # Linearization: sum_k [h[k](1-h[k+m]0) + (1-h0[k+m])*(h[k]-h0[k])(-1)]
            # = sum_k h0[k](1-h0[k+m])  (current value)
            #   + sum_k [(1-h0[k+m])*(h[k]-h0[k]) - h0[k+m]*... ]
            
            # Actually, the linearization of f(h) = sum_k h[k]*(1-h[k+m]) at h0:
            # f(h) ≈ f(h0) + sum_k [(1-h0[k+m])] * (h[k] - h0[k])
            #         + sum_k h0[k] * [-(h[k+m] - h0[k+m])]
            # = f(h0) + sum_k (1-h0[k+m])*(h[k]-h0[k]) - sum_k h0[k]*(h[k+m]-h0[k+m])
            
            # The gradient at h0 is:
            # grad[j] = (1-h0[j+m]) * 1_{j+m valid} - h0[j-m] * 1_{j-m valid}
            
            grad_k = np.zeros(n)
            v1 = (idxs + m >= 0) & (idxs + m < n)
            v2 = (idxs - m >= 0) & (idxs - m < n)
            grad_k[v1] += (1.0 - h[idxs[v1] + m])
            grad_k[v2] -= h[idxs[v2] - m]
            
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS, verbose=False)
                except Exception:
                    break
        
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.3
            h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def random_perturb(h, n, target, sigma=0.1):
    noise = np.random.normal(0, sigma, n)
    h_new = project_feasible(h + noise, target)
    return h_new
