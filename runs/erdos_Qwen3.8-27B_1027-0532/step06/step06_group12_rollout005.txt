    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Build LP: fix g = 1-h, optimize h
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        n_half = n - 1
        # For each lag k from -(n-1) to (n-1)
        for k in range(-n_half, n_half + 1):
            abs_k = abs(k)
            if abs_k == 0:
                coeff = g * dx  # (1-h[i]) * dx
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n-k] = g[k:] * dx
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = g[:n-m] * dx
            constraints.append(coeff @ h_var <= t_var)
        
        # Add regularization to stay near current h
        reg = 1e-4
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:

                break
        
        if h_var.value is None:

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.7 * h + 0.3 * h_new  # Damping

        if c5_val < best_c5 - 1e-14:

def lp_step(h_fixed, n, target):

    n_half = n - 1
