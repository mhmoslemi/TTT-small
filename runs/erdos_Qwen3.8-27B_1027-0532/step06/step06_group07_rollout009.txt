def subgradient_step(h, n, target, top_k=20, step_size=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k lags
    top_indices = np.argsort(corr)[-top_k:]
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for idx in top_indices:
        k = int(idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= top_k
    grad -= np.mean(grad)  # Remove mean to preserve sum constraint
    h_new = project_feasible(h - step_size * grad, target)

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_fn(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def _smooth_obj_wrapper(h, alpha):
    return smooth_obj_fn(h, alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            def obj_fn(x):
                return smooth_obj_fn(x, alpha)
            res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha,
                                                       'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd(h, n, target, time_limit, max_sweeps=10000, n_cand=50, n_bisect=15):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 120:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 50:
                break

    return best_h, best_c5


def topk_subgradient(h, n, target, time_limit, top_k=30, max_steps=5000,
                     base_lr=0.005, decay=0.999):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break

        top_indices = np.argsort(corr)[-top_k:]
        for idx in top_indices:
            if corr[idx] < np.max(corr) - dx / n:
                top_indices = top_indices[top_indices >= idx]
                break

        grad = np.zeros(n)
        for idx in top_indices:
            k = int(idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        n_active = len(top_indices)
        if n_active == 0:
            break
        grad /= n_active
        grad -= np.mean(grad)

        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= decay
            if lr < 1e-8:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_gaussian(h, sigma_idx):
    n = len(h)
    if sigma_idx < 0.5:
        return h.copy()
    k = int(sigma_idx * 3)
    if k < 1:
        k = 1
    x = np.arange(-k, k + 1)
    kernel = np.exp(-0.5 * (x / sigma_idx) ** 2)
    kernel /= kernel.sum()
    h_padded = np.concatenate([h[-k:], h, h[:k]])
    h_smooth = np.convolve(h_padded, kernel, mode='valid')
    return h_smooth


def multi_restart_cd(h_best, n, target, time_per_restart, seed_val, max_restarts=5):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)

    for r in range(max_restarts):
        if time.time() - start > time_per_restart:
            break
        scales = [0.02, 0.05, 0.1, 0.2]
        scale = scales[r % len(scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        t_left_r = max(1.0, time_per_restart - (time.time() - start))
        h_try, c5_try = pair_cd(h_pert, n, target, time_limit=t_left_r * 0.8,
                                max_sweeps=3000, n_cand=40, n_bisect=14)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    return best_h, best_c5
