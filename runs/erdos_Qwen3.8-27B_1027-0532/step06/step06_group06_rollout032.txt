- Continue the binary search

For the gradient of the top-k objective:
∇(∑_{k∈top_k} f_k(h)) = ∑_{k∈top_k} ∇f_k(h)

where ∇f_k(h)[i] = (1-h[i-k]) - h[i+k] (from the bilinear form)

∂f_k/∂h[i] = (1-h[i-k]) - ∑_{j: j+k=i} h[j](1-h[j-... 

Hmm, this is bilinear, so the gradient is:
∂f_k/∂h[i] = (1-h[i-k]) + ∂/∂h[i] ∑_j h[j](1-h[j-k]) where j-k=i, i.e., j=i+k
= (1-h[i-k]) - h[i+k]

So ∇f_k(h)[i] = (1-h[i-k]) - h[i+k] (with h out of bounds = 0)

This is the same formula as the change I computed above! So the subgradient of the max is just the change vector for the lag that achieves the max.

For the projected gradient step:
h_new = project(h - α * ∇f_{k*}(h))

where project clips to [0,1] and adjusts sum to n/2.

This is very cheap to compute! Let me include this.

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]

            h_r[np.where(h_r == 1)[0][0]] = 0
        else:
            h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r.astype(float)

        # grad[i] = mean over top T of [(1-h[i-k]) - h[i+k]]

f_k(h) = ∑_i h[i](1-h[i-k])
For h + δ:

h[i]δ_{i-k} = δ_{i-k} if h[i]=1, else 0.

So the linearized change is:
Δf_k = ∑_i [δ_i * 1_{h[i-k]=0} - δ_{i-k} * 1_{h[i]=1}]

Let me re-index the second term: let j = i-k, so i = j+k:
∑_j δ_{j} * 1_{h[j+k]=1}

So: Δf_k = ∑_i δ_i * 1_{h[i-k]=0} - ∑_j δ_j * 1_{h[j+k]=1}
= ∑_i δ_i * (1_{h[i-k]=0} - 1_{h[i+k]=1})

Note that 1_{h[i-k]=0} - 1_{h[i+k]=1} can be:
- 1 - 1 = 0 (if h[i-k]=0 and h[i+k]=1)
- 1 - 0 = 1 (if h[i-k]=0 and h[i+k]=0)
- 0 - 1 = -1 (if h[i-k]=1 and h[i+k]=1)
- 0 - 0 = 0 (if h[i-k]=1 and h[i+k]=0)

So the linearized change in f_k is:
Δf_k = ∑_i δ_i * c_i(k)

where c_i(k) = 1_{h[i-k]=0} - 1_{h[i+k]=1} ∈ {-1, 0, 1}.

The constraint is: ∑_i δ_i = 0 (maintain sum)

minimize t
subject to:

def lp_continuous_refine(h_binary, target, eps=0.3):
    """Use LP to find the best perturbation of a binary solution."""
    from scipy.optimize import linprog
    n = len(h_binary)
    dx = 2.0 / n
    
    h = h_binary.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Variables: δ_0, ..., δ_{n-1}, t
    # Total: n+1 variables
    # 
    # Minimize: t (last variable)
    # 
    # Constraints:
    # For each lag k: ∑_i c_i(k) * δ_i - t ≤ -f_k(h)
    # ∑_i δ_i = 0
    # -eps ≤ δ_i ≤ 0 for h[i]=1
    # 0 ≤ δ_i ≤ eps for h[i]=0
    
    n_lags = 2*n - 1
    c_matrix = np.zeros((n_lags, n))
    
    i_vals = np.arange(n)
    for m in range(n_lags):
        k = m - (n - 1)
        for i in range(n):
            c_out = 0
            if 0 <= i - k < n:
                c_out += (1 - h[i-k])
            if 0 <= i + k < n:
                c_out -= h[i+k]
            c_matrix[m, i] = c_out
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each lag m: ∑ c_i(m) * δ_i - t ≤ -f_k(m)
    A_ub = np.hstack([c_matrix, -np.ones((n_lags, 1))])
    b_ub = -corr.astype(float)
    
    # Equality constraint: ∑ δ_i = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds on δ_i
    bounds = []
    for i in range(n):
        if h[i] > 0.5:  # was 1, can only decrease
            bounds.append((-eps, 0.0))
        else:  # was 0, can only increase
            bounds.append((0.0, eps))
    bounds.append((None, None))  # t is unbounded above (we minimize it)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        delta = result.x[:n]
        h_new = np.clip(h + delta, 0, 1)
        h_new = project_box_sum(h_new, target)
        return h_new
    
    return h

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    elif current_sum < target_sum:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    while int(h_r.sum()) != target_sum:
        if int(h_r.sum()) > target_sum:
            h_r[np.where(h_r == 1)[0][0]] = 0
        else:
            h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r.astype(float)

def binary_greedy_step(h, corr, ones, zeros, T, n):
    n_zeros = len(zeros)
    T_eff = min(T, 2 * n - 1)
    if T_eff >= 2 * n - 1:
        top_T = np.arange(2 * n - 1)
    else:
        top_T = np.argpartition(corr, -T_eff)[-T_eff:]
    k_vals = top_T - (n - 1)
    t_mk = ones[:, None] - k_vals[None, :]
    t_pk = ones[:, None] + k_vals[None, :]
    vtm = (t_mk >= 0) & (t_mk < n)
    vtp = (t_pk >= 0) & (t_pk < n)
    htm = np.where(vtm, h[np.clip(t_mk, 0, n - 1)], 0.0)
    htp = np.where(vtp, h[np.clip(t_pk, 0, n - 1)], 0.0)
    ch_ones = -(1.0 - htm) + htp
    s_mk = zeros[:, None] - k_vals[None, :]
    s_pk = zeros[:, None] + k_vals[None, :]
    vsm = (s_mk >= 0) & (s_mk < n)
    vsp = (s_pk >= 0) & (s_pk < n)
    hsm = np.where(vsm, h[np.clip(s_mk, 0, n - 1)], 0.0)
    hsp = np.where(vsp, h[np.clip(s_pk, 0, n - 1)], 0.0)
    ch_zeros = (1.0 - hsm) - hsp
    ct = corr[top_T]
    new_vals = ct[None, None, :] + ch_ones[:, None, :] + ch_zeros[None, :, :]
    new_max = new_vals.max(axis=2)
    best_flat = int(new_max.argmin())
    bi = best_flat // n_zeros
    bj = best_flat % n_zeros
    new_max_val = float(new_max[bi, bj])
    cur_max = float(corr.max())
    return bi, bj, new_max_val, cur_max

                h[ones[ri]] = 0
                h[zeros[rj]] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(corr.max()) * dx
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h.copy()
                    no_improve = 0
            no_improve += 1
    return best_h, best_c5

def lp_refine(h_binary, target, eps=0.4, max_iters=5):
    n = len(h_binary)
    dx = 2.0 / n
    h = h_binary.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_lags = 2 * n - 1
        c_matrix = np.zeros((n_lags, n))
        i_vals = np.arange(n)
        for m in range(n_lags):
            k = m - (n - 1)
            i_mk = i_vals - k
            i_pk = i_vals + k
            valid_m = (i_mk >= 0) & (i_mk < n)
            valid_p = (i_pk >= 0) & (i_pk < n)
            c_matrix[m] = np.where(valid_m, 1.0 - h[np.clip(i_mk, 0, n - 1)], 0.0) - np.where(valid_p, h[np.clip(i_pk, 0, n - 1)], 0.0)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([c_matrix, -np.ones((n_lags, 1))])
        b_ub = -corr.astype(float)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for i in range(n):
            if h[i] > 0.5:
                bounds.append((-eps, 0.0))
            else:
                bounds.append((0.0, eps))
        bounds.append((None, None))
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[:n]
            h_new = np.clip(h + delta, 0.0, 1.0)
            h_new = project_box_sum(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < best_c5:
                best_c5 = sc_new
                best_h = h_new.copy()
                h = h_new
            else:
                break
        else:
            break
    return best_h, best_c5

def continuous_subgradient(h_start, target, T=10, n_steps=50, alpha_scale=1.0, deadline=None):
    n = len(h_start)
    h = project_box_sum(h_start.copy(), target)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_scale * dx
    i_vals = np.arange(n)
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_T = np.argpartition(corr, -T)[-T:]
        k_list = (top_T - (n - 1)).astype(int).tolist()
        grad = np.zeros(n)
        for k in k_list:
            i_mk = i_vals - k
            i_pk = i_vals + k
            v1 = (i_mk >= 0) & (i_mk < n)
            v2 = (i_pk >= 0) & (i_pk < n)
            h1 = np.where(v1, h[np.clip(i_mk, 0, n - 1)], 0.0)
            h2 = np.where(v2, h[np.clip(i_pk, 0, n - 1)], 0.0)
            grad += (1.0 - h1) - h2
        grad /= T
        grad *= dx
        h = h - alpha * grad
        h = project_box_sum(h, target)
        sc = compute_c5(h)
        if sc < best_c5:
            best_c5 = sc
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_binary(h_init, target_int)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive fast binary search
    deadline1 = start_time + min(120, budget_s * 0.10)
    h_b, c5_b = fast_binary_search(best_h, target_int, deadline1, T=30, plateau=5000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b

    # Phase 2: LP refinement
    deadline2 = start_time + min(200, budget_s * 0.18)
    if time.time() < deadline2 - 5:
        h_lp, c5_lp = lp_refine(best_h, target, eps=0.4, max_iters=3)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        h_bin2 = project_binary(h_lp, target_int)
        deadline2b = time.time() + min(60, deadline2 - time.time())
        h_b2, c5_b2 = fast_binary_search(h_bin2, target_int, deadline2b, T=40, plateau=3000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2

    # Phase 3: Multi-restart with LP in loop
    deadline3 = start_time + budget_s * 0.60
    n_restarts = 12
    for r in range(n_restarts):
        if time.time() > deadline3:
            break
        remaining = deadline3 - time.time()
        if remaining < 15:
            break

        if r % 4 == 0:
            h_start_r = best_h.copy()
            n_flips = rng.randint(5, 15)
            ones = np.where(h_start_r == 1)[0]
            zeros = np.where(h_start_r == 0)[0]
            fi = rng.choice(len(ones), min(n_flips, len(ones)), replace=False)
            fj = rng.choice(len(zeros), min(n_flips, len(zeros)), replace=False)
            h_start_r[ones[fi]] = 0
            h_start_r[zeros[fj]] = 1
        elif r % 4 == 1:
            noise = rng.normal(0, 0.15, n)
            h_start_r = project_binary(project_box_sum(best_h + noise, target), target_int)
        elif r % 4 == 2:
            h_start_r = best_h.copy()
            corr = np.correlate(h_start_r, 1.0 - h_start_r, mode='full')
            worst_idx = int(np.argmax(corr))
            worst_k = worst_idx - (n - 1)
            ones = np.where(h_start_r == 1)[0]
            cands = [i for i in ones if 0 <= i + worst_k < n and h_start_r[i + worst_k] == 0]
            if len(cands) > 0:
                flip_i = cands[rng.randint(len(cands))]
                zeros = np.where(h_start_r == 0)[0]
                flip_j = zeros[rng.randint(len(zeros))]
                h_start_r[flip_i] = 0
                h_start_r[flip_j] = 1
        else:
            h_start_r = project_binary(best_h, target_int)

        t_bin = max(5, remaining * 0.4)
        h_br, c5_br = fast_binary_search(h_start_r, target_int, time.time() + t_bin, T=30, plateau=2000)

        t_lp = max(3, remaining * 0.15)
        if time.time() < time.time() + t_lp and c5_br < best_c5 + 0.001:
            h_lpr, c5_lpr = lp_refine(h_br, target, eps=0.3, max_iters=2)
            if c5_lpr < best_c5:
                best_c5 = c5_lpr
                best_h = h_lpr
            h_binr = project_binary(h_lpr, target_int)
            h_brr, c5_brr = fast_binary_search(h_binr, target_int, time.time() + min(10, remaining * 0.1), T=30, plateau=1000)
            if c5_brr < best_c5:
                best_c5 = c5_brr
                best_h = h_brr
        elif c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br

    # Phase 4: Long polishing with alternating binary + LP
    deadline4 = start_time + budget_s - 30
    if time.time() < deadline4:
        h_cur = best_h.copy()
        if h_cur.min() >= 0.99 and h_cur.max() <= 1.01:
            h_cur = project_binary(h_cur, target_int)
        while time.time() < deadline4:
            t_bin = min(30, deadline4 - time.time())
            if t_bin < 3:
                break
            h_cur, c5_cur = fast_binary_search(h_cur, target_int, time.time() + t_bin, T=50, plateau=2000)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur
            if time.time() < deadline4 - 5:
                h_lpc, c5_lpc = lp_refine(h_cur, target, eps=0.3, max_iters=3)
                if c5_lpc < best_c5:
                    best_c5 = c5_lpc
                    best_h = h_lpc
                    h_cur = h_lpc
                else:
                    h_cur = project_binary(h_lpc, target_int)
            else:
                break

    # Phase 5: Final LP polish on the best continuous solution
    if time.time() < start_time + budget_s - 5:
        h_best_cont = project_box_sum(best_h, target)
        h_fin, c5_fin = lp_refine(h_best_cont, target, eps=0.2, max_iters=4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
