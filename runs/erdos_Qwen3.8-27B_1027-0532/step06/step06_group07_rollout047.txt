def subgradient_nesterov(h_init, n, target, time_limit, eta0=0.5, tol=1e-10, max_iter=5000):
    """
    Nesterov-accelerated subgradient method for min max_k corr[k]*dx
    subject to 0 <= h <= 1, sum(h) = target
    """
    start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h_prev = h.copy()
    momentum = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    beta = 1.0  # Nesterov momentum parameter
    step = eta0
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Nesterov extrapolation
        h_nest = h + momentum * (h - h_prev)
        h_nest = np.clip(h_nest, 0.0, 1.0)
        
        corr = np.correlate(h_nest, 1.0 - h_nest, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_nest.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            momentum *= 0.5
            if momentum < 1e-4:
                break
        
        # Identify active set
        c_max = np.max(corr)
        active_tol = max(1e-8, 0.01 * c_max)
        active = corr >= c_max - active_tol
        
        # Compute subgradient (average over active set)
        n_active = int(np.sum(active))
        grad = np.zeros(n)
        for s_idx in np.where(active)[0]:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h_nest[idxs[v1] - k])
            grad[v2] -= h_nest[idxs[v2] + k]
        grad /= n_active
        
        # Make gradient consistent with equality constraint
        grad -= np.mean(grad)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Step size (diminishing)
        step = step * 0.999  # slight decay
        # Or: step = eta0 / (1 + 0.01 * it)
        
        # Take step
        h_try = project_feasible(h - step * grad, target)
        
        # Check if we improved on the actual (non-extrapolated) point
        h_eval = project_feasible(h_nest, target)
        c5_eval = compute_c5(h_eval)
        if c5_eval < best_c5 - 1e-13:
            best_c5 = c5_eval
            best_h = h_eval.copy()
        
        h_prev = h
        h = h_try
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0 - 1.0
    
    return best_h, best_c5

    Cutting plane method for min max_k g_k(h).
    Each iteration adds a linear constraint and solves an LP.
    """

    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Cut collection: list of (gradient, value) pairs
    cuts = []  # Each cut: (grad_vec, offset) representing grad·(h-h₀) + offset <= t/dx
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Build LP: min t
        # s.t. grad_k · (h - h_k) + g_k(h_k) <= t/dx  for each cut k
        #      0 <= h[i] <= 1
        #      sum(h) = target
        
        # Variables: [h[0], ..., h[n-1], t]  (n+1 variables)
        # Minimize t: c = [0,...,0, 1]
        
        from scipy.optimize import linprog
        
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []

        # Rearranged: grad_k · h - (1/dx) * t <= g_k(h_k) - grad_k · h_k

            row[:n] = grad_k

        # A_ub: inequality constraints (Ax <= b)
        # A_eq: equality constraints (Ax = b)
        # bounds: variable bounds

        # And: -h[i] <= 0 (also in bounds)
        # So just the cut constraints as A_ub
        
        if len(cuts) == 0:
            # No cuts yet, just return current
            break
        
        A_ub = np.array(A_ub_list) if A_ub_list else None
        b_ub = np.array(b_ub_list) if b_ub_list else None
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds = [(0.0, 1.0)] * n + [(None, None)]
        
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

            break

        t_new = res.x[n] * dx

Total: maybe 5-100s. Feasible!

But there's a subtlety: the linearized cuts are only valid locally. As h moves, the cuts become less accurate. The cutting plane method for non-smooth convex functions works because the cuts are valid globally (since the function is convex). But our function g_k(h) = ∑h[i](1-h[i+k]) is NOT convex in h (it has the h[i]*h[i+k] term which is indefinite).

So the standard cutting plane method doesn't directly apply. The linearization is only a local approximation.

However, we can still use it as a heuristic: solve the LP, take a step toward the LP solution, and re-evaluate. This is like a "sequential linear programming" (SLP) approach.

Actually, let me think about this more carefully. The function g_k(h) = ∑ h[i](1-h[i+k]) can be written as:
g_k(h) = ∑ h[i] - ∑ h[i]h[i+k]

The first term is linear, the second is bilinear (quadratic but not positive/negative definite).

At a local optimum of the max problem, we could potentially use the second-order information. But for the SLP approach, we just use the first-order approximation and iterate.

The SLP approach:
1. At current point h_m, compute gradient ∇g_k(h_m) for active k
2. Solve: min t s.t. g_k(h_m) + ∇g_k(h_m)·(h - h_m) ≤ t for k in S, 0≤h≤1, sum(h)=target
3. Get solution h_{m+1}
4. Go to step 1

This is a valid approach (similar to SQP but with linear instead of quadratic subproblems). It should converge locally.

The issue is that with n=600 variables, solving this LP at each iteration might be slow. Let me estimate:
- 601 variables, ~200 inequality constraints, 1 equality constraint
- HiGHS solver: probably 1-10ms per solve
- With 100 iterations: 0.1-1s

That's very fast! Let me implement this.

But wait, I need to be careful. The LP solution might be far from the current point, making the linearization invalid. I should use a trust region: limit ||h_{m+1} - h_m|| ≤ Δ.

Or, simpler: just take a fraction of the way to the LP solution. Like h_{m+1} = (1-α)h_m + α * h_LP.

```python
def slp_method(h_init, n, target, time_limit, max_iter=200, trust_delta=None):
    """
    Sequential Linear Programming for min max_k g_k(h).
    """
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    active_tol = 5e-4  # tolerance for "active" shifts
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Identify active set
        active = corr >= c_max - active_tol
        active_s = np.where(active)[0]
        
        # Compute gradients for active shifts
        grads = np.zeros((len(active_s), n))
        for ii, s in enumerate(active_s):
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grads[ii][v1] += (1.0 - h[idxs[v1] - k])
            grads[ii][v2] -= h[idxs[v2] + k]
        
        # Solve LP: min t
        # s.t. g_k(h) + grad_k · (h_new - h) <= t  for k in active
        #      0 <= h_new <= 1
        #      sum(h_new) = target
        
        from scipy.optimize import linprog
        
        n_var = n + 1
        c_lp = np.zeros(n_var)
        c_lp[n] = 1.0  # minimize t
        
        A_ub_list = []
        b_ub_list = []
        
        for ii in range(len(active_s)):
            s = active_s[ii]
            k = int(s) - mid
            g_val = corr[s]
            gk = grads[ii]
            # Constraint: g_val + gk · (h_new - h) <= t
            # => gk · h_new - t <= g_val - gk · h
            row = np.zeros(n_var)
            row[:n] = gk
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(g_val - np.dot(gk, h))
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 5})
        
        if not res.success:
            break
        
        h_lp = res.x[:n]
        t_lp = res.x[n]
        
        # Verify actual objective at LP solution
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        # Take a step toward LP solution (trust region)
        if trust_delta is not None:
            step_vec = h_lp - h
            step_norm = np.linalg.norm(step_vec)
            if step_norm > trust_delta:
                h_lp = h + trust_delta * step_vec / step_norm
                h_lp = project_feasible(h_lp, target)
        
        h = h_lp
        
        # Check convergence
        if abs(c5_lp - c5_val) < 1e-10:
            break
    
    return best_h, best_c5
```
