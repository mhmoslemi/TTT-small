def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_min_t(g, n, target, active_set, half_n, reg_coeff=1e-9, h_ref=None, scs_eps=1e-12):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in sorted(active_set):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    obj_terms = [t_var]
    if reg_coeff > 0 and h_ref is not None:
        obj_terms.append(reg_coeff * cp.sum_squares(h_var - h_ref))
    prob = cp.Problem(cp.Minimize(cp.sum(obj_terms)), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
    except Exception:
        return None, None, None
    if h_var.value is None:
        return None, None, None
    h_sol = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    t_sol = float(t_var.value) if t_var.value is not None else None
    duals = []
    for c in prob.constraints[3:]:
        try:
            if c.dual_value is not None:
                duals.append(float(c.dual_value))
            else:
                duals.append(0.0)
        except Exception:
            duals.append(0.0)
    return h_sol, t_sol, duals

        grad[valid2] -= h[idxs[valid2] + k]

def smooth_softmax_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -350.0, 0.0)
    val = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    return val

def lbfgs_softmax(h_init, n, target, alphas, maxiter, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        def obj(h_flat):
            return smooth_softmax_obj(h_flat, alpha)

        def grad(h_flat):
            corr = np.correlate(h_flat, 1.0 - h_flat, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -350.0, 0.0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            g = np.zeros(n)
            idxs = np.arange(n)
            for k_idx in range(len(corr)):
                w = weights[k_idx]
                if w < 1e-10:
                    continue
                k = k_idx - (n - 1)
                valid1 = (0 <= idxs - k) & ((idxs - k) < n)
                valid2 = (0 <= idxs + k) & ((idxs + k) < n)
                g[valid1] += w * (1.0 - h_flat[idxs[valid1] - k])
                g[valid2] -= w * h_flat[idxs[valid2] + k]
            g *= dx
            return g

        def con_fun(h_flat):
            return np.sum(h_flat) - target

        def con_jac(h_flat):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                best_h = h_lp600
                n = n600
                target = target600
            elif sc600 < best_score:
                best_score = sc600
                best_h = h600
                n = n600
                target = target600

    # Phase 8: Try n=800 (higher resolution for finer control)
    if tleft() > 120:
        n800 = 800
        target800 = n800 / 2.0
        h800 = interpolate_to(best_h, n800, target800)
        sc800 = compute_c5(h800)
        if tleft() > 80:
            lp_time = min(60, tleft() * 0.3)
            h_lp800, c5_lp800 = alternating_lp_improved(h800, n800, target800,
                                                       max_iter=15000, time_limit=lp_time,
                                                       n_active=80, scs_eps=1e-11)
            if c5_lp800 < best_score:
                best_score = c5_lp800
                best_h = h_lp800
                n = n800
                target = target800
            elif sc800 < best_score:
                best_score = sc800
                best_h = h800
                n = n800
                target = target800

    # Phase 9: Final LP polish at current resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(50, tleft() * 0.6)
        h_lp, c5_lp = alternating_lp_improved(best_h, n_use, target_use,
                                              max_iter=20000, time_limit=lp_time,
                                              n_active=150, scs_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Final pairwise refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pw, c5_pw = pairwise_improve(best_h, n_use, target_use,
                                       max_iters=50000, time_limit=min(30, tleft() * 0.7),
                                       rng=rng)
        if c5_pw < best_score:
            best_score = c5_pw
            best_h = h_pw

    # Phase 11: L-BFGS final polish with high alpha
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lb, c5_lb = lbfgs_softmax(best_h, n_use, target_use,
                                     alphas=[5000, 20000], maxiter=300,
                                     time_limit=min(30, tleft() * 0.7))
        if c5_lb < best_score:
            best_score = c5_lb
            best_h = h_lb

    # Final: multi-start perturbation and quick LP
    for noise in [0.05, 0.15]:
        if tleft() < 50:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        if tleft() > 40:
            lp_time = min(30, tleft() * 0.5)
            h_lp, c5_lp = alternating_lp_improved(h_pert, n_use, target_use,
                                                  max_iter=10000, time_limit=lp_time,
                                                  n_active=100, scs_eps=1e-11)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
