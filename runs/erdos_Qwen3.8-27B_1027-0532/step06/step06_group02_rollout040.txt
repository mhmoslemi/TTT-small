```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)
```

```python
def adam_continuous_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(80.0, time_limit - (time.time() - start_time))
        
        # Adam parameters
        lr = 1.0 / (alpha * 0.5)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        
        m = np.zeros(n)
        v = np.zeros(n)
        
        n_iter = max(100, int(phase_time * 50))
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            # Smooth gradient
            max_c = corr.max()
            exp_c = np.exp(alpha * (corr - max_c))
            w = exp_c / exp_c.sum()
            # Gradient of soft-max w.r.t. h
            N_fft = 1
            while N_fft < 3 * n - 2:
                N_fft <<= 1
            W_fft = np.fft.fft(w, N_fft)
            h_pad = np.zeros(N_fft)
            h_pad[:n] = h
            h_fft = np.fft.fft(h_pad)
            conv_full = np.fft.ifft(W_fft * h_fft).real
            conv = conv_full[:n]
            grad = 1.0 - 2.0 * conv
            
            # Adam update
            t = it + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
            
            if it % 100 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score
```

```
conv = convolve(w, h)  # one-sided, giving sum_k w_k * h_{i-k}
grad = 1 - 2 * conv
```

This is the gradient of sum_k w_k * corr_k with respect to h_i, under the approximation that w_k is fixed (doesn't depend on h). This is a valid approximation when alpha is not too large (w is spread out).

For the purpose of this problem, this should be sufficient. Let me proceed.

Now, let me think about the binary local search more carefully.

**Key idea: "Hill climbing with lookahead"**

Instead of accepting a move if it improves the current C5, look ahead 1-2 steps:
- For each candidate move, compute the new C5
- Among improving moves, pick the one that gives the most improvement
- This is greedy, but with a larger neighborhood it can be more effective

The previous code already does something like this in `fast_greedy` (it tries top 25 candidates for i and j).

**New idea: " pair optimization"**

At each step, find the (i, j) pair that gives the maximum decrease in C5. But instead of trying all pairs (O(n²)), use the gradient to narrow down candidates:
- i should be a position where h_i = 1 and the "cost" of removing it is high (gradient suggests it's bad to have it there)
- j should be a position where h_j = 0 and the "benefit" of adding it is high

This is what `sa_focused` does in the previous code (every 4th iteration, it looks at top correlation lags and finds good i, j pairs).

Let me enhance this:
- Maintain a running "damage" array: for each position i where h_i = 1, compute how much it contributes to the max correlation
- Similarly for positions where h_j = 0
- Always pick the i with highest damage and j with highest benefit

This can be updated incrementally.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_gradient(h, corr_full, alpha):
    n = len(h)
    max_c = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - max_c))
    w = exp_c / exp_c.sum()
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    return 1.0 - 2.0 * conv

def adam_refine(h0, target, start_time, time_limit):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    alphas = [3, 8, 20, 50, 150, 400, 1200, 4000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        # Adaptive learning rate
        lr = 0.5 / np.sqrt(alpha + 1.0)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        max_iters = max(50, int(remaining * 40))
        for it in range(max_iters):
            t += 1
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            grad = compute_gradient(h, corr, alpha)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h, target)
            if it % 200 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score

def guided_sa(h, max_iters, t_init, t_final, rng, start_time, time_limit, k_arr, alpha_guide=50):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)
    stall = 0
    max_stall = 50000
    grad_cache = None
    grad_iter = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Update gradient every 1000 iterations
        if grad_iter % 1000 == 0 or grad_cache is None:
            grad_cache = compute_gradient(h, full, alpha_guide)
        
        if it % 3 == 0 and grad_cache is not None:
            # Gradient-guided selection
            # Prefer i (1->0) where grad is high (bad to have 1 there)
            # Prefer j (0->1) where grad is low/negative (good to have 1 there)
            i_scores = grad_cache[ones]
            j_scores = -grad_cache[zeros]
            # Top-K candidates
            k_sel = min(50, len(ones))
            top_i = ones[np.argsort(i_scores)[::-1][:k_sel]]
            top_j = zeros[np.argsort(j_scores)[::-1][:k_sel]]
            # Randomly pick from top candidates
            i = top_i[rng.randint(len(top_i))]
            j = top_j[rng.randint(len(top_j))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        grad_iter += 1
    return best_h, best_c5

def segment_sa(h, max_iters, t_init, t_final, rng, start_time, time_limit, k_arr,

        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)

        if rng.random() < seg_frac:

            blen = rng.randint(seg_min, min(seg_max, n // 5) + 1)

        else:
            # Swap move
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())

                    stall = 0

                    stall += 1
            else:
