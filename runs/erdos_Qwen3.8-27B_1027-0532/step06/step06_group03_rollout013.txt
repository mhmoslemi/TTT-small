    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

def _con_fun(h):
    return np.sum(h) - _g_target

def _con_jac(h):
    return np.ones(_g_n)

_g_n = 0
_g_target = 0.0

def slsqp_solve(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_neighbor_sums(h, shifts):
    n = len(h)
    nshifts = len(shifts)
    N = np.zeros((n, nshifts))
    for ki in range(nshifts):
        k = int(shifts[ki])
        if k >= 0:
            if k < n:
                N[k:, ki] += h[:n - k]
                N[:n - k, ki] += h[k:]
        else:
            kk = -k
            if kk < n:
                N[:n - kk, ki] += h[kk:]
                N[kk:, ki] += h[:n - kk]
    return N

def binary_local_search(h_bin, target, time_limit, max_iter=100000, rng=None, top_k=5):
    n = len(h_bin)
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.astype(float).copy()
    current_corr = compute_corr(h)
    current_max = float(np.max(current_corr))
    best_corr = current_corr.copy()
    best_max = current_max
    best_h = h.copy()
    start_t = time.time()
    no_improve = 0
    one_positions = np.where(h == 1)[0]
    zero_positions = np.where(h == 0)[0]
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        if len(one_positions) == 0 or len(zero_positions) == 0:
            break
        rest_mask = np.ones(len(current_corr), dtype=bool)
        if top_k < len(current_corr):
            top_idx = np.argsort(current_corr)[-top_k:]
            rest_mask[top_idx] = False
            rest_max = float(np.max(current_corr[rest_mask]))
        else:
            top_idx = np.arange(len(current_corr))
            rest_max = 0.0
        corr_top = current_corr[top_idx]
        N_all = compute_neighbor_sums(h, top_idx)
        removal_score = np.max(corr_top + N_all, axis=1)
        one_scores = removal_score[one_positions]
        one_sorted = one_positions[np.argsort(one_scores)]
        addition_score = np.max(N_all, axis=1)
        zero_scores = addition_score[zero_positions]
        zero_sorted = zero_positions[np.argsort(zero_scores)[::-1]]
        M = min(40, len(one_sorted))
        Mz = min(40, len(zero_sorted))
        best_delta_max = current_max + 1.0
        best_i = -1
        best_j = -1
        for ii in range(M):
            i = one_sorted[ii]
            Ni = N_all[i]
            base = corr_top + Ni
            base_max = float(np.max(base))
            if base_max - 0.5 > best_delta_max:
                continue
            for jj in range(Mz):
                j = zero_sorted[jj]
                if i == j:
                    continue
                new_vals = base - N_all[j]
                new_max = max(float(np.max(new_vals)), rest_max)
                if new_max < best_delta_max:
                    best_delta_max = new_max
                    best_i = i
                    best_j = j
        if best_i >= 0 and best_delta_max < current_max - 1e-12:
            h[best_i] = 0.0
            h[best_j] = 1.0
            current_corr = compute_corr(h)
            current_max = float(np.max(current_corr))
            no_improve = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                best_corr = current_corr.copy()
        else:
            no_improve += 1
            if no_improve > 150:
                i_rand = one_positions[rng.randint(len(one_positions))]
                j_rand = zero_positions[rng.randint(len(zero_positions))]
                h[i_rand] = 0.0
                h[j_rand] = 1.0
                current_corr = compute_corr(h)
                current_max = float(np.max(current_corr))
                no_improve = 0
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
        one_positions = np.where(h == 1)[0]
        zero_positions = np.where(h == 0)[0]
    return best_h, best_max * dx

def continuous_refine(h_start, n, target, time_limit, alpha=200, max_steps=50000, mu=0.9, base_lr=0.08):
    start_t = time.time()
    h = h_start.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = _smooth_grad(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        accepted = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15 or c5_try < compute_c5(h) - 1e-15:
                h = h_try
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h.copy()
                accepted = True
                break
        if not accepted and step % 500 == 499:
            h_bin = (h > 0.5).astype(float)
            diff = target - np.sum(h_bin)
            if diff > 0:
                zidx = np.where(h_bin == 0)[0]
                h_bin_sorted = zidx[np.argsort(-h[zidx])]
                h_bin[h_bin_sorted[:int(diff)]] = 1.0
            elif diff < 0:
                oidx = np.where(h_bin == 1)[0]
                h_bin_sorted = oidx[np.argsort(h[oidx])]
                h_bin[h_bin_sorted[:int(abs(diff))]] = 0.0
            h = h_bin + 0.01 * np.random.randn(n)
            h = project_feasible(h, target)
            v = np.zeros(n)
    return best_h, best_c5

def pair_cd_improved(h_start, n, target, time_limit, max_sweeps=20000, n_cand=40, top_k=3):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 500:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_alpha = 80.0
        w_top = np.exp(soft_alpha * (corr[top_idx] - c_max))
        w_top = w_top / (w_top.sum() + 1e-30)
        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(25):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_binary_inits(n, target, rng):
    inits = []
    h1 = np.zeros(n)
    h1[:int(target)] = 1.0
    h1 = h1[rng.permutation(n)]
    inits.append(h1)
    block = max(1, n // 20)
    h2 = np.zeros(n)
    for s in range(0, n, 2 * block):
        end = min(s + block, n)
        h2[s:end] = 1.0
    h2 = h2[rng.permutation(n)]
    cur = np.sum(h2)
    if cur < target:
        zidx = np.where(h2 == 0)[0]
        h2[zidx[:int(target - cur)]] = 1.0
    elif cur > target:
        oidx = np.where(h2 == 1)[0]
        h2[oidx[:int(cur - target)]] = 0.0
    inits.append(h2)
    h3 = (rng.rand(n) < 0.5).astype(float)
    cur = np.sum(h3)
    if cur < target:
        zidx = np.where(h3 == 0)[0]
        h3[zidx[:int(target - cur)]] = 1.0
    elif cur > target:
        oidx = np.where(h3 == 1)[0]
        h3[oidx[:int(cur - target)]] = 0.0
    inits.append(h3)
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)

    # Phase 1: Binary local search from multiple starts
    if tleft() > 300:
        h_bin_start = (best_h > 0.5).astype(float)
        diff = target - np.sum(h_bin_start)
        if diff > 0:
            zidx = np.where(h_bin_start == 0)[0]
            h_bin_start[zidx[:int(diff)]] = 1.0
        elif diff < 0:
            oidx = np.where(h_bin_start == 1)[0]
            h_bin_start[oidx[:int(abs(diff))]] = 0.0
        per_start_time = min(80, tleft() / 5.0)
        best_bin = h_bin_start.copy()
        best_bin_c5 = compute_c5(h_bin_start)
        h_bin_best, c5_bin_best = binary_local_search(h_bin_start, target, per_start_time, max_iter=80000, rng=rng, top_k=5)
        if c5_bin_best < best_bin_c5:
            best_bin = h_bin_best
            best_bin_c5 = c5_bin_best
        cont_from_bin = project_feasible(best_bin, target)
        if compute_c5(cont_from_bin) < best_score:
            best_score = compute_c5(cont_from_bin)
            best_h = cont_from_bin
        # Additional binary restarts
        for restart in range(3):
            if tleft() < 100:
                break
            inits = structured_binary_inits(n, target, rng)
            for h_b in inits:
                if tleft() < 60:
                    break
                h_b_best, c5_b = binary_local_search(h_b, target, min(40, tleft() / 3.0), max_iter=40000, rng=rng, top_k=5)
                if c5_b < best_bin_c5:
                    best_bin = h_b_best
                    best_bin_c5 = c5_b
                    cont_from_bin = project_feasible(best_bin, target)
                    if compute_c5(cont_from_bin) < best_score:
                        best_score = compute_c5(cont_from_bin)
                        best_h = cont_from_bin

    # Phase 2: Continuous refinement from best binary
    if tleft() > 250:
        h_cont, c5_cont = continuous_refine(best_h, n, target,
                                            time_limit=min(120, tleft() * 0.4),
                                            alpha=200, max_steps=50000, mu=0.85, base_lr=0.06)
        if c5_cont < best_score:
            best_score = c5_cont
            best_h = h_cont
        # Try higher alpha
        if tleft() > 100:
            h_cont2, c5_cont2 = continuous_refine(best_h, n, target,
                                                  time_limit=min(60, tleft() * 0.3),
                                                  alpha=1000, max_steps=30000, mu=0.9, base_lr=0.03)
            if c5_cont2 < best_score:
                best_score = c5_cont2
                best_h = h_cont2

    # Phase 3: SLSQP polish
    if tleft() > 150:
        alphas = [100, 500, 2000]
        h_slq = slsqp_solve(best_h, n, target, alphas, 2000, 1e-16, min(120, tleft() * 0.5))
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 4: Pair CD
    if tleft() > 150:
        h_cd, c5_cd = pair_cd_improved(best_h, n, target,
                                       time_limit=min(150, tleft() * 0.4),
                                       max_sweeps=15000, n_cand=50, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP with high alpha
    if tleft() > 100:
        h_slq2 = slsqp_solve(best_h, n, target, [2000, 10000], 1500, 1e-16, min(80, tleft() * 0.5))
        sc2 = compute_c5(h_slq2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_slq2

    # Phase 6: Binary snap + local search + refine (explore new basin)
    if tleft() > 200:
        h_snap = (best_h > 0.5).astype(float)
        diff = target - np.sum(h_snap)
        if diff > 0:
            zidx = np.where(h_snap == 0)[0]
            h_snap[zidx[:int(diff)]] = 1.0
        elif diff < 0:
            oidx = np.where(h_snap == 1)[0]
            h_snap[oidx[:int(abs(diff))]] = 0.0
        # Perturb and re-search
        h_pert = h_snap.copy()
        n_pert = max(5, n // 20)
        pidx = rng.choice(n, size=n_pert, replace=False)
        h_pert[pidx] = 1.0 - h_pert[pidx]
        # Fix sum
        diff = target - np.sum(h_pert)
        if diff > 0:
            zidx = np.where(h_pert == 0)[0]
            h_pert[zidx[:int(diff)]] = 1.0
        elif diff < 0:
            oidx = np.where(h_pert == 1)[0]
            h_pert[oidx[:int(abs(diff))]] = 0.0
        h_b2, c5_b2 = binary_local_search(h_pert, target, min(60, tleft() / 3.0),
                                          max_iter=50000, rng=rng, top_k=5)
        if c5_b2 < best_score:
            best_score = c5_b2
            best_h = project_feasible(h_b2, target)
        if tleft() > 60:
            h_c3, c5_c3 = continuous_refine(best_h, n, target,
                                            time_limit=min(50, tleft() * 0.4),
                                            alpha=500, max_steps=30000, mu=0.85, base_lr=0.05)
            if c5_c3 < best_score:
                best_score = c5_c3
                best_h = h_c3

    # Phase 7: More pair CD with different params
    if tleft() > 100:
        h_cd2, c5_cd2 = pair_cd_improved(best_h, n, target,
                                         time_limit=min(80, tleft() * 0.4),
                                         max_sweeps=10000, n_cand=70, top_k=5)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Final SLSQP
    if tleft() > 40:
        h_final_slq = slsqp_solve(best_h, n, target, [5000, 20000], 1000, 1e-16,
                                  min(30, tleft() * 0.5))
        sc_final = compute_c5(h_final_slq)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_slq

    # Phase 9: Final pair CD
    if tleft() > 30:
        h_cd3, c5_cd3 = pair_cd_improved(best_h, n, target,
                                         time_limit=min(25, tleft() * 0.5),
                                         max_sweeps=5000, n_cand=50, top_k=3)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
