import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _power_sum_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Power sum approximation of max
    positive = np.maximum(corr, 0)
    return float(np.sum(positive**p)) / (2*len(x)-1)  # Scale to be comparable

def _smooth_logsumobj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_power_sum(h_start, n, target, powers, maxiter_per_p, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for p in powers:
        try:
            res = minimize(_power_sum_obj, h_cur, args=(p,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_p, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

|S ∩ (S+k)^c| = |S| - |S ∩ (S+k)| = m - |S ∩ (S+k)|

corr_k = sum_i h[i] * (1 - h[i+k])

So we want to minimize max_k (|S| - |S ∩ (S-k)|) = |S| - min_k |S ∩ (S-k)|.

To minimize this, we want to maximize min_k |S ∩ (S-k)|, i.e., make all the autocorrelations as large as possible.

For a "random" subset of size m from n=2m elements, the expected |S ∩ (S-k)| is m^2/n = m/2. So the expected overlap is m - m/2 = m/2.

The integral of the overlap is (m/2) * dx = (m/2) * (2/n) = m/n = 1/2. Wait, that's the average, not the maximum.

For the maximum, by the pigeonhole principle, there exists some k with |S ∩ (S-k)| >= average = m/2. So the maximum overlap is at most m - m/2 = m/2, giving C₅ <= m/2 * 2/n = m/n = 1/2.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=25, n_cand_end=80, max_no_improve=400,
                 bs_iters=12):
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bs_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * (2.0 / n)
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_cd(h_init, n, target, time_limit, eta_init=0.005,
                   max_iters=200000, tol=1e-15, active_frac=0.01):
    start = time.time()
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta_init
    no_improve = 0
    shift_list = list(range(-n + 1, n))

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 20000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = float(np.max(corr))
        c5_val = max_corr * (2.0 / n)

        if c5_val < best_c5 - tol:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        threshold = max_corr - active_frac * max_corr
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val - 1e-15:
            h = h_try
        else:
            h_try2 = h - 0.25 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val - 1e-15:
                h = h_try2
                eta *= 0.5
            else:
                eta *= 0.8

        if it > 100:
            eta = max(eta * 0.9997, 1e-7)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_local_search(h_init, n, target, time_limit, max_steps=50000,
                            base_scale=0.04):
    start = time.time()
    rng = np.random.RandomState(98765)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        scale = base_scale * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 7, 10]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

            h_ref = slsqp_round(h_pert, n0, target0,
                                [200, 1000, 5000], 500, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_final, c5_final = fast_pair_cd(best_h, n0, target0,
                                            time_limit=min(40, tleft() * 0.6),
                                            max_sweeps=10000,
                                            n_cand_start=40, n_cand_end=90,
                                            max_no_improve=200)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_final_pol = slsqp_round(best_h, n0, target0,
                                  [20000, 80000], 500, 1e-14)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
