    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            # (1-h[j+d]) for j = 0 to n-1-d
            grad[:n-d] += wk * (1.0 - h[d:])
            # -h[j-d] for j = d to n-1
            grad[d:] -= wk * h[:n-d]
        else:
            dd = -d
            # (1-h[j-d]) for j = dd to n-1
            grad[dd:] += wk * (1.0 - h[:n-dd])
            # -h[j+d] for j = 0 to n-1-dd
            grad[:n-dd] -= wk * h[dd:]
    
    grad *= dx
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()

    bounds = [(0.0, 1.0)] * n

    def make_ls(beta, penalty):
        def obj(h):
            return obj_ls(h, beta, target, penalty)
        def grad(h):
            return grad_ls(h, beta, target, penalty)
        return obj, grad

    # Phase 1: Continuous L-BFGS with increasing beta
    h_cont = best_h.copy()
    beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        if tleft() < 40:
            break
        obj_f, grad_f = make_ls(beta, 500.0)
        time_alloc = min(50, tleft() * 0.12)
        h_opt = run_lbfgs(h_cont, obj_f, grad_f, target, 300, time_alloc, start_time)
        sc = compute_c5(h_opt)
        h_cont = h_opt
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

    # Phase 2: Binary local search
    if tleft() > 60:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(80, tleft() * 0.2)
        h_ls, c5_ls = binary_swap_search(h_bin, target_int, time_alloc, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 3: Continuous refinement from best binary
    if tleft() > 50:
        h_cont2 = project_box_sum(best_h.copy(), target)
        for beta in [100, 300, 800, 2000]:
            if tleft() < 25:
                break
            obj_f, grad_f = make_ls(beta, 1000.0)
            time_alloc = min(35, tleft() * 0.15)
            h_opt = run_lbfgs(h_cont2, obj_f, grad_f, target, 250, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_cont2 = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 4: Top-k optimization
    if tleft() > 60:
        h_topk = project_box_sum(best_h.copy(), target)
        for k_top in [5, 10, 20, 50]:
            if tleft() < 30:
                break
            time_alloc = min(30, tleft() * 0.15)
            h_opt = run_lbfgs(h_topk,
                              lambda h: obj_topk(h, k_top, target, 800.0),
                              lambda h: grad_topk(h, k_top, target, 800.0),
                              target, 200, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_topk = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 5: SA from best binary
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.25)
        max_it = int(min(150000, time_alloc * 1500))
        h_sa, c5_sa = sa_binary(h_bin, target_int, max_it, 0.006, 0.00001,
                                rng, time_alloc, start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Try quadratic residue starting point
    if tleft() > 80:
        h_qr = quadratic_residue_init(n, rng)
        h_qr_proj = project_box_sum(h_qr, target)
        sc_qr = compute_c5(h_qr_proj)
        h_qr_bin = project_binary(h_qr, target_int)
        sc_qr_bin = compute_c5(h_qr_bin)
        h_best_qr = h_qr_proj if sc_qr < sc_qr_bin else h_qr_bin
        best_qr_c5 = min(sc_qr, sc_qr_bin)
        if tleft() > 50:
            for beta in [50, 200, 1000]:
                if tleft() < 25:
                    break
                obj_f, grad_f = make_ls(beta, 500.0)
                time_alloc = min(25, tleft() * 0.15)
                h_opt = run_lbfgs(h_best_qr, obj_f, grad_f, target, 200, time_alloc, start_time)
                sc = compute_c5(h_opt)
                h_best_qr = h_opt
                if sc < best_qr_c5:
                    best_qr_c5 = sc
        if best_qr_c5 < best_c5:
            best_c5 = best_qr_c5
            best_h = h_best_qr.copy()

    # Phase 7: Multi-restart from perturbed best
    if tleft() > 60:
        for restart in range(3):
            if tleft() < 30:
                break
            pert = rng.normal(0, 0.03, n)
            h_pert = project_box_sum(best_h + pert, target)
            for beta_r in [200, 800]:
                if tleft() < 20:
                    break
                obj_f, grad_f = make_ls(beta_r, 800.0)
                time_alloc = min(25, tleft() * 0.2)
                h_opt = run_lbfgs(h_pert, obj_f, grad_f, target, 200, time_alloc, start_time)
                sc = compute_c5(h_opt)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_opt.copy()

    # Phase 8: Try different n
    n_options = [n + 4, n + 8, n + 16, n + 32]
    for n_try in n_options:
        if tleft() < 50:
            break
        if n_try % 2 != 0:
            n_try += 1
        target_try = float(n_try / 2)
        target_try_int = n_try // 2
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_box_sum(h_interp, target_try)
        sc_i = compute_c5(h_interp)
        if sc_i < best_c5:
            best_c5 = sc_i
            best_h = h_interp.copy()
        if tleft() < 30:
            continue
        obj_f, grad_f = make_ls(200, 500.0)
        time_alloc = min(25, tleft() * 0.15)
        h_opt = run_lbfgs(h_interp, obj_f, grad_f, target_try, 200, time_alloc, start_time)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()
        h_bin_try = project_binary(h_opt, target_try_int)
        sc_b = compute_c5(h_bin_try)
        if sc_b < best_c5:
            best_c5 = sc_b
            best_h = h_bin_try.copy()
            time_alloc2 = min(20, tleft() * 0.1)
            h_ls2, c5_ls2 = binary_swap_search(h_bin_try, target_try_int, time_alloc2, start_time)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()

    # Phase 9: Final SA polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(50, tleft() - 10)
        max_it = int(min(80000, time_alloc * 2000))
        h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.003, 0.000005,
                                rng, time_alloc, start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 10: Final continuous polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_fin = project_box_sum(best_h.copy(), target_use)
        for beta_f in [500, 2000, 5000]:
            if tleft() < 10:
                break
            obj_f, grad_f = make_ls(beta_f, 2000.0)
            h_pol = run_lbfgs(h_fin, obj_f, grad_f, target_use, 100, min(15, tleft() * 0.4), start_time)
            h_fin = h_pol
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()

    # Final projection and check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
