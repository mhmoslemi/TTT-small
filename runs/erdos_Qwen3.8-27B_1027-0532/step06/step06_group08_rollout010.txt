def golden_ratio_init(n, rng):
    """Initialize using golden ratio spacing for near-uniform distribution."""
    h = np.zeros(n)
    n_ones = n // 2
    phi = (1 + np.sqrt(5)) / 2
    # Use golden ratio to place ones
    positions = np.sort(np.floor(np.arange(n_ones) * n / phi + rng.randn() * 0.5).astype(int) % n)
    # Handle duplicates
    seen = set()
    unique_pos = []
    for p in positions:
        if p not in seen:
            seen.add(p)
            unique_pos.append(p)
    # Fill remaining
    while len(unique_pos) < n_ones:
        p = rng.randint(n)
        if p not in seen:
            seen.add(p)
            unique_pos.append(p)
    h[np.array(unique_pos[:n_ones])] = 1.0
    return h

def reweight_greedy(h, max_iters, beta, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    n_lags = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        
        # Update weights
        corr_max = float(corr.max())
        weights = np.exp(beta * (corr - corr_max))
        weights /= weights.sum()
        # Pad to circular length n for FFT
        w_circ = np.zeros(n)
        w_circ[:n] = weights[:n]  # lags 0 to n-1
        W = np.fft.fft(w_circ)
        H = np.fft.fft(h)
        
        # Compute S[p] = sum_t w[t] * (h[p+t] + h[p-t]) approximately
        # A[p] = sum_t w[t] * h[(p+t) mod n] = ifft(fft(w_circ) * fft(h))
        A = np.fft.ifft(W * H).real
        # B[p] = sum_t w[t] * h[(p-t) mod n] = ifft(conj(fft(w_circ)) * fft(h))
        B = np.fft.ifft(np.conj(W) * H).real
        S = A + B
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Best swap: minimize S[i] - S[j], i.e., pick i with min S among ones, j with max S among zeros
        i = ones[np.argmin(S[ones])]
        j = zeros[np.argmax(S[zeros])]
        
        # Verify with actual delta
        delta = compute_delta(h, i, j)
        new_corr = corr + delta
        new_max = float(new_corr.max())
        if new_max < float(corr.max()):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            stall = 0

def compute_c5(h):

    dx = 2.0 / n

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def reweight_greedy(h, max_iters, beta, rng, time_limit, start_time):

        if time.time() - start_time > time_limit:

            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            corr_max = float(corr.max())
            weights = np.exp(beta * (corr - corr_max))
            weights /= weights.sum()

        T = compute_T(h, weights)

        ones = np.where(h == 1.0)[0]

                    found = True

            c5 = float(corr.max()) * dx

def momentum_subgradient(h, n_iters, alpha_init, alpha_growth, lr_init, lr_decay,
                         time_limit, start_time, momentum=0.85):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    v = np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break

        alpha = min(alpha_init * (alpha_growth ** (it / max(n_iters, 1))), 5000.0)
        lr = lr_init * (lr_decay ** (it / max(n_iters, 1)))

        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        exp_w = np.exp(shifted)
        wsum = float(exp_w.sum())
        if wsum < 1e-30:
            break
        W = exp_w / wsum

        grad = np.zeros(n)
        active = W > 1e-10
        for k_idx in np.where(active)[0]:
            wk = W[k_idx]
            s = shifts[k_idx]
            if s >= 0:
                if s < n:
                    grad[:n - s] += wk * (1.0 - h[s:])
                    grad[s:] += wk * (-h[:n - s])
            else:
                ak = -s
                if ak < n:
                    grad[ak:] += wk * (1.0 - h[:n - ak])
                    grad[:n - ak] += wk * (-h[ak:])

        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_box_sum(h_new, target)

        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        curr_c5 = float(corr.max()) * dx

        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            v *= 0.95

        h = h_new

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    def tleft():
        return budget_s - 8.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Reweighting greedy
    allowed = min(50.0, tleft() * 0.08)
    if allowed > 10:
        for beta in [3.0, 10.0, 30.0]:
            if tleft() < 15:
                break
            bg = max(5.0, allowed / 3.0 - 2.0)
            h_rg, c5_rg = reweight_greedy(best_h, max_iters=100000, beta=beta,
                                           rng=rng, time_limit=bg, start_time=start_time)
            if c5_rg < best_score:
                best_score = c5_rg
                best_h = h_rg.copy()

    # Phase 2: Main adaptive SA
    allowed = min(300.0, tleft() * 0.30)
    if allowed > 30:
        n_sa = int(min(12000000, allowed * 60000))
        h_sa, c5_sa = adaptive_sa(best_h, n_sa, 0.012, 0.00001, rng,
                                   time_limit=allowed, start_time=start_time, topk=4)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Continuous relaxation with momentum
    h_cont = project_box_sum(best_h.astype(float), float(target))
    allowed = min(120.0, tleft() * 0.15)
    if allowed > 20:
        n_iters = int(min(400000, allowed * 4000))
        h_ms, c5_ms = momentum_subgradient(h_cont, n_iters,
                                            alpha_init=15.0, alpha_growth=80.0,
                                            lr_init=0.015, lr_decay=0.005,
                                            time_limit=allowed, start_time=start_time)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = project_binary(h_ms, target)
        h_cont = project_box_sum(h_ms, float(target))

    # Phase 4: SA from continuous-informed binary
    h_bin_cont = project_binary(h_cont, target)
    c5_bc = compute_c5(h_bin_cont)
    if c5_bc < best_score:
        best_score = c5_bc
        best_h = h_bin_cont

    allowed = min(200.0, tleft() * 0.20)
    if allowed > 30:
        n_sa = int(min(10000000, allowed * 60000))
        h_sa2, c5_sa2 = adaptive_sa(best_h, n_sa, 0.008, 0.000005, rng,
                                     time_limit=allowed, start_time=start_time, topk=5)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Diverse initializations
    for init_fn in [golden_ratio_init, random_binary_init]:
        if tleft() < 40:
            break
        allowed = min(80.0, tleft() * 0.10)
        if allowed < 20:
            continue
        h_try = init_fn(n, rng)
        n_sa = int(min(500000, allowed * 30000))
        h_try, c5_try = adaptive_sa(h_try, n_sa, 0.02, 0.00005, rng,
                                     time_limit=allowed, start_time=start_time, topk=3)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: SLSQP polish on continuous
    if tleft() > 40:
        h_cont = project_box_sum(best_h.astype(float), float(target))
        allowed = min(80.0, tleft() * 0.12)
        if allowed > 15:
            h_ref, sc_ref = slsqp_refine(h_cont, n, float(target),
                                          alphas=[50, 200, 1000],
                                          maxiter=2000, ftol=1e-13,
                                          time_limit=allowed, start_time=start_time)
            sc_bin = compute_c5(project_binary(h_ref, target))
            if sc_ref < best_score:
                best_score = sc_ref
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = project_binary(h_ref, target)

    # Phase 7: Final intensive SA with restart
    if tleft() > 60:
        allowed = min(tleft() - 20, 150.0)
        n_sa = int(min(10000000, allowed * 50000))
        h_fin, c5_fin = adaptive_sa(best_h, n_sa, 0.006, 0.000002, rng,
                                     time_limit=allowed, start_time=start_time, topk=5)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    # Phase 8: Final greedy polish
    if tleft() > 15:
        allowed = min(25.0, tleft() - 8)
        h_pol, c5_pol = reweight_greedy(best_h, max_iters=50000, beta=10.0,
                                         rng=rng, time_limit=allowed, start_time=start_time)
        if c5_pol < best_score:
            best_score = c5_pol
            best_h = h_pol.copy()

    final_h = project_binary(best_h, target)
    final_score = compute_c5(final_h)
    return final_h, final_score, n
