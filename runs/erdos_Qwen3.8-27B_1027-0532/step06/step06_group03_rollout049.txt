    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

    _global_target = target
    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    start_t = time.time()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

            break
        try:
            res = minimize(soft_max_objective, h_cur, args=(alpha,),
                          jac=soft_max_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)

                best_h = h_new.copy()
            if c5_new < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_V(h):
    n = len(h)
    k_vals = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    V = np.zeros((n, n_corr))
    for p in range(n):
        idx1 = p - k_vals
        m = (idx1 >= 0) & (idx1 < n)
        V[p, m] += h[idx1[m]]
        idx2 = p + k_vals
        m2 = (idx2 >= 0) & (idx2 < n)
        V[p, m2] += h[idx2[m2]]
    return V

def soft_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

def slsqp_run(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _gn, _gt
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(soft_max_obj, h_cur, args=(alpha,),
                          jac=soft_max_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def vectorized_pair_cd(h_start, n, target, time_limit, n_cand, n_bisect, stall_limit):
    start_t = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    dx = 2.0 / n
    no_improve = 0

    while time.time() - start_t < time_limit and no_improve < stall_limit:
        corr = get_corr(h)
        c5_val = float(np.max(corr)) * dx

        else:
            no_improve += 1

        w = w / w.sum()

        mid = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        pairs_i = []
        pairs_j = []
        for ii in high_set:
            for jj in low_set:
                if ii != jj:
                    pairs_i.append(ii)
                    pairs_j.append(jj)

        if len(pairs_i) == 0:
            break

        pairs_i = np.array(pairs_i, dtype=np.intp)
        pairs_j = np.array(pairs_j, dtype=np.intp)
        n_pairs = len(pairs_i)

        delta_max = np.minimum(h[pairs_i], 1.0 - h[pairs_j])
        valid = delta_max > 1e-10
        if not np.any(valid):
            no_improve += 50
            continue
        pairs_i = pairs_i[valid]
        pairs_j = pairs_j[valid]
        delta_max = delta_max[valid]
        n_pairs = len(pairs_i)
        if n_pairs == 0:
            break

        V = compute_V(h)

        t_hi = np.ones(n_pairs)
        new_corr = np.empty((n_pairs, n_corr))

            t_mid = 0.5 * (t_lo + t_hi)
            delta_att = t_mid * delta_max

            new_corr += corr[None, :]

            diff_masked = np.where(active, diff_mat, -np.inf)
            deriv = np.max(diff_masked, axis=1)
            inc = deriv > 0
            t_hi[inc] = t_mid[inc]
            t_lo[~inc] = t_mid[~inc]

        t_opt = 0.5 * (t_lo + t_hi)
        delta_opt = t_opt * delta_max
        np.multiply(delta_opt[:, None], diff_mat, out=new_corr)
        new_corr += corr[None, :]
        final_maxes = np.max(new_corr, axis=1)

        best_p = np.argmin(final_maxes)
        best_trial = final_maxes[best_p] * dx

        if best_trial < best_c5 - 1e-13:
            i = int(pairs_i[best_p])
            j = int(pairs_j[best_p])
            delta = delta_opt[best_p]
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def momentum_search(h_start, n, target, time_limit, alpha, max_steps, mu, base_lr):
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = soft_max_grad(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.02 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h) - 1e-14:
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
    return best_h, best_c5

def stochastic_perturb_cd(h_start, n, target, time_limit, rng, n_cd_cand):
