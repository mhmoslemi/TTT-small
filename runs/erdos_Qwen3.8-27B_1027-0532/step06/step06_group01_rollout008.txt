- Reducing n_cand to 15: O(15*15*10*1200) = 2.7M ops (3x faster)
- Or using a smarter approach: for each candidate i, find the best j without exhaustive search

Actually, for a given candidate i (from the "high" set), the optimal j is the one that maximizes the improvement. The improvement from moving delta from i to j is:
improve(delta, i, j) = max(corr) - max(corr + delta * (profile_i - profile_j))

For small delta, this is approximately delta * (max(corr) - max(corr + (profile_i - profile_j)))
≈ delta * (corr[k*] - (corr[k*] + (profile_i[k*] - profile_j[k*])))
= delta * (profile_j[k*] - profile_i[k*])

So for small moves, the best j is the one with the largest profile_j[k*] - profile_i[k*]. Since profile_i[k*] is fixed for a given i, we just need the j with the largest profile_j[k*].

This means we can find the best j for each i in O(n_cand) instead of O(n_cand * binary_search). Then we only do the binary search for the top few candidates.

Actually, let me just write a clean, efficient solution and not overthink it. The key improvements are:
1. Faster CD (smaller candidate set, fewer binary search iterations)
2. More efficient use of time (spend more time on CD, less on SLSQP)
3. Restart strategy
4. Higher resolution at the end

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, alpha)
    n_corr = len(corr)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    nonzero = np.where(w > 1e-10)[0]
    for m in nonzero:
        k = int(m) - mid
        wi = float(w[m])
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    return grad, corr

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * (2.0 / n)
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit, n_cand=20, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        ks = np.arange(-(n - 1), n)
        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            pi = profiles[ci]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = pi - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-13)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-13:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def proj_grad_bb(h_init, n, target, time_limit, max_iters, alpha, top_k=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    prev_grad = None
    prev_step = None
    bb_num = 0.0
    bb_den = 0.0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

        w = smooth_max_weights(corr, alpha)
        grad = np.zeros(n)
        nonzero = np.where(w > 1e-10)[0]
        for m in nonzero:
            k = int(m) - mid
            wi = float(w[m])
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        if prev_grad is not None:
            s = h - prev_step
            y = grad - prev_grad
            sy = float(np.dot(s, y))
            ss = float(np.dot(s, s))
            if sy > 1e-16:
                bb_num = float(np.dot(s, s))
                bb_den = sy
        if bb_den > 1e-16:
            lr = bb_num / bb_den
            lr = min(lr, 5.0)
        else:
            lr = 0.5

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = lr * lr_mult
            h_try = h - lr_try * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                prev_grad = grad.copy()
                prev_step = h.copy()
                h = h_try
                improved = True
                break
        if not improved and it > 200:
            break

    return best_h, best_c5

def multi_restart_optimize(h_init, n, target, time_limit, rng, n_restarts=3):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    per_restart_time = time_limit / (n_restarts + 1)
    start = time.time()

    for r in range(n_restarts):
        if time.time() - start > time_limit - 30:
            break
        sigma = 0.02 / (r + 1)
        h_pert = h_init + rng.normal(0, sigma, n)
        h_pert = project_feasible(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()

        remaining = time_limit - (time.time() - start) - 20
        if remaining < 30:
            break

        h_cd, c5_cd = fast_pair_cd(h_pert, n, target, time_limit=min(60, remaining * 0.4),
                                    n_cand=15, max_sweeps=30000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        remaining = time_limit - (time.time() - start) - 20
        if remaining < 20:
            break

        h_h, c5_h = proj_grad_bb(best_h, n, target, time_limit=min(40, remaining * 0.4),
                                  max_iters=5000, alpha=5000, top_k=5)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h_h.copy()

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
