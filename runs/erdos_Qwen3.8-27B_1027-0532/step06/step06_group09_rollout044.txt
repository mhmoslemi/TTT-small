import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = weights.sum()
    if total_w < 1e-300:
        return c_max * dx, np.zeros(n)
    weights = weights / total_w
    obj = (c_max + np.log(total_w) / alpha) * dx
    grad = np.zeros(n)
    mid = n - 1
    for s in range(-(n - 1), n):
        w = weights[s + mid]
        if w < 1e-14:
            continue
        if s >= 0:
            if n - s > 0:
                j1 = np.arange(0, n - s)
                grad[j1] += w * (1.0 - h[j1 + s])
            if s < n:
                j2 = np.arange(s, n)
                grad[j2] -= w * h[j2 - s]
        else:
            t = -s
            j1 = np.arange(t, n)
            grad[j1] += w * (1.0 - h[j1 - t])
            if n - t > 0:
                j2 = np.arange(0, n - t)
                grad[j2] -= w * h[j2 + t]
    grad *= dx
    return float(obj), grad


def lse_wrap(h, alpha):
    return lse_obj_and_grad(h, alpha)[0]


def lse_grad_wrap(h, alpha):
    return lse_obj_and_grad(h, alpha)[1]


def lbfags_optimize(h_init, n, target, alpha, maxiter=5000, ftol=1e-16):
    def obj(x):
        return lse_obj_and_grad(x, alpha)[0]

    def grad(x):
        return lse_obj_and_grad(x, alpha)[1]

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_init.astype(float), jac=grad, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h_init


def slsqp_optimize(h_init, n, target, alpha, maxiter=2000):
    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(lse_obj_and_grad, h_init.astype(float), args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h_init


def pair_cd(h_init, n, target, time_limit, max_sweeps=5000,
            n_cand=40, patience=100):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            h_i_sp = np.zeros(n_corr)
            ip = (i + shifts >= 0) & (i + shifts < n)
            h_i_sp[ip] = h[i + shifts[ip]]
            h_i_sm = np.zeros(n_corr)
            im = (i - shifts >= 0) & (i - shifts < n)
            h_i_sm[im] = h[i - shifts[im]]

            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm_arr = np.zeros(n_corr)
                jpm_arr[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm_arr = np.zeros(n_corr)
                jsm_arr[jm] = h[j - shifts[jm]]
                g = h_i_sp + h_i_sm - jpm_arr - jsm_arr
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = max(b_d, delta_lo + 1e-10)
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def mirror_descent_nesterov(h_init, n, target, time_limit, max_steps=20000,
                            mu=0.75, base_lr=0.05, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        lr = base_lr * max(0.05, 1.0 - step / (max_steps * 1.2))

        improved = False
        for attempt in range(10):
            h_try = project_feasible(h - lr * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            lr *= 0.5

        if not improved:
            base_lr *= 0.95
            if base_lr < 0.001:
                break

        exp_arg = np.clip(2.0 * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.6 * lam + 0.4 * lam_new

    return best_h, best_c5


def binary_snap(h, n, target, threshold=0.05):
    h_snap = h.copy()
    h_snap[h_snap < threshold] = 0.0
    h_snap[h_snap > 1.0 - threshold] = 1.0
    h_snap = project_feasible(h_snap, target)
    return h_snap


def multi_restart_search(best_h, n, target, time_limit, seed, max_restarts=5):
    start = time.time()
    rng = np.random.RandomState(seed)
    best = best_h.copy()
    best_c5 = compute_c5(best)

    for r in range(max_restarts):
        if time.time() - start > time_limit:
            break
        noise = 0.1 + 0.1 * r
        h_pert = best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        t_part = max(20, (time_limit - (time.time() - start)) / (max_restarts - r))
        h_ref = lbfags_optimize(h_pert, n, target, 2000, maxiter=800)
        sc = compute_c5(h_ref)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best = h_ref
            h_cd, c5_cd = pair_cd(best, n, target,
                                  time_limit=min(30, t_part * 0.3),
                                  max_sweeps=2000, n_cand=40, patience=80)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best = h_cd

    return best, best_c5


def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    for alpha in alphas:
        if tleft() < 60:
            break
        t_alloc = min(80, tleft() * 0.15)
        maxit = int(max(500, t_alloc * 20))
        h_try = lbfags_optimize(best_h, n, target, alpha, maxiter=maxit,
                                ftol=1e-16)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try
        # Quick pair CD between LSE iterations
        if tleft() > 80:
            h_cd, c5_cd = pair_cd(best_h, n, target,
                                  time_limit=min(15, tleft() * 0.05),
                                  max_sweeps=500, n_cand=30, patience=50)
            if c5_cd < best_c5 - 1e-14:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 2: Extended pair CD
    if tleft() > 150:
        cd_time = min(300, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=10000, n_cand=40, patience=200)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: Mirror descent with Nesterov
    if tleft() > 100:
        md_time = min(100, tleft() * 0.15)
        h_md, c5_md = mirror_descent_nesterov(best_h, n, target,
                                              time_limit=md_time,
                                              max_steps=30000, mu=0.75,
                                              base_lr=0.04, m_top=12)
        if c5_md < best_c5 - 1e-14:
            best_c5 = c5_md
            best_h = h_md
        # Polish after mirror descent
        if tleft() > 50:
            h_ref = lbfags_optimize(best_h, n, target, 10000, maxiter=2000)
            sc = compute_c5(h_ref)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_ref

    # Phase 4: Binary snap attempt
    if tleft() > 80:
        for thresh in [0.03, 0.05, 0.08, 0.12]:
            if tleft() < 40:
                break
            h_snap = binary_snap(best_h, n, target, threshold=thresh)
            # Re-optimize from snapped starting point
            h_ref = lbfags_optimize(h_snap, n, target, 5000, maxiter=1500)
            sc = compute_c5(h_ref)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_ref
                break

    # Phase 5: Second extended pair CD
    if tleft() > 120:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=8000, n_cand=50, patience=150)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 6: Higher alpha LSE polishing
    if tleft() > 80:
        for alpha in [50000, 100000, 300000]:
            if tleft() < 30:
                break
            h_try = lbfags_optimize(best_h, n, target, alpha, maxiter=1500,
                                    ftol=1e-16)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try

    # Phase 7: Multi-restart
    if tleft() > 100:
        mr_time = min(100, tleft() * 0.2)
        best_h, best_c5 = multi_restart_search(best_h, n, target,
                                               time_limit=mr_time,
                                               seed=seed, max_restarts=4)

    # Phase 8: SLSQP for final polish
    if tleft() > 50:
        for alpha in [50000, 200000]:
            if tleft() < 25:
                break
            h_try = slsqp_optimize(best_h, n, target, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try

    # Phase 9: Try higher resolution
    if tleft() > 150:
        for n_try in [500, 600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to_n(best_h, n_try, target_try)
            h_hires = lbfags_optimize(h_interp, n_try, target_try, 3000, maxiter=1500)
            sc = compute_c5(h_hires)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h_hires
                n = n_try
                target = target_try
                # Refine at higher res
                if tleft() > 60:
                    h_hires = lbfags_optimize(best_h, n, target, 20000, maxiter=1000)
                    sc2 = compute_c5(h_hires)
                    if sc2 < best_c5 - 1e-12:
                        best_c5 = sc2
                        best_h = h_hires
                if tleft() > 60:
                    h_cd2, c5_cd2 = pair_cd(best_h, n, target,
                                            time_limit=min(40, tleft() * 0.2),
                                            max_sweeps=3000, n_cand=40,
                                            patience=100)
                    if c5_cd2 < best_c5 - 1e-12:
                        best_c5 = c5_cd2
                        best_h = h_cd2

    # Phase 10: Final pair CD + LSE
    if tleft() > 40:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=min(20, tleft() * 0.3),
                              max_sweeps=3000, n_cand=50, patience=60)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd
    if tleft() > 20:
        h_final = lbfags_optimize(best_h, n, target, 100000, maxiter=500)
        sc = compute_c5(h_final)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    h_out = project_feasible(h_out, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_use
