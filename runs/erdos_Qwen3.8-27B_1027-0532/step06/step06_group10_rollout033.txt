    ws = w.sum()
    if ws < 1e-300:

        w[int(np.argmax(corr))] = 1.0

        w = w / ws

    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    no_improve = 0
    
    for sweep in range(50000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break
        
        # Multi-shift weighted gradient
        c_max = np.max(corr)
        top_k = np.argsort(corr)[::-1][:n_top_shifts]
        w = np.exp(50.0 * (corr[top_k] - c_max))
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Find best pair to transfer
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                # Compute the effect on all correlations
                g = np.zeros(2*n-1)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)

            h[best_i] -= best_delta
            h[best_j] += best_delta

            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_val_alpha(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _smooth_grad_alpha(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def _eq_fun(x, target):
    return np.sum(x) - target

def _eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        res = minimize(
            _smooth_val_alpha, h, args=(alpha,),
            jac=_smooth_grad_alpha,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'args': (target,),
                          'jac': _eq_jac, 'args2': (n,)}],
            options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def multi_shift_cd(h0, n, target, time_limit, n_top=5, n_cand=15, bs_iters=18):
    start = time.time()
    deadline = start + time_limit
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    no_improve = 0
    for sweep in range(100000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            break
        c_max = np.max(corr)
        top_k = np.argsort(corr)[::-1][:n_top]
        exp_w = np.exp(60.0 * (corr[top_k] - c_max))
        w = exp_w / exp_w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(2 * n - 1)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(bs_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active=20, max_rounds=40):
    start = time.time()
    deadline = start + time_limit
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.CVXOPT, cp.SCS]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_h(h, target, seed, scale):
    rng = np.random.RandomState(seed)
    noise = rng.normal(0.0, scale, len(h))
    return project_feasible(h + noise, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at base resolution
    if time.time() < deadline - 400:
        dl1 = min(start_time + 120.0, deadline - 350.0)
        h1, s1 = slsqp_optimize(best_h, n, target,
                                 alphas=[30, 100, 500, 2000, 10000, 50000],
                                 maxiter=2000, ftol=1e-14, deadline=dl1)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Multi-shift CD at base resolution
    if time.time() < deadline - 350:
        cd_t = min(120.0, deadline - start_time - 300.0)
        h2, s2 = multi_shift_cd(best_h, n, target,
                                 time_limit=cd_t, n_top=5, n_cand=15, bs_iters=18)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Perturbation and re-optimization
    n_perturbations = 3
    for p in range(n_perturbations):
        if time.time() > deadline - 300:
            break
        scale = 0.015 + 0.01 * p
        h_pert = perturb_h(best_h, target, seed + 100 + p, scale)
        sl_dl = min(start_time + (130 + 40 * p), deadline - (280 - 30 * p))
        h_p, s_p = slsqp_optimize(h_pert, n, target,
                                   alphas=[100, 500, 2000, 10000],
                                   maxiter=1000, ftol=1e-13, deadline=sl_dl)
        if s_p < best_score - 1e-6:
            cd_dl2 = min(start_time + (170 + 40 * p), deadline - (250 - 30 * p))
            h_pc, s_pc = multi_shift_cd(h_p, n, target,
                                         time_limit=max(5.0, cd_dl2 - time.time()),
                                         n_top=5, n_cand=12, bs_iters=15)
            if s_pc < best_score:
                best_score = s_pc
                best_h = h_pc
            elif s_p < best_score:
                best_score = s_p
                best_h = h_p

    # Phase 4: Upsample to n=800
    n_up = 800
    target_up = float(n_up) / 2.0
    if time.time() < deadline - 250:
        h_up = interpolate_to(best_h, n_up, target_up)
        sl_dl = min(start_time + 350.0, deadline - 200.0)
        h_up, s_up = slsqp_optimize(h_up, n_up, target_up,
                                     alphas=[30, 100, 500, 2000, 10000],
                                     maxiter=1500, ftol=1e-14, deadline=sl_dl)
        cd_dl3 = min(start_time + 450.0, deadline - 150.0)
        h_up2, s_up2 = multi_shift_cd(h_up, n_up, target_up,
                                       time_limit=max(5.0, cd_dl3 - time.time()),
                                       n_top=5, n_cand=12, bs_iters=15)
        if s_up2 < best_score:
            best_score = s_up2
            best_h = h_up2
        elif s_up < best_score:
            best_score = s_up
            best_h = h_up

    # Phase 5: Try n=1200
    n_up2 = 1200
    target_up2 = float(n_up2) / 2.0
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        h_up2 = interpolate_to(best_h, n_up2, target_up2)
        sl_dl2 = min(start_time + 580.0, deadline - 120.0)
        h_up2, s_up3 = slsqp_optimize(h_up2, n_up2, target_up2,
                                       alphas=[50, 200, 1000, 5000],
                                       maxiter=1000, ftol=1e-13, deadline=sl_dl2)
        cd_dl4 = min(start_time + 680.0, deadline - 80.0)
        h_up3, s_up4 = multi_shift_cd(h_up2, n_up2, target_up2,
                                       time_limit=max(5.0, cd_dl4 - time.time()),
                                       n_top=5, n_cand=10, bs_iters=12)
        if s_up4 < best_score:
            best_score = s_up4
            best_h = h_up3
        elif s_up3 < best_score:
            best_score = s_up3
            best_h = h_up2

    # Phase 6: LP refinement at best resolution
    if time.time() < deadline - 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        lp_tl = min(60.0, deadline - time.time() - 30.0)
        h_lp, s_lp = lp_refine(best_h, n_use, target_use,
                                time_limit=lp_tl, n_active=25, max_rounds=30)
        if s_lp < best_score:
            best_score = s_lp
            best_h = h_lp

    # Phase 7: Final CD polish
    if time.time() < deadline - 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        cd_tl = min(30.0, deadline - time.time() - 10.0)
        h_cd, s_cd = multi_shift_cd(best_h, n_use, target_use,
                                     time_limit=cd_tl, n_top=5, n_cand=12, bs_iters=14)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
