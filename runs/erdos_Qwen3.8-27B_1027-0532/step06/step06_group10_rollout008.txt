def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5_and_corr(h):

def multi_shift_gradient(h, corr, n, n_top=3):
    idxs = np.arange(n)
    mid = n - 1
    top_shifts = np.argsort(corr)[::-1][:n_top]
    grad = np.zeros(n)
    weights = np.zeros(n_top)
    c_max = np.max(corr)
    for t, s_idx in enumerate(top_shifts):
        weights[t] = np.exp(20.0 * (corr[s_idx] - c_max))
    weights /= weights.sum()
    for t, s_idx in enumerate(top_shifts):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[t] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[t] * h[idxs[v2] + k]

def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    idxs = np.arange(n)
    mid = n - 1

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break

        def _obj(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
            return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

        def _grad(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros_like(w)
                w[int(np.argmax(corr))] = 1.0
            else:
                w = w / ws
            grad = np.zeros(n)
            active = np.where(w > 1e-6)[0]
            for s in active:
                k = int(s) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
                grad[v2] -= w[s] * x[idxs[v2] + k]
            grad *= dx
            grad -= np.mean(grad)
            return grad

        def _con(x):
            return np.sum(x) - target

        def _con_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
