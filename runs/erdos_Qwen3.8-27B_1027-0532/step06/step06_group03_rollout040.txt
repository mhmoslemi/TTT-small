            top_act = np.argsort(corr)[-10:]
            active = top_act

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        n_step = int(min(n // 10, 50))
        sorted_idx = np.argsort(grad)
        targets = sorted_idx[:n_step]
        sources = sorted_idx[-n_step:]

        total_to_move = 0.0
        for s in sources:
            total_to_move += h[s]
        capacity = 0.0
        for t in targets:
            capacity += (1.0 - h[t])
        move_amt = min(total_to_move, capacity) * 0.3

        if move_amt > 1e-15:
            src_weights = h[sources]
            src_weights = src_weights / max(src_weights.sum(), 1e-30)
            tgt_weights = (1.0 - h[targets])
            tgt_weights = tgt_weights / max(tgt_weights.sum(), 1e-30)

            for mu in [1.0, 0.7, 0.5, 0.3, 0.1]:
                h_try = h.copy()
                h_try[sources] -= move_amt * mu * src_weights
                h_try[targets] += move_amt * mu * tgt_weights
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_round(h, n, target):
    h_b = np.where(h > 0.5, 1.0, 0.0)
    h_b = project_feasible(h_b, target)
    return h_b


def local_perturb_search(h_init, n, target, max_tries, time_limit, seed=12345):
    start_t = time.time()
    rng_local = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5]))
        n_pert = min(n_pert, n)
        idxs_p = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.02, 0.05, 0.1]))
        h_try = h_cur.copy()
        h_try[idxs_p] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def momentum_cd(h_start, n, target, time_limit, alpha, mu, base_lr, max_steps):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        g_norm = np.linalg.norm(v)
        if g_norm < 1e-15:
            break
        v = v / g_norm
        lr = base_lr / (1.0 + 0.05 * step)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
            elif c5_try < compute_c5(h) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            v *= 0.5

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick pair CD at current resolution to get immediate gains
    if tleft() > 300:
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=min(80, tleft() * 0.15),
                                   max_sweeps=5000, n_cand=40, top_k=3,
                                   use_exact_step=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: SLSQP polish at current resolution
    if tleft() > 250:
        _global_n = n
        _global_target = target
        h_sls = slsqp_opt(best_h, n, target, [100, 500, 2000],
                          1000, 1e-15, min(60, tleft() * 0.15))
        sc = compute_c5(h_sls)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 3: Try higher resolution n=600
    n_hires = 600
    target_hires = n_hires / 2.0
    if tleft() > 500:
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        h_hr = slsqp_opt(h_interp, n_hires, target_hires,
                         [50, 200, 1000],
                         800, 1e-14, min(80, tleft() * 0.15))
        sc_hr = compute_c5(h_hr)
        if sc_hr < best_score:
            best_score = sc_hr
            best_h = h_hr
            n = n_hires
            target = target_hires

    # Phase 4: Aggressive pair CD at current (possibly higher) resolution
    if tleft() > 300:
        cd_time = min(400, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=20000, n_cand=45, top_k=3,
                                   use_exact_step=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Active set CD
    if tleft() > 200:
        h_as, c5_as = active_set_cd(best_h, n, target,
                                    time_limit=min(80, tleft() * 0.2),
                                    max_sweeps=5000, top_k=3)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 6: Another round of pair CD with different parameters
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=15000, n_cand=60, top_k=5,
                                   use_exact_step=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Momentum-based continuous descent
    if tleft() > 150:
        for alpha_m in [200, 1000]:
            if tleft() < 60:
                break
            h_mom, c5_mom = momentum_cd(best_h, n, target,
                                        time_limit=min(40, tleft() * 0.2),
                                        alpha=alpha_m, mu=0.8, base_lr=0.03,
                                        max_steps=20000)
            if c5_mom < best_score:
                best_score = c5_mom
                best_h = h_mom

    # Phase 8: Try even higher resolution if time allows
    n_hires2 = 800
    target_hires2 = n_hires2 / 2.0
    if tleft() > 400 and n < n_hires2:
        h_interp2 = interpolate_to(best_h, n_hires2, target_hires2)
        h_hr2 = slsqp_opt(h_interp2, n_hires2, target_hires2,
                          [100, 500],
                          600, 1e-14, min(60, tleft() * 0.2))
        sc_hr2 = compute_c5(h_hr2)
        if sc_hr2 < best_score:
            best_score = sc_hr2
            best_h = h_hr2
            n = n_hires2
            target = target_hires2

        if tleft() > 200:
            h_cd2, c5_cd2 = fast_pair_cd(best_h, n, target,
                                         time_limit=min(150, tleft() * 0.3),
                                         max_sweeps=10000, n_cand=40, top_k=3,
                                         use_exact_step=True)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 9: SLSQP with high alpha for fine polishing
    if tleft() > 100:
        h_pol = slsqp_opt(best_h, n, target, [5000, 20000],
                          800, 1e-16, min(50, tleft() * 0.3))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Final pair CD sweep
    if tleft() > 60:
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=min(40, tleft() * 0.4),
                                   max_sweeps=5000, n_cand=50, top_k=4,
                                   use_exact_step=True)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Local perturbation to escape potential plateaus
    if tleft() > 50:
        h_pert, c5_pert = local_perturb_search(best_h, n, target,
                                               max_tries=2000,
                                               time_limit=min(20, tleft() * 0.3),
                                               seed=99999)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if tleft() > 30:
            h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                       time_limit=min(15, tleft() * 0.3),
                                       max_sweeps=3000, n_cand=40, top_k=3,
                                       use_exact_step=True)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 25:
        h_final = slsqp_opt(best_h, n, target, [10000, 50000],
                            600, 1e-16, min(15, tleft() * 0.4))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
