from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    return c_max + np.log(w_sum) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-14:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return grad

    value = c_max + np.log(w_sum) / alpha

def adam_optimize(h0, target_sum, time_limit, start_time, alphas, lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = smooth_obj(h, alpha)
        
        while True:
            if time.time() - start_time > time_limit:
                break
            t += 1
            val, grad = obj_and_grad(h, alpha)
            if val < local_best - 1e-10:

                stall = 0
            else:
