    """Project h onto {sum(h)=target, 0<=h<=1}"""
    # Binary search on lambda for: h_i = clip(h_i - lambda, 0, 1)
    lo, hi = -max(h) - 1, max(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Final adjustment
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h_proj < 1.0)[0]
        else:
            idx = np.where(h_proj > 0.0)[0]
        if len(idx) > 0:
            h_proj[idx] += residual / len(idx)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, target, rng):
    """Convert h to binary with exactly target ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != int(target):
        if n_ones > int(target):
            ones = np.where(h_bin == 1)[0]
            # Remove the ones with lowest "value" (lowest h value among ones)
            order = np.argsort(-h[ones])  # most confident ones first
            h_bin[ones[order[int(target):]]] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            # Add the zeros with highest h value
            order = np.argsort(-h[zeros])
            h_bin[zeros[order[:int(target)-n_ones]]] = 1
    return h_bin

def local_search(h_init, n, target, time_limit, rng, max_iter=500000):
    """Local search with simulated annealing on binary solutions."""
    start = time.time()
    h = to_binary(h_init, n, target, rng)
    dx = 2.0 / n
    
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    current_max = float(np.max(overlaps))
    best_h = h.copy()
    best_max = current_max
    
    T = 0.001
    T_min = 0.00001
    cooling = 0.99999
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if T < T_min:
            break
        
        # Choose a random move
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:

        h[a] = 0
        h[b] = 1
        new_overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
        new_max = float(np.max(new_overlaps))

def smooth_max_overlaps(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - n_half
        abs_lag = abs(lag)
        if abs_lag == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:]) * dx
            grad[lag:] += w * (-h[:n - lag]) * dx
        else:
            m2 = -lag
            grad[m2:] += w * (1.0 - h[:n - m2]) * dx
            grad[:n - m2] += w * (-h[m2:]) * dx
    return grad

def l_bfgsb_optimize(h_start, n, target, time_limit, alphas, maxiter=2000):
    """Optimize using L-BFGS-B with augmented Lagrangian for the sum constraint."""
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            def obj(x):
                return smooth_max_overlaps(x, alpha)
            
            def grad(x):
                return smooth_max_grad(x, alpha)
            
            # Use penalty for sum constraint
            penalty = 100.0
            
            def obj_pen(x):
                sum_pen = (np.sum(x) - target) ** 2 * penalty
                return smooth_max_overlaps(x, alpha) + sum_pen
            
            def grad_pen(x):
                return smooth_max_grad(x, alpha) + 2 * penalty * (np.sum(x) - target) * np.ones(n)
            
            res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_score

def subgradient_maxmin(h_init, n, target, time_limit, K=50):
    """Subgradient method for max-min autocorrelation (equivalent to min max overlap)."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    
    for t in range(1000000):
        if time.time() - start > time_limit:
            break
        
        # Compute overlaps
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        n_half = n - 1
        
        # Active set: top K overlaps
        top_idx = np.argsort(corr)[::-1][:K]
        
        # Compute subgradient (direction to decrease max overlap)
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n - k] += (1.0 - h[k:]) * dx
                g[k:] += (-h[:n - k]) * dx
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m]) * dx
                g[:n - m] += (-h[m:]) * dx
        g /= K
        
        # FW step: find binary v minimizing g·v (i.e., set v[i]=1 where g[i] is smallest)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        
        # Update
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        
        h = h_new
        
        if t > 5000 and t - best_t > 50000:
            break
    
    return best_h, best_c5
