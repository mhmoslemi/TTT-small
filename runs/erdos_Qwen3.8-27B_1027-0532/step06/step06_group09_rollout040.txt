import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_smooth(h, alpha):
    """Smooth approximation of max using log-sum-exp."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def dual_weighted_gradient(h, dual_weights, dx):
    """Compute gradient of weighted sum of shift overlaps."""
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx, w in enumerate(dual_weights):
        if w < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def dual_weighted_descent(h, n, target, time_limit, max_steps=20000, eta=2.0, m_top=15):
    """Descend using dual-weighted gradient with adaptive learning rate."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break
        
        # Update dual weights
        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_sum = lam_new.sum()
        if lam_sum < 1e-10:
            lam_new = np.ones(n_corr) / n_corr
        else:
            lam_new /= lam_sum
        lam = 0.6 * lam + 0.4 * lam_new
        
        # Focus on top m for efficiency
        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()
        
        grad = np.zeros(n)
        idxs = np.arange(n)
        for wi, s_idx in zip(lam_top, top_m):

            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        
        # Line search
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        improved = False
        for _ in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.15, 0.5)
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            best_lr *= 0.85
            if best_lr < 0.002:

def lbfags_polish(h_start, n, target, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(logsumexp_smooth, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def multi_start_search(h_init, n, target, time_limit, n_starts=5, seed=42):
    """Try multiple random perturbations and keep the best."""
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for s in range(n_starts):
        if time.time() - start > time_limit - 30:
            break
        scale = 0.05 * (s + 1)
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        
        # Quick optimization
        h_opt, c5_opt = dual_weighted_descent(h_pert, n, target,
                                              time_limit=min(30, (time_limit - 30) / (n_starts - s)),
                                              max_steps=3000, eta=2.0, m_top=10)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:

        weights = np.zeros_like(corr)

        weights /= w_sum

        grad[v1] += w * (1.0 - h[idxs[v1] - k])

def slsqp_smooth(h, n, target, alpha, maxiter=1000):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n

def pair_cd_ex(h, n, target, time_limit, max_sweeps=50000, n_cand_base=30, patience=250):
    """Enhanced pair CD with adaptive candidates and dual-weighted gradient."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    no_improve = 0
    
    # Precompute shift index arrays for speed
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        # Adaptive candidate size
        nc = n_cand_base
        if no_improve > patience // 3:
            nc = min(n_cand_base + 10, 50)
        if no_improve > patience // 2:
            nc = min(n_cand_base + 20, 60)
        
        # Use dual-weighted gradient for pair selection
        # Top shifts
        top_k = 5
        top_shifts = np.argsort(corr)[-top_k:]
        # Equal weights on top shifts (simple but effective)
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= top_k
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]
        
        best_trial_c5 = c5_val
        best_move = None
        
        # Try pairs
        for i in high_set:
            ip = (i + shifts >= 0) & (i + shifts < n)
            h_ip = np.zeros(n_corr)
            h_ip[ip] = h[i + shifts[ip]]
            im = (i - shifts >= 0) & (i - shifts < n)
            h_im = np.zeros(n_corr)
            h_im[im] = h[i - shifts[im]]
            
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                jp = (j + shifts >= 0) & (j + shifts < n)
                jpm = np.zeros(n_corr)
                jpm[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                jsm = np.zeros(n_corr)
                jsm[jm] = h[j - shifts[jm]]
                
                g = h_ip + h_im - jpm - jsm
                
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    continue
                
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = float(np.max(g[active]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_move = (int(i), int(j), delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = project_feasible(h, target)
    
    return best_h, best_c5
