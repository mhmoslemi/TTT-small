    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

    n = len(h)

    return float(np.max(corr)) * dx

        return h

        return h

2. Find the minimizer of <g, h> over the feasible set (this is the FW step)

4. Update h

- Sort g
- Set h[i] = 1 for the smallest g[i] values (to bring sum to target)
- Set h[i] = 0 for the rest
- (This is because the constraint is sum(h) = target, and we want to minimize <g, h>)
