def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    """SLSQP with smooth max, multiple alpha values."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    idxs = np.arange(n)
    mid = n - 1
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, alpha) * (2.0 / n)
    
    def obj_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = smooth_max_weights(corr, alpha)
        grad = compute_grad_from_weights(w, x, n)
        grad *= (2.0 / n)
        grad -= np.mean(grad)
        return grad
    
    def con(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                lambda x: obj(x, alpha), h,
                jac=lambda x: obj_grad(x, alpha),
                method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def cd_multi_shift(h0, n, target, time_limit, n_cand=20, n_top=5):
    """Coordinate descent with multi-shift weighted gradient."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for sweep in range(20000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        
        # Top shifts
        top_idx = np.argsort(corr)[::-1][:n_top]
        c_max = corr[top_idx[0]]
        w = np.exp(50.0 * (corr[top_idx] - c_max))
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select pairs
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-12:
                    continue
                for frac in [0.5, 0.8, 1.0]:
                    d = frac * d_max
                    h[i] -= d
                    h[j] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    c5_try = compute_c5(h)
                    if c5_try < best_c5_try:
                        best_c5_try = c5_try
                        best_move = (i, j, d)
                    else:
                        # revert
                        h[i] += d
                        h[j] -= d
                        h = np.clip(h, 0.0, 1.0)
                        h = project_feasible(h, target)
                        break
        
        if best_move is None:
            # try smaller steps
            for i in high_set[:10]:
                for j in low_set[:10]:
                    if i == j:
                        continue
                    d_max = min(h[i], 1.0 - h[j])
                    if d_max < 1e-12:
                        continue

                    h[i] -= d
                    h[j] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    c5_try = compute_c5(h)
                    if c5_try < best_c5_try:
                        best_c5_try = c5_try
                        best_move = (i, j, d)
                        break
                if best_move is not None:
                    break
        
        if best_move is None:
            no_improve += 5
            if no_improve > 500:
                break
    
    return best_h, best_c5

def lp_cutting_plane(h0, n, target, time_limit, n_active_max=30, max_rounds=50):
    """LP cutting plane: grow active set, solve LP."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    active_set = []
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        # Add top shifts to active set
        top_now = np.argsort(corr)[::-1]
        for s in top_now:
            if int(s) not in active_set and len(active_set) < n_active_max:
                active_set.append(int(s))
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for s_idx in active_set:
            k = s_idx - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin = corr[s_idx] + float(grad_k @ (h_var - h))
            constraints.append(lin <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                try:
                    prob.solve(solver=cp.CVXOPT)
                except Exception:
                    break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            # Try mixing
            found = False
            for mix in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix * h + (1-mix) * h_lp, target)
                if compute_c5(h_mix) < c5_val - 1e-10:
                    h = h_mix
                    found = True
                    break
            if not found:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    return w

    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:

        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    return grad

def fista_smooth(h0, n, target, time_limit, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()
    t_k = 1.0
    base_alpha = 200.0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        alpha = base_alpha * (1.0 + 0.02 * step)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        w = smooth_max_weights(corr_y, alpha)
        grad = grad_from_weights(w, y, n)
        grad *= dx
        grad -= np.mean(grad)
        L = alpha * dx * n * 2.0
        lr = 1.0 / L
        h_new = project_feasible(y - lr * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        t_k = t_new
        h = h_new
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def make_slsqp_obj(alpha, n, target):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, alpha) * dx
    return obj

def make_slsqp_grad(alpha, n, target):
    dx = 2.0 / n
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = smooth_max_weights(corr, alpha)
        grad = grad_from_weights(w, x, n)
        grad *= dx
        grad -= np.mean(grad)
        return grad
    return obj_grad

def make_slsqp_con(target):
    def con(x):
        return np.sum(x) - target
    return con

def make_slsqp_con_jac(n):
    def con_jac(x):
        return np.ones(n)
    return con_jac

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        obj_fn = make_slsqp_obj(alpha, n, target)
        grad_fn = make_slsqp_grad(alpha, n, target)
        con_fn = make_slsqp_con(target)
        jac_fn = make_slsqp_con_jac(n)
        cons = [{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}]
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def cd_multi_shift(h0, n, target, time_limit, n_cand=20, n_top=5):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    shifts_all = np.arange(-(n - 1), n)
    no_improve = 0
    for sweep in range(30000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        top_idx = np.argsort(corr)[::-1][:n_top]
        c_max = corr[top_idx[0]]
        w = np.exp(80.0 * (corr[top_idx] - c_max))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_move = None
        best_approx = c5_val
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts_all >= 0) & (i + shifts_all < n)
                g[v_ip] += h[i + shifts_all[v_ip]]
                v_im = (i - shifts_all >= 0) & (i - shifts_all < n)
                g[v_im] += h[i - shifts_all[v_im]]
                v_jp = (j + shifts_all >= 0) & (j + shifts_all < n)
                g[v_jp] -= h[j + shifts_all[v_jp]]
                v_jm = (j - shifts_all >= 0) & (j - shifts_all < n)
                g[v_jm] -= h[j - shifts_all[v_jm]]
                a_d = 0.0
                b_d = d_max
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_growing(h0, n, target, time_limit, n_active_max=30, max_rounds=80):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    active_set = []
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        top_now = np.argsort(corr)[::-1]
        added = False
        for s in top_now:
            si = int(s)
            if si not in active_set and len(active_set) < n_active_max:
                active_set.append(si)
                added = True
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active_set:
            k = s_idx - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin = corr[s_idx] + float(grad_k.dot(h_var - h))
            constraints.append(lin <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=8000, abstol=1e-9, reltol=1e-9)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, max_iter=8000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            found = False
            for mix_f in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix_f * h + (1.0 - mix_f) * h_lp, target)
                if compute_c5(h_mix) < c5_val - 1e-10:
                    h = h_mix
                    found = True
                    break
            if not found:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_perturb(h0, n, target, rng):
    x = np.linspace(0, 2, n)
    cands = []
    for freq in [1, 2, 3, 4, 5, 7, 10, 13, 17]:
        for amp in [0.05, 0.1, 0.15, 0.2]:
            for phase in [0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi]:
                p = amp * np.sin(freq * np.pi * x + phase)
                cands.append(p)
    for block in [8, 16, 32]:
        for _ in range(3):
            n_b = n // block
            vals = rng.uniform(-0.2, 0.2, size=n_b)
            p = np.repeat(vals, block)[:n]
            cands.append(p)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for p in cands:
        h_try = project_feasible(h0 + p, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: Structured perturbations for diverse starts
    if tleft() > 300:
        h_sp, s_sp = structured_perturb(best_h, n, target, rng)
        if s_sp < best_score:
            best_score = s_sp
            best_h = h_sp

    # Phase 2: FISTA fast progress
    if tleft() > 400:
        h_f, s_f = fista_smooth(best_h, n, target, min(100.0, tleft() * 0.15), max_steps=40000)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f

    # Phase 3: SLSQP with ramping alpha
    if tleft() > 400:
        alphas_p3 = [50, 200, 800, 3000, 10000, 30000, 80000]
        h_s, s_s = slsqp_round(best_h, n, target, alphas_p3, 2500, 1e-14,
                                min(200.0, tleft() * 0.25), start_time)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s

    # Phase 4: Multi-shift CD
    if tleft() > 300:
        h_cd, s_cd = cd_multi_shift(best_h, n, target, min(150.0, tleft() * 0.25), n_cand=18, n_top=5)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Phase 5: LP with growing active set
    if tleft() > 200:
        h_lp, s_lp = lp_growing(best_h, n, target, min(120.0, tleft() * 0.25), n_active_max=30, max_rounds=60)
        if s_lp < best_score:
            best_score = s_lp
            best_h = h_lp

    # Phase 6: SLSQP high alpha refinement
    if tleft() > 200:
        alphas_p6 = [2000, 8000, 30000, 100000, 300000]
        h_s, s_s = slsqp_round(best_h, n, target, alphas_p6, 3000, 1e-15,
                                min(150.0, tleft() * 0.25), start_time)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s

    # Phase 7: CD with wider candidate set
    if tleft() > 150:
        h_cd, s_cd = cd_multi_shift(best_h, n, target, min(100.0, tleft() * 0.3), n_cand=25, n_top=8)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Phase 8: Resolution upgrade
    if tleft() > 400:
        for n_try in [800, 1200]:
            if tleft() < 250:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            if tleft() > 150:
                h_hr, s_hr = slsqp_round(h_interp, n_try, target_try,
                                           [50, 200, 800, 3000, 10000, 30000],
                                           2000, 1e-14, min(100.0, tleft() * 0.25), start_time)
                if s_hr < best_score:
                    best_score = s_hr
                    best_h = h_hr
            if tleft() > 100:
                n_use = len(best_h)
                target_use = float(n_use) / 2.0
                h_cd2, s_cd2 = cd_multi_shift(best_h, n_use, target_use,
                                               min(80.0, tleft() * 0.25), n_cand=15, n_top=5)
                if s_cd2 < best_score:
                    best_score = s_cd2
                    best_h = h_cd2
            if tleft() > 80:
                n_use = len(best_h)
                target_use = float(n_use) / 2.0
                h_lp2, s_lp2 = lp_growing(best_h, n_use, target_use,
                                           min(60.0, tleft() * 0.25), n_active_max=25, max_rounds=30)
                if s_lp2 < best_score:
                    best_score = s_lp2
                    best_h = h_lp2

    # Phase 9: LP final
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_lp, s_lp = lp_growing(best_h, n_use, target_use, min(60.0, tleft() * 0.4), n_active_max=35, max_rounds=40)
        if s_lp < best_score:
            best_score = s_lp
            best_h = h_lp

    # Phase 10: SLSQP final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_fin, s_fin = slsqp_round(best_h, n_use, target_use,
                                     [10000, 50000, 200000, 500000],
                                     2000, 1e-15, min(30.0, tleft() * 0.7), start_time)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    # Phase 11: Final CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, s_cd = cd_multi_shift(best_h, n_use, target_use,
                                     min(15.0, tleft() * 0.7), n_cand=20, n_top=5)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
