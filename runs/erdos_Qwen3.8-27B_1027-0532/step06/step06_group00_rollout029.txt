def subgradient_step(h, n, target, c5_target, max_iter=10000, startup_iters=100):
    """Polyak-like subgradient method with adapt

    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Polyak parameters
    gamma = 1.0  # step size parameter
    avg_c5 = best_c5
    avg_count = 1
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Running average for Polyak step
        avg_c5 = (avg_c5 * avg_count + c5_val) / (avg_count + 1)
        avg_count += 1
        
        if avg_c5 <= c5_target + 1e-8:
            break
        
        # Subgradient: gradient of the max correlation
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        g_norm_sq = np.sum(grad ** 2)
        if g_norm_sq < 1e-16:
            break
        
        # Polyak step
        if it < startup_iters:
            eta = 0.01 / np.sqrt(g_norm_sq + 1e-10)
        else:
            eta = gamma * (avg_c5 - c5_target) / (g_norm_sq + 1e-10)
            eta = min(eta, 0.1)  # cap step size
        
        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < c5_val:
            h = h_try
            gamma = min(gamma * 1.05, 10.0)  # increase step
        else:
            gamma = max(gamma * 0.8, 0.1)  # decrease step
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0
_g_n_top = 5


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_top_m(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_top = _g_n_top
    idx = np.argsort(corr)[::-1][:n_top]
    top = corr[idx]
    c_max = top[0]
    shifted = np.clip(_g_alpha * (top - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, n_top=5, ftol=1e-14):
    global _g_n, _g_target, _g_alpha, _g_n_top
    _g_n = n
    _g_target = target
    _g_n_top = n_top
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj_top_m, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=25, n_cand_end=75, max_no_improve=500,
               n_shifts=3, kick_interval=350, kick_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        elapsed = time.time() - start
        if elapsed > time_limit - 3:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if kick_interval > 0 and consecutive_no_improve >= kick_interval:
            rng = np.random.RandomState(42 + sweep)
            h = h + rng.randn(n) * kick_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            continue

        top_indices = np.argsort(corr)[::-1][:n_shifts]
        grad = np.zeros(n)
        for mi in range(n_shifts):
            k = int(top_indices[mi]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(n_shifts)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            consecutive_no_improve += 1

    return best_h, best_c5


def multi_start_slsqp(h_init, n, target, time_limit, n_restarts=4):
    start = time.time()
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    rng = np.random.RandomState(777)

    for r in range(n_restarts):
        if time.time() - start > time_limit - 30:
            break
        noise_scale = 0.02 + 0.03 * r
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        alphas = [500, 5000, 20000]
        h_pol, sc = slsqp_round(h_pert, n, target, alphas, 600, n_top=4, ftol=1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
