    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break

        c_max = np.max(corr)
        active_threshold = c_max - 0.05 * (c_max - np.min(corr) + 1e-15)
        active_shifts = np.where(corr >= active_threshold)[0]
        active_shifts = active_shifts[:n_active]

        grad_combined = np.zeros(n)
        for s_idx in active_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = (corr[int(s_idx)] - c_max + active_threshold)
            grad_combined[v1] += w * (1.0 - h[idxs[v1] - k])
            grad_combined[v2] -= w * h[idxs[v2] + k]
        grad_combined -= np.mean(grad_combined)

        sorted_idx = np.argsort(grad_combined)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def binary_swap_search(h_init, n, target, time_limit, max_rounds=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improvement = 0.0
        best_pair = None

        for _ in range(500):
            if time.time() - start > time_limit:
                break
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            if i == j:
                continue

            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g = np.zeros(n_corr)
            g[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g[v_im] += h[i - shifts[v_im]]
            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]

            new_max = np.max(corr + g)
            improvement = float(np.max(corr)) - new_max
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            h = project_feasible(h, target)
        else:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_obj_alpha(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Gradient momentum optimization (fast, replaces initial SLSQP)
    if tleft() > 200:
        t_alloc = min(150, tleft() * 0.15)
        h_gm, c5_gm = gradient_momentum_optimize(best_h, n, target, t_alloc,
                                                  alpha_start=2000, alpha_end=50000,
                                                  max_steps=30000, beta=0.8, lr_start=0.003)
        if c5_gm < best_score:
            best_score = c5_gm
            best_h = h_gm

    # Phase 2: Main pair CD (the workhorse - give it generous time)
    if tleft() > 300:
        t_alloc = min(350, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_refined(best_h, n, target, t_alloc,
                                       max_sweeps=15000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Multi-active pair CD (focus on multiple active shifts)
    if tleft() > 200:
        t_alloc = min(200, tleft() * 0.2)
        h_ma, c5_ma = multi_active_pair_cd(best_h, n, target, t_alloc,
                                            max_sweeps=8000, n_active=5, n_cand=45, n_bisect=14)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    # Phase 4: Gradient momentum again (refine after CD)
    if tleft() > 150:
        t_alloc = min(100, tleft() * 0.12)
        h_gm2, c5_gm2 = gradient_momentum_optimize(best_h, n, target, t_alloc,
                                                    alpha_start=10000, alpha_end=300000,
                                                    max_steps=30000, beta=0.7, lr_start=0.002)
        if c5_gm2 < best_score:
            best_score = c5_gm2
            best_h = h_gm2

    # Phase 5: Second pair CD round with different params
    if tleft() > 200:
        t_alloc = min(250, tleft() * 0.25)
        h_cd2, c5_cd2 = pair_cd_refined(best_h, n, target, t_alloc,
                                         max_sweeps=12000, n_cand=65, n_bisect=16)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Binary swap search (exploit binary structure)
    if tleft() > 100:
        t_alloc = min(80, tleft() * 0.1)
        h_bs, c5_bs = binary_swap_search(best_h, n, target, t_alloc, max_rounds=30)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs
        # Follow up with CD if binary search found something
        if c5_bs < best_score + 0.0001:
            if tleft() > 100:
                h_cd3, c5_cd3 = pair_cd_refined(h_bs, n, target, min(60, tleft() * 0.1),
                                                 max_sweeps=5000, n_cand=50, n_bisect=14)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3

    # Phase 7: Multi-restart with perturbation
    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.1]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        t_alloc = min(60, tleft() * 0.12)
        h_try, c5_try = pair_cd_refined(h_pert, n, target, t_alloc,
                                         max_sweeps=5000, n_cand=45, n_bisect=14)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 8: Try higher resolution
    if tleft() > 250:
        for n_try in [800, 1000]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            # Quick gradient optimization at higher res
            t_alloc = min(60, tleft() * 0.15)
            h_hr, c5_hr = gradient_momentum_optimize(h_interp, n_try, target_try, t_alloc,
                                                      alpha_start=2000, alpha_end=20000,
                                                      max_steps=10000, beta=0.8, lr_start=0.003)
            if tleft() > 100 and c5_hr < best_score + 0.002:
                t_alloc2 = min(80, tleft() * 0.2)
                h_hr2, c5_hr2 = pair_cd_refined(h_hr, n_try, target_try, t_alloc2,
                                                 max_sweeps=8000, n_cand=45, n_bisect=14)
                if c5_hr2 < best_score:
                    best_score = c5_hr2
                    best_h = h_hr2

    # Phase 9: Final focused pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_alloc = min(80, tleft() * 0.4)
        h_cd4, c5_cd4 = pair_cd_refined(best_h, n_use, target_use, t_alloc,
                                         max_sweeps=10000, n_cand=70, n_bisect=16)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Phase 10: Final gradient polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_alloc = min(25, tleft() * 0.5)
        h_fin, c5_fin = gradient_momentum_optimize(best_h, n_use, target_use, t_alloc,
                                                    alpha_start=50000, alpha_end=500000,
                                                    max_steps=20000, beta=0.5, lr_start=0.001)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
