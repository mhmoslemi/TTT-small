import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_iteration(h, n, target, active_k, solver):
    """One LP iteration: fix g=1-h, solve LP for new h."""
    dx = 2.0 / n
    g = 1.0 - h
    half_n = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in active_k:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo-k:hi-k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if h_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), float(t_var.value)
    except:
        pass
    # Fallback
    try:
        prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), float(t_var.value) if t_var.value is not None else None
    except:
        pass
    return h, None

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    val = c_max + np.log(np.sum(exp_s)) / alpha
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    half_n = n - 1
    omh = 1.0 - h
    nz = np.where(weights > 1e-10)[0]
    for k_idx in nz:
        w = weights[k_idx]
        k = k_idx - half_n
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * omh[lo1+k:hi1+k] * dx
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k] * dx
    return val, grad

def lbfgs_optimize(h0, target, alphas, maxiter, time_limit):
    n = len(h0)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_score = compute_c5(h0)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        penalty = 1e6  # heavy penalty for sum constraint
        
        def obj(h):
            v, _ = smooth_max_and_grad(h, alpha)
            return v + 0.5 * penalty * (np.sum(h) - target)**2
        
        def objg(h):
            v, g = smooth_max_and_grad(h, alpha)
            g = g + penalty * (np.sum(h) - target)
            return v, g
        
        try:
            res = minimize(objg, best_h, jac=True, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new
        except:
            pass
    return best_h, best_score

def coord_descent_fast(h, target, n_active, max_passes, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    start = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        for i in range(n):
            # For each active k, compute the slope and intercept of f_k(h[i])
            # f_k = sum_j h[j]*(1-h[j+k])*dx
            # Treating h[i] as the variable:
            # Terms involving h[i]: h[i]*(1-h[i+k])*dx (if 0<=i+k<n) + h[i-k]*(1-h[i])*dx (if 0<=i-k<n)
            # = h[i]*dx - h[i]*h[i+k]*dx + h[i-k]*dx - h[i-k]*h[i]*dx
            # As a function of h[i] (treating others as fixed):
            # = h[i]*(dx - h[i+k]*dx - h[i-k]*dx) + h[i-k]*dx
            # = slope * h[i] + intercept
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active[idx]
                k = k_idx - half_n
                A = dx
                B = 0.0
                if 0 <= i+k < n:
                    B = h[i+k] * dx
                if 0 <= i-k < n:
                    B += h[i-k] * dx
                slopes[idx] = A - B
                intercepts[idx] = corr[k_idx] - slopes[idx] * h[i]
            
            # Find h[i] in [0,1] that minimizes max_k (slopes[k]*h[i] + intercepts[k])
            # This is a 1D convex problem (max of linear functions)
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = slopes * mid + intercepts
                active_set = np.abs(vals - np.max(vals)) < 1e-10 * max(1.0, np.max(np.abs(vals)))
                if np.all(slopes[active_set] > 1e-12):
                    hi = mid
                elif np.all(slopes[active_set] < -1e-12):
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        active = np.argsort(corr)[-n_active:]
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_subproblem(h_cur, n, target, active_k, solver):
    dx = 2.0 / n
    g = 1.0 - h_cur
    half_n = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in active_k:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if h_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), (float(t_var.value) if t_var.value is not None else None)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.SCS, eps=1e-13, max_iters=200000, verbose=False)
        if h_var.value is not None:
            return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), (float(t_var.value) if t_var.value is not None else None)
    except Exception:
        pass
    return h_cur, None

def smooth_max_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    val = c_max + np.log(np.sum(exp_s)) / alpha
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    half_n = n - 1
    omh = 1.0 - h
    nz = np.where(weights > 1e-8)[0]
    for k_idx in nz:
        w = weights[k_idx]
        k = int(k_idx) - half_n
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * omh[lo1 + k:hi1 + k] * dx
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - k:hi2 - k] * dx
    return float(val), grad

def lbfgs_phase(h0, target, alphas, maxiter, time_limit, pen):
    n = len(h0)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_score = compute_c5(h0)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def _obj(h):
            v, _ = smooth_max_val_grad(h, alpha)
            return v + 0.5 * pen * (np.sum(h) - target) ** 2
        def _objg(h):
            v, g = smooth_max_val_grad(h, alpha)
            g = g + pen * (np.sum(h) - target)
            return v, g
        try:
            res = minimize(_objg, best_h, jac=True, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 30})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

        h_cur = project_feasible(h_cur, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
