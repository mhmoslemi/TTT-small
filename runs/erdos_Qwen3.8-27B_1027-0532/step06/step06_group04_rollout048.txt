    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return (c_max + logsum / alpha) * dx

def _cons_fun(v, target):
    return np.sum(v) - target

def _cons_jac(v, target):
    return np.ones(len(v))

def slsqp_optimize(h, n, target, alphas, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda v: _cons_fun(v, target), 'jac': lambda v: _cons_jac(v, target)}]
    h_cur = project_box_sum(h.copy(), target)
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n, target),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_improve(h, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = project_box_sum(h.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half = n - 1
    idxs = np.arange(n)
    stall = 0
    while time.time() - start < time_limit:
        corr = get_corr(h_cur)
        k_star = int(np.argmax(corr)) - half
        v1 = idxs - k_star
        v2 = idxs + k_star
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        grad = np.zeros(n)
        grad[m1] += (1.0 - h_cur[v1[m1]])
        grad[m2] -= h_cur[v2[m2]]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_box_sum(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_box_sum(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: LP with 150 constraints
    if tleft() > 200:
        lp_time = min(320, tleft() * 0.32)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time,
                                      150, 0.85, 5e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent
    if tleft() > 80:
        h_cd, c5_cd = coord_descent(best_h, target, 120, min(70, tleft() * 0.12), rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP with 250 constraints
    if tleft() > 200:
        lp_time = min(280, tleft() * 0.28)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time,
                                      250, 0.9, 1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP
    if tleft() > 100:
        h_ref = slsqp_optimize(best_h, n, target, [200, 1000, 5000], 1200, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: LP with 350 constraints
    if tleft() > 180:
        lp_time = min(220, tleft() * 0.28)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time,
                                      350, 0.92, 5e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: Subgradient
    if tleft() > 50:
        h_sg, c5_sg = subgradient_improve(best_h, n, target, min(35, tleft() * 0.3))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Fourier perturbations
    for mode in [3, 5, 7, 11, 17]:
        if tleft() < 70:
            break
        x = np.linspace(0, 2, n)
        pert = 0.02 * np.sin(mode * np.pi * x)
        h_p = project_box_sum(best_h + pert, target)
        h_cd, c5_cd = coord_descent(h_p, target, 100, min(15, tleft() * 0.15), rng)
        sc = compute_c5(h_cd)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 8: Final LP
    if tleft() > 70:
        lp_time = min(90, tleft() * 0.4)
        h_lp, c5_lp = alternating_lp(best_h, n, target, lp_time,
                                      400, 0.95, 1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: Final CD
    if tleft() > 35:
        h_cd, c5_cd = coord_descent(best_h, target, 150, min(25, tleft() * 0.5), rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 25:
        h_ref = slsqp_optimize(best_h, n, target, [5000, 20000], 500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
