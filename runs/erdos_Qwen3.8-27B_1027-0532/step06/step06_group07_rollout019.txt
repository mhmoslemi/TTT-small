def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_c5_and_grad(h, alpha, dx):

    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    f_val = dx * (max_c + np.log(sum_exp) / alpha)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    sig = np.where(p > 1e-10)[0]

        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return f_val, grad

def _slsqp_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

        return np.sum(h) - _global_target

def structured_perturbation(h, n, seed_val, n_modes=5):

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.04)
        h_pert += amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h_pert, n / 2.0)

            best_h = h_pol2

    # Phase 7: Structured perturbation restarts
    rng = np.random.RandomState(seed)
    n_restarts = 4
    for ri in range(n_restarts):
        if tleft() < 120:
            break
        h_pert = structured_perturbation(best_h, n, seed + ri * 1000 + 7, n_modes=3)
        # Quick Nesterov from perturbed
        nt = min(60, tleft() * 0.2)
        h_try, c5_try = nesterov_optimize(h_pert, n, target, time_limit=nt,
                                          alpha_start=500, alpha_end=10000, max_iter=50000)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try
        # Quick pair CD from best of the two
        if tleft() < 80:
            break
        cd_t = min(50, tleft() * 0.2)
        start_cd = best_h if best_score < c5_try else h_try
        h_try2, c5_try2 = active_pair_cd(start_cd, n, target, time_limit=cd_t,
                                         epsilon_frac=0.02, n_cand=45, n_bisect=13)
        if c5_try2 < best_score:
            best_score = c5_try2
            best_h = h_try2

    # Phase 8: Final active pair CD
    if tleft() > 80:
        cd_t = min(100, tleft() * 0.4)
        h_cd3, c5_cd3 = active_pair_cd(best_h, n, target, time_limit=cd_t,
                                       epsilon_frac=0.015, n_cand=65, n_bisect=15)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 9: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use, [50000, 150000, 400000], 800, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
