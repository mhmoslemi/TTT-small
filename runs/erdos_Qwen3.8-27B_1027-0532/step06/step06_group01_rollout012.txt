import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _con_fun(x):
    return np.sum(x) - _TARGET


def _con_jac(x):
    return np.ones(len(x))


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit):

def compute_multi_shift_grad(h, n, corr, top_k):
    idxs = np.arange(n)
    mid = n - 1
    sorted_corr = np.argsort(corr)[-top_k:]
    cur_max = float(np.max(corr))
    alpha_w = 20.0
    w = np.exp(alpha_w * (corr[sorted_corr] - cur_max))
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, si in zip(w, sorted_corr):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    return grad
