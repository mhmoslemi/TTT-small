**Move types for binary SA:**
1. **Single swap**: Pick i with h[i]=1, j with h[j]=0. Set h[i]=0, h[j]=1.

   - This is the basic move.

   - Can create more complex patterns.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):

    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):

    else:
        idx = np.where(h > 0.0)[0]

        return h
    h = h.copy()

            if badness > 0:
                bad_ones.append((i, badness))

        for j in np.where(h == 0.0)[0]:
            goodness = 0
            if valid_plus[j]:
                if h[idx_plus[j]] == 1.0:
                    goodness += 1
            if valid_minus[j]:
                if h[idx_minus[j]] == 1.0:
                    goodness += 1
            if goodness > 0:
                good_zeros.append((j, goodness))

        if not bad_ones or not good_zeros:
            no_improve += 1
            continue

        bad_ones.sort(key=lambda x: -x[1])
        good_zeros.sort(key=lambda x: -x[1])

        found = False
        for i, _ in bad_ones[:10]:
            for j, _ in good_zeros[:10]:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < float(full.max()):
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improvement_count += 1
                    found = True
                    break
            if found:
                break

        if not found:
            no_improve += 1
        else:
            no_improve = 0

    return best_h, best_c5

def continuous_softmax_refine(h0, target, start_time, time_limit, alphas, lr_schedule='adaptive'):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(120.0, max(5.0, time_limit - (time.time() - start_time)))
        lr = 1.5 / (alpha + 1.0)
        momentum = 0.8
        v = np.zeros(n)
        n_iter = int(phase_time * 300)
        n_iter = max(n_iter, 100)
        patience = 500
        no_improve = 0

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = float(corr.max())
            shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(len(corr)) / len(corr)

            N_fft = 1
            while N_fft < 3 * n - 2:
                N_fft <<= 1
            W_fft = np.fft.fft(w, N_fft)
            h_pad = np.zeros(N_fft)
            h_pad[:n] = h
            h_fft = np.fft.fft(h_pad)
            conv_full = np.fft.ifft(W_fft * h_fft).real
            conv = conv_full[:n]
            grad = 1.0 - 2.0 * conv

            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new

            if it % 20 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                if no_improve > patience:
                    break

    return best_h, best_score

def double_swap_delta(h, i1, i2, j1, j2, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    for i, sign in [(i1, 1.0), (i2, 1.0), (j1, -1.0), (j2, -1.0)]:
        idx = i + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += sign * h[idx[mask]]
        idx = i - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += sign * h[idx[mask]]
    return result

def sa_double_swap(h0, target, n_max_iters, t_init, t_final, rng, time_limit,
                   start_time, k_arr, positions, single_frac=0.5):
    n = len(h0)
    h = h0.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    current_c5 = float(full.max()) * dx
    best_c5 = current_c5
    best_h = h.copy()
    stall = 0
    max_stall = 30000

    for it in range(n_max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 4 or len(zeros) < 4:
            break

        max_val = float(full.max())

        if rng.random() < single_frac:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            idx1 = rng.choice(len(ones), size=2, replace=False)
            idx2 = rng.choice(len(zeros), size=2, replace=False)
            i1, i2 = ones[idx1[0]], ones[idx1[1]]
            j1, j2 = zeros[idx2[0]], zeros[idx2[1]]
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            full_new = full_corr_fast(h)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                current_c5 = c5_new
                full = full_new
                if c5_new < best_c5:
                    best_c5 = c5_n
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                stall += 1

    return best_h, best_c5

def make_periodic_init(n, period, ones_per_period, rng):
    h = np.zeros(n)
    target = n // 2
    pattern = np.zeros(period)
    ones_pos = rng.choice(period, size=min(ones_per_period, period), replace=False)
    pattern[ones_pos] = 1.0
    n_full = n // period
    h[:n_full * period] = np.tile(pattern, n_full)
    h[n_full * period:] = pattern[:n % period]
    return project_binary(h, target)

def make_staircase_init(n, rng):
    target = n // 2
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:target]] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n // 2
    k_arr = np.arange(2 * n - 1) - (n - 1)
    positions = np.arange(n)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Lag-focused local search on binary (quick improvements)
    allowed = min(60, tleft() * 0.08)
    h_ls, c5_ls = lag_focused_local_search(best_h, target, rng,
                                            time_limit=allowed, start_time=start_time,
                                            k_arr=k_arr, positions=positions,
                                            max_rounds=50000)
    if c5_ls < best_score:
        best_score = c5_ls
        best_h = h_ls.copy()

    # Phase 2: Intensive binary SA with tabu (multi-start)
    starts = [best_h]
    # Add diverse starts
    starts.append(make_periodic_init(n, 7, 4, rng))
    starts.append(make_periodic_init(n, 11, 6, rng))
    starts.append(make_staircase_init(n, rng))
    # Interpolated version
    x_coarse = np.linspace(0, 1, 100)
    x_fine = np.linspace(0, 1, n)
    h_interp = np.interp(x_fine, x_coarse, project_binary(h_init, target)[:100])
    starts.append(project_binary(h_interp, target))

    for si, h_start in enumerate(starts):
        if tleft() < 100:
            break
        if si == 0:
            allowed = min(250, tleft() * 0.25)
            t_init = 0.03
            t_final = 0.00005
            n_iters = int(allowed * 50000)
        elif si == 1:
            allowed = min(120, tleft() * 0.12)
            t_init = 0.05
            t_final = 0.0001
            n_iters = int(allowed * 40000)
        else:
            allowed = min(80, tleft() * 0.08)
            t_init = 0.04
            t_final = 0.0002
            n_iters = int(allowed * 40000)

        h_sa, c5_sa = sa_tabu_binary(h_start, target, n_iters, t_init, t_final,
                                      rng, time_limit=allowed, start_time=start_time,
                                      k_arr=k_arr, positions=positions,
                                      tabu_size=120, max_stall=30000,
                                      focus_frac=0.8, top_k_focus=3, n_candidates=6)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: SA with double swaps from best binary
    if tleft() > 120:
        allowed = min(150, tleft() * 0.18)
        n_iters = int(allowed * 25000)
        h_ds, c5_ds = sa_double_swap(best_h, target, n_iters, 0.02, 0.0001,
                                      rng, time_limit=allowed, start_time=start_time,
                                      k_arr=k_arr, positions=positions,
                                      single_frac=0.6)
        if c5_ds < best_score:
            best_score = c5_ds
            best_h = h_ds.copy()

    # Phase 4: Continuous soft-max refinement
    if tleft() > 100:
        target_f = float(n) / 2.0
        alphas = [10, 30, 100, 300, 1000, 3000, 10000]
        allowed = min(250, tleft() * 0.25)
        h_cont, sc_cont = continuous_softmax_refine(best_h, target_f,
                                                     start_time, allowed, alphas)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()

    # Phase 5: Binary projection of continuous + SA
    if tleft() > 80:
        h_bin = project_binary(best_h, target)
        allowed = min(100, tleft() * 0.15)
        n_iters = int(allowed * 45000)
        h_bs, c5_bs = sa_tabu_binary(h_bin, target, n_iters, 0.015, 0.00003,
                                      rng, time_limit=allowed, start_time=start_time,
                                      k_arr=k_arr, positions=positions,
                                      tabu_size=100, max_stall=25000,
                                      focus_frac=0.85, top_k_focus=4, n_candidates=8)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs.copy()

    # Phase 6: Lag-focused local search again on best
    if tleft() > 60:
        allowed = min(80, tleft() * 0.12)
        h_ls2, c5_ls2 = lag_focused_local_search(best_h, target, rng,
                                                   time_limit=allowed, start_time=start_time,
                                                   k_arr=k_arr, positions=positions,
                                                   max_rounds=80000)
        if c5_ls2 < best_score:
            best_score = c5_ls2
            best_h = h_ls2.copy()

    # Phase 7: Another round of continuous refinement with different alphas
    if tleft() > 60:
        target_f = float(n) / 2.0
        alphas2 = [50, 200, 1000, 5000, 20000, 100000]
        allowed = min(120, tleft() * 0.15)
        h_cont2, sc_cont2 = continuous_softmax_refine(best_h, target_f,
                                                       start_time, allowed, alphas2)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2.copy()

    # Phase 8: Final intensive SA on binary projection
    if tleft() > 40:
        h_bin_final = project_binary(best_h, target)
        allowed = max(5.0, tleft() - 10)
        n_iters = int(allowed * 50000)
        h_final_sa, c5_final = sa_tabu_binary(h_bin_final, target, n_iters,
                                               0.01, 0.00001, rng,
                                               time_limit=allowed, start_time=start_time,
                                               k_arr=k_arr, positions=positions,
                                               tabu_size=80, max_stall=20000,
                                               focus_frac=0.9, top_k_focus=5, n_candidates=10)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_sa.copy()

    # Final: ensure feasibility and compute score
    target_f = float(n) / 2.0
    h_final = project_box_sum(best_h, target_f)
    final_score = compute_c5(h_final)

    if final_score >= best_score:
        h_final = best_h.copy()
        final_score = best_score

    return h_final, final_score, n
