A tighter bound: the overlap at shift k is bounded by min(n-|k|, n/2) (since there are at most n/2 ones and n/2 zeros in any window).

The best case is if h has no structure that creates large overlaps at any shift. This is related to "balanced sequences" and "low autocorrelation sequences."

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        margins = 1.0 - h
        total = margins.sum()
        if total > 1e-10:
            h = h + (diff / total) * margins
    else:
        total = h.sum()
        if total > 1e-10:
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0).copy()
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

For the candidate generation, I want to focus on swaps that are likely to reduce the bottleneck. The bottleneck is the shift with the maximum overlap. A swap that reduces this overlap is:
- Moving a 1 away from a position where h=0 at distance k_star
- Moving a 0 to a position where h=1 at distance k_star

The "A" array identifies positions where moving a 1 away would help (high A means the 1 at that position contributes a lot to the overlap at k_star).
The "B" array identifies positions where moving a 0 would help (high B means placing a 1 at that position would reduce the overlap at k_star).

So the best swap is: i from high-A ones, j from high-B zeros.

But we also need to consider the effect on OTHER shifts. A swap that helps k_star might hurt k_star+1.

The compute_delta function handles this by computing the effect on all shifts.
