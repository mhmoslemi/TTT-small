def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):

def compute_gradient_direct(h, corr, n):
    """Compute gradient using direct summation (vectorized over j)."""
    # corr has length 2n-1, indexed by m = k + (n-1) for shift k
    # w[m] proportional to exp(alpha * corr[m])
    # For efficiency, use the fact that most weight is on the top few shifts
    
    # Get top shifts
    top_k = 20
    top_idx = np.argsort(corr)[::-1][:top_k]
    w = np.zeros(2*n-1)
    max_c = corr.max()
    exp_vals = np.exp(50 * (corr[top_idx] - max_c))
    w[top_idx] = exp_vals / exp_vals.sum()
    
    # For each top shift, compute its contribution to the gradient
    grad = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        wm = w[m]
        # ∂C(k)/∂h[j] = (1-h[j+k])*I(valid) - h[j-k]*I(valid)
        if k >= 0:
            # j+k valid: j ∈ [0, n-1-k]
            valid1 = np.arange(0, n-k)
            grad[valid1] += wm * (1 - h[valid1 + k])
            # j-k valid: j-k ∈ [0,n-1] → j ∈ [k, n-1]
            valid2 = np.arange(k, n)
            grad[valid2] -= wm * h[valid2 - k]
        else:
            # k < 0, let k' = -k > 0
            k_abs = -k
            # j+k valid: j-k' valid: j ∈ [k', n-1]
            valid1 = np.arange(k_abs, n)
            grad[valid1] += wm * (1 - h[valid1 - k_abs])
            # j-k valid: j+k' valid: j ∈ [0, n-1-k']
            valid2 = np.arange(0, n-k_abs)
            grad[valid2] -= wm * h[valid2 + k_abs]
    
    return grad

What about the "checkerboard" solution? h[i] = 1 for even i, 0 for odd i (approximately).

So C(2) ≈ 0.

Max is at k=1: C₅ ≈ 300 * 2/600 = 1.0. Also terrible.

What about a more structured solution? The optimal solution should have the property that for every shift k, the "disagreement" between h and h shifted by k is roughly constant.

This is related to "differentially syndromable" sequences or "low autocorrelation" sequences.

For a binary sequence of length n with n/2 ones, the "overlap" at shift k is the number of positions where h[i]=1 and h[i+k]=0 (or vice versa).

We want to minimize the maximum over k of this overlap.

This is related to the theory of "B_seqs" or "Fermat sequences" or similar combinatorial designs.

For the continuous relaxation, the optimal solution would be a "fractional" version of such a sequence.

My plan:
1. Start from initial_h_values
2. Use projected gradient on smoothed max (top-k weighted)
3. Fine-tune with binary SA
4. Try a few different n values
5. Return the best

Let me also think about a key improvement: **using the top-k correlation profile as the objective**.

Instead of max_k C(k), use:
f(h) = (1/K) * sum of top K values of C(k)

This is a smoother objective (less peaked) and encourages a flatter correlation profile. It's a good proxy for the max.

For the gradient:
∂f/∂h[j] = (1/K) * sum_{k in top_K} ∂C_k/∂h[j]

This is easy to compute!

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr

    result[mask] += h[idx[mask]]
    idx = i - k_arr

    if current_sum == target_sum:

    else:

def top_k_gradient(h, corr, K=20, alpha=50):
    """Compute gradient using top-K weighted approach."""
    n = len(h)
    # Weight the top K correlations
    sort_idx = np.argsort(corr)[::-1]
    top_idx = sort_idx[:K]
    
    # Soft weights within top K
    max_c = corr[top_idx].max()
    w = np.zeros(2*n-1)
    exp_vals = np.exp(alpha * (corr[top_idx] - max_c))
    w[top_idx] = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for m in top_idx:
        k = m - (n - 1)
        wm = w[m]
        if abs(k) >= n:
            continue
        k_abs = abs(k)
        if k >= 0:
            # j+k valid: j ∈ [0, n-1-k]
            v1 = np.arange(0, n - k_abs)
            grad[v1] += wm * (1.0 - h[v1 + k_abs])
            # j-k valid: j ∈ [k, n-1]
            v2 = np.arange(k_abs, n)
            grad[v2] -= wm * h[v2 - k_abs]
        else:
            k_abs = -k
            # j+k = j-k' valid: j ∈ [k', n-1]
            v1 = np.arange(k_abs, n)
            grad[v1] += wm * (1.0 - h[v1 - k_abs])
            # j-k = j+k' valid: j ∈ [0, n-1-k']
            v2 = np.arange(0, n - k_abs)
            grad[v2] -= wm * h[v2 + k_abs]
    
    return grad
