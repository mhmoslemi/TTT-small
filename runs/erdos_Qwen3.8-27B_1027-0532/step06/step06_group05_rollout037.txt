                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.4:
            worst = np.argsort(corr)[-10:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.8:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def subgradient_step(h, target_sum, N_fft, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    k_star = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j - k_star < n:
            grad[j] += (1.0 - h[j - k_star])
        if 0 <= j + k_star < n:
            grad[j] -= h[j + k_star]
    h_new = h - lr * grad
    return project_box_sum(h_new, target_sum)


def subgradient_optimize(h0, target_sum, N_fft, time_start, time_budget, lr_init, lr_final):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    total_iters = 20000
    # Run subgradient with averaging
    average_h = np.zeros(n)
    n_iter = 0
    for it in range(total_iters):
        if time.time() - time_start > time_budget:
            break
        ratio = it / max(total_iters - 1, 1)
        lr = lr_init * (lr_final / lr_init) ** ratio
        h = subgradient_step(h, target_sum, N_fft, lr)
        average_h += h
        n_iter += 1
        avg_proj = project_box_sum(average_h / n_iter, target_sum)
        if it % 500 == 0:
            sc = compute_c5(avg_proj)
            if sc < best_score:
                best_score = sc
                best_h = avg_proj.copy()
    if n_iter > 0:
        avg_proj = project_box_sum(average_h / n_iter, target_sum)
        sc = compute_c5(avg_proj)
        if sc < best_score:
            best_score = sc
            best_h = avg_proj.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 10.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy binary
    h_bin = project_binary(best_h, int(target))
    greedy_time = min(40.0, tleft() * 0.08)
    if greedy_time > 5:
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=50000,
                                  time_limit=greedy_time, start_time=start_time, scan_width=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()
        # Also try continuous from binary
        h_from_bin = h_g * 1.0
        h_from_bin = project_box_sum(h_from_bin, target)

    # Phase 2: Main Adam with FFT gradient
    if tleft() > 300:
        adam_time = min(450, tleft() - 300)
        alphas_main = [20, 50, 100, 250, 500, 1000, 2000, 5000, 10000, 20000, 40000]
        h_a, s_a = adam_fft(best_h, target, N_fft, start_time, adam_time,
                            alphas_main, lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 3: L-BFGS-B refinement
    if tleft() > 200:
        for alpha_lb in [2000, 8000, 30000]:
            if tleft() < 50:
                break
            h_lb, sc_lb = lbfgsb_refine(best_h, n, target, N_fft, alpha_lb,
                                        maxiter=500, time_start=start_time,
                                        time_budget=min(80, tleft() - 50))
            if sc_lb < best_score:
                best_score = sc_lb
                best_h = h_lb.copy()

    # Phase 4: Multi-restart Adam
    n_restarts = 8
    if tleft() > 250:
        for ri in range(n_restarts):
            if tleft() < 80:
                break
            jitter_scale = 0.003 + 0.004 * (ri % 4)
            h_pert = best_h + rng.randn(n) * jitter_scale
            h_pert = project_box_sum(h_pert, target)
            rt = min(120, (tleft() - 80) / max(n_restarts - ri, 1))
            alphas_r = [50, 200, 1000, 5000, 20000, 60000]
            h_ar, s_ar = adam_fft(h_pert, target, N_fft, start_time, rt,
                                  alphas_r, lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8)
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()
            # Quick L-BFGS-B on the restart result
            if tleft() > 60 and s_ar < best_score + 0.001:
                h_lb2, sc_lb2 = lbfgsb_refine(h_ar, n, target, N_fft, 10000,
                                              maxiter=200, time_start=start_time,
                                              time_budget=min(40, tleft() - 30))
                if sc_lb2 < best_score:
                    best_score = sc_lb2
                    best_h = h_lb2.copy()

    # Phase 5: Subgradient with averaging
    if tleft() > 100:
        sg_time = min(80, tleft() - 50)
        h_sg, s_sg = subgradient_optimize(best_h, target, N_fft, start_time, sg_time,
                                          lr_init=0.05, lr_final=0.001)
        if s_sg < best_score:
            best_score = s_sg
            best_h = h_sg.copy()

    # Phase 6: Binary SA
    if tleft() > 80:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(100, tleft() - 40)
        max_it = int(min(300000, sa_time * 800))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                0.005, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 7: Final L-BFGS-B polish
    if tleft() > 40:
        h_fin, sc_fin = lbfgsb_refine(best_h, n, target, N_fft, 50000,
                                      maxiter=300, time_start=start_time,
                                      time_budget=min(60, tleft() - 15))
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin.copy()

    # Phase 8: Final Adam polish
    if tleft() > 25:
        polish_time = max(10.0, tleft() - 10)
        h_p, s_p = adam_fft(best_h, target, N_fft, start_time, polish_time,
                            alphas=[10000, 30000, 80000, 200000],
                            lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
