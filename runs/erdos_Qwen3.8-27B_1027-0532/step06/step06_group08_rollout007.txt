    if n_src == n_dst:
        return project_box_sum_exact(src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, src)
    return project_box_sum_exact(h_out, target_dst)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5_val(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def smooth_obj(x, a=alpha):
                corr = fft_corr(x)
                c_max = float(np.max(corr))
                shifted = np.clip(a * (corr - c_max), -400.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / a

            def cons_fun(x):
                return np.sum(x) - target

            def cons_jac(x):
                return np.ones(n)

            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum_exact(res.x, target)
            sc = compute_c5_val(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_cont_h = project_box_sum_exact(h_init, target_init)
    best_cont_score = compute_c5_val(best_cont_h)

    best_bin_h = project_binary(h_init, int(n_init // 2))
    best_bin_score = compute_c5_val(best_bin_h)

    # Phase 1: Nesterov smooth optimization (moderate alpha)
    allowed = min(300.0, max(50.0, tleft() * 0.25))
    if allowed > 20:
        h1, s1 = nesterov_opt(best_cont_h, target_init,
                              max_iter=5000000, alpha_start=20.0, alpha_end=200.0,
                              lr_init=0.03, lr_end=0.0005, topk=15, beta=0.9,
                              time_limit=allowed, start_time=start_time)
        if s1 < best_cont_score:
            best_cont_score = s1
            best_cont_h = h1.copy()

    # Phase 2: Subgradient with momentum (for large alpha / near-max behavior)
    allowed = min(200.0, max(30.0, tleft() * 0.15))
    if allowed > 15:
        h2, s2 = subgrad_opt(best_cont_h, target_init,
                             max_iter=3000000, lr_init=0.02, lr_end=0.0002,
                             topk=5, time_limit=allowed, start_time=start_time)
        if s2 < best_cont_score:
            best_cont_score = s2
            best_cont_h = h2.copy()

    # Phase 3: Binary greedy from continuous solution
    h_bin3 = project_binary(best_cont_h, int(n_init // 2))
    allowed = min(80.0, max(20.0, tleft() * 0.06))
    if allowed > 10:
        h3, s3 = greedy_binary(h_bin3, max_iters=100000, rng=rng,
                                time_limit=allowed, start_time=start_time)
        if s3 < best_bin_score:
            best_bin_score = s3
            best_bin_h = h3.copy()

    # Phase 4: Binary SA
    allowed = min(250.0, max(40.0, tleft() * 0.20))
    if allowed > 20:
        n_sa = int(min(10000000, allowed * 80000))
        h4, s4 = binary_sa(best_bin_h, n_sa, 0.015, 0.00003, rng,
                           time_limit=allowed, start_time=start_time)
        if s4 < best_bin_score:
            best_bin_score = s4
            best_bin_h = h4.copy()

    # Phase 5: Interpolate binary to continuous and re-optimize
    allowed = min(150.0, max(30.0, tleft() * 0.12))
    if allowed > 15:
        h_interp = project_box_sum_exact(best_bin_h, target_init)
        h5, s5 = nesterov_opt(h_interp, target_init,
                              max_iter=3000000, alpha_start=50.0, alpha_end=500.0,
                              lr_init=0.02, lr_end=0.0003, topk=10, beta=0.85,
                              time_limit=allowed, start_time=start_time)
        if s5 < best_cont_score:
            best_cont_score = s5
            best_cont_h = h5.copy()

    # Phase 6: High-alpha subgradient polish
    allowed = min(120.0, max(20.0, tleft() * 0.10))
    if allowed > 10:
        h6, s6 = subgrad_opt(best_cont_h, target_init,
                             max_iter=2000000, lr_init=0.01, lr_end=0.0001,
                             topk=3, time_limit=allowed, start_time=start_time)
        if s6 < best_cont_score:
            best_cont_score = s6
            best_cont_h = h6.copy()

    # Phase 7: Multi-resolution
    for n_try in [n_init + 4, n_init + 8, n_init - 4]:
        if tleft() < 80:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_cont_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, n_try // 2)

        allowed = min(30.0, max(10.0, tleft() * 0.04))
        if allowed < 8:
            continue
        h_g, s_g = greedy_binary(h_bin_try, max_iters=30000, rng=rng,
                                  time_limit=allowed * 0.5, start_time=start_time)

        allowed2 = min(40.0, max(15.0, tleft() * 0.05))
        if allowed2 < 10:
            continue
        h_pg, s_pg = nesterov_opt(h_g, target_try,
                                   max_iter=1000000, alpha_start=30.0, alpha_end=300.0,
                                   lr_init=0.025, lr_end=0.0004, topk=12, beta=0.9,
                                   time_limit=allowed2, start_time=start_time)
        if s_pg < best_cont_score:
            best_cont_score = s_pg
            best_cont_h = h_pg.copy()

    # Phase 8: Final SLSQP
    if tleft() > 40:
        n_use = len(best_cont_h)
        h_ref, sc_ref = slsqp_refine(best_cont_h, n_use, float(n_use) / 2.0,
                                      alphas=[100, 500, 2000],
                                      maxiter=3000, ftol=1e-14,
                                      time_limit=max(10.0, tleft() - 15),
                                      start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 9: Final binary refinement from best continuous
    if tleft() > 30:
        n_use = len(best_cont_h)
        h_bin_f = project_binary(best_cont_h, int(n_use // 2))
        allowed = min(25.0, max(10.0, tleft() * 0.15))
        if allowed > 8:
            n_sa_f = int(min(3000000, allowed * 60000))
            h_f, s_f = binary_sa(h_bin_f, n_sa_f, 0.008, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time)
            if s_f < best_bin_score:
                best_bin_score = s_f
                best_bin_h = h_f.copy()

    # Final: choose best
    h_final_cont = project_box_sum_exact(best_cont_h, float(len(best_cont_h)) / 2.0)
    final_cont_score = compute_c5_val(h_final_cont)

    h_final_bin = project_binary(best_bin_h, int(len(best_bin_h) // 2))
    final_bin_score = compute_c5_val(h_final_bin)

    if final_bin_score < final_cont_score:
        return h_final_bin, final_bin_score, len(h_final_bin)
    else:
        return h_final_cont, final_cont_score, len(h_final_cont)
