    n = len(h)
    pos = np.arange(n)
    grad = np.zeros(n)
    idx = pos - l_star
    m = (idx >= 0) & (idx < n)
    grad[m] += (1.0 - h[idx[m]])
    idx = pos + l_star

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        need = current - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Final check
    s = int(h_r.sum())
    if s != target_sum:
        if s > target_sum:
            extra = s - target_sum
            ones = np.where(h_r == 1)[0]
            h_r[ones[:extra]] = 0
        else:
            extra = target_sum - s
            zeros = np.where(h_r == 0)[0]
            h_r[zeros[:extra]] = 1

        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Choose move type
        r = rng.random()
        if r < 0.15:
            # Block reversal
            blen = rng.randint(3, min(15, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            if h_new.sum() != h.sum():
                continue
            corr_new = full_corr_fft(h_new)
            c5_new = float(corr_new.max()) * dx
            dc5 = c5_new - current_c5
            if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                h = h_new
                corr = corr_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        else:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_fast(h, i, j, k_arr)
            new_max = float((corr + delta).max())
            c5_new = new_max * dx
            dc5 = c5_new - current_c5
            if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    return best_h, best_c5
