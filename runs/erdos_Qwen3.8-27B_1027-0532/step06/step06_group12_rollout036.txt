import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if np.sum(mask) == 0:
        return h
    h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask2 = h < 1.0 - 1e-12
        else:
            mask2 = h > 1e-12
        if np.sum(mask2) > 0:
            h[mask2] += residual / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_loss_and_grad(h, alpha):
    """Compute smooth log-sum-exp loss and its gradient."""
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - max_ov), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total  # softmax weights
    loss = (max_ov + np.log(total) / alpha)
    
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    grad = np.zeros(n)
    
    for idx in range(len(p)):
        if p[idx] < 1e-12:
            continue
        k = idx - n_half
        w = p[idx] * dx
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] += w * (-h[m:])
    
    return loss, grad

def adam_phase(h_init, n, target, time_limit, start_time,
               alpha_start=20.0, alpha_end=5000.0, lr_start=0.01, lr_end=0.001):
    """Adam optimization with smooth log-sum-exp objective."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    max_iter = 100000
    
    for t in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        progress = min(1.0, t / max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        loss, grad = smooth_loss_and_grad(h, alpha)
        
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**(t+1))
        v_hat = v_v / (1 - beta2**(t+1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def coord_descent_step(h, i, n, target, overlaps):
    """Find optimal value for coordinate i given current overlaps."""
    dx = 2.0 / n
    B = np.zeros(2*n-1)
    
    for idx in range(2*n-1):
        k = idx - (n-1)
        if k >= 0:
            b = 0.0
            if i + k < n:
                b += (1.0 - h[i+k])
            if i >= k:
                b -= h[i-k]
        else:
            m_abs = -k
            b = 0.0
            if m_abs <= i:
                b += (1.0 - h[i - m_abs])
            if i + m_abs < n:
                b -= h[i + m_abs]
        B[idx] = b * dx
    
    A = overlaps - h[i] * B
    
    # Ternary search for min of max(A + B*x)
    lo, hi = 0.0, 1.0
    for _ in range(50):

        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(A + B * m1)
        f2 = np.max(A + B * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2.0

def coord_descent(h_init, n, target, time_limit, start_time, max_passes=500):
