```python
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = free.sum()
            if nf == 0:
                break
            h[free] = np.minimum(h[free] + diff / nf, 1.0)
        else:
            free = h > 1e-15
            nf = free.sum()
            if nf == 0:
                break
            h[free] = np.maximum(h[free] + diff / nf, 0.0)
    return h

def get_shift_gradient(h, s, n):
    """Gradient of overlap at shift s with respect to h.
    overlap(s) = sum_i h[i]*(1-h[i+s]) for valid i.
    d/ dh[i] = (1-h[i+s]) - h[i-s] (with boundary handling)
    """
    g = np.zeros(n)
    # First term: (1-h[i+s]) for i where i+s in [0,n-1]
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
    else:
        for i in range(-s, n):
            g[i] += (1.0 - h[i + s])
    # Second term: -h[i-s] for i where i-s in [0,n-1]
    if s >= 0:
        for i in range(s, n):
            g[i] -= h[i - s]
    else:
        for i in range(n + s):
            g[i] -= h[i - s]
    return g
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full')
```

```python
    weights /= wsum  # These are the softmax weights

        s = int(idx) - (n - 1)
```

Wait, it's not that simple because h[i] appears in multiple overlaps (for different s values). Specifically, h[i] appears in:
- overlap(s) as h[i]*(1-h[i+s]) when i+s is valid
- overlap(s) as (1-h[i-s])*h[i] when i-s is valid (i.e., as the second factor)

overlap(s) = x*(1-h[i+s]) + [rest] for s where i+s valid
overlap(s) = (1-h[i-s])*x + [rest] for s where i-s valid

This is linear in x for each s! So the max over s is a convex piecewise linear function of x. We can minimize it in O(K log K) where K is the number of shifts where x appears.

```python
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break

    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        g[-s:] += (1.0 - h[:n+s])
        g[:n+s] -= h[-s:]

    n = len(corr)
    if k >= n:
        return np.arange(n)
    # Use partial sort for efficiency
    idx = np.argpartition(corr, -k)[-k:]
```

```python
def linearized_LP(h_current, n, target, n_constraints=50):
    """Solve the linearized problem at h_current."""
    dx = 2.0 / n
    mid = n - 1
    corr_current = np.correlate(h_current, 1.0 - h_current, mode='full') * dx
    
    # Select top-n_constraints shifts
    top_idx = np.argpartition(corr_current, -n_constraints)[-n_constraints:]
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    
    for k_idx in top_idx:
        s = int(k_idx) - mid
        grad = _shift_grad(h_current, s, n)
        # Linearized: corr_current[k_idx] + grad · (h - h_current) ≤ t
        constraints.append(corr_current[k_idx] + dx * grad @ (h - h_current) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return h_current, compute_c5(h_current)
    
    if h.value is not None:
        h_new = project_feasible(h.value, target)
        return h_new, compute_c5(h_new)
    return h_current, compute_c5(h_current)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = int(free.sum())
            if nf == 0:
                break
            h[free] = np.minimum(h[free] + diff / nf, 1.0)
        else:
            free = h > 1e-15
            nf = int(free.sum())
            if nf == 0:
                break
            h[free] = np.maximum(h[free] + diff / nf, 0.0)
    return h

def _shift_grad(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        g[-s:] += (1.0 - h[:n+s])
        g[:n+s] -= h[-s:]
    return g

def slp_optimize(h_start, n, target, time_limit, max_iters, n_constraints, step_size=1.0):
    """Successive Linear Programming optimization."""
    dx = 2.0 / n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        # Select active constraints (top shifts)
        if n_constraints >= 2*n - 1:
            top_idx = np.arange(2*n-1)
        else:
            top_idx = np.argpartition(corr, -n_constraints)[-n_constraints:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        mid = n - 1
        
        for k_idx in top_idx:
            s = int(k_idx) - mid
            grad = _shift_grad(h_cur, s, n)
            # Linearized constraint: corr[k] + grad·(h - h_cur) ≤ t
            constraints.append(corr[k_idx] + dx * (grad @ (h_var - h_cur)) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = h_var.value
        # Step size: full step or damped
        h_candidate = h_cur + step_size * (h_lp - h_cur)
        h_candidate = project_feasible(h_candidate, target)
        
        c5_new = compute_c5(h_candidate)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_candidate.copy()
            h_cur = h_candidate
        elif c5_new < best_c5 + 1e-14:
            h_cur = h_candidate
        else:
            # Try smaller step
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h_cur + alpha * (h_lp - h_cur), target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
                    break
                elif c5_try < c5_new:
                    h_cur = h_try
            else:
                # No improvement, try to continue with same h_cur
                pass
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

# Globals for SLSQP
_G_n = 0
_G_target = 0.0
_G_top_idx = None
_G_alpha = 0.0
_G_dx = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            nf = int(free.sum())
            if nf == 0:
                break
            h[free] = np.minimum(h[free] + diff / nf, 1.0)
        else:
            free = h > 1e-15
            nf = int(free.sum())
            if nf == 0:
                break
            h[free] = np.maximum(h[free] + diff / nf, 0.0)
    return h

def _shift_grad(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        g[-s:] += (1.0 - h[:n+s])
        g[:n+s] -= h[-s:]
    return g

def _glb_con_fun(x):
    return np.sum(x) - _G_target

def _glb_con_jac(x):
    return np.ones(_G_n)

def _glb_obj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    vals = c[_G_top_idx] * _G_dx
    vmax = np.max(vals)
    return float(vmax + np.log(np.sum(np.exp(_G_alpha * (vals - vmax)))) / _G_alpha)

def _glb_grad(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    vals = c[_G_top_idx] * _G_dx
    vmax = np.max(vals)
    w = np.exp(_G_alpha * (vals - vmax))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w = w / wsum
    g = np.zeros(_G_n)
    mid = _G_n - 1
    for i in range(len(_G_top_idx)):
        s = int(_G_top_idx[i]) - mid
        g += w[i] * _shift_grad(x, s, _G_n)
    return g * _G_dx

def slsqp_active(h_start, n, target, time_limit, n_top, alphas, maxiter_per_alpha):
    global _G_n, _G_target, _G_top_idx, _G_alpha, _G_dx
    _G_n = n
    _G_target = target
    _G_dx = 2.0 / n
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _glb_con_fun, 'jac': _glb_con_jac}]
    
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    start = time.time()
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _G_alpha = alpha
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        if n_top >= 2*n - 1:
            _G_top_idx = np.arange(2*n-1)
        else:
            _G_top_idx = np.argpartition(corr, -n_top)[-n_top:]
        
        try:
            res = minimize(_glb_obj, h_cur, jac=_glb_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_c5

def slp_optimize(h_start, n, target, time_limit, max_iters, n_constraints, damping=1.0):
    """Successive Linear Programming."""
    dx = 2.0 / n
    mid = n - 1
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 50:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        if n_constraints >= 2*n - 1:
            top_idx = np.arange(2*n-1)
        else:
            top_idx = np.argpartition(corr, -n_constraints)[-n_constraints:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k_idx in top_idx:
            s = int(k_idx) - mid
            grad = _shift_grad(h_cur, s, n)
            constraints.append(corr[k_idx] + dx * (grad @ (h_var - h_cur)) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.array(h_var.value).flatten()
        h_cand = project_feasible(h_cur + damping * (h_lp - h_cur), target)
        c5_new = compute_c5(h_cand)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_cand.copy()
            h_cur = h_cand
            no_improve = 0
        else:
            no_improve += 1
            # Try damped step
            for dmp in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h_cur + dmp * (h_lp - h_cur), target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
                    no_improve = 0
                    break
    
    return best_h, best_c5

def pair_cd_topk(h_start, n, target, time_limit, max_sweeps, n_cand, n_top_shifts):
    """Pair coordinate descent considering top-K shifts."""
    dx = 2.0 / n
    mid = n - 1
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top-K shift indices
        if n_top_shifts >= 2*n-1:
            top_shifts = np.arange(2*n-1)
        else:
            top_shifts = np.argpartition(corr, -n_top_shifts)[-n_top_shifts:]
        
        # For each pair (i,j), compute delta_corr for top shifts
        # Use gradient-based selection: find i with best "donor" and j with best "receiver"
        # for the top shifts
        grad = np.zeros(n)
        for k_idx in top_shifts:
            s = int(k_idx) - mid
            grad += _shift_grad(h, s, n)
        grad /= len(top_shifts)
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_move = None
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                # delta_corr[s] for moving δ from i to j
                dcorr = np.zeros(len(top_shifts))
                for t_idx, k_idx in enumerate(top_shifts):
                    s = int(k_idx) - mid
                    sg = _shift_grad(h, s, n)
                    dcorr[t_idx] = sg[i] - sg[j]
                
                # Binary search for optimal δ
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr[top_shifts] + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                trial = float(np.max(corr[top_shifts] + delta_star * dcorr)) * dx
                # Also check the full corr (not just top shifts)

                h_trial[j] += delta_star
                h_trial = project_feasible(h_trial, target)
                trial_full = compute_c5(h_trial)
                
                if trial_full < best_trial:
                    best_trial = trial_full
                    best_move = (i, j, delta_star)
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = project_feasible(h, target)
        # else: no improving move found
    
    return best_h, best_c5

    return project_feasible(h_out, target_dst)
