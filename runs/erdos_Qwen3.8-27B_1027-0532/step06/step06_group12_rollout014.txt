import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, active_indices, n):
    dx = 2.0 / n
    n_half = n - 1
    g = np.zeros(n)
    for idx in active_indices:
        k = idx - n_half
        abs_k = abs(k)
        if abs_k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n - k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n - m]) * dx
            g[:n - m] += (-h[m:]) * dx
    return g / len(active_indices)

def build_lp(h_fixed, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        abs_k = abs(k)
        if abs_k == 0:
            coeff = (1.0 - h_fixed) * dx
            constraints.append(coeff @ h_var <= t_var)
        elif k > 0:
            coeff = np.zeros(n)
            coeff[:n - k] = (1.0 - h_fixed[k:]) * dx
            constraints.append(coeff @ h_var <= t_var)
        else:
            m = -k
            coeff = np.zeros(n)
            coeff[m:] = (1.0 - h_fixed[:n - m]) * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=200,

        lags = top_idx - n_half
        progress = min(1.0, it / max(1, max_iter * 0.3))

        reg = reg_start * (reg_end / reg_start) ** progress
        h_var, t_var, constraints = build_lp(g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:

            for _ in range(n_perturb):
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i != j and h_pert[i] > 0.05 and h_pert[j] < 0.95:
                    d = min(0.03, h_pert[i] * 0.5, (1.0 - h_pert[j]) * 0.5)
                    h_pert[i] -= d
                    h_pert[j] += d
            h_pert = project_feasible(h_pert, target)
            h = h_pert
            no_improve = 0
            continue
        
        n_half = n - 1
        top_indices = np.argsort(overlaps)[::-1][:top_k]
        active_lags = top_indices - n_half
        
        check_indices = np.argsort(overlaps)[::-1][:check_k]
        check_lags = check_indices - n_half
        
        high_i = np.where(h > 0.15)[0]
        low_j = np.where(h < 0.85)[0]
        
        if len(high_i) == 0 or len(low_j) == 0:
            no_improve += 1
            continue
        
        if len(high_i) > 80:
            rng_sort = np.random.RandomState(round_count % (2**31))
            rng_sort.shuffle(high_i)
            high_i = high_i[:80]
        if len(low_j) > 80:
            rng_sort = np.random.RandomState((round_count + 7919) % (2**31))
            rng_sort.shuffle(low_j)
            low_j = low_j[:80]
        
        best_move = None
        best_new_c5 = current_max
        best_improvement = 1e-14
        
        for d in delta_values:
            for i in high_i:
                if h[i] < d:
                    continue
                for j in low_j:
                    if i == j:
                        continue
                    if h[j] + d > 1.0:
                        continue
                    
                    max_delta = 0.0
                    for kl in active_lags:
                        do = compute_delta_o(h, i, j, kl, n)
                        if do > max_delta:
                            max_delta = do
                        if max_delta > -0.001 * d:
                            break
                    
                    if max_delta > -0.001 * d:
                        continue
                    
                    h[i] -= d
                    h[j] += d
                    h = project_feasible(h, target)
                    new_c5 = compute_c5(h)
                    h[i] += d
                    h[j] -= d
                    h = project_feasible(h, target)
                    
                    if new_c5 < best_new_c5 - 1e-15:
                        best_new_c5 = new_c5
                        best_move = (i, j, d)
                        best_improvement = current_max - new_c5
                        
            if best_move is not None and best_improvement > 0.001:
                break
        
        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = project_feasible(h, target)
        else:
            no_improve += 1
    
    return best_h, best_c5

def fine_local_search_v2(h, n, target, time_limit, top_k=50, check_k=100,
                         max_rounds=500000):
    start = time.time()
    h = h.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    dx = 2.0 / n
    n_half = n - 1
    
    delta_schedule = [0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.0025, 0.0015, 0.001]
    delta_idx = 0
    
    for round_count in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        current_max = float(np.max(overlaps))
        
        if current_max < best_c5 - 1e-14:
            best_c5 = current_max
            best_h = h.copy()
            no_improve = 0
        
        if no_improve > 500:
            if no_improve > 2000:
                break
            rng = np.random.RandomState((round_count * 7919 + 13) % (2**31))
            for _ in range(rng.randint(2, 6)):
                i = rng.randint(0, n)
                j = rng.randint(0, n)
                if i != j and h[i] > 0.05 and h[j] < 0.95:
                    d = min(0.02, h[i] * 0.4, (1.0 - h[j]) * 0.4)
                    h[i] -= d
                    h[j] += d
            h = project_feasible(h, target)
            no_improve = 0
            continue
        
        top_indices = np.argsort(overlaps)[::-1][:top_k]
        active_lags = top_indices - n_half
        
        candidates_i = np.where(h > 0.2)[0]
        candidates_j = np.where(h < 0.8)[0]
        
        if len(candidates_i) == 0 or len(candidates_j) == 0:
            no_improve += 1
            continue
        
        if len(candidates_i) > 100:
            scores_i = np.zeros(len(candidates_i))
            for ci, idx in enumerate(candidates_i):
                s = 0.0
                for kl in active_lags:
                    if kl >= 0:
                        if idx + kl < n:
                            s += (1.0 - h[idx + kl])
                        if idx - kl >= 0:
                            s -= h[idx - kl]
                    else:
                        m = -kl
                        if idx - m >= 0:
                            s += (1.0 - h[idx - m])
                        if idx + m < n:
                            s -= h[idx + m]
                scores_i[ci] = s
            top_i_mask = scores_i > np.median(scores_i)
            candidates_i = candidates_i[top_i_mask]
            if len(candidates_i) > 60:
                candidates_i = candidates_i[:60]
        
        if len(candidates_j) > 100:
            scores_j = np.zeros(len(candidates_j))
            for cj, idx in enumerate(candidates_j):
                s = 0.0
                for kl in active_lags:
                    if kl >= 0:
                        if idx + kl < n:
                            s += h[idx + kl]
                        if idx - kl >= 0:
                            s += h[idx - kl]
                    else:
                        m = -kl
                        if idx - m >= 0:
                            s += h[idx - m]
                        if idx + m < n:
                            s += h[idx + m]
                scores_j[cj] = s
            top_j_mask = scores_j > np.median(scores_j)
            candidates_j = candidates_j[top_j_mask]
            if len(candidates_j) > 60:
                candidates_j = candidates_j[:60]
        
        d = delta_schedule[min(delta_idx, len(delta_schedule) - 1)]
        if no_improve > 100:
            delta_idx = min(delta_idx + 1, len(delta_schedule) - 1)
        elif no_improve == 0:
            delta_idx = max(delta_idx - 1, 0)
        
        best_move = None
        best_new_c5 = current_max
        
        n_i = len(candidates_i)
        n_j = len(candidates_j)
        
        for ii in range(n_i):
            i = candidates_i[ii]
            if h[i] < d * 1.1:
                continue
            for jj in range(n_j):
                j = candidates_j[jj]
                if i == j or h[j] + d > 1.0:
                    continue
                
                bad = False
                for kl in active_lags:
                    if kl >= 0:
                        do = 0.0
                        if i + kl < n:
                            do += (1.0 - h[i + kl])
                        if j + kl < n:
                            do -= (1.0 - h[j + kl])
                        if i - kl >= 0:
                            do += -h[i - kl]
                        if j - kl >= 0:
                            do += h[j - kl]
                    else:
                        m = -kl
                        do = 0.0
                        if i - m >= 0:
                            do += (1.0 - h[i - m])
                        if j - m >= 0:
                            do -= (1.0 - h[j - m])
                        if i + m < n:
                            do += -h[i + m]
                        if j + m < n:
                            do += h[j + m]
                    if do * dx > -0.0005 * d:
                        bad = True
                        break
                
                if bad:
                    continue
                
                h[i] -= d
                h[j] += d
                h = project_feasible(h, target)
                new_c5 = compute_c5(h)
                h[i] += d
                h[j] -= d
                h = project_feasible(h, target)
                
                if new_c5 < best_new_c5 - 1e-15:
                    best_new_c5 = new_c5
                    best_move = (i, j, d)
        
        if best_move is not None:
            i, j, d_move = best_move
            h[i] -= d_move
            h[j] += d_move
            h = project_feasible(h, target)
        else:
            no_improve += 1
    
    return best_h, best_c5

def time_remaining(start_time, budget_s):
    return budget_s - 20.0 - (time.time() - start_time)
