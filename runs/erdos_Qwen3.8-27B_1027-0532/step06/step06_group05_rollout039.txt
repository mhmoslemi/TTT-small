    n = len(h)
    h = h.astype(float).copy()

    corr = np.correlate(h, 1-h, mode='full')

    best_h = h.copy()
    stall = 0
    
    while time.time() - start_time < start_time + time_budget:
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n-1)
        
        # Find candidates
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]

        i_cand = [i for i in ones if 0 <= i+k_star < n and h[i+k_star] < 0.5]
        # Best j: h[j]=0, h[j+k*]=1 (reduces overlap at k*)  
        # Wait, we need h[j+k*]=1 for the +(1-h[j+k*]) term to be 0
        # Actually we want +(1-h[j+k*]) to be small, so h[j+k*]=1
        j_cand = [j for j in zeros if 0 <= j+k_star < n and h[j+k_star] >= 0.5]

            # Try negative shift
            k_star = -k_star
            if k_star == 0:
                stall += 100
                continue
            i_cand = [i for i in ones if 0 <= i+k_star < n and h[i+k_star] < 0.5]
            j_cand = [j for j in zeros if 0 <= j+k_star < n and h[j+k_star] >= 0.5]
            if not i_cand or not j_cand:
                stall += 100
                continue

        best_improve = 0
        best_i, best_j = -1, -1
        
        for i in i_cand[:50]:  # Limit candidates
            for j in j_cand[:50]:
                # Compute delta for all shifts

                dk = 0.0
                if 0 <= i+k_star < n:

                if 0 <= i-k_star < n:

                if 0 <= j+k_star < n:
                    dk += (1-h[j+k_star])
                if 0 <= j-k_star < n:
                    dk -= h[j-k_star]

                if max_val + dk >= max_val - 0.01:  # Not a clear improvement

                # Full evaluation
                h[i] = 0
                h[j] = 1
                corr_new = np.correlate(h, 1-h, mode='full')
                new_max = float(corr_new.max())
                h[i] = 1
                h[j] = 0
                
                if max_val - new_max > best_improve:
                    best_improve = max_val - new_max
                    best_i, best_j = i, j
                    corr_new = corr_new  # Save for later
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1-h, mode='full')

    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-12 and float(np.min(h)) >= -1e-12 and float(np.max(h)) <= 1.0 + 1e-12:

        else:

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r

    grad = np.zeros(n)
