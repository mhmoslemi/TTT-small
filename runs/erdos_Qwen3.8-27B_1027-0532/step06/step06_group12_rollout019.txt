        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def get_subgrad(h, K):
    n = len(h)
    dx = 2.0 / n
    n_half = n - 1
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:K]
    g = np.zeros(n)
    for idx in top_idx:
        k = idx - n_half
        abs_k = abs(k)
        if abs_k == 0:
            g += (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n - k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n - k]) * dx
        else:
            m = -k
            g[m:] += (1.0 - h[:n - m]) * dx
            g[:n - m] += (-h[m:]) * dx
    return g / K, overlaps, top_idx

def bb_method(h_init, n, target, time_limit, K_start=20, K_end=100, max_iter=200000):
    """BB subgradient with increasing K and Nesterov momentum."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_prev = h.copy()
    g_prev = None
    m = np.zeros(n)
    no_improve = 0
    gamma = 0.005
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, t / max(1, max_iter * 0.5))
        K = int(K_start + (K_end - K_start) * progress)
        
        g, overlaps, _ = get_subgrad(h, K)
        cur = float(np.max(overlaps))
        if cur < best_c5 - 1e-14:
            best_c5 = cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if g_prev is not None:
            s = h - h_prev
            r = g - g_prev
            rr = np.dot(r, r)
            if rr > 1e-20:
                sr = np.dot(s, r)
                if sr > 1e-8:
                    gamma = min(max(sr / rr, 0.0001), 0.05)
                else:
                    ss = np.dot(s, s)
                    gamma = min(max(0.5 * ss / (abs(sr) + 1e-10), 0.0001), 0.03)
        
        # Momentum update
        mu = 0.85
        m = mu * m - gamma * g
        h = project_feasible(h + m, target)
        
        h_prev = h.copy()
        g_prev = g
        
        if no_improve > 80000 and t > 20000:
            break
    
    return best_h, best_c5

def fw_method(h_init, n, target, time_limit, K=50, max_iter=200000):
    """Frank-Wolfe with adaptive step."""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g, overlaps, top_idx = get_subgrad(h, K)
        cur = float(np.max(overlaps))
        if cur < best_c5 - 1e-14:
            best_c5 = cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # FW: find v = argmin <g, v> over feasible set
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        
        gamma = 2.0 / (t + 2)
        h = (1.0 - gamma) * h + gamma * v
        h = project_feasible(h, target)
        
        if no_improve > 100000:
            break
    
    return best_h, best_c5

def alternating_lp(h_init, n, target, max_iter, time_limit, n_top=300,
                   damping=0.85, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 50 and stuck > 100:
            break
        
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            abs_k = abs(k)
            if abs_k == 0:
                coeff = (1.0 - h) * dx
                constraints.append(coeff @ h_var <= t_var)
            elif k > 0:
                coeff = np.zeros(n)
                coeff[:n - k] = (1.0 - h[k:]) * dx
                coeff[n - k:] = 0.0  # h[k:] doesn't wrap
                constraints.append(coeff @ h_var <= t_var)
            else:
                m = -k
                coeff = np.zeros(n)
                coeff[m:] = (1.0 - h[:n - m]) * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-12, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter=3000):
    """SLSQP with log-sum-exp smoothing."""
    N = n
    TGT = target
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(h, alpha):
        n_local = len(h)
        dx = 2.0 / n_local
        n_half = n_local - 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        exp_vals = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        weights = exp_vals / np.sum(exp_vals)
        
        g = np.zeros(n_local)
        total = len(corr)
        for idx in range(total):
            w = weights[idx]
            if w < 1e-10:
                continue
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                g += w * (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n_local - k] += w * (1.0 - h[k:]) * dx
                g[k:] += w * (-h[:n_local - k]) * dx
            else:
                m = -k
                g[m:] += w * (1.0 - h[:n_local - m]) * dx
                g[:n_local - m] += w * (-h[m:]) * dx
        return g
    
    def con_fun(h):
        return np.sum(h) - TGT
    
    def con_jac(h):
        return np.ones(N)
    
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj_func(x, alpha), h_cur,
                          jac=lambda x: obj_grad(x, alpha),
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def upsample(h, factor):
    """Upsample h by a factor using linear interpolation."""
    n = len(h)
    n_new = n * factor
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, n_new)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    return h_new
