def qcd_single_step_fast(h, n, target, corr, n_cand=15, n_bisect=25):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    # Compute gradient for ranking
    k_star = int(np.argmax(corr)) - mid
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    
    sorted_idx = np.argsort(grad)
    low_set = sorted_idx[:n_cand]
    high_set = sorted_idx[-n_cand:]
    
    best_obj = float(np.max(corr)) * dx
    best_i = -1
    best_j = -1
    best_d = 0.0
    
    for i in high_set:
        hi = h[i]
        for j in low_set:
            if i == j:
                continue
            d_max = min(hi, 1.0 - h[j])
            if d_max < 1e-12:
                continue
            
            # Compute A vector: A[t] = -h[j-t] + h[i-t] + h[i+t] - h[j+t] (with 0 for OOB)
            # where t = s - mid, ranging from -(n-1) to (n-1)
            t = np.arange(-(n-1), n)  # all possible shifts
            # h[i+t] with 0 for OOB
            h_it = np.zeros(n_corr)
            mask_it = (i + t >= 0) & (i + t < n)
            h_it[mask_it] = h[i + t[mask_it]]
            h_jt = np.zeros(n_corr)
            mask_jt = (j + t >= 0) & (j + t < n)
            h_jt[mask_jt] = h[j + t[mask_jt]]
            h_it_neg = np.zeros(n_corr)
            mask_itn = (i - t >= 0) & (i - t < n)
            h_it_neg[mask_itn] = h[i - t[mask_itn]]
            h_jt_neg = np.zeros(n_corr)
            mask_jtn = (j - t >= 0) & (j - t < n)
            h_jt_neg[mask_jtn] = h[j - t[mask_jtn]]
            
            A = -h_jt_neg + h_it_neg + h_it - h_jt
            
            # B vector: B[mid] = -2, B[mid+(j-i)] = 1, B[mid+(i-j)] = 1
            B = np.zeros(n_corr)
            B[mid] = -2.0
            if mid + (j - i) != mid:  # j != i
