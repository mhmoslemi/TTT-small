def compute_delta(h, i, j, k_arr):

def nesterov_refine(h0, target, start_time, time_limit, alpha_schedule):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        lr = 1.0 / np.sqrt(alpha)
        lr = min(lr, 0.5)
        momentum = 0.85
        n_iter = max(200, int(30 * np.sqrt(alpha)))
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            obj, grad = smooth_obj_and_grad(h, alpha, corr)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 100 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    return best_h, best_score

            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                accept_count += 1
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            if focus_top > 0 and it % 6 == 0:
                top_indices = np.argsort(full)[::-1][:focus_top]
                move_found = False
                for ti in top_indices:
                    k_star = int(ti) - (n - 1)
                    if k_star == 0:
                        continue
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    i_mask = (h == 1.0) & (A > 0.5)
                    j_mask = (h == 0.0) & (B < 1.5)
                    i_cands = np.where(i_mask)[0]
                    j_cands = np.where(j_mask)[0]
                    if len(i_cands) > 0 and len(j_cands) > 0:
                        i = i_cands[rng.randint(len(i_cands))]
                        j = j_cands[rng.randint(len(j_cands))]
                        move_found = True
                        break
                if not move_found:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                accept_count += 1
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def sa_block_focus(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr,
                   max_block_len=20):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 40000
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        r = rng.random()
        if r < 0.4:
            blen = rng.randint(2, min(max_block_len, n // 6) + 1)
            a = rng.randint(0, n - blen + 1)
            block = h[a:a+blen].copy()
            if block.sum() == 0 or block.sum() == blen:
                stall += 1
                continue
            b = rng.randint(0, n - blen + 1)
            if abs(a - b) < blen:
                stall += 1
                continue
            target_block = h[b:b+blen]
            if target_block.sum() != block.sum():
                stall += 1
                continue
            h_new = h.copy()
            h_new[a:a+blen] = target_block
            h_new[b:b+blen] = block
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        elif r < 0.7:
            blen = rng.randint(3, min(max_block_len, n // 8) + 1)
            a = rng.randint(0, n - blen + 1)
            h_new = h.copy()
            h_new[a:a+blen] = h[a:a+blen][::-1]
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def greedy_improve(h, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    end_time = start_time + time_limit
    while time.time() < end_time:
        if stall > 500:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(40, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(40, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def multi_start_sa(h_base, n_restarts, t_init, rng, time_limit, start_time, k_arr):
    n = len(h_base)
    best_h = h_base.copy()
    best_score = compute_c5(h_base)
    end_time = start_time + time_limit
    for r in range(n_restarts):
        if time.time() > end_time:
            break
        h_pert = h_base.copy()
        n_pert = rng.randint(4, 10)
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        if r % 2 == 0:
            blen = rng.randint(5, 15)
            a = rng.randint(0, n - blen)
            h_pert[a:a+blen] = h_pert[a:a+blen][::-1]
        allowed = end_time - time.time()
        if allowed < 15:
            break
        per_alloc = allowed / max(1, n_restarts - r)
        n_sa = int(min(400000, per_alloc * 40000))
        h_sa, c5_sa = sa_adaptive(h_pert, n_sa, t_init * (0.9 ** r), rng,
                                  time_limit=per_alloc + time.time(),
                                  start_time=start_time, k_arr=k_arr,
                                  block_frac=0.2, max_block_len=12, focus_top=5)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_smooth_obj(h, alpha, corr_prev=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_box_sum(h_start.copy(), target)
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, curr_score

def variance_refine(h0, target, start_time, time_limit):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    lr = 0.05
    momentum = 0.7
    n_iter = int((time_limit - (time.time() - start_time)) * 80)
    n_iter = max(n_iter, 50)
    end_time = start_time + time_limit
    for it in range(n_iter):
        if time.time() > end_time:
            break
        corr = full_corr_fast(h)
        corr_mean = corr.mean()
        grad_v = 2.0 * (corr - corr_mean)
        N_fft = 1
        while N_fft < 3 * n - 2:
            N_fft <<= 1
        w_full = np.zeros(N_fft)
        w_full[:2*n-1] = grad_v
        w_rev = np.zeros(N_fft)
        w_rev[:2*n-1] = grad_v[::-1]
        h_pad = np.zeros(N_fft)
        h_pad[:n] = h
        H_fft = np.fft.fft(h_pad)
        conv1 = np.fft.ifft(np.fft.fft(w_full) * H_fft).real[:n]
        conv2 = np.fft.ifft(np.fft.fft(w_rev) * H_fft).real[:n]
        grad = conv1 + conv2
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_box_sum(h_new, target)
        h = h_new
        if it % 100 == 0:
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy (fast)
    if tleft() > 20:
        allowed = min(20, tleft() * 0.03)
        h_g, c5_g = greedy_improve(best_h, allowed, start_time, k_arr)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Variance reduction on continuous
    if tleft() > 120:
        target_use = float(n_init) / 2.0
        allowed = min(80, tleft() * 0.10)
        h_vr, sc_vr = variance_refine(best_h, target_use, start_time, allowed)
        if sc_vr < best_score:
            best_score = sc_vr
            best_h = h_vr.copy()

    # Phase 3: Nesterov continuous (main optimization)
    if tleft() > 150:
        target_use = float(n_init) / 2.0
        alpha_schedule = [8, 25, 80, 250, 800, 2500, 8000, 20000]
        allowed = min(250, tleft() * 0.22)
        h_ref, sc = nesterov_refine(best_h, target_use, start_time, allowed, alpha_schedule)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 4: Binary SA (adaptive)
    if tleft() > 100:
        h_bin = project_binary(best_h, target_init)
        allowed = min(180, tleft() * 0.18)
        n_sa = int(min(4000000, allowed * 55000))
        h_sa, c5_sa = sa_adaptive(h_bin, n_sa, 0.025, rng,
                                  time_limit=allowed + start_time,
                                  start_time=start_time, k_arr=k_arr,
                                  block_frac=0.22, max_block_len=14, focus_top=6)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Block-focused SA
    if tleft() > 80:
        h_bin = project_binary(best_h, target_init)
        allowed = min(150, tleft() * 0.15)
        n_sa = int(min(1500000, allowed * 25000))
        h_sa, c5_sa = sa_block_focus(h_bin, n_sa, 0.02, 0.0001, rng,
                                     time_limit=allowed + start_time,
                                     start_time=start_time, k_arr=k_arr,
                                     max_block_len=18)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Multi-start
    if tleft() > 80:
        h_bin = project_binary(best_h, target_init)
        allowed = min(150, tleft() * 0.14)
        h_ms, c5_ms = multi_start_sa(h_bin, n_restarts=7, t_init=0.018, rng=rng,
                                     time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 7: Second round of continuous optimization
    if tleft() > 80:
        target_use = float(n_init) / 2.0
        alpha_schedule2 = [50, 200, 800, 3000, 10000, 30000]
        allowed = min(120, tleft() * 0.12)
        h_ref2, sc2 = nesterov_refine(best_h, target_use, start_time, allowed, alpha_schedule2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 8: Binary SA after second continuous
    if tleft() > 60:
        h_bin = project_binary(best_h, target_init)
        allowed = min(80, tleft() * 0.08)
        n_sa = int(min(2000000, allowed * 50000))
        h_sa, c5_sa = sa_adaptive(h_bin, n_sa, 0.012, rng,
                                  time_limit=allowed + start_time,
                                  start_time=start_time, k_arr=k_arr,
                                  block_frac=0.15, max_block_len=10, focus_top=8)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 9: Resolution exploration
    if tleft() > 50:
        for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8, n_init + 12]:
            if tleft() < 20:
                break
            if n_try < 100 or n_try % 2 != 0:
                continue
            target_try = n_try // 2
            k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
            h_interp = interpolate_to(best_h, n_try, float(target_try))
            h_bin_try = project_binary(h_interp, target_try)
            allowed = min(12, tleft() * 0.08)
            n_sa = int(min(150000, allowed * 35000))
            h_st, c5_st = sa_adaptive(h_bin_try, n_sa, 0.015, rng,
                                      time_limit=allowed + time.time(),
                                      start_time=start_time, k_arr=k_arr_try,
                                      block_frac=0.2, max_block_len=10, focus_top=4)
            if c5_st < best_score:
                best_score = c5_st
                best_h = h_st.copy()
                n_init = n_try
                target_init = target_try
                k_arr = k_arr_try

    # Phase 10: SLSQP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_slsqp = [100, 500, 2000, 8000, 30000]
        allowed = min(tleft() - 30, 120)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas_slsqp,
                                 2000, time_limit=allowed + start_time,
                                 start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 11: Final SA burst
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        k_arr_f = np.arange(2 * n_use - 1) - (n_use - 1)
        h_bin_final = project_binary(best_h, target_use)
        allowed = max(15.0, tleft() - 12.0)
        n_sa_f = int(min(1200000, allowed * 60000))
        h_sa_f, c5_f = sa_adaptive(h_bin_final, n_sa_f, 0.006, rng,
                                   time_limit=allowed + time.time(),
                                   start_time=start_time, k_arr=k_arr_f,
                                   block_frac=0.1, max_block_len=8, focus_top=10)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 12: Very final polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [5000, 20000, 60000], 1500,
                                 time_limit=max(10.0, tleft() - 5.0) + start_time,
                                 start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
