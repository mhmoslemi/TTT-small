```python
def smooth_objective(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # grad[m] = sum_k weights[k] * ((1-h[m-k]) - h[m+k]) * dx
    # where k ranges from -(n-1) to n-1, weights indexed by k+(n-1)
    
    omh = 1.0 - h
    grad = np.zeros(n)
    
    for m in range(n):
        # First part: sum_{k: 0<=m-k<n} weights[k+n-1] * (1-h[m-k])
        # m-k in [0,n) → k in [m-n+1, m]
        k_lo = max(0, m-n+1)  # Wait, k ranges from -(n-1) to n-1, so k+n-1 ranges from 0 to 2n-2
        # k in [m-n+1, m] ∩ [-(n-1), n-1] = [max(m-n+1, -(n-1)), min(m, n-1)]
        k_lo = max(m-n+1, -(n-1))
        k_hi = min(m, n-1)
        if k_hi >= k_lo:
            k_vals = np.arange(k_lo, k_hi+1)
            h_idx = m - k_vals
            grad[m] += np.dot(weights[k_vals + n - 1], omh[h_idx])
        
        # Second part: sum_{k: 0<=m+k<n} weights[k+n-1] * h[m+k]
        # m+k in [0,n) → k in [-m, n-1-m]
        k_lo2 = max(-m, -(n-1))
        k_hi2 = min(n-1-m, n-1)
        if k_hi2 >= k_lo2:
            k_vals = np.arange(k_lo2, k_hi2+1)
            h_idx = m + k_vals
            grad[m] -= np.dot(weights[k_vals + n - 1], h[h_idx])
    
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad
```
