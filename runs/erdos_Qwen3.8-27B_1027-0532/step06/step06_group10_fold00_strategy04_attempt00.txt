    if abs(current - target) < 1e-12:
        return h
    # First adjustment: proportional change
    diff = target - current
    h = h + (diff / len(h))
    h = np.clip(h, 0, 1)
    # Final adjustment: locate extreme points
    if diff != 0:
        unsorted = np.abs([1-h, h])
        h = min(h.tolist(), key=lambda x: [min(x), max(x)][unsorted])
    return h
