We want to minimize max_k (corr[k] + delta * g[k]) over delta ∈ [0, dmax].

Let's define f(delta) = max_k (corr[k] + delta * g[k]). This is a convex piecewise-linear function of delta.

The minimum of a convex piecewise-linear function on [0, dmax] occurs at:
- delta = 0 if f'(0+) ≥ 0
- delta = dmax if f'(dmax-) ≤ 0
- Otherwise, at a breakpoint where f'(delta) changes sign

f'(delta) at a non-breakpoint = max_{k in active(delta)} g[k], where active(delta) = {k : corr[k] + delta*g[k] = f(delta)}.

A breakpoint occurs when two (or more) functions corr[k] + delta*g[k] are equal and are the maximum.

So the algorithm is:
1. Find the set of k that achieve the max at delta=0: A = {k: corr[k] = cur_max}
2. If max_{k in A} g[k] ≥ 0, the optimal is delta = 0 (no improvement)
3. Otherwise, find the largest delta where the max is still achieved by members of A. This is:
   delta_break = min over pairs (k1, k2) where k1 in A, k2 not in A, of (corr[k1] - corr[k2]) / (g[k2] - g[k1]) (when g[k1] < g[k2])
   Also consider delta_break from the constraint delta ≤ dmax.
4. At delta_break, the active set changes. Add the newly active k's and repeat.

This is the "lower envelope" algorithm for 1D convex piecewise-linear minimization. It's exact and fast (O(n_corr) per breakpoint, and there are at most n_corr breakpoints).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

_tgt = 0.0

def _cf(x):
    return np.sum(x) - _tgt

def _cj(x):
    return np.ones(len(x))

def lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400.0, 0.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = np.abs(corr)
    cmax = np.max(c)
    if cmax < 1e-30:
        return 0.0
    cn = c / cmax
    return float(cmax * np.power(np.sum(cn ** p), 1.0 / p))

def slsqp_opt(h0, n, tgt, obj_fn, params_list, maxit, ftol):
    global _tgt
    _tgt = tgt
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    for p in params_list:
        try:
            r = minimize(obj_fn, h_cur, args=(p,), method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': maxit, 'ftol': ftol})
            h_new = project_feasible(r.x, tgt)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_sc

def pair_cd(h0, n, tgt, tlim, maxsw, ncand):
    st = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    nc = 2 * n - 1
    noimp = 0
    idxs = np.arange(n)

    for sw in range(maxsw):
        if time.time() - st > tlim:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = float(np.max(corr))
        cv = cm * dx
        if cv < best_c5 - 1e-13:
            best_c5 = cv
            best_h = h.copy()
            noimp = 0
        else:
            noimp += 1
            if noimp > 400:
                break

        top = np.argsort(corr)[-3:]
        aw = 30.0
        w = np.exp(aw * (corr[top] - cm))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top):
            k = int(si) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        hi_set = np.argsort(-grad)[:ncand]
        lo_set = np.argsort(grad)[:ncand]
        all_c = np.unique(np.concatenate([hi_set, lo_set]))
        ncl = len(all_c)
        pos = np.full(n, -1, dtype=int)
        for p in range(ncl):
            pos[int(all_c[p])] = p

        profs = np.zeros((ncl, nc))
        for ci in range(ncl):
            m = int(all_c[ci])
            ks = np.arange(-(n - 1), n)
            i1 = m - ks
            i2 = m + ks
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            if np.any(v1):
                profs[ci, v1] += h[i1[v1]]
            if np.any(v2):
                profs[ci, v2] += h[i2[v2]]

        best_imp = 0.0
        best_mv = None

        for i in hi_set:
            hv = float(h[i])
            if hv < 1e-12:
                continue
            ci = pos[int(i)]
            if ci < 0:
                continue
            for j in lo_set:
                if int(i) == int(j):
                    continue
                htv = float(h[j])
                dm = min(hv, 1.0 - htv)
                if dm < 1e-12:
                    continue
                cj = pos[int(j)]
                if cj < 0:
                    continue
                g = profs[ci] - profs[cj]
                lo = 0.0
                hidd = dm
                for _ in range(15):
                    mid = 0.5 * (lo + hidd)
                    vals = corr + mid * g
                    fm = float(np.max(vals))
                    act = vals >= (fm - 1e-14)
                    if not np.any(act):
                        break
                    fp = float(np.max(g[act]))
                    if fp > 1e-14:
                        hidd = mid
                    else:
                        lo = mid
                ds = 0.5 * (lo + hidd)
                if ds < 1e-10:
                    continue
                nm = float(np.max(corr + ds * g))
                imp = cm - nm
                if imp > best_imp:
                    best_imp = imp
                    best_mv = (int(i), int(j), ds)

        if best_mv is not None:
            i, j, d = best_mv
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, tgt)

    return best_h, best_c5

def multi_shift_subgrad(h0, n, tgt, tlim, max_steps, n_shifts, base_lr):
    st = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cm = float(np.max(corr))
        cn = cm * dx
        if cn < best_c5 - 1e-13:
            best_c5 = cn
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_shifts:]
        aw = 25.0
        w = np.exp(aw * (corr[top_idx] - cm))
        w /= w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, tgt)
            c5_try = compute_c5(h_try)
            if c5_try < cn - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 800:
            break

    return best_h, best_c5

def interp_to(hs, nd, td):
    ns = len(hs)
    if ns == nd:

    xo = np.linspace(0.0, 1.0, ns)
    return project_feasible(np.interp(xn, xo, hs), td)
