def project_box_sum(h, target):
    n = len(h)
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    while int(h_r.sum()) != target_sum:
        if h_r.sum() > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
    return h_r.astype(float)

        d += 1.0
    if b + t == a:
        d += 1.0
    return d

def greedy_multi_lag(h, target_sum, n_critical, time_limit, start_time, rng):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200
    tabu = {}
    tabu_len = 50

    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        if it > 0 and it % tabu_len == 0:
            tabu = {}

        max_val = float(corr.max())
        threshold = max_val * 0.85
        critical_idx = np.where(corr >= threshold)[0]
        if len(critical_idx) == 0:
            critical_idx = np.array([int(np.argmax(corr))])
        critical_idx = critical_idx[:n_critical]
        critical_lags = critical_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        relevant_ones = []
        for i in ones:
            if any(0 <= i + t < n or 0 <= i - t < n for t in critical_lags):
                relevant_ones.append(i)
        relevant_zeros = []
        for j in zeros:
            if any(0 <= j + t < n or 0 <= j - t < n for t in critical_lags):
                relevant_zeros.append(j)

        if len(relevant_ones) == 0 or len(relevant_zeros) == 0:
            stall += 10
            continue

        max_ones = min(len(relevant_ones), 60)
        max_zeros = min(len(relevant_zeros), 60)
        sel_ones = relevant_ones[:max_ones]
        sel_zeros = relevant_zeros[:max_zeros]

        best_improve = 0.0
        best_a = -1
        best_b = -1

        for a in sel_ones:
            if (a, a) in tabu:
                continue
            for b in sel_zeros:
                if (b, a) in tabu:
                    continue
                new_max = max_val
                feasible = True
                for t in critical_lags:
                    d = swap_effect_on_lag(h, a, b, t, n)
                    v = corr[n - 1 + t] + d
                    if v > new_max:
                        feasible = False
                        break
                if feasible:
                    improve = max_val - new_max
                    if improve > best_improve:
                        best_improve = improve
                        best_a = a
                        best_b = b

        if best_a >= 0:
            h[best_a] = 0
            h[best_b] = 1
            corr = compute_corr(h)
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            tabu[(best_b, best_a)] = it
        else:
            stall += 1

    return best_h, best_c5

def greedy_single_lag(h, target_sum, time_limit, start_time, scan_width=40):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300

    for it in range(500000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        top_threshold = max_val * 0.75
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        top_indices = top_indices[:30]

        i_cand = [i for i in ones if (0 <= i + k_star < n and h[i + k_star] == 0) or (0 <= i - k_star < n and h[i - k_star] == 1)]
        j_cand = [j for j in zeros if (0 <= j + k_star < n and h[j + k_star] == 1) or (0 <= j - k_star < n and h[j - k_star] == 0)]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 5
            continue

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        best_improve = 0.0
        best_i = -1
        best_j = -1

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    if i + k == j:
                        d += 1.0
                    if j + k == i:
                        d += 1.0
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = compute_corr(h)
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def sa_binary(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, focus_on_worst=True):
    n = len(h_start)
    h = h_start.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 3000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.5:
            if focus_on_worst:
                top_k = np.argsort(corr)[-10:]
                worst = top_k[rng.randint(len(top_k))]
            else:
                worst = rng.randint(2 * n - 1)
            k = worst - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.80:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        elif r < 0.92:
            if len(ones) >= 2 and len(zeros) >= 2:
                idx_o = rng.choice(len(ones), 2, replace=False)
                idx_z = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[idx_o[0]]] = 0
                h_new[ones[idx_o[1]]] = 0
                h_new[zeros[idx_z[0]]] = 1
                h_new[zeros[idx_z[1]]] = 1
                moved = True
        else:
            block_len = rng.randint(2, 6)
            if len(ones) > block_len:
                start_i = rng.randint(0, n - block_len)
                block_ones = [i for i in range(start_i, start_i + block_len) if h[i] == 1]
                if len(block_ones) >= 2:
                    start_j = rng.randint(0, n - block_len)
                    block_zeros = [j for j in range(start_j, start_j + block_len) if h[j] == 0]
                    n_swap = min(len(block_ones), len(block_zeros), 3)
                    for s in range(n_swap):
                        h_new[block_ones[s]] = 0
                        h_new[block_zeros[s]] = 1
                    moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def iterated_local_search(h_start, target_sum, rng, time_limit, start_time, n_iterations=10):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    for iter_idx in range(n_iterations):
        if time.time() - start_time > time_limit:
            break

        perturb_size = rng.randint(5, 20)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(perturb_size):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        remaining = time_limit - (time.time() - start_time)
        if remaining < 10:
            break

        alloc = remaining / (n_iterations - iter_idx) * 0.8
        time_alloc = min(alloc, 60.0)

        h_ls, c5_ls = greedy_single_lag(h_pert, target_sum, time_limit=time_alloc,
                                         start_time=start_time, scan_width=30)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    return best_h, best_c5

def continuous_optimize(h_start, target, beta, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        shifted = beta * (corr - c_max)
        shifted -= shifted.max()
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
        d = float(np.sum(h)) - target
        val += penalty * d * d
        return val * dx

    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        shifted = beta * (corr - c_max)
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()

        t_arr = np.arange(n)
        one_minus_h = 1.0 - h
        g = np.zeros(n)
        for j in range(n):
            idx1 = j - t_arr + n - 1
            idx2 = t_arr - j + n - 1
            g[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
        g *= dx
        if penalty > 0:
            g += 2.0 * penalty * (float(np.sum(h)) - target) * dx
        return g

    h_cur = h_start.copy()
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15,
                                   'gtol': 1e-10, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            h_obj = obj(h_candidate)
            h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin_init = project_binary(h_init, target_int)
    c5_bin_init = compute_c5(h_bin_init)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()

    # Phase 1: Quick greedy from binary init
    if tleft() > 30:
        time_alloc = min(30, tleft() * 0.1)
        h_g, c5_g = greedy_single_lag(h_bin_init, target_int, time_limit=time_alloc,
                                       start_time=start_time, scan_width=30)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Continuous optimization to find good starting points
    h_cont = project_box_sum(best_h.copy(), target)
    for beta in [20, 50, 100, 200, 500]:
        if tleft() < 40:
            break
        time_alloc = min(30, tleft() * 0.08)
        h_opt = continuous_optimize(h_cont, target, beta, 500.0, 200,
                                     time_alloc, start_time)
        h_cont = h_opt
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()
        h_bin_try = project_binary(h_opt, target_int)
        sc_bin = compute_c5(h_bin_try)
        if sc_bin < best_c5:
            best_c5 = sc_bin
            best_h = h_bin_try.copy()

    # Phase 3: Multi-lag greedy (different from previous single-lag)
    if tleft() > 40:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(60, tleft() * 0.12)
        h_ml, c5_ml = greedy_multi_lag(h_bin, target_int, n_critical=15,
                                        time_limit=time_alloc, start_time=start_time, rng=rng)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()

    # Phase 4: Iterated local search
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(200, tleft() * 0.2)
        h_ils, c5_ils = iterated_local_search(h_bin, target_int, rng,
                                               time_limit=time_alloc,
                                               start_time=start_time,
                                               n_iterations=15)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()

    # Phase 5: SA with focus on worst lags
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(200, tleft() * 0.2)
        max_it = int(min(150000, time_alloc * 3000))
        h_sa, c5_sa = sa_binary(h_bin, target_int, max_it, 0.006, 0.00001,
                                 rng, time_limit=time_alloc, start_time=start_time,
                                 focus_on_worst=True)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: More ILS with different perturbation sizes
    if tleft() > 60:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.15)
        h_ils2, c5_ils2 = iterated_local_search(h_bin, target_int, rng,
                                                  time_limit=time_alloc,
                                                  start_time=start_time,
                                                  n_iterations=12)
        if c5_ils2 < best_c5:
            best_c5 = c5_ils2
            best_h = h_ils2.copy()

    # Phase 7: SA with different parameters
    if tleft() > 60:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.15)
        max_it = int(min(120000, time_alloc * 2500))
        h_sa2, c5_sa2 = sa_binary(h_bin, target_int, max_it, 0.003, 0.000005,
                                    rng, time_limit=time_alloc, start_time=start_time,
                                    focus_on_worst=False)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 8: Multi-resolution
    n_options = [n - 4, n + 4, n - 8, n + 8, n + 12]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]

    for n_try in n_options:
        if tleft() < 40:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)

        time_alloc = min(25, tleft() * 0.1)
        h_opt = continuous_optimize(h_interp, target_try, 50, 300.0, 150,
                                     time_alloc, start_time)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

        if tleft() < 25:
            continue
        h_bin_try = project_binary(h_opt, target_try_int)
        time_alloc2 = min(20, tleft() * 0.1)
        h_ls2, c5_ls2 = greedy_single_lag(h_bin_try, target_try_int,
                                           time_limit=time_alloc2,
                                           start_time=start_time, scan_width=25)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 9: Final multi-lag greedy + SA polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        
        time_alloc = min(30, tleft() * 0.4)
        h_ml_f, c5_ml_f = greedy_multi_lag(h_bin_final, target_use, n_critical=20,
                                            time_limit=time_alloc, start_time=start_time, rng=rng)
        if c5_ml_f < best_c5:
            best_c5 = c5_ml_f
            best_h = h_ml_f.copy()

    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(25, tleft() * 0.5)
        max_it = int(min(80000, time_alloc * 2000))
        h_sf, c5_sf = sa_binary(h_bin_final, target_use, max_it, 0.002, 0.000005,
                                 rng, time_limit=time_alloc, start_time=start_time,
                                 focus_on_worst=True)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
