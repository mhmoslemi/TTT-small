            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand_low]
        high_set = sorted_idx[-n_cand_high:]
        
        best_delta = 0.0
        best_new_c5 = c5_val
        best_i = -1
        best_j = -1
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                # Try a range of delta values
                max_d = min(h[i], 1.0 - h[j]) * 0.5
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                    d = max_d * frac
                    if d < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                    h_try = project_feasible(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_new_c5 - 1e-14:
                        best_new_c5 = c5_try
                        best_delta = d
                        best_i = i
                        best_j = j
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)
    
    return best_h, best_c5
