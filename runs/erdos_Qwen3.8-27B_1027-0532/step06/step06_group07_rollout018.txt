    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

def project_feasible(h, target_sum):
    h = h.astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_out = np.clip(h + lam, 0.0, 1.0)
    residual = target_sum - np.sum(h_out)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h_out < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h_out > 1e-15)[0]

            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            delta_new = approx_max - c5_val
            if delta_new < -1e-15:
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
            elif delta_new < T and rng.random() < np.exp(-delta_new / T):
                if approx_max < best_trial_c5 + T:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_restart_cd(h_init, n, target, time_limit, seed_val=42):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    n_scales = [0.02, 0.05, 0.1, 0.15, 0.25]
    for scale in n_scales:
        if time.time() - start > time_limit - 30:
            break
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(60, (time_limit - (time.time() - start)) * 0.3)
        if cd_t < 10:
            break
        h_try, c5_try = enhanced_pair_cd(h_pert, n, target, time_limit=cd_t,
                                         max_sweeps=1500, n_cand=40, n_bisect=16)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _g_alpha
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient, gradual alpha
    if tleft() > 150:
        alphas = [2**i for i in range(3, 15)]
        _g_alpha = alphas[0]
        h_pol = slsqp_round_analytic(best_h, n, target, alphas, 400)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Projected gradient with smooth objective
    if tleft() > 200:
        alphas_pg = [2**i for i in range(6, 16)]
        h_pg, c5_pg = projected_gradient_smooth(best_h, n, target,
                                                 alphas_pg, 200, start_time,
                                                 time_limit=budget_s - 40)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 3: Enhanced pair CD (main workhorse)
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.35)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                        time_limit=cd_time,
                                        max_sweeps=8000, n_cand=60, n_bisect=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP polish after CD
    if tleft() > 100:
        alphas2 = [2**i for i in range(10, 18)]
        h_pol = slsqp_round_analytic(best_h, n, target, alphas2, 500)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Simulated annealing CD for escape
    if tleft() > 150:
        sa_time = min(200, tleft() * 0.3)
        h_sa, c5_sa = simulated_annealing_cd(best_h, n, target,
                                              time_limit=sa_time,
                                              max_sweeps=5000, n_cand=45,
                                              n_bisect=16, T_init=0.003,
                                              T_final=0.00005)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 6: Multi-restart with perturbations
    if tleft() > 150:
        mr_time = min(200, tleft() * 0.35)
        h_mr, c5_mr = multi_restart_cd(best_h, n, target,
                                        time_limit=mr_time, seed_val=seed)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 7: Try higher resolution n=800
    if tleft() > 250:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_h = [2**i for i in range(6, 14)]
        h_hires = slsqp_round_analytic(h_interp, n_try, target_try, alphas_h, 300)
        sc_h = compute_c5(h_hires)
        if tleft() > 120 and sc_h < best_score + 0.002:
            cd_t = min(100, tleft() * 0.3)
            h_hires2, sc_h2 = enhanced_pair_cd(h_hires, n_try, target_try,
                                                time_limit=cd_t,
                                                max_sweeps=3000, n_cand=50,
                                                n_bisect=18)
            if sc_h2 < best_score:
                best_score = sc_h2
                best_h = h_hires2
                n = n_try
                target = target_try

    # Phase 8: Try n=1000 if time permits
    if tleft() > 250:
        n_try = 1000
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_h = [2**i for i in range(8, 15)]
        h_hires = slsqp_round_analytic(h_interp, n_try, target_try, alphas_h, 250)
        sc_h = compute_c5(h_hires)
        if tleft() > 100 and sc_h < best_score + 0.002:
            cd_t = min(100, tleft() * 0.3)
            h_hires2, sc_h2 = enhanced_pair_cd(h_hires, n_try, target_try,
                                                time_limit=cd_t,
                                                max_sweeps=2500, n_cand=45,
                                                n_bisect=18)
            if sc_h2 < best_score:
                best_score = sc_h2
                best_h = h_hires2
                n = n_try
                target = target_try

    # Phase 9: Final CD at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.4)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n_use, target_use,
                                        time_limit=cd_time,
                                        max_sweeps=3000, n_cand=65, n_bisect=22)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_final = [2**i for i in range(14, 20)]
        h_final = slsqp_round_analytic(best_h, n_use, target_use,
                                        alphas_final, 400)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, len(h_out)
