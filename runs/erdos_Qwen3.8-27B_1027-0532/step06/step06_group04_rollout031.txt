    """Compute gradients of f_k for active k values.
    Returns array of shape (n_active, n) where grad[k_idx, j] = df_k/dh[j]."""

    half_n = n - 1

    grads = np.zeros((n_active, n))

def project_feasible(h, target_sum):

        return h

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_linearized_lp(h, target, n, k_indices):
    dx = 2.0 / n
    half_n = n - 1
    idxs = np.arange(n)
    m = len(k_indices)

    A_grad = np.zeros((m, n))
    b_offset = np.zeros(m)

    for i, ki in enumerate(k_indices):
        k = ki - half_n
        vd = (idxs - k >= 0) & (idxs - k < n)
        vu = (idxs + k >= 0) & (idxs + k < n)
        A_grad[i, vd] += (1.0 - h[idxs[vd] - k])
        A_grad[i, vu] -= h[idxs[vu] + k]
        A_grad[i] *= dx
        lo = max(0, -k)
        hi = min(n, n - k)
        f_val = np.sum(h[lo + k:hi + k] * (1.0 - h[lo:hi])) * dx
        b_offset[i] = A_grad[i] @ h - f_val

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, :n] = A_grad
    A_ub[:m, n] = -1.0
    b_ub[:m] = b_offset
    A_ub[m:m + n, :n] = -np.eye(n)
    b_ub[m:m + n] = 0.0
    A_ub[m + n:m + 2 * n, :n] = np.eye(n)
    b_ub[m + n:m + 2 * n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-12})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def linearized_prog(h_init, target, n, time_limit, top_k=300):
    start_t = time.time()
    dx = 2.0 / n
    half_n = n - 1
    n_total = 2 * n - 1

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    k_indices = np.argsort(corr)[-top_k:]
    k_indices = np.sort(k_indices)

    stall = 0
    for it in range(200000):
        if time.time() - start_t > time_limit:
            break

        h_lp = solve_linearized_lp(h, target, n, k_indices)
        if h_lp is None:
            break
        h_lp = np.clip(h_lp, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)

        cur_c5 = compute_c5(h)
        best_alpha = -1.0
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_try = (1.0 - alpha) * h + alpha * h_lp
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                best_alpha = alpha
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.try if hasattr(h, 'try') else h_tr
                    best_h = h_try.copy()
                break

        if best_alpha == -1.0:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0

        if it % 80 == 79:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_cur = float(np.max(corr)) * dx
            threshold = c5_cur - 0.003
            active_all = np.where(corr * dx >= threshold)[0]
            if len(active_all) > top_k * 2:
                active_all = np.argsort(corr * dx)[-top_k * 2:]
            k_indices = np.sort(active_all)

    return best_h, best_c5

def coord_descent(h_init, target, n, time_limit, n_active=300):
    start_t = time.time()
    dx = 2.0 / n
    half_n = n - 1

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for it in range(500000):
        if time.time() - start_t > time_limit:
            break

        for i in range(n):
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] * dx - A * h[i]

            lo = 0.0
            hi = 1.0
            for _ in range(50):
                mid = 0.5 * (lo + hi)
                vals = slopes * mid + intercepts
                max_val = np.max(vals)
                tol = 1e-14 * max(1.0, abs(max_val))
                active = np.abs(vals - max_val) < tol
                if not np.any(active):
                    active = np.zeros(n_active, dtype=bool)
                    active[np.argmax(vals)] = True
                if np.all(slopes[active] > 1e-14):
                    hi = mid
                elif np.all(slopes[active] < -1e-14):
                    lo = mid
                else:
                    break
            h[i] = 0.5 * (lo + hi)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        if it % 100 == 99:
            new_active = np.argsort(corr)[-n_active:]
            if not np.array_equal(np.sort(new_active), np.sort(active_k)):
                active_k = new_active

    return best_h, best_c5

def subgradient_step(h, target, n, time_limit):
    start_t = time.time()
    dx = 2.0 / n
    half_n = n - 1
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    idxs = np.arange(n)
    stall = 0

    for step in range(20000):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - half_n
        idx_v1 = (idxs - k >= 0) & (idxs - k < n)
        idx_v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[idx_v1] += (1.0 - h_cur[idxs[idx_v1] - k])
        grad[idx_v2] -= h_cur[idxs[idx_v2] + k]
        grad *= dx

        improved = False
        for lr in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def smooth_max_obj(h, alpha, corr):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, target, n, alpha, maxiter):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max_obj(h, alpha, corr)
    def cons_fun(h):
        return np.array([np.sum(h) - target])
    def cons_jac(h):
        return np.array([np.ones(n)])
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Linearized LP with all active constraints
    if tleft() > 200:
        lp_time = min(300, tleft() * 0.3)
        h_lp, c5_lp = linearized_prog(best_h, target, n, lp_time, top_k=400)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent polish
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.25)
        h_cd, c5_cd = coord_descent(best_h, target, n, cd_time, n_active=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Another LP round
    if tleft() > 150:
        lp_time = min(180, tleft() * 0.25)
        h_lp, c5_lp = linearized_prog(best_h, target, n, lp_time, top_k=500)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP with moderate smoothness
    if tleft() > 120:
        h_ref = slsqp_refine(best_h, target, n, alpha=200, maxiter=3000)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Coordinate descent again
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.2)
        h_cd, c5_cd = coord_descent(best_h, target, n, cd_time, n_active=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Try n=600
    if tleft() > 300:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        if tleft() > 200:
            lp_time = min(150, tleft() * 0.3)
            h_lp600, c5_lp600 = linearized_prog(h_interp, target_try, n_try, lp_time, top_k=400)
            if c5_lp600 < best_score:
                best_score = c5_lp600
                best_h = h_lp600
                n = n_try
                target = target_try
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp
                n = n_try
                target = target_try
        if tleft() > 100 and len(best_h) == n_try:
            cd_time = min(80, tleft() * 0.3)
            h_cd, c5_cd = coord_descent(best_h, target, n, cd_time, n_active=500)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 7: Subgradient refinement
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_step(best_h, target_use, n_use, min(60, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: LP with perturbed start
    if tleft() > 150:
        for noise_level in [0.03, 0.08, 0.15]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_level
            h_pert = project_feasible(h_pert, target)
            n_use = len(best_h)
            target_use = n_use / 2.0
            lp_time = min(60, tleft() * 0.3)
            h_lp, c5_lp = linearized_prog(h_pert, target_use, n_use, lp_time, top_k=300)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    # Phase 9: Final coordinate descent
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_use, cd_time, n_active=600)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final LP
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(40, tleft() * 0.6)
        h_lp, c5_lp = linearized_prog(best_h, target_use, n_use, lp_time, top_k=600)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final subgradient
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_step(best_h, target_use, n_use, min(20, tleft() * 0.6))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final: SLSQP high alpha
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_refine(best_h, target_use, n_use, alpha=5000, maxiter=1500)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
