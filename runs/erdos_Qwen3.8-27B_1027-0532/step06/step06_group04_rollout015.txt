def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_active_set(h, n_active, min_margin=1e-6):
    n = len(h)
    half_n = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')

    c5_val = np.max(corr) * dx
    # Return indices of top n_active constraints
    top_idx = np.argsort(corr)[-n_active:]
    return top_idx, c5_val, corr

    # Initial active set

            h_var <= 1.0

            hi = min(n, k + n)
            if hi > lo:

                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        # df_k/dh[j] = dx * [(1-h[j+k]) - h[j-k]] (with bounds)
        j = np.arange(n)
        valid1 = (j + k >= 0) & (j + k < n)
        valid2 = (j - k >= 0) & (j - k < n)

        dfdj[valid1] += (1.0 - h[j[valid1] + k])
        dfdj[valid2] -= h[j[valid2] - k]
        dfdj *= dx
        grad += weight[k_idx] * dfdj
    
    return obj_val, grad

_g_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_sub(h_fix, g, n, target, active_k_list, reg):
    dx = 2.0 / n
    half_n = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k_idx in active_k_list:
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    objective = t_var + reg * cp.sum_squares(h_var - h_fix)
    prob = cp.Problem(cp.Minimize(objective), constraints)
    solvers = [
        (cp.GLOP, {}),
        (cp.SCS, {'eps': 1e-12, 'max_iters': 200000}),
        (cp.ECOS, {'max_iter': 200000})
    ]
    for solver, kwargs in solvers:
        try:
            prob.solve(solver=solver, verbose=False, **kwargs)
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        except Exception:
            continue
    return None

def alternating_lp(h_init, n, target, max_iter, time_limit, n_act_init=40, n_act_max=120):
