def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()

    h = np.clip(h, 0.0, 1.0)

        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def _smooth_grad(h, alpha):

def pair_cd_v2(h_init, target, time_limit, max_iters, n_high, n_low, top_k, n_skip):
    """
    Fast pair coordinate descent.
    n_high: number of high-gradient candidates
    n_low: number of low-gradient candidates  
    top_k: number of top shifts to consider for gradient
    n_skip: recompute corr from scratch every n_skip iterations
    """
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    
    start_t = time.time()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        
        # Recompute corr periodically to avoid drift
        if it > 0 and it % n_skip == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        c_max = float(np.max(corr))
        c5_val = c_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1000:
                break
        
        # Find top shifts
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Compute weighted gradient
        alpha_w = 50.0
        top_vals = corr[top_idx] - c_max
        w = np.exp(alpha_w * top_vals)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - (n - 1)
            ip = idxs + k
            vp = (ip >= 0) & (ip < n)
            grad[vp] += wi * (1.0 - h[ip[vp]])
            im = idxs - k
            vn = (im >= 0) & (im < n)
            grad[vn] -= wi * h[im[vn]]
        
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_low]
        high_set = sorted_idx[-n_high:]
        
        # Compute effects for candidates
        # effect[i] = vector of shape (n_corr,)
        # effect[i][k+mid] = (1-h[i+k])_{valid} - h[i-k]_{valid}
        n_cand = len(high_set) + len(low_set)
        cand_idx = np.concatenate([high_set, low_set])
        
        # Vectorized effect computation
        # For each candidate i, compute the effect vector
        # effect[i, k+mid] = (1-h[i+k]) if 0≤i+k<n, else 0
        #                        - h[i-k] if 0≤i-k<n, else 0
        k_range = np.arange(-(n-1), n)  # shape (n_corr,)
        
        # For all candidates at once:
        # cand_idx shape: (n_cand,)
        # i + k: shape (n_cand, n_corr)
        ip_all = cand_idx[:, None] + k_range[None, :]  # (n_cand, n_corr)
        im_all = cand_idx[:, None] - k_range[None, :]  # (n_cand, n_corr)
        
        vp_all = (ip_all >= 0) & (ip_all < n)  # (n_cand, n_corr)
        vn_all = (im_all >= 0) & (im_all < n)  # (n_cand, n_corr)
        
        # Get h values
        # h[ip_all] where valid
        h_pos = np.where(vp_all, h[ip_all.clip(0, n-1)], 0.0)
        h_neg = np.where(vn_all, h[im_all.clip(0, n-1)], 0.0)
        
        effects = (1.0 - h_pos) * vp_all - h_neg * vn_all  # (n_cand, n_corr)
        
        # Now try pairs: (i in high, j in low)
        # high_set is first n_high of cand_idx, low_set is next n_low
        # effects for high: effects[:n_high]
        # effects for low: effects[n_high:]
        
        # For pair (hi, lo): d = effects[hi] - effects[lo]
        # We want to find the best pair.
        
        # Quick score: for pair (i,j), the expected improvement is related to
        # how well d "aligns" with reducing the max.
        # A simple proxy: -min_k d[k] for active k (we want d to be negative at active k)
        # Even simpler: just try the top few pairs by gradient difference.
        
        # For efficiency, try at most max(n_high * n_low, 100) pairs but cap at reasonable number
        n_pairs_to_try = min(n_high * n_low, 200)
        
        # Score pairs by gradient difference
        scores = np.zeros((n_high, n_low))
        for a in range(n_high):
            for b in range(n_low):
                scores[a, b] = grad[cand_idx[a]] - grad[cand_idx[n_high + b]]
        
        # Get top pairs
        flat_scores = scores.flatten()
        n_actual = min(n_pairs_to_try, len(flat_scores))
        top_pairs = np.argpartition(flat_scores, -n_actual)[-n_actual:]
        
        best_pair_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_d = None
        
        for pi in top_pairs:
            a = int(pi // n_low)
            b = int(pi % n_low)
            i = int(cand_idx[a])
            j = int(cand_idx[n_high + b])
            
            if i == j:
                continue
            
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(h[i], 1.0 - h[j])
            if d_hi - d_lo < 1e-12:
                continue
            
            d = effects[a] - effects[n_high + b]
            
            # Bisection for optimal delta
            a_bis, b_bis = d_lo, d_hi
            for _ in range(35):
                mid_d = 0.5 * (a_bis + b_bis)
                vals = corr + mid_d * d
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-13
                if np.max(d[active]) > 1e-13:
                    b_bis = mid_d
                elif np.min(d[active]) < -1e-13:
                    a_bis = mid_d
                else:
                    break
            
            delta_opt = 0.5 * (a_bis + b_bis)
            if abs(delta_opt) < 1e-13:
                continue
            
            c5_try = np.max(corr + delta_opt * d) * dx
            if c5_try < best_pair_c5:
                best_pair_c5 = c5_try
                best_i = i
                best_j = j
                best_delta = delta_opt
                best_d = d
        
        if best_i >= 0:
            h[best_i] += best_delta
            h[best_j] -= best_delta
            np.clip(h, 0.0, 1.0, out=h)
            corr += best_delta * best_d
            # Quick feasibility fix
            s = np.sum(h)
            if abs(s - target) > 1e-10:
                diff = target - s
                # Distribute among non-boundary elements
                mask = (h > 1e-8) & (h < 1.0 - 1e-8)
                if mask.any():
                    h[mask] += diff / mask.sum()
                    np.clip(h, 0.0, 1.0, out=h)

                    corr = np.correlate(h, 1.0 - h, mode='full')
                else:
                    h = project_feasible(h, target)
                    corr = np.correlate(h, 1.0 - h, mode='full')
    
    return best_h, best_c5
