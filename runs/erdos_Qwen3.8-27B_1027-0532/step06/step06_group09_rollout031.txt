    if kmax1 >= 0:
        ks = np.arange(0, kmax1 + 1)
        e[mid + ks] += (1.0 - h[i - ks])
    kmax2 = min(n - 1 - i, n - 1)
    if kmax2 >= 0:
        ks = np.arange(0, kmax2 + 1)
        e[mid + ks] -= h[i + ks]
    # Negative shifts k=0..n-1: e[mid-k] = (1-h[i+k])*[i+k<n] - h[i-k]*[i>=k]
    if kmax2 >= 0:
        e[mid - ks] += (1.0 - h[i + ks])
    if kmax1 >= 0:
        ks1 = np.arange(0, kmax1 + 1)
        e[mid - ks1] -= h[i - ks1]
    return e

def pair_cd_fast(h0, n, target, time_limit, n_cand=30, patience=200):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        k_star = int(np.argmax(corr)) - (n - 1)
        
        # Compute gradient for the worst shift
        if k_star >= 0:
            grad = np.zeros(n)
            k = k_star
            if k == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                grad[k:] = (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            s = -k_star
            grad = np.zeros(n)
            grad[:n-s] = (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Compute effect vectors
        hi_list = [int(x) for x in high_set]
        lo_list = [int(x) for x in low_set]
        
        hi_effects = [effect_vector(i, h, n) for i in hi_list]
        lo_effects = [effect_vector(j, h, n) for j in lo_list]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_dc = 0.0
        
        for ai, i in enumerate(hi_list):
            ei = hi_effects[ai]
            for aj, j in enumerate(lo_list):
                if i == j:
                    continue
                g = ei - lo_effects[aj]
                
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-14:
                    continue
                
                # Binary search
                a_d, b_d = 0.0, d_max
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-13
                    g_act = g[active]
                    if np.min(g_act) > 1e-14:
                        b_d = mid_d
                    elif np.max(g_act) < -1e-14:
                        a_d = mid_d
                    else:
                        break
                d_star = 0.5 * (a_d + b_d)
                if d_star < 1e-14:
                    continue
                
                trial = np.max(corr + d_star * g) * dx
                if trial < best_trial - 1e-14:
                    best_trial = trial
                    best_i = i
                    best_j = j
                    best_dc = d_star
        
        if best_i >= 0 and best_dc > 0:
            h[best_i] -= best_dc
            h[best_j] += best_dc
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return
