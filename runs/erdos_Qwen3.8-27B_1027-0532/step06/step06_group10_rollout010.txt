    return np.ones(len(x))

def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (target,)}]
        try:
            res = minimize(_slsqp_obj, h, args=(alpha, n), jac=_slsqp_jac,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def fast_greedy_cd(h0, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 800:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = int(i)
                    best_j = int(j)
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def subgrad_gd(h0, n, target, time_limit, max_steps, lr_init):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stagnant = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 500:
            break
        grad = max_grad(h, n)
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_now - 1e-15:
            h = h_try
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active, max_rounds):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            for mix in [0.7, 0.5, 0.3]:
                h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    break
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturbation_escape(h0, n, target, time_limit, n_tries, rng):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for t in range(n_tries):
        if time.time() > time_limit:
            break
        noise_scale = 0.05 * (1.0 - t / max(n_tries, 1))
        noise = rng.standard_normal(n)
        noise -= np.mean(noise)
        h_pert = project_feasible(h0 + noise_scale * noise, target)
        h_try, c5_try = slsqp_round(h_pert, n, target, [200, 1000, 5000, 20000], 500, 1e-13,
                                     min(60.0, time_limit - (time.time() - time_limit)), time.time())
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP with moderate alpha to ensure local optimality
    if tleft() > 180:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [100, 500, 2000, 10000, 50000],
                                2000, 1e-14, min(150.0, tleft() * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: CD fine-tuning at current resolution
    if tleft() > 200:
        cd_time = min(180.0, tleft() * 0.35)
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=cd_time, max_sweeps=15000, n_cand=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Very high alpha SLSQP
    if tleft() > 150:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [5000, 20000, 100000, 500000],
                                3000, 1e-15, min(120.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: Subgradient method
    if tleft() > 100:
        h_sg, c5_sg = subgrad_gd(best_h, n, target,
                                  time_limit=min(80.0, tleft() * 0.3),
                                  max_steps=5000, lr_init=0.05)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: CD again
    if tleft() > 150:
        cd_time = min(130.0, tleft() * 0.35)
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=cd_time, max_sweeps=12000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: LP refinement with more active constraints
    if tleft() > 100:
        h_lp, c5_lp = lp_refine(best_h, n, target,
                                 time_limit=min(80.0, tleft() * 0.35),
                                 n_active=40, max_rounds=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Try higher resolution n=800
    if tleft() > 300:
        n_hi = 800
        target_hi = float(n_hi) / 2.0
        h_hi = interpolate_to(best_h, n_hi, target_hi)
        h_hi, s_hi = slsqp_round(h_hi, n_hi, target_hi,
                                  [100, 500, 2000, 10000, 50000, 200000],
                                  2000, 1e-14, min(200.0, tleft() * 0.5), start_time)
        if s_hi < best_score:
            best_score = s_hi
            best_h = h_hi
            n = n_hi
            target = target_hi
        if tleft() > 150:
            cd_time = min(120.0, tleft() * 0.4)
            h_cd, c5_cd = fast_greedy_cd(best_h, len(best_h), float(len(best_h)) / 2.0,
                                          time_limit=cd_time, max_sweeps=10000, n_cand=20)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
        if tleft() > 80:
            h_p, s_p = slsqp_round(best_h, len(best_h), float(len(best_h)) / 2.0,
                                    [5000, 20000, 100000, 500000],
                                    2000, 1e-15, min(60.0, tleft() * 0.4), start_time)
            if s_p < best_score:
                best_score = s_p
                best_h = h_p

    # Phase 8: Perturbation escape
    if tleft() > 150:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_pe, c5_pe = perturbation_escape(best_h, n_use, target_use,
                                           time_limit=min(120.0, tleft() * 0.5),
                                           n_tries=5, rng=rng)
        if c5_pe < best_score:
            best_score = c5_pe
            best_h = h_pe

    # Phase 9: CD at current best
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        cd_time = min(60.0, tleft() * 0.4)
        h_cd, c5_cd = fast_greedy_cd(best_h, n_use, target_use,
                                      time_limit=cd_time, max_sweeps=8000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_f, s_f = slsqp_round(best_h, n_use, target_use,
                                [10000, 50000, 200000, 1000000],
                                2000, 1e-15, min(30.0, tleft() * 0.6), start_time)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f

    # Phase 11: Final CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = fast_greedy_cd(best_h, n_use, target_use,
                                      time_limit=min(15.0, tleft() * 0.7),
                                      max_sweeps=5000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
