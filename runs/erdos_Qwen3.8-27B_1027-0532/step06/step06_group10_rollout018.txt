def project_onto_simplex_box(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Handle residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    """Compute smooth approximation of max(corr)"""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_weights(corr, alpha):
    """Compute softmax weights for the smooth max"""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    return w

Actually, I want to add one more technique that I think could be very effective: **solving the KKT conditions directly**.

At the optimum, for the active set A with weights w (sum w = 1, w ≥ 0), we need:
1. For all j with 0 < h[j] < 1: sum_{s in A} w_s * grad_s[j] = λ
2. For all j with h[j] = 0: sum_{s in A} w_s * grad_s[j] ≥ λ
3. For all j with h[j] = 1: sum_{s in A} w_s * grad_s[j] ≤ λ
4. sum_j h[j] = n/2
5. corr_s = t for all s in A (active constraints are tight)
6. corr_s ≤ t for all s not in A

This is a system of nonlinear equations. I can solve it iteratively:
- Fix the active set A and the free set F = {j: 0 < h[j] < 1}
- The conditions become: sum_s w_s * grad_s[j] = λ for j in F
- This gives |F| equations in |F| + |A| + 1 unknowns (h_F, w_A, λ)
- Plus the tightness conditions: corr_s = t for s in A (|A| equations)
- Total: |F| + |A| equations in |F| + |A| + 1 unknowns (we need sum w = 1 as well, so |F| + |A| + 1 equations in |F| + |A| + 1 unknowns)

This is a square system! I can solve it with Newton's method.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        if resid > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += resid / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha, n):
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        if k >= 0:
            grad[:n-k] += w[s] * (1.0 - h[k:])
            grad[k:] -= w[s] * h[:n-k]
        else:
            m = -k
            grad[m:] += w[s] * (1.0 - h[:n-m])
            grad[:n-m] -= w[s] * h[m:]
    grad *= dx
    return grad

def lp_iteration(h, n, target, active_set):
    dx = 2.0 / n
    corr = get_corr(h)
    mid = n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for s_idx in active_set:
        k = int(s_idx) - mid
        if k >= 0:
            g = np.zeros(n)
            g[:n-k] = (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            m = -k
            g = np.zeros(n)
            g[m:] = (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
        lin = corr[s_idx] + g @ (h_var - h)
        constraints.append(dx * lin <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(h_var.value, 0.0, 1.0)

def coordinate_descent(h0, n, target, time_budget, start_time, n_swap=20):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

        corr = get_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        if k_star >= 0:
            grad[:n-k_star] += (1.0 - h[k_star:])
            grad[k_star:] -= h[:n-k_star]
        else:
            m = -k_star
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_swap]
        high_set = sorted_idx[-n_swap:]
        
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        
        shifts = np.arange(-(n-1), n)
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(2*n-1)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:

                    best_delta = delta_star
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
