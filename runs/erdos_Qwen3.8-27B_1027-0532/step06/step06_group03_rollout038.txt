def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            w = weights[s_vals + mid]
            k_vals = j - s_vals
            grad[j] += np.dot(w, 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            w2 = weights[s_vals2 + mid]
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(w2, h[k_vals2])
    grad *= dx
    return grad

def _slsqp_constraint_sum(h, target):
    return np.sum(h) - target

def _slsqp_constraint_jac(h):
    return np.ones(len(h))

def slsqp_polish(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        target_local = target
        cons = [{'type': 'eq', 'fun': _slsqp_constraint_sum, 'jac': _slsqp_constraint_jac}]
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_local)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand, top_k, start_time):
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    streak = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            streak = 0
        else:
            streak += 1
        if streak > 300:
            break

        c_max = np.max(corr)
        if top_k >= n_corr:
            top_idx = np.arange(n_corr)
        else:
            top_idx = np.argsort(corr)[-top_k:]

        grad_proxy = np.zeros(n)
        for s_idx in top_idx:
            k_shift = int(s_idx) - mid
            v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
            v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
            grad_proxy[v1] += (1.0 - h[idxs[v1] - k_shift])
            grad_proxy[v2] -= h[idxs[v2] + k_shift]
        grad_proxy -= np.mean(grad_proxy)

        sorted_idx = np.argsort(grad_proxy)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(25):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def momentum_descent(h_start, n, target, alpha, time_limit, max_steps, base_lr, mu, start_time):
    dx = 2.0 / n
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        grad = smooth_grad(h, alpha)
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr / (1.0 + 0.005 * step)
        accepted = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * lr_mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h) - 1e-15:
                h = h_try
                accepted = True
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
        if not accepted:
            v *= 0.5
    return best_h, best_c5

def symmetrize(h, n):
    h_sym = 0.5 * (h + h[::-1].copy())
    return h_sym

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_cd(h_start, n, target, time_limit, max_sweeps, block_size, n_trials, start_time):
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng_bl = np.random.RandomState(12345)
    streak = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            streak = 0
        else:
            streak += 1
        if streak > 100:
            break

        found = False
        for _ in range(n_trials):
            start_pos = rng_bl.randint(0, n - block_size)
            block = slice(start_pos, start_pos + block_size)
            h_block = h[block].copy()
            block_sum = np.sum(h_block)

            for direction in [1, -1]:
                h_attempt = h_block + direction * rng_bl.randn(block_size) * 0.1
                h_attempt = np.clip(h_attempt, 0.0, 1.0)
                h_attempt = h_attempt * (block_sum / max(np.sum(h_attempt), 1e-15))
                h_attempt = np.clip(h_attempt, 0.0, 1.0)
                h_new = h.copy()
                h_new[block] = h_attempt
                h_new = project_feasible(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h) - 1e-14:
                    h = h_new
                    found = True
                    break
            if found:
                break
    return best_h, best_c5
