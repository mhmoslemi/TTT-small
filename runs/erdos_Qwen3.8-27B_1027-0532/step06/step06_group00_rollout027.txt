import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        # Compute gradient using top n_shifts correlations

            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

- If 0 <= p <= n-1-|k|: term from h[p+|k|] is -h[p+|k|] ... wait no.

f_{-K} = sum_{i=K}^{n-1} h[i] * (1 - h[i-K])

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_start=40, n_cand_end=100, max_no_improve=500,
                       n_shifts=5, bisection_iters=8):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        grad = np.zeros(n)
        total_w = 0.0
        for s_idx in top_shifts:
            w = corr[s_idx]
            total_w += w
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        if total_w > 0:
            grad /= total_w
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        n_h = len(high_set)
        n_l = len(low_set)

        H = high_set[:, None].astype(np.intp)
        L = low_set[None, :].astype(np.intp)
        valid = (H != L)

        pair_i = H[valid].flatten()
        pair_j = L[valid].flatten()
        P = len(pair_i)

        if P == 0:
            continue

        G = change[pair_j] - change[pair_i]
        dlo = np.maximum(h[pair_i] - 1.0, -h[pair_j])
        dhi = np.minimum(h[pair_i], 1.0 - h[pair_j])

        valid2 = (dhi - dlo) > 1e-12
        if not valid2.any():
            continue
        pair_i = pair_i[valid2]
        pair_j = pair_j[valid2]
        G = G[valid2]
        dlo = dlo[valid2]
        dhi = dhi[valid2]
        P = len(pair_i)
        if P == 0:
            continue

        mid_d = 0.5 * (dlo + dhi)
        for _ in range(bisection_iters):
            modified = corr[None, :] + mid_d[:, None] * G
            max_vals = modified.max(axis=1)
            active_mask = (modified >= max_vals[:, None] - 1e-12)
            G_active = np.where(active_mask, G, -np.inf)
            deriv = G_active.max(axis=1)
            inc_mask = deriv > 1e-12
            dhi[inc_mask] = mid_d[inc_mask]
            dlo[~inc_mask] = mid_d[~inc_mask]
            mid_d = 0.5 * (dlo + dhi)

        final_modified = corr[None, :] + mid_d[:, None] * G
        final_max = final_modified.max(axis=1) * dx

        best_trial = float(final_max.min())
        if best_trial < c5_val - 1e-15:
            best_idx = int(final_max.argmin())
            i = int(pair_i[best_idx])
            j = int(pair_j[best_idx])
            delta = float(mid_d[best_idx])
            h[i] -= delta
            h[j] += delta
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def contrast_enhance(h, target, alpha=1.5):
    h_new = (h - 0.5) * alpha + 0.5
    return project_feasible(h_new, target)


def multi_start_vcd(h_init, n, target, time_limit, n_starts=4, noise_scales=None):
    if noise_scales is None:
        noise_scales = [0.08, 0.15, 0.25, 0.4]
    start = time.time()
    rng = np.random.RandomState(777)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for s in range(n_starts):
        remaining = time_limit - (time.time() - start)
        if remaining < 40:
            break
        scale = noise_scales[s] if s < len(noise_scales) else 0.2
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        time_alloc = remaining * 0.85 / (n_starts - s)
        h_cd, c5_cd = vectorized_pair_cd(h_pert, n, target,
                                          time_limit=time_alloc,
                                          max_sweeps=20000,
                                          n_cand_start=40, n_cand_end=90,
                                          max_no_improve=300, n_shifts=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    return best_h, best_c5
