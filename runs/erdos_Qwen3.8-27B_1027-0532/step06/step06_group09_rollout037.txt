```python
def newton_step(h, n, target, alpha_smoothing, top_m):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    
    # Get top m lags
    top_indices = np.argsort(corr)[-top_m:]
    top_lags = top_indices - mid
    
    # Weights for softmax
    exp_vals = np.exp(alpha_smoothing * (corr[top_indices] - np.max(corr)))
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient and Hessian
    grad = np.zeros(n)
    H = np.zeros((n, n))
    
    idxs = np.arange(n)
    for k, w in zip(top_lags, weights):
        # f_k(h) = sum over valid i of h[i] * (1 - h[i+k])
        # For k >= 0: valid i are 0 to n-1-k
        # For k < 0: valid i are -k to n-1
        if k >= 0:
            valid_i = np.arange(0, n - k)
            h_shifted = h[valid_i + k]
            # grad contribution: w * (1 - h[i+k]) for i in valid_i
            #                    w * (-h[i-k]) for i-k in [0, n-1] i.e., i in [k, n-1]
            grad[valid_i] += w * (1.0 - h_shifted)
            # For the -h[i]*h[i+k] term, derivative w.r.t h[j] also gets -h[j-k] contribution
            valid_j = np.arange(k, n)
            grad[valid_j] += -w * h[valid_j - k]
        else:
            k_abs = -k
            valid_i = np.arange(0, n - k_abs)
            h_shifted = h[valid_i + k_abs]
            grad[valid_i] += w * (1.0 - h_shifted)
            valid_j = np.arange(k_abs, n)
            grad[valid_j] += -w * h[valid_j - k_abs]
        
        # Hessian contribution: -w * (e_i e_{i+k}^T + e_{i+k} e_i^T) for valid pairs
        if k > 0:
            for i in range(n - k):
                H[i, i+k] -= w
                H[i+k, i] -= w
        elif k < 0:
            k_abs = -k
            for i in range(n - k_abs):
                H[i, i+k_abs] -= w
                H[i+k_abs, i] -= w
    
    grad *= dx
    grad -= np.mean(grad)  # Account for sum constraint
    
    return grad, H, top_indices, weights
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_objective(h, n, target, top_m=10, temperature=50.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    
    # Softmax weights for top m
    max_corr = np.max(corr)
    exp_vals = np.exp(temperature * (corr - max_corr))
    # Focus on top m
    top_idx = np.argsort(corr)[-top_m:]
    weights = np.zeros(len(corr))
    weights[top_idx] = exp_vals[top_idx]
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(corr))
        weights[top_idx] = 1.0 / top_m
        w_sum = 1.0
    weights /= w_sum
    
    # Compute smooth objective
    obj = np.sum(weights * corr) * dx
    
    # Compute gradient
    idxs = np.arange(n)
    grad = np.zeros(n)
    H = np.zeros((n, n))
    
    for s_idx, w in zip(top_idx, weights[top_idx]):
        k = int(s_idx) - mid
        if k == 0:
            # f_0 = sum h[i]^2 ... wait no
            # f_0 = sum h[i] * (1-h[i]) = sum h[i] - sum h[i]^2
            # grad = 1 - 2*h[i]
            grad[:] += w * (1.0 - 2.0 * h)
            H[np.arange(n), np.arange(n)] += w * (-2.0)
        else:
            abs_k = abs(k)
            # f_k = sum_{valid i} h[i] * (1 - h[i+k])  (for k>0)
            # or f_k = sum_{valid i} h[i+k] * (1 - h[i])  (for k<0, reindexed)
            # In either case: f = sum valid h[a] - sum valid h[a]*h[b]
            # where b = a + k (if k>0) or a + |k| (if k<0, with different indexing)
            
            if k > 0:
                # valid: i from 0 to n-1-k
                # term i: h[i] * (1 - h[i+k]) = h[i] - h[i]*h[i+k]
                valid = np.arange(0, n - k)
                grad[valid] += w * (1.0 - h[valid + k])  # d/dh[i] of h[i] - h[i]*h[i+k]
                # d/dh[i+k] of -h[i]*h[i+k] = -h[i]
                grad[k:] += -w * h[:n-k]
                # Hessian: d²/dh[i]dh[i+k] = -w
                H[valid, valid + k] += -w
                H[valid + k, valid] += -w
            else:
                # k < 0, let k_abs = -k
                # f_k = sum_{i=0}^{n-1-k_abs} h[i+k_abs] * (1 - h[i])
                #     = sum h[i+k_abs] - sum h[i+k_abs]*h[i]
                # Same structure as k>0 just with roles reversed
                k_abs = -k
                valid = np.arange(0, n - k_abs)
                grad[valid + k_abs] += w * (1.0 - h[valid])  # d/dh[i+k_abs] of h[i+k_abs] - h[i+k_abs]*h[i]
                grad[valid] += -w * h[valid + k_abs]  # d/dh[i] of -h[i+k_abs]*h[i]
                H[valid + k_abs, valid] += -w
                H[valid, valid + k_abs] += -w
    
    grad *= dx
    H *= dx
    # Account for sum constraint: project gradient to sum-zero subspace
    grad -= np.mean(grad)
    # For Hessian, we'll handle the constraint separately
    
    return obj, grad, H, weights, top_idx

def damped_newton(h, n, target, time_limit, top_m=10, temperature=50.0, 
                  max_iter=500, init_lam=1.0):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = init_lam
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        obj, grad, H, weights, top_idx = active_set_objective(h, n, target, top_m, temperature)
        
        # Regularize Hessian for positive definiteness
        # Add diagonal to ensure PD
        diag = np.diag(H).copy()
        min_eig_approx = min(0.0, np.min(diag)) - 1.0
        H_reg = H + (lam - min_eig_approx) * np.eye(n)
        
        # Solve for Newton direction
        try:
            d = np.linalg.solve(H_reg, -grad)
        except np.linalg.LinAlgError:
            d = -grad * 0.01
        
        # Project direction to sum-zero (for feasibility)
        d -= np.mean(d)
        
        # Line search
        best_alpha = 0.0
        c5_now = compute_c5(h)
        for alpha in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            h_try = h + alpha * d
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                best_alpha = alpha
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                    no_improve = 0
                break
        
        if best_alpha == 0.0:
            no_improve += 1
            lam *= 2.0
            if no_improve > 50:
                break
        else:
            lam *= 0.9
            lam = max(lam, 1e-3)
    
    return best_h, best_c5
```

```python
def exact_pair_cd(h, n, target, time_limit, max_sweeps=10000, 
                  n_candidates=15, n_grid=200, patience=200):
    """Pair coordinate descent with exact 1D optimization."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        # Compute gradient of max (subgradient: direction of max lag)
        # Use weighted top-m for smoother gradient
        k_star = int(np.argmax(corr)) - mid
        
        grad = np.zeros(n)
        if k_star > 0:
            for i in range(n - k_star):
                grad[i] += (1.0 - h[i + k_star])
                grad[i + k_star] -= h[i]
        elif k_star < 0:
            k_abs = -k_star
            for i in range(n - k_abs):
                grad[i + k_abs] += (1.0 - h[i])
                grad[i] -= h[i + k_abs]
        else:
            grad[:] = 1.0 - 2.0 * h
        grad *= dx
        grad -= np.mean(grad)
        
        # Candidate pairs
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_candidates]
        high_set = sorted_idx[-n_candidates:]
        
        best_c5_trial = c5_val
        best_move = None
        
        for ii in range(min(8, len(high_set))):
            i = int(high_set[ii])
            for jj in range(min(8, len(low_set))):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-10:
                    continue
                
                # Compute a, b vectors
                a, b = compute_pair_ab(h, n, i, j)
                
                # Grid search for optimal d
                ds = np.linspace(0, d_hi, n_grid)
                # Vectorized: compute all values
                # corr[None, :] + a[None, :]*ds[:, None] + b[None, :]*ds[:, None]**2
                # This is n_grid x n_corr matrix
                values = corr[np.newaxis, :] + a[np.newaxis, :] * ds[:, np.newaxis] + b[np.newaxis, :] * ds[:, np.newaxis] ** 2
                max_vals = np.max(values, axis=1)
                best_grid_idx = np.argmin(max_vals)

            h[i] -= d_star
            h[j] += d_star
            h = project_feasible(h, target)
        else:
            if no_improve > patience // 3:
                break
    
    return best_h, best_c5
```

We can reformulate the problem as:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_pair_ab(h, n, i, j):
    n_corr = 2 * n - 1
    mid = n - 1
    a = np.zeros(n_corr)
    b = np.zeros(n_corr)
    for s in range(n_corr):
        k = s - mid
        a_s = 0.0
        b_s = 0.0
        t_set = set()
        for t_val in (i, i - k, j, j - k):
            if 0 <= t_val < n:
                t_set.add(t_val)
        for t in t_set:
            tk = t + k
            if tk < 0 or tk >= n:
                continue
            df = 0
            ds = 0
            if t == i:
                df = -1
            elif t == j:
                df = 1
            if tk == i:
                ds = -1
            elif tk == j:
                ds = 1
            if df == 0 and ds == 0:
                continue
            ht = h[t]
            htk = h[tk]
            if df != 0 and ds == 0:
                a_s += df * (1.0 - htk)
            elif df == 0 and ds != 0:
                a_s += -ht * ds
            else:
                a_s += df * (1.0 - htk) - ht * ds
                b_s += -df * ds
        a[s] = a_s
        b[s] = b_s
    return a, b

def exact_pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
                  n_candidates=12, n_grid=150, patience=300, m_top=8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    values_buf = np.empty((n_grid, n_corr))

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        if k_star > 0:
            vi = np.arange(0, n - k_star)
            grad[vi] += (1.0 - h[vi + k_star])
            grad[vi + k_star] -= h[vi]
        elif k_star < 0:
            ka = -k_star
            vi = np.arange(0, n - ka)
            grad[vi + ka] += (1.0 - h[vi])
            grad[vi] -= h[vi + ka]
        else:
            grad[:] = 1.0 - 2.0 * h
        grad *= dx
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_candidates]
        high_set = sorted_idx[-n_candidates:]

        candidate_pairs = set()
        for ii in range(min(6, len(high_set))):
            for jj in range(min(6, len(low_set))):
                candidate_pairs.add((int(high_set[ii]), int(low_set[jj])))
                candidate_pairs.add((int(low_set[jj]), int(high_set[ii])))

        top_s = np.argsort(corr)[-m_top:]
        for s_idx in top_s:
            k = int(s_idx) - mid
            if k > 0:
                vi = np.arange(0, n - k)
                terms = h[vi] * (1.0 - h[vi + k])
                top_terms = vi[np.argsort(terms)[-4:]]
                for tv in top_terms:
                    candidate_pairs.add((int(tv), int(tv + k)))
            elif k < 0:
                ka = -k
                vi = np.arange(0, n - ka)
                terms = h[vi + ka] * (1.0 - h[vi])
                top_terms = (vi + ka)[np.argsort(terms)[-4:]]
                for tv in top_terms:
                    candidate_pairs.add((int(tv), int(tv - ka)))

        best_c5_trial = c5_val
        best_move = None

        for (pi, pj) in list(candidate_pairs)[:40]:
            d_hi = min(h[pi], 1.0 - h[pj])
            if d_hi < 1e-10:
                continue
            a, b = compute_pair_ab(h, n, pi, pj)
            ds = np.linspace(0.0, d_hi, n_grid)
            np.multiply(ds[:, np.newaxis], b[np.newaxis, :], out=values_buf)
            values_buf *= ds[:, np.newaxis]
            np.multiply(ds[:, np.newaxis], a[np.newaxis, :], out=values_buf[:, :])
            values_buf += corr[np.newaxis, :]
            actual_vals = corr[np.newaxis, :] + a[np.newaxis, :] * ds[:, np.newaxis] + b[np.newaxis, :] * ds[:, np.newaxis] ** 2
            max_vals = np.max(actual_vals, axis=1)
            gi = int(np.argmin(max_vals))
            if max_vals[gi] * dx < best_c5_trial - 1e-14:
                best_c5_trial = max_vals[gi] * dx
                best_move = (pi, pj, float(ds[gi]))

        if best_move is not None:
            pi, pj, d_star = best_move
            h[pi] -= d_star
            h[pj] += d_star
            h = project_feasible(h, target)
        else:
            if no_improve > patience // 3:
                break
            if no_improve % 100 == 0 and no_improve > 50:
                rng_esc = np.random.RandomState(sweep)
                h_pert = h + rng_esc.randn(n) * 0.03
                h_pert = project_feasible(h_pert, target)
                if compute_c5(h_pert) < best_c5:
                    h = h_pert
                    no_improve = 0

    return best_h, best_c5
