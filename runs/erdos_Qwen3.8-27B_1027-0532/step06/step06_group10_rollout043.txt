def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def topk_smooth_val(h, alpha, k_top):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    # Get top k values
    if k_top >= n_corr:
        idx_top = np.arange(n_corr)
    else:
        idx_top = np.argpartition(corr, -k_top)[-k_top:]
    vals = corr[idx_top]
    c_max = np.max(vals)
    shifted = alpha * (vals - c_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return float((c_max + logsum / alpha) * dx), idx_top, vals

def topk_smooth_grad(h, alpha, k_top, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if k_top >= n_corr:

    c_max = np.max(vals)
    shifted = alpha * (vals - c_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(vals)] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for wi, s in zip(w, idx_top):
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def projected_subgradient(h, n, target, top_k, lr, alpha, n_steps):

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_opt = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def topk_corr(h, k_top):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if k_top >= n_corr:
        idx_top = np.arange(n_corr)
    else:
        idx_top = np.argpartition(corr, -k_top)[-k_top:]
    return corr, idx_top

def topk_smooth_val(h, alpha, k_top):
    corr, idx_top = topk_corr(h, k_top)
    vals = corr[idx_top]
    c_max = np.max(vals)
    shifted = np.clip(alpha * (vals - c_max), -50.0, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    dx = 2.0 / len(h)
    return float((c_max + logsum / alpha) * dx)

def topk_smooth_grad(h, alpha, k_top, n):
    dx = 2.0 / n
    corr, idx_top = topk_corr(h, k_top)
    vals = corr[idx_top]
    c_max = np.max(vals)
    shifted = np.clip(alpha * (vals - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(vals)] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for ai in active:
        s = int(idx_top[ai])
        k = s - mid
        wi = w[ai]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def _obj_fn(x):
    return topk_smooth_val(x, _opt['alpha'], _opt['k_top'])

def _grad_fn(x):
    return topk_smooth_grad(x, _opt['alpha'], _opt['k_top'], _opt['n'])

def _con_fn(x):
    return np.sum(x) - _opt['target']

def _con_jac_fn(x):
    return np.ones(_opt['n'])

def concave_opt(h0, n, target, time_limit, alpha_start, alpha_end, n_alphas, maxiter, k_top_start=10, k_top_end=200):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    k_tops = np.linspace(k_top_start, k_top_end, n_alphas).astype(int)
    _opt['n'] = n
    _opt['target'] = target
    for i, alpha in enumerate(alphas):
        if time.time() - start > time_limit:
            break
        _opt['alpha'] = float(alpha)
        _opt['k_top'] = int(k_tops[i])
        cons = [{'type': 'eq', 'fun': _con_fn, 'jac': _con_jac_fn}]
        try:
            res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def cd_refine(h0, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def multi_restart(h0, n, target, time_limit, n_restarts, noise_scale, seed_base):
    start = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_restarts):
        if time.time() - start > time_limit:

        h_try = project_feasible(h0 + noise, target)

def upgrade_resolution(h_src, n_dst, target_dst, time_limit):
    n_src = len(h_src)
    if n_src == n_dst:
        h_src = project_feasible(h_src.copy(), target_dst)
        return h_src, compute_c5(h_src)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_interp = np.interp(x_new, x_old, h_src)
    h_interp = project_feasible(h_interp, target_dst)
    c5 = compute_c5(h_interp)
    t_budget = time_limit * 0.55
    h_opt, c5_opt = concave_opt(h_interp, n_dst, target_dst,
                                 time_limit=t_budget,
                                 alpha_start=100, alpha_end=50000,
                                 n_alphas=8, maxiter=1500,
                                 k_top_start=10, k_top_end=150)
    if c5_opt < c5:
        h_interp = h_opt
        c5 = c5_opt
    remaining = time_limit - (time.time() - (time.time() - time_limit))
    if remaining > 10:
        t_cd = min(time_limit * 0.3, remaining)
        h_cd, c5_cd = cd_refine(h_interp, n_dst, target_dst,
                                 time_limit=t_cd, max_sweeps=5000, n_cand=15)
        if c5_cd < c5:
            h_interp = h_cd
            c5 = c5_cd
    return h_interp, c5

def projected_subgrad(h0, n, target, top_k, lr, alpha, n_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h_curr = h0.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_curr.copy()
        k = min(top_k, len(corr))
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
        c_max = np.max(vals)
        w = np.exp(alpha * (vals - c_max))
        w = w / w.sum()
        grad = np.zeros(n)
        active = np.where(w > 1e-6)[0]
        for ai in active:
            s = int(top_idx[ai])
            ki = s - mid
            wi = w[ai]
            v1 = (idxs - ki >= 0) & (idxs - ki < n)
            v2 = (idxs + ki >= 0) & (idxs + ki < n)
            grad[v1] += wi * (1.0 - h_curr[idxs[v1] - ki])
            grad[v2] -= wi * h_curr[idxs[v2] + ki]
        grad *= dx
        grad -= np.mean(grad)
        h_curr = project_feasible(h_curr - lr * grad, target)
    return best_h, best_c5

def lp_round(h0, n, target, time_limit, n_active, max_rounds):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1.0 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5
