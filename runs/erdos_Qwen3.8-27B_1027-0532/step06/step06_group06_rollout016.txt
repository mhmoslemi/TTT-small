subject to 0 ≤ h[i] ≤ 1, Σ h[i] = n/2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

    n = len(h)
    dx = 2.0 / n

        if not i_cand or not j_cand:

            i_cand = [i for i in ones if 0 <= i + k_star < n]

                # corr[k] = sum_t h[t]*(1-h[t+k])

                # From i: k = i-t for t in {i, i-k...} → actually k = anything where i+k or i-k is in range
                # From j: similarly
                # So affected lags are: k where (0 <= i+k < n) or (0 <= i-k < n) or (0 <= j+k < n) or (0 <= j-k < n)
                
                # This is potentially all lags. Let's be smarter.
                # For the new max to be >= current max, we need some lag k where corr[k] + delta[k] > max_val
                # delta[k] = -(1-h[i+k]) + h[i-k] + (1-h[j+k]) - h[j-k] (with 0 for out of bounds)

                top_lags = np.argsort(corr)[-20:]

                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_continuous(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_obj = smooth_objective(h_cur, beta, target, penalty)
    
    for attempt in range(3):
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_objective, h_cur,
                          args=(beta, target, penalty),
                          jac=smooth_gradient,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            sc = compute_c5(h_candidate)
            if sc < compute_c5(h_cur):
                h_cur = h_candidate
        except Exception:
            pass
    return project_box_sum(h_cur, target)

Wait, let me be more careful.

corr[m] = sum_t h[t] * (1-h[t-k]) where k = m-(n-1) and the sum is over t such that 0≤t<n and 0≤t-k<n.

If h[i] changes by delta:
- In the sum, the term where t=i: contributes delta * (1-h[i-k]) if 0≤i-k<n
- The term where t-k=i, i.e., t=i+k: contributes h[i+k] * (1 - (h[i]+delta)) - h[i+k]*(1-h[i]) = -h[i+k]*delta if 0≤i+k<n

So: delta_corr[m] = delta * [(1-h[i-k]) * I(0≤i-k<n) - h[i+k] * I(0≤i+k<n)]

For a swap (h[i] goes 1→0, h[j] goes 0→1):
delta_corr[m] = -[(1-h[i-k]) * I(0≤i-k<n) - h[i+k] * I(0≤i+k<n)] + [(1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)]

This is O(1) per lag m! So for a single swap, we can compute the new max in O(n) time by computing delta_corr for all m and checking if corr[m] + delta_corr[m] exceeds the current max.

This is much faster than recomputing the full correlation (which is O(n^2) or O(n log n)).

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0

def project_binary(h, target_sum):
