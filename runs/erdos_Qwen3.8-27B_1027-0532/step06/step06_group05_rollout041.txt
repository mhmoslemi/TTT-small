def project_box_sum(h, target_sum):
    # Project onto {0 ≤ h ≤ 1, sum(h) = target_sum}
    n = len(h)
    lo, hi = float(np.min(h)) - 2.0, float(np.max(h)) + 2.0
    for _ in range(100):

        h_p = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_p)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def smooth_max_value_and_grad(corr, alpha):

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    return value, c_max, weights

def grad_smooth_max(h, alpha, corr=None, weights=None):
    n = len(h)
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    if weights is None:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = float(np.min(h)) - 2.0, float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_p)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    grad = np.zeros(n)
    for p in range(n):
        w = weights[n - 1 + p]
        if w < 1e-14:
            continue
        if p == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    
    return value, grad, corr, weights

def adam_optimize(h0, target, start_time, time_budget, alphas, lr, beta1, beta2, eps):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad, _, _ = smooth_obj_grad(h, alpha)
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)
        
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
    
    return best_h, best_actual

def compute_grad_matrix(h):
    """Compute g_inc[i][p] for all i and p (p = 0..n-1, corresponding to corr index n-1+p)"""
    n = len(h)
    g = np.zeros((n, n))  # g[i, p] = change in corr[n-1+p] when increasing h[i] by 1
    for p in range(n):
        if p == 0:
            g[:, 0] = 1.0 - 2.0 * h
        else:
            g[:n-p, p] += (1.0 - h[p:])
            g[p:, p] -= h[:n-p]
    return g

        g = compute_grad_matrix(h)

        best_move = None  # (i, j, delta)

def lp_polish(h, target, start_time, time_limit, max_iters=100):
    """Use linearization + LP to polish the solution"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = float(corr.max())
        
        # Identify active set (top K lags)
        K = min(10, 2 * n - 1)
        top_ks = np.argsort(corr)[-K:]
        
        # Compute gradients for active lags
        # For lag k (0-indexed from n-1), corr_idx = k
        # If k >= n-1, lag = k - (n-1) >= 0
        # If k < n-1, lag = k - (n-1) < 0
        
        grad_active = np.zeros((K, n))
        for idx, k_idx in enumerate(top_ks):
            lag = int(k_idx) - (n - 1)
            if lag >= 0:
                # corr[n-1+lag] = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
                # d/dh[j] = (1-h[j+lag]) if j <= n-1-lag, -h[j-lag] if j >= lag
                if lag == 0:
                    grad_active[idx] = 1.0 - 2.0 * h
                else:
                    grad_active[idx, :n-lag] += (1.0 - h[lag:])
                    grad_active[idx, lag:] -= h[:n-lag]
            else:
                abs_lag = -lag
                # corr[n-1-abs_lag] = sum_{i=abs_lag}^{n-1} h[i](1-h[i-abs_lag])
                # d/dh[j] = (1-h[j-abs_lag]) if j >= abs_lag, -h[j+abs_lag] if j+abs_lag < n
                if abs_lag == 0:
                    grad_active[idx] = 1.0 - 2.0 * h
                else:
                    grad_active[idx, abs_lag:] += (1.0 - h[:n-abs_lag])
                    grad_active[idx, :n-abs_lag] -= h[abs_lag:]
        
        # Build LP: min t s.t. grad_active[k] @ h <= t + b_k, 0 <= h <= 1, sum(h) = target
        # where b_k = -grad_active[k] @ h (the constant term from linearization)
        # Actually: corr_k_linear(h_new) = corr_k(h) + grad_k @ (h_new - h)
        #          = (corr_k(h) - grad_k @ h) + grad_k @ h_new
        # Let c_k = corr_k(h) - grad_k @ h
        # We want: min t s.t. c_k + grad_k @ h_new <= t for all active k
        #         0 <= h_new <= 1, sum(h_new) = target, t free
        
        c_vec = np.zeros(K)
        for idx in range(K):
            k_idx = int(top_ks[idx])
            c_vec[idx] = corr[k_idx] - float(grad_active[idx] @ h)
        
        # LP variables: [h_0, ..., h_{n-1}, t]  (n+1 variables)
        # Minimize: t → c_obj = [0, ..., 0, 1]
        # Constraints:
        #   c_k + grad_k @ h - t <= 0  for each k  →  [grad_k, -1] @ [h, t] <= -c_k
        #   sum(h) = target  →  [1, ..., 1, 0] @ [h, t] = target
        #   0 <= h_i <= 1
        #   t >= 0 (actually t can be anything, but the constraints imply t >= max(c_k))
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for idx in range(K):
            A_ub[idx, :n] = grad_active[idx]
            A_ub[idx, n] = -1.0
            b_ub[idx] = -c_vec[idx]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
            if res.success:
                h_lp = res.x[:n]
                h_lp = np.clip(h_lp, 0.0, 1.0)
                h_lp = project_box_sum(h_lp, target)
                
                # Line search: try different step sizes
                best_step = 0.0
                best_val = compute_c5(h)
                for step in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    h_try = h + step * (h_lp - h)
                    h_try = project_box_sum(h_try, target)
                    val_try = compute_c5(h_try)
                    if val_try < best_val:

                        best_c5 = compute_c5(h)
                        best_h = h.copy()
        
        except Exception:
            pass
    
    return best_h, best_c5

            break
        if stall > 50:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = float(np.max(corr))
        g = compute_grad_matrix(h)
        dec_pos = np.where(h > 0.02)[0]
        inc_pos = np.where(h < 0.98)[0]
        if len(dec_pos) == 0 or len(inc_pos) == 0:
            break
        ns = 25
        if len(dec_pos) > ns:
            dec_s = rng.choice(dec_pos, ns, replace=False)
        else:
            dec_s = dec_pos
        if len(inc_pos) > ns:
            inc_s = rng.choice(inc_pos, ns, replace=False)
        else:
            inc_s = inc_pos
        best_improve = 0.0
        best_move = None
        for i in dec_s:
            gi = g[i]
            for j in inc_s:
                if i == j:
                    continue
                diff = g[j] - gi
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-5:
                    continue
                best_d = 0.0
                best_f = max_corr
                for frac in [0.1, 0.3, 0.5, 0.8, 1.0]:
                    d = delta_max * frac
                    fm = float(np.max(corr + d * diff))
                    if fm < best_f:
                        best_f = fm
                        best_d = d
                improvement = max_corr - best_f
                if improvement > best_improve:
                    best_improve = improvement
                    best_move = (i, j, best_d)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = project_box_sum(h, target)
            sc = compute_c5(h)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width=30, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(1)
    h = (h >= 0.5).astype(float)
    cur_sum = int(h.sum())
    if cur_sum > target_sum:
        ones = np.where(h == 1)[0]
        extra = cur_sum - target_sum
        h[ones[:extra]] = 0
    elif cur_sum < target_sum:
        zeros = np.where(h == 0)[0]
        need = target_sum - cur_sum
        h[zeros[:need]] = 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 150:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        top_thresh = max_val * 0.9
        top_indices = np.where(corr >= top_thresh)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        best_improve = 0.0
        best_i = -1
        best_j = -1
        ni = min(len(i_cand), scan_width)
        nj = min(len(j_cand), scan_width)
        for ii in range(ni):
            i = i_cand[ii]
            for jj in range(nj):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_c5 = compute_c5(best_h)

    if tleft() > 60:
        h_bin = (best_h >= 0.5).astype(float)
        cs = int(h_bin.sum())
        if cs > int(target):
            ones = np.where(h_bin == 1)[0]
            h_bin[ones[:cs - int(target)]] = 0
        elif cs < int(target):
            zeros = np.where(h_bin == 0)[0]
            h_bin[zeros[:int(target) - cs]] = 1
        t_greedy = min(40, tleft() * 0.1)
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=50000,
                                   time_limit=t_greedy, start_time=start_time,
                                   scan_width=30, rng=rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    if tleft() > 200:
        t_adam = min(450, tleft() - 150)
        alphas = [50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600, 51200, 100000]
        h_a, s_a = adam_exact(best_h, target, start_time, t_adam,
                               alphas=alphas, lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8)
        if s_a < best_c5:
            best_c5 = s_a
            best_h = h_a.copy()

    if tleft() > 250:
        for ri in range(3):
            if tleft() < 120:
                break
            jitter = 0.003 + 0.002 * ri
            h_p = best_h + rng.randn(n) * jitter
            h_p = project_box_sum(h_p, target)
            rt = min(120, (tleft() - 100) / (3 - ri))
            alphas_r = [500, 2000, 8000, 30000, 80000, 200000]
            h_ar, s_ar = adam_exact(h_p, target, start_time, rt,
                                     alphas=alphas_r, lr=0.002, beta1=0.9,
                                     beta2=0.999, eps=1e-8)
            if s_ar < best_c5:
                best_c5 = s_ar
                best_h = h_ar.copy()

    if tleft() > 200:
        t_lp = min(300, tleft() - 80)
        h_lp, s_lp = lp_polish(best_h, target, start_time, t_lp, max_iters=1000)
        if s_lp < best_c5:
            best_c5 = s_lp
            best_h = h_lp.copy()

    if tleft() > 150:
        t_ls = min(250, tleft() - 60)
        h_ls, s_ls = exact_local_search(best_h, target, start_time, t_ls,
                                         max_iters=20000, rng=rng)
        if s_ls < best_c5:
            best_c5 = s_ls
            best_h = h_ls.copy()

    if tleft() > 80:
        t_lp2 = min(150, tleft() - 40)
        h_lp2, s_lp2 = lp_polish(best_h, target, start_time, t_lp2, max_iters=500)
        if s_lp2 < best_c5:
            best_c5 = s_lp2
            best_h = h_lp2.copy()

    if tleft() > 60:
        t_final = max(10.0, tleft() - 10)
        alphas_f = [50000, 100000, 200000, 500000]
        h_f, s_f = adam_exact(best_h, target, start_time, t_final,
                               alphas=alphas_f, lr=0.001, beta1=0.9,
                               beta2=0.999, eps=1e-8)
        if s_f < best_c5:
            best_c5 = s_f
            best_h = h_f.copy()

    if tleft() > 30:
        t_ls2 = max(10.0, tleft() - 5)
        h_ls2, s_ls2 = exact_local_search(best_h, target, start_time, t_ls2,
                                           max_iters=5000, rng=rng)
        if s_ls2 < best_c5:
            best_c5 = s_ls2
            best_h = h_ls2.copy()

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score <= best_c5 + 1e-10:
        best_h = h_final
        best_c5 = final_score

    return best_h, best_c5, n
