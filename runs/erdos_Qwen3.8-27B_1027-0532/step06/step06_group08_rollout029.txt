```python
def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def cons_fun(h):
        return np.sum(h) - target
    
    cons = [{'type': 'eq', 'fun': cons_fun}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
```

So W_flip_pad[:2n-1] = W[::-1], and the rest is zeros. Let me fix:

```python
W_flip_pad = np.zeros(N)
W_flip_pad[:2*n-1] = W[::-1]
```

That's cleaner. Let me verify with the example:
W = [W_{-2}, W_{-1}, W_0, W_1, W_2] (n=3, 5 elements)
W_flipped = [W_2, W_1, W_0, W_{-1}, W_{-2}] = W[::-1] ✓

OK good. Now let me also precompute the FFT of W and W_flipped if W doesn't change between iterations. But W does change (it's the softmax of the correlation), so we can't precompute.

However, we CAN precompute the FFT of h and (1-h) if h doesn't change. But h changes every iteration too.

So we need 4 FFTs per iteration (W, W_flip, h, 1-h). But actually, FFT(1-h) = FFT(1) - FFT(h), and FFT(1) is just [N, 0, 0, ...]. So we only need 3 FFTs: FFT(W), FFT(W_flip), FFT(h).

Wait, FFT(1-h) where 1 is a vector of ones of length n (zero-padded to N):
FFT(1-h)_pad = FFT(ones_n_pad) - FFT(h_pad)

And FFT(ones_n_pad) is a known vector: it's [sum of ones, 0, 0, ...] = [n, 0, 0, ...] in the frequency domain? No, that's not right.

FFT of [1, 1, ..., 1, 0, 0, ..., 0] (n ones followed by N-n zeros):
This is e^{-2πi*0*k/N} + e^{-2πi*1*k/N} + ... + e^{-2πi*(n-1)*k/N} for k = 0, ..., N-1

This is a geometric series: (1 - e^{-2πi*n*k/N}) / (1 - e^{-2πi*k/N}) for k != 0, and n for k=0.

It's not a simple vector, so let's just compute it directly. 4 FFTs per iteration is fine.

Actually, for N=2048, each FFT takes about 5-10 microseconds in numpy. So 4 FFTs + 2 IFFTs = 6 transforms ≈ 30-60 microseconds. Plus the multiplications and slicing. Total per gradient: maybe 100-200 microseconds. That's 20-50x faster than the direct O(n^2) approach!

In 200 seconds, we can do 1-2 million iterations. That's plenty.

Let me now write the complete solution with this FFT-based gradient.

Actually wait, I realize there might be an even simpler approach. Since we're computing:
grad = A[n-1:2n-1] - B[n-1:2n-1]

And A = IFFT(FFT(W_flip) * FFT(v)), B = IFFT(FFT(W) * FFT(h))

We can combine: grad = IFFT(FFT(W_flip) * FFT(v) - FFT(W) * FFT(h))[n-1:2n-1]

That saves one IFFT!

Let me also think about numerical stability. The FFT approach should be accurate to machine precision, so it should be fine.

Now let me put together the complete solution:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx2 = i - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    result[mask2] += h[idx2[mask2]]
    idx3 = j - k
    mask3 = (idx3 >= 0) & (idx3 < n)
    result[mask3] -= h[idx3[mask3]]
    idx4 = j + k
    mask4 = (idx4 >= 0) & (idx4 < n)
    result[mask4] -= h[idx4[mask4]]
    return result

def fft_gradient(h, W, N):
    """Compute gradient using FFT. W is softmax weights of length 2n-1."""
    n = len(h)
    W_pad = np.zeros(N)
    W_pad[:2*n-1] = W
    Wf_pad = np.zeros(N)
    Wf_pad[:2*n-1] = W[::-1]
    
    h_pad = np.zeros(N)
    h_pad[:n] = h
    v_pad = np.zeros(N)
    v_pad[:n] = 1.0 - h
    
    FH = np.fft.fft(h_pad)
    FV = np.fft.fft(v_pad)
    FW = np.fft.fft(W_pad)
    FWf = np.fft.fft(Wf_pad)
    
    # grad = IFFT(FWf * FV - FW * FH)[n-1:2n-1]
    combined = FWf * FV - FW * FH
    result = np.fft.ifft(combined).real
    return result[n-1:2*n-1]
```

Hmm wait, I realize I should double-check this formula. Let me re-derive.

We want: grad[j] = sum_k W[k] * [(1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]]

I showed above that:
A[j] = sum_k W[k] * v[j+k]*[0<=j+k<n] = conv(W_flip, v)[j + n - 1]
B[j] = sum_k W[k] * h[j-k]*[0<=j-k<n] = conv(W, h)[j + n - 1]

where conv is linear convolution.

So grad[j] = A[j] - B[j] = [conv(W_flip, v) - conv(W, h)][j + n - 1]

Using FFT: conv(a, b) = IFFT(FFT(a) * FFT(b)) (with zero-padding)

So grad = IFFT(FFT(Wf) * FFT(v) - FFT(W) * FFT(h))[n-1 : 2n-1]

Yes, that's what I have. 

Now, one concern: numerical precision. The FFT approach introduces small numerical errors (~1e-15 relative). For our purposes, this should be fine since we're working with values in [0,1] and the gradient magnitudes are O(1).

Let me now write the full Adam optimizer with FFT gradient:

```python
def adam_softmax_fft(h_init, n_iters, alpha_init, alpha_end, lr, beta1, beta2, eps,
                     time_limit, start_time):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    N = 1
    while N < 3*n - 2:
        N <<= 1
    
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    stall_limit = 5000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > stall_limit:
            break
        
        t += 1
        ratio = it / max(n_iters - 1, 1)
        alpha = alpha_init * (alpha_end / alpha_init) ** ratio
        
        max_corr = float(corr.max())
        shifted = np.clip(alpha * (corr - max_corr), -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        
        grad = fft_gradient(h, W, N)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target)
        
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        curr_c5 = float(corr.max()) * dx
        
        if curr_c5 < best_c5 - 1e-15:
            best_c5 = curr_c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
    
    return best_h, best_c5
```
