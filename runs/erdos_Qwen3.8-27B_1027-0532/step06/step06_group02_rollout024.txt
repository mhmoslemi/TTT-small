                    mask = (h[valid] == 1.0) & (h[valid + k_star] == 0.0)
                    cands = valid[mask]
                    if len(cands) == 0:
                        continue
                    i = cands[rng.randint(len(cands))]
                    j = i + k_star
                else:
                    m = -k_star
                    valid = np.arange(m, n)
                    mask = (h[valid] == 1.0) & (h[valid - m] == 0.0)
                    cands = valid[mask]
                    if len(cands) == 0:
                        continue
                    i = cands[rng.randint(len(cands))]
                    j = i - m
                
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                delta_c5 = c5_new - current_c5
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                    found_good = True
                    break
            
            if not found_good:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                delta_c5 = c5_new - current_c5
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
    
    return best_h, best_c5
