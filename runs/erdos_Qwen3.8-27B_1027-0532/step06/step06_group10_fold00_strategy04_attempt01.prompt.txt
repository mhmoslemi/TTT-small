def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_step(h, n, target, n_top=10, lr=0.02):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1][:n_top]

    w = np.exp(alpha_w * (corr[sorted_idx] - c_max))
    w = w / w.sum()
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for wi, s_idx in zip(w, sorted_idx):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)

        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    h_new = project_feasible(h - lr * grad, target)
