from scipy.optimize import minimize
import time

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on lambda."""
    h = h.astype(float).copy()
    n = len(h)
    # Bisection on lambda: h*_i = clip(h_i - lambda, 0, 1), sum(h*) = target
    lo, hi = -2.0, 2.0  # lambda range
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Final adjustment for exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Distribute the remaining difference
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj
