Wait, I realize there's an issue with the lambda approach. The lambda is meant to enforce the sum constraint. But in the vectorized bisection, I'm adding lambda to all slopes. That's not quite right.

The correct approach: we want to minimize max_k f_k(x) subject to the sum constraint. The Lagrangian for coordinate i is:
L(x_i) = max_k f_k(x_i) + λ * x_i

where f_k(x_i) = slopes[i,k] * (x_i - h[i]) + corr[k_idx]

So L(x_i) = max_k [slopes[i,k] * (x_i - h[i]) + corr[k_idx]] + λ * x_i
          = max_k [(slopes[i,k] + λ) * x_i - slopes[i,k]*h[i] + corr[k_idx]]
          = max_k [(slopes[i,k] + λ) * (x_i - h[i]) + corr[k_idx] + λ*h[i]]

Hmm, this is getting complicated. Let me just use a simpler approach: after the bisection, project to feasibility. The lambda is just a heuristic to bias the solution.

Actually, a cleaner approach: don't use lambda in the bisection at all. Just solve the pure min-max problem for each coordinate, then project the full vector to be feasible. The projection is cheap (just add a constant to all entries and clip).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    for _ in range(30):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

    active_k_idx = np.argsort(corr)[-n_active:]

    prev_active = np.sort(active_k_idx).copy()

    for p in range(max_passes):

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        idx_valid1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        idx_valid2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k_star])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k_star]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h_lp

    # Phase 2: Vectorized coordinate descent
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.2)
        h_cd, c5_cd = vectorized_cd(best_h, target, n_active=150,
                                    max_passes=50000, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP with larger active set
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.25)
        h_lp, c5_lp = lp_optimize(best_h, n, target,
                                  max_iter=5000, time_limit=lp_time,
                                  n_active=150, solver_name='PDLP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: CD with more active constraints
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.18)
        h_cd, c5_cd = vectorized_cd(best_h, target, n_active=200,
                                    max_passes=50000, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP with smooth approximation
    if tleft() > 100:
        h_ref, sc = slsqp_refine(best_h, n, target, [500, 2000, 8000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: LP polish with tight settings
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.2)
        h_lp, c5_lp = lp_optimize(best_h, n, target,
                                  max_iter=5000, time_limit=lp_time,
                                  n_active=200, solver_name='GLOP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Vectorized CD final
    if tleft() > 80:
        cd_time = min(100, tleft() * 0.2)
        h_cd, c5_cd = vectorized_cd(best_h, target, n_active=300,
                                    max_passes=50000, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Try higher resolution n=500
    if tleft() > 300:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        # LP at higher resolution
        if tleft() > 250:
            lp_time = min(150, tleft() * 0.3)
            h_lp500, c5_lp500 = lp_optimize(h_interp, n_try, target_try,
                                            max_iter=3000, time_limit=lp_time,
                                            n_active=120, solver_name='GLOP')
            if c5_lp500 < best_score:
                best_score = c5_lp500
                best_h = h_lp500
                n = n_try
                target = target_try
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp
                n = n_try
                target = target_try
        # CD at higher resolution
        if tleft() > 150:
            cd_time = min(120, tleft() * 0.3)
            h_cd500, c5_cd500 = vectorized_cd(best_h, target, n_active=200,
                                              max_passes=30000, time_limit=cd_time)
            if c5_cd500 < best_score:
                best_score = c5_cd500
                best_h = h_cd500

    # Phase 9: SLSQP high alpha
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                  [5000, 20000, 60000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Subgradient refinement
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=3000, time_limit=min(35, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 11: Final coordinate descent
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(35, tleft() * 0.6)
        h_cd, c5_cd = vectorized_cd(best_h, target_use, n_active=250,
                                    max_passes=50000, time_limit=cd_time)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final LP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(30, tleft() * 0.7)
        h_lp, c5_lp = lp_optimize(best_h, n_use, target_use,
                                  max_iter=2000, time_limit=lp_time,
                                  n_active=250, solver_name='GLOP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 13: Final subgradient micro-refinement
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=2000, time_limit=min(10, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
