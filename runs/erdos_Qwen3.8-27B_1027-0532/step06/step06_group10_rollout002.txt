def project_feasible(h, target_sum):

            h[idx] += inc
            diff -= inc

            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_local_search(h0, n, target, time_limit, max_rounds=200):
    start = time.time()
    dx = 2.0 / n
    h_binary = np.zeros(n)
    h_binary[h0 >= 0.5] = 1.0
    n_ones = int(np.sum(h_binary))
    if n_ones != n // 2:
        order = np.where(h_binary == 1.0)[0]
        if n_ones > n // 2:
            for idx in order[:n_ones - n // 2]:
                h_binary[idx] = 0.0
        else:
            zero_order = np.where(h_binary == 0.0)[0]
            for idx in zero_order[:n // 2 - n_ones]:
                h_binary[idx] = 1.0
    h = h_binary.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        top_idx = int(np.argmax(corr))
        k_star = top_idx - mid
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_swap = None
        best_val = c5_val
        for i in ones:
            for j in zeros:
                gi = index_grad(h, int(i), n)
                gj = index_grad(h, int(j), n)
                dg = gj - gi
                approx_max = float(np.max(corr + dg)) * dx
                if approx_max < best_val - 1e-14:
                    best_val = approx_max
                    best_swap = (int(i), int(j))

        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP smooth from parent
    if tleft() > 100:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [50, 200, 1000, 5000, 20000],
                                2000, 1e-14, min(60.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Fast local search
    if tleft() > 120:
        h_fs, s_fs = fast_local_search(best_h, n, target,
                                        time_limit=min(80.0, tleft() * 0.35),
                                        n_top=12, n_cand=25, max_iters=3000)
        if s_fs < best_score:
            best_score = s_fs
            best_h = h_fs

    # Phase 3: Power-sum SLSQP
    if tleft() > 100:
        h_ps, s_ps = psqp_round(best_h, n, target,
                                 [20, 50, 100, 300],
                                 2000, min(50.0, tleft() * 0.35), start_time)
        if s_ps < best_score:
            best_score = s_ps
            best_h = h_ps

    # Phase 4: LP refinement
    if tleft() > 80:
        h_lp, s_lp = lp_round(best_h, n, target,
                               time_limit=min(50.0, tleft() * 0.3),
                               n_active=12, max_rounds=25)
        if s_lp < best_score:
            best_score = s_lp
            best_h = h_lp

    # Phase 5: Local search again
    if tleft() > 100:
        h_fs2, s_fs2 = fast_local_search(best_h, n, target,
                                          time_limit=min(80.0, tleft() * 0.35),
                                          n_top=15, n_cand=30, max_iters=3000)
        if s_fs2 < best_score:
            best_score = s_fs2
            best_h = h_fs2

    # Phase 6: Higher resolution n=600
    if tleft() > 200:
        n2 = 600
        target2 = float(n2) / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        h_hires, s_hires = fast_local_search(h_interp, n2, target2,
                                              time_limit=min(60.0, tleft() * 0.25),
                                              n_top=10, n_cand=20, max_iters=2000)
        if s_hires < best_score:
            best_score = s_hires
            best_h = h_hires
        if tleft() > 80:
            h_hires2, s_hires2 = slsqp_round(best_h if best_score < compute_c5(h_interp) else h_interp,
                                              n2, target2,
                                              [50, 200, 1000, 5000],
                                              1500, 1e-14, min(50.0, tleft() * 0.3), start_time)
            if s_hires2 < best_score:
                best_score = s_hires2
                best_h = h_hires2
        if tleft() > 50:
            h_hires3, s_hires3 = fast_local_search(best_h, len(best_h), float(len(best_h))/2.0,
                                                    time_limit=min(40.0, tleft() * 0.3),
                                                    n_top=12, n_cand=25, max_iters=1500)
            if s_hires3 < best_score:
                best_score = s_hires3
                best_h = h_hires3

    # Phase 7: Try n=800 if time allows
    if tleft() > 150:
        n3 = 800
        target3 = float(n3) / 2.0
        h_interp3 = interpolate_to(best_h, n3, target3)
        h_hires4, s_hires4 = fast_local_search(h_interp3, n3, target3,
                                                time_limit=min(60.0, tleft() * 0.3),
                                                n_top=10, n_cand=18, max_iters=1500)
        if s_hires4 < best_score:
            best_score = s_hires4
            best_h = h_hires4
        if tleft() > 50:
            h_hires5, s_hires5 = psqp_round(best_h, len(best_h), float(len(best_h))/2.0,
                                             [30, 100, 300],
                                             1500, min(40.0, tleft() * 0.35), start_time)
            if s_hires5 < best_score:
                best_score = s_hires5
                best_h = h_hires5

    # Phase 8: Binary local search at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_bin, s_bin = binary_local_search(best_h, n_use, target_use,
                                            time_limit=min(40.0, tleft() * 0.4),
                                            max_rounds=100)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_fin, s_fin = slsqp_round(best_h, n_use, target_use,
                                    [200, 1000, 5000, 20000],
                                    1500, 1e-15, min(30.0, tleft() * 0.6), start_time)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    # Phase 10: Final local search
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_fin2, s_fin2 = fast_local_search(best_h, n_use, target_use,
                                            time_limit=min(15.0, tleft() * 0.7),
                                            n_top=10, n_cand=20, max_iters=1000)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
