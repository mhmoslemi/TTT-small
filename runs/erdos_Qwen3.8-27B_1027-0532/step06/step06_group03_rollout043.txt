    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)

    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp

    grad = np.zeros(n)

        # corr[k] = sum_i h[i] * (1-h[i+k-mid]) where valid indices
        # d corr[k] / d h[j]:
        #   (1 - h[j+k-mid]) if 0 <= j+k-mid < n  [from h[j] in first factor]
        #   -h[j-k+mid] if 0 <= j-k+mid < n  [from h in second factor]
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:

            k_indices = j + s_vals  # = mid + shift_index
            # shift_index = s_vals, actual shift = s_vals
            # h[j + shift] = h[k_indices]
            grad[j] += np.dot(weights[k_indices], 1.0 - h[k_indices])
        
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_indices2 = j - s_vals2
            grad[j] -= np.dot(weights[k_indices2 + mid], h[k_indices2])
    
    grad *= dx
    return grad

The key idea: maintain a distribution λ over the 2n-1 shifts. At each iteration:
1. Compute the gradient of ∑_k λ_k f_k(h)
2. Take a projected gradient step
3. Update λ using multiplicative weights based on the current f_k values

The gradient of ∑_k λ_k f_k(h) can be computed efficiently if λ is sparse (which it will be after a few iterations).

Let me code this up properly. For n=400, the full gradient computation (looping over all 799 shifts) takes O(n * n_corr) = O(400 * 799) ≈ 320,000 operations, which is fast in numpy if vectorized.

Actually, let me vectorize the gradient computation. For each shift k, the contribution to grad is:
- For k ≥ 0: grad[0:n-k] += λ_k * (1 - h[k:n]) and grad[k:n] -= λ_k * h[0:n-k]
- For k < 0: grad[-k:n] += λ_k * (1 - h[0:n+k]) and grad[0:n+k] -= λ_k * h[-k:n]

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _weighted_grad(h, lam, n, dx):
    """Compute gradient of sum_k lam[k] * f_k(h)."""
    grad = np.zeros(n)
    mid = n - 1
    for k_idx in range(len(lam)):
        w = lam[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - mid
        if k >= 0:
            valid = n - k
            if valid > 0:
                grad[:valid] += w * (1.0 - h[k:k+valid])
                grad[k:k+valid] -= w * h[:valid]
        else:
            m = -k
            valid = n - m
            if valid > 0:
                grad[m:m+valid] += w * (1.0 - h[:valid])
                grad[:valid] -= w * h[m:m+valid]
    grad *= dx
    return grad

def minimax_opt(h_init, n, target, time_limit, max_outer=200, eta=5.0, inner_iters=30):
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    mid = n - 1
    
    obj_prev = None
    
    for outer in range(max_outer):
        if time.time() - start_t > time_limit:
            break
        
        # Inner optimization: projected gradient for sum_k lam[k] f_k(h)
        step_size = 0.1 / (1.0 + 0.05 * outer)
        for inner in range(inner_iters):
            if time.time() - start_t > time_limit:
                break
            grad = _weighted_grad(h, lam, n, dx)
            grad -= np.mean(grad)
            
            # Backtracking
            for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step_size * mult * grad, target)
                obj_new = _weighted_obj(h_try, lam, n, dx)
                if obj_new < (obj_prev if obj_prev is not None else obj_new + 1.0):
                    h = h_try
                    obj_prev = obj_new
                    break
        
        # Compute f_k values
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Multiplicative weight update
        f_max = np.max(corr)
        f_min = np.min(corr)
        rng = f_max - f_min
        if rng > 1e-12:
            f_norm = (corr - f_min) / rng
        else:
            f_norm = np.zeros(n_corr)

            best_h = h.copy()
