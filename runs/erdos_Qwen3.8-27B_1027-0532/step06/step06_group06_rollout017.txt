def smooth_objective(h, beta, target, penalty):

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)

    grad = np.zeros(N)

        idx2 = t_arr - j + N - 1

    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

        else:

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        s1 = 0.0
        s2 = 0.0
        for k in range(-(N - 1), N):
            if 0 <= j + k < N:
                s1 += weights[k + N - 1] * one_minus_h[j + k]
            if 0 <= j - k < N:
                s2 += weights[k + N - 1] * h[j - k]
        grad[j] = s1 - s2
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

    N = len(h)
    dx = 2.0 / N

        else:
            stall += 10

    return best_h, best_c5

def sa_adaptive(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 12000
    neighborhood_width = 1

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            neighborhood_width = min(neighborhood_width + 1, 4)
            stall = 0

        ratio = min(it / max(max_iters - 1, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < neighborhood_width or len(zeros) < neighborhood_width:
            break

        h_new = h.copy()
        moved = False

        r = rng.random()
        if r < 0.6:
            worst_count = min(12, max(neighborhood_width, 4))
            worst_idx = np.argsort(corr)[-worst_count:]
            k = int(worst_idx[rng.randint(len(worst_idx))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            if ip:
                ip_filtered = [i for i in ip if h[i + k] == 0]
                if ip_filtered:
                    ip = ip_filtered
            jp = [j for j in zeros if 0 <= j + k < n]
            if jp:
                jp_filtered = [j for j in jp if h[j + k] == 1]
                if jp_filtered:
                    jp = jp_filtered
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.85:
            h_new[ones[rng.randint(len(ones))]] = 0
            h_new[zeros[rng.randint(len(zeros))]] = 1
            moved = True
        else:
            if neighborhood_width >= 2 and len(ones) >= 2 and len(zeros) >= 2:
                nmoves = min(neighborhood_width, 3)
                idx1 = rng.choice(len(ones), nmoves, replace=False)
                idx2 = rng.choice(len(zeros), nmoves, replace=False)
                for idx in idx1:
                    h_new[ones[idx]] = 0
                for idx in idx2:
                    h_new[zeros[idx]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        accept = False
        if delta <= 0:
            accept = True
        elif t > 1e-15:
            accept = rng.random() < np.exp(-delta / t)

        if accept:
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
                if neighborhood_width > 1:
                    neighborhood_width = 1
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 8.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin_init = project_binary(h_init, target_int)
    c5_bin_init = compute_c5(h_bin_init)
    if c5_bin_init < best_c5:
        best_c5 = c5_bin_init
        best_h = h_bin_init.copy()

    # Phase 1: Continuous warmup
    if tleft() > 60:
        h_cont = best_h.copy()
        time_alloc = min(60, tleft() * 0.12)
        h_opt = lbfgs_phase(h_cont, [20, 50, 100], target, 500.0, 200, time_alloc, start_time)
        sc = compute_c5(h_opt)
        h_cont = h_opt
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()

    # Phase 2: High-beta continuous optimization
    if tleft() > 100:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta_chunk in [[200, 400], [600, 1000], [2000, 4000], [6000, 10000]]:
            if tleft() < 25:
                break
            time_alloc = min(40, tleft() * 0.15)
            h_opt = lbfgs_phase(h_cont, beta_chunk, target, 1000.0, 250, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 3: Thorough binary greedy local search
    if tleft() > 80:
        h_bin = project_binary(best_h, target_int)
        time_alloc = min(120, tleft() * 0.22)
        h_ls, c5_ls = enhanced_greedy_binary(h_bin, target_int, time_limit=time_alloc,
                                              start_time=start_time, scan_width=70, top_k_count=12)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: Continue binary greedy from best
    if tleft() > 50:
        h_bin = project_binary(best_h, int(len(best_h) / 2))
        time_alloc = min(70, tleft() * 0.12)
        h_ls2, c5_ls2 = enhanced_greedy_binary(h_bin, int(len(best_h) / 2), time_limit=time_alloc,
                                                start_time=start_time, scan_width=90, top_k_count=15)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 5: SA with adaptive neighborhood
    if tleft() > 60:
        h_bin = project_binary(best_h, int(len(best_h) / 2))
        time_alloc = min(100, tleft() * 0.18)
        max_it = int(min(400000, time_alloc * 1200))
        h_sa, c5_sa = sa_adaptive(h_bin, int(len(best_h) / 2), max_it, 0.005, 0.000005,
                                   rng, time_limit=time_alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Continuous refinement from best binary
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_cont = project_box_sum(best_h.copy(), target_use)
        for beta_chunk in [[500, 1500], [3000, 6000], [10000, 20000]]:
            if tleft() < 20:
                break
            time_alloc = min(30, tleft() * 0.15)
            h_opt = lbfgs_phase(h_cont, beta_chunk, target_use, 2000.0, 200, time_alloc, start_time)
            sc = compute_c5(h_opt)
            h_cont = h_opt
            if sc < best_c5:
                best_c5 = sc
                best_h = h_opt.copy()

    # Phase 7: Alternate binary/continuous
    if tleft() > 80:
        for cycle in range(4):
            if tleft() < 30:
                break
            n_use = len(best_h)
            target_use = int(n_use / 2)
            h_bin_c = project_binary(best_h, target_use)
            time_alloc_b = min(35, tleft() * 0.15)
            h_b, c5_b = enhanced_greedy_binary(h_bin_c, target_use, time_limit=time_alloc_b,
                                                start_time=start_time, scan_width=60, top_k_count=10)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            if tleft() < 20:
                break
            h_cont_c = project_box_sum(best_h.copy(), float(n_use / 2))
            time_alloc_c = min(25, tleft() * 0.12)
            h_c = lbfgs_phase(h_cont_c, [2000, 8000], float(n_use / 2), 2000.0, 150, time_alloc_c, start_time)
            sc_c = compute_c5(h_c)
            if sc_c < best_c5:
                best_c5 = sc_c
                best_h = h_c.copy()

    # Phase 8: Multi-restart binary
    if tleft() > 60:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        for restart in range(4):
            if tleft() < 25:
                break
            sigma = 0.08 + 0.03 * restart
            perturbation = rng.normal(0, sigma, n_use)
            h_pert = best_h + perturbation
            h_bin_r = project_binary(h_pert, target_use)
            time_alloc_r = min(25, tleft() * 0.12)
            h_r, c5_r = enhanced_greedy_binary(h_bin_r, target_use, time_limit=time_alloc_r,
                                                start_time=start_time, scan_width=50, top_k_count=8)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 9: Try nearby grid sizes
    n_options = [n + 2, n - 2, n + 4, n + 6]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]
    for n_try in n_options:
        if tleft() < 35:
            break
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        time_alloc_i = min(20, tleft() * 0.1)
        h_opt = lbfgs_phase(h_interp, [100, 500], target_try, 500.0, 150, time_alloc_i, start_time)
        sc = compute_c5(h_opt)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt.copy()
        if tleft() < 20:
            continue
        h_bin_t = project_binary(h_opt, target_try_int)
        time_alloc_g = min(15, tleft() * 0.08)
        h_gt, c5_gt = enhanced_greedy_binary(h_bin_t, target_try_int, time_limit=time_alloc_g,
                                              start_time=start_time, scan_width=40, top_k_count=8)
        if c5_gt < best_c5:
            best_c5 = c5_gt
            best_h = h_gt.copy()

    # Phase 10: Final SA polish
    if tleft() > 35:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        time_alloc_f = min(40, tleft() - 5)
        max_it_f = int(min(200000, time_alloc_f * 1500))
        h_sf, c5_sf = sa_adaptive(h_bin_f, target_use, max_it_f, 0.003, 0.000003,
                                   rng, time_limit=time_alloc_f, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 11: Final continuous polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_final = project_box_sum(best_h.copy(), target_use)
        time_alloc_p = min(12, tleft() - 3)
        h_polished = lbfgs_phase(h_final, [5000, 15000], target_use, 3000.0, 100, time_alloc_p, start_time)
        sc_p = compute_c5(h_polished)
        if sc_p < best_c5:
            best_c5 = sc_p
            best_h = h_polished.copy()

    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
