def smooth_obj_and_grad(x, n, target, alpha):
    """Log-sum-exp objective and gradient."""
    h = np.zeros(n)
    h[:n-1] = x
    h[n-1] = target - np.sum(x)
    h = np.clip(h, 0.0, 1.0)  # Clip for evaluation (but gradient won't account for this)
    
    overlaps = compute_overlaps(h)
    # Shift to avoid overflow
    max_o = np.max(overlaps)
    shifted = alpha * (overlaps - max_o)
    shifted = np.clip(shifted, -500, 0)
    log_sum = max_o * alpha + np.log(np.sum(np.exp(shifted)))
    obj = log_sum / alpha
    
    # Gradient
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # d O(k) / d h[i]
    dx = 2.0 / n
    n_half = n - 1
    grad_h = np.zeros(n)
    for idx in range(len(overlaps)):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - n_half
        if k == 0:
            grad_h += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad_h[:n-k] += w * (1.0 - h[k:]) * dx
            grad_h[k:] += w * (-h[:n-k]) * dx
        else:
            m = -k
            grad_h[m:] += w * (1.0 - h[:n-m]) * dx
            grad_h[:n-m] += w * (-h[m:]) * dx
    
    # Chain rule: x = h[0:n-1], h[n-1] = target - sum(x)
    grad_x = grad_h[:n-1] - grad_h[n-1]
    return obj, grad_x

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

            # Contributing positions: i where h[i]=1 and h[i+k_star]=0
            contrib = np.array([i for i in range(n - k_star) 
                              if h[i] == 1.0 and h[i+k_star] == 0.0])
        elif k_star < 0:

            # Contributing positions: i+m where h[i+m]=1 and h[i]=0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += room * (diff / tr)
        else:
            av = h.copy()
            ta = av.sum()
            if ta < 1e-14:
                break
            h -= av * ((-diff) / ta)
        h = np.clip(h, 0.0, 1.0)

    return float(mo + np.log(np.sum(np.exp(sh))) / alpha)
