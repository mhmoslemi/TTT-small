def compute_c5_correlation(h):
    """Compute C5 from correlation array directly (for binary h)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / len(h))

def compute_c5_from_corr(corr, n):
    return float(np.max(corr)) * (2.0 / n)

def build_from_intervals(intervals, n):
    """Build binary h from list of (start, length) intervals."""
    h = np.zeros(n)
    for start, length in intervals:
        s = int(round(start))
        l = int(round(length))
        if s >= 0 and s + l <= n and l > 0:
            h[s:s+l] = 1.0
    return h

def c5_from_binary_array(h):
    """Compute C5 for a binary array."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def interval_objective(x, n, target_sum):
    """
    x = [start1, len1, start2, len2, start3, len3, ...]
    Objectives: minimize C5 subject to sum of lengths = target_sum
    """
    n_intervals = len(x) // 2
    h = np.zeros(n)
    total_len = 0
    for i in range(n_intervals):
        s = x[2*i]
        l = x[2*i+1]
        total_len += l
        s_i = int(np.clip(s, 0, n-1))
        l_i = int(np.clip(l, 0, n))
        if l_i > 0:
            h[s_i:s_i+l_i] = 1.0
    
    # Enforce exact sum
    actual_sum = int(h.sum())
    if actual_sum != int(target_sum):
        # Penalty

    to h[a]=0,h[b]=1.
    Returns the new max correlation value (before scaling by dx).
    """

    # corr has length 2n-1, index s corresponds to lag d = s-(n-1)

    # c[d] for d>=0: sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
    # c[d] for d<0:  sum_{i=-d}^{n-1} h[i+d]*(1-h[i])

    # For the sum sum_i h[i+d]*(1-h[i]):
    #   i=a: h[a+d]*(1-h[a]) → h[a+d]*(1-0) = h[a+d]. Change: +h[a+d] (since h[a] was 1, term was 0)
    #   i=a-d: h[a]*(1-h[a-d]) → 0*(1-h[a-d]) = 0. Change: -(1-h[a-d]) = h[a-d]-1
    #   i=b: h[b+d]*(1-h[b]) → h[b+d]*(1-1) = 0. Change: -h[b+d] (since h[b] was 0, term was h[b+d])

    #   - If i = a is in range: term was h[a+d]*0 = 0, becomes h[a+d]*1 = h[a+d]. Δ = +h[a+d]
    #     BUT if a+d = b, h[b] also changed from 0 to 1, so h[a+d] = h[b] = 1 (new value)

    #   If b-d = a: h_new[b-d] = h_new[a] = 0, so new = 1, Δ = +1
    #   If b-d ≠ a: h_new[b-d] = h[b-d], so new = 1-h[b-d], Δ = 1-h[b-d]
    #   In general: Δ = 1 - h_new[b-d]
    
    # So the total change at output index s (lag d = s-(n-1)):
    # Δ = 0
    #   + (if i=a valid): h_new[a+d]
    #   + (if i=a-d valid): -(1-h[a-d])   [using old h for a-d since only a and b changed, and a-d≠a]
    #     but if a-d = b, this is -(1-h[b]) = -(1-0) = -1
    #     and h_new[a-d] = h_new[b] = 1, but we use the old value in the formula... 
    #     Wait, I said Δ = -(1-h[a-d]) where h[a-d] is the OLD value. If a-d=b, old h[b]=0, so Δ = -1. ✓
    #   + (if i=b valid): -h[b+d]   [old value; if b+d=a, old h[a]=1, Δ=-1] ✓
    #   + (if i=b-d valid): 1-h_new[b-d]
    #     if b-d=a: h_new[a]=0, Δ=1. Otherwise: Δ=1-h[b-d] (old value = new value since b-d≠a,b)

    # i=a valid: 0 <= a <= n-1 (always) AND 0 <= a+d <= n-1
    #   In terms of s: 0 <= a + s - (n-1) <= n-1, i.e., (n-1-a) <= s <= (n-1+a)

    #   0 <= a - s + (n-1) means s <= a + (n-1)
    #   a - s + (n-1) <= n-1 means a <= s, i.e., s >= a
    #   So: a <= s <= a + (n-1)
    #   But also need the original sum to be valid: this term exists in the sum only if
    #   i = a-d is in [max(0,-d), min(n-1, n-1-d)]
    #   a-d >= max(0, -d) and a-d <= n-1-d
    #   a-d >= 0 → a >= d → s-(n-1) <= a → s <= a+(n-1) ✓ (already have)
    #   a-d >= -d → a >= 0 ✓
    #   a-d <= n-1-d → a <= n-1 ✓
    #   a-d <= n-1 → a-s+(n-1) <= n-1 → a <= s ✓ (already have)
    #   So validity: a <= s <= a+(n-1)
    
    # i=b valid: 0 <= b+d <= n-1, i.e., (n-1-b) <= s <= (n-1+b)
    
    # i=b-d valid: 0 <= b-d <= n-1, which gives b <= s <= b+(n-1)
    
    return changes  # sum of the contributions

  For a one-for-one swap (set h[b]=1 and h[a]=0):

  - Option (b): a = i+d* (a 1 that has a 0 at distance d* behind it), b = some 0

  The question is: which specific a and b to choose?
  - For option (a): b is determined by the worst lag (a 0 near a 1 at distance d*). a should be a 1 whose removal doesn't hurt other lags too much.
  - For option (b): a is determined by the worst lag. b should be a 0 whose addition doesn't hurt other lags.

  A good heuristic: try all b's of type (a) and a few a's, pick the best.

def targeted_local_search(h, n, target_sum, time_limit, start_time, rng):
    """Local search that targets the worst lags."""
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h = h.copy()
    current_max = float(corr.max())
    best_max = current_max
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    stall = 0
    max_stall = 2000
    
    while time.time() - start_time < time_limit and stall < max_stall:
        # Focus on top-3 worst lags
        top_idx = np.argsort(corr)[-3:][::-1]
        
        best_improve = 0.0
        best_a = -1
        best_b = -1
        best_new_corr = None
        
        for ts in top_idx:
            d = ts - (n-1)
            
            if d > 0:
                # c[d] = #{i in [0, n-1-d] : h[i+d]=1, h[i]=0}
                # Move type A: set h[i]=1 for i where h[i]=0, h[i+d]=1
                #   b = i, a = some 1
                # Move type B: set h[i+d]=0 for i where h[i+d]=1, h[i]=0
                #   a = i+d, b = some 0
                
                # Get candidate b's for type A (0s that have a 1 at distance d)
                b_cands = [i for i in zeros_list if i <= n-1-d and h[i+d] == 1]
                # Get candidate a's for type B (1s that have a 0 at distance d behind)
                a_cands = [i+d for i in range(n-d) if h[i+d] == 1 and h[i] == 0]
                
                # Try type A: b from b_cands, a from random subset of ones
                for b in b_cands[:20]:
                    # a should be a 1 whose removal helps
                    # Try a few random ones
                    for _ in range(5):
                        a = ones_list[rng.randint(len(ones_list))]
                        if a == b:
                            continue
                        delta = swap_delta(corr, h, a, b, n)
                        new_max = float(np.max(corr + delta))
                        if current_max - new_max > best_improve:
                            best_improve = current_max - new_max
                            best_a = a
                            best_b = b
                            best_new_corr = corr + delta
                
                # Try type B: a from a_cands, b from random subset of zeros
                for a in a_cands[:20]:
                    for _ in range(5):
                        b = zeros_list[rng.randint(len(zeros_list))]
                        if a == b:
                            continue
                        delta = swap_delta(corr, h, a, b, n)
                        new_max = float(np.max(corr + delta))
                        if current_max - new_max > best_improve:
                            best_improve = current_max - new_max
                            best_a = a
                            best_b = b
                            best_new_corr = corr + delta
            elif d < 0:
                d_prime = -d
                # c[-d'] = #{j in [0, n-1-d'] : h[j]=1, h[j+d']=0}
                # Move type A: set h[j+d']=1 where h[j]=1, h[j+d']=0
                #   b = j+d', a = some 1
                # Move type B: set h[j]=0 where h[j]=1, h[j+d']=0
                #   a = j, b = some 0
                
                b_cands = [j+d_prime for j in range(n-d_prime) if h[j] == 1 and h[j+d_prime] == 0]
                a_cands = [j for j in range(n-d_prime) if h[j] == 1 and h[j+d_prime] == 0]
                
                for b in b_cands[:20]:
                    for _ in range(5):
                        a = ones_list[rng.randint(len(ones_list))]
                        if a == b:
                            continue
                        delta = swap_delta(corr, h, a, b, n)
                        new_max = float(np.max(corr + delta))
                        if current_max - new_max > best_improve:
                            best_improve = current_max - new_max
                            best_a = a
                            best_b = b
                            best_new_corr = corr + delta
                
                for a in a_cands[:20]:
                    for _ in range(5):
                        b = zeros_list[rng.randint(len(zeros_list))]
                        if a == b:
                            continue
                        delta = swap_delta(corr, h, a, b, n)
                        new_max = float(np.max(corr + delta))
                        if current_max - new_max > best_improve:
                            best_improve = current_max - new_max
                            best_a = a
                            best_b = b
                            best_new_corr = corr + delta
        
        # Also try some random swaps
        for _ in range(20):
            a = ones_list[rng.randint(len(ones_list))]
            b = zeros_list[rng.randint(len(zeros_list))]
            delta = swap_delta(corr, h, a, b, n)
            new_max = float(np.max(corr + delta))
            if current_max - new_max > best_improve:
                best_improve = current_max - new_max
                best_a = a
                best_b = b
                best_new_corr = corr + delta
        
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            corr = best_new_corr
            current_max = float(corr.max())
            ones_list = list(np.where(h == 1)[0])
            zeros_list = list(np.where(h == 0)[0])
            if current_max < best_max - 1e-12:
                best_max = current_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_max * (2.0 / n)
