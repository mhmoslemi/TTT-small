        h = np.clip(h, 0.0, 1.0)

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    val = c_max + np.log(total) / alpha

    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < min_delta:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def triple_cd(h_init, n, target, time_limit, max_sweeps=50000,
              n_cand=25, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_config = None

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                for kk in range(jj + 1, n_l):
                    k = int(low_set[kk])
                    if k == i:
                        continue

                    g_net = change[j] + change[k] - change[i]
                    max_delta_i = min(h[i], (1.0 - h[j]) + (1.0 - h[k]))
                    max_delta_j = (1.0 - h[j])
                    max_delta_k = (1.0 - h[k])
                    max_delta_jk = max_delta_j + max_delta_k

                    if max_delta_i < 1e-12:
                        continue

                    g_vals = corr + np.linspace(0, max_delta_i, 40)[:, None] * g_net[None, :]
                    row_max = np.max(g_vals, axis=1)
                    best_row = int(np.argmin(row_max))
                    delta_i_candidate = np.linspace(0, max_delta_i, 40)[best_row]

                    if delta_i_candidate < 1e-10:
                        continue

                    approx_max = float(np.min(row_max)) * dx
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max

                        split = 0.5
                        if max_delta_j > 0:
                            dj = min(delta_i_candidate * split, max_delta_j)
                        else:
                            dj = 0.0
                        dk = min(delta_i_candidate - dj, max_delta_k)
                        dj = delta_i_candidate - dk

                        best_config = (i, j, k, delta_i_candidate, dj, dk)

        if best_config is not None:
            i, j, k, di, dj, dk = best_config
            h[i] -= di
            h[j] += dj
            h[k] += dk
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            update_change_row(change, k, h, n, shifts)

    return best_h, best_c5

def interpolated_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def projected_smooth_gradient(h_init, n, target, time_limit, alpha=2000,
                               top_k=20, max_iter=200000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = 2.0 / alpha
    momentum = np.zeros(n)
    beta = 0.7

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 50000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_s = np.exp(shifted)
        total = np.sum(exp_s)
        weights = exp_s / total

        top_idx = np.argsort(corr)[-top_k:]
        top_w = weights[top_idx]
        sw = top_w.sum()
        if sw < 1e-15:
            continue
        top_w /= sw

        grad = np.zeros(n)
        for ti in range(len(top_idx)):
            if top_w[ti] < 1e-10:
                continue
            s_idx = int(top_idx[ti])
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += top_w[ti] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= top_w[ti] * h[idxs[v2] + k]

        grad -= np.mean(grad)

        momentum = beta * momentum - lr * grad
        h_try = project_feasible(h + momentum, target)
        new_c5 = compute_c5(h_try)

        if new_c5 < c5_val - 1e-15:
            h = h_try
            lr = min(lr * 1.02, 20.0 / alpha)
        else:
            lr *= 0.5
            momentum = np.zeros(n)
            if lr < 1e-8:
                lr = 2.0 / alpha
                momentum = np.zeros(n)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol, sc = slsqp_polish(best_h, n0, target0, min(40, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 1000
    target1 = n1 / 2.0
    h1 = interpolated_h(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    if tleft() > 150:
        h_cd, c5_cd = pair_cd_adaptive(h1, n1, target1,
                                        time_limit=min(120, tleft() * 0.4),
                                        max_sweeps=40000,
                                        n_cand_start=20, n_cand_end=50,
                                        max_no_improve=300, top_shifts=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n2 = 1600
    target2 = n2 / 2.0
    h2 = interpolated_h(best_h, n2, target2)
    sc2 = compute_c5(h2)
    if sc2 < best_score:
        best_score = sc2
        best_h = h2

    if tleft() > 200:
        h_cd2, c5_cd2 = pair_cd_adaptive(h2, n2, target2,
                                          time_limit=min(250, tleft() * 0.5),
                                          max_sweeps=50000,
                                          n_cand_start=25, n_cand_end=55,
                                          max_no_improve=400, top_shifts=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    if tleft() > 100:
        h_poly, sc_poly = slsqp_polish(best_h, n_cur, target_cur,
                                        min(80, tleft() * 0.4))
        if sc_poly < best_score:
            best_score = sc_poly
            best_h = h_poly

    if tleft() > 150:
        h_cd3, c5_cd3 = pair_cd_adaptive(best_h, n_cur, target_cur,
                                          time_limit=min(120, tleft() * 0.45),
                                          max_sweeps=40000,
                                          n_cand_start=30, n_cand_end=65,
                                          max_no_improve=500, top_shifts=4)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 100:
        h_tri, c5_tri = triple_cd(best_h, n_cur, target_cur,
                                   time_limit=min(70, tleft() * 0.5),
                                   max_sweeps=20000, n_cand=20,
                                   max_no_improve=200)
        if c5_tri < best_score:
            best_score = c5_tri
            best_h = h_tri

    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    if tleft() > 100:
        h_pg, c5_pg = projected_smooth_gradient(best_h, n_cur, target_cur,
                                                  time_limit=min(70, tleft() * 0.5),
                                                  alpha=5000, top_k=15)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg
            n_cur = len(best_h)
            target_cur = n_cur / 2.0

    if tleft() > 80:
        h_cd4, c5_cd4 = pair_cd_adaptive(best_h, n_cur, target_cur,
                                          time_limit=min(60, tleft() * 0.5),
                                          max_sweeps=30000,
                                          n_cand_start=35, n_cand_end=70,
                                          max_no_improve=300, top_shifts=3)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    if tleft() > 50:
        h_pf, sc_pf = slsqp_polish(best_h, len(best_h), len(best_h) / 2.0,
                                    min(40, tleft() * 0.6))
        if sc_pf < best_score:
            best_score = sc_pf
            best_h = h_pf

    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 40:
        h_cd5, c5_cd5 = pair_cd_adaptive(best_h, n_cur, target_cur,
                                          time_limit=min(30, tleft() * 0.6),
                                          max_sweeps=20000,
                                          n_cand_start=40, n_cand_end=75,
                                          max_no_improve=200, top_shifts=3)
        if c5_cd5 < best_score:
            best_score = c5_cd5
            best_h = h_cd5

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
