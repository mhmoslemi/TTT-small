def slp_all_constraints(h0, n, target, time_limit, max_rounds=100):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # SLP step: solve LP with linearized constraints at all shifts
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for s in range(n_corr):
            k = s - mid
            # corr[s] = sum_i h[i] * (1-h[i+k]) (with boundary)
            # Linearize around h:
            # corr[s] ≈ corr[s]_current + grad @ (h_var - h)
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s] + grad_k @ (h_var - h)
            constraints.append(lin_val * dx <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_val - 1e-12:
            h = h_new
        else:
            # Try mixing
            for alpha_mix in [0.5, 0.75, 0.25]:
                h_mix = project_feasible(alpha_mix * h + (1 - alpha_mix) * h_new, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-12:
                    h = h_mix
                    break
            else:
                break
    
    return best_h, best_c5

    # Phase 1: SLSQP with progressive alpha
    if tleft() > 180:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [100, 500, 2000, 10000, 50000, 200000],
                                5000, 1e-16, min(180.0, tleft() * 0.45), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: SLP with all constraints (key new feature)
    if tleft() > 250:
        h_slp, s_slp = slp_all_constraints(best_h, n, target,
                                            time_limit=min(250.0, tleft() * 0.4),
                                            max_rounds=300, trust_init=0.03)
        if s_slp < best_score:
            best_score = s_slp
            best_h = h_slp

    # Phase 3: SLSQP polish after SLP
    if tleft() > 120:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [50000, 200000, 500000, 1000000],
                                5000, 1e-17, min(120.0, tleft() * 0.35), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: SLP top-k for additional refinement
    if tleft() > 150:
        h_slp, s_slp = slp_top_k(best_h, n, target,
                                   time_limit=min(150.0, tleft() * 0.35),
                                   max_rounds=80, k_top=40)
        if s_slp < best_score:
            best_score = s_slp
            best_h = h_slp

    # Phase 5: Coordinate descent
    if tleft() > 120:
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=min(120.0, tleft() * 0.3),
                                      max_sweeps=15000, n_cand=25)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Top-k subgradient (Nesterov accelerated)
    if tleft() > 100:
        h_sg, c5_sg = topk_subgradient(best_h, n, target,
                                        time_limit=min(100.0, tleft() * 0.3),
                                        max_steps=30000, topk=3)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: SLP all constraints again
    if tleft() > 150:
        h_slp, s_slp = slp_all_constraints(best_h, n, target,
                                            time_limit=min(150.0, tleft() * 0.35),
                                            max_rounds=200, trust_init=0.02)
        if s_slp < best_score:
            best_score = s_slp
            best_h = h_slp

    # Phase 8: SLSQP very high alpha
    if tleft() > 80:
        h_p, s_p = slsqp_round(best_h, n, target,
                                [200000, 500000, 1000000, 2000000],
                                8000, 1e-17, min(80.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 9: Random perturbation attempts
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for attempt in range(3):
            if tleft() < 40:
                break
            noise_scale = 0.02 * (attempt + 1)
            h_pert = project_feasible(best_h + rng.randn(n) * noise_scale, target)
            h_p2, s_p2 = slsqp_round(h_pert, n, target,
                                      [10000, 50000, 200000],
                                      3000, 1e-16, min(30.0, tleft() * 0.3), start_time)
            if s_p2 < best_score:
                best_score = s_p2
                best_h = h_p2

    # Phase 10: CD final
    if tleft() > 50:
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=min(50.0, tleft() * 0.4),
                                      max_sweeps=8000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: SLP final
    if tleft() > 60:
        h_slp, s_slp = slp_top_k(best_h, n, target,
                                   time_limit=min(60.0, tleft() * 0.45),
                                   max_rounds=50, k_top=60)
        if s_slp < best_score:
            best_score = s_slp
            best_h = h_slp

    # Phase 12: Try higher resolution if time allows
    if tleft() > 180:
        n_try = n * 2
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, s_hires = slsqp_round(h_interp, n_try, target_try,
                                        [500, 5000, 50000, 200000],
                                        3000, 1e-15, min(100.0, tleft() * 0.4), start_time)
        if s_hires < best_score:
            best_score = s_hires
            best_h = h_hires
        if tleft() > 80:
            h_slp2, s_slp2 = slp_all_constraints(best_h, len(best_h), len(best_h)/2.0,
                                                  time_limit=min(80.0, tleft() * 0.4),
                                                  max_rounds=100, trust_init=0.02)
            if s_slp2 < best_score:
                best_score = s_slp2
                best_h = h_slp2

    # Phase 13: Final SLSQP + CD polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_round(best_h, n_use, target_use,
                                        [500000, 1000000, 2000000],
                                        5000, 1e-17,
                                        min(25.0, tleft() * 0.6), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = fast_greedy_cd(best_h, n_use, target_use,
                                      time_limit=min(12.0, tleft() * 0.7),
                                      max_sweeps=3000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check and output
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
