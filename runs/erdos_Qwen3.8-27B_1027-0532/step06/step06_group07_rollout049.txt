import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):

def frank_wolfe_step(g, n, target):
    """Given gradient g, find argmin_{h in feasible} <g, h>"""
    # Sort by gradient value
    order = np.argsort(g)
    # We want h[j]=1 for small g, h[j]=0 for large g, one partial
    # Target sum = target
    # Greedy: assign 1 to smallest g until we hit target
    h_fw = np.zeros(n)
    remaining = target
    for i in order:
        if remaining >= 1.0 - 1e-12:
            h_fw[i] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            h_fw[i] = remaining
            remaining = 0.0
        else:
            break
    return h_fw

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd_precise(h, n, target, time_limit, max_sweeps=5000, n_cand=50, n_bisect=18, n_active_weight=3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 100:
            break

        if n_active_weight <= 1:
            k_list = [int(np.argmax(corr)) - mid]
            w_list = [1.0]
        else:
            top_idx = np.argsort(corr)[-n_active_weight:][::-1]
            k_list = [int(si) - mid for si in top_idx]
            gaps = corr[top_idx] - corr[top_idx[0]]
            w_list = np.exp(8.0 * gaps)
            w_list = w_list / w_list.sum()

        grad = np.zeros(n)
        for k_val, w_val in zip(k_list, w_list):
            v1 = (idxs - k_val >= 0) & (idxs - k_val < n)
            v2 = (idxs + k_val >= 0) & (idxs + k_val < n)
            grad[v1] += w_val * (1.0 - h[idxs[v1] - k_val])
            grad[v2] -= w_val * h[idxs[v2] + k_val]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            if hi < 1e-12:
                continue
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5


def adaptive_momentum_subgrad(h, n, target, time_limit, max_iter=5000, n_active=5, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    grad_sq = np.ones(n) * 1e-8
    velocity = np.zeros(n)
    base_lr = 0.08
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break

        top_idx = np.argsort(corr)[-n_active:][::-1]
        gaps = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(6.0 * gaps)
        weights = weights / weights.sum()

        g = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += w * (1.0 - h[idxs[v1] - k])
            g[v2] -= w * h[idxs[v2] + k]

        g -= np.mean(g)
        grad_sq += g * g
        lr = base_lr / np.sqrt(grad_sq)

        velocity = momentum * velocity + lr * g
        h_try = project_feasible(h - velocity, target)
        c5_try = compute_c5(h_try)

        if c5_try < c5_val - 1e-14:
            h = h_try
        else:
            velocity *= 0.5
            h_try2 = project_feasible(h - velocity, target)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < c5_val - 1e-14:
                h = h_try2
            else:
                base_lr *= 0.98

    return best_h, best_c5


def gradient_descent_fw(h, n, target, time_limit, max_iter=3000, n_active=5):

        if no_improve > 300:
            break

        top_idx = np.argsort(corr)[-n_active:][::-1]
        gaps = corr[top_idx] - corr[top_idx[0]]

        weights = weights / weights.sum()

        g = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += w * (1.0 - h[idxs[v1] - k])
            g[v2] -= w * h[idxs[v2] + k]

        g -= np.mean(g)

        order = np.argsort(g)
        h_fw = np.zeros(n)
        remaining = target
        for idx in order:
            if remaining >= 1.0 - 1e-12:
                h_fw[idx] = 1.0
                remaining -= 1.0
            elif remaining > 1e-12:
                h_fw[idx] = remaining
                remaining = 0.0
            else:
                break

        gamma = 2.0 / (it + 2.0)
        gamma = min(gamma, 0.3)
        h_try = (1.0 - gamma) * h + gamma * h_fw
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)

        if c5_try < c5_val - 1e-15:
            h = h_try

    return best_h, best_c5

        h_cd, c5_cd = pair_cd_precise(h_sg, n, target, time_limit=cd_time,

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP quick polish
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 10000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Precise pair CD with multi-active gradient
    if tleft() > 250:
        cd_time = min(300, tleft() * 0.25)
        h_cd, c5_cd = pair_cd_precise(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=6000, n_cand=60, n_bisect=20, n_active_weight=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP mid-alpha
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Adaptive momentum subgradient
    if tleft() > 150:
        ms_time = min(180, tleft() * 0.18)
        h_ms, c5_ms = adaptive_momentum_subgrad(best_h, n, target, time_limit=ms_time,
                                                 max_iter=5000, n_active=5, momentum=0.92)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 5: Another precise pair CD round (larger candidates)
    if tleft() > 200:
        cd_time = min(220, tleft() * 0.2)
        h_cd, c5_cd = pair_cd_precise(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=5000, n_cand=75, n_bisect=20, n_active_weight=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: SLSQP high alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [20000, 80000, 200000], 1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Perturb and optimize (escape local optimum)
    if tleft() > 200:
        pt_time = min(200, tleft() * 0.2)
        h_pt, c5_pt = perturb_and_optimize(best_h, n, target, time_limit=pt_time, seed_val=seed)
        if c5_pt < best_score:
            best_score = c5_pt
            best_h = h_pt

    # Phase 8: FW gradient descent (complementary direction)
    if tleft() > 100:
        fw_time = min(100, tleft() * 0.12)
        h_fw, c5_fw = gradient_descent_fw(best_h, n, target, time_limit=fw_time,
                                           max_iter=3000, n_active=5)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 9: Final precise pair CD
    if tleft() > 100:
        cd_time = min(120, tleft() * 0.2)
        h_cd, c5_cd = pair_cd_precise(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=4000, n_cand=70, n_bisect=20, n_active_weight=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP polish
    if tleft() > 30:
        h_final = slsqp_round(best_h, n, target, [50000, 200000], 600, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
