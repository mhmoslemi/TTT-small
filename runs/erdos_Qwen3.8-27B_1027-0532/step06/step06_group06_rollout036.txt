def compute_swap_delta(h, i, j):
    """Compute the delta vector for swapping h[i] (1→0) and h[j] (0→1)."""
    n = len(h)
    delta = np.zeros(2*n - 1)  # same size as correlation array
    
    # For shift k (corresponding to array index k + n - 1):
    # delta_k = -(1-h[i+k])·[0≤i+k<n] + h[i-k]·[0≤i-k<n] + (1-h[j+k])·[0≤j+k<n] - h[j-k]·[0≤j-k<n]
    
    # Term 1: -(1-h[i+k]) for k where 0 ≤ i+k < n, i.e., -i ≤ k < n-i
    k_min1 = max(-i, -(n-1))
    k_max1 = min(n-1-i, n-1)
    for k in range(k_min1, k_max1+1):
        delta[k + n - 1] -= (1.0 - h[i+k])
    
    # Term 2: +h[i-k] for k where 0 ≤ i-k < n, i.e., i-n+1 < k ≤ i, i.e., i-n+1 ≤ k ≤ i
    k_min2 = max(-(n-1), i-n+1)
    k_max2 = min(n-1, i)
    for k in range(k_min2, k_max2+1):
        delta[k + n - 1] += h[i-k]
    
    # Term 3: +(1-h[j+k]) for k where 0 ≤ j+k < n
    k_min3 = max(-j, -(n-1))
    k_max3 = min(n-1-j, n-1)
    for k in range(k_min3, k_max3+1):
        delta[k + n - 1] += (1.0 - h[j+k])
    
    # Term 4: -h[j-k] for k where 0 ≤ j-k < n
    k_min4 = max(-(n-1), j-n+1)
    k_max4 = min(n-1, j)
    for k in range(k_min4, k_max4+1):
        delta[k + n - 1] -= h[j-k]
    
    return delta

    # delta[k+n-1] = -(1-h[i+k])*[0<=i+k<n] + h[i-k]*[0<=i-k<n] + (1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(np.clip(h, 0, 1)).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    while int(h_r.sum()) != target_sum:
        diff = int(h_r.sum()) - target_sum
        if diff > 0:
            idx = np.where(h_r == 1)[0]
            h_r[idx[np.argmin(h[idx])]] = 0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[np.argmax(h[idx])]] = 1
    return h_r.astype(float)

def topk_obj(h, K, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_array(h) * dx
    top_vals = np.sort(corr)[-K:]
    obj = float(np.mean(top_vals))
    if penalty > 0:
        d = float(np.sum(h)) - target
        obj += penalty * d * d
    return obj

def topk_grad(h, K, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_array(h)
    top_k_idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for idx in top_k_idx:
        k = int(idx) - (n - 1)
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            kp = -k
            grad[kp:] += (1.0 - h[:n-kp])
            grad[:n-kp] -= h[kp:]
    grad *= dx / K
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def run_lbfgs(h_start, K, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    for attempt in range(3):
        if time.time() - start_time >= time_limit:
            break
        try:
            res = minimize(topk_obj, h_cur,
                          args=(K, target, penalty),
                          jac=topk_grad,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1})
            h_candidate = project_box_sum(res.x, target)
            c5 = compute_c5(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
            h_cur = h_candidate
        except Exception:
            pass
    return best_h, best_c5

            t1 = np.where(mask_ik, 1.0 - h[idx_ik], 0.0)
            idx_ik2 = ii - ks
            mask_ik2 = (idx_ik2 >= 0) & (idx_ik2 < n)
            t2 = np.where(mask_ik2, h[idx_ik2], 0.0)
            delta_i = np.where(mask_ik, -(1.0 - h[idx_ik]), 0.0) + np.where(mask_ik2, h[idx_ik2], 0.0)
            for jj in j_cand:
                idx_jk = jj + ks
                mask_jk = (idx_jk >= 0) & (idx_jk < n)
                delta_j = np.where(mask_jk, 1.0 - h[idx_jk], 0.0)
                idx_jk2 = jj - ks
                mask_jk2 = (idx_jk2 >= 0) & (idx_jk2 < n)
                delta_j -= np.where(mask_jk2, h[idx_jk2], 0.0)
                new_max = float(np.max(corr + delta_i + delta_j))
                improve = max_val - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = ii
                    best_j = jj
        if best_i >= 0 and best_improve > 1e-10:
            h[best_i] = 0
            h[best_j] = 1
            corr = compute_corr_array(h)
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def sa_continuous(h_start, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5
    stall = 0
    max_stall = 3000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        h_new = h.copy()
        n_pert = rng.randint(1, 5)
        idx = rng.choice(n, n_pert, replace=False)
        h_new[idx] = np.clip(h_new[idx] + rng.normal(0, 0.1, n_pert), 0, 1)
        h_new = project_box_sum(h_new, target)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = float(n0 / 2)

    best_h = project_box_sum(h_init.copy(), target0)
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(h_init, int(n0 / 2))
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    K_schedule = [25, 15, 10, 7, 5, 4, 3, 3, 2, 2, 1, 1]
    for K in K_schedule:
        if tleft() < 25:
            break
        time_alloc = min(55, tleft() * 0.09)
        h_opt, c5_opt = run_lbfgs(best_h, K, target0, 500.0, 250, time_alloc, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        else:
            h_cur_proj = project_box_sum(h_opt, target0)
            c5_cur = compute_c5(h_cur_proj)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur_proj

    if tleft() > 40:
        n_cur = len(best_h)
        h_bin_cur = project_binary(best_h, int(n_cur / 2))
        c5_b = compute_c5(h_bin_cur)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin_cur.copy()
        time_alloc = min(80, tleft() * 0.15)
        h_ls, c5_ls = fast_binary_ls(best_h, int(n_cur / 2), time_alloc, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    larger_ns = [n0 + 50, n0 + 100, n0 + 200, n0 + 400, n0 + 600]
    larger_ns = [x for x in larger_ns if x % 2 == 0 and x > n0]
    for n_try in larger_ns:
        if tleft() < 50:
            break
        target_try = float(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        K_sch = [12, 7, 4, 3, 2, 1]
        for K in K_sch:
            if tleft() < 20:
                break
            time_alloc = min(30, tleft() * 0.06)
            h_opt, c5_opt = run_lbfgs(h_interp, K, target_try, 500.0, 150, time_alloc, start_time)
            h_interp = h_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        if tleft() < 25:
            continue
        h_bin_try = project_binary(h_interp, int(n_try / 2))
        c5_bt = compute_c5(h_bin_try)
        if c5_bt < best_c5:
            best_c5 = c5_bt
            best_h = h_bin_try.copy()
        time_alloc2 = min(30, tleft() * 0.05)
        h_ls2, c5_ls2 = fast_binary_ls(h_interp, int(n_try / 2), time_alloc2, start_time, max_scan=10)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = float(n_cur / 2)
        time_alloc = min(80, tleft() * 0.15)
        max_it = int(min(50000, time_alloc * 3000))
        h_sa, c5_sa = sa_continuous(best_h, target_cur, max_it, 0.05, 0.0001, rng, time_alloc, start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = float(n_cur / 2)
        for K in [2, 1, 1, 2, 1]:
            if tleft() < 10:
                break
            time_alloc = min(20, tleft() * 0.35)
            h_opt, c5_opt = run_lbfgs(best_h, K, target_cur, 1000.0, 200, time_alloc, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    if tleft() > 15:
        n_cur = len(best_h)
        h_bin_final = project_binary(best_h, int(n_cur / 2))
        c5_bf = compute_c5(h_bin_final)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bin_final.copy()
        time_alloc = min(25, tleft() - 5)
        h_ls_final, c5_lsf = fast_binary_ls(h_bin_final, int(n_cur / 2), time_alloc, start_time, max_scan=10)
        if c5_lsf < best_c5:
            best_c5 = c5_lsf
            best_h = h_ls_final.copy()

    n_out = len(best_h)
    best_h = project_box_sum(best_h, float(n_out / 2))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_out
