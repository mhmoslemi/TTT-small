def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

    return np.sum(h) - _global_target

    return np.ones(_global_n)


def slsqp_analytic(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=_smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def simulated_annealing(h_start, n, target, time_limit, max_iter, t_init, t_final, cooling):
    start_t = time.time()
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break

        t = t_init * (t_final / t_init) ** (it / max(1, max_iter - 1))

        n_move = rng.choice([1, 2, 3, 5, 8])
        n_move = min(n_move, n)
        idxs = rng.choice(n, size=n_move, replace=False)
        scale = 0.1 + 0.3 * t / t_init
        perturb = rng.randn(n_move) * scale
        h_try = h.copy()
        h_try[idxs] += perturb
        h_try = project_feasible(h_try, target)

        c5_try = compute_c5(h_try)
        delta = c5_try - best_c5

        if delta < 0:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
        else:
            accept_prob = np.exp(-delta / max(t, 1e-10))
            if rng.random() < accept_prob:
                h = h_try

    return best_h, best_c5


def momentum_subgrad(h_start, n, target, time_limit, max_steps, alpha, mu, base_lr):

    best_c5 = compute_c5(h)

            break
        grad = _smooth_grad(h, alpha)
        grad -= np.mean(grad)

        lr = base_lr / (1.0 + 0.005 * step)

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        for noise_scale in [0.02, 0.05, 0.08]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_cd, c5_cd = multi_scale_pair_cd(h_pert, n_cur, target_cur,
                                              time_limit=min(30, tleft() * 0.3),
                                              max_sweeps=5000, n_cand=30, top_k=5)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Final SLSQP polish
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        if n_cur <= 800:
            h_final = slsqp_analytic(best_h, n_cur, target_cur,
                                     [2000, 10000, 50000],
                                     600, 1e-16, min(25, tleft() * 0.5))
        else:
            h_final = best_h
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 9: Final pair CD polish
    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd, c5_cd = multi_scale_pair_cd(best_h, n_cur, target_cur,
                                          time_limit=min(20, tleft() * 0.5),
                                          max_sweeps=5000, n_cand=25, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
