**Level-set method:**
1. Start with t = current C₅
2. At each iteration:
   a. Compute the "violation" v_k = max(0, O(k) - t) for each shift k
   b. Compute the gradient of sum_k v_k²
   c. Take a step in the negative gradient direction
   d. If any v_k > 0, decrease the step size
   e. If all v_k ≤ 0, decrease t (try to lower the target)

This is essentially a penalty method for the constraint O(k) ≤ t.

        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = np.zeros(2 * n - 1)
                for p in range(n):
                    idx1 = p + n - 1
                    offset = idx1 - i
                    mask_pos = (offset >= 0) & (offset < 2 * n - 1)
                    if mask_pos:
                        dcorr[offset] += 1.0
                    offset2 = idx1 - j
                    mask_neg = (offset2 >= 0) & (offset2 < 2 * n - 1)
                    if mask_neg:
                        dcorr[offset2] -= 1.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                a_d = 0.0
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial - 1e-15:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def project_feasible(vec, target_sum):
    vec = np.clip(vec.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-13:
        return vec
    if diff > 0:
        idx = np.where(vec < 1.0 - 1e-12)[0]
    else:
        idx = np.where(vec > 1e-12)[0]
    if len(idx) == 0:
        return vec
    vec = vec.copy()
    vec[idx] += diff / len(idx)
    vec = np.clip(vec, 0.0, 1.0)
    residual = target_sum - np.sum(vec)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(vec < 1.0)[0]
        else:
            idx2 = np.where(vec > 0.0)[0]
        if len(idx2) > 0:
            vec[idx2] += residual / len(idx2)
            vec = np.clip(vec, 0.0, 1.0)
    return vec

def gradient_ascent_h(h_start, n, target, time_limit, max_steps, n_top, lr, momentum):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    vel = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 3000:
            break
        corr = compute_corr(h)
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        vel = momentum * vel - lr * grad
        h_new = project_feasible_h(h + vel, target)
        h = h_new
    return best_h, best_c5

def project_feasible_h(vec, target_sum):
    vec = np.clip(vec.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(vec)
    if abs(diff) < 1e-13:
        return vec
    if diff > 0:
        idx = np.where(vec < 1.0 - 1e-12)[0]
    else:
        idx = np.where(vec > 1e-12)[0]
    if len(idx) == 0:
        return vec
    vec = vec.copy()
    vec[idx] += diff / len(idx)
    vec = np.clip(vec, 0.0, 1.0)

    if abs(residual) > 1e-10:

            idx2 = np.where(vec < 1.0)[0]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible_h(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible_h(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible_h(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 180:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [50, 200, 1000, 5000, 20000, 100000],
                                   1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_sb, sc = soft_bottleneck_optimize(best_h, n, target,
                                            time_limit=min(100, tleft() * 0.25),
                                            max_steps=50000,
                                            alpha_init=50.0, alpha_end=50000.0,
                                            base_lr=0.03, beta1=0.9, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_sb

    if tleft() > 150:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [5000, 20000, 100000, 500000, 2000000],
                                   2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, sc = pair_cd_fast(best_h, n, target,
                                time_limit=min(120, tleft() * 0.3),
                                max_sweeps=2000, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 100:
        h_ga, sc = gradient_ascent_h(best_h, n, target,
                                     time_limit=min(60, tleft() * 0.2),
                                     max_steps=50000, n_top=8, lr=0.02, momentum=0.95)
        if sc < best_score:
            best_score = sc
            best_h = h_ga

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n, target,
                                   [20000, 100000, 500000, 2000000],
                                   1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    for noise_scale in [0.005, 0.01, 0.02, 0.04]:
        if tleft() < 120:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible_h(h_pert, target)
        h_ref, sc = slsqp_optimize(h_pert, n, target,
                                   [200, 1000, 5000, 20000, 100000, 500000],
                                   1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 120:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = slsqp_optimize(h_interp, n_try, target_try,
                                     [100, 500, 2000, 10000, 50000, 200000],
                                     1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try

    if tleft() > 100:
        h_sb, sc = soft_bottleneck_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                            time_limit=min(60, tleft() * 0.3),
                                            max_steps=30000,
                                            alpha_init=100.0, alpha_end=100000.0,
                                            base_lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8)
        if sc < best_score:
            best_score = sc
            best_h = h_sb

    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [50000, 200000, 1000000, 5000000],
                                     1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 30:
        h_final, sc = pair_cd_fast(best_h, len(best_h), len(best_h) / 2.0,
                                   time_limit=min(20, tleft() * 0.5),
                                   max_sweeps=1000, n_cand=15)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible_h(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
