The inner problem is: min_h ∑_k π_k corr_k(h) + λ(∑h - target) + μ^T (h - 1) + ν^T h
= min_h [quadratic in h] + [linear in h]

This is a QP! And for a fixed π, it's convex if the Hessian is PSD.

The Hessian of ∑_k π_k corr_k(h) is: -∑_k π_k (T_k + T_k^T) where T_k is the shift matrix by k.

This matrix is a weighted sum of "anti-diagonal" matrices. Whether it's PSD depends on the weights π.

If we can find a π such that the Hessian is PSD, then the inner problem is a convex QP, and we can solve it efficiently. Then we just need to optimize over π (which is a simplex, so it's a convex optimization problem in the dual).

This is a promising direction! Let me think about it more carefully.

corr_k(h) = ∑_i h[i](1-h[i+k]) = ∑_i h[i] - ∑_i h[i]h[i+k]

The Hessian of ∑_i h[i]h[i+k] is: for the (j,l) entry, it's 1 if l = j+k or l = j-k (i.e., if |l-j| = k and both indices are valid).

So the Hessian of -∑_i h[i]h[i+k] is: -1 at (j, j+k) and (j+k, j) positions.

The total Hessian of ∑_k π_k corr_k is: -∑_k π_k * (off-diagonal at distance k)

This is a symmetric matrix with 0 on the diagonal and negative off-diagonal entries (weighted by π_k). A matrix with 0 diagonal and negative off-diagonals is... not necessarily PSD. In fact, it's likely indefinite.

For example, with just π_1 = 1, the Hessian is:
H = -[0, -1, 0, ..., 0; -1, 0, -1, ..., 0; ...]
This is the negative of the adjacency matrix of a path graph, which has both positive and negative eigenvalues. So it's indefinite.

This means the dual approach doesn't directly give us a convex subproblem. However, we can still use it as a heuristic:
- Pick a π (e.g., uniform over active set)
- Solve the inner QP (even if non-convex, it might be solved to a local minimum)
- Update π based on which constraints are violated
- Iterate

This is essentially a "bundle method" or "cutting plane" approach for the non-smooth problem.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(lse_objective, h_cur, args=(alpha,), jac=lse_gradient,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)

def power_mean_objective(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr ** p) ** (1.0 / p)) * dx


def power_mean_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    pm = np.sum(corr ** p) ** (1.0 / p)
    weights = (corr ** (p - 1)) / np.sum(corr ** (p - 1))
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx * (pm / pm)
    return grad

def simulated_annealing(h_start, n, target, time_limit, max_steps, t_start, t_end):
    start_t = time.time()
    rng = np.random.RandomState(12345)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        progress = step / max_steps
        T = t_start * (t_end / t_start) ** progress

        n_pert = rng.choice([1, 2, 3, 4, 5, 8])
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = rng.choice([0.02, 0.05, 0.1, 0.2])
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)

        delta = c5_try - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                     1500, 1e-15, min(50, tleft()))
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Final: pair CD at best resolution
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = active_set_pair_cd(best_h, n_use, target_use,
                                         time_limit=min(25, tleft() * 0.5),
                                         max_sweeps=20000,
                                         n_cand=80, top_k=10,
                                         soft_alpha=60.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
