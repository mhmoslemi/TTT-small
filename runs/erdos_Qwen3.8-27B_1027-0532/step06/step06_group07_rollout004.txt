    n = len(h)

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def water_fill_project(z, target_sum):
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_obj_and_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    max_c = np.max(corr)
    exps = np.exp(alpha * (corr - max_c))
    exp_sum = exps.sum()
    w = exps / exp_sum
    obj = dx * (max_c + np.log(exp_sum) / alpha)
    mid = n - 1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k_idx in range(n_corr):
        wk = w[k_idx] * dx
        if wk < 1e-16:
            continue
        k = k_idx - mid
        if k >= 0:
            grad[:n - k] += wk * one_minus_h[k:]
            grad[k:] -= wk * h[:n - k]
        else:
            grad[-k:] += wk * one_minus_h[:n + k]
            grad[:n + k] -= wk * h[-k:]
    return obj, grad, corr


def projected_gradient_phase(h_init, n, target, time_limit, max_total_iters=8000):

    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        obj, grad, corr = smooth_obj_and_grad(h, alpha, dx)
        for it in range(iters_per_alpha):
            if time.time() - start > time_limit:

            grad_dir = grad - np.mean(grad)
            gnorm_sq = float(np.dot(grad_dir, grad_dir))
            if gnorm_sq < 1e-26:

            step = min(2.0 * dx / (np.sqrt(gnorm_sq) + 1e-12), 5.0 / (alpha * dx + 1.0))
            step = max(step, 1e-5)

def subgradient_polyak(h_init, n, target, time_limit, max_iters=20000):

    c5_final = compute_c5(h)

    for noise_scale in [0.02, 0.05, 0.1]:
        if time.time() - start > time_limit:
            break
        h_pert = h_best + rng.randn(n) * noise_scale
        h_pert = water_fill_project(h_pert, target)
        cd_t = min(80, (time_limit - (time.time() - start)) * 0.4)
        if cd_t < 10:
            break
        h_try, c5_try = pair_cd_enhanced(h_pert, n, target, time_limit=cd_t,
                                          n_cand=30, n_bisect=12, max_sweeps=1500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5


def slsqp_polish(h_init, n, target, time_limit, alphas, maxiter):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            def obj_fn(x):
                return smooth_obj_and_grad(x, alpha, dx)[0]

            def obj_grad_fn(x):
                _, g, _ = smooth_obj_and_grad(x, alpha, dx)
                return g

            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                            'jac': lambda x: np.ones(n)}]
            bounds = [(0.0, 1.0)] * n
            res = minimize(obj_fn, h, jac=obj_grad_fn, method='SLSQP',
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = water_fill_project(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5
