import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

# Polyak step subgradient method
def polyak_subgrad(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, mu_init=1.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_lower = 0.1  # some lower bound
    idxs = np.arange(n)
    mid = n - 1
    step_count = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step % 20 == 0 and time.time() - start > time_limit:
            break
            
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
        
        if c5_now < best_lower:
            best_lower = c5_now
            
        # Get top n_shifts active constraints
        top_idx = np.argsort(corr)[-n_shifts:]
        weights = np.exp(50.0 * (corr[top_idx] - cur_max))
        weights = weights / weights.sum()
        
        # Compute subgradient (average of active constraint gradients)
        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        
        # Polyak step size
        gap = cur_max - best_lower  # in "corr" units
        gnorm2 = float(np.dot(grad, grad))
        if gnorm2 < 1e-20:
            break
        step_size = gap / gnorm2
        step_size = min(step_size, 2.0)  # cap step size
        
        # Try the step
        h_try = h - step_size * grad
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < c5_now - 1e-15:
            h = h_try
            step_count += 1
        else:
            # Try smaller step
            h_try = h - 0.5 * step_size * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                step_count += 1
            else:
                # Try even smaller
                h_try = h - 0.1 * step_size * grad
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_now - 1e-15:
                    h = h_try
                    step_count += 1
                else:
                    if step_count > 1000:
                        break
    
    return best_h, best_c5

# Improved pair CD with multi-shift gradients and block moves
def improved_pair_cd(h_init, n, target, time_limit, max_sweeps=50000, n_cand=30, n_shifts=3):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Multi-shift gradient
        top_idx = np.argsort(corr)[-n_shifts:]
        weights = np.exp(30.0 * (corr[top_idx] - cur_max))
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        # Candidate selection
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles for candidates
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        
        # Find best pair
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Sample pairs to save time (don't check all)
        n_high = len(high_set)
        n_low = len(low_set)
        
        if n_high * n_low > 400:
            # Subsample
            perm_h = np.random.permutation(n_high)
            perm_l = np.random.permutation(n_low)
            high_sub = high_set[perm_h[:20]]
            low_sub = low_set[perm_l[:20]]
        else:
            high_sub = high_set
            low_sub = low_set
        
        for i in high_sub:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_sub:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

# Block move CD
def block_cd(h_init, n, target, time_limit, max_sweeps=10000, block_size=3):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
        
        k_star = int(np.argmax(corr)) - (n - 1)
        
        # Compute gradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Find best block of high-grad positions and block of low-grad
        # Use rolling sum of gradient to find best blocks
        window = block_size
        block_grads = np.convolve(grad, np.ones(window) / window, mode='valid')
        
        best_src_start = int(np.argmax(block_grads))
        best_dst_start = int(np.argmin(block_grads))
        
        src_block = range(best_src_start, best_src_start + window)
        dst_block = range(best_dst_start, best_dst_start + window)
        
        # Check if blocks overlap
        overlap = set(src_block) & set(dst_block)
        if overlap:
            continue
        
        # Compute total delta available
        total_src = sum(float(h[i]) for i in src_block)
        total_dst_cap = sum(1.0 - float(h[i]) for i in dst_block)
        delta_max = min(total_src, total_dst_cap)
        
        if delta_max < 1e-10:
            continue
        
        # Try to move delta_max / n_block from each src to each dst
        # Actually, let's try uniform redistribution
        per_delta = delta_max / window
        
        # Check if individually feasible
        feasible = True
        for i in src_block:
            if float(h[i]) < per_delta - 1e-12:
                feasible = False
                break
        if feasible:
            for j in dst_block:
                if float(h[j]) + per_delta > 1.0 + 1e-12:
                    feasible = False
                    break
        
        if not feasible:
            # Scale down
            per_delta = min([float(h[i]) for i in src_block])
            per_delta = min(per_delta, min([1.0 - float(h[j]) for j in dst_block]))
            if per_delta < 1e-10:
                continue
        
        h_try = h.copy()
        for i in src_block:
            h_try[i] -= per_delta
        for j in dst_block:
            h_try[j] += per_delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        
        c5_try = compute_c5(h_try)
        if c5_try < c5_val - 1e-13:
            h = h_try
    
    return best_h, best_c5

# SLSQP with adaptive alpha
def slsqp_adaptive(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        
        def obj(x):
            return smooth_obj(x, alpha, target)
        
        def con_fun(x):
            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(len(x))
        
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    
    return best_h, best_c5

# Discrete-to-continuous refinement
def refine_resolution(h_low, n_high, target_high, time_limit=60):
    x_low = np.linspace(0.0, 1.0, len(h_low))
    x_high = np.linspace(0.0, 1.0, n_high)
    h_high = np.interp(x_high, x_low, h_low)
    h_high = project_feasible(h_high, target_high)
    return h_high

            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            prof_i = profiles[ci]
            for j in low_sub:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = prof_i - profiles[cj]
                
                # Binary search
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, max_sweeps=5000, block_size=3):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break
        
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Find blocks using rolling average
        window = block_size
        if window >= n:
            break
        block_grads = np.convolve(grad, np.ones(window) / window, mode='valid')
        
        best_src = int(np.argmax(block_grads))
        best_dst = int(np.argmin(block_grads))
        
        if best_src + window > best_dst or best_dst + window > best_src:
            # Overlapping, try next best
            sorted_src = np.argsort(-block_grads)
            sorted_dst = np.argsort(block_grads)
            found = False
            for si in sorted_src[:5]:
                for di in sorted_dst[:5]:
                    if si + window <= di or di + window <= si:
                        best_src = int(si)
                        best_dst = int(di)
                        found = True
                        break
                if found:
                    break
            if not found:
                continue
        
        src_range = range(best_src, best_src + window)
        dst_range = range(best_dst, best_dst + window)
        
        min_src = min(float(h[i]) for i in src_range)
        min_dst_cap = min(1.0 - float(h[j]) for j in dst_range)
        per_delta = min(min_src, min_dst_cap)
        
        if per_delta < 1e-10:
            continue
        
        h_try = h.copy()
        for i in src_range:
            h_try[i] -= per_delta
        for j in dst_range:
            h_try[j] += per_delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        
        c5_try = compute_c5(h_try)
        if c5_try < c5_val - 1e-13:
            h = h_try
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
