def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(-y)
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can = y[idx]
            if can > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can = 1.0 - y[idx]
            if can > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current > target_sum:
        need = current - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1

def slsqp_smooth(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        exp_s = np.exp(shifted)
        val = c_max + float(np.log(exp_s.sum())) / alpha
        # Gradient
        w = exp_s / exp_s.sum() / alpha
        shifts = np.arange(2 * n - 1) - (n - 1)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = shifts[k]
            if s >= 0:
                if s < n:
                    grad[:n-s] += wk * (1.0 - x[s:])
                    grad[s:] += wk * (-x[:n-s])
            else:
                as_ = -s
                if as_ < n:
                    grad[as_:] += wk * (1.0 - x[:n-as_])
                    grad[:n-as_] += wk * (-x[as_:])
        return val, grad
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: obj_and_grad(x, alpha)[0], h_cur,
                           jac=lambda x: obj_and_grad(x, alpha)[1],
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-14:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def _slsqp_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(corr.max())
    shifted = np.clip(_slsqp_alpha * (corr - c_max), -400.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / (exp_s.sum() * _slsqp_alpha)
    n = _slsqp_n
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = shifts[k]
        if s >= 0:
            if s < n:
                grad[:n-s] += wk * (1.0 - x[s:])
                grad[s:] += wk * (-x[:n-s])
        else:
            as_ = -s
            if as_ < n:
                grad[as_:] += wk * (1.0 - x[:n-as_])
                grad[:n-as_] += wk * (-x[as_:])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(-y)
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can = y[idx]
            if can > remaining:
                y[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= can
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can = 1.0 - y[idx]
            if can > remaining:
                y[idx] += remaining
                remaining = 0.0
            else:
                remaining -= can
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current > target_sum:
        need = current - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def multi_sa(h, n_iters, t_init, t_final, max_k, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best = float(corr.max()) * dx
    best_h = h.copy()
    curr = best
    stall = 0
    stall_limit = 30000
    reheat_count = 0
    max_reheats = 6
    cur_t_init = t_init

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > stall_limit:
            if reheat_count < max_reheats:
                cur_t_init = max(cur_t_init * 0.7, t_final * 10)
                stall = 0
                reheat_count += 1
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                n_pert = rng.randint(3, 12)
                for _p in range(n_pert):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    h[ii], h[jj] = 0.0, 1.0
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                corr = np.correlate(h, 1.0 - h, mode='full')
                curr = float(corr.max()) * dx
            else:
                break

        ratio = min(it / max(n_iters - 1, 1), 1.0)
        t = cur_t_init * (t_final / cur_t_init) ** ratio
        t = max(t, t_final)

        r = rng.random()
        if r < 0.30:
            k = 1
        elif r < 0.55:
            k = 2
        elif r < 0.72:
            k = 3
        elif r < 0.87:
            k = 4
        else:
            k = rng.randint(1, max_k + 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < k or len(zeros) < k:
            k = min(len(ones), len(zeros), 1)
            if k == 0:
                break

        if k == 1:
            sel_i = ones[rng.randint(len(ones))]
            sel_j = zeros[rng.randint(len(zeros))]
            h[sel_i] = 0.0
            h[sel_j] = 1.0
        else:
            sel_i = rng.choice(ones, size=k, replace=False)
            sel_j = rng.choice(zeros, size=k, replace=False)
            h[sel_i] = 0.0
            h[sel_j] = 1.0

        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(corr.max()) * dx
        d = new_c5 - curr

        if d < 0 or (t > 1e-15 and rng.random() < np.exp(-d / t)):
            curr = new_c5
            if new_c5 < best:
                best = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if k == 1:
                h[sel_i] = 1.0
                h[sel_j] = 0.0
            else:
                h[sel_i] = 1.0
                h[sel_j] = 0.0
            stall += 1

    return best_h, best

def grad_momentum(h, n_iters, alpha, lr_init, lr_end, mom, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    v = np.zeros(n)
    best = compute_c5(h)
    best_h = h.copy()
    stall = 0
    max_stall = 6000
    shifts = np.arange(2 * n - 1) - (n - 1)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = float(corr.max())
        shifted = alpha * (corr - max_c)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        grad = np.zeros(n)
        for kk in range(2 * n - 1):
            wk = w[kk]
            if wk < 1e-12:
                continue
            s = shifts[kk]
            if s >= 0:
                if s < n:
                    grad[:n-s] += wk * (1.0 - h[s:])
                    grad[s:] += wk * (-h[:n-s])
            else:
                as_ = -s
                if as_ < n:
                    grad[as_:] += wk * (1.0 - h[:n-as_])
                    grad[:n-as_] += wk * (-h[as_:])

        v = mom * v + lr * grad
        h_new = h - v
        h = project_box_sum(h_new, target)

        curr = compute_c5(h)
        if curr < best - 1e-14:
            best = curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    shifts = np.arange(2 * n - 1) - (n - 1)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj_fn(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = float(corr.max())
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

            def obj_grad(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = float(corr.max())
                shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
                exp_s = np.exp(shifted)
                w = exp_s / (exp_s.sum() * alpha)
                g = np.zeros(n)
                for kk in range(2 * n - 1):
                    wk = w[kk]
                    if wk < 1e-12:
                        continue
                    s = shifts[kk]
                    if s >= 0:
                        if s < n:
                            g[:n-s] += wk * (1.0 - x[s:])
                            g[s:] += wk * (-x[:n-s])
                    else:
                        as_ = -s
                        if as_ < n:
                            g[as_:] += wk * (1.0 - x[:n-as_])
                            g[:n-as_] += wk * (-x[as_:])
                return c_max + float(np.log(exp_s.sum())) / alpha, g

            def cons_fun(x):
                return np.sum(x) - target

            def cons_jac(x):
                return np.ones(n)

            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(obj_grad, h_cur, jac=None,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-14:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)
