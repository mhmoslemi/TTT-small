- h[j+k] is large (ideally 1): position j is "followed by 1" at shift k
- h[i-k] is large (ideally 1): position i is "preceded by 1" at shift k  
- h[j-k] is small (ideally 0): position j is "preceded by 0" at shift k

So a good candidate i for shift k is one where h[i]=1, h[i+k]=0, h[i-k]=1 (if in range).
A good candidate j for shift k is one where h[j]=0, h[j+k]=1, h[j-k]=0 (if in range).

    for k in range(-n+1, n):

        # n values that matter: i, i-k, j, j-k
        delta = 0.0

        if 0 <= i + k < n:
            # term: h[i+k]*(1-h[i]), old: h[i+k]*0=0, new: h[i+k]_new*1
            # h[i+k]_new = 1 if i+k==j else h[i+k]
            if i + k == j:
                delta += 1.0  # new term = 1*1 = 1, old = 0
            else:
                delta += h[i+k]
        # n = i-k (if valid: 0<=i-k<n, always 0<=(i-k)+k=i<n)
        if 0 <= i - k < n:

            # change: -(1-h[i-k])

                delta = 0  # reset and redo
                # Actually let me just handle all 4 terms together
            else:

            delta += (1.0 - h[j-k])

        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0

            final_sum -= 1
        else:

    h = h_start.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    max_iter = 100000
    stall = 0
    max_stall = 50
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Find top bad shifts
        max_val = float(corr.max())
        mean_val = float(corr.mean())
        
        # Get shifts above mean (or top k)
        threshold = mean_val + 0.5 * (max_val - mean_val)
        bad_mask = corr >= threshold
        bad_indices = np.where(bad_mask)[0]
        if len(bad_indices) == 0:
            bad_indices = np.argsort(corr)[-top_k:]
        elif len(bad_indices) > top_k:
            bad_indices = np.argsort(corr)[-top_k:]
        
        # Generate candidate swaps
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for k_idx in bad_indices:
            k = k_idx - (n - 1)
            
            # Candidate i's: h[i]=1, and the swap would help at shift k
            # h[i+k]=0 would remove a positive contribution
            # h[i-k]=1 would remove a positive contribution (since 1-h[i-k]=0)
            i_cands = []
            for i in ones:
                score = 0.0
                if 0 <= i + k < n and h[i+k] == 0:
                    score += 1.0
                if 0 <= i - k < n and h[i-k] == 1:
                    score += 1.0
                if score > 0:
                    i_cands.append((score, i))
            i_cands.sort(reverse=True)
            i_cands = [x[1] for x in i_cands[:scan_width]]
            
            # Candidate j's: h[j]=0, and the swap wouldn't hurt at shift k
            # h[j+k]=1 would add a positive contribution (bad)
            # h[j-k]=0 would not add (good, since 1-h[j-k]=1)
            j_cands = []
            for j in zeros:
                score = 0.0
                if 0 <= j + k < n and h[j+k] == 1:
                    score -= 1.0  # bad: would add
                if 0 <= j - k < n and h[j-k] == 0:
                    score += 1.0  # good: 1-h[j-k]=1, adds
                # We want j's that don't increase c[k] much
                # Actually, for j: the change is +(1-h[j-k]) - h[j+k]
                # We want this to be small (or negative)
                penalty = 0.0
                if 0 <= j + k < n:
                    penalty += h[j+k]
                if 0 <= j - k < n:
                    penalty -= (1.0 - h[j-k])
                # We want penalty < 0 for this shift
                j_cands.append((penalty, j))
            j_cands.sort()
            j_cands = [x[1] for x in j_cands[:scan_width]]
            
            # Try all combinations
            for i in i_cands:
                for j in j_cands:
                    if i == j:
                        continue
                    # Quick estimate: compute Δc for the worst shift
                    # Full evaluation:
                    h[i] = 0
                    h[j] = 1
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(new_corr.max())
                    h[i] = 1
                    h[j] = 0
                    
                    improve = max_val - new_max
                    if improve > best_improve:
                        best_improve = improve
                        best_i = i
                        best_j = j
                        best_new_corr = new_corr.copy()
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = best_new_corr
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

                s = 0.0
                if 0 <= i + k < n:
                    s += h[i + k]  # removing a 1 that was contributing via h[i+k]*(1-h[i])
                    # Actually: the term at n=i is h[i+k]*(1-h[i]). Since h[i]=1, this is 0.
                    # The term at n=i-k is h[i]*(1-h[i-k]). Since h[i]=1, this is 1-h[i-k].
                    # Removing i (setting h[i]=0):
                    # - Term at n=i: was 0, becomes h[i+k]*1 = h[i+k]. Change: +h[i+k]
                    # - Term at n=i-k: was 1-h[i-k], becomes 0. Change: -(1-h[i-k])
                    # Net: h[i+k] - (1-h[i-k]) = h[i+k] + h[i-k] - 1

                if 0 <= i + k < n:

                # We want s < 0 to reduce c[k]
                i_scores.append((s, i))

            i_cands = [x[1] for x in i_scores[:20]]

            j_scores = []

                # Adding j (setting h[j]=1):
                # - Term at n=j: was h[j+k]*1, becomes 0. Change: -h[j+k]
                # - Term at n=j-k: was 0, becomes 1*(1-h[j-k]). Change: +(1-h[j-k])
                # Net: (1-h[j-k]) - h[j+k]
                if 0 <= j + k < n:
                    s -= h[j + k]
                if 0 <= j - k < n:
                    s += (1.0 - h[j - k])
                # We want s < 0 to reduce c[k] (but this is the "adding" part)
                # Total change from swap (i,j) at shift k = i_contribution + j_contribution
                j_scores.append((s, j))
            
            j_scores.sort()
            j_cands = [x[1] for x in j_scores[:20]]
            
            # Try combinations
            for i_val in i_cands:
                for j_val in j_cands:
                    if i_val == j_val:
                        continue
                    h[i_val] = 0
                    h[j_val] = 1
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(new_corr.max())
                    h[i_val] = 1
                    h[j_val] = 0

                        best_i = i_val
                        best_j = j_val
                        best_new_corr = new_corr.copy()

            h[best_j] = 1
            corr = best_new_corr
            sc = float(corr.max()) * dx

                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            # Try a random move to escape

                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                sc = float(corr.max()) * dx
                # Accept with some probability (simple SA)
                if sc < best_c5 * 1.01 or rng.random() < 0.1:
                    pass  # Keep the random move
    
    return best_h, best_c5

        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

            if j_min >= j_max:
                continue
            g[j_min:j_max] -= wk * h[j_min-k_offset:j_max-k_offset]
        
        g *= dx
        g += 2.0 * penalty * (float(np.sum(h)) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    
    # Run L-BFGS-B
    try:

    n = len(h_start)
    h = h_start.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 80

    while time.time() - start_time < time_limit and stall < max_stall:
        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        if len(ones_arr) < 2 or len(zeros_arr) < 2:
            break
        max_val = float(corr.max())

        best_improve = 1e-12
        best_ij = None
        best_corr = None

        for _ in range(n_try):
            i_idx = rng.choice(len(ones_arr), 2, replace=False)
            j_idx = rng.choice(len(zeros_arr), 2, replace=False)
            i1, i2 = int(ones_arr[i_idx[0]]), int(ones_arr[i_idx[1]])
            j1, j2 = int(zeros_arr[j_idx[0]]), int(zeros_arr[j_idx[1]])

            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            nc = np.correlate(h, 1.0 - h, mode='full')
            nm = float(nc.max())
            h[i1] = 1.0; h[i2] = 1.0
            h[j1] = 0.0; h[j2] = 0.0

            imp = max_val - nm
            if imp > best_improve:
                best_improve = imp
                best_ij = (i1, i2, j1, j2)
                best_corr = nc

        if best_ij is not None:
            i1, i2, j1, j2 = best_ij
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            corr = best_corr
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def sa_polish(h_start, time_limit, start_time, rng, t_init=0.005, t_final=0.00001):
    n = len(h_start)
    h = h_start.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 3000

    max_iters = int((time_limit - 5) * 3000)
    max_iters = min(max_iters, 500000)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones_arr = np.where(h == 1)[0]
        zeros_arr = np.where(h == 0)[0]
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break

        r = rng.random()
        h_new = h.copy()

        if r < 0.5:
            ip = int(ones_arr[rng.randint(len(ones_arr))])
            jp = int(zeros_arr[rng.randint(len(zeros_arr))])
            h_new[ip] = 0.0
            h_new[jp] = 1.0
        elif r < 0.8:
            worst = np.argsort(corr)[-10:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [x for x in ones_arr if 0 <= x + k < n]
            jp = [x for x in zeros_arr if 0 <= x + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[int(ip[rng.randint(len(ip))])] = 0.0
                h_new[int(jp[rng.randint(len(jp))])] = 1.0
            else:
                ip2 = int(ones_arr[rng.randint(len(ones_arr))])
                jp2 = int(zeros_arr[rng.randint(len(zeros_arr))])
                h_new[ip2] = 0.0
                h_new[jp2] = 1.0
        else:
            if len(ones_arr) >= 2 and len(zeros_arr) >= 2:
                ii = rng.choice(len(ones_arr), 2, replace=False)
                jj = rng.choice(len(zeros_arr), 2, replace=False)
                h_new[int(ones_arr[ii[0]])] = 0.0
                h_new[int(ones_arr[ii[1]])] = 0.0
                h_new[int(zeros_arr[jj[0]])] = 1.0
                h_new[int(zeros_arr[jj[1]])] = 1.0
            else:
                continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def continuous_refine(h_start, target, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h_cur = h_start.copy()

    def obj_fn(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        c_max = float(corr.max())
        beta_local = 300.0
        shifted = beta_local * (corr - c_max)
        shifted -= shifted.max()
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta_local
        d = float(np.sum(hv)) - target
        val += 800.0 * d * d
        return val

    def grad_fn(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        c_max = float(corr.max())
        beta_local = 300.0
        shifted = beta_local * (corr - c_max)
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()

        g = np.zeros(n)
        for k_off in range(-n + 1, n):
            wk = w[k_off + n - 1]
            if wk < 1e-12:
                continue
            i_min = max(0, -k_off)
            i_max = min(n, n - k_off)
            if i_min < i_max:
                g[i_min:i_max] += wk * (1.0 - hv[i_min+k_off:i_max+k_off])
            j_min = max(0, k_off)
            j_max = min(n, n + k_off)
            if j_min < j_max:
                g[j_min:j_max] -= wk * hv[j_min-k_off:j_max-k_off]
        g *= dx
        g += 2.0 * 800.0 * (float(np.sum(hv)) - target)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h_cur, jac=grad_fn, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1})
        h_cur = project_box_sum(res.x, target)
    except Exception:
        pass

    return h_cur

def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    h_bin = project_binary(h_init, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Focused 1-move local search
    if tleft() > 100:
        time_alloc = min(250, tleft() * 0.3)
        h_ls, c5_ls = local_search_1move(best_h, time_limit=time_alloc,
                                          start_time=start_time, rng=rng,
                                          top_k=5, cand_width=15)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 2: 2-move local search
    if tleft() > 80:
        time_alloc = min(150, tleft() * 0.2)
        h_2m, c5_2m = local_search_2move(best_h, time_limit=time_alloc,
                                          start_time=start_time, rng=rng, n_try=300)
        if c5_2m < best_c5:
            best_c5 = c5_2m
            best_h = h_2m.copy()

    # Phase 3: SA polish
    if tleft() > 80:
        time_alloc = min(150, tleft() * 0.2)
        h_sa, c5_sa = sa_polish(best_h, time_limit=time_alloc,
                                 start_time=start_time, rng=rng,
                                 t_init=0.004, t_final=0.00001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Second round of local search (alternate 1-move and 2-move)
    if tleft() > 60:
        time_alloc = min(100, tleft() * 0.2)
        h_ls2, c5_ls2 = local_search_1move(best_h, time_limit=time_alloc * 0.6,
                                            start_time=start_time, rng=rng,
                                            top_k=7, cand_width=12)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
        if tleft() > 40:
            time_alloc2 = min(80, tleft() * 0.2)
            h_2m2, c5_2m2 = local_search_2move(best_h, time_limit=time_alloc2,
                                                start_time=start_time, rng=rng, n_try=400)
            if c5_2m2 < best_c5:
                best_c5 = c5_2m2
                best_h = h_2m2.copy()

    # Phase 5: Continuous refinement
    if tleft() > 50:
        h_cont = project_box_sum(best_h.copy(), target)
        time_alloc = min(80, tleft() * 0.2)
        h_ref = continuous_refine(h_cont, target, time_limit=time_alloc,
                                   start_time=start_time)
        sc = compute_c5(h_ref)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref.copy()
        # Also try binary projection of the refined solution
        h_bin_ref = project_binary(h_ref, target_int)
        sc_bin = compute_c5(h_bin_ref)
        if sc_bin < best_c5:
            best_c5 = sc_bin
            best_h = h_bin_ref.copy()

    # Phase 6: Multiple restarts with perturbation
    for restart in range(4):
        if tleft() < 80:
            break
        # Perturb: flip 10-30 random pairs
        n_flip = rng.randint(10, 31)
        h_pert = best_h.copy()
        ones_r = np.where(h_pert == 1)[0]
        zeros_r = np.where(h_pert == 0)[0]
        for _ in range(n_flip):
            if len(ones_r) == 0 or len(zeros_r) == 0:
                break
            ip = int(ones_r[rng.randint(len(ones_r))])
            jp = int(zeros_r[rng.randint(len(zeros_r))])
            h_pert[ip] = 0.0
            h_pert[jp] = 1.0
            ones_r = np.where(h_pert == 1)[0]
            zeros_r = np.where(h_pert == 0)[0]

        time_alloc = min(120, tleft() * 0.25)
        h_rs, c5_rs = local_search_1move(h_pert, time_limit=time_alloc * 0.5,
                                          start_time=start_time, rng=rng,
                                          top_k=5, cand_width=15)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
        if tleft() > 40:
            time_alloc2 = min(60, tleft() * 0.15)
            h_rs2, c5_rs2 = local_search_2move(h_rs, time_limit=time_alloc2,
                                                start_time=start_time, rng=rng, n_try=250)
            if c5_rs2 < best_c5:
                best_c5 = c5_rs2
                best_h = h_rs2.copy()

    # Phase 7: Try different grid sizes
    n_base = len(best_h)
    for n_try in [n_base + 2, n_base - 2, n_base + 4, n_base + 6]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = float(n_try / 2)
        target_try_int = int(n_try / 2)
        h_interp = interpolate_to_n(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try_int)
        sc_try = compute_c5(h_bin_try)
        if tleft() > 40:
            time_alloc = min(50, tleft() * 0.15)
            h_ls_try, c5_try = local_search_1move(h_bin_try, time_limit=time_alloc,
                                                   start_time=start_time, rng=rng,
                                                   top_k=5, cand_width=12)
            sc_try = c5_try
        if sc_try < best_c5:
            best_c5 = sc_try
            best_h = h_bin_try if c5_try == sc_try else h_ls_try
            # Actually we need to track which one gave the best
            pass

    # Phase 8: Final SA polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        time_alloc = min(80, tleft() - 10)
        h_sf, c5_sf = sa_polish(h_bin_final, time_limit=time_alloc,
                                 start_time=start_time, rng=rng,
                                 t_init=0.003, t_final=0.000005)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 9: Final 1-move polish
    if tleft() > 20:
        time_alloc = min(40, tleft() - 5)
        h_fp, c5_fp = local_search_1move(best_h, time_limit=time_alloc,
                                          start_time=start_time, rng=rng,
                                          top_k=8, cand_width=10)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(np.clip(best_h, 0, 1), target_final)
    # Ensure sum constraint
    best_h = project_binary(best_h, int(n_final / 2)) if n_final % 2 == 0 else best_h
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
