        best_j = -1

            delta_hi = min(h[i], 1.0 - h[j])

                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:

                else:
                    a_d = mid_d

                continue

            if approx_max < best_trial_c5:

                best_i = i
                best_j = j
                best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    
    return best_h, best_c5

        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(11):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_search(h_init, n, target, time_limit, max_steps=30000):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        scale = 0.04 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [100, 500, 2000, 10000],
                               600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        tl = _time_left(start_time, budget_s)
        if tl > 250:
            h_cd, c5_cd = pair_cd_fast(h1, n1, target1,
                                       time_limit=min(280, tl * 0.6),
                                       max_sweeps=60000,
                                       n_cand_start=25, n_cand_end=70,
                                       top_k_eval=50, max_no_improve=800,
                                       n_shifts_grad=2)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [500, 2000, 8000, 20000],
                               1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 200:
        h_cd2, c5_cd2 = pair_cd_fast(best_h, n0, target0,
                                     time_limit=min(220, tl * 0.7),
                                     max_sweeps=50000,
                                     n_cand_start=35, n_cand_end=80,
                                     top_k_eval=60, max_no_improve=1000,
                                     n_shifts_grad=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_ms, c5_ms = pair_cd_multishift(best_h, n0, target0,
                                         time_limit=min(90, tl * 0.7),
                                         max_sweeps=30000, n_cand=50,
                                         top_k_eval=50, n_shifts=3,
                                         max_no_improve=400)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [5000, 20000, 80000],
                               800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = _time_left(start_time, budget_s)
        if tl > 200:
            h_cd3, c5_cd3 = pair_cd_fast(h2, n2, target2,
                                         time_limit=min(180, tl * 0.7),
                                         max_sweeps=30000,
                                         n_cand_start=25, n_cand_end=60,
                                         top_k_eval=50, max_no_improve=500,
                                         n_shifts_grad=2)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                n0 = n2
                target0 = target2

            tl = _time_left(start_time, budget_s)
            if tl > 50:
                h_pol3 = slsqp_optimize(best_h, n0, target0,
                                        [2000, 10000],
                                        600, 1e-15)
                sc3 = compute_c5(h_pol3)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_pol3

    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_ss, c5_ss = stochastic_search(best_h, n0, target0,
                                        time_limit=min(40, tl * 0.6),
                                        max_steps=20000)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.06, 0.12]:
            tl = _time_left(start_time, budget_s)
            if tl < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_optimize(h_pert, n0, target0,
                                   [200, 1000, 5000, 20000],
                                   500, 1e-15)

            if sc < best_score:

                best_h = h_ref

    if tl > 60:
        h_cd_final, c5_final = pair_cd_fast(best_h, n0, target0,
                                            time_limit=min(50, tl * 0.7),
                                            max_sweeps=20000,
                                            n_cand_start=40, n_cand_end=90,
                                            top_k_eval=70, max_no_improve=300,
                                            n_shifts_grad=3)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = _time_left(start_time, budget_s)
    if tl > 30:
        h_final_pol = slsqp_optimize(best_h, n0, target0,
                                     [10000, 50000],
                                     500, 1e-15)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
