def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff > 0:
        margins = 1.0 - y
        total = margins.sum()
        if total > 1e-10:
            y = y + (diff / total) * margins
    else:
        total = y.sum()
        if total > 1e-10:
            y = y + (diff / total) * y
    return np.clip(y, 0.0, 1.0)

def compute_lse_gradient(corr, W, shifts, n):
    grad = np.zeros(n)
    for k in range(len(W)):
        wk = W[k]
        if wk < 1e-10:
            continue
        s = shifts[k]
        if s == 0:
            grad += wk * (1.0 - 2.0 * np.zeros(n))
            grad -= wk * (2.0 * np.array([h_global_placeholder] * n) if False else 0)
        if s >= 0:
            if s == 0:
                grad += wk * (1.0 - 2.0 * _h_ref)
            else:
                grad[s:] += wk * (1.0 - _h_ref[:n - s])
                grad[:n - s] -= wk * _h_ref[s:]
        else:
            sp = -s
            grad[:n - sp] += wk * (1.0 - _h_ref[sp:])
            grad[sp:] -= wk * _h_ref[:n - sp]
    return grad

def lse_gradient_descent(h_init, n_iters, alpha_start, alpha_end, lr, time_limit, start_time, threshold=1e-8):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 5000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio

        c_max = float(corr.max())
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        exp_vals = np.exp(shifted)
        W = exp_vals / exp_vals.sum()

        grad = np.zeros(n)
        active = W > 1e-8
        for k in np.where(active)[0]:
            wk = W[k]
            s = shifts[k]
            if s >= 0:
                if s == 0:
                    grad += wk * (1.0 - 2.0 * h)
                else:
                    grad[s:] += wk * (1.0 - h[:n - s])
                    grad[:n - s] -= wk * h[s:]
            else:
                sp = -s
                grad[:n - sp] += wk * (1.0 - h[sp:])
                grad[sp:] -= wk * h[:n - sp]

        # Line search
        step = lr
        for _ in range(10):
            h_trial = project_box_sum(h - step * grad, target)
            corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            curr_c5 = float(corr_trial.max()) * dx
            if curr_c5 < best_c5:
                h = h_trial
                corr = corr_trial
                best_c5 = curr_c5
                best_h = h.copy()
                stall = 0
                break
            step *= 0.5

        if not (best_c5 < float(corr.max()) * dx * 0 + best_c5):
            pass
        if stall > 0 and it > 0:
            h_trial2 = project_box_sum(h - step * 0.1 * grad, target)
            corr_t = np.correlate(h_trial2, 1.0 - h_trial2, mode='full')
            if float(corr_t.max()) * dx < best_c5:
                h = h_trial2
                corr = corr_t
                best_c5 = float(corr.max()) * dx
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if best_c5 == float(corr.max()) * dx:
                stall += 1

    return best_h, best_c5

def topk_gradient_descent(h_init, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 8000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                if s == 0:
                    grad += (1.0 - 2.0 * h)
                else:
                    grad[s:] += (1.0 - h[:n - s])
                    grad[:n - s] -= h[s:]
            else:
                sp = -s
                grad[:n - sp] += (1.0 - h[sp:])
                grad[sp:] -= h[:n - sp]
        grad /= topk

        h = project_box_sum(h - lr * grad, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-13:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def slsqp_polish(h_start, target, alpha, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = float(np.max(c))
        sh = np.clip(alpha * (c - cm), -400.0, 0.0)
        return cm + float(np.log(np.sum(np.exp(sh)))) / alpha

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = float(np.max(c))
        sh = np.clip(alpha * (c - cm), -400.0, 0.0)
        ev = np.exp(sh)
        W = ev / ev.sum()
        shifts = np.arange(2 * n - 1) - (n - 1)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = W[k]
            if wk < 1e-10:
                continue
            s = shifts[k]
            if s >= 0:
                if s == 0:
                    grad += wk * (1.0 - 2.0 * x)
                else:
                    grad[s:] += wk * (1.0 - x[:n - s])
                    grad[:n - s] -= wk * x[s:]
            else:
                sp = -s
                grad[:n - sp] += wk * (1.0 - x[sp:])
                grad[sp:] -= wk * x[:n - sp]
        return grad

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    result = minimize(obj, h_start.copy(), jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    h_out = project_box_sum(result.x, target)
    sc = compute_c5(h_out)
    return h_out, sc

def slsqp_polish_batch(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    best_h = h_start.copy()
    best_sc = compute_c5(best_h)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        h_new, sc = slsqp_polish(best_h, target, alpha, maxiter, ftol, time_limit, start_time)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new
    return best_h, best_sc

def binary_greedy(h_init, max_iters, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 300:
            break

        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        A = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]

        max_val = float(corr.max())
        found = False
        for i in top_i:
            for j in top_j:
                # Compute delta
                delta = np.zeros(2 * n - 1)
                k = np.arange(2 * n - 1) - (n - 1)
                idx_i = i + k
                m = (idx_i >= 0) & (idx_i < n)
                delta[m] += h[idx_i[m]]
                idx_i2 = i - k
                m = (idx_i2 >= 0) & (idx_i2 < n)
                delta[m] += h[idx_i2[m]]
                idx_j = j - k
                m = (idx_j >= 0) & (idx_j < n)
                delta[m] -= h[idx_j[m]]
                idx_j2 = j + k
                m = (idx_j2 >= 0) & (idx_j2 < n)
                delta[m] -= h[idx_j2[m]]

                new_max = float((corr + delta).max())
                if new_max < max_val - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    stall = 0
                    nc = float(corr.max()) * dx
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            stall += 1

    return best_h, best_c5

def sa_binary(h_init, n_iters, t_init, t_final, time_limit, start_time, rng):
    n = len(h_init)
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 40000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        mode = rng.random()
        if mode < 0.6:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                ci = ones[A[ones] < 1.0]
                if len(ci) == 0:
                    ci = ones
                i = ci[rng.randint(len(ci))]
                B = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                cz = zeros[B[zeros] > 1.0]
                if len(cz) == 0:
                    cz = zeros
                j = cz[rng.randint(len(cz))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        # Compute delta
        delta = np.zeros(2 * n - 1)
        k = np.arange(2 * n - 1) - (n - 1)
        idx_i = i + k
        m = (idx_i >= 0) & (idx_i < n)
        delta[m] += h[idx_i[m]]
        idx_i2 = i - k
        m = (idx_i2 >= 0) & (idx_i2 < n)
        delta[m] += h[idx_i2[m]]
        idx_j = j - k
        m = (idx_j >= 0) & (idx_j < n)
        delta[m] -= h[idx_j[m]]
        idx_j2 = j + k
        m = (idx_j2 >= 0) & (idx_j2 < n)
        delta[m] -= h[idx_j2[m]]

        new_max = float((corr + delta).max())
        c5_new = new_max * dx
        dc = c5_new - current_c5

        if dc < 0 or (t > 1e-15 and rng.random() < np.exp(-dc / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def multi_alpha_lse(h_init, n, target, alphas, lrs, time_limit, start_time):
    best_h = h_init.copy()
    best_sc = compute_c5(best_h)
    for i, (alpha, lr) in enumerate(zip(alphas, lrs)):
        if time.time() - start_time > time_limit:
            break
        frac = (i + 1) / len(alphas)
        tl = start_time + time_limit * frac
        h_new, sc = lse_grad_descent_v2(best_h, 500000, alpha, alpha * 2, lr, lr * 0.1, tl, start_time)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new
    return best_h, best_sc

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_cont_h = project_box_sum(h_init, target_init)
    best_cont_score = compute_c5(best_cont_h)
    best_bin_h = project_binary(h_init, int(n_init / 2))
    best_bin_score = compute_c5(best_bin_h)

    # Phase 1: Binary greedy (quick)
    allowed = min(40.0, max(10.0, tleft() * 0.05))
    if allowed > 10:
        h_g, c5_g = binary_greedy(best_bin_h, max_iters=30000, time_limit=allowed, start_time=start_time)
        if c5_g < best_bin_score:
            best_bin_score = c5_g
            best_bin_h = h_g.copy()

    # Phase 2: Binary SA (moderate)
    allowed = min(120.0, max(20.0, tleft() * 0.12))
    if allowed > 20:
        n_sa = int(min(3000000, allowed * 30000))
        h_sa, c5_sa = sa_binary(best_bin_h, n_sa, 0.015, 0.0001, allowed, start_time, rng)
        if c5_sa < best_bin_score:
            best_bin_score = c5_sa
            best_bin_h = h_sa.copy()

    # Phase 3: Continuous topk gradient descent
    allowed = min(180.0, max(30.0, tleft() * 0.15))
    if allowed > 30:
        n_iters = int(min(500000, allowed * 5000))
        h_pg, c5_pg = topk_gradient_descent(best_cont_h, n_iters, topk=5,
                                            lr_init=0.06, lr_end=0.001,
                                            time_limit=allowed, start_time=start_time)
        if c5_pg < best_cont_score:
            best_cont_score = c5_pg
            best_cont_h = h_pg.copy()

    # Phase 4: LogSumExp gradient descent (adaptive alpha)
    allowed = min(200.0, max(30.0, tleft() * 0.18))
    if allowed > 30:
        # Multiple alpha schedules
        alphas = [(5.0, 50.0, 0.04), (20.0, 200.0, 0.03), (50.0, 2000.0, 0.02)]
        for a_start, a_end, lr in alphas:
            if time.time() - start_time > start_time + allowed:
                break
            sub_allowed = min(allowed / 3.0, max(15.0, (budget_s - 15) - (time.time() - start_time)))
            n_it = int(min(300000, sub_allowed * 5000))
            h_lse, sc_lse = lse_grad_descent_v2(best_cont_h, n_it, a_start, a_end,
                                                lr, lr * 0.05,
                                                time_limit=sub_allowed, start_time=start_time)
            if sc_lse < best_cont_score:
                best_cont_score = sc_lse
                best_cont_h = h_lse.copy()

    # Phase 5: SLSQP polish
    allowed = min(200.0, max(30.0, tleft() * 0.20))
    if allowed > 30:
        h_ref, sc_ref = slsqp_polish_batch(best_cont_h, target_init,
                                            alphas=[50, 200, 1000, 5000],
                                            maxiter=3000, ftol=1e-14,
                                            time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 6: Iterate - perturb + reoptimize
    for iteration in range(4):
        if tleft() < 100:
            break

        # Perturb
        n_pert = rng.randint(5, 15)
        h_pert = best_cont_h.copy()
        for _ in range(n_pert):
            idx1 = rng.randint(n_init)
            idx2 = rng.randint(n_init)
            h_pert[idx1] = rng.uniform(0.0, 1.0)
            h_pert[idx2] = rng.uniform(0.0, 1.0)
        h_pert = project_box_sum(h_pert, target_init)

        # Quick LSE
        allowed = min(40.0, max(10.0, tleft() * 0.08))
        if allowed > 15:
            h_p2, sc_p2 = lse_grad_descent_v2(h_pert, 100000, 20.0, 500.0,
                                               0.03, 0.001,
                                               time_limit=allowed, start_time=start_time)
            if sc_p2 < best_cont_score:
                best_cont_score = sc_p2
                best_cont_h = h_p2.copy()

        # Topk from best
        if tleft() < 80:
            break
        allowed = min(50.0, max(15.0, tleft() * 0.10))
        if allowed > 15:
            h_p3, sc_p3 = topk_gradient_descent(best_cont_h, 200000, topk=3,
                                                 lr_init=0.04, lr_end=0.0005,
                                                 time_limit=allowed, start_time=start_time)
            if sc_p3 < best_cont_score:
                best_cont_score = sc_p3
                best_cont_h = h_p3.copy()

        # SLSQP
        if tleft() < 60:
            break
        allowed = min(60.0, max(20.0, tleft() * 0.12))
        if allowed > 20:
            h_p4, sc_p4 = slsqp_polish_batch(best_cont_h, target_init,
                                              alphas=[200, 1000, 5000],
                                              maxiter=2000, ftol=1e-15,
                                              time_limit=allowed, start_time=start_time)
            if sc_p4 < best_cont_score:
                best_cont_score = sc_p4
                best_cont_h = h_p4.copy()

    # Phase 7: Try different resolutions
    for n_try in [n_init + 2, n_init - 2, n_init + 6, n_init + 10]:
        if tleft() < 60:
            break
        if n_try < 200:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_cont_h, n_try, target_try)

        allowed = min(25.0, max(10.0, tleft() * 0.06))
        if allowed < 10:
            continue
        h_r, sc_r = topk_gradient_descent(h_interp, 100000, topk=4,
                                           lr_init=0.04, lr_end=0.001,
                                           time_limit=allowed, start_time=start_time)
        if sc_r < best_cont_score:
            best_cont_score = sc_r
            best_cont_h = h_r.copy()

        if tleft() < 40:
            break
        allowed2 = min(30.0, max(10.0, tleft() * 0.08))
        if allowed2 > 15:
            h_r2, sc_r2 = slsqp_polish(h_interp, target_try, alpha=500,
                                        maxiter=3000, ftol=1e-14,
                                        time_limit=allowed2, start_time=start_time)
            if sc_r2 < best_cont_score:
                best_cont_score = sc_r2
                best_cont_h = h_r2.copy()

    # Phase 8: Final SLSQP burst
    if tleft() > 20:
        n_use = len(best_cont_h)
        target_use = n_use / 2.0
        allowed = max(10.0, tleft() - 10)
        h_final_ref, sc_final = slsqp_polish_batch(best_cont_h, target_use,
                                                    alphas=[500, 2000, 10000],
                                                    maxiter=5000, ftol=1e-15,
                                                    time_limit=allowed, start_time=start_time)
        if sc_final < best_cont_score:
            best_cont_score = sc_final
            best_cont_h = h_final_ref.copy()

    # Final: choose best
    n_final = len(best_cont_h)
    h_out = project_box_sum(best_cont_h, n_final / 2.0)
    sc_out = compute_c5(h_out)

    h_bin_out = project_binary(best_bin_h, int(n_init / 2))
    sc_bin = compute_c5(h_bin_out)

    if sc_bin < sc_out:
        return h_bin_out, sc_bin, n_init
    else:
        return h_out, sc_out, n_final
