import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    g = 1.0 - h
    half_n = n - 1

    # the coefficient of h[i] is g[i+k]*dx if 0<=i+k<n else 0, where k=j-(n-1)

    # Actually, let's be more careful.

    # So coeff[j, i] = g[i+k] * dx if 0 <= i+k < n, else 0
    
    # Build the matrix A (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        k = j - half_n
        # valid i: 0 <= i <= n-1 and 0 <= i+k <= n-1
        i_lo = max(0, -k)
        i_hi = min(n, n-k)  # exclusive
        if i_hi > i_lo:

    h_var = cp.Variable(n)

    constraints = [

        h_var >= 0,
        h_var <= 1,

def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    """Coordinate descent on the max of linear functions."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    half_n = n - 1
    idxs = np.arange(n)
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            
            # Solve: min_x max_k (slopes[k]*x + intercepts[k]) s.t. 0<=x<=1
            C = slopes
            if np.all(C > 1e-14):
                h[i] = 0.0
            elif np.all(C < -1e-14):
                h[i] = 1.0
            else:
                lo = 0.0
                hi = 1.0
                for _ in range(40):
                    mid = 0.5 * (lo + hi)
                    vals = C * mid + intercepts
                    max_val = np.max(vals)
                    tol = 1e-12 * max(1.0, abs(max_val))
                    active = np.abs(vals - max_val) < tol
                    if not np.any(active):
                        active = np.zeros(n_active, dtype=bool)
                        active[np.argmax(vals)] = True
                    if np.all(C[active] > 1e-14):
                        hi = mid
                    elif np.all(C[active] < -1e-14):
                        lo = mid
                    else:
                        break
                h[i] = 0.5 * (lo + hi)
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active
    
    return best_h, best_c5

    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)

def compute_c5(h):

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 60:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Full alternating LP - primary optimization
    if tleft() > 200:
        lp_time = min(350, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=10000, time_limit=lp_time,
                                          d0=0.97, reg0=5e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent
    if tleft() > 120:
        cd_time = min(100, tleft() * 0.12)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=120,
                                    max_passes=15000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Another LP round with tighter parameters
    if tleft() > 180:
        lp_time = min(200, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                          max_iter=10000, time_limit=lp_time,
                                          d0=0.99, reg0=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP medium alpha
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [500, 2000, 10000], 2000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: LP with ECOS solver
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.15)
        h = best_h.copy()
        h = project_feasible(h, target)
        best_h5 = h.copy()
        best_c5_5 = compute_c5(h)
        start5 = time.time()
        reg5 = 1e-5
        damp5 = 0.99
        for it in range(5000):
            if time.time() - start5 > lp_time:
                break
            h_sol, _ = solve_lp_full(h, n, target, reg5, solver='ECOS')
            h_updated = (1.0 - damp5) * h + damp5 * h_sol
            h_updated = project_feasible(h_updated, target)
            c5_val = compute_c5(h_updated)
            if c5_val < best_c5_5 - 1e-13:
                best_c5_5 = c5_val
                best_h5 = h_updated.copy()
                damp5 = min(0.995, damp5 + 0.005)
            else:
                damp5 = max(0.6, damp5 * 0.97)
            h = h_updated
            if it % 40 == 39 and reg5 > 1e-8:
                reg5 = max(1e-8, reg5 * 0.5)
        if best_c5_5 < best_score:
            best_score = best_c5_5
            best_h = best_h5

    # Phase 6: Coordinate descent with more active constraints
    if tleft() > 100:
        cd_time = min(80, tleft() * 0.1)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=200,
                                    max_passes=15000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Subgradient refinement
    if tleft() > 60:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=3000, time_limit=min(40, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Try n=300
    if tleft() > 200:
        n_try = 300
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cur = h_interp.copy()
        sc_cur = compute_c5(h_cur)
        if tleft() > 150:
            lp_time = min(100, tleft() * 0.3)
            h_lp300, c5_lp300 = alternating_lp_full(h_cur, n_try, target_try,
                                                     max_iter=8000, time_limit=lp_time,
                                                     d0=0.97, reg0=5e-5)
            if c5_lp300 < sc_cur:
                sc_cur = c5_lp300
                h_cur = h_lp300
        if tleft() > 80:
            cd_time = min(60, tleft() * 0.3)
            h_cd300, c5_cd300 = coord_descent(h_cur, target_try, n_active=100,
                                               max_passes=10000, time_limit=cd_time, rng=rng)
            if c5_cd300 < sc_cur:
                sc_cur = c5_cd300
                h_cur = h_cd300
        if sc_cur < best_score:
            best_score = sc_cur
            best_h = h_cur
            n = n_try
            target = target_try

    # Phase 9: Try n=500
    if tleft() > 200:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cur = h_interp.copy()
        sc_cur = compute_c5(h_cur)
        if tleft() > 150:
            lp_time = min(100, tleft() * 0.3)
            h_lp500, c5_lp500 = alternating_lp_full(h_cur, n_try, target_try,
                                                     max_iter=8000, time_limit=lp_time,
                                                     d0=0.97, reg0=5e-5)
            if c5_lp500 < sc_cur:
                sc_cur = c5_lp500
                h_cur = h_lp500
        if tleft() > 80:
            cd_time = min(60, tleft() * 0.3)
            h_cd500, c5_cd500 = coord_descent(h_cur, target_try, n_active=120,
                                               max_passes=10000, time_limit=cd_time, rng=rng)
            if c5_cd500 < sc_cur:
                sc_cur = c5_cd500
                h_cur = h_cd500
        if sc_cur < best_score:
            best_score = sc_cur
            best_h = h_cur
            n = n_try
            target = target_try

    # Phase 10: Perturbed restarts
    for noise in [0.03, 0.08, 0.15]:
        if tleft() < 80:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        h_ref = slsqp_round(h_pert, n_use, target_use, [500, 2000, 10000], 1500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final LP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(50, tleft() * 0.5)
        h_lp, c5_lp = alternating_lp_full(best_h, n_use, target_use,
                                          max_iter=5000, time_limit=lp_time,
                                          d0=0.99, reg0=1e-7)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final coordinate descent
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(30, tleft() * 0.6)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=200,
                                    max_passes=10000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 13: Final SLSQP micro-polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(best_h, n_use, target_use, [50000, 200000, 1000000], 600, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 14: Final subgradient
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=2000, time_limit=min(10, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
