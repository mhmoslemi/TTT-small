        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts):
    """Solve min_{x in [0,1]} max_i (slopes[i]*x + intercepts[i])"""
    n = len(slopes)
    # Check boundaries
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    
    if val_0 <= val_1:
        # Minimum might be at 0 or at a crossing
        # Check if all slopes of active lines at 0 are >= 0
        active_0 = np.abs(intercepts - val_0) < 1e-13 * max(1.0, abs(val_0))
        if np.all(slopes[active_0] >= -1e-13):
            return 0.0
    else:
        # Minimum might be at 1 or at a crossing
        active_1 = np.abs((slopes + intercepts) - val_1) < 1e-13 * max(1.0, abs(val_1))
        if np.all(slopes[active_1] <= 1e-13):
            return 1.0
    
    # Bisection
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-13 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(n, dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(slopes[active] > 1e-13):
            lo = mid
        elif np.all(slopes[active] < -1e-13):
            hi = mid
        else:
            break
    return 0.5 * (lo + hi)

def jacobi_cd(h_init, target, n_active, max_passes, time_limit):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k_idx = np.argsort(corr)[-n_active:]
    half_n = n - 1
    K = active_k_idx - half_n  # actual lag values
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        # Build slope and intercept matrices
        # S[idx, i] = (1 - h[i+K[idx]] - h[i-K[idx]]) * dx
        # I[idx, i] = corr[active_k_idx[idx]] - h[i] * S[idx, i]
        
        S = np.zeros((n_active, n))
        I = np.zeros((n_active, n))
        
        for idx in range(n_active):
            k = K[idx]
            # h[i+k] for valid i
            lo = max(0, -k)
            hi = min(n, n - k)
            # S[idx, i] += (1 - h[i+k]) for i in [lo, hi)
            S[idx, lo:hi] += (1.0 - h[lo+k:hi+k]) * dx
            # h[i-k] for valid i
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            # S[idx, i] += -h[i-k] for i in [lo2, hi2)
            S[idx, lo2:hi2] += (-h[lo2-k:hi2-k]) * dx
            
            I[idx, :] = corr[active_k_idx[idx]] - h * S[idx, :]
        
        # Jacobi update: for each i, solve min_{x∈[0,1]} max_idx (S[idx,i]*x + I[idx,i])
        # Vectorized bisection
        lo = np.zeros(n)
        hi = np.ones(n)
        for _ in range(35):
            mid = 0.5 * (lo + hi)
            vals = S @ mid + I  # n_active × n
            max_val = np.max(vals, axis=0)  # n
            tol = 1e-12 * np.maximum(1.0, np.abs(max_val))
            active = np.abs(vals - max_val[None, :]) < tol[:, None]  # n_active × n
            # For each i: if all active slopes > 0 → lo = mid; if all < 0 → hi = mid
            # min_active_slope[i] = min of S[idx, i] where active[idx, i]
            # max_active_slope[i] = max of S[idx, i] where active[idx, i]
            
            # Use where to set inactive to ±inf
            min_as = np.where(active, S, np.inf).min(axis=0)
            max_as = np.where(active, S, -np.inf).max(axis=0)
            
            lo[np.all(active * (S > 1e-12), axis=0)] = mid[np.all(active * (S > 1e-12), axis=0)]
            # This is getting complicated. Let me simplify.
            
            # If min active slope > 0 at i: lo[i] = mid[i]
            # If max active slope < 0 at i: hi[i] = mid[i]
            # Otherwise: converged at i
            
            go_lo = min_as > 1e-12
            go_hi = max_as < -1e-12
            lo = np.where(go_lo, mid, lo)
            hi = np.where(go_hi, mid, hi)
        
        h_new = 0.5 * (lo + hi)
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # Project to feasible
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        h = h_new
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Update active set
        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k_idx)):
            active_k_idx = new_active
            K = active_k_idx - half_n
    
    return best_h, best_c5

Let me also think about a completely different approach: **direct optimization of the max using the smooth max (log-sum-exp)**.

The smooth max: f_α(h) = (1/α) * log(sum_k exp(α * O_k(h)))

As α → ∞, this approaches max_k O_k(h). We can minimize f_α(h) using gradient descent or SLSQP.

The gradient of O_k with respect to h[i] is:
d O_k / d h[i] = (1 - h[i+k] - h[i-k]) * dx (same as before)

So the gradient of f_α is:
∇f_α = sum_k p_k * ∇O_k / α
where p_k = exp(α * O_k) / sum_j exp(α * O_j)

This is a weighted average of the gradients of the individual O_k's, weighted by the soft-max probabilities.

For large α, this focuses on the active (max) lags. For small α, it's a smooth approximation.

We can use this with SLSQP or with projected gradient descent.

def smooth_max_cd(h_init, target, max_steps, time_limit, alphas):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    
    for alpha in alphas:
        for step in range(max_steps):
            if time.time() - start > time_limit:
                break
            
            corr = np.correlate(h, 1.0 - h, mode='full') / dx  # O_k values (without dx)
            # Actually, O_k = corr[k_idx] * dx
            # Let's work with the raw corr values
            O = corr * dx  # actual overlap values
            
            # Soft-max weights
            O_shifted = alpha * (O - np.max(O))
            p = np.exp(O_shifted) / np.sum(np.exp(O_shifted))
            
            # Gradient of O_k w.r.t. h[i]: (1 - h[i+k] - h[i-k]) * dx
            # Weighted gradient: sum_k p_k * (1 - h[i+k] - h[i-k]) * dx
            grad = np.zeros(n)
            for k_idx in range(1, len(corr) - 1):  # skip k=-(n-1) and k=n-1 (zero overlap)
                if p[k_idx] < 1e-15:
                    continue
                k = k_idx - half_n
                # d O_k / d h[i] = (1 - h[i+k] - h[i-k]) * dx
                valid1 = (0 <= idxs - k) & ((idxs - k) < n)
                valid2 = (0 <= idxs + k) & ((idxs + k) < n)
                grad[valid1] += p[k_idx] * (1 - h[idxs[valid1] - k]) * dx
                grad[valid2] += -p[k_idx] * h[idxs[valid2] + k] * dx
            
            # Actually wait, let me recheck. O_k = sum_i h[i](1-h[i+k]) * dx

            # = (1 - h[i+k] - h[i-k]) * dx
            # Yes, that's correct.
            
            # Line search
            improved = False
            for lr in [0.01, 0.03, 0.1, 0.3, 0.5]:
                h_try = project_feasible(h - lr * grad, target)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_line(slopes, intercepts):
    n = len(slopes)
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    if val_0 <= val_1:
        tol = 1e-13 * max(1.0, abs(val_0))
        active_0 = np.abs(intercepts - val_0) < tol
        if np.all(slopes[active_0] >= -1e-13):
