import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def compute_c5(h):

def full_corr_fast(h):

    N = 1

    b = np.zeros(N)

    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr

    result[mask] -= h[idx[mask]]

    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def compute_lag_gradient(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:n])
        g[k:] -= h[:n-k]
    else:
        k2 = -k
        g[k2:] += (1.0 - h[:n-k2])
        g[:n-k2] -= h[k2:n]
    return g

def smooth_gradient_top(h, corr, alpha, top_m=30):
    n = len(h)
    indices = np.argsort(corr)[::-1][:top_m]
    vals = corr[indices]
    shifted = alpha * (vals - vals.max())
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for w, idx in zip(weights, indices):
        k = idx - (n - 1)
        grad += w * compute_lag_gradient(h, k, n)
    return grad

def slsqp_smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    cons_sum = sum(h_start)
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(x, a=alpha):
            return slsqp_smooth_obj(x, a)
        def con_fun(x, t=target):
            return np.sum(x) - t
        def con_jac(x, t=target):
            return np.ones(n)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

        allowed2 = min(20.0, tleft() * 0.05)
        n_sa_ms = int(allowed2 * 6000)
        h_ms_s, c5_ms_s = sa_with_2swap(h_ms_g, n_sa_ms, 0.01, 0.00005, rng,
                                         time_limit=allowed2, start_time=start_time, k_arr=k_arr,
                                         use_2swap_frac=0.3)
        c5_ms = c5_ms_s if c5_ms_s < c5_ms_g else c5_ms_g
        h_ms_b = h_ms_s if c5_ms_s < c5_ms_g else h_ms_g
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms_b.copy()

    # Phase 10: Final continuous polish
    if tleft() > 50:
        allowed = min(tleft() - 20, 80.0)
        alphas_final = [500, 2000, 10000, 40000]
        h_fin, c5_fin = continuous_optimize(best_h, float(target), start_time,
                                            start_time + allowed, alphas_final, top_m=20)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    # Phase 11: Final binary SA burst
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target)
        allowed = max(5.0, min(30.0, tleft() - 10.0))
        n_sa_f = int(allowed * 8000)
        h_f, c5_f = sa_with_2swap(h_bin_final, n_sa_f, 0.008, 0.00001, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                    use_2swap_frac=0.4)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, float(target))
    final_score = compute_c5(h_final)

    return h_final, final_score, n
