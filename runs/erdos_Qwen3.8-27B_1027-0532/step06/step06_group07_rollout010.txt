def improved_pair_cd(h, n, target, time_limit, max_sweeps, n_cand, top_k):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive = 0
        else:
            consecutive += 1
        if consecutive > 120:
            break

        thresh = np.max(corr) * (1.0 - 1e-6) - top_k * 0.01 / n
        active = np.where(corr >= np.max(corr) - 0.02 / dx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        active = active[:top_k]

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_bisect = 13

        for i in high_set:
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(2 * n - 1)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi_val, 1.0 - h[j])
                if delta_hi < 1e-14:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_set = vals >= f_mid - 1e-12
                    f_prime = np.max(g[active_set]) if active_set.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-14:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial - 1e-15:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_onto_set(h, target)
        elif consecutive > 40:
            break

    return best_h, best_c5


def random_perturb_and_optimize(h_base, n, target, time_limit, rng, n_perturb):
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    for _ in range(n_perturb):
        if time.time() % 1 > time_limit:
            break
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.15]))
        h_try = h_base + rng.randn(n) * scale
        h_try = project_onto_set(h_try, target)
        csc = compute_c5(h_try)
        if csc < best_c5 - 1e-13:
            best_c5 = csc
            best_h = h_try.copy()
    return best_h, best_c5


def slsqp_optimize(h_start, n, target, alphas, maxiter):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    boundso = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=boundso,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_onto_set(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _global_target


def _cons_jac(h):
    return np.ones(_global_n)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_onto_set(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_onto_set(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    M = n / 2.0
    rng = np.random.RandomState(seed)

    best_h = project_onto_set(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: P-norm projected gradient with increasing p
    p_list = [10, 30, 80, 200]
    for p in p_list:
        if tleft() < 60:
            break
        tl = min(120, tleft() * 0.2)
        h_pg, c5_pg = pnorm_projected_grad(best_h, n, target, p, tl,
                                           lr_init=0.05, momentum=0.85,
                                           max_iter=2000)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 2: Improved pair CD with multi-lag
    if tleft() > 150:
        cd_tl = min(250, tleft() * 0.25)
        h_cd, c5_cd = improved_pair_cd(best_h, n, target, cd_tl,
                                       max_sweeps=5000, n_cand=50, top_k=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: More p-norm rounds
    if tleft() > 100:
        for p in [300, 500, 1000]:
            if tleft() < 50:
                break
            tl = min(80, tleft() * 0.15)
            h_pg, c5_pg = pnorm_projected_grad(best_h, n, target, p, tl,
                                               lr_init=0.02, momentum=0.9,
                                               max_iter=1500)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg

    # Phase 4: Pair CD again
    if tleft() > 120:
        cd_tl = min(200, tleft() * 0.2)
        h_cd, c5_cd = improved_pair_cd(best_h, n, target, cd_tl,
                                       max_sweeps=4000, n_cand=60, top_k=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if tleft() > 80:
        tl = min(100, tleft() * 0.15)
        h_pol = slsqp_optimize(best_h, n, target, [5000, 20000, 80000], 800)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Random restarts + local opt
    if tleft() > 100:
        for noise_scale in [0.03, 0.07, 0.12]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_onto_set(h_pert, target)
            # Quick p-norm from perturbed start
            tl = min(40, tleft() * 0.1)
            h_try, c5_try = pnorm_projected_grad(h_pert, n, target, 50, tl,
                                                 lr_init=0.03, momentum=0.85,
                                                 max_iter=800)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try
            # Quick pair CD
            if tleft() > 40:
                cd_tl = min(40, tleft() * 0.1)
                h_try2, c5_try2 = improved_pair_cd(h_try, n, target, cd_tl,
                                                   max_sweeps=1500, n_cand=40, top_k=8)
                if c5_try2 < best_score:
                    best_score = c5_try2
                    best_h = h_try2

    # Phase 7: Try higher resolution
    if tleft() > 200:
        for n_try in [800, 1000]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            # Quick p-norm at high res
            tl = min(80, tleft() * 0.2)
            h_hr, c5_hr = pnorm_projected_grad(h_interp, n_try, target_try, 50, tl,
                                               lr_init=0.03, momentum=0.85,
                                               max_iter=1000)
            if c5_hr < best_score and tleft() > 60:
                cd_tl = min(60, tleft() * 0.15)
                h_hr2, c5_hr2 = improved_pair_cd(h_hr, n_try, target_try, cd_tl,
                                                 max_sweeps=2000, n_cand=50, top_k=10)
                if c5_hr2 < best_score:
                    best_score = c5_hr2
                    best_h = h_hr2

    # Phase 8: Final pair CD
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_tl = min(60, tleft() * 0.5)
        h_cd, c5_cd = improved_pair_cd(best_h, n_use, target_use, cd_tl,
                                       max_sweeps=3000, n_cand=55, top_k=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_optimize(best_h, n_use, target_use, [20000, 100000], 600)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final projection
    n_use = len(best_h)
    target_use = n_use / 2.0
    best_h = project_onto_set(best_h, target_use)
    final_score = compute_c5(best_h)

    return best_h, final_score, len(best_h)
