            mask_pos = idxs <= (n - 1 - k)
            mask_neg = idxs >= k
            contrib = np.zeros(n)
            contrib[mask_pos] = 1.0 - h[idxs[mask_pos] + k]
            contrib[mask_neg] += -h[idxs[mask_neg] - k]
            grad += contrib / n_active
        else:
            k_neg = -k
            # corr[n-1-k] = Σ_j h[j+k_neg](1-h[j]) for valid j
            # ∂/∂h[i] = (1-h[i])*[i>=k_neg, i<=n-1] + (-h[i+k_neg])*[i<=n-1-k_neg]
            # Actually by symmetry with k>0 case:
            mask_pos = idxs <= (n - 1 - k_neg)
            mask_neg = idxs >= k_neg
            contrib = np.zeros(n)
            contrib[mask_pos] = 1.0 - h[idxs[mask_pos] + k_neg]
            contrib[mask_neg] += -h[idxs[mask_neg] - k_neg]
            grad += contrib / n_active
    
    return grad, c_max, active

def projected_subgradient(h_init, n, target, time_limit, lr_init=0.01, epsilon=0.005, n_avg=50):
    """Projected subgradient method with averaging."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_sum = h.copy()
    n_iter = 0
    lr = lr_init
    
    while time.time() - start < time_limit:
        grad, c_max, active = subgradient_all_active(h, n, epsilon)
        grad -= np.mean(grad)  # Remove DC component to preserve sum
        
        # Polyak-like step size
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        if c5_try < c_max * (2.0 / n):
            h = h_try
            h_sum += h
            n_iter += 1
            lr *= 1.01  # Slight increase
        else:
            lr *= 0.7
        
        # Adaptive epsilon
        if n_iter % 100 == 0 and n_iter > 0:
            corr = compute_corr(best_h)
            c_max_best = np.max(corr)
            sorted_corr = np.sort(corr)[::-1]
            epsilon = max(0.001, 0.5 * (sorted_corr[0] - sorted_corr[min(10, len(sorted_corr)-1)]))
    
    return best_h, best_c5

Hmm, this subgradient approach might be slow because of the O(n) loop over active constraints. Let me think of a faster way.

The key observation: for a given set of active lags S, the gradient is:
grad[i] = (1/|S|) Σ_{k∈S} ∂corr[k]/∂h[i]

For k=0: contribution is (1-2h[i])/|S|
For k>0: contribution is [(1-h[i+k])·1_{i≤n-1-k} - h[i-k]·1_{i≥k}]/|S|
For k<0: similar by symmetry

If S is dense (many active lags), this is expensive. But if we only use a few (say top 5-20), it's manageable.

Actually, let me think about this differently. The most efficient approach might be:

1. Compute corr
2. Find the top M active lags (M = 5 to 20)
3. Compute the average gradient for those M lags
4. Take a projected step

This is O(M·n) per iteration, which for M=10, n=600 is 6000 operations - very fast.

Let me also think about a "smart" pair CD that uses the subgradient information:
- The subgradient tells us which h[i] should increase and which should decrease
- We can do a "mass transport" from high-gradient positions to low-gradient positions
- Use binary search for the optimal transfer amount

Actually, the previous code already does this with the fast_pair_cd. Let me think about what else to try.

**New idea: Mirror-averaging**
Since the problem is symmetric under h → 1-h, if we have a candidate h, we can try:
- h_sym = (h + (1-h))/2 = 1/2 (useless)
- But we can try h and 1-h as separate candidates, keep the better one

More useful: if h is a local min, then 1-h is also a local min (by the symmetry argument above). So we can explore both basins.

**New idea: Phase shift in frequency domain**
If we take the DFT of h, we can rotate the phases and take the IDFT to get a new feasible point. This preserves the power spectrum (and hence the autocorrelation envelope) but changes the time-domain structure.

Wait, that would change the autocorrelation. The autocorrelation depends on |H[k]|², not on the phases. So rotating phases of H preserves the autocorrelation!

This means: if h has a certain autocorrelation, then so does IDFT(|DFT(h)| · e^{i·θ(k)}) for any phase function θ(k). The constraint is that the new h must satisfy h[j] ∈ [0,1] and sum(h) = n/2.

DC component: H[0] = Σh = n/2 (fixed)
For k>0: H[k] and H[n-k] are conjugates (since h is real), so we have n/2 free phase parameters.

This gives us a way to explore the solution space while keeping the autocorrelation fixed! We can use this as a "manifold" to move along, then optimize the amplitudes.

Algorithm:
1. Start with h0
2. Compute H = DFT(h0)
3. Randomly perturb the phases: H_new[k] = |H[k]| · e^{i·(arg(H[k]) + δ[k])}
4. h_new = IDFT(H_new)
5. Project to feasible set
6. Evaluate and keep if better
7. Iterate

This is interesting but might be hard to make efficient.

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, n, k):
    """Gradient of corr[n-1+k] with respect to h."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    elif k > 0:
        mask_pos = idxs <= (n - 1 - k)
        mask_neg = idxs >= k
        grad[mask_pos] = 1.0 - h[idxs[mask_pos] + k]
        grad[mask_neg] += -h[idxs[mask_neg] - k]
    else:
        k_abs = -k
        mask_pos = idxs <= (n - 1 - k_abs)
        mask_neg = idxs >= k_abs
        grad[mask_pos] = 1.0 - h[idxs[mask_pos] + k_abs]
        grad[mask_neg] += -h[idxs[mask_neg] - k_abs]
    return grad

def smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm])

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 30:
                break

    return best_h, best_c5

def multi_active_subgrad(h_init, n, target, time_limit, n_active=10, lr_init=0.005):
    """Subgradient method using top-n active constraints."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    dx = 2.0 / n
    mid = n - 1
    stagnation = 0
    
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 200:
            break
        
        # Find top n_active lags
        top_indices = np.argsort(corr)[::-1][:n_active]
        
        # Average gradient over active lags
        grad_avg = np.zeros(n)
        for s_idx in top_indices:
            k = int(s_idx) - mid
            grad_avg += grad_for_lag(h, n, k)
        grad_avg /= len(top_indices)
        grad_avg -= np.mean(grad_avg)
        
        # Take step
        h_try = project_feasible(h - lr * grad_avg, target)
        corr_try = compute_corr(h_try)
        c5_try = float(np.max(corr_try)) * dx
        
        if c5_try < c5_val:
            h = h_try
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c5

def chirp_init(n, num_chirps=3):
    """Generate initial h based on chirp sequences (flat power spectrum)."""
    idxs = np.arange(n)
    h = np.full(n, 0.5)
    phase = np.random.RandomState(123)
    for m in range(1, num_chirps + 1):
        freq = m
        # Linear chirp
        phi = phase.uniform(0, 2 * np.pi)
        amp = 0.15 / np.sqrt(m)
        h += amp * np.cos(2 * np.pi * freq * idxs / n + phi + 0.5 * 2 * np.pi * freq * idxs**2 / (n * n))
    return project_feasible(h, n / 2.0)

def zadoff_chu_init(n, num_seqs=2):
    """Zadoff-Chu-like initialization."""
    idxs = np.arange(n)
    h = np.full(n, 0.5)
    rng = np.random.RandomState(456)
    for m in range(1, num_seqs + 1):
        amp = 0.2 / np.sqrt(m)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * m * idxs * (idxs + 1) / (2 * n) + phase)
    return project_feasible(h, n / 2.0)

def frequency_phase_explore(h_init, n, target, time_limit, n_trials=50):
    """Explore by rotating phases in frequency domain (preserves autocorrelation)."""
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(789)
    H = np.fft.rfft(h_init)
    
    for trial in range(n_trials):
        if time.time() - start > time_limit:
            break
        
        # Random phase perturbation
        dphi = rng.randn(len(H)) * 0.3
        dphi[0] = 0  # Keep DC fixed
        H_new = np.abs(H) * np.exp(1j * (np.angle(H) + dphi))
        # Enforce conjugate symmetry for real signal
        H_new_ir = np.fft.irfft(H_new, n=n)
        h_try = project_feasible(H_new_ir, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            H = np.fft.rfft(best_h)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def penalty_target_method(h_init, n, target, time_limit, T_init, T_target, max_steps=2000):
    start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1

    T_lo = T_target
    T_hi = T_init
    n_bisect_T = 8

    for T_iter in range(n_bisect_T):
        if time.time() - start_time > time_limit:
            break
        T_mid = 0.5 * (T_lo + T_hi)
        T_corr = T_mid / dx

        h_cur = h.copy()
        lr = 0.05
        for step in range(max_steps):
            if time.time() - start_time > time_limit:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c5_now = float(np.max(corr)) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_cur.copy()

            violations = np.maximum(corr - T_corr, 0.0)
            if violations.max() < 1e-8:
                break

            grad = np.zeros(n)
            active_idx = np.where(corr > T_corr + 1e-10)[0]
            for s_idx in active_idx:
                k = int(s_idx) - mid
                w = (corr[int(s_idx)] - T_corr)
                grad += w * grad_for_lag(h_cur, n, k)

            grad *= 2.0 * dx
            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

            improved = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-6:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_cur.copy()

        if c5_test <= T_mid + 1e-5:
            T_lo = T_mid
            h = h_cur.copy()
        else:
            T_hi = T_mid

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, n, k):
    idxs = np.arange(n)
    grad = np.zeros(n)
    if k == 0:
        grad = 1.0 - 2.0 * h
    else:
        ka = abs(k)
        mask_pos = idxs <= (n - 1 - ka)
        mask_neg = idxs >= ka
        grad[mask_pos] = 1.0 - h[idxs[mask_pos] + ka]
        grad[mask_neg] += -h[idxs[mask_neg] - ka]
    return grad


def smooth_obj(h, alpha):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=45, n_bisect=14):

    best_h = h.copy()

    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):

        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val

            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:

                delta_hi = min(hi, 1.0 - h[j])

                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-7:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_cur.copy()

        if c5_test <= T_mid + 1e-5:
            T_lo = T_mid
            h = h_cur.copy()
        else:
            T_hi = T_mid

    return best_h, best_c5


def chirp_init(n, num_chirps, seed_val):
    idxs = np.arange(n)
    h = np.full(n, 0.5)
    rng = np.random.RandomState(seed_val)
    for m in range(1, num_chirps + 1):
        amp = 0.18 / np.sqrt(m)
        phi = rng.uniform(0, 2 * np.pi)
        chirp_arg = 2 * np.pi * m * idxs / n + phi + np.pi * m * idxs ** 2 / n
        h += amp * np.cos(chirp_arg)
    return project_feasible(h, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Multi-active subgradient
    if tleft() > 150:
        sg_time = min(100, tleft() * 0.12)
        h_sg, c5_sg = multi_active_subgrad(best_h, n, target, time_limit=sg_time, n_active=15, lr_init=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 2: SLSQP polish
    if tleft() > 130:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 10000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Main pair CD
    if tleft() > 250:
        cd_time = min(280, tleft() * 0.22)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=6000, n_cand=55, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP after CD
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Second pair CD round with different params
    if tleft() > 180:
        cd_time = min(200, tleft() * 0.18)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=5000, n_cand=65, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Frequency phase exploration
    if tleft() > 100:
        fpe_time = min(50, tleft() * 0.08)
        h_fpe, c5_fpe = frequency_phase_explore(best_h, n, target, time_limit=fpe_time, n_trials=150)
        if c5_fpe < best_score:
            best_score = c5_fpe
            best_h = h_fpe
            if tleft() > 60:
                h_cd2, c5_cd2 = fast_pair_cd(best_h, n, target, time_limit=min(50, tleft() * 0.2), max_sweeps=3000, n_cand=45, n_bisect=13)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2

    # Phase 7: Penalty target method
    if tleft() > 120:
        pen_time = min(100, tleft() * 0.12)
        T_init_val = best_score + 0.004
        T_target_val = best_score - 0.004
        h_pen, c5_pen = penalty_target_method(best_h, n, target, time_limit=pen_time, T_init=T_init_val, T_target=T_target_val, max_steps=1500)
        if c5_pen < best_score:
            best_score = c5_pen
            best_h = h_pen

    # Phase 8: SLSQP high alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [20000, 100000, 300000], 1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 9: Multi-start with perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.04, 0.08, 0.15]:
        if tleft() < 90:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(70, tleft() * 0.15)
        h_try, c5_try = fast_pair_cd(h_pert, n, target, time_limit=cd_t, max_sweeps=3000, n_cand=45, n_bisect=13)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 10: Chirp-based initialization alternative
    if tleft() > 120:
        for nch in [4, 8]:
            if tleft() < 80:
                break
            h_ch = chirp_init(n, nch, seed_val=42 + nch)
            h_ch_sym = project_feasible(1.0 - h_ch, target)
            for h_start in [h_ch, h_ch_sym]:
                if tleft() < 60:
                    break
                sc_start = compute_c5(h_start)
                if sc_start < best_score + 0.03:
                    cd_t = min(50, tleft() * 0.18)
                    h_try, c5_try = fast_pair_cd(h_start, n, target, time_limit=cd_t, max_sweeps=4000, n_cand=50, n_bisect=14)
                    if c5_try < best_score:
                        best_score = c5_try
                        best_h = h_try

    # Phase 11: Higher resolution
    if tleft() > 180:
        for n_try in [800, 1000]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try, [500, 2000, 8000], 800, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 80 and sc < best_score + 0.003:
                cd_t = min(60, tleft() * 0.25)
                h_hires2, sc2 = fast_pair_cd(h_hires, n_try, target_try, time_limit=cd_t, max_sweeps=3000, n_cand=40, n_bisect=13)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 12: Final pair CD
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use, time_limit=cd_time, max_sweeps=4000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 13: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use, [50000, 200000, 500000], 800, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
