def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            break

```python
def smooth_gradient_fast(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    mid = n - 1
    grad = np.zeros(n)
    
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-16:
            continue
        k = s - mid
        if k > 0:
            grad[:n-k] += w * (1.0 - x[k:])
            grad[k:] -= w * x[:n-k]
        elif k < 0:
            kp = -k
            grad[kp:] += w * (1.0 - x[:n-kp])
            grad[:n-kp] -= w * x[kp:]
        else:
            grad[:] += w * (1.0 - 2.0 * x)
    
    return grad
```

This is O(n²) per gradient evaluation. For n=400, that's 160K operations per gradient. With SLSQP doing maybe 100-1000 gradient evaluations, that's 16M-160M operations total, which should be fine.

    h_prev = h.copy()
    
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Initial BB step size
    s = np.ones(n) * 0.01
    y = np.ones(n) * 0.01
    gamma = np.dot(s, y) / np.dot(y, y)
    gamma = max(gamma, 1e-6)
    
    no_improve = 0

            break

def lBFGS_round(h_start, n, target, time_limit, alphas):
    start = time.time()
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        # Penalty parameter (large to enforce constraint)
        penalty = 1e6
        
        def objective(x):
            base = smooth_objective(x, alpha)
            constraint_viol = np.sum(x) - target
            return base + penalty * constraint_viol ** 2
        
        def gradient(x):
            base_grad = smooth_gradient_fast(x, alpha)
            constraint_grad = penalty * 2 * (np.sum(x) - target) * np.ones(n)
            return base_grad + constraint_grad
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(objective, h_cur, jac=gradient, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_score

def annealed_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand=40, T_start=0.01, T_end=0.0001):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 500:
            break
        
        # Temperature schedule
        progress = min(sweep / max_sweeps, 1.0)
        T = T_start * (T_end / T_start) ** progress

        top_k = 3
        top_idxs = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] / np.sum(corr[top_idxs])
            k = int(s_idx) - mid

            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
