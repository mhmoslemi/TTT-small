    # For all j where 0<=j-s<n: grad[j] -= w_s * h[j-s]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

            extra = final_sum - target_sum

        else:

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def compute_gradient_smooth(h, alpha, corr_full):
    n = len(h)
    max_c = np.max(corr_full)
    shifted = alpha * (corr_full - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(corr_full))
        w[np.argmax(corr_full)] = 1.0
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        ws = w[k_idx]
        if ws < 1e-14:
            continue
        s = k_idx - (n - 1)
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start < j_end:
            grad[j_start:j_end] += ws * (1.0 - h[j_start + s:j_end + s])
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= ws * h[j_start2 - s:j_end2 - s]
    return grad

def nesterov_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    m = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = 0.85
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_budget = min(80.0, time_limit - (time.time() - start_time))
        n_iter = max(200, int(phase_budget * 500))
        lr = 3.0 / (alpha + 1.0)
        corr = full_corr_fast(h)
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            grad = compute_gradient_smooth(h, alpha, corr)
            m = momentum * m - lr * grad
            y = h + m
            h_new = project_box_sum(y, target)
            corr_new = full_corr_fast(h_new)
            obj_new = (1.0 / alpha) * np.log(np.sum(np.exp(np.clip(alpha * (corr_new - np.max(corr_new)), -500, 0))))
            h = h_new
            corr = corr_new
            if it % 100 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

    if tleft() > 100:
        target_cont2 = float(n_init) / 2.0
        alphas_c2 = [10, 30, 100, 300, 800, 2000, 5000, 12000]
        allowed = min(300, tleft() * 0.3)
        h_cont2, sc_cont2 = nesterov_refine(best_h, target_cont2, start_time, allowed, alphas_c2)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2.copy()

    # Phase 7:.Round + focused local search
    if tleft() > 50:
        h_bin2 = project_binary(best_h, target_init)
        allowed = min(60, tleft() * 0.12)
        h_ls, c5_ls = local_search_binary(h_bin2, rng,
                                          time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls.copy()

    # Phase 8: Multi-restart with diverse perturbations
    for restart in range(12):
        if tleft() < 60:
            break
        if restart % 4 == 0:
            h_p = perturbed_copy(best_h, rng, rng.randint(3, 8), 1, 10)
        elif restart % 4 == 1:
            h_p = perturbed_copy(best_h, rng, rng.randint(8, 15), 1, 12)
        elif restart % 4 == 2:
            h_p = perturbed_copy(best_h, rng, rng.randint(2, 5), 2, 8)
        else:
            h_p = perturbed_copy(best_h, rng, rng.randint(5, 10), 2, 10)

        allowed = min(45, tleft() * 0.08)
        n_rrt_r = int(min(800000, allowed * 40000))
        h_rr, c5_rr = rrt_binary(h_p, n_rrt_r, 0.002, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_rr < best_score:
            best_score = c5_rr
            best_h = h_rr.copy()

    # Phase 9: Final continuous polish
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_final = [20, 60, 200, 600, 1500, 4000, 10000, 30000]
        allowed = min(200, tleft() * 0.4)
        h_final_cont, sc_final = nesterov_refine(best_h, target_use, start_time, allowed, alphas_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cont.copy()

    # Phase 10: Final binary local search
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(30, tleft() * 0.3)
        h_ls_f, c5_ls_f = local_search_binary(h_bin_f, rng,
                                              time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_ls_f < best_score:
            best_score = c5_ls_f
            best_h = h_ls_f.copy()

    # Phase 11: Very short RRT burst
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_b = project_binary(best_h, target_use)
        allowed = min(20, tleft() * 0.4)
        n_burst = int(min(500000, allowed * 60000))
        h_b, c5_b = rrt_binary(h_bin_b, n_burst, 0.001, 0.000005, rng,
                                time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_b < best_score:
            best_score = c5_b
            best_h = h_b.copy()

    # Final: ensure feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_alt = project_binary(best_h, int(n_final / 2.0))
        alt_score = compute_c5(h_alt)
        if alt_score < final_score:
            h_final = h_alt
            final_score = alt_score

    return h_final, final_score, n_final
