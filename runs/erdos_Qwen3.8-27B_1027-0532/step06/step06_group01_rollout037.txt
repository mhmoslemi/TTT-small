def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_matrix(h):
    """Build the full gradient matrix G where G[m, n-1+k] = dc_k/dh[m]."""
    n = len(h)

    G = np.zeros((n, n_corr))

    # B part: G[m, n-1:2n-1-m] -= h[m:]

**Alternative: Use the dual problem**

The primal is:
minimize max_k c_k(h)
subject to 0 ≤ h ≤ 1, 1^T h = n/2

The dual (informally) involves finding a probability distribution over k that "witnesses" the minimum. At the optimum, there exists a distribution λ over k such that:
∑_k λ_k c_k(h) = max_k c_k(h) (this is the minimax theorem)

And h minimizes ∑_k λ_k c_k(h) over the feasible set.

So the algorithm would be:
1. Maintain a distribution λ over k
2. Solve: minimize ∑_k λ_k c_k(h) subject to 0≤h≤1, 1^T h = n/2
3. Update λ to put more weight on the maximizing k
4. Repeat

Step 2 is a QP (since ∑λ_k c_k(h) is quadratic in h). Step 3 is easy (just find the max and update).

This is essentially the multiplicative weights / exponentiated gradient method for zero-sum games!

The QP in step 2:
minimize ∑_k λ_k [∑_j h[j+k](1-h[j])]
= minimize ∑_k λ_k [∑_j h[j+k] - ∑_j h[j+k]h[j]]
= minimize (linear in h) - (quadratic in h)

This is a concave quadratic minimization (since the quadratic part is negative definite... or is it? Let me check).

∑_k λ_k ∑_j h[j+k]h[j] = h^T (∑_k λ_k A_k) h (where A_k is the shift matrix)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_one_k(h, k, n):
    g = np.zeros(n)
    j1s = max(0, k)
    j1e = min(n, n + k)
    if j1s < j1e:
        g[j1s:j1e] += (1.0 - h[j1s - k:j1e - k])
    j2s = max(0, -k)
    j2e = min(n, n - k)
    if j2s < j2e:
        g[j2s:j2e] -= h[j2s + k:j2e + k]
    return g

            row[-1] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(dx * (np.dot(g, h) - corr[idx]))

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            n_cons = max(20, n_cons // 2)
            no_improve += 5
            if no_improve > 50:
                break
            continue

        h_new = res.x[:n]
        c5_new = compute_c5(h_new)

        if c5_new < c5_now - 1e-14:
            h = h_new
            n_cons = min(n_corr, int(n_cons * 1.4))
        elif c5_new < c5_now + 1e-12:
            h_try = 0.6 * h + 0.4 * h_new
            h_try = project_feasible(h_try, target)
            if compute_c5(h_try) < c5_now - 1e-14:
                h = h_try
            else:
                h_try2 = 0.4 * h + 0.6 * h_new
                h_try2 = project_feasible(h_try2, target)
                c5_try2 = compute_c5(h_try2)
                if c5_try2 < c5_now - 1e-14:
                    h = h_try2
                else:
                    no_improve += 3
        else:
            h_try = 0.3 * h + 0.7 * h_new
            h_try = project_feasible(h_try, target)
            if compute_c5(h_try) < c5_now - 1e-14:
                h = h_try
            else:
                no_improve += 5

    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit, n_cand=25, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit - 5:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(15):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit):
    start_t = time.time()
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    class ConFun:
        def fun(self, x):
            return np.sum(x) - target
        def jac(self, x):
            return np.ones(len(x))

    con = {'type': 'eq', 'fun': ConFun().fun, 'jac': ConFun().jac}

    for alpha in alphas:
        if time.time() - start_t > time_limit - 5:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=[con],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP optimization (primary new method)
    if tleft() > 200:
        lp_time = min(200, tleft() * 0.35)
        h_lp, c5_lp = lp_optimize(best_h, n, target, time_limit=lp_time, n_cons_start=50)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Pair CD refinement
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=25, max_sweeps=80000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP again with more constraints
    if tleft() > 150:
        lp_time = min(150, tleft() * 0.35)
        h_lp, c5_lp = lp_optimize(best_h, n, target, time_limit=lp_time, n_cons_start=100)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP polish
    if tleft() > 60:
        hp, sc = slsqp_polish(best_h, n, target, [500, 2000, 10000], 1500, time_limit=min(50, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Try higher resolution n=800
    if tleft() > 350:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        c5_interp = compute_c5(h_interp)

        h_lp2, c5_lp2 = lp_optimize(h_interp, n_new, target_new, time_limit=min(180, tleft() * 0.4), n_cons_start=50)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2
            n = n_new
            target = target_new

        if tleft() > 120:
            h_cd2, c5_cd2 = fast_pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.4), n_cand=20, max_sweeps=40000)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n = n_new
                target = target_new

    # Phase 6: LP at current resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp3, c5_lp3 = lp_optimize(best_h, n_use, target_use, time_limit=min(100, tleft() * 0.4), n_cons_start=80)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3
            n = n_use
            target = target_use

    # Phase 7: Pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd3, c5_cd3 = fast_pair_cd(best_h, n_use, target_use, time_limit=min(50, tleft() * 0.5), n_cand=30, max_sweeps=30000)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            n = n_use
            target = target_use

    # Phase 8: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_polish(best_h, n_use, target_use, [5000, 20000, 100000], 1000, time_limit=min(30, tleft() * 0.6))
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_use
            target = target_use

    # Phase 9: Final LP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp4, c5_lp4 = lp_optimize(best_h, n_use, target_use, time_limit=min(25, tleft() * 0.7), n_cons_start=100)
        if c5_lp4 < best_score:
            best_score = c5_lp4
            best_h = h_lp4
            n = n_use
            target = target_use

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
