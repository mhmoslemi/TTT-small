def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h

    for _ in range(50):

        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs >= k) & (idxs - k < n) if k >= 0 else (idxs + k >= 0)
    v2 = (idxs + k >= 0) & (idxs + k < n) if k >= 0 else (idxs < -k)

    #          - h[p+k] if p+k valid (from h[p] being the "second" term)

    # g[p] -= h[p+k] if p+k in [0,n)  [p is the "dest" at position p, and p+k is the "source"]

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    # p-k valid: p in range [max(0,k), min(n, n+k))
    v1 = (idxs - k >= 0) & (idxs - k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])

    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v2] -= h[idxs[v2] + k]
    return g

Now, the main optimization function. Let me think about the alpha schedule for log-sum-exp.

The log-sum-exp approximation of max is:
soft_max_alpha(C) = (1/alpha) * log(sum(exp(alpha * (C - max_C)))) + max_C

As alpha → ∞, this approaches max(C). For finite alpha, it's a smooth approximation.

The gradient of soft_max_alpha is:
(1/alpha) * d/dh log(sum(exp(alpha * (C(k) - max_C)))) 
= sum_k exp(alpha*(C(k)-max_C)) / sum_j exp(alpha*(C(j)-max_C)) * dC(k)/dh

So the gradient is a weighted average of the individual gradients, with weights proportional to exp(alpha*(C-Cmax)).

For the optimization to work well:
- Low alpha: spreads the weight more evenly, may find a good "balanced" solution
- High alpha: focuses on the maximum, which is what we want

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - np.sum(h)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v2] -= h[idxs[v2] + k]
    return g

def softmax_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    C_max = np.max(C)
    shifted = alpha * (C - C_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = C_max + log_sum / alpha
    # Gradient
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for idx in active:
        k = int(idx) - mid
        grad += weights[idx] * get_grad_shift(h, k, n)
    grad *= dx
    return val, grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

# Global state for SLSQP
_G = {}

def _s_obj(h):
    return softmax_obj_and_grad(h, _G['alpha'])[0]

def _s_grad(h):
    return softmax_obj_and_grad(h, _G['alpha'])[1]

def _s_cons_fun(h):
    return np.sum(h) - _G['target']

def _s_cons_jac(h):
    return np.ones(_G['n'])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    _G['n'] = n
    _G['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_cons_fun, 'jac': _s_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        _G['alpha'] = alpha
        try:
            res = minimize(_s_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=_s_grad,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5
