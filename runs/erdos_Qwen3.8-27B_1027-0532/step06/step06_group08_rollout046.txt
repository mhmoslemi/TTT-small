∑_{s: c_s > 0} c_s * (something) = ∑_{s: c_s < 0} |c_s| * (something)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff < 0:
        remaining = -diff
        order = np.argsort(h)[::-1]
        for idx in order:
            if remaining <= 1e-13:
                break
            if h[idx] >= remaining:
                h[idx] -= remaining
                remaining = 0.0
            else:
                remaining -= h[idx]
                h[idx] = 0.0
    else:
        remaining = diff
        order = np.argsort(h)
        for idx in order:
            if remaining <= 1e-13:
                break
            cap = 1.0 - h[idx]
            if cap >= remaining:
                h[idx] += remaining
                remaining = 0.0
            else:
                remaining -= cap
                h[idx] = 1.0
    return h

def project_binary(h, target):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    cs = int(h_r.sum())
    if cs == target:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if cs > target:
        need = cs - target
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target - cs
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_smooth_grad(h, corr, alpha, topk):
    n = len(h)
    mc = corr.max()
    lw = alpha * (corr - mc)
    lw = np.clip(lw, -500.0, 0.0)
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= ws
    topk_idx = np.argsort(w)[-topk:]
    grad = np.zeros(n)
    for m in topk_idx:
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wm * (1.0 - h[:n - s])
            grad[:n - s] -= wm * h[s:]
        else:
            r = -s
            grad[:n - r] += wm * (1.0 - h[r:])
            grad[n - r:] -= wm * h[:r]
    return grad

def nesterov_optimize(h_init, target, alphas, iters_per_alpha, lr_init, lr_end,
                      momentum, topk, time_limit, start_time):

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    mom = np.zeros(n)
    stall = 0
    max_stall = 20000
    n_alphas = len(alphas)
    for ai in range(n_alphas):
        alpha = alphas[ai]
        lr = lr_init * (lr_end / lr_init) ** (float(ai) / max(n_alphas - 1, 1))
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            if stall > max_stall:
                break
            grad = compute_smooth_grad(h, corr, alpha, topk)
            mom = momentum * mom + grad
            h = h - lr * mom
            h = project_box_sum(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur = float(corr.max()) * dx
            if cur < best_c5 - 1e-12:

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if time.time() - start_time > time_limit:
            break

            break
    return best_h, best_c5

                h2 = h.copy()
                h2[int(i1)] = 0.0
                h2[int(j1)] = 1.0
                delta2 = compute_delta(h2, int(i2), int(j2))
                new_corr = corr + delta1 + delta2
                new_max = float(new_corr.max())
                dc5 = new_max * dx - cur_c5
                if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
                    h[int(i1)] = 0.0
                    h[int(j1)] = 1.0
                    h[int(i2)] = 0.0
                    h[int(j2)] = 1.0
                    corr = new_corr
                    cur_c5 = new_max * dx
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
                continue
        if it == -1:
            continue
        i = int(i)
        j = int(j)
        delta = compute_delta(h, i, j)
        new_max = float((corr + delta).max())
        dc5 = new_max * dx - cur_c5
        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            cur_c5 = new_max * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(x, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        cm = float(np.max(c))
        sh = np.clip(alpha * (c - cm), -400.0, 0.0)
        return cm + float(np.log(np.sum(np.exp(sh)))) / alpha

    h_cur = h_start.copy()
    best = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_bin = project_binary(h_init, int(n_init // 2))
    best_bin_score = compute_c5(best_bin)
    best_cont = project_box_sum(h_init, target_init)
    best_cont_score = compute_c5(best_cont)

    # Phase 1: Binary greedy (60s)
    allowed = min(60.0, tleft() * 0.08)
    if allowed > 15:
        hb, sb = binary_greedy_diverse(best_bin, int(n_init // 2), 80000, rng,
                                       allowed, start_time)
        if sb < best_bin_score:
            best_bin_score = sb
            best_bin = hb.copy()

    # Phase 2: Binary SA (150s)
    allowed = min(150.0, tleft() * 0.15)
    if allowed > 30:
        n_sa = int(min(6000000, allowed * 50000))
        hs, ss = binary_sa_diverse(best_bin, int(n_init // 2), n_sa,
                                   0.015, 0.00003, rng, allowed, start_time)
        if ss < best_bin_score:
            best_bin_score = ss
            best_bin = hs.copy()

    # Phase 3: Continuous Nesterov - main (500s)
    allowed = min(500.0, tleft() * 0.40)
    if allowed > 60:
        alphas = [5.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
        iters = int(min(200000, allowed * 400))
        hc, sc = nesterov_optimize(best_cont, target_init, alphas, iters,
                                   lr_init=0.08, lr_end=0.001,
                                   momentum=0.9, topk=80,
                                   time_limit=allowed, start_time=start_time)
        if sc < best_cont_score:
            best_cont_score = sc
            best_cont = hc.copy()

        # Back to binary from continuous
        if tleft() > 50:
            hb2 = project_binary(best_cont, int(n_init // 2))
            sb2 = compute_c5(hb2)
            if sb2 < best_bin_score:
                best_bin_score = sb2
                best_bin = hb2.copy()

    # Phase 4: Second Nesterov pass with different params (200s)
    allowed = min(200.0, tleft() * 0.18)
    if allowed > 40:
        alphas2 = [20.0, 80.0, 200.0, 500.0, 1000.0]
        iters2 = int(min(150000, allowed * 500))
        hc2, sc2 = nesterov_optimize(best_cont, target_init, alphas2, iters2,
                                     lr_init=0.05, lr_end=0.0005,
                                     momentum=0.92, topk=50,
                                     time_limit=allowed, start_time=start_time)
        if sc2 < best_cont_score:
            best_cont_score = sc2
            best_cont = hc2.copy()

    # Phase 5: Binary refinement (80s)
    allowed = min(80.0, tleft() * 0.08)
    if allowed > 20:
        hb3 = project_binary(best_cont, int(n_init // 2))
        sb3 = compute_c5(hb3)
        if sb3 < best_bin_score:
            best_bin_score = sb3
            best_bin = hb3.copy()
        n_sa3 = int(min(3000000, allowed * 60000))
        hs3, ss3 = binary_sa_diverse(best_bin, int(n_init // 2), n_sa3,
                                     0.01, 0.00002, rng, allowed, start_time)
        if ss3 < best_bin_score:
            best_bin_score = ss3
            best_bin = hs3.copy()

    # Phase 6: Multi-resolution (200s)
    for n_try in [n_init + 4, n_init + 8, n_init + 16, n_init + 32]:
        if tleft() < 80:
            break
        if n_try % 2 != 0 or n_try < 200:
            continue
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_cont, n_try, target_try)
        allowed6 = min(40.0, tleft() * 0.15)
        if allowed6 < 20:
            continue
        alphas6 = [20.0, 100.0, 400.0]
        iters6 = int(min(80000, allowed6 * 2000))
        hc6, sc6 = nesterov_optimize(h_interp, target_try, alphas6, iters6,
                                     lr_init=0.06, lr_end=0.001,
                                     momentum=0.9, topk=60,
                                     time_limit=allowed6, start_time=start_time)
        if sc6 < best_cont_score:
            best_cont_score = sc6
            best_cont = hc6.copy()

    # Phase 7: SLSQP polish (100s)
    allowed = min(100.0, tleft() * 0.10)
    if allowed > 30:
        n_use = len(best_cont)
        h_ref, sc_ref = slsqp_refine(best_cont, float(n_use) / 2.0,
                                     alphas=[100, 500, 2000],
                                     maxiter=3000, ftol=1e-14,
                                     time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont = h_ref.copy()

    # Phase 8: Final Nesterov burst
    if tleft() > 40:
        n_use = len(best_cont)
        target_use = float(n_use) / 2.0
        allowed8 = max(20.0, tleft() - 15.0)
        alphas8 = [200.0, 500.0, 1000.0]
        iters8 = int(min(100000, allowed8 * 3000))
        hc8, sc8 = nesterov_optimize(best_cont, target_use, alphas8, iters8,
                                     lr_init=0.03, lr_end=0.0001,
                                     momentum=0.93, topk=40,
                                     time_limit=allowed8, start_time=start_time)
        if sc8 < best_cont_score:
            best_cont_score = sc8
            best_cont = hc8.copy()

    # Final selection
    n_final = len(best_cont)
    target_final = float(n_final) / 2.0
    h_final_cont = project_box_sum(best_cont, target_final)
    final_cont_score = compute_c5(h_final_cont)

    # Also try binary at final resolution
    if len(best_bin) == n_final:
        h_final_bin = project_binary(best_bin, int(n_final // 2))
    else:
        h_final_bin = project_binary(h_final_cont, int(n_final // 2))
    final_bin_score = compute_c5(h_final_bin)

    if final_bin_score < final_cont_score:
        return h_final_bin, final_bin_score, n_final
    else:
        return h_final_cont, final_cont_score, n_final
