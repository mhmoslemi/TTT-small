Actually, let me think about this more carefully. The problem is:
- minimize max_k f_k(h) subject to g(h) = 0, 0 <= h <= 1

where f_k(h) = ∫ h(x)(1-h(x+k)) dx.

A common approach for min-max is to use the Dinkelbach method or parametric approach:
- For a given C, check if there exists h such that f_k(h) <= C * n/2 for all k, sum(h) = n/2, 0<=h<=1
- This is a feasibility problem (LP if h is not in the objective, but here it's quadratic)

Actually, f_k(h) = sum_i h[i]*(1-h[i+k-shifted]) is bilinear in h, so it's not linear. This makes the feasibility problem non-convex.

But we can linearize: for a fixed C, the constraint f_k(h) <= C*n/2 is:
sum_i h[i] - sum_i h[i]*h[i+k'] <= C*n/2

The term sum_i h[i]*h[i+k'] is quadratic. We can bound it from below (since we want the upper bound on the left side):
Actually, we want sum_i h[i]*(1-h[i+k']) <= C*n/2
=> sum_i h[i] - sum_i h[i]*h[i+k'] <= C*n/2
=> sum_i h[i] * h[i+k'] >= sum_i h[i] - C*n/2 = n/2 - C*n/2 = n(1-C)/2

So we need: min_k sum_i h[i]*h[i+k'] >= n(1-C)/2

This is still non-convex (maximizing a minimum of quadratic functions).

For a convex relaxation, we could use: h[i]*h[i+k'] >= 0 (trivial) or use the McCormick envelope for the product h[i]*h[i+k'] given 0<=h<=1.

But this might not be tight enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def enforce_symmetry(h):
    return 0.5 * (h + h[::-1])

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad *= dx
    return grad

def topk_smooth_obj(h, alpha, top_k):
    """Smoothing that only considers top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_idx = np.argsort(corr)[-top_k:]
    corr_top = corr[top_idx]
    c_max_top = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max_top), -500.0, 0.0)
    return float(np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def symmetric_slsqp(h_start, n, target, alphas, maxiter, ftol, time_limit, top_k=None):
    """SLSQP on symmetric reduced variables."""
    M = n // 2
    x0 = h_start[:M].copy()
    bounds = [(0.0, 1.0)] * M
    
    def get_full(x):
        h = np.zeros(n)
        h[:M] = x
        h[M:] = x[::-1]
        h = project_feasible(h, target)
        h = enforce_symmetry(h)
        h = project_feasible(h, target)
        return h
    
    start_t = time.time()
    h_cur = h_start.copy()
    
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        
        def obj(x):
            h = get_full(x)
            return smooth_obj(h, alpha)
        
        def grad(x):
            h = get_full(x)
            g = smooth_grad(h, alpha)
            g_sym = 0.5 * (g + g[::-1])
            return g_sym[:M]
        
        def con_fun(x):
            h = get_full(x)
            return np.sum(h) - target
        
        cons = ({'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(M)})
        
        try:
            res = minimize(obj, x0, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = get_full(res.x)
            if compute_c5(h_new) <= compute_c5(h_cur) + 1e-14:
                h_cur = h_new
                x0 = h_cur[:M].copy()
        except Exception:
            pass
    
    return h_cur

def subgradient_descent(h_start, n, target, time_limit, alpha_param, eta0, decay):
    """Subgradient descent on smoothed objective."""
    start_t = time.time()
    h = h_start.copy()
    M = n // 2
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t = 0
    while time.time() - start_t < time_limit:
        t += 1
        eta = eta0 / (1.0 + decay * t)
        g = smooth_grad(h, alpha_param)
        g = 0.5 * (g + g[::-1])  # Symmetrize
        g -= np.mean(g)  # Remove mean (for sum constraint)
        
        h_try = h - eta * g
        h_try = project_feasible(h_try, target)
        h_try = enforce_symmetry(h_try)
        h_try = project_feasible(h_try, target)
        
        c = compute_c5(h_try)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_try.copy()
        h = h_try
    
    return best_h, best_c5

def make_ears_pattern(n, center_width=200):
    """Create a symmetric 'ears' pattern: 1 near edges, 0 in center."""

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:

        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        top_idx = np.argsort(corr)[-top_k:]
        correction = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - (n - 1)
            g = np.zeros(n)
            valid1 = (idxs - k >= 0) & (idxs - k < n)
            g[valid1] += (1.0 - h[idxs[valid1] - k])
            valid2 = (idxs + k >= 0) & (idxs + k < n)
            g[valid2] -= h[idxs[valid2] + k]
            correction += (-g)
        correction = 0.5 * (correction + correction[::-1])
        correction -= np.mean(correction)
        norm = np.linalg.norm(correction)
        if norm < 1e-12:
            break
        correction = correction / norm * step
        best_accepted = False
        for trial_scale in [1.0, 0.5, 0.25, 0.1]:
            h_try = sym_project(h + trial_scale * correction, target)
            c_try = compute_c5(h_try)
            if c_try < c5_val - 1e-14:
                h = h_try
                step = min(step * 1.05, 0.1)
                best_accepted = True
                break
        if not best_accepted:
            step *= 0.5
            if step < 1e-5:
                break
    return best_h, best_c5

def binary_push_sym(h, n, target, time_limit, max_tries):
    start_t = time.time()
    M = n // 2
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(99999)
    for t in range(max_tries):
        if time.time() - start_t > time_limit:
            break
        i = int(rng.randint(M))
        target_val = rng.choice([0.0, 1.0])
        h_try = h.copy()
        h_try[i] = target_val
        h_try[n - 1 - i] = target_val
        h_try = sym_project(h_try, target)
        c = compute_c5(h_try)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_h = h_try.copy()
            h = h_try
    return best_h, best_c5

def subgradient_sym(h_start, n, target, time_limit, alpha_param, eta0=0.02, decay=0.003):
    start_t = time.time()
    h = sym_project(h_start.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    while time.time() - start_t < time_limit:
        t += 1
        eta = eta0 / (1.0 + decay * t)
        g = smooth_grad(h, alpha_param)
        g = 0.5 * (g + g[::-1])
        g -= np.mean(g)
        h_try = sym_project(h - eta * g, target)
        c = compute_c5(h_try)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_try.copy()
        h = h_try
    return best_h, best_c5

def make_ears(n, gap):
    h = np.zeros(n)
    edge = (n - gap) // 2
    h[:edge] = 1.0
    h[n - edge:] = 1.0
    return sym_project(h, n / 2.0)

def make_center(n, width):
    h = np.zeros(n)
    s = (n - width) // 2
    h[s:s + width] = 1.0
    return sym_project(h, n / 2.0)

def make_sinusoid(n, freq, phase):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return sym_project(h, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return sym_project(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return sym_project(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)

    if tleft() > 80:
        h8 = symmetric_slsqp(best_h, n, target,
                             [10000, 50000], 1500, 1e-16, min(50, tleft()))
        s8 = compute_c5(h8)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    rng = np.random.RandomState(seed)
    if tleft() > 150:
        patterns = [make_ears(n, 120), make_ears(n, 200), make_center(n, 200),
                    make_sinusoid(n, 1.0, 0.0), make_sinusoid(n, 2.0, 0.5)]
        for pat in patterns:
            if tleft() < 40:

            h_p = symmetric_slsqp(pat, n, target, [50, 500, 5000], 1000, 1e-14, min(25, tleft() * 0.25))
            s_p = compute_c5(h_p)

    if tleft() > 80:

    return best_h, final_score, n_final
