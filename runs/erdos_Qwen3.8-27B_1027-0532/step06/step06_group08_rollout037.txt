def continuous_optimize(h_start, n, time_limit, start_time):
    """Continuous optimization using subgradient method"""
    dx = 2.0 / n
    target = n / 2.0
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    lr = 0.02
    stall = 0
    
    while time.time() - start_time < time_limit:
        if stall > 3000:
            break
            
        # Find active set (shifts with near-max correlation)
        max_corr = float(corr.max())
        threshold = max_corr - 0.01 * max_corr
        active = np.where(corr >= threshold)[0]
        
        # Subgradient: average of gradients over active set
        # c_s = sum_t h[t]*(1-h[t+s]) for shift s = shifts[k]
        # dc_s/dh[t] = (1-h[t+s]) - h[t-s] (for valid indices)
        
        grad = np.zeros(n)
        for k in active:
            s = shifts[k]
            if s >= 0:
                # c_s = sum_{t=s}^{n-1} h[t]*(1-h[t-s])
                # d/dh[t] for t in [s, n-1]: (1-h[t-s])
                # d/dh[t] for t in [0, n-1-s]: -h[t+s]
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
            else:
                a = -s
                # c_s = sum_{t=0}^{n-1-a} h[t]*(1-h[t+a])
                grad[:n-a] += (1.0 - h[a:])
                grad[a:] += (-h[:n-a])
        
        grad /= len(active)
        
        # Update with backtracking
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        new_c5 = float(corr_new.max()) * dx
        
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            stall = 0
            # Slowly decrease lr
            lr *= 0.999
        else:
            h = h_new  # still move, but don't update best
            corr = corr_new
            stall += 1
            lr *= 0.99
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

                flip_o = ones[rng.choice(len(ones), nf, replace=False)]
            flip_z = zeros[rng.choice(len(zeros), nf, replace=False)]
            
            h_new = h.copy()
            h_new[flip_o] = 0.0
            h_new[flip_z] = 1.0
            full_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            nm = float(full_new.max())
            c5_new = nm * dx
            dc = c5_new - cur_c5

            if dc < 0 or (t > 1e-15 and rng.random() < np.exp(-dc / t)):
                h = h_new
                full = full_new
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Single flip
            if ks != 0 and rng.random() < 0.7:
                A = np.zeros(n)
                idx = positions + ks
                v = (idx >= 0) & (idx < n)
                A[v] += h[idx[v]]
                idx = positions - ks
                v = (idx >= 0) & (idx < n)
                A[v] += h[idx[v]]
                # Prefer low A (positions whose neighbors at shift ks are both 0)
                scores = A[ones]
                if t > 0.005:
                    w = np.exp(-scores / (2.0 * t + 1e-10))
                    w /= w.sum()
                    i = ones[rng.choice(len(ones), p=w)]
                else:
                    i = ones[np.argmin(scores)]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            
            delta = compute_delta_single(h, i, j)
            nm = float(np.max(full + delta))
            c5_new = nm * dx
            dc = c5_new - cur_c5

            if dc < 0 or (t > 1e-15 and rng.random() < np.exp(-dc / t)):
                h[i] = 0.0
                h[j] = 1.0
                full = full + delta
                cur_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def smooth_slsqp(h_start, n, target, alpha_seq, maxiter, ftol, time_limit, start_time):
    """Run SLSQP with a sequence of smoothing parameters"""
    bounds = [(0.0, 1.0)] * n

    def obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr))
        shifted = np.clip(alpha * (corr - cmax), -400.0, 0.0)
        return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)

    for alpha in alpha_seq:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_cur, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def subgrad_continuous(h_start, n, time_limit, start_time, lr_init=0.03, lr_min=0.001):
    """Subgradient descent with adaptive step size"""
    dx = 2.0 / n
    target = n / 2.0
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)
    lr = lr_init
    stall = 0

    while time.time() - start_time < time_limit:
        if stall > 5000:
            break

        max_corr = float(corr.max())
        # Active set: shifts within 5% of max
        thresh = max_corr - 0.05 * max_corr
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in active:
            s = shifts[k]
            if s >= 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
            else:
                a = -s
                grad[:n-a] += (1.0 - h[a:])
                grad[a:] += (-h[:n-a])
        grad /= len(active)

        h_new = project_box_sum(h - lr * grad, target)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        nc = float(corr_new.max()) * dx

        if nc < best_c5 - 1e-13:
            best_c5 = nc
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            stall = 0
        else:
            corr = corr_new
            h = h_new
            stall += 1

        # Adaptive lr
        if stall % 100 == 0:
            lr = max(lr_min, lr * 0.95)

    return best_h, best_c5

def make_diverse_inits(n, seed):
    """Generate diverse binary initializations of length n with n/2 ones"""
    target = n // 2
    rng = np.random.RandomState(seed)
    inits = []

    # 1. Random
    h = np.zeros(n)
    ones_pos = rng.choice(n, target, replace=False)
    h[ones_pos] = 1.0
    inits.append(h.copy())

    # 2. Beatty with golden ratio
    alpha = (1.0 + 5.0**0.5) / 2.0
    frac = np.arange(n) * alpha - np.floor(np.arange(n) * alpha)
    h = (frac < 0.5).astype(float)
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(-frac[ones])[:cur - target]
        h[ones[order]] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(frac[zeros])[:target - cur]
        h[zeros[order]] = 1
    inits.append(h.copy())

    # 3. Beatty with sqrt(2)
    alpha = 2.0**0.5
    frac = np.arange(n) * alpha - np.floor(np.arange(n) * alpha)
    h = (frac < 0.5).astype(float)
    cur = int(h.sum())
    if cur > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(-frac[ones])[:cur - target]
        h[ones[order]] = 0
