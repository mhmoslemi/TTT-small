        return c_max + float(np.log(shifted.sum())) / alpha

    c_max = float(corr.max())
    shifted = np.exp(alpha * (corr - c_max))
    return c_max + float(np.log(shifted.sum())) / alpha

def smooth_grad(h, alpha):
    corr = full_corr_fast(h)
    c_max = float(corr.max())
    shifted = np.exp(alpha * (corr - c_max))
    w = shifted / shifted.sum()
    n = len(h)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    return 1.0 - 2.0 * conv

def nesterov_continuous(h0, target, alpha, deadline):
    n = len(h0)
    h = project_box_sum(h0.copy().astype(float), target)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_smooth = smooth_obj(h, alpha)

    grad0 = smooth_grad(h, alpha)
    grad0 -= grad0.mean()
    gnorm = float(np.dot(grad0, grad0))
    if gnorm < 1e-30:
        return best_h, best_c5

    lr = min(1.0 / max(alpha, 1.0) / 2.0, 5.0 / max(np.sqrt(gnorm), 1e-8))
    lr = max(lr, 1e-10)
    momentum = 0.85
    iter_count = 0

    while time.time() < deadline:
        y = h + momentum * v
        y = project_box_sum(y, target)

        grad = smooth_grad(y, alpha)
        grad -= grad.mean()
        f_y = smooth_obj(y, alpha)
        grad_norm_sq = float(np.dot(grad, grad))

        if grad_norm_sq < 1e-30:
            break

        step = lr
        accepted = False
        for _ in range(12):
            h_new = project_box_sum(y - step * grad, target)
            f_new = smooth_obj(h_new, alpha)
            if f_new <= f_y - 0.01 * step * grad_norm_sq:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            h_new = project_box_sum(y - step * grad, target)
            f_new = smooth_obj(h_new, alpha)
            if f_new > f_y + 0.1:
                momentum = max(0.3, momentum * 0.8)
                lr = max(1e-12, lr * 0.5)
                h_new = h.copy()
                f_new = f_y

        v = h_new - h
        h = h_new
        iter_count += 1

        if iter_count % 50 == 0 or f_new < best_smooth:
            if f_new < best_smooth:
                best_smooth = f_new
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()

    return best_h, best_c5

def slsqp_smooth(x, alpha):
    corr = full_corr_fast(x)
    c_max = float(np.max(corr))
    shifted = np.exp(alpha * (corr - c_max))
    f = c_max + float(np.log(shifted.sum())) / alpha
    w = shifted / shifted.sum()
    n = len(x)
    N_fft = 1
    while N_fft < 3 * n:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = x
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    g = 1.0 - 2.0 * conv
    g -= g.mean()
    return f, g

def slsqp_refine(h_start, n, target, alphas, max_iter, deadline):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    h_cur = project_box_sum(h_start.copy().astype(float), target)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-12:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def generate_smooth_init(n, rng):
    x = np.linspace(0, 1, n)
    h = np.zeros(n)
    for _ in range(5):
        freq = rng.randint(2, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        h += amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = 0.5 + h
    return project_box_sum(h, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy().astype(float), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    abs_deadline = start_time + budget_s - 12.0
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_h = project_box_sum(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    alphas_phase1 = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 5000.0, 20000.0, 100000.0]
    n_a1 = len(alphas_phase1)
    phase1_end = start_time + min(450.0, (budget_s - 12.0) * 0.45)
    time_per_a1 = (phase1_end - start_time) / n_a1

    for i, a in enumerate(alphas_phase1):
        dl = min(phase1_end, abs_deadline)
        if time.time() > dl:
            break
        t_end = min(start_time + (i + 1) * time_per_a1, dl)
        h_new, sc = nesterov_continuous(best_h, target_init, a, t_end)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    if time.time() < abs_deadline - 30:
        phase2_end = min(start_time + (budget_s - 12.0) * 0.7, abs_deadline)
        time_per_phase2 = max(5.0, (phase2_end - time.time()) / 4.0)
        alphas_phase2 = [500.0, 5000.0, 50000.0, 200000.0]
        for a in alphas_phase2:
            if time.time() > phase2_end:
                break
            t_end = min(time.time() + time_per_phase2, phase2_end, abs_deadline)
            h_new, sc = nesterov_continuous(best_h, target_init, a, t_end)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()

    if time.time() < abs_deadline - 10:
        sl_deadline = min(time.time() + 100.0, abs_deadline)
        alphas_sl = [2000.0, 20000.0, 200000.0]
        h_ref, sc_ref = slsqp_refine(best_h, n_init, target_init, alphas_sl, 3000, sl_deadline)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    if time.time() < abs_deadline - 30:
        h_sm = generate_smooth_init(n_init, rng)
        sm_end = min(time.time() + 60.0, abs_deadline)
        alphas_sm = [10.0, 100.0, 1000.0, 10000.0, 100000.0]
        n_sm = len(alphas_sm)
        for i, a in enumerate(alphas_sm):
            if time.time() > sm_end:
                break
            t_end = min(time.time() + (sm_end - time.time()) / (n_sm - i), sm_end)
            h_new, sc = nesterov_continuous(h_sm, target_init, a, t_end)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_sm = h_new.copy()

    if time.time() < abs_deadline - 15:
        for n_try in [n_init + 2, n_init + 4, n_init - 2, n_init - 4]:
            if time.time() > abs_deadline - 10:
                break
            if n_try < 50 or n_try % 2 != 0:
                continue
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            alphas_try = [500.0, 5000.0, 50000.0]
            t_end = min(time.time() + 15.0, abs_deadline)
            n_try_a = len(alphas_try)
            for i, a in enumerate(alphas_try):
                if time.time() > t_end:
                    break
                t_end_i = min(time.time() + (t_end - time.time()) / (n_try_a - i), t_end)
                h_new, sc = nesterov_continuous(h_interp, target_try, a, t_end_i)
                h_interp = h_new.copy()
            sc_final_try = compute_c5(h_interp)
            if sc_final_try < best_score:
                best_score = sc_final_try
                best_h = h_interp.copy()

    if time.time() < abs_deadline - 5:
        final_end = abs_deadline
        alphas_final = [10000.0, 100000.0, 500000.0]
        n_fin = len(alphas_final)
        for i, a in enumerate(alphas_final):
            if time.time() > final_end - 1:
                break
            t_end = min(time.time() + max(3.0, (final_end - time.time()) / (n_fin - i)), final_end)
            h_new, sc = nesterov_continuous(best_h, float(len(best_h)) / 2.0, a, t_end)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
