def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    weights = np.exp(shifted)

    _g_target = target

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                       [100, 500, 2000], 600, 1e-15)
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 150:
            h1_cd, c5_1cd = fast_pair_cd(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.6),
                                         max_sweeps=25000,
                                         n_cand_start=20, n_cand_end=80,
                                         max_no_improve=500)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 60:
            h1_gd, c5_1gd = gradient_descent_active(h1, n1, target1,
                                                    time_limit=min(40, tleft() * 0.4),
                                                    max_steps=30000, base_lr=0.02,
                                                    n_active=6)
            if c5_1gd < best_score:
                best_score = c5_1gd
                best_h = h1_gd
                best_n = n1
                best_target = target1

    n2 = 800
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 280:
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [200, 1000, 5000], 800, 1e-15)
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 200:
            h2_cd, c5_2cd = fast_pair_cd(h2, n2, target2,
                                         time_limit=min(250, tleft() * 0.55),
                                         max_sweeps=30000,
                                         n_cand_start=25, n_cand_end=90,
                                         max_no_improve=500)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_gd, c5_2gd = gradient_descent_active(h2, n2, target2,
                                                    time_limit=min(60, tleft() * 0.4),
                                                    max_steps=40000, base_lr=0.015,
                                                    n_active=10)
            if c5_2gd < best_score:
                best_score = c5_2gd
                best_h = h2_gd
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_cd2, c5_2cd2 = multi_active_cd(h2, n2, target2,
                                              time_limit=min(60, tleft() * 0.5),
                                              n_cand=50, n_active=5)
            if c5_2cd2 < best_score:
                best_score = c5_2cd2
                best_h = h2_cd2

    if tleft() > 80:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [10000, 50000, 100000], 2000, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 100:
        h_cd_final, c5_final = fast_pair_cd(best_h, best_n, best_target,
                                            time_limit=min(120, tleft() * 0.6),
                                            max_sweeps=20000,
                                            n_cand_start=40, n_cand_end=100,
                                            max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 60:
        h_gd_final, c5_gf = gradient_descent_active(best_h, best_n, best_target,
                                                    time_limit=min(40, tleft() * 0.5),
                                                    max_steps=30000, base_lr=0.01,
                                                    n_active=12)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gd_final

    if tleft() > 40:
        h_pol3, sc3 = slsqp_round(best_h, best_n, best_target,
                                  [50000, 200000], 1000, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 30:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 20:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = slsqp_round(h_pert, best_n, best_target,
                                    [500, 5000, 50000], 500, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 20:
        h_cd_last, c5_last = fast_pair_cd(best_h, best_n, best_target,
                                          time_limit=min(15, tleft() * 0.5),
                                          max_sweeps=5000,
                                          n_cand_start=50, n_cand_end=100,
                                          max_no_improve=100)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
