import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    m = np.max(scaled)
    weights = np.exp(scaled - m) / np.sum(np.exp(scaled - m))
    obj = (m + np.log(np.sum(np.exp(scaled - m)))) / alpha
    mid = n - 1
    idxs = np.arange(n)
    active = np.where(weights > 1e-10)[0]
    if len(active) > 30:
        order = np.argsort(weights[active])[::-1][:30]
        active = active[order]
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - mid
        w_k = dx * weights[k_idx]
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        term = v1.astype(float)
        if np.any(v1):
            term[v1] -= h[idxs[v1] + k]
        if np.any(v2):
            term[v2] -= h[idxs[v2] - k]
        grad += w_k * term
    grad -= np.mean(grad)
    return obj, grad


def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        a = alpha
        try:
            def obj_wrapper(x):
                o, _ = smooth_obj_and_grad(x, a)
                return o
            def jac_wrapper(x):
                _, g = smooth_obj_and_grad(x, a)
                return g
            res = minimize(obj_wrapper, h_cur, jac=jac_wrapper,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def lbfgsb_smooth(h_start, n, target, alphas, maxiter_total=5000):
    h_cur = h_start.copy()
    total_iter = 0
    for alpha in alphas:
        remain = maxiter_total - total_iter
        if remain <= 0:
            break
        a = alpha
        def obj_wrapper(x):
            o, _ = smooth_obj_and_grad(x, a)
            return o
        def jac_wrapper(x):
            _, g = smooth_obj_and_grad(x, a)
            return g
        try:
            res = minimize(obj_wrapper, h_cur, jac=jac_wrapper,
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': remain, 'ftol': 1e-14, 'gtol': 1e-8})
            total_iter += res.nit
            h_new = project_feasible(res.x, target)
            h_cur = h_new
        except Exception:
            break
    return h_cur


def projected_momentum(h_start, n, target, alpha, max_steps=5000,
                        lr_init=0.01, momentum=0.9, decay=0.999):
    h = h_start.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for step in range(max_steps):
        obj, grad = smooth_obj_and_grad(h, alpha)
        vel = momentum * vel - lr * grad
        h_try = project_feasible(h + vel, target)
        obj_try, _ = smooth_obj_and_grad(h_try, alpha)
        if obj_try < obj + 1e-15:
            h = h_try
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
        if step % 100 == 0:
            lr = max(lr * decay, 1e-8)
    return best_h, best_c5


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 100:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(len(shifts))
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 40:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_smooth(h_interp, n_try, target_try, [500, 2000, 8000], 600, 1e-14)
            sc_hires = compute_c5(h_hires)
            if tleft() > 80 and sc_hires < best_score + 0.002:
                cd_t = min(100, tleft() * 0.3)
                h_hires2, sc2 = fast_pair_cd(h_hires, n_try, target_try,
                                              time_limit=cd_t,
                                              max_sweeps=3000, n_cand=45, n_bisect=14)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 10: L-BFGS-B final push with very high alpha
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lb = lbfgsb_smooth(best_h, n_use, target_use, [10000, 30000, 100000], maxiter_total=5000)
        sc = compute_c5(h_lb)
        if sc < best_score:
            best_score = sc
            best_h = h_lb

    # Phase 11: Final multi-active CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.5)
        h_cd, c5_cd = multi_active_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=3000, n_cand=60,
                                       n_bisect=15, n_active=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_smooth(best_h, n_use, target_use, [20000, 100000, 300000], 600, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, len(h_out)
