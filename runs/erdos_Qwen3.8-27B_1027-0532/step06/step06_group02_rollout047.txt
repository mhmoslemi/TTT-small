def compute_c5(h):

def full_corr_fast(h):
    n = len(h)

    a[:n] = h
    b = np.zeros(N)

    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    result = np.zeros(2 * n - 1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k_arr
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum

    t = np.zeros(n, dtype=np.float64)

def gen_period_doubling(n):
    p = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        if i % 2 == 0:
            p[i] = p[i // 2]
        else:
            p[i] = 1.0 - p[i // 2]
    return p

def gen_walsh_row(n, k):
    out = np.zeros(n, dtype=np.float64)
    for i in range(n):
        v = i ^ k
        c = 0
        while v:
            c += v & 1
            v >>= 1
        out[i] = 1.0 if c % 2 == 0 else 0.0
    return out

def continuous_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(40.0, time_limit - (time.time() - start_time))
        if phase_time <= 0:
            break
        lr = 1.0 / np.sqrt(alpha + 1.0)
        momentum = 0.75
        v = np.zeros(n)
        n_iter = max(int(phase_time * 250), 80)
        N_fft = 1
        while N_fft < 3 * n - 2:
            N_fft <<= 1
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = float(corr.max())
            shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
            exp_sum = float(np.sum(np.exp(shifted)))
            w = np.exp(shifted) / max(exp_sum, 1e-30)
            W_fft = np.fft.fft(w, N_fft)
            h_pad = np.zeros(N_fft)
            h_pad[:n] = h
            h_fft = np.fft.fft(h_pad)
            conv_full = np.fft.ifft(W_fft * h_fft).real
            conv = conv_full[:n]
            grad = 1.0 - 2.0 * conv
            step_scale = min(1.0, 0.5 / (np.linalg.norm(grad) + 0.01))
            v = momentum * v - lr * step_scale * grad
            h_new = h + v
            h = project_box_sum(h_new, target)
            if it % 30 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _slsqp_target

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

            h_best = hfg.copy()
        if tleft() > 40:
            allowed4 = min(20, tleft() * 0.05)
            n_saf = int(min(400000, allowed4 * 30000))
            hfs, c5fs = sa_binary(hfg, n_saf, 0.008, 0.00003, rng,
                                  time_limit=allowed4, start_time=start_time,
                                  k_arr=k_arr, block_frac=0.2, max_block_len=8)
            if c5fs < best_score:
                best_score = c5fs
                h_best = hfs.copy()

    if tleft() > 50:
        n_use = len(h_best)
        target_use = float(n_use) / 2.0
        alphas_slsqp = [50, 200, 800, 3000, 12000, 50000]
        allowed = min(tleft() - 25, 180)
        h_ref, sc = slsqp_refine(h_best, n_use, target_use, alphas_slsqp,
                                 3000, 1e-14, time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            h_best = h_ref.copy()

    if tleft() > 40:
        n_use = len(h_best)
        target_use = float(n_use) / 2.0
        alphas2 = [30, 100, 500, 2000, 10000, 50000]
        allowed = min(tleft() - 20, 100)
        h_ref2, sc2 = continuous_refine(h_best, target_use, start_time, allowed, alphas2)
        if sc2 < best_score:
            best_score = sc2
            h_best = h_ref2.copy()

    if tleft() > 25:
        h_bin_final = project_binary(h_best, target)
        n_sa_f = int(min(600000, max(30000, (tleft() - 12) * 35000)))
        hsf, c5f = sa_binary(h_bin_final, n_sa_f, 0.005, 0.00001, rng,
                             time_limit=max(3.0, tleft() - 10),
                             start_time=start_time, k_arr=k_arr,
                             block_frac=0.1, max_block_len=6)
        if c5f < best_score:
            best_score = c5f
            h_best = hsf.copy()

    if tleft() > 15:
        n_use = len(h_best)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(h_best, n_use, target_use,
                                 [10000, 50000, 200000], 2000, 1e-15,
                                 time_limit=max(3.0, tleft() - 8),
                                 start_time=start_time)
        if sc < best_score:
            best_score = sc
            h_best = h_ref.copy()

    n_final = len(h_best)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(h_best, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
