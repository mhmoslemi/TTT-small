import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def lbfags_smooth(h_start, n, target, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h_start, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def active_set_balance(h, n, target, M=10, alpha=50000, maxiter=5000):
    """Minimize the max by equalizing the top M correlation lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-M:]
    top_vals = corr[top_idx]
    mean_top = np.mean(top_vals)
    
    # Objective: minimize sum of squared deviations from mean of top M
    # plus penalty for any lag exceeding the mean
    def obj(h_local):
        corr_local = np.correlate(h_local, 1.0 - h_local, mode='full')
        top_vals_local = corr_local[top_idx]
        mean_local = np.mean(top_vals_local)
        spread = np.sum((top_vals_local - mean_local)**2)
        over = np.sum(np.maximum(corr_local - (mean_local + 1e-6), 0.0)**2)
        return (spread + over) * dx * dx
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def multi_lag_gradient_descent(h, n, target, time_limit, max_steps=50000, m_top=10, eta=5.0):
    """Gradient descent using top-M lag weights (game-theoretic approach)."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.02
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(10):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                best_lr = min(lr * 1.2, 0.5)
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            best_lr *= 0.85
            if best_lr < 0.001:
                break
        
        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -40, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = 0.6 * lam + 0.4 * lam_new
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def local_perturbation(h_init, n, target, max_tries, time_limit, seed=777):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):

        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Phase 12: Final pair CD
    if tleft() > 35:
        h_try, sc = pair_cd(best_h, n, target, time_limit=min(25, tleft() * 0.5),
                             max_sweeps=6000, n_cand=60, patience=100)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Phase 13: Local perturbation
    if tleft() > 25:
        h_try, sc = local_perturbation(best_h, n, target,
                                        max_tries=5000, time_limit=min(15, tleft() * 0.5),
                                        seed=seed + 100)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
