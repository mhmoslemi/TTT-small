```python
def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_active(h, alpha, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top-m smoothing
    srt = np.sort(corr)[::-1]
    m = min(top_m, len(srt))
    # Weight: active shifts get full weight, others get exponentially decaying weight
    threshold = srt[m-1]
    weights = np.exp(alpha * (corr - threshold))
    weights /= np.sum(weights)
    # Return weighted sum (this is a smooth approximation of the max)
    return float(np.sum(weights * corr))

def active_set_subgrad_descent(h_init, n, target, time_limit, alpha=500, top_m=10, max_steps=5000):

    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    
    lr = 0.01
    no_improve = 0

        if time.time() - start > time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full')

        if c5_val < best_c5 - 1e-13:

            no_improve = 0
```

```python
def compute_gradient_matrix(h, n):
    """Compute the gradient matrix G where G[i, m] = dcorr[m]/dh[i]"""
    # G has shape (n, 2n-1)
    G = np.zeros((n, 2*n-1))
    shifts = np.arange(-(n-1), n)  # shift k for correlation index k+n-1
    idxs = np.arange(n)
    
    for i in range(n):
        # ∂corr/∂h[i] at shift k:
        # = (1-h[i-k]) if i-k is valid (from h[i] term)
        #   - h[i+k] if i+k is valid (from (1-h[i]) term, which is h[i+k] appearing in (1-h) at position i)
        # 
        # More carefully: corr[m] = sum_i h[i+k]*(1-h[i]) where k = m-(n-1)
        # ∂corr[m]/∂h[j] = (1-h[j-k]) [if j-k valid] - h[j+k] [if j+k valid]
        k = shifts  # array of all possible shifts
        # For each shift k:
        # term1: (1-h[i-k]) if 0 <= i-k < n
        v1 = (i - k >= 0) & (i - k < n)
        v2 = (i + k >= 0) & (i + k < n)
        G[i, v1] += (1.0 - h[i - k[v1]])
        G[i, v2] -= h[i + k[v2]]
    
    return G
```

```python
        else:
```
```python
            no_improve += 1

            break
        
        k_star = int(np.argmax(corr)) - mid

        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

        grad[v2] -= h[idxs[v2] + k_star]
```

```python
            break

        # But "active" changes with delta, so we approximate using current active set.
        
        # Simpler approach for "union":
        # For each valid pair (p,q), find the first k where corr[k] + delta * pair_grads[p,q,k] exceeds the current max
        # delta_limit[p,q] = min over k (where corr[k] >= max - eps) of (max - corr[k]) / pair_grads[p,q,k] (if pair_grads > 0)
        
        c_max = np.max(corr)
        active_mask = corr >= c_max - 1e-10  # boolean (n_corr,)
        
        # For each pair, compute the max increase before any active constraint is violated
        # delta_pair_limit[p,q] = min over k in active of (c_max - corr[k]) / pair_grads[p,q,k] (only if pair_grads > 0)
        # If pair_grads[p,q,k] <= 0 for all active k, the pair doesn't increase any active constraint → unlimited
        
        active_pair_grads = pair_grads[:, :, active_mask]  # (n_high, n_low, n_active)
        active_excess = (c_max - corr[active_mask])[None, None, :]  # (1, 1, n_active)

        pair_delta_limit = np.min(limits, axis=2)  # (n_high, n_low)
        pair_delta_limit = np.minimum(pair_delta_limit, delta_limits)
        pair_delta_limit = np.maximum(pair_delta_limit, 0.0)
        pair_delta_limit[~valid] = 0.0
        
        if pair_delta_limit.max() < 1e-12:
            break
        
        # Now we have a safe delta for each pair. Use a fraction for safety.
        # Evaluate the objective at delta = 0.5 * pair_delta_limit for each pair
        eval_delta = 0.5 * pair_delta_limit
        new_corrs = corr[None, None, :] + eval_delta[:, :, None] * pair_grads
        maxes = np.max(new_corrs, axis=2) * dx
        maxes[~valid] = np.inf
        
        best_flat = int(np.argmin(maxes))
        best_p = best_flat // n_cand
        best_q = best_flat % n_cand
        best_delta = eval_delta[best_p, best_q]
        best_c5_try = maxes[best_p, best_q]
        
        if best_c5_try < c5_val - 1e-14 and best_delta > 1e-12:
            i = high_set[best_p]
            j = low_set[best_q]
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            G = compute_grad_matrix(h, n)
        else:
            no_improve += 2  # Count non-improving sweeps more aggressively
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_slsqp_n = 0
_slsqp_target = 0.0
_slsqp_alpha = 1.0

def _slsqp_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_slsqp_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _slsqp_alpha)

def _slsqp_cons(h):
    return np.sum(h) - _slsqp_target

def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    G = np.zeros((n, n_corr))
    for i in range(n):
        v1 = (i - shifts >= 0) & (i - shifts < n)
        v2 = (i + shifts >= 0) & (i + shifts < n)
        G[i, v1] = (1.0 - h[i - shifts[v1]])
        G[i, v2] += -h[i + shifts[v2]]
    return G

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, time_limit=None, start_time=None):
    global _slsqp_n, _slsqp_target, _slsqp_alpha
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and (time.time() - start_time) > (time_limit - 5):
            break
        _slsqp_alpha = float(alpha)
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def subgradient_method(h_init, n, target, time_limit, top_m=15, max_steps=3000):

    dx = 2.0 / n

    mid = n - 1
    lr = 0.02
    no_improve = 0
    momentum = np.zeros(n)
    beta_mom = 0.9

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        srt = np.sort(corr)[::-1]
        m = min(top_m, len(srt))
        threshold = srt[m - 1]
        active_idx = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr * lr_mult
            h_try = project_feasible(h - lr_try * momentum, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-14:
                h = h_try
                lr = lr_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5

            best_h = h_sym
        cd_time = min(100, tleft() * 0.15)
        h_cd, c5_cd = fast_pair_cd(h_sym, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=45, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.10, 0.20]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)
        h_start = h_pert
        if sc_pert < best_score:
            best_score = sc_pert
            best_h = h_pert
        cd_time = min(60, tleft() * 0.12)
        h_cd, c5_cd = fast_pair_cd(h_start, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=2000, n_cand=40, n_bisect=13)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Try higher resolution
    if tleft() > 150:
        for n_try in [800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            cd_time = min(80, tleft() * 0.3)
            h_hires, sc2 = fast_pair_cd(h_interp, n_try, target_try,
                                        time_limit=cd_time,
                                        max_sweeps=2000, n_cand=40, n_bisect=12)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 9: Final pair CD (precision)
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=50, n_bisect=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use,
                                     [10000, 30000, 100000],
                                     500, 1e-16,
                                     time_limit=min(25, tleft() - 5), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
