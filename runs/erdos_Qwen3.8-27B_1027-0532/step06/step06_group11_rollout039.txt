```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    g = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _obj_slq(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def _cons_fun(h):
    return np.sum(h) - _G_target

def _cons_jac(h):
    return np.ones(_G_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_obj_slq, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps=1e-8):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = max(1, int(n_top_start + (n_top_end - n_top_start) * frac))
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_structured_init(n, pattern_type, rng):
    """Create various structured initializations"""
    target = n / 2.0
    if pattern_type == 'sawtooth':
        period = int(rng.choice([10, 14, 16, 20, 24, 30]))
        duty = rng.uniform(0.35, 0.65)
        h = np.zeros(n)
        for i in range(n):
            if (i % period) < int(period * duty):
                h[i] = 1.0
        h = project_feasible(h, target)
    elif pattern_type == 'blocks':
        n_blk = int(rng.choice([3, 4, 5, 6, 8]))
        h = np.zeros(n)
        blk = n // n_blk
        for b in range(n_blk):
            if b % 2 == 0:
                h[b*blk:(b+1)*blk] = 1.0
        h = project_feasible(h, target)
    elif pattern_type == 'sine':
        freq = rng.uniform(1.0, 5.0)
        amp = rng.uniform(0.3, 0.7)
        x = np.linspace(0, 2, n)
        h = 0.5 + amp * np.sin(2 * np.pi * freq * x)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target)
    elif pattern_type == 'noise':
        h = rng.uniform(0, 1, n)
        h = project_feasible(h, target)
    elif pattern_type == 'bimodal':
        # Two blocks at the ends
        frac = rng.uniform(0.3, 0.5)
        h = np.zeros(n)
        h[:int(frac*n)] = 1.0
        h[int((1-frac)*n):] = 1.0
        h = project_feasible(h, target)
    else:
        h = np.ones(n) * 0.5
        h = project_feasible(h, target)
    return h

def weighted_subgrad_refine(h_start, n, target, time_limit, max_steps, n_weights=5):
    """Weighted subgradient with softmax weighting on active constraints"""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    lr = 0.03
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 100 and no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_now = float(np.max(corr))
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = np.max(corr)
        # Softmax weights with increasing sharpness
        sharpness = 30.0 + 10.0 * (step / max_steps)
        weights = np.exp(sharpness * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        # Only use constraints with significant weight
        sig = np.where(weights > 1e-4)[0]
        if len(sig) == 0:
            sig = np.argsort(corr)[-n_weights:]
        
        grad = np.zeros(n)
        for s_idx in sig:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad /= weights[sig].sum()
        grad -= np.mean(grad)
        
        # Adaptive lr
        lr_step = lr / (1.0 + 0.005 * step)
        h_new = project_feasible(h - lr_step * grad, target)
        h = h_new
    
    return best_h, best_c5
```
