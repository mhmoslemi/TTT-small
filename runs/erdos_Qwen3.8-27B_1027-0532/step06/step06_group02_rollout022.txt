def compute_gradient_fast(h, alpha, corr_full):
    n = len(h)
    # Softmax weights
    max_c = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - max_c))
    w = exp_c / exp_c.sum()  # shape (2n-1,)
    
    # Convert to indexed by lag k: k = -(n-1), ..., n-1
    # w[0] corresponds to k = -(n-1), w[n-1] to k=0, w[2n-2] to k=n-1
    w_0 = w[n-1]  # k=0
    
    # v[k] = w[k+(n-1)] + w[-k+(n-1)] for k=1,...,n-1
    # v[k] = w[n-1+k] + w[n-1-k]
    v = np.zeros(n)  # v[0]=0, v[1]...v[n-1]
    for k in range(1, n):
        v[k] = w[n-1+k] + w[n-1-k]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_smooth_gradient(h, alpha, corr_full):
    n = len(h)
    max_c = float(corr_full.max())
    # Softmax weights (numerically stable)
    exp_c = np.exp(alpha * (corr_full - max_c))
    w = exp_c / exp_c.sum()
    
    w_0 = w[n-1]  # weight for lag 0
    
    # v[k] = w[n-1+k] + w[n-1-k] for k=1,...,n-1
    v = np.zeros(n)
    v[1:] = w[n:] + w[n-2::-1]  # v[k] = w[n-1+k] + w[n-1-k]
    
    # A[j] = sum of v[1..j]
    A = np.zeros(n)
    A[1:] = np.cumsum(v[1:])
    
    # B[j] = sum of v[k]*h[j-k] for k=1..j (linear convolution)
    N_fft = 1
    while N_fft < 2 * n - 1:
        N_fft <<= 1
    h_lin = np.zeros(N_fft)
    h_lin[:n] = h
    v_lin = np.zeros(N_fft)
    v_lin[:n] = v
    B_full = np.fft.ifft(np.fft.fft(v_lin) * np.fft.fft(h_lin)).real
    B = B_full[:n]
    
    # C[j] = D[n-1-j] where D = linear convolution of v and h_flip
    h_flip = np.zeros(N_fft)
    h_flip[:n] = h[::-1]
    D_full = np.fft.ifft(np.fft.fft(v_lin) * np.fft.fft(h_flip)).real
    C = D_full[n-1-np.arange(n)]
    
    grad = w_0 * (1.0 - 2.0 * h) + A - B - C
    return grad

def project_box_sum(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

        N_fft <<= 1
    h_lin = np.zeros(N_fft)
    h_lin[:n] = h
    v_lin = np.zeros(N_fft)
    v_lin[:n] = v
    B_full = np.fft.ifft(np.fft.fft(v_lin) * np.fft.fft(h_lin)).real
    B = B_full[:n]
    h_flip = np.zeros(N_fft)
    h_flip[:n] = h[::-1]
    D_full = np.fft.ifft(np.fft.fft(v_lin) * np.fft.fft(h_flip)).real
    C = D_full[n-1-np.arange(n)]
    grad = w_0 * (1.0 - 2.0 * h) + A - B - C
    return grad

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            extra = final_sum - target_sum
            ones2 = np.where(h_r == 1)[0]
            if extra <= len(ones2):
                h_r[ones2[:extra]] = 0
        else:
            extra = target_sum - final_sum
            zeros2 = np.where(h_r == 0)[0]
            if extra <= len(zeros2):
                h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def sa_binary_fast(h, n_iters, t_init, t_final, rng, deadline):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 20000

    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        full_new = full_corr_fft(h)
        c5_new = float(full_new.max()) * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            full = full_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def continuous_optimize(h0, target, deadline, alphas, lr_init=0.005, momentum_rate=0.85):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    corr = full_corr_fft(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    lr = lr_init
    v_mom = np.zeros(n)
    stall_counter = 0
    max_stall = 3000

    for alpha in alphas:
        if time.time() > deadline:
            break
        # Estimate iterations possible in remaining time for this alpha phase
        phase_deadline = min(deadline, time.time() + min(60.0, (deadline - time.time()) / max(len(alphas), 1)))
        n_iter = int((phase_deadline - time.time()) * 1000)
        n_iter = max(n_iter, 100)

        for it in range(n_iter):
            if time.time() > deadline:
                break
            if stall_counter > max_stall:
                break
            corr = full_corr_fft(h)
            grad = compute_smooth_gradient(h, alpha, corr)
            v_mom = momentum_rate * v_mom - lr * grad
            h_new = h + v_mom
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 20 == 0 or it == n_iter - 1:
                corr = full_corr_fft(h)
                c5_now = float(corr.max()) * (2.0 / n)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                    stall_counter = 0
                else:
                    stall_counter += 1
            if it % 200 == 199:
                if stall_counter > 2000:
                    lr *= 0.5
                elif stall_counter > 1000:
                    lr *= 0.7
    return best_h, best_c5

def greedy_binary(h, max_iters, rng, deadline):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 400
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(20, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(20, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                h[i] = 0.0
                h[j] = 1.0
                full_new = full_corr_fft(h)
                new_max = float(full_new.max())
                if new_max < max_val:
                    full = full_new
                    stall = 0
                    new_c5 = new_max * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def continuous_sa_block(h0, target, n_iters, t_init, t_final, rng, deadline, noise_scale=0.04):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    corr = full_corr_fft(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000

    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        n_perturb = rng.randint(3, 25)
        idx = rng.choice(n, size=n_perturb, replace=False)
        noise = rng.randn(n_perturb) * noise_scale
        h_pert = h.copy()
        h_pert[idx] += noise
        h_pert = project_box_sum(h_pert, target)
        corr_new = full_corr_fft(h_pert)
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h = h_pert
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    def tleft():
        return deadline - time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    target_int = int(target)

    best_h = project_binary(h_init, target_int)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy binary (50s)
    if tleft() > 30:
        dl = time.time() + min(50, tleft() * 0.04)
        h_g, c5_g = greedy_binary(best_h, max_iters=100000, rng=rng, deadline=dl)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Binary SA (100s)
    if tleft() > 50:
        dl = time.time() + min(100, tleft() * 0.08)
        n_sa = int((dl - time.time()) * 3000)
        h_sa, c5_sa = sa_binary_fast(best_h, n_sa, 0.02, 0.0001, rng, dl)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Continuous gradient (350s) - THE KEY PHASE
    if tleft() > 100:
        dl = time.time() + min(350, tleft() * 0.3)
        alphas = [3, 8, 20, 50, 100, 300]
        h_cont, sc_cont = continuous_optimize(best_h, target, dl, alphas, lr_init=0.004, momentum_rate=0.85)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()

    # Phase 4: Binary-continuous cycles (200s)
    for cyc in range(4):
        if tleft() < 60:
            break
        # Round to binary
        h_bin = project_binary(best_h, target_int)
        # Short SA
        if tleft() > 40:
            dl_sa = time.time() + min(20, tleft() * 0.04)
            n_sac = int((dl_sa - time.time()) * 3000)
            h_sac, c5_sac = sa_binary_fast(h_bin, n_sac, 0.012, 0.00008, rng, dl_sa)
            if c5_sac < best_score:
                best_score = c5_sac
                best_h = h_sac.copy()
            h_use = h_sac if c5_sac < compute_c5(h_bin) else h_bin
        else:
            h_use = h_bin
        # Continuous refine
        if tleft() > 30:
            dl_cont = time.time() + min(40, tleft() * 0.06)
            alphas_c = [50, 150, 400, 1000]
            h_cc, sc_cc = continuous_optimize(h_use, target, dl_cont, alphas_c, lr_init=0.002, momentum_rate=0.82)
            if sc_cc < best_score:
                best_score = sc_cc
                best_h = h_cc.copy()

    # Phase 5: Continuous SA for escape (80s)
    if tleft() > 40:
        dl = time.time() + min(80, tleft() * 0.08)
        n_csa = int((dl - time.time()) * 3000)
        h_csa, sc_csa = continuous_sa_block(best_h, target, n_csa, 0.025, 0.001, rng, dl, noise_scale=0.03)
        if sc_csa < best_score:
            best_score = sc_csa
            best_h = h_csa.copy()

    # Phase 6: Multi-resolution
    for n_try in [n + 20, n + 40, n - 20]:
        if tleft() < 50:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        target_try_int = int(target_try)
        h_interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n), best_h)
        h_bin_try = project_binary(h_interp, target_try_int)
        # Greedy at new res
        if tleft() > 30:
            dl_g = time.time() + min(15, tleft() * 0.03)
            h_gt, c5_gt = greedy_binary(h_bin_try, max_iters=50000, rng=rng, deadline=dl_g)
            h_best_try = h_gt
            best_try = c5_gt
            # Continuous at new res
            if tleft() > 30:
                dl_ct = time.time() + min(25, tleft() * 0.04)
                alphas_t = [20, 80, 300]
                h_ct, sc_ct = continuous_optimize(h_best_try, target_try, dl_ct, alphas_t, lr_init=0.002, momentum_rate=0.8)
                if sc_ct < best_try:
                    best_try = sc_ct
                    h_best_try = h_ct
            # Interpolate back
            h_back = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_try), h_best_try)
            h_back = project_box_sum(h_back, target)
            sc_back = compute_c5(h_back)
            if sc_back < best_score:
                best_score = sc_back
                best_h = h_back.copy()

    # Phase 7: Final high-alpha polish (60s)
    if tleft() > 30:
        dl = time.time() + min(60, tleft() * 0.08)
        alphas_final = [300, 1000, 3000]
        h_fin, sc_fin = continuous_optimize(best_h, target, dl, alphas_final, lr_init=0.001, momentum_rate=0.8)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin.copy()

    # Phase 8: Final binary + continuous burst
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target_int)
        if tleft() > 20:
            dl_sa_f = time.time() + min(15, tleft() * 0.03)
            n_sf = int((dl_sa_f - time.time()) * 3000)
            h_sf, c5_sf = sa_binary_fast(h_bin_final, n_sf, 0.008, 0.00005, rng, dl_sa_f)
            if c5_sf < best_score:
                best_score = c5_sf
                best_h = h_sf.copy()
        if tleft() > 15:
            dl_f = time.time() + min(20, tleft() * 0.03)
            alphas_last = [500, 2000]
            h_lf, sc_lf = continuous_optimize(h_bin_final, target, dl_f, alphas_last, lr_init=0.001, momentum_rate=0.8)
            if sc_lf < best_score:
                best_score = sc_lf
                best_h = h_lf.copy()

    # Final check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
