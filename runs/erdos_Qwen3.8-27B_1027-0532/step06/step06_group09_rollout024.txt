import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    n = len(h)
    sum_h = np.sum(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return sum_h - corr

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    """Find delta that maximizes min_k (autocorr[k] + delta * g[k])"""
    # Focus on the m smallest autocorr values (most likely to be the minimum)
    # But also include any k where g[k] is very large (could become minimum)
    min_idx = np.argsort(autocorr)[:m]
    
    # For the current top-m constraints, find the optimal delta
    # The minimum of (a_k + delta * g_k) over k in min_idx is a concave function of delta
    # Maximum is where two are equal, or at a boundary
    
    a_sub = autocorr[min_idx]
    g_sub = g[min_idx]
    
    # Sort by a_k (ascending)
    order = np.argsort(a_sub)
    a_sub = a_sub[order]
    g_sub = g_sub[order]
    
    # The minimum at delta=0 is a_sub[0]
    # As delta increases, the k with smallest a_k + delta*g_k changes
    # Find breakpoints
    
    best_delta = 0.0
    best_val = a_sub[0]
    
    # Check boundaries
    for d in [delta_lo, delta_hi]:
        vals = a_sub + d * g_sub
        if np.min(vals) > best_val:
            best_val = np.min(vals)
            best_delta = d
    
    # Check internal breakpoints
    for i in range(len(a_sub)):
        for j in range(i+1, len(a_sub)):
            if abs(g_sub[i] - g_sub[j]) < 1e-14:

            d_break = (a_sub[j] - a_sub[i]) / (g_sub[i] - g_sub[j])
            if delta_lo <= d_break <= delta_hi:
                vals = a_sub + d_break * g_sub
                min_val = np.min(vals)
                if min_val > best_val:
                    best_val = min_val
                    best_delta = d_break
    
    return best_delta

def lp_subgradient_step(h, n, target, max_step=0.1):
    """Take a step using LP-based subgradient method"""
    from scipy.optimize import linprog
    
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute A[k, j] = (1-h[j+k_offset]) - h[j-k_offset] for all k, j
    # where k_offset = k - mid
    A = np.zeros((n_corr, n))
    for k in range(n_corr):

        for j in range(n):
            val = 0.0
            if 0 <= j + k_off < n:
                val += (1.0 - h[j + k_off])
            if 0 <= j - k_off < n:
                val -= h[j - k_off]
            A[k, j] = val

    #      sum(delta) = 0
    #      -h[i] ≤ delta[i] ≤ 1-h[i]
    #      t ≥ 0

    c_obj = np.zeros(n + 1)

    # In matrix form: [A, -1] @ [delta; t] ≤ -corr
    
    A_ub = np.hstack([A, -np.ones((n_corr, 1))])
    b_ub = -corr

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i] ≤ delta[i] ≤ 1-h[i], t ≥ 0
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(0, None)]
    
    # Solve LP
    try:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

Actually, you know what, let me think about this from a completely different angle.

The problem is: minimize max_k corr_k where corr_k = sum_i h[i]*(1-h[i+k]).

This is a minimax problem. The minimax theorem tells us:
min_h max_k corr_k = max_{p ∈ Δ} min_h sum_k p_k * corr_k

where Δ is the probability simplex.

So we can think of it as a game:
- Player 1 (h-chooser) picks h to minimize
- Player 2 (lag-chooser) picks a distribution p over lags to maximize
- The payoff is sum_k p_k * corr_k = sum_k p_k * sum_i h[i]*(1-h[i+k])

For a fixed p, Player 1's problem is:
min_h sum_k p_k * sum_i h[i]*(1-h[i+k])
= min_h sum_i h[i] * (1 - sum_k p_k * h[i+k])

This is a non-convex problem in h (because of h[i]*h[i+k] terms). But if we fix the "other" values, it's linear in each h[i].

This suggests an alternating minimization:
1. Fix p, optimize h
2. Fix h, optimize p (which is just picking the max lag, or a mixture near the max)

This is essentially what the game-theoretic descent does! But we can improve it:

**Step 1 (optimize h given p)**:
The objective is: sum_k p_k * sum_i h[i]*(1-h[i+k])
= sum_i h[i] * (1 - sum_k p_k * h[i+k])
= sum_i h[i] - sum_i h[i] * (sum_k p_k * h[i+k])

Let w_i = sum_k p_k * h[i+k] (the "expected value" at position i under distribution p)
Then the objective is: sum_i h[i] - sum_i h[i] * w_i = sum_i h[i]*(1 - w_i)

This is linear in h! So for fixed p (and fixed w), the optimal h is:
- h[i] = 1 if w_i < 1/2 (and we need more mass)
- h[i] = 0 if w_i > 1/2 (and we need less mass)
- h[i] = fractional if w_i = 1/2

- For each i, given all other h values, w_i is fixed
- Optimize h[i] to minimize h[i]*(1-w_i) + (terms involving h[i] in other positions' w_j)

        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Compute effect vectors for candidates (only for top 10 lags)
        top10 = np.argsort(corr)[-10:]
        
        low_effs = np.zeros((len(low_set), len(top10)))
        high_effs = np.zeros((len(high_set), len(top10)))
        
        for ci, j in enumerate(low_set):
            for ti, k_s in enumerate(top10):
                k_off = int(k_s) - mid
                val = 0.0
                if 0 <= j + k_off < n:
                    val += h[j + k_off]
                if 0 <= j - k_off < n:
                    val += h[j - k_off]
                low_effs[ci, ti] = val
        
        for ci, i in enumerate(high_set):
            for ti, k_s in enumerate(top10):
                k_off = int(k_s) - mid
                val = 0.0
                if 0 <= i + k_off < n:
                    val += h[i + k_off]
                if 0 <= i - k_off < n:
                    val += h[i - k_off]
