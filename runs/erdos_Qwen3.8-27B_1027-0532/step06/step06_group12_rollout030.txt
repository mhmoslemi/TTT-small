                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

            def cons_fun(h):

            if compute_c5(h_new) < compute_c5(h_cur):

            pass

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, target):
    n = len(h)
    target_int = int(target)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target_int]] = 1.0
    return h_bin

def frank_wolfe(h_init, n, target, time_limit, max_iter=200000, K=40):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    dx = 2.0 / n
    n_half = n - 1
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_all_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:K]
        g = np.zeros(n)
        for idx in top_idx:
            k = idx - n_half
            abs_k = abs(k)
            if abs_k == 0:
                g += (1.0 - 2.0 * h) * dx
            elif k > 0:
                g[:n - k] += (1.0 - h[k:]) * dx
                g[k:] += (-h[:n - k]) * dx
            else:
                m = -k
                g[m:] += (1.0 - h[:n - m]) * dx
                g[:n - m] += (-h[m:]) * dx
        g /= len(top_idx)
        sort_idx = np.argsort(g)
        v = np.zeros(n)
        v[sort_idx[:int(target)]] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
        h = h_new
        if t > 5000 and t - best_t > 30000:
            break
    return best_h, best_c5

                ca = rng.randint(n)
                cb = rng.randint(n)
                if h[ca] == 1 and h[cb] == 0:
                    a = ca
                    b = cb
                    break
            if a is None:
                continue
        if a == b:
            continue
        h[a] = 0
        h[b] = 1
        new_overlaps = compute_all_overlaps(h)
        new_c5 = float(np.max(new_overlaps))
        if new_c5 <= best_c5 + 1e-15:
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                best_overlaps = new_overlaps
            no_improve = 0
        else:
            h[a] = 1
            h[b] = 0
            no_improve += 1
    return best_h, best_c5

def upsample(h, factor=2):
    return np.repeat(h, factor)

def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            def obj(h):
                corr = np.correlate(h, 1.0 - h, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
            def cons_fun(h):
                return np.sum(h) - target
            def cons_jac(h):
                return np.ones(n)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - 45.0 - elapsed()

    # Phase 1: Quick FW polish on n=400
    tl = remaining()
    if tl > 60:
        h_fw, c5_fw = frank_wolfe(best_h, n0, target0, min(50, tl * 0.1), max_iter=80000, K=40)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 2: Upsample to n=800 and FW
    n1 = n0 * 2
    target1 = n1 / 2.0
    h_up = upsample(best_h, 2)
    tl = remaining()
    if tl > 100:
        h_fw2, c5_fw2 = frank_wolfe(h_up, n1, target1, min(80, tl * 0.12), max_iter=80000, K=50)
        if c5_fw2 < best_score:
            best_score = c5_fw2
            best_h = h_fw2
        else:
            best_h_n800 = h_fw2
        if c5_fw2 >= best_score:
            best_h_n800 = frank_wolfe(h_up, n1, target1, 30, max_iter=30000, K=50)[0]
        else:
            best_h_n800 = best_h
        if best_score <= 0.38085:
            h_work = best_h
            n_work = len(best_h)
        else:
            h_work = best_h_n800 if c5_fw2 < best_score else h_fw2
            n_work = n1

    # Phase 3: Guided local search on n=800 binary
    tl = remaining()
    if tl > 200:
        h_bin1 = binarize(h_work, n_work / 2.0)
        ls_time = min(500, tl * 0.5)
        h_ls, c5_ls = guided_local_search(h_bin1, n_work, ls_time,
                                           max_iter=30000000, rng_seed=seed, K=5)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls

    # Phase 4: SLSQP polish
    tl = remaining()
    if tl > 80:
        n_best = len(best_h)
        h_slq = slsqp_optimize(best_h, n_best, n_best / 2.0,
                                [100, 500, 2000, 10000, 50000, 200000], maxiter=3000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: More local search after SLSQP
    tl = remaining()
    if tl > 100:
        h_bin2 = binarize(best_h, len(best_h) / 2.0)
        ls_time = min(300, tl * 0.5)
        h_ls2, c5_ls2 = guided_local_search(h_bin2, len(best_h), ls_time,
                                             max_iter=20000000, rng_seed=seed + 100, K=7)
        if c5_ls2 < best_score:
            best_score = c5_ls2
            best_h = h_ls2

    # Phase 6: Another SLSQP
    tl = remaining()
    if tl > 60:
        n_best = len(best_h)
        h_slq2 = slsqp_optimize(best_h, n_best, n_best / 2.0,
                                 [500, 2000, 10000, 50000, 200000, 1000000], maxiter=3000)
        sc = compute_c5(h_slq2)
        if sc < best_score:
            best_score = sc
            best_h = h_slq2

    # Phase 7: Final local search
    tl = remaining()
    if tl > 50:
        h_bin3 = binarize(best_h, len(best_h) / 2.0)
        ls_time = min(80, tl * 0.7)
        h_ls3, c5_ls3 = guided_local_search(h_bin3, len(best_h), ls_time,
                                             max_iter=10000000, rng_seed=seed + 200, K=3)
        if c5_ls3 < best_score:
            best_score = c5_ls3
            best_h = h_ls3

    # Phase 8: Final SLSQP
    tl = remaining()
    if tl > 40:
        n_best = len(best_h)
        h_slq3 = slsqp_optimize(best_h, n_best, n_best / 2.0,
                                 [2000, 10000, 50000], maxiter=2000)
        sc = compute_c5(h_slq3)
        if sc < best_score:
            best_score = sc
            best_h = h_slq3

    # Final feasibility
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
