The ∑_s w_ext[s] * h[j+s] * I[valid] is a cross-correlation of w_ext and h.

The second sum ∑_s w_ext[s] * h[j-s] * I[valid] is also a cross-correlation.

def compute_gradient_fast(h, w, n):
    """Compute gradient of smoothed max objective using FFT."""
    # w has length 2n-1, indexed 0 to 2n-2
    # w_ext[s] = w[s+n-1] for s in [-(n-1), n-1]
    # 
    # g[j] = sum_s w_ext[s] * (1-h[j+s]) * I[0<=j+s<n] 
    #        - sum_s w_ext[s] * h[j-s] * I[0<=j-s<n]
    
    # First term: sum_s w_ext[s] * I[0<=j+s<n]
    # For a given j, this is sum over s such that 0 <= j+s < n, i.e., -j <= s < n-j
    # So it's sum of w_ext[s] for s in [max(-(n-1),-j), min(n-1, n-j-1)]
    # This is a sliding window sum of w_ext
    
    # Second term: sum_s w_ext[s] * h[j+s] * I[0<=j+s<n]
    # This is the cross-correlation of w_ext and h (with appropriate truncation)
    
    # Third term: sum_s w_ext[s] * h[j-s] * I[0<=j-s<n]
    # This is the cross-correlation of w_ext and h in the other direction
    
    # Let's use np.correlate for these
    # np.correlate(a, b, 'full')[k] = sum_n a[n+k] * b[n]
    
    # For the first term, define ones = np.ones(n)
    # sum_s w_ext[s] * I[0<=j+s<n] = sum_s w_ext[s] * ones[j+s] * I[valid for ones]
    # This is correlate(ones, w_ext, 'full') at appropriate index... 
    
    # Actually, let me just do it directly with correlate
    # 
    # Term 1: T1[j] = sum_{s: 0<=j+s<n} w_ext[s]
    #         = sum_{i: 0<=i<n} w_ext[i-j] (where i=j+s)
    #         = cross-correlation of ones_n and w_ext at position j
    #         = np.correlate(np.ones(n), w_ext_padded, 'full')[offset + j]
    
    # This is getting complicated with indexing. Let me just use the O(n^2) approach
    # since n=600 is small enough.
    
    idxs = np.arange(n)
    g = np.zeros(n)
    
    # For each shift k (from -(n-1) to n-1), w_val = w[k+n-1]
    # Contribution to g[j]: w_val * (1-h[j+k]) for j+k in [0,n)
    #                      : w_val * (-h[j-k]) for j-k in [0,n)
    
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-18:
            continue
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(v1):
            g[v1] += wk * (1.0 - h[idxs[v1] + k])
        if np.any(v2):
            g[v2] -= wk * h[idxs[v2] - k]
    
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

        if w_sum < 1e-300:

        g -= g_full[n-1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_smoothed(h_init, n, target, alpha, maxiter):
    """L-BFGS-B on reparameterized problem with log-sum-exp smoothing."""
    idxs = np.arange(n)
    x0 = h_init[:n-1].copy()
    
    def obj(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target - np.sum(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def grad(x):
        h = np.empty(n)
        h[:n-1] = x
        h[n-1] = target - np.sum(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.ones(len(corr)) / len(corr)
        else:
            w = w / w_sum
        
        g_full = np.zeros(n)
        for k in range(-(n-1), n):
            wk = float(w[k + n - 1])
            if abs(wk) < 1e-18:
                continue
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            if np.any(v1):
                g_full[v1] += wk * (1.0 - h[idxs[v1] + k])
            if np.any(v2):
                g_full[v2] -= wk * h[idxs[v2] - k]
        
        g = g_full[:n-1].copy()
        g -= g_full[n-1]
        return g
    
    bounds = [(0.0, 1.0)] * (n-1)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = np.empty(n)
        h_new[:n-1] = res.x
        h_new[n-1] = target - np.sum(res.x)
        h_new = project_feasible(h_new, target)
    except Exception:
        h_new = h_init.copy()
    
    return h_new, compute_c5(h_new)

def pair_cd_optimized(h_init, n, target, time_limit, n_cand=15, max_sweeps=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    ks = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Use the argmax shift
        k_star = int(np.argmax(corr)) - (n - 1)
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        sorted_idx = np.argsort(-grad)
        high_set = sorted_idx[:n_cand]
        low_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Compute profiles
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            p1 = (m + ks >= 0) & (m + ks < n)
            p2 = (m - ks >= 0) & (m - ks < n)
            if np.any(p1):
                profiles[ci, p1] += (1.0 - h[m + ks[p1]])
            if np.any(p2):
                profiles[ci, p2] -= h[m - ks[p2]]
        
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        # Identify "active" correlation positions (near max)
        active_thresh = cur_max - 1e-8
        active_mask = corr >= active_thresh
        n_active = int(np.sum(active_mask))
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-14:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                if hj_val >= 1.0 - 1e-14:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                delta_max = min(hi_val, 1.0 - hj_val)
                
                # Try delta = 0.5 * delta_max first
                delta = 0.5 * delta_max
                if n_active < n_corr:
                    new_max = float(np.max(corr[active_mask] + delta * g[active_mask]))
                else:
                    new_max = float(np.max(corr + delta * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta
        
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def subgrad_method(h_init, n, target, time_limit, max_steps=100000):
    """Projected subgradient with momentum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    v = np.zeros(n)  # momentum
    beta = 0.9
    base_lr = 0.01
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Use top shifts
        n_shifts = 3
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        # Momentum
        v = beta * v + grad
        lr = base_lr / (1.0 + 0.01 * step)
        
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        
        # Line search
        c5_try = compute_c5(h_new)
        if c5_try < c5_now:
            h = h_new
        else:
            # Try smaller step
            h_new = h - 0.5 * lr * v
            h_new = project_feasible(h_new, target)
            c5_try = compute_c5(h_new)
            if c5_try < c5_now:
                h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha, n, target, idxs):
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = target - np.sum(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _smooth_grad(x, alpha, n, target, idxs):
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = target - np.sum(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.ones(len(corr)) / len(corr)
    else:
        w = w / w_sum
    g_full = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = float(w[k + n - 1])
        if abs(wk) < 1e-18:
            continue
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(v1):
            g_full[v1] += wk * (1.0 - h[idxs[v1] + k])
        if np.any(v2):
            g_full[v2] -= wk * h[idxs[v2] - k]
    g = g_full[:n - 1].copy()
    g -= g_full[n - 1]
    return g


def lbfgsb_smoothed(h_init, n, target, alpha, maxiter):
    idxs = np.arange(n)
    x0 = h_init[:n - 1].copy()
    bounds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(_smooth_obj, x0, args=(alpha, n, target, idxs),
                       jac=_smooth_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = np.empty(n)
        h_new[:n - 1] = res.x
        h_new[n - 1] = target - np.sum(res.x)
        h_new = project_feasible(h_new, target)
    except Exception:
        h_new = h_init.copy()
    return h_new, compute_c5(h_new)


def pair_cd_optimized(h_init, n, target, time_limit, n_cand=15, max_sweeps=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    ks = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_idx = np.argsort(-grad)
        high_set = sorted_idx[:n_cand]
        low_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)

        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            p1 = (m + ks >= 0) & (m + ks < n)
            p2 = (m - ks >= 0) & (m - ks < n)
            if np.any(p1):
                profiles[ci, p1] += (1.0 - h[m + ks[p1]])
            if np.any(p2):
                profiles[ci, p2] -= h[m - ks[p2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        active_thresh = cur_max - 1e-8
        active_mask = corr >= active_thresh
        n_active = int(np.sum(active_mask))

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-14:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                if hj_val >= 1.0 - 1e-14:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-14:
                    continue

                delta = 0.5 * delta_max
                if n_active < n_corr:
                    new_max = float(np.max(corr[active_mask] + delta * g[active_mask]))
                else:
                    new_max = float(np.max(corr + delta * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta

                delta2 = 0.25 * delta_max
                if n_active < n_corr:
                    new_max2 = float(np.max(corr[active_mask] + delta2 * g[active_mask]))
                else:
                    new_max2 = float(np.max(corr + delta2 * g))
                improve2 = cur_max - new_max2
                if improve2 > best_improve:
                    best_improve = improve2
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta2

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, max_steps=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.015

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        n_shifts = 4
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 25.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)
        v = beta * v + grad
        lr = base_lr / (1.0 + 0.005 * step)

        h_new = project_feasible(h - lr * v, target)
        c5_try = compute_c5(h_new)
        if c5_try < c5_now - 1e-15:
            h = h_new
        else:
            h_new = project_feasible(h - 0.5 * lr * v, target)
            c5_try2 = compute_c5(h_new)
            if c5_try2 < c5_now - 1e-15:
                h = h_new

    return best_h, best_c5
