def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

                       options={'maxiter': maxiter, 'ftol': 1e-14})

        j_minus = low_set[:, None] - shifts[None, :]
        j_plus = low_set[:, None] + shifts[None, :]
        jm_valid = (j_minus >= 0) & (j_minus < n)
        jp_valid = (j_plus >= 0) & (j_plus < n)
        B = np.zeros((n_low, n_corr))
        B[jm_valid] += (1.0 - h[j_minus[jm_valid]])
        B[jp_valid] -= h[j_plus[jp_valid]]

        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_g = None
        best_new_max = c_max_corr + 1e-10

        for i in high_set:
            i_minus = i - shifts
            i_plus = i + shifts
            im_valid = (i_minus >= 0) & (i_minus < n)
            ip_valid = (i_plus >= 0) & (i_plus < n)
            A = np.zeros(n_corr)
            A[ip_valid] += h[i_plus[ip_valid]]
            A[im_valid] -= (1.0 - h[i_minus[im_valid]])

            G = A[None, :] + B
            delta_hi = np.minimum(h[i], 1.0 - h[low_set])
            valid_mask = delta_hi > 1e-12
            if not valid_mask.any():
                continue
            Gv = G[valid_mask]
            dhv = delta_hi[valid_mask]
            vi = np.where(valid_mask)[0]

            a_d = np.zeros(len(Gv))
            b_d = dhv.copy()

            for _ in range(max_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr[None, :] + mid_d[:, None] * Gv
                f_mid = vals.max(axis=1)
                active = vals >= (f_mid[:, None] - 1e-12)
                Gm_pos = np.where(active, Gv, -np.inf)
                f_prime_max = Gm_pos.max(axis=1)
                Gm_neg = np.where(active, Gv, np.inf)
                f_prime_min = Gm_neg.min(axis=1)
                go_left = f_prime_min > 1e-12
                go_right = f_prime_max < -1e-12
                b_d = np.where(go_left, mid_d, b_d)
                a_d = np.where(go_right, a_d, mid_d)

            delta_star = 0.5 * (a_d + b_d)
            final_vals = corr[None, :] + delta_star[:, None] * Gv
            final_max = final_vals.max(axis=1)

            bi = np.argmin(final_max)
            if final_max[bi] < best_new_max:
                best_new_max = final_max[bi]
                best_delta = float(delta_star[bi])
                best_i = int(i)
                best_j = int(low_set[vi[bi]])
                best_g = A + B[vi[bi]]

        if best_i >= 0 and abs(best_delta) > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            if h[best_i] < 0:
                h[best_i] = 0.0
            if h[best_j] > 1:
                h[best_j] = 1.0
            corr += best_delta * best_g
            moves_since_recalc += 1

            if moves_since_recalc >= recalc_every:
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                moves_since_recalc = 0
        else:
            if no_improve > 30:
                break

    return best_h, best_c5

def multi_shift_cd(h_init, n, target, time_limit, n_cand=30, m_top=5, max_bisect=14, recalc_every=60):
    start_t = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    moves_since_recalc = 0

    while time.time() - start_t < time_limit:
        c_max_corr = float(np.max(corr))
        c5_now = c_max_corr * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                rng_p = np.random.RandomState(int(time.time() * 9999) & 0xFFFFFFFF)
                h_pert = h + rng_p.randn(n) * 0.03
                h_pert = project_feasible(h_pert, target)
                corr_pert = np.correlate(h_pert, 1.0 - h_pert, mode='full').astype(float)
                if float(np.max(corr_pert)) * dx < best_c5:
                    h = h_pert
                    corr = corr_pert
                    no_improve = 0
                    moves_since_recalc = 0
                else:
                    break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = np.ones(m_top) / m_top

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_low = len(low_set)
        j_minus = low_set[:, None] - shifts[None, :]
        j_plus = low_set[:, None] + shifts[None, :]
        jm_valid = (j_minus >= 0) & (j_minus < n)
        jp_valid = (j_plus >= 0) & (j_plus < n)
        B = np.zeros((n_low, n_corr))
        B[jm_valid] += (1.0 - h[j_minus[jm_valid]])
        B[jp_valid] -= h[j_plus[jp_valid]]

        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_g = None
        best_new_max = c_max_corr + 1e-10

        for i in high_set:
            i_minus = i - shifts
            i_plus = i + shifts
            im_valid = (i_minus >= 0) & (i_minus < n)
            ip_valid = (i_plus >= 0) & (i_plus < n)
            A = np.zeros(n_corr)
            A[ip_valid] += h[i_plus[ip_valid]]
            A[im_valid] -= (1.0 - h[i_minus[im_valid]])

            G = A[None, :] + B
            delta_hi = np.minimum(h[i], 1.0 - h[low_set])
            valid_mask = delta_hi > 1e-12
            if not valid_mask.any():
                continue
            Gv = G[valid_mask]
            dhv = delta_hi[valid_mask]
            vi = np.where(valid_mask)[0]

            a_d = np.zeros(len(Gv))
            b_d = dhv.copy()

            for _ in range(max_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr[None, :] + mid_d[:, None] * Gv
                f_mid = vals.max(axis=1)
                active = vals >= (f_mid[:, None] - 1e-12)
                Gm_pos = np.where(active, Gv, -np.inf)
                f_prime_max = Gm_pos.max(axis=1)
                Gm_neg = np.where(active, Gv, np.inf)
                f_prime_min = Gm_neg.min(axis=1)
                go_left = f_prime_min > 1e-12
                go_right = f_prime_max < -1e-12
                b_d = np.where(go_left, mid_d, b_d)
                a_d = np.where(go_right, a_d, mid_d)

            delta_star = 0.5 * (a_d + b_d)
            final_vals = corr[None, :] + delta_star[:, None] * Gv
            final_max = final_vals.max(axis=1)

            bi = np.argmin(final_max)
            if final_max[bi] < best_new_max:
                best_new_max = final_max[bi]
                best_delta = float(delta_star[bi])
                best_i = int(i)
                best_j = int(low_set[vi[bi]])
                best_g = A + B[vi[bi]]

        if best_i >= 0 and abs(best_delta) > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            if h[best_i] < 0:
                h[best_i] = 0.0
            if h[best_j] > 1:
                h[best_j] = 1.0
            corr += best_delta * best_g
            moves_since_recalc += 1

            if moves_since_recalc >= recalc_every:
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                moves_since_recalc = 0
        else:
            if no_improve > 20:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast pair CD at original resolution
    if tleft() > 200:
        cd_time = min(450, tleft() * 0.42)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   n_cand=35, max_bisect=16, recalc_every=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Multi-shift CD for a different angle
    if tleft() > 150:
        ms_time = min(120, tleft() * 0.15)
        h_ms, c5_ms = multi_shift_cd(best_h, n, target, time_limit=ms_time,
                                     n_cand=30, m_top=5, max_bisect=14, recalc_every=60)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 3: L-BFGS-B polish
    if tleft() > 100:
        for alpha in [500, 2000, 8000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Second round pair CD with larger candidate set
    if tleft() > 150:
        cd_time = min(180, tleft() * 0.22)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   n_cand=50, max_bisect=16, recalc_every=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Higher resolution
    if tleft() > 200:
        n_hires = 500
        target_hires = n_hires / 2.0
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        sc_int = compute_c5(h_interp)

        h_hires = lbfags_polish(h_interp, n_hires, target_hires, 2000, maxiter=1200)
        sc_pol = compute_c5(h_hires)
        if sc_pol < sc_int:
            h_interp = h_hires
            sc_int = sc_pol

        if tleft() > 100:
            cd_time = min(150, tleft() * 0.4)
            h_cd, c5_cd = fast_pair_cd(h_interp, n_hires, target_hires,
                                       time_limit=cd_time,
                                       n_cand=35, max_bisect=16, recalc_every=40)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        if tleft() > 60:
            h_ms, c5_ms = multi_shift_cd(best_h, len(best_h), len(best_h) / 2.0,
                                         time_limit=min(50, tleft() * 0.3),
                                         n_cand=30, m_top=6, max_bisect=14, recalc_every=50)
            if c5_ms < best_score:
                best_score = c5_ms
                best_h = h_ms

    # Phase 6: Final L-BFGS-B + pair CD
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_try = lbfags_polish(best_h, n_use, target_use, 10000, maxiter=1000)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try
        if tleft() > 50:
            h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                       time_limit=min(40, tleft() * 0.4),
                                       n_cand=40, max_bisect=16, recalc_every=25)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
