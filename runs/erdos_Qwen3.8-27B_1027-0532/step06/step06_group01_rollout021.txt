    corr = np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def multi_shift_pair_cd(h_init, n, target, time_limit, n_cand=30, n_shifts=5, max_sweeps=50000):
    """Pair CD using gradient from multiple critical shifts."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Get top n_shifts shifts
        top_idx = np.argsort(corr)[-n_shifts:]
        # Softmax weights
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        # Select candidates
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles for each candidate
        n_corr = 2 * n - 1
        ks = np.arange(-(n - 1), n)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        # Find best pair
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-13)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        # Try swap as well (lower cost)
        elif best_i < 0 or best_improve <= 1e-13:
            # Try a few random swaps
            if sweep % 10 == 0:
                for _ in range(5):
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                    if i != j:
                        h_temp = h.copy()
                        h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
                        h_temp = project_feasible(h_temp, target)
                        c5_temp = compute_c5(h_temp)
                        if c5_temp < c5_val - 1e-14:
                            h = h_temp
                            break
    
    return best_h, best_c5


def subgradient_descent(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    """Multi-shift subgradient descent with backtracking."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        # Multi-shift weighted gradient
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        # Backtracking line search
        improved = False
        lr = base_lr
        for _ in range(8):
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            break
    
    return best_h, best_c5


def perturb_restart(h, n, target, magnitude=0.05):
    """Add structured perturbation and re-project."""
    h_pert = h + magnitude * (np.random.rand(n) - 0.5)
    return project_feasible(h_pert, target)

        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with multiple alphas from initial
    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000, 20000], 1200, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Multi-shift pair CD (main workhorse, longer time)
    if tleft() > 250:
        cd_time = min(450, tleft() * 0.35)
        hcd, c5cd = multi_shift_pair_cd(best_h, n, target, time_limit=cd_time,
                                         n_cand=40, n_shifts=5, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: Subgradient descent for polishing
    if tleft() > 50:
        hsg, c5sg = subgradient_cd(best_h, n, target, time_limit=min(35, tleft() * 0.3),
                                    max_steps=50000, n_shifts=5, base_lr=0.03)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 4: SLSQP polish with higher alphas
    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1200, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Pair CD with wider candidate set and fewer shifts
    if tleft() > 180:
        cd_time = min(280, tleft() * 0.3)
        hcd, c5cd = multi_shift_pair_cd(best_h, n, target, time_limit=cd_time,
                                         n_cand=50, n_shifts=3, max_sweeps=80000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: Perturbation restart + re-optimize
    if tleft() > 120:
        for mag in [0.02, 0.04]:
            if tleft() < 100:
                break
            h_pert = perturb_restart(best_h, n, target, magnitude=mag)
            if tleft() > 80:
                hp, sc = slsqp_round(h_pert, n, target, [1000, 5000], 800, 1e-13)
                if tleft() > 60:
                    hcd, c5cd = multi_shift_pair_cd(hp, n, target, time_limit=min(50, tleft() * 0.4),
                                                     n_cand=35, n_shifts=5, max_sweeps=40000)
                    if c5cd < best_score:
                        best_score = c5cd
                        best_h = hcd
                elif sc < best_score:
                    best_score = sc
                    best_h = hp

    # Phase 7: SLSQP with very high alpha for fine polishing
    if tleft() > 50:
        hp, sc = slsqp_round(best_h, n, target, [50000, 200000, 500000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Final pair CD
    if tleft() > 50:
        hcd, c5cd = multi_shift_pair_cd(best_h, n, target, time_limit=min(40, tleft() * 0.5),
                                         n_cand=45, n_shifts=5, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Final SLSQP
    if tleft() > 25:
        hp, sc = slsqp_round(best_h, n, target, [100000, 500000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 10: Ultimate pair CD
    if tleft() > 20:
        hcd, c5cd = multi_shift_pair_cd(best_h, n, target, time_limit=min(15, tleft() * 0.5),
                                         n_cand=35, n_shifts=5, max_sweeps=20000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
