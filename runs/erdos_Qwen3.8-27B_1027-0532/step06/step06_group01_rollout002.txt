import numpy as np
from scipy.optimize import minimize
import time

_SLQ_TARGET = 0.0
_SLQ_N = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _slq_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET


def _slq_con_jac(x):
    return np.ones(_SLQ_N)


def slsqp_round(h_start, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET, _SLQ_N
    n = len(h_start)
    target = n / 2.0
    _SLQ_TARGET = target
    _SLQ_N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slq_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def multi_shift_pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=200000,
                        n_active=3, bisection_iters=20, restart_every=300,
                        restart_frac=0.03):
    start = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * (2.0 / n)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > restart_every:
                rng = np.random.default_rng(sweep)
                n_perturb = max(3, int(n * restart_frac))
                perm = rng.permutation(n)
                perturb_idx = perm[:n_perturb]
                jitter = rng.uniform(-0.1, 0.1, size=n_perturb)
                h = h.copy()
                h[perturb_idx] = np.clip(h[perturb_idx] + jitter, 0.0, 1.0)
                h = project_feasible(h, target)
                no_improve = 0
                continue

        top_idx = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for si in top_idx:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]

        grad /= float(n_active)
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)

        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        high_list = high_set[:min(n_cand, 40)]
        low_list = low_set[:min(n_cand, 40)]

        for i in high_list:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_list:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def kkt_refine(h_init, n, target, time_limit, max_iter=500):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * (2.0 / n)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        tol = cur_max * 0.01
        active = np.where(corr >= cur_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for si in active:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))

        grad -= np.mean(grad)
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)

            break

        step_size = 0.01 / (np.max(np.abs(grad[interior])) + 1e-10)

        h_try[interior] -= step_size * grad[interior]

        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try

    return best_h, best_c5


def binary_icm(h_init, n, target, time_limit, max_sweeps=50000):

        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cands = ones[np.argsort(-grad[ones])[:20]]

        best_j = -1

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    target_dst = n_dst / 2.0
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
