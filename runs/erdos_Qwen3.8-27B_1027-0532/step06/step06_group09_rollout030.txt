So correlate(a, b, 'full')[mid + k] where mid = len(b) - 1 = 2:
- k=-2: a0*b2 (only a[0] * b[0-(-2)] = a[0]*b[2])
- k=-1: a0*b1 + a1*b2
- k=0: a0*b0 + a1*b1 + a2*b2
- k=1: a1*b0 + a2*b1
- k=2: a2*b0

So correlate(a, b, 'full')[mid+k] = sum_i a[i] * b[i+k] for valid i and i+k.

Great, so:
- sum_j d[j]*h[j+k] = correlate(d, h, 'full')[mid+k] ✓
- sum_j h[j]*d[j+k] = correlate(h, d, 'full')[mid+k] ✓

So Delta[k] = sum(d) - correlate(d, h)[mid+k] - correlate(h, d)[mid+k]

And since sum(d) = 0 (we subtracted the mean), Delta[k] = -correlate(d, h)[mid+k] - correlate(h, d)[mid+k]

The length of correlate(d, h, 'full') is 2*n-1, same as corr. So Delta has the same shape as corr. ✓

This is correct and efficient (2 correlate calls per step, each O(n^2) or O(n log n) with FFT).

    if abs(diff) < 1e-14:
        return h

    h = np.clip(h, 0.0, 1.0)

    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

        return h_start

        return smooth_obj(h, alpha)

        return np.ones(n)

                        h = h_try
                        break
        else:
            # Direction is not helpful; try opposite
            pass
    
    return best_h, best_c5

            d /= dn
        else:
            exp_arg = np.clip(alpha * (corr - c_max), -30, 0)
            weights = np.exp(exp_arg)
            top_idx = np.argsort(corr)[-top_m:]
            mask = np.zeros(n_corr, dtype=bool)
            mask[top_idx] = True
            weights[~mask] = 0
            ws = weights.sum()
            if ws < 1e-14:
                break
            weights /= ws

            g = np.zeros(n)
            for s_idx in top_idx:
                k = int(s_idx) - mid
                w = weights[s_idx]
                if w < 1e-10:
                    continue
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += w * (1.0 - h[idxs[v1] - k])
                g[v2] -= w * h[idxs[v2] + k]
            g *= dx
            g -= np.mean(g)
            d = -g

        if use_momentum:
            d = mu * d_prev + (1.0 - mu) * d
            d_prev = d

        dn = np.linalg.norm(d)
        if dn < 1e-14:
            no_improve += 5
            continue
        d /= dn

        d_pos = d > 1e-14
        d_neg = d < -1e-14
        t_max = 2.0
        if d_pos.any():
            t_max = min(t_max, np.min((1.0 - h[d_pos]) / d[d_pos]))
        if d_neg.any():
            t_max = min(t_max, np.min(h[d_neg] / (-d[d_neg])))
        if t_max < 1e-12:
            no_improve += 5
            continue

        dh = np.correlate(d, h, mode='full')
        hd = np.correlate(h, d, mode='full')
        Delta = -dh - hd

        best_t = 0.0
        best_val = c5_now
        t_try = t_max * t_scale * t_decay
        for _ in range(25):
            if t_try <= 1e-13:
                break
            vals = corr + t_try * Delta
            val = float(np.max(vals)) * dx
            if val < best_val - 1e-15:
                best_val = val
                best_t = t_try
                if val < best_c5 - 1e-6:
                    break
            t_try /= 1.35

        improved_this_step = False
        if best_t > 0:
            h_new = h + best_t * d
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_now - 1e-15:
                h = h_new
                improved_this_step = True
                t_scale = min(t_scale * 1.05, 0.3)
                t_decay = min(t_decay * 1.02, 1.0)
            else:
                for frac in [0.5, 0.25, 0.1]:
                    h_try = h + best_t * frac * d
                    h_try = project_feasible(h_try, target)
                    if compute_c5(h_try) < c5_now - 1e-15:
                        h = h_try
                        improved_this_step = True
                        break
        if not improved_this_step:
            t_scale = max(t_scale * 0.9, 0.01)
            t_decay *= 0.98

    return best_h, best_c5

def coordinate_descent_v2(h_init, n, target, time_limit, n_cand=20, patience=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        top_shifts = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        pos_indices = np.where(grad > 0.01)[0]
        neg_indices = np.where(grad < -0.01)[0]
        if len(pos_indices) == 0 or len(neg_indices) == 0:
            no_improve += 3
            continue

        donor_idx = pos_indices[np.argsort(-grad[pos_indices])][:n_cand]
        recip_idx = neg_indices[np.argsort(grad[neg_indices])][:n_cand]

        # Precompute g for all candidates
        g_cache = {}
        for idx_arr, sign in [(donor_idx, -1), (recip_idx, 1)]:
            for idx in idx_arr:
                idx = int(idx)
                if idx in g_cache:
                    continue
                g_v = np.zeros(n_corr)
                for s in range(n_corr):
                    k = s - mid
                    val = 0.0
                    if 0 <= idx + k < n:
                        val += (1.0 - h[idx + k])
                    if 0 <= idx - k < n:
                        val -= h[idx - k]
                    g_cache[idx] = val

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in donor_idx[:12]:
            i = int(i)
            g_i = g_cache[i]
            for j in recip_idx[:12]:
                j = int(j)
                if i == j:
                    continue
                g_j = g_cache[j]
                g = g_j - g_i
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                a = 0.0
                b = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a + b)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-12
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b = mid_d
                    else:
                        a = mid_d
                delta_star = 0.5 * (a + b)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)

    return best_h, best_c5

def local_perturbation_search(h_init, n, target, max_tries, time_limit, seed=777):
    start = time.time()
    rng_local = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    scales = [0.01, 0.02, 0.04, 0.08, 0.12]

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 4, 6, 10, 15, 25]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice(scales))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Ray shooting with subgradient (main optimization)
    if tleft() > 200:
        rs_time = min(300, tleft() * 0.25)
        h_rs, c5_rs = ray_shooting_descent(best_h, n, target,
                                           time_limit=rs_time,
                                           top_m=10, alpha=80,
                                           use_momentum=True, mu=0.6,
                                           random_frac=0.1, seed=seed,
                                           patience=350)
        if c5_rs < best_score:
            best_score = c5_rs
            best_h = h_rs

    # Phase 2: Coordinate descent for fine-tuning
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.2)
        h_cd, c5_cd = coordinate_descent_v2(best_h, n, target,
                                            time_limit=cd_time,
                                            n_cand=20, patience=180)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B polish with multiple alphas
    if tleft() > 100:
        for alpha in [200, 800, 3000, 10000]:
            if tleft() < 30:
                break
            h_try = lbfags_smooth(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Second round of ray shooting (different params)
    if tleft() > 150:
        rs_time = min(200, tleft() * 0.2)
        h_rs, c5_rs = ray_shooting_descent(best_h, n, target,
                                           time_limit=rs_time,
                                           top_m=15, alpha=200,
                                           use_momentum=True, mu=0.7,
                                           random_frac=0.05, seed=seed+1,
                                           patience=300)
        if c5_rs < best_score:
            best_score = c5_rs
            best_h = h_rs

    # Phase 5: Coordinate descent round 2
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.15)
        h_cd, c5_cd = coordinate_descent_v2(best_h, n, target,
                                            time_limit=cd_time,
                                            n_cand=30, patience=150)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: SLSQP polish
    if tleft() > 80:
        for alpha in [5000, 20000]:
            if tleft() < 30:
                break
            h_try = slsqp_smooth(best_h, n, target, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 7: Multi-resolution
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            # Quick L-BFGS-B at higher resolution
            h_hires = lbfags_smooth(h_interp, n_try, target_try, 1000, maxiter=1000)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try
                # Further refine at this resolution
                if tleft() > 100:
                    h_rs, c5_rs = ray_shooting_descent(best_h, n, target,
                                                       time_limit=min(80, tleft()*0.3),
                                                       top_m=10, alpha=100,
                                                       use_momentum=True, mu=0.6,
                                                       random_frac=0.1, seed=seed+2,
                                                       patience=250)
                    if c5_rs < best_score:
                        best_score = c5_rs
                        best_h = h_rs
                    if tleft() > 60:
                        h_cd, c5_cd = coordinate_descent_v2(best_h, n, target,
                                                            time_limit=min(50, tleft()*0.3),
                                                            n_cand=20, patience=100)
                        if c5_cd < best_score:
                            best_score = c5_cd
                            best_h = h_cd

    # Phase 8: Local perturbation search
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert, c5_pert = local_perturbation_search(best_h, n_use, target_use,
                                                    max_tries=5000,
                                                    time_limit=min(40, tleft()*0.3),
                                                    seed=seed+3)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 9: Final round of ray shooting + CD
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_rs, c5_rs = ray_shooting_descent(best_h, n_use, target_use,
                                           time_limit=min(50, tleft()*0.4),
                                           top_m=8, alpha=50,
                                           use_momentum=False, mu=0.5,
                                           random_frac=0.15, seed=seed+4,
                                           patience=200)
        if c5_rs < best_score:
            best_score = c5_rs
            best_h = h_rs
        if tleft() > 40:
            h_cd, c5_cd = coordinate_descent_v2(best_h, n_use, target_use,
                                                time_limit=min(30, tleft()*0.4),
                                                n_cand=15, patience=80)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 10: Final L-BFGS-B polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_try = lbfags_smooth(best_h, n_use, target_use, 50000, maxiter=2000)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
