import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_max_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def _obj(x):
            return smooth_max_obj(x, alpha)
        
        def _grad(x):
            return smooth_max_grad(x, alpha, n)
        
        def _con(x):
            return np.sum(x) - target
        
        def _con_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lp_bisection(h0, n, target, T_target, time_limit, max_rounds=50, n_active=30):
    """Iterative linearization LP to check if C5 <= T_target is achievable."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        if c5_val <= T_target:
            break  # Feasible!
        
        # Find active constraints (lags near the maximum)
        sorted_lags = np.argsort(corr)[::-1]
        active = sorted_lags[:n_active]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val * dx <= t_var)
        
        constraints.append(t_var <= T_target)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            # Try mixing
            for mix in [0.3, 0.5, 0.7]:
                h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    break
            else:
                break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
