  sum_{k in A} λ_k * ∇g_k(h*) = μ * 1 (for the sum constraint)
  where μ is the Lagrange multiplier for the equality constraint.

This means the weighted sum of gradients of active shifts should be constant (proportional to 1 vector).

This gives us a optimality check! At the current point, if we compute the weighted gradient and it's not constant, we know we're not at the optimum and can take a step to fix this.

The direction should be to reduce the maximum while maintaining the constraint. A good direction is:
d = -(∇_weighted - mean(∇_weighted))

where ∇_weighted = sum λ_k * ∇g_k for the active shifts.

This is essentially what the subgradient methods do, but using only the active set (not all shifts with softmax weights) should be more focused.

Let me implement this as a "focused subgradient" method:
1. Identify active set (top few shifts)
2. Compute weighted gradient using only active shifts
3. Center it (subtract mean)
4. Take a step
5. Project

This should be faster and more focused than the full softmax approach.

    _GL_target = target

    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _GL_alpha = alpha
        try:
            res = minimize(_slsqp_lse_obj, h_cur, jac=_slsqp_lse_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def focused_subgrad(h_start, n, target, time_limit, max_steps, n_top, lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 600:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            try_lr = lr * lr_mult
            h_try = project_feasible(h - try_lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 3
    return best_h, best_c5

def precompute_f(h, n):
    f = -np.ones((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    for p in range(n):
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += h[idx1[mask1]]
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    return f

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        f_arr = precompute_f(h, n)
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = f_arr[i] - f_arr[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = project_feasible(h, target)
    return best_h, best_c5

def nesterov_top(h_start, n, target, time_limit, max_steps, n_top, lr, mu):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 800:
            break
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr_y)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(y, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        y_new = y - lr * grad
        h_new = (1.0 - mu) * h + mu * y_new
        h_new = project_feasible(h_new, target)
        y = project_feasible(h_new + mu * (h_new - h), target)
        h = h_new
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if step % 100 == 0 and step > 0:
            mu = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def local_perturb_refine(h_init, n, target, max_tries, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.01, 0.03, 0.05, 0.08, 0.12]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 300:
        h_pol, sc = slsqp_grad(best_h, n, target,
                                [50, 200, 1000, 5000, 20000, 100000, 500000],
                                3000, 1e-18)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_fg, sc = focused_subgrad(best_h, n, target,
                                    time_limit=min(80, tleft() * 0.3),
                                    max_steps=50000, n_top=5, lr=0.01)
        if sc < best_score:
            best_score = sc
            best_h = h_fg

    if tleft() > 250:
        h_pol, sc = slsqp_grad(best_h, n, target,
                                [5000, 50000, 500000, 2000000],
                                2500, 1e-18)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(100, tleft() * 0.4),
                           max_sweeps=5000, n_cand=25)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 200:
        h_pol, sc = slsqp_grad(best_h, n, target,
                                [10000, 100000, 1000000],
                                2000, 1e-18)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_nes, sc = nesterov_top(best_h, n, target,
                                  time_limit=min(50, tleft() * 0.3),
                                  max_steps=30000, n_top=5, lr=0.008, mu=0.9)
        if sc < best_score:
            best_score = sc
            best_h = h_nes

    if tleft() > 150:
        h_cd, sc = pair_cd(best_h, n, target,
                           time_limit=min(60, tleft() * 0.4),
                           max_sweeps=3000, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    rng = np.random.RandomState(seed)
    if tleft() > 200:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 100:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref, sc = slsqp_grad(h_pert, n, target,
                                    [500, 5000, 50000, 500000],
                                    1500, 1e-16)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 200:
        n_try = 600
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_hires, sc = slsqp_grad(h_interp, n_try, target_try,
                                  [100, 1000, 10000, 100000, 1000000],
                                  2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_hires
            n = n_try
            target = target_try

    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, sc = pair_cd(best_h, n_use, target_use,
                           time_limit=min(50, tleft() * 0.4),
                           max_sweeps=2000, n_cand=20)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 60:
        h_lp, sc = local_perturb_refine(best_h, len(best_h), len(best_h) / 2.0,
                                         max_tries=2000,
                                         time_limit=min(30, tleft() * 0.4), rng=rng)
        if sc < best_score:
            best_score = sc
            best_h = h_lp

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_grad(best_h, n_use, target_use,
                                  [100000, 1000000, 5000000],
                                  1000, 1e-18)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
