                # For positions i where i+s < n: (1-h[i+s])
                valid_first = range(0, n - s)
                grad[:n-s] += wk * (1.0 - h[s:])
                # For positions i where i-s >= 0: -h[i-s]
                grad[s:] += wk * (-h[:n-s])
            else:
                abs_s = -s
                # For positions i where i+abs_s < n: (1-h[i+abs_s])
                grad[:n-abs_s] += wk * (1.0 - h[abs_s:])
                # For positions i where i-abs_s >= 0: -h[i-abs_s]
                grad[abs_s:] += wk * (-h[:n-abs_s])

        # Momentum update
        v = momentum * v - lr * grad
        h = project_box_sum(h + v, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-14:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def binary_greedy_multi(h, max_iters, rng, time_limit, start_time, topk_shifts=5, n_candidates=80):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break

        max_val = float(full.max())
        
        # Get top shifts
        top_idx = np.argsort(full)[-topk_shifts:]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # For each top shift, find good candidates
        candidate_pairs = set()
        for k_idx in top_idx:
            k_star = int(k_idx) - (n - 1)
            if k_star == 0:
                continue
            # For this shift, we want to reduce c_k_star
            # delta at shift k_star for swap (i:1→0, j:0→1):
            # = h[i+k*] + h[i-k*] - h[j-k*] - h[j+k*]
            # We want this to be negative (reduce c at k_star)
            # So we want: h[i+k*] + h[i-k*] < h[j-k*] + h[j+k*]
            
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            
            # For ones (i candidates): want low A[i]
            top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
            # For zeros (j candidates): want high B[j]
            top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
            
            for i in top_i[:10]:
                for j in top_j[:10]:
                    candidate_pairs.add((int(i), int(j)))

        if len(candidate_pairs) == 0:
            # Fallback: random
            candidate_pairs = set()
            for _ in range(n_candidates):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                candidate_pairs.add((i, j))

        found = False
        # Evaluate all candidates
        best_delta = None
        best_new_max = max_val
        best_pair = None
        for (i, j) in list(candidate_pairs)[:n_candidates]:
            if h[i] != 1.0 or h[j] != 0.0:
                continue
            delta = compute_delta(h, i, j)
            new_full = full + delta
            nm = float(new_full.max())
            if nm < best_new_max:
                best_new_max = nm
                best_pair = (i, j)
                best_delta = delta
                if nm < max_val - 0.1:  # Good enough
                    break

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            full += best_delta
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stall += 1

    return best_h, best_c5

def binary_sa_block(h, n_iters, t_init, t_final, rng, time_limit, start_time, block_sizes=(1, 2, 3)):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 30000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        mode = rng.random()
        if mode < 0.7:
            # Single pair swap
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0 and rng.random() < 0.7:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                candidates_i = ones[A[ones] <= 1.0]
                if len(candidates_i) == 0:
                    candidates_i = ones
                i = int(candidates_i[rng.randint(len(candidates_i))])
                j = int(zeros[rng.randint(len(zeros))])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
            
            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Block swap: flip two pairs simultaneously
            i1 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            i2 = int(ones[rng.randint(len(ones))])
            j2 = int(zeros[rng.randint(len(zeros))])
            
            if i1 == i2 or j1 == j2:
                stall += 1
                continue
                
            delta1 = compute_delta(h, i1, j1)
            # After first swap, h changes, so we need to recompute
            # But for small perturbations, approximate as sum
            delta2_approx = compute_delta(h, i2, j2)
            delta = delta1 + delta2_approx
            
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / (t * 1.5))):
                h[i1] = 0.0
                h[j1] = 1.0
                full += delta1
                h[i2] = 0.0
                h[j2] = 1.0
                # Recompute delta2 with updated h
                # Actually, we already added delta1, so let's just recompute full
                full = np.correlate(h, 1.0 - h, mode='full')
                current_c5 = float(full.max()) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_box_sum(y, target):
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) < 1e-13:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-13:
                break
            can = y[idx]
            if can > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-13:
                break
            can = 1.0 - y[idx]
            if can > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can
                y[idx] = 1.0
    return y

def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def momentum_grad_descent(h, n_iters, alpha_init, alpha_end, lr_init, lr_end, momentum, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    v = np.zeros(n)
    stall = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 2000:
            nr = int(n * 0.02)
            nz = int(n * 0.02)
            h = h.copy()
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            if len(ones_idx) > nr and len(zeros_idx) > nz:
                pi = np.random.choice(ones_idx, min(nr, len(ones_idx)), replace=False)
                pj = np.random.choice(zeros_idx, min(nz, len(zeros_idx)), replace=False)
                h[pi] = 0.0
                h[pj] = 1.0
                h = project_box_sum(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                if float(corr.max()) * dx < best_c5:
                    pass
                stall = 0
            else:
                break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        alpha = alpha_init * (alpha_end / alpha_init) ** ratio

        max_corr = float(corr.max())
        shifted = alpha * (corr - max_corr)
        shifted = np.clip(shifted, -400.0, 0.0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()

        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = W[k_idx]
            if wk < 1e-10:
                continue
            s = shifts[k_idx]
            if s >= 0:
                grad[:n - s] += wk * (1.0 - h[s:])
                grad[s:] += wk * (-h[:n - s])
            else:
                abs_s = -s
                grad[:n - abs_s] += wk * (1.0 - h[abs_s:])
                grad[abs_s:] += wk * (-h[:n - abs_s])

        v = momentum * v - lr * grad
        h = project_box_sum(h + v, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-14:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def binary_greedy_multi(h, max_iters, rng, time_limit, start_time, topk_shifts=8, n_candidates=120):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 300:
            break

        max_val = float(full.max())
        top_idx = np.argsort(full)[-topk_shifts:]

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        candidate_pairs = []
        seen = set()
        for k_idx in top_idx:
            k_star = int(k_idx) - (n - 1)
            if k_star == 0:
                continue
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]

            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]

            top_i = ones[np.argsort(A[ones])[:min(25, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(25, len(zeros))]]

            for i in top_i[:12]:
                for j in top_j[:12]:
                    key = (int(i), int(j))
                    if key not in seen:
                        seen.add(key)
                        candidate_pairs.append(key)

        if len(candidate_pairs) == 0:
            for _ in range(min(n_candidates, 40)):
                candidate_pairs.append((int(ones[rng.randint(len(ones))]),
                                       int(zeros[rng.randint(len(zeros))))))

        best_new_max = max_val
        best_pair = None
        best_delta = None
        for (i, j) in candidate_pairs[:n_candidates]:
            if h[i] != 1.0 or h[j] != 0.0:
                continue
            delta = compute_delta(h, i, j)
            nm = float((full + delta).max())
            if nm < best_new_max:
                best_new_max = nm
                best_pair = (i, j)
                best_delta = delta
                if best_new_max < max_val - 0.5:
                    break

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            full += best_delta
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stall += 1

    return best_h, best_c5

def binary_sa_block(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 25000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        mode = rng.random()
        if mode < 0.75:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0 and rng.random() < 0.8:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                candidates_i = ones[A[ones] <= 1.0]
                if len(candidates_i) == 0:
                    candidates_i = ones
                i = int(candidates_i[rng.randint(len(candidates_i))])
                j = int(zeros[rng.randint(len(zeros))])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])

            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i1 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            i2 = int(ones[rng.randint(len(ones))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if i1 == i2 or j1 == j2 or i1 == j2 or i2 == j1:
                stall += 1
                continue
            h[i1] = 0.0
            h[j1] = 1.0
            h[i2] = 0.0
            h[j2] = 1.0
            new_full = np.correlate(h, 1.0 - h, mode='full')
            c5_new = float(new_full.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / (t * 1.5))):
                full = new_full
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i1] = 1.0
                h[j1] = 0.0
                h[i2] = 1.0
                h[j2] = 0.0
                stall += 1

    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)
