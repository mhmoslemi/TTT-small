            if 0 <= i - k < n:
                g += (1.0 - h[i - k])
            if 0 <= i + k < n:
                g -= h[i + k]
            grad[i] += pk * g * dx
    
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    curr = np.sum(h)
    diff = target_sum - curr
    if abs(diff) < 1e-14:
        return h
    # L2 projection via bisection
    lo, hi = -1.0, 1.0
    if diff > 0:
        lo, hi = 0.0, 1.0

        lo, hi = -1.0, 0.0
    for _ in range(200):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    tau = 0.5 * (lo + hi)
    result = np.clip(h + tau, 0.0, 1.0)
    # Final fix
    residual = target_sum - np.sum(result)

        else:

            h[best_i_idx] -= best_delta
            h[best_j_idx] += best_delta
            h = project_feasible(h, target_sum)
    
    return best_h, best_c5

    lo = h.min()

    h = np.clip(h, 0.0, 1.0)
    curr = h.sum()
    diff = target_sum - curr
    if abs(diff) < 1e-14:
        return h
    # Bisection for L2 projection
    a, b = -1.0, 1.0
    if diff > 0:
        a = 0.0
    else:
        b = 0.0
    for _ in range(100):
        m = 0.5 * (a + b)
        s = np.clip(h + m, 0.0, 1.0).sum()
        if s > target_sum:
            b = m
        else:
            a = m
    tau = 0.5 * (a + b)
    result = np.clip(h + tau, 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-14)[0]
        else:
            idx = np.where(result > 1e-14)[0]
        if len(idx) > 0:
            result = result.copy()
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -300.0, 0.0)
    return cmax + np.log(np.sum(np.exp(shifted)) / len(corr))

def compute_infl(h):
    n = len(h)
    n_corr = 2 * n - 1
    infl = np.zeros((n, n_corr))
    omh_rev = (1.0 - h)[::-1]
    for i in range(n):
        infl[i, i:i+n] += omh_rev
        infl[i, n-1-i:2*n-1-i] -= h
    return infl

def pair_cd_fast(h_init, n, target_sum, time_limit, max_sweeps=50000, n_cand_start=30, n_cand_end=80, patience=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        frac = min(1.0, sweep / max(1, patience))
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * frac)
        
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        infl = compute_infl(h)
        
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_score = c5_val
        
        for i_pos in range(len(high_set)):
            i = high_set[i_pos]
            for j_pos in range(len(low_set)):
                j = low_set[j_pos]
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = infl[j] - infl[i]
                
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e-10)
                if b_d - a_d < 1e-10:
                    b_d = a_d + 1e-10
                
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = np.max(g[active])
                    else:
                        fp = 0.0
                    if fp > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx = float(np.max(corr + delta_star * g)) * dx
                if approx < best_score:
                    best_score = approx
                    best_delta = delta_star
                    best_i = int(i)
                    best_j = int(j)
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target_sum)
    
    return best_h, best_c5

def smoothed_subgrad(h_init, n, target_sum, time_limit, alphas=None, max_steps=20000, base_lr=0.05, mu=0.7):
    start = time.time()
    if alphas is None:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000]
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    no_improve = 0
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        alpha = alphas[min(step // (max_steps // len(alphas)), len(alphas)-1)]
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        shifted = np.clip(alpha * (corr - cmax), -300.0, 0.0)
        p = np.exp(shifted)
        p /= p.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-12)[0]
        for k_idx in active:
            k = int(k_idx) - mid
            pk = p[k_idx] * dx
            v1 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += pk * (1.0 - h[idxs[v1] - k])
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v2] -= pk * h[idxs[v2] + k]
        
        grad -= grad.mean()
        v = mu * v + (1 - mu) * grad
        lr = base_lr * max(0.1, 1.0 - step / max_steps)
        
        c5_now = compute_c5(h)
        for bt in range(15):
            h_try = project_feasible(h - lr * v, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                break
            lr *= 0.5
        
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
    
    return best_h, best_c5
