def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        y = y.copy()
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = y[idx]
            if can_reduce > remaining:
                y[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                y[idx] = 0.0
    else:
        order = np.argsort(y)
        y = y.copy()
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - y[idx]
            if can_inc > remaining:
                y[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                y[idx] = 1.0
    return y

def projected_grad_descent(h, n_iters, topk, lr_init, lr_end, time_limit, start_time):

            break
        if stall > max_stall:
            break

        lr = lr_init * (lr_end / lr_init) ** ratio

        topk_idx = np.argsort(corr)[-topk:]

        for k in topk_idx:

                abs_s = -s

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
