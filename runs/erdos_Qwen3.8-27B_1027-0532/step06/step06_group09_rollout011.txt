```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(20):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h


def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def gradient_at_lag(h, k, n):
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad


def game_theoretic_phase(h, n, target, time_limit, eta=4.0, m_top=12, temp=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1

    corr0 = np.correlate(h, 1.0 - h, mode='full')
    lam = np.maximum(corr0, 1e-10)
    lam = lam / lam.sum()

    no_improve = 0
    step = 0
    lr_base = 0.08

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                break

        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / lam_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            grad += wi * gradient_at_lag(h, k, n)
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = lr_base * max(0.05, 1.0 - step / 8000.0)
        for _ in range(12):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            lr *= 0.45

        if not improved:
            lr_base *= 0.92
            if lr_base < 0.003:
                break

        exp_arg = np.clip(eta * (corr - c_max) / (c_max + 1e-10), -40, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / lam_new.sum()
        lam = (1.0 - temp) * lam + temp * lam_new
        step += 1

    return best_h, best_c5


def fast_pair_cd(h, n, target, time_limit, n_cand_start=35, n_cand_max=100, patience=250):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    n_cand = n_cand_start

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

        if no_improve > patience * 0.35:
            n_cand = min(n_cand + 8, n_cand_max)
        elif no_improve < patience * 0.1:
            n_cand = max(n_cand - 4, n_cand_start)

        top_lags = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for s_idx in top_lags:
            k = int(s_idx) - mid
            grad += gradient_at_lag(h, k, n)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        hi_grads = np.zeros((len(high_set), n_corr))
        lo_grads = np.zeros((len(low_set), n_corr))

        for ii, i in enumerate(high_set):
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            hi_grads[ii, v_ip] = 1.0 - h[i + shifts[v_ip]]
            hi_grads[ii, v_im] -= h[i - shifts[v_im]]

        for jj, j in enumerate(low_set):
            jp = (j + shifts >= 0) & (j + shifts < n)
            jm = (j - shifts >= 0) & (j - shifts < n)
            lo_grads[jj, jp] = 1.0 - h[j + shifts[jp]]
            lo_grads[jj, jm] -= h[j - shifts[jm]]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        npair_h = len(high_set)
        npair_l = len(low_set)

        for ii in range(npair_h):
            i = int(high_set[ii])
            g_i = hi_grads[ii]
            for jj in range(npair_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(0.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = g_i - lo_grads[jj]
                a_d = -1e-10
                b_d = 1e-10
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def evolutionary_phase(h, n, target, time_limit, pop_size=10, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)

    pop = [h.copy()]
    for i in range(1, pop_size):
        scale = 0.03 + 0.08 * i / pop_size
        pop.append(project_feasible(h + rng.randn(n) * scale, target))

    fitness = [compute_c5(p) for p in pop]
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]

    sigma = 0.04
    no_improve = 0

    while time.time() - start < time_limit:
        cands = rng.choice(pop_size, size=min(3, pop_size), replace=False)
        parent_idx = int(cands[np.argmin([fitness[c] for c in cands])])

        n_pert = int(rng.choice([2, 3, 5, 8, 12, 16]))
        n_pert = min(n_pert, n)
        pert_idxs = rng.choice(n, size=n_pert, replace=False)
        mutant = pop[parent_idx].copy()
        mutant[pert_idxs] += rng.randn(n_pert) * sigma
        mutant = project_feasible(mutant, target)
        mut_fit = compute_c5(mutant)

        if mut_fit < best_c5 - 1e-13:
            best_c5 = mut_fit
            best_h = mutant.copy()
            no_improve = 0
            sigma *= 0.88
        else:
            no_improve += 1
            sigma = min(sigma * 1.03, 0.25)

        if no_improve > 400:
            break

        worst_idx = int(np.argmax(fitness))
        pop[worst_idx] = mutant.copy()
        fitness[worst_idx] = mut_fit

        if no_improve > 150 and rng.random() < 0.15:
            esc = project_feasible(best_h + rng.randn(n) * min(sigma * 2.5, 0.2), target)
            esc_fit = compute_c5(esc)
            pop[int(np.argmax(fitness))] = esc
            fitness[int(np.argmax(fitness))] = esc_fit

    return best_h, best_c5


def lbfags_polish(h_start, n, target, alpha, maxiter=1500):
    dx = 2.0 / n
    penalty_weight = 200.0

    def obj_pen(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm = smooth_max(corr, alpha)
        penalty = penalty_weight * (np.sum(h) - target) ** 2
        return sm * dx + penalty

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    dx = 2.0 / n
    penalty_weight = 500.0

    def obj_func(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm = smooth_max(corr, alpha)
        penalty = penalty_weight * (np.sum(h) - target) ** 2
        return sm * dx + penalty

    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj_func(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
```
