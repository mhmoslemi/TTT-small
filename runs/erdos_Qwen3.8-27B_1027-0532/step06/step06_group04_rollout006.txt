    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:

        idx = np.where(h > 1e-10)[0]

    if abs(residual) > 1e-10:

            idx2 = np.where(h < 1.0)[0]
        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(slopes)
    if n == 0:
        return 0.0

    # At x=1: max of (slopes + intercepts)
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    if val_0 <= val_1 + 1e-14:

        # Check if the function is non-decreasing everywhere
        # If all slopes >= 0, answer is 0
        if np.all(slopes >= -1e-14):
            return 0.0
    if val_1 <= val_0 + 1e-14:
        if np.all(slopes <= 1e-14):
            return 1.0
    
    # Binary search for the point where the max is minimized
    lo, hi = 0.0, 1.0
    for _ in range(50):

        vals = slopes * mid + intercepts
        max_val = np.max(vals)

            hi = mid
        elif np.all(active_slopes < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

    best_c5 = compute_c5(h)

    # i.e., k ∈ [-i, n-1-i] ∪ [i-(n-1), i]
    # But we can just compute on the fly

            #   slope contribution: (1-h[i+k]) * dx

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max(slopes, intercepts):
    """Solve min_{x in [0,1]} max_j (slopes[j]*x + intercepts[j]) efficiently."""
    # Quick checks
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    if val_0 <= val_1:
        # Check if non-decreasing
        if np.all(slopes >= -1e-15):
            return 0.0
    else:
        if np.all(slopes <= 1e-15):
            return 1.0
    
    # Binary search
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid + intercepts
        mx = np.max(vals)
        tol = 1e-13 * max(1.0, abs(mx))
        active = vals >= mx - tol
        if not np.any(active):
            active = np.zeros(len(slopes), dtype=bool)
            active[np.argmax(vals)] = True
        s = slopes[active]
        if np.min(s) > 1e-13:
            hi = mid
        elif np.max(s) < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent_full(h_init, target, max_passes, time_limit, rng, n_lags='all'):
    """Coordinate descent. n_lags='all' uses all lags, or an int for top-k."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    total_lags = 2 * n - 1
    
    # Precompute lag offsets
    k_all = np.arange(total_lags) - half_n  # shape (2n-1,)
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Determine which lags to use
        if n_lags == 'all' or n_lags >= total_lags:
            lag_indices = np.arange(total_lags)
        else:
            lag_indices = np.argsort(corr)[-n_lags:]
        
        k_sel = k_all[lag_indices]  # actual lag values for selected lags
        corr_sel = corr[lag_indices]
        
        # Precompute valid masks for each i
        order = rng.permutation(n)
        for i in order:
            valid1 = (i + k_sel >= 0) & (i + k_sel < n)
            valid2 = (i - k_sel >= 0) & (i - k_sel < n)
            
            slopes = np.zeros(len(k_sel))
            slopes[valid1] += (1.0 - h[i + k_sel[valid1]]) * dx
            slopes[valid2] -= h[i - k_sel[valid2]] * dx
            
            intercepts = corr_sel - slopes * h[i]
            h[i] = solve_1d_max(slopes, intercepts)
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

```python
def prox_simplex_box(z, target):
    """Project z onto {h: sum(h)=target, 0<=h<=1}."""
    n = len(z)
    # Bisection on threshold τ such that sum(clip(z - τ, 0, 1)) = target
    lo = z.min() - 1
    hi = z.max() + 1
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5*(lo+hi), 0.0, 1.0)

def proximal_gradient(h_init, target, max_iter, time_limit, init_step=0.01):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    step = init_step
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - half_n
        
        # Compute gradient of corr[k_star] with respect to h
        grad = np.zeros(n)
        idxs = np.arange(n)
        valid1 = (idxs + k >= 0) & (idxs + k < n)
        valid2 = (idxs - k >= 0) & (idxs - k < n)
        grad[valid1] += (1.0 - h[idxs[valid1] + k])
        grad[valid2] -= h[idxs[valid2] - k]
        grad *= dx
        
        # Proximal gradient step
        z = h - step * grad
        h_new = prox_simplex_box(z, target)
        
        # Line search (backtracking)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Simple step size adaptation
        if c5_new < compute_c5(h):
            step = min(step * 1.1, 0.5)
        else:
            step = max(step * 0.7, 0.001)
        
        h = h_new
    
    return best_h, best_c5

The issue with proximal gradient is that it only uses the gradient of the single active constraint (the max one). This is a subgradient method, which converges slowly. To speed it up, we could use:
- **Smoothing**: Use log-sum-exp approximation of max
- **Multiple active constraints**: Use a weighted combination of the top few active constraints
- **Momentum**: Add Nesterov momentum

Let me use the **smoothed proximal gradient** (a.k.a. accelerated proximal gradient for nonsmooth objectives):

f(h) ≈ (1/α) * log(sum_k exp(α * corr_k(h) * dx))

∇f_smooth(h) = (1/sum_k exp(α * corr_k * dx)) * sum_k exp(α * corr_k * dx) * ∇corr_k(h) * dx

This gives a smooth approximation whose gradient is a weighted average of the individual gradients. For large α, it approaches the true gradient of the max.

The computational cost is: O(n^2) for all correlations + O(n * 2n) for the weighted gradient = O(n^2).

Actually, computing ∇corr_k(h) for all k and taking a weighted sum is expensive. Let me think of a more efficient way.

∇corr_k(h)[i] = dx * [(1-h[i+k]) * 1{0≤i+k<n} - h[i-k] * 1{0≤i-k<n}]

So the weighted gradient is:
∇f_smooth(h)[i] = dx * sum_k w_k * [(1-h[i+k]) * 1{valid} - h[i-k] * 1{valid}]

where w_k = exp(α * corr_k * dx) / sum_j exp(α * corr_j * dx)

This is like a weighted average of the "shifted" versions of h. Can we compute this efficiently?

sum_k w_k * h[i-k] * 1{0≤i-k<n} = sum_j w_{i-j} * h[j] * 1{0≤i-j<n}

This is a convolution of h with w! So we can use FFT to compute it in O(n log n).

Similarly, sum_k w_k * (1-h[i+k]) = sum_j w_k * 1 - sum_j w_k * h[j] where the indexing works out to another convolution.

So the total per iteration is:
1. Compute all correlations: O(n^2) [or O(n log n) with FFT, but np.correlate is probably fine]
2. Compute weights: O(n)
3. Compute gradient via convolution: O(n log n)
4. Proximal step: O(n log n) (bisection on threshold)

Total: O(n^2) dominated by the correlation computation.

For the smoothed objective, we also need to compute f_smooth itself for line search, which is O(n) once we have the correlations.

The 1D min-max can be solved by:
- The optimal x_i is where the upper envelope of the lines {slope[i,k] * x + intercept[i,k]} achieves its minimum on [0,1]
- This is a convex piecewise-linear function in x
- The minimum is at the point where the subdifferential contains 0, or at 0 or 1

def solve_1d_max_vectorized(slopes, intercepts):
    """
    For each row i, solve min_{x∈[0,1]} max_j (slopes[i,j]*x + intercepts[i,j])
    Returns optimal x for each i.
    slopes, intercepts: shape (n, m)
    """
    n, m = slopes.shape
    lo = np.zeros(n)
    hi = np.ones(n)
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid[:, None] + intercepts  # shape (n, m)
        max_vals = np.max(vals, axis=1)  # shape (n,)
        tol = 1e-13 * np.maximum(1.0, np.abs(max_vals))
        active = (vals >= max_vals[:, None] - tol[:, None])  # shape (n, m)
        # For rows with no active, take argmax
        no_active = ~np.any(active, axis=1)
        if np.any(no_active):
            argmax_idx = np.argmax(vals[no_active], axis=1)
            active[no_active, argmax_idx] = True
        
        active_slopes_min = np.where(active, slopes, np.inf).min(axis=1)
        active_slopes_max = np.where(active, slopes, -np.inf).max(axis=1)
        
        moving_hi = active_slopes_min > 1e-13
        moving_lo = active_slopes_max < -1e-13
        hi[moving_hi] = mid[moving_hi]
        lo[moving_lo] = mid[moving_lo]
    
    return 0.5 * (lo + hi)
```

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def vec_solve_1d_max(slopes, intercepts):
    n, m = slopes.shape
    lo = np.zeros(n)
    hi = np.ones(n)
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid[:, None] + intercepts
        max_vals = np.max(vals, axis=1)
        tol = 1e-13 * np.maximum(1.0, np.abs(max_vals))
        active = (vals >= max_vals[:, None] - tol[:, None])
        no_active = ~np.any(active, axis=1)
        if np.any(no_active):
            idx = np.argmax(vals[no_active], axis=1)
            rows = np.where(no_active)[0]
            active[rows, idx] = True
        amin = np.where(active, slopes, np.inf).min(axis=1)
        amax = np.where(active, slopes, -np.inf).max(axis=1)
        hi[amin > 1e-13] = mid[amin > 1e-13]
        lo[amax < -1e-13] = mid[amax < -1e-13]
    return 0.5 * (lo + hi)
