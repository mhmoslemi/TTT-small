    c_max = np.max(corr)

    sum_exp = np.sum(exp_shifted)

    mid = n - 1
    grad = np.zeros(n)

def pair_cd_fast(h_start, n, target, time_limit, max_sweeps=10000, n_cand=40, top_k=5):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:]
        soft_a = 100.0
        w_top = np.exp(soft_a * (corr[top_idx] - c_max))
        w_top = w_top / w_top.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w_top, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def simulated_annealing_cd(h_start, n, target, time_limit, max_iters=100000):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)
    T = 0.05
    cool = 0.9997

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        T *= cool
        if T < 0.001:
            T = 0.05
            cool = 0.9997

        n_pert = int(rng.choice([1, 2, 3, 5]))
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.01, 0.03, 0.05, 0.1]))
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        c5_cur = compute_c5(h)

        if c5_try < c5_cur or rng.random() < np.exp(-(c5_try - c5_cur) / max(T, 1e-10)):
            h = h_try
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_active_cd(h_start, n, target, time_limit, max_sweeps=5000):
    start_t = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    mid = n - 1

        if c5_val < best_c5 - 1e-13:

            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        c_max = np.max(corr)
        threshold = c_max - 0.02 * c_max
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for s_idx in active:

            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        n_cand = max(20, n // 15)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1

        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:

            h[best_j] += best_delta

            h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast subgradient at original resolution
    if tleft() > 200:
        h_sg, c5_sg = fast_subgradient_method(best_h, n, target,
                                             time_limit=min(180, tleft()),
                                             max_steps=80000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 2: SLSQP polish
    if tleft() > 150:
        h_poly = slsqp_opt(best_h, n, target, [50, 200, 1000], 2000, min(120, tleft()))
        sc = compute_c5(h_poly)
        if sc < best_score:
            best_score = sc
            best_h = h_poly

    # Phase 3: Multi-active pair CD
    if tleft() > 200:
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                      time_limit=min(150, tleft()),
                                      max_sweeps=8000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Simulated annealing to escape local optima
    if tleft() > 150:
        h_sa, c5_sa = simulated_annealing_cd(best_h, n, target,
                                             time_limit=min(100, tleft()),
                                             max_iters=50000)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 5: SLSQP after SA
    if tleft() > 100:
        h_poly = slsqp_opt(best_h, n, target, [200, 1000, 5000], 1500, min(80, tleft()))
        sc = compute_c5(h_poly)
        if sc < best_score:
            best_score = sc
            best_h = h_poly

    # Phase 6: Try higher resolution n=500
    if tleft() > 300:
        n_hi = 500
        target_hi = n_hi / 2.0
        h_hi = interpolate_to(best_h, n_hi, target_hi)
        sc_hi = compute_c5(h_hi)

        if tleft() > 250:
            h_hi2, c5_hi2 = fast_subgradient_method(h_hi, n_hi, target_hi,
                                                    time_limit=min(100, tleft()),
                                                    max_steps=40000)
            if c5_hi2 < sc_hi:
                sc_hi = c5_hi2
                h_hi = h_hi2

        if tleft() > 150:
            h_poly_hi = slsqp_opt(h_hi, n_hi, target_hi, [50, 200, 1000], 1500, min(100, tleft()))
            sc_p = compute_c5(h_poly_hi)
            if sc_p < sc_hi:
                sc_hi = sc_p
                h_hi = h_poly_hi

        if tleft() > 100:
            h_cd_hi, c5_cd_hi = pair_cd_fast(h_hi, n_hi, target_hi,
                                             time_limit=min(80, tleft()),
                                             max_sweeps=5000, n_cand=35, top_k=5)
            if c5_cd_hi < sc_hi:
                sc_hi = c5_cd_hi
                h_hi = h_cd_hi

        if sc_hi < best_score:
            best_score = sc_hi
            best_h = h_hi
            n = n_hi
            target = target_hi

    # Phase 7: Try n=600
    if tleft() > 250:
        n_hi2 = 600
        target_hi2 = n_hi2 / 2.0
        h_hi2 = interpolate_to(best_h, n_hi2, target_hi2)
        sc_hi2 = compute_c5(h_hi2)

        if tleft() > 200:
            h_hi3, c5_hi3 = fast_subgradient_method(h_hi2, n_hi2, target_hi2,
                                                    time_limit=min(80, tleft()),
                                                    max_steps=30000)
            if c5_hi3 < sc_hi2:
                sc_hi2 = c5_hi3
                h_hi2 = h_hi3

        if tleft() > 120:
            h_poly2 = slsqp_opt(h_hi2, n_hi2, target_hi2, [50, 500], 1000, min(80, tleft()))
            sc_p2 = compute_c5(h_poly2)
            if sc_p2 < sc_hi2:
                sc_hi2 = sc_p2
                h_hi2 = h_poly2

        if tleft() > 80:
            h_cd2, c5_cd2 = pair_cd_fast(h_hi2, n_hi2, target_hi2,
                                         time_limit=min(60, tleft()),
                                         max_sweeps=4000, n_cand=30, top_k=4)
            if c5_cd2 < sc_hi2:
                sc_hi2 = c5_cd2
                h_hi2 = h_cd2

        if sc_hi2 < best_score:
            best_score = sc_hi2
            best_h = h_hi2
            n = n_hi2
            target = target_hi2

    # Phase 8: Final intensive pair CD at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 80:
        h_cd, c5_cd = pair_cd_fast(best_h, n_use, target_use,
                                   time_limit=min(60, tleft()),
                                   max_sweeps=5000, n_cand=50, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final subgradient sweep
    if tleft() > 50:
        h_sg2, c5_sg2 = fast_subgradient_method(best_h, n_use, target_use,
                                                time_limit=min(40, tleft()),
                                                max_steps=30000)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    # Phase 10: Final SLSQP
    if tleft() > 25:
        h_final = slsqp_opt(best_h, n_use, target_use, [1000, 5000], 500, min(20, tleft()))
        sc_f = compute_c5(h_final)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, len(h_out)
