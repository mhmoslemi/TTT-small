def project_binary(h, target_sum):

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):

    final_sum = int(h_r.sum())

            final_sum -= 1
        else:

            final_sum += 1
    return h_r.astype(float)

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)

        except Exception:

    return project_box_sum(h_cur, target)

def greedy_binary_fast(h, time_limit, start_time, max_iter=500000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(int)
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find shifts with high correlation
        threshold = max_val * 0.85
        hot_shifts = np.where(corr >= threshold)[0]
        
        # Score each position
        # For a one at position a: removing it changes C[k] by -(1-h[a+k]) + h[a-k]
        # For a zero at position b: adding it changes C[k] by (1-h[b+k]) - h[b-k]
        # We want to minimize the max of (C[k] + delta) over k
        
        # Fast approach: for each candidate (a,b), compute the worst shift after the swap
        # Limit candidates to reasonable number
        n_scan = min(len(ones), 50)
        n_scan_z = min(len(zeros), 50)
        
        # Heuristic selection: pick ones that contribute most to hot shifts,
        # and zeros that would most reduce hot shifts
        a_candidates = ones[:n_scan]  # We'll refine this
        b_candidates = zeros[:n_scan_z]
        
        best_improve = 1e-10
        best_a = -1
        best_b = -1
        
        for a in a_candidates:
            # delta from removing 1 at a: for shift k, -(1-h[a+k]) + h[a-k]
            a_effect = np.zeros(len(corr))
            for idx, m in enumerate(hot_shifts):
                k = m - (n - 1)
                # -(1-h[a+k]) where a+k valid
                if 0 <= a + k < n:
                    a_effect[m] -= (1.0 - h[a + k])
                else:
                    a_effect[m] -= 1.0  # h outside = 0, so 1-h = 1
                if 0 <= a - k < n:
                    a_effect[m] += h[a - k]
            
            for b in b_candidates:
                # delta from adding 1 at b: (1-h[b+k]) - h[b-k]
                worst = 0.0
                for idx, m in enumerate(hot_shifts):
                    k = m - (n - 1)
                    b_effect = 0.0
                    if 0 <= b + k < n:
                        b_effect += (1.0 - h[b + k])
                    else:
                        b_effect += 1.0
                    if 0 <= b - k < n:
                        b_effect -= h[b - k]
                    
                    new_val = corr[m] + a_effect[m] + b_effect
                    if new_val > worst:
                        worst = new_val
                        if worst >= max_val - best_improve:
                            break
                
                if worst < max_val - best_improve:
                    best_improve = max_val - worst
                    best_a = a
                    best_b = b
        
        if best_a >= 0:
            h[best_a] = 0
            h[best_b] = 1
            corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h.copy()
        else:
            break
    
    return best_h.astype(float), best_c5

def sa_binary_improved(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time,

    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 20000
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        h_new = h.copy()
        moved = False
        
        if r < 0.6:
            # Single swap, biased towards "worst" contributions
            if use_warm_start:
                # Pick from positions that contribute to high-correlation shifts
                top_k = 5
                hot = np.argsort(corr)[-top_k:]
                # For each hot shift, find positions that contribute
                # A one at position a contributes (1-h[a+k]) to shift k
                # So positions where h[a+k]=0 are "bad" ones

                hot_zeros = set()
                for m in hot:
                    k = m - (n - 1)
                    for i in range(n):
                        if h[i] == 1 and 0 <= i + k < n and h[i + k] == 0:
                            hot_ones.add(i)
                        if h[i] == 0 and 0 <= i + k < n and h[i + k] == 1:
                            hot_zeros.add(i)
                hot_ones_arr = np.array(list(hot_ones)) if len(hot_ones) > 0 else ones
                hot_zeros_arr = np.array(list(hot_zeros)) if len(hot_zeros) > 0 else zeros
                a = hot_ones_arr[rng.randint(len(hot_ones_arr))]
                b = hot_zeros_arr[rng.randint(len(hot_zeros_arr))]
            else:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
            h_new[a] = 0
            h_new[b] = 1
            moved = True
        elif r < 0.85:
            # Random single swap
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_new[a] = 0
            h_new[b] = 1
            moved = True
        else:

            if len(ones) >= 2 and len(zeros) >= 2:

                b1, b2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[a1]] = 0
                h_new[ones[a2]] = 0
                h_new[zeros[b1]] = 1
                h_new[zeros[b2]] = 1
                moved = True

            continue
        
        h_new_float = h_new.astype(float)
        corr_new = np.correlate(h_new_float, 1.0 - h_new_float, mode='full')
        c5_new = float(corr_new.max()) * dx

            current_c5 = c5_new

                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h.astype(float), best_c5
