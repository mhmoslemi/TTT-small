import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if mask.sum() == 0:
        return h
    h = h.copy()
    h[mask] += diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Second pass for residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        if mask2.sum() > 0:
            h[mask2] += residual / mask2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grads_active(h, corr, n, k_active, weights):
    """Compute weighted gradient from active shifts."""
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for wi, ki in zip(weights, k_active):
        v1 = (idxs - ki >= 0) & (idxs - ki < n)
        v2 = (idxs + ki >= 0) & (idxs + ki < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - ki])
        grad[v2] += wi * (-h[idxs[v2] + ki])
    return grad

    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1

    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break
        
        # Find active shifts
        c_max = np.max(corr)
        # Use top n_active shifts with softmax weighting
        corr_norm = corr - c_max
        weights_raw = np.exp(50.0 * corr_norm)  # alpha=50 for concentration
        weights_raw = weights_raw / weights_raw.sum()
        
        # Get indices of top shifts
        active_idx = np.argsort(-corr)[:n_active]
        k_active = (active_idx - mid).astype(int)
        w_active = weights_raw[active_idx]
        w_active = w_active / w_active.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, ki in zip(w_active, k_active):

            grad[v2] += wi * (-h[idxs[v2] + ki])
        grad -= grad.mean()  # Account for sum constraint

                g[v_jm] -= h[j - shifts[v_jm]]

def nesterov_subgrad(h, n, target, time_limit, max_iter=5000):
    """Nesterov accelerated projected subgradient method."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h_cur = h.copy()
    h_prev = h.copy()
    y_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    
    t = 1.0
    base_step = 0.1  # Base step size
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute objective at y
        corr = np.correlate(y_cur, 1.0 - y_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = project_feasible(y_cur, target)
        
        # Find active set (top 5 shifts)
        c_max = np.max(corr)
        active_threshold = c_max - 0.01 / dx  # Within 0.01 of max in C5 units
        active_mask = corr >= active_threshold
        
        if not active_mask.any():
            active_mask = corr == c_max
        
        active_idx = np.where(active_mask)[0]
        # Limit to top 10 for efficiency
        if len(active_idx) > 10:
            top_scores = corr[active_idx]
            top_order = np.argsort(-top_scores)[:10]
            active_idx = active_idx[top_order]
        
        # Compute subgradient (uniform weighting over active set)
        grad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y_cur[idxs[v1] - k])
            grad[v2] += (-y_cur[idxs[v2] + k])
        grad /= len(active_idx)
        grad -= grad.mean()  # Sum constraint
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (t - 1.0) / t_new
        step = base_step / (t_new ** 0.5)  # Decreasing step
        
        y_new = h_cur + alpha * (h_cur - h_prev)
        # Gradient step
        y_new = y_new - step * grad
        # Project
        h_new = project_feasible(y_new, target)
        
        h_prev = h_cur
        h_cur = h_new
        y_cur = h_cur + ((t_new - 1.0) / t_new) * (h_cur - h_prev)
        t = t_new
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            def smooth_obj(h, a=alpha, t=target, n=n):
                corr = np.correlate(h, 1.0 - h, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
            
            def cons_fun(h, t=target):
                return h.sum() - t
            
            def cons_jac(h, n=n):
                return np.ones(n)
            
            res = minimize(smooth_obj, h_cur,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def block_cd(h, n, target, time_limit, block_size=5, max_sweeps=5000):

            best_h = h.copy()
            no_improve = 0

        # Select a random block of coordinates to update
        block = rng.choice(n, size=block_size, replace=False)
        block_h = h[block].copy()
        
        # For this block, find the best perturbation that reduces C5
        best_block_h = block_h.copy()
        best_block_c5 = c5_val
        
        # Try small random perturbations
        for attempt in range(20):
            pert = rng.randn(block_size) * 0.05
            trial = block_h + pert
            trial = np.clip(trial, 0.0, 1.0)
            # Adjust to maintain sum

                best_block_h = trial.copy()
        
        h[block] = best_block_h
        h = project_feasible(h, target)
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def multi_active_cd(h, n, target, time_limit, max_sweeps=8000,
                    n_active=5, n_cand=45, n_bisect=15):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 120:
            break

        c_max = float(np.max(corr))
        active_idx = np.argsort(-corr)[:n_active]
        k_active = (active_idx - mid).astype(int)
        c_vals = corr[active_idx]
        w_raw = np.exp(80.0 * (c_vals - c_max))
        w_raw = w_raw / w_raw.sum()

        grad = np.zeros(n)
        for wi, ki in zip(w_raw, k_active):
            v1 = (idxs - ki >= 0) & (idxs - ki < n)
            v2 = (idxs + ki >= 0) & (idxs + ki < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - ki])
            grad[v2] += wi * (-h[idxs[v2] + ki])
        grad -= grad.mean()

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            if hi < 1e-10:
                continue
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                hj = h[j]
                if hj > 1.0 - 1e-10:
                    continue
                delta_hi = min(hi, 1.0 - hj)
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if no_improve > 40:
                break

    return best_h, best_c5


def proximal_smoothed(h, n, target, time_limit, alpha_start=10.0,
                      alpha_end=100.0, max_iter=10000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    step = 0.5 / (n * alpha_start)
    staleness = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 500 == 0 and staleness > 200:
            break

        progress_ratio = it / float(max_iter)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress_ratio

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_cur.copy()
            staleness = 0
        else:
            staleness += 1

        corr_shifted = alpha * (corr - np.max(corr))
        p = np.exp(corr_shifted)
        p_max = np.max(p)
        if p_max < 1e-300:
            p = np.zeros_like(corr)
            p[np.argmax(corr)] = 1.0
        else:
            p /= p.sum()

        active_thresh = np.max(corr) - 10.0 / alpha
        active_mask = corr >= active_thresh
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(corr))])
        if len(active_idx) > 20:
            top_order = np.argsort(-corr[active_idx])[:20]
            active_idx = active_idx[top_order]

        p_a = p[active_idx]
        s = p_a.sum()
        if s < 1e-300:
            p_a = np.ones(len(active_idx)) / len(active_idx)
        else:
            p_a /= s

        grad = np.zeros(n)
        for wi, s_idx in zip(p_a, active_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] += wi * (-h_cur[idxs[v2] + k])
        grad -= grad.mean()

        h_try = project_exact(h_cur - step * grad, target)
        corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        c5_try = float(np.max(corr_try)) * dx

        if c5_try < c5_val - 1e-15:
            h_cur = h_try
            step = min(step * 1.05, 2.0 / (n * alpha))
        else:
            step *= 0.7
            if step < 1e-10:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [500, 2000], 400, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        cd_time = min(280, tleft() * 0.35)
        h_cd, c5_cd = multi_active_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_active=5,
                                       n_cand=45, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        px_time = min(100, tleft() * 0.2)
        h_px, c5_px = proximal_smoothed(best_h, n, target,
                                         time_limit=px_time,
                                         alpha_start=8.0,
                                         alpha_end=150.0,
                                         max_iter=8000)
        if c5_px < best_score:
            best_score = c5_px
            best_h = h_px

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        cd_time = min(200, tleft() * 0.3)
        h_qs, c5_qs = equalize_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=6000, n_top=8, n_bisect=15)
        if c5_qs < best_score:
            best_score = c5_qs
            best_h = h_qs

    if tleft() > 100:
        rp_time = min(80, tleft() * 0.15)
        h_rp, c5_rp = random_pair_cd(best_h, n, target,
                                      time_limit=rp_time,
                                      max_sweeps=3000, n_bisect=12)
        if c5_rp < best_score:
            best_score = c5_rp
            best_h = h_rp

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [20000, 80000], 600, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.10]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(50, tleft() * 0.15)
        h_try, c5_try = multi_active_cd(h_pert, n, target,
                                         time_limit=cd_t,
                                         max_sweeps=2000, n_active=5,
                                         n_cand=40, n_bisect=13)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    if tleft() > 250:
        for n_try in [800, 1000]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [500, 2000], 400, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 100 and sc < best_score + 0.002:
                cd_t = min(120, tleft() * 0.35)
                h_hires2, sc2 = multi_active_cd(h_hires, n_try, target_try,
                                                 time_limit=cd_t,
                                                 max_sweeps=4000, n_active=5,
                                                 n_cand=50, n_bisect=15)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        px_time = min(50, tleft() * 0.3)
        h_px, c5_px = proximal_smoothed(best_h, n_use, target_use,
                                         time_limit=px_time,
                                         alpha_start=15.0,
                                         alpha_end=200.0,
                                         max_iter=6000)
        if c5_px < best_score:
            best_score = c5_px
            best_h = h_px

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(40, tleft() * 0.4)
        h_cd, c5_cd = multi_active_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=3000, n_active=5,
                                       n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000], 500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
