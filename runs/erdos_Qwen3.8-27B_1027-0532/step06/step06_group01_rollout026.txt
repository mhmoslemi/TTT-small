```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    # Gradient
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)  # weights for each shift
    
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w[k_idx] * (1.0 - x[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w[k_idx] * x[idxs[v2] + k]
    
    grad *= dx
    return obj, grad

def slsqp_with_grad(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            def obj_wrapper(x):
                return smooth_obj_and_grad(x, alpha, target)[0]
            def grad_wrapper(x):
                return smooth_obj_and_grad(x, alpha, target)[1]
            res = minimize(obj_wrapper, h_cur, jac=grad_wrapper,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5
```

```python
def multi_shift_pair_cd(h_init, time_limit, K=10, n_cand=30):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute weights for top-K shifts
        sorted_idx = np.argsort(corr)[-K:]
        w = np.zeros(n_corr)
        # Use a softmax-like weighting
        top_values = corr[sorted_idx]
        alpha_w = 50.0
        w_top = np.exp(alpha_w * (top_values - c_max))
        w_top /= w_top.sum()
        w[sorted_idx] = w_top
        
        # Compute scores: score[m] = sum_k w[k] * profile[m][k]
        score = np.zeros(n)
        for k_pos in sorted_idx:
            k = k_pos - mid
            wk = w[k_pos]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                score[v1] += wk * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                score[v2] -= wk * h[idxs[v2] + k]
        
        # Select candidates
        # Decrease: high score, h > 0
        # Increase: low score, h < 1
        dec_mask = h > 0.01
        inc_mask = h < 0.99
        
        dec_scores = np.where(dec_mask, score, -np.inf)
        inc_scores = np.where(inc_mask, score, np.inf)
        
        dec_cands = np.argsort(-dec_scores)[:n_cand]
        inc_cands = np.argsort(inc_scores)[:n_cand]
        
        # For each pair, compute the gain under linear approx
        best_improve = 0
        best_i, best_j, best_delta = -1, -1, 0
        
        # Precompute profiles for candidates
        all_cands = np.unique(np.concatenate([dec_cands, inc_cands]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            for k_pos in sorted_idx:
                k = k_pos - mid
                # profile[m][k_pos] = (1-h[m-k]) - h[m+k] (with bounds)
                val = 0.0
                if 0 <= m - k < n:
                    val += (1.0 - h[m - k])
                if 0 <= m + k < n:
                    val -= h[m + k]
                profiles[ci, k_pos] = val
        
        # For speed, only evaluate the top few pairs
        for ci in range(min(n_cand, len(dec_cands))):
            i = int(dec_cands[ci])
            hi_val = h[i]
            if hi_val < 1e-6:
                continue
            for cj in range(min(n_cand, len(inc_cands))):
                j = int(inc_cands[cj])
                if i == j:
                    continue
                hj_val = h[j]
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-6:
                    continue
                
                # Find the profile indices
                ci_pos = np.where(all_cands == i)[0][0]
                cj_pos = np.where(all_cands == j)[0][0]
                
                g = profiles[cj_pos] - profiles[ci_pos]
                
                # Line search: find delta that minimizes max(corr + delta * g)
                # Bisection on the subgradient
                lo, hi_d = 0.0, delta_max
                for _ in range(20):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= (f_mid - 1e-10)
                    f_prime = np.max(g[active])
                    if f_prime > 1e-12:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-8:
                    continue
                
                new_max = np.max(corr + delta_star * g)
                improve = c_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i, best_j = i, j
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            break
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    return c_max + np.log(np.sum(np.exp(shifted))) / _G_ALPHA

                    g[v2] -= h[idx_iak[v2]]
                if np.any(v1):
                    g[v1] += h[idx_ik[v1]]
                if np.any(v3):
                    g[v3] -= h[idx_jk[v3]]
                if np.any(v4):
                    g[v4] += h[idx_jak[v4]]
                new_max = float(np.max(corr - g))
                improve = c_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] = 0.0
            h[best_j] = 1.0
            h = project_feasible(h, target)
        else:
            if no_improve > 100:
                n_swaps = rng.randint(2, 6)
                for _ in range(n_swaps):
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    pi = rng.choice(ones)
                    pj = rng.choice(zeros)
                    h[pi] = 0.0
                    h[pj] = 1.0
                h = project_feasible(h, target)
                no_improve = 0

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient from initial
    if tleft() > 120:
        hp, sc = slsqp_round(best_h, [200, 1000, 5000], 1500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Multi-shift pair CD (main workhorse)
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.3)
        hcd, c5cd = multi_shift_pair_cd(best_h, cd_time, K=8, n_cand=30, max_no_improve=400)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: SLSQP polish
    if tleft() > 80:
        hp, sc = slsqp_round(best_h, [5000, 20000, 100000], 1000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Binary local search (different landscape)
    if tleft() > 150:
        blt = min(150, tleft() * 0.3)
        hb, c5b = binary_local_search(best_h, blt, n_cand=40, max_no_improve=300)
        if c5b < best_score:
            best_score = c5b
            best_h = hb

    # Phase 5: SLSQP after binary LS
    if tleft() > 60:
        hp, sc = slsqp_round(best_h, [10000, 50000, 200000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 6: More multi-shift CD with different K
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        hcd, c5cd = multi_shift_pair_cd(best_h, cd_time, K=15, n_cand=40, max_no_improve=400)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 7: Interpolate to higher resolution n=1200
    if tleft() > 350:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, [100, 500, 2000, 10000], 1500, 1e-14)
        if tleft() > 200:
            hcd, c5cd = multi_shift_pair_cd(hp, min(180, tleft() * 0.4), K=10, n_cand=30, max_no_improve=300)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
            elif sc < best_score:
                best_score = sc
                best_h = hp
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 8: Binary LS at current best resolution
    if tleft() > 120:
        blt = min(100, tleft() * 0.3)
        hb, c5b = binary_local_search(best_h, blt, n_cand=30, max_no_improve=200)
        if c5b < best_score:
            best_score = c5b
            best_h = hb
            n = len(hb)
            target = n / 2.0

    # Phase 9: Final multi-shift CD
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(100, tleft() * 0.4)
        hcd, c5cd = multi_shift_pair_cd(best_h, cd_time, K=12, n_cand=35, max_no_improve=300)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: SLSQP with very high alpha
    if tleft() > 50:
        n_use = len(best_h)
        hp, sc = slsqp_round(best_h, [100000, 500000, 1000000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final quick CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = multi_shift_pair_cd(best_h, min(25, tleft() * 0.5), K=10, n_cand=25, max_no_improve=200)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 12: Projected gradient with high alpha as final touch
    if tleft() > 20:
        _G_ALPHA = 500000.0
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = projected_gradient(best_h, min(15, tleft() * 0.5), 500000.0, 200,
                                     [0.005, 0.002, 0.001, 0.0005, 0.0001])
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
```
