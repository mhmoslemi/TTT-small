def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):

    b = np.zeros(N)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        need = current_sum - target_sum
        h_r[ones[order[:need]]] = 0.0
    else:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        need = target_sum - int(h_r.sum())
        h_r[zeros[order[:need]]] = 1.0
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[:final_sum - target_sum]] = 0.0
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[:target_sum - final_sum]] = 1.0
    return h_r

def compute_delta_for_pos(h, m):
    n = len(h)

    idx1 = m + k_vals
    idx2 = m - k_vals
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta = np.zeros(2*n-1)
    delta[mask1] += (1.0 - h[idx1[mask1]])
    delta[mask2] -= h[idx2[mask2]]
    return delta

            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0; h[j1] = 1.0; h[i2] = 0.0; h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_swap(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0; h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    return best_h, best_c5

def greedy_best_improve(h0, k_arr, max_rounds=50, time_limit=30.0, start_time=0.0):
    n = len(h0)
    h = h0.astype(float).copy()
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            break
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_i = ones[np.argsort(A[ones])[:min(40, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(40, len(zeros))]]
        best_delta = max_val
        best_i, best_j = -1, -1
        best_d = None
        for i in top_i:
            for j in top_j:
                delta = compute_delta_swap(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < best_delta:
                    best_delta = new_max
                    best_i, best_j = i, j
                    best_d = delta
        if best_d is None:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        full += best_d
        cur_c5 = float(full.max()) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)
    
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Coordinate descent (continuous)
    if tleft() > 60:
        allowed = min(120.0, tleft() * 0.2)
        h_cd, sc_cd = coordinate_descent(h_init, target_init, max_sweeps=800, rng=rng,
                                         time_limit=allowed, start_time=start_time)
        if sc_cd * (2.0 / n_init) < best_score:
            best_score = sc_cd * (2.0 / n_init)
            best_h = h_cd.copy()
    
    # Phase 2: Project to binary, greedy improve
    if tleft() > 40:
        h_bin = project_binary(best_h, target_init)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
        allowed = min(40.0, tleft() * 0.1)
        h_g, sc_g = greedy_best_improve(best_h, k_arr, max_rounds=200,
                                         time_limit=allowed, start_time=start_time)
        if sc_g < best_score:
            best_score = sc_g
            best_h = h_g.copy()
    
    # Phase 3: Adaptive SA
    if tleft() > 60:
        allowed = min(180.0, tleft() * 0.25)
        n_sa = int(min(5000000, allowed * 40000))
        h_sa, sc_sa = sa_adaptive(best_h, n_sa, 0.02, 0.00005, rng,
                                   time_limit=allowed, start_time=start_time,
                                   k_arr=k_arr, focus_frac=0.6)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()
    
    # Phase 4: Double-swap SA
    if tleft() > 50:
        allowed = min(100.0, tleft() * 0.15)
        n_sd = int(min(2000000, allowed * 10000))
        h_sd, sc_sd = sa_double_swap(best_h, n_sd, 0.015, 0.00003, rng,
                                      time_limit=allowed, start_time=start_time,
                                      k_arr=k_arr)
        if sc_sd < best_score:
            best_score = sc_sd
            best_h = h_sd.copy()
    
    # Phase 5: Multiple coordinate descent restarts
    for restart in range(3):
        if tleft() < 60:
            break
        h_pert = best_h.copy()
        n_pert = rng.randint(5, 15)
        ones = np.where(h_pert > 0.5)[0]
        zeros = np.where(h_pert < 0.5)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
        h_pert = project_box_sum(h_pert, target_init)
        allowed = min(60.0, tleft() * 0.12)
        h_cd2, sc_cd2 = coordinate_descent(h_pert, target_init, max_sweeps=400, rng=rng,
                                            time_limit=allowed, start_time=start_time)
        sc_cd2_final = sc_cd2 * (2.0 / n_init)
        h_bin2 = project_binary(h_cd2, target_init)
        sc_bin2 = compute_c5(h_bin2)
        if sc_cd2_final < best_score:
            best_score = sc_cd2_final
            best_h = h_cd2.copy()
        if sc_bin2 < best_score:
            best_score = sc_bin2
            best_h = h_bin2.copy()
    
    # Phase 6: Try different resolutions
    for n_try in [n_init + 20, n_init - 20, n_init + 40]:
        if tleft() < 80:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        h_interp = interpolate_to(best_h, n_try)
        h_interp = project_box_sum(h_interp, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed1 = min(30.0, tleft() * 0.1)
        h_cd3, sc_cd3 = coordinate_descent(h_bin_try, target_try, max_sweeps=300, rng=rng,
                                            time_limit=allowed1, start_time=start_time)
        sc_cd3_final = sc_cd3 * (2.0 / n_try)
        h_bin3 = project_binary(h_cd3, target_try)
        sc_bin3 = compute_c5(h_bin3)
        allowed2 = min(40.0, tleft() * 0.1)
        n_sa_t = int(min(1500000, allowed2 * 30000))
        h_sa_t, sc_sa_t = sa_adaptive(h_bin3, n_sa_t, 0.015, 0.00003, rng,
                                       time_limit=allowed2, start_time=start_time,
                                       k_arr=k_arr_try, focus_frac=0.5)
        candidates = [(sc_cd3_final, h_cd3, n_try), (sc_bin3, h_bin3, n_try), (sc_sa_t, h_sa_t, n_try)]
        for sc_c, h_c, n_c in candidates:
            if sc_c < best_score:
                best_score = sc_c
                best_h = h_c.copy()
    
    # Phase 7: Final greedy + SA polish on best_h at its resolution
    n_final = len(best_h)
    target_final = n_final // 2
    k_arr_final = np.arange(2 * n_final - 1) - (n_final - 1)
    h_work = project_binary(best_h, target_final)
    if tleft() > 30:
        allowed = min(30.0, tleft() * 0.2)
        h_gf, sc_gf = greedy_best_improve(h_work, k_arr_final, max_rounds=100,
                                           time_limit=allowed, start_time=start_time)
        if sc_gf < best_score:
            best_score = sc_gf
            best_h = h_gf.copy()
            h_work = h_gf.copy()
    if tleft() > 20:
        allowed = min(40.0, tleft() * 0.3)
        n_sa_f = int(min(2000000, allowed * 40000))
        h_sf, sc_sf = sa_adaptive(h_work, n_sa_f, 0.008, 0.00001, rng,
                                   time_limit=allowed, start_time=start_time,
                                   k_arr=k_arr_final, focus_frac=0.7)
        if sc_sf < best_score:
            best_score = sc_sf
            best_h = h_sf.copy()
    
    # Phase 8: Final coordinate descent + binary projection
    if tleft() > 20:
        n_f2 = len(best_h)
        t_f2 = n_f2 // 2
        allowed = min(30.0, tleft() * 0.4)
        h_fcd, sc_fcd = coordinate_descent(best_h, t_f2, max_sweeps=500, rng=rng,
                                            time_limit=allowed, start_time=start_time)
        sc_fcd_final = sc_fcd * (2.0 / n_f2)
        h_fbin = project_binary(h_fcd, t_f2)
        sc_fbin = compute_c5(h_fbin)
        if sc_fcd_final < best_score:
            best_score = sc_fcd_final
            best_h = h_fcd.copy()
        if sc_fbin < best_score:
            best_score = sc_fbin
            best_h = h_fbin.copy()
    
    # Final feasibility
    n_ret = len(best_h)
    target_ret = n_ret // 2
    h_ret = project_box_sum(best_h, target_ret)
    final_score = compute_c5(h_ret)
    
    if final_score > best_score:
        h_ret = best_h
        final_score = best_score
    
    return h_ret, final_score, n_ret
