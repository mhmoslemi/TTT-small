        return h
    # Water-filling style projection
    if diff > 0:
        # Need to add mass: add to entries not at 1
        order = np.argsort(1.0 - h)  # entries closest to 1 first
    else:
        # Need to remove mass: remove from entries not at 0
        order = np.argsort(h)  # entries closest to 0 first
    
    h = h.copy()
    remaining = diff
    for idx in order:
        if remaining > 0:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
        else:
            sub = min(h[idx], -remaining)
            h[idx] -= sub
            remaining += sub
        if abs(remaining) < 1e-12:
            break
    return h

    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t = 1.0
    y = h.copy()
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        
        alpha = min(100000.0, 10.0 * (it + 1) ** 1.5)  # growing alpha
        
        # Compute overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        # Use only absolute lags (symmetric)
        # overlaps has 2n-1 entries, indexed 0 to 2n-2
        # Lag k = idx - (n-1), m = |k| = |idx - (n-1)|
        # m ranges from 0 to n-1
        # For m > 0, there are two entries (idx = n-1+m and idx = n-1-m), both equal
        # For m = 0, one entry (idx = n-1)
        
        # Compute weights
        max_ov = np.max(overlaps)
        shifted = alpha * (overlaps - max_ov)
        exp_shifted = np.exp(np.clip(shifted, -500, 0))
        p = exp_shifted / np.sum(exp_shifted)
        
        # Compute weighted subgradient
        g = np.zeros(n)
        n_half = n - 1
        for idx in range(len(overlaps)):
            if p[idx] < 1e-10:
                continue
            k = idx - n_half
            m = abs(k)
            w = p[idx] * dx
            if m == 0:
                g += w * (1.0 - 2.0 * h)
            else:
                g[m:] += w * (1.0 - h[:n-m])
                g[:n-m] += w * (-h[m:])
        
        # FISTA step
        # Step size: 1/L where L is the Lipschitz constant
        # L ≈ alpha * 2 * dx * max_ov_scale (rough)
        # Use a conservative estimate
        L = alpha * 2.0 * dx * 2.0  # conservative
        step = 1.0 / L
        
        z = y - step * g
        h_new = np.clip(z, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        
        h = h_new
        t = t_new
        
        # Update best
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

For FISTA to work correctly, we need the proximal operator of the indicator function of the feasible set, which is the Euclidean projection. The Euclidean projection onto {h : 0 ≤ h ≤ 1, sum h = target} is:
1. Find λ such that sum_i clip(h_i - λ, 0, 1) = target
2. Return clip(h - λ, 0, 1)

This can be found by bisection on λ. It's O(n log(precision)).

```python
def project_polytope(v, target):
    # Project v onto {h : 0 <= h <= 1, sum h = target}
    # Find lambda such that sum clip(v_i - lambda, 0, 1) = target
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(v - mid, 0.0, 1.0)
```
