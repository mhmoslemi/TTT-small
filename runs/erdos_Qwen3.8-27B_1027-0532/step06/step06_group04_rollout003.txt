```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_direction(h, n, active_k, corr, eps=1e-9):
    dx = 2.0 / n
    half_n = n - 1
    idxs = np.arange(n)
    
    delta_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(delta_var) == 0,
        delta_var >= -h + 1e-10,
        delta_var <= (1.0 - h) - 1e-10
    ]
    
    for k_idx in active_k:
        k = int(k_idx) - half_n
        grad = np.zeros(n)
        v1 = (0 <= idxs + k) & ((idxs + k) < n)
        v2 = (0 <= idxs - k) & ((idxs - k) < n)
        grad[v1] += (1.0 - h[idxs[v1] + k])
        grad[v2] -= h[idxs[v2] - k]
        grad *= dx
        constraints.append(corr[k_idx] * dx + grad @ delta_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=50000, verbose=False)
    except Exception:
        return None, None
    
    if t_var.value is None or delta_var.value is None:
        return None, None
    
    delta = np.asarray(delta_var.value).ravel()
    if not np.all(np.isfinite(delta)):
        return None, None
    
    t_opt = float(t_var.value)
    return delta, t_opt

    idxs = np.arange(n)
    stall = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_cur = float(np.max(corr)) * dx
        k_star = int(np.argmax(corr))
        k = k_star - half_n
        
        # Gradient of active constraint
        grad = np.zeros(n)
        v1 = (0 <= idxs + k) & ((idxs + k) < n)
        v2 = (0 <= idxs - k) & ((idxs - k) < n)
        grad[v1] += (1.0 - h_cur[idxs[v1] + k])
        grad[v2] -= h_cur[idxs[v2] - k]
        grad *= dx
        
        # Generate move: gradient + noise
        move = -grad * 0.005 + rng.randn(n) * T
        move -= move.mean()
        h_try = project_feasible(h_cur + move, target)
        c5_try = compute_c5(h_try)
        
        # Metropolis acceptance
        if c5_try < best_c5:
            h_cur = h_try
            best_c5 = c5_try
            best_h = h_try.copy()
            stall = 0
        elif rng.rand() < np.exp(-(c5_try - best_c5) / max(T, 1e-15)):
            h_cur = h_try
            stall += 1
        else:
            stall += 1
        
        T *= 0.99995
        if T < 1e-7:
            T = T_init * 0.3
            stall = 0
        
        if stall > 2000:
            break
    
    return best_h, best_c5

def smooth_obj(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_max = np.max(corr) * dx
    shifted = np.clip(alpha * (corr * dx - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_bfgs(h, n, target, time_limit, rng):
    from scipy.optimize import minimize
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    for alpha in [50, 200, 500, 1000, 2000, 5000, 10000]:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            return smooth_obj(x, n, alpha)
        
        def con_fun(x):
            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', bounds=bounds,
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    
    return best_h
