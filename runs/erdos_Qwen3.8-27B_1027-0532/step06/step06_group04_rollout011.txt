        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max = float(np.max(corr_y))
        weights = np.exp(alpha * (corr_y - c_max))
        weights /= np.sum(weights)
        
        # Gradient of f_smooth at y
        grad = np.zeros(n)
        # Use only top contributing k's for efficiency
        top_k = np.argsort(weights)[-50:]
        for k_idx in top_k:
            k = k_idx - half_n
            w = weights[k_idx] * dx
            idx_v1 = (0 <= np.arange(n) + k) & ((np.arange(n) + k) < n)
            idx_v2 = (0 <= np.arange(n) - k) & ((np.arange(n) - k) < n)
            grad[idx_v1] += w * (1.0 - y[np.arange(n)[idx_v1] + k])
            grad[idx_v2] += w * (-y[np.arange(n)[idx_v2] - k])
        
        # Step size (for smooth function, L ≈ 2*alpha*dx*n is an upper bound on Lipschitz constant)
        # Use a smaller step for safety
        lr = 0.5 / (alpha * dx * n * 2.0 + 1e-10)
        lr = min(lr, 0.5)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t += 1
        
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_max_line(slopes, intercepts, lam):
    C = slopes + lam
    if np.all(C > 1e-14):
        return 0.0
    if np.all(C < -1e-14):
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = C * mid + intercepts
        max_val = np.max(vals)
        tol = 1e-13 * max(1.0, abs(max_val))
        active = np.abs(vals - max_val) < tol
        if not np.any(active):
            active = np.zeros(len(C), dtype=bool)
            active[np.argmax(vals)] = True
        if np.all(C[active] > 1e-14):
            hi = mid
        elif np.all(C[active] < -1e-14):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h_init, target, n_active, max_passes, time_limit, rng):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    lam = 0.0
    lam_rate = 0.005

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break

        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            h[i] = solve_1d_max_line(slopes, intercepts, lam)

        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-8:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-10)[0]
            else:
                idx = np.where(h > 1e-10)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
                h = np.clip(h, 0.0, 1.0)
        lam = np.clip(lam + lam_rate * (target - np.sum(h)) / n, -1.0, 1.0)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h.copy()

        new_active = np.argsort(corr)[-n_active:]
        if not np.array_equal(np.sort(new_active), np.sort(active_k)):
            active_k = new_active

        if p % 30 == 29:
            lam_rate = max(0.001, lam_rate * 0.97)

    return best_h, best_c5


def alternating_lp(h_init, n, target, max_iter, time_limit, d0, reg0, n_active, scs_eps=1e-11):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = d0
    current_reg = reg0
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = list(np.argsort(corr)[-n_active:])

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in active_k:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=50000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        added = False
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in set(active_k) and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 3e-5:
                    active_k = list(active_k) + [k_idx]
                    added = True
                    break
        all_candidates = set(top_k) | set(active_k)
        active_k = list(all_candidates)
        if len(active_k) > 5 * n_active:
            sorted_act = sorted(active_k, key=lambda x: corr[x], reverse=True)
            active_k = sorted_act[:5 * n_active]

        if c5_val >= best_c5 - 1e-15:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.99, damping + 0.02)
        if it % 40 == 0 and current_reg > 1e-9:
            current_reg = max(1e-9, current_reg * 0.7)

    return best_h, best_c5

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -300.0, 0.0)

            hp, cp = alternating_lp(h_up2, n2, target2, max_iter=5000,
                                    time_limit=min(60, tleft() * 0.2), d0=0.92,
                                    reg0=1e-5, n_active=70, scs_eps=1e-11)
            if cp < sc_up2:
                h_up2, sc_up2 = hp, cp

        if tleft() > 80:
            hp, cp = optimized_cd(h_up2, target2, n_active=80,
                                  max_passes=10000, time_limit=min(40, tleft() * 0.15),
                                  rng=rng, n_top=30)
            if cp < sc_up2:
                h_up2, sc_up2 = hp, cp

        if tleft() > 50:
            h_ref = slsqp_round(h_up2, n2, target2, [300, 1000, 5000], 1000, 1e-14)
            sc = compute_c5(h_ref)
            if sc < sc_up2:
                h_up2, sc_up2 = h_ref, sc

        if sc_up2 < best_score:
            best_score, best_h = sc_up2, h_up2
            best_n, best_target = n2, target2

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, cp = alternating_lp(best_h, n_use, target_use, max_iter=5000,
                                time_limit=min(40, tleft() * 0.4), d0=0.97,
                                reg0=1e-6, n_active=100, scs_eps=1e-12)
        if cp < best_score:
            best_score, best_h = cp, hp

    for noise in [0.01, 0.03, 0.06, 0.1]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(h_pert, n_use, target_use, [500, 2000, 8000], 1000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score, best_h = sc, h_ref

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, cp = coord_descent(best_h, target_use, n_active=100,
                               max_passes=15000, time_limit=min(35, tleft() * 0.5), rng=rng)
        if cp < best_score:
            best_score, best_h = cp, hp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(best_h, n_use, target_use, [5000, 20000, 80000], 800, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score, best_h = sc, h_ref

    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, cp = subgradient_improve(best_h, n_use, target_use,
                                      max_steps=3000, time_limit=min(12, tleft() * 0.8))
        if cp < best_score:
            best_score, best_h = cp, hp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
