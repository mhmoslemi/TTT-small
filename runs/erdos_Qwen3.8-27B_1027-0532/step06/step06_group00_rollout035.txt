def frank_wolfe(h_init, n, target, time_limit, max_iter=10000, alpha_mode='adaptive'):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute gradient of smooth objective
        alpha = max(100.0, 100.0 * (1 + it / 100))  # increasing sharpness
        weights = np.exp(alpha * corr / dx - np.max(alpha * corr / dx))
        weights /= weights.sum()
        
        # Gradient of weighted sum of f_k
        grad = np.zeros(n)
        for m in range(2*n - 1):
            if weights[m] < 1e-15:
                continue
            k = m - (n - 1)
            # f_k = sum_{i: valid} h[i](1-h[i+k])
            # df_k/dh[i] = (1-h[i+k]) - h[i-k] (where defined)
            for i in range(n):
                contrib = 0.0
                if 0 <= i + k < n:
                    contrib += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    contrib -= h[i - k]
                grad[i] += weights[m] * contrib
        
        # Vertex: set to 1 where grad is smallest
        v = np.zeros(n)
        v[np.argsort(grad)[:int(target)]] = 1.0
        
        # Line search
        best_alpha = 0.0
        best_obj = c5_val
        for frac in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            h_try = (1 - frac) * h + frac * v
            h_try = np.clip(h_try, 0, 1)
            obj = compute_c5(h_try)
            if obj < best_obj:
                best_obj = obj
                best_alpha = frac
        
        if best_alpha > 0:
            h = (1 - best_alpha) * h + best_alpha * v
            h = np.clip(h, 0, 1)
        else:
            break
    
    return best_h, best_c5

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_exact(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on threshold."""
    n = len(x)
    # We want clip(x - tau, 0, 1) such that sum = target_sum
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        proj = np.clip(x - tau, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(x - tau, 0.0, 1.0)

        for frac in [2.0/(it+2.0), 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            frac = min(frac, 1.0)
            h_try = (1 - frac) * h + frac * v
            h_try = project_exact(h_try, target)
            obj = compute_c5(h_try)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_alpha_step = frac
        
        # Also try away step
        if no_improve > 100:
            # Find vertex in current support to move away from
            support = np.where(h > 0.01)[0]
            if len(support) > 0:
                v_support = np.zeros(n)
                v_away_idx = support[np.argmax(grad[support])]
                v_support[v_away_idx] = 1.0
                # Move: h - gamma * v_support + (gamma/2) * v (normalized)
                for gamma in [0.3, 0.1, 0.05]:
                    h_try = h - gamma * v_support
                    h_try = h_try + gamma * v
                    h_try = project_exact(h_try, target)
                    obj = compute_c5(h_try)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_alpha_step = -gamma  # negative means away step
        
        if best_alpha_step > 0:
            h = (1 - best_alpha_step) * h + best_alpha_step * v
            h = project_exact(h, target)
        elif best_alpha_step < 0:
            gamma = -best_alpha_step
            # Reconstruct the away step
            support = np.where(h > 0.01)[0]
            if len(support) > 0:
                v_away_idx = support[np.argmax(grad[support])]
                v_support = np.zeros(n)
                v_support[v_away_idx] = 1.0
                h = h - gamma * v_support + gamma * v
                h = project_exact(h, target)
        
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def gradient_descent(h_init, n, target, time_limit, max_iter=100000,
                     alpha_start=50, alpha_end=10000, lr_init=0.01):
    """Gradient descent with projection and adaptive learning rate."""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0
    prev_obj = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            # Restart with smaller lr
            lr *= 0.5
            if lr < 1e-6:
                break
            no_improve = 0
        
        progress = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        grad, _ = smooth_gradient(h, n, alpha)
        
        # Backtracking line search
        best_obj = compute_c5(h)
        best_h_try = h
        step = lr
        for _ in range(10):
            h_try = project_exact(h - step * grad, target)
            obj = compute_c5(h_try)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_h_try = h_try
                break
            step *= 0.5
        
        h = best_h_try
        cur_c5 = compute_c5(h)
        
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.05)  # slightly increase lr on success
        else:
            no_improve += 1
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=5000,
            n_cand=20, max_no_improve=200):
    """Pair coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, time_limit=60):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def project_exact(x, target_sum):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        proj = np.clip(x - tau, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(x - tau, 0.0, 1.0)


def smooth_gradient(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= w.sum()
    grad = np.zeros(n)
    mid = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - mid
        if k >= 0:
            if n - k > 0:
                grad[:n - k] += wm * (1.0 - h[k:])
                grad[k:] -= wm * h[:n - k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += wm * (1.0 - h[:n - kk])
                grad[:n - kk] -= wm * h[kk:]
    return grad


def frank_wolfe(h_init, n, target, time_limit, max_iter=50000,
                alpha_start=50, alpha_end=5000, max_no_improve=600):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        progress = min(float(it) / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        grad = smooth_gradient(h, n, alpha)

        v = np.zeros(n)
        v[np.argsort(grad)[:int(round(target))]] = 1.0

        best_obj = compute_c5(h)
        best_step = 0.0

        fw_step = min(2.0 / (it + 2.0), 1.0)
        candidates = [fw_step, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
        for frac in candidates:
            h_try = (1.0 - frac) * h + frac * v
            h_try = project_exact(h_try, target)
            obj = compute_c5(h_try)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_step = frac

        if no_improve > 150:
            support = np.where(h > 0.05)[0]
            if len(support) > 0:
                away_idx = support[np.argmax(grad[support])]
                for gamma in [0.3, 0.15, 0.05]:
                    h_try = h.copy()
                    h_try[away_idx] -= gamma
                    h_try += gamma * v
                    h_try = project_exact(h_try, target)
                    obj = compute_c5(h_try)
                    if obj < best_obj - 1e-15:
                        best_obj = obj
                        best_step = -gamma

        if best_step > 0:
            h = (1.0 - best_step) * h + best_step * v
            h = project_exact(h, target)
        elif best_step < 0:
            gamma = -best_step
            support = np.where(h > 0.05)[0]
            if len(support) > 0:
                away_idx = support[np.argmax(grad[support])]
                h = h.copy()
                h[away_idx] -= gamma
                h += gamma * v
                h = project_exact(h, target)

        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5
