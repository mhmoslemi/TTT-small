import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def coord_desc(h_init, n, target, time_limit, grid_size=201, rng=None, max_sweeps=50000, no_improve_limit=100):
    start = time.time()
    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_full = 2 * n - 1
    n_half = n - 1
    indices = np.arange(n)
    v_grid = np.linspace(0.0, 1.0, grid_size)
    s_vals = np.arange(n_full) - n_half

    i_vals = np.arange(n)
    diff_mat = i_vals[:, None] - s_vals[None, :]
    vf_mask = (diff_mat >= 0) & (diff_mat < n)
    sum_mat = i_vals[:, None] + s_vals[None, :]
    vs_mask = (sum_mat >= 0) & (sum_mat < n)

    c = np.correlate(h, 1.0 - h, mode='full')
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if rng is not None and sweep > 0:
            rng.shuffle(indices)

        max_delta = 0.0
        for i in indices:
            old_val = h[i]
            vf = vf_mask[i]
            vs = vs_mask[i]

            slopes = np.zeros(n_full)
            if np.any(vf):
                slopes[vf] += dx * (1.0 - h[i - s_vals[vf]])
            if np.any(vs):
                slopes[vs] -= dx * h[i + s_vals[vs]]

            d_grid = v_grid - old_val
            vals = c[None, :] + d_grid[:, None] * slopes[None, :]
            obj_vals = vals.max(axis=1)

            best_idx = int(obj_vals.argmin())
            new_val = float(v_grid[best_idx])

            if 0 < best_idx < grid_size - 1:
                v_lo = float(v_grid[best_idx - 1])
                v_hi = float(v_grid[best_idx + 1])
                v_fine = np.linspace(v_lo, v_hi, 40)
                d_fine = v_fine - old_val
                vals_fine = c[None, :] + d_fine[:, None] * slopes[None, :]
                obj_fine = vals_fine.max(axis=1)
                new_val = float(v_fine[int(obj_fine.argmin())])

            new_val = max(0.0, min(1.0, new_val))
            delta = new_val - old_val
            if abs(delta) > 1e-13:
                h[i] = new_val
                c += delta * slopes
                if abs(delta) > max_delta:
                    max_delta = abs(delta)

        h = project_feasible(h, target)
        c = np.correlate(h, 1.0 - h, mode='full')
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= no_improve_limit:
            break

    return best_h, best_c5


def perturb_h(h, n, target, rng, n_perturb, strength):
    h_new = h.copy()
    idx = rng.choice(n, size=n_perturb, replace=False)
    h_new[idx] = h_new[idx] * (1.0 - strength) + strength * rng.uniform(0.0, 1.0, size=n_perturb)
    return project_feasible(h_new, target)


def binary_perturb(h, n, target, rng, n_flip):
    h_new = h.copy()
    idx = rng.choice(n, size=n_flip, replace=False)
    h_new[idx] = 1.0 - h_new[idx]
    return project_feasible(h_new, target)


def smooth_run_slsqp(h_start, n, target, time_limit):
    start_t = time.time()
    dx = 2.0 / n
    n_full = 2 * n - 1

    def _obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -300.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    alphas = [30, 100, 500, 2000]
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(_obj, h_cur, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_try = project_feasible(res.x, target)
            if compute_c5(h_try) < compute_c5(h_cur):
                h_cur = h_try
        except Exception:
            pass
    return h_cur
