import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_box_hyperplane(h, target):
    """Exact projection onto {h: 0<=h[i]<=1, sum(h)=target}"""
    n = len(h)
    # Binary search for lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def log_sum_exp_obj(h, alpha):
    """Smooth approximation of max overlap using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def log_sum_exp_grad(h, alpha):
    """Gradient of the log-sum-exp approximation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # grad of sum_k w_k * overlap_k where overlap_k = dx * sum_i h[i](1-h[i+k])
    # For each k, d(overlap_k)/dh[i] depends on k
    g = np.zeros(n)
    n_half = n - 1
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - n_half
        abs_k = abs(k)
        if abs_k == 0:
            g += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            g[:n-k] += w * (1.0 - h[k:]) * dx
            g[k:] += w * (-h[:n-k]) * dx
        else:
            m = -k
            g[m:] += w * (1.0 - h[:n-m]) * dx
            g[:n-m] += w * (-h[m:]) * dx
    return g

                       n_steps, time_limit, step_scale=1.0):
    """Nesterov-accelerated proximal gradient for log-sum-exp objective"""
    start = time.time()
    h = h_init.copy()
    h = project_box_hyperplane(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_lr = step_scale / np.sqrt(n)
    
    for t in range(n_steps):
        if time.time() - start > time_limit:
            break
        # Adaptive alpha
        progress = min(1.0, t / max(1, n_steps * 0.5))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Nesterov momentum
        momentum = (t - 1.0) / (t + 1.0) if t > 1 else 0.0
        y = h + momentum * (h - h_prev)
        
        # Gradient step
        grad = log_sum_exp_grad(y, alpha)
        y_new = y - step_lr * grad
        h_prev = h.copy()
        h = project_box_hyperplane(y_new, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def compute_grad_fft(h, W, dx):
    """Compute gradient using FFT-based convolution.
    W is an array of length 2n-1, W[n-1+k] = weight for shift k."""
    n = len(h)
    W_flipped = W[::-1]  # W_flipped[n-1+k] = W[-k]
    
    # Use FFT for linear convolution
    # Length needed: max(n + (2n-1) - 1) = 3n-2
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    fft_size = 2 * fft_size  # extra for safety
    
    h_hat = np.fft.rfft(h, fft_size)
    W_hat = np.fft.rfft(W, fft_size)
    Wf_hat = np.fft.rfft(W_flipped, fft_size)
    one_minus_h_hat = np.fft.rfft(1.0 - h, fft_size)
    
    conv_hW = np.fft.irfft(h_hat * W_hat, fft_size)
    conv_1mhWf = np.fft.irfft(one_minus_h_hat * Wf_hat, fft_size)
    
    # S2[j] = (h * W)[j + n - 1] for j = 0, ..., n-1
    S2 = conv_hW[n-1 : n-1+n]
    # S1[j] = ((1-h) * W_flipped)[j + n - 1] for j = 0, ..., n-1
    S1 = conv_1mhWf[n-1 : n-1+n]
    
    g = dx * (S1 - S2)
    return g

The objective is f(h) = LSE(h; alpha) + (mu/2) * (sum(h) - target)^2

The gradient is Lipschitz continuous with constant L. I need to estimate L.

For the LSE part: The Hessian of the LSE is a weighted sum of the Hessian of each overlap, plus some extra terms. Each overlap has Hessian with spectral norm ≤ 2*dx (since it's a bilinear form). The LSE Hessian has spectral norm ≤ 2*dx (the weights sum to 1 and the cross terms are bounded).

For the penalty part: The Hessian is mu * ones * ones^T, which has spectral norm mu * n.

So L ≈ 2*dx + mu*n = 2*(2/n) + mu*n = 4/n + mu*n.

For n=400, dx=0.005: 2*dx = 0.01. If mu=1e-4, then mu*n = 0.04. So L ≈ 0.05, and step size = 1/L ≈ 20.

Actually, I should be more careful. The Hessian of overlap_k is:
For k=0: -2*dx * I (spectral norm 2*dx)
For k≠0: -dx * (P + P^T) where P is a permutation-like matrix (spectral norm ≤ 2*dx)

The Hessian of LSE is:
∇²LSE = sum_k w_k * ∇²overlap_k + alpha * sum_{k,l} (w_k w_l - w_k^2 δ_{kl}) * (∇overlap_k)^T (∇overlap_l)

The second term is a sum of rank-1 matrices. Its spectral norm is at most alpha * (max |∇overlap_k|^2) * (sum of all weights) = alpha * (max |∇overlap_k|^2).

|∇overlap_k|^2 = dx^2 * sum_i [(1-h[i+k])^2 + h[i-k]^2] ≤ dx^2 * 2n = 2*dx^2*n = 2*(2/n)^2*n = 8/n

So the spectral norm of the second term is at most alpha * 8/n.

For alpha = 1000, n = 400: alpha * 8/n = 1000 * 0.02 = 20. That's large!

So L ≈ 2*dx + alpha * 8/n ≈ 0.01 + alpha * 0.02.

For alpha = 100: L ≈ 2.01, step = 0.5
For alpha = 1000: L ≈ 20.01, step = 0.05
For alpha = 10000: L ≈ 200.01, step = 0.005

def nesterov_proximal_lse(h_init, n, target, alpha_schedule, n_steps, time_limit, step_factor=0.9):

    h = h_init.copy()
    h = project_box_hyperplane(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t_nest = 1.0
    
    for t in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        alpha = alpha_schedule(t, n_steps)
        dx = 2.0 / n
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest**2)) / 2.0
        momentum = (t_nest - 1.0) / t_new
        t_nest = t_new
        
        y = h + momentum * (h - h_prev)
        
        # Compute gradient of LSE at y
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        c_max = np.max(corr)

        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Step size
        L = 2.0 * dx + alpha * 8.0 / n
        step = step_factor / L
        
        # Gradient via FFT
        g = compute_grad_fft(y, weights, dx)
        
        h_prev = h.copy()
        h = project_box_hyperplane(y - step * g, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_hyperplane(h, target):
    """Exact projection onto {h: 0<=h[i]<=1, sum(h)=target} using bisection"""
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def lse_grad_fft(h, weights, dx):
    """Compute gradient of weighted sum of overlaps using FFT."""
    n = len(h)
    # weights is an array of length 2n-1 (the full correlation weights)
    W_flipped = weights[::-1]
    fft_size = 1
    needed = 3 * n
    while fft_size < needed:
        fft_size *= 2
    
    h_hat = np.fft.rfft(h, fft_size)
    W_hat = np.fft.rfft(weights, fft_size)
    Wf_hat = np.fft.rfft(W_flipped, fft_size)
    omh_hat = np.fft.rfft(1.0 - h, fft_size)
    
    conv_hW = np.fft.irfft(h_hat * W_hat, fft_size)
    conv_omhWf = np.fft.irfft(omh_hat * Wf_hat, fft_size)
    
    offset = n - 1
    S1 = conv_omhWf[offset : offset + n]
    S2 = conv_hW[offset : offset + n]
    
    return dx * (S1 - S2)

def lse_value_and_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    value = c_max + np.log(sum_exp) / alpha
    return value, weights

def nesterov_lse(h_init, n, target, alpha_start, alpha_end, n_steps, time_limit, step_factor=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_box_hyperplane(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t_nest = 1.0
    for t in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        progress = t / max(1, n_steps)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = (t_nest - 1.0) / t_new
        t_nest = t_new
        
        y = h + momentum * (h - h_prev)
        
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        _, weights = lse_value_and_weights(corr, alpha)
        
        L = 2.0 * dx + alpha * 8.0 / n + 1.0
        step = step_factor / L
        
        g = lse_grad_fft(y, weights, dx)
        
        h_prev = h.copy()
        h = project_box_hyperplane(y - step * g, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def lbfgsb_polish(h_init, n, target, alpha, penalty, time_limit, maxiter=5000):
    dx = 2.0 / n
    
    def obj_and_grad(h_flat):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        val, weights = lse_value_and_weights(corr, alpha)
        sum_h = np.sum(h)
        pen_val = penalty * (sum_h - target) ** 2
        total_val = val + pen_val
        
        g = lse_grad_fft(h, weights, dx)
        g_flat = g + penalty * 2.0 * (sum_h - target)
        return total_val, g_flat
    
    bounds = [(0.0, 1.0)] * n
    h0 = h_init.copy()
    
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_box_hyperplane(res.x, target)
        return h_new
    except Exception:
        return h_init

def slsqp_polish(h_init, n, target, alphas, time_limit, maxiter=3000):
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:

        ]
        
        try:

                          bounds=[(0.0, 1.0)] * n, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_box_hyperplane(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new

            pass
    
    return h, best_c5
