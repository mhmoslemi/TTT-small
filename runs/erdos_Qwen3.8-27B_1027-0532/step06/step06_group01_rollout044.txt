def frank_wolfe_smooth(h, n, target, alpha, t=2.0):
    # Compute correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    # Softmax weights
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)  # numerical stability
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Compute gradient
    idxs = np.arange(n)
    mid = n - 1  # offset in 'full' mode
    grad = np.zeros(n)
    
    for k_idx in range(len(w)):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (ids - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w[k_idx] * h[idxs[v2] + k]
    
    # Frank-Wolfe: find vertex that minimizes grad · v
    # v is a permutation: sort by grad, set top n/2 to 1
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    v[sorted_idx[:target]] = 1.0  # target = n/2
    
    # Line search or fixed step
    # h_new = (1-t)*h + t*v
    h_new = (1 - t/n) * h + (t/n) * v  # using t/n as a conservative step
    # Actually, standard FW uses step size 2/(iter+2)
    
    return h_new, grad

            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_proximal(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = max(20.0, 100.0 * (step / 1000.0))
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        w = w * dx

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 2000:
            for lr_mult in [0.03, 0.015, 0.0075]:
                lr = base_lr * lr_mult
                h_try = h - lr * grad
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_now - 1e-15:
                    h = h_try
                    improved = True
                    break

    return best_h, best_c5


def frank_wolfe_opt(h_init, n, target, alpha, time_limit, max_iters=50000):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid
    = n - 1
    target_int = int(round(target))
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        grad = np.zeros(n)
        for ki in range(len(w)):
            if w[ki] < 1e-13:
                continue
            k = ki - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w[ki] * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w[ki] * h[idxs[v2] + k]

        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:target_int]] = 1.0

        t = 2.0 / (it + 2)
        h_new = (1.0 - t) * h + t * v
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def noisy_restart(h_init, n, target, seed):
    rng = np.random.RandomState(seed)
    noise = rng.standard_normal(n) * 0.05
    h_new = h_init + noise
    h_new = project_feasible(h_new, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_try = noisy_restart(best_h, n, target, seed + 1)
        sc_try = compute_c5(h_try)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 100:
        cd_time = min(200, tleft() * 0.25)
        hcd, c5cd = greedy_pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=300000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        cd_time = min(180, tleft() * 0.25)
        hcd, c5cd = exhaustive_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=35, max_sweeps=80000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [20000, 100000, 500000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 80:

        hfw, c5fw = frank_wolfe_opt(best_h, n, target, alpha=10000, time_limit=fw_time, max_iters=30000)
        if c5fw < best_score:
            best_score = c5fw
            best_h = hfw

    if tleft() > 60:
        hsg, c5sg = multi_shift_proximal(best_h, n, target, time_limit=min(50, tleft() * 0.3),
                                         max_steps=50000, n_shifts=5, base_lr=0.03)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 100:
        cd_time = min(120, tleft() * 0.25)
        hcd, c5cd = greedy_pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=200000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_new = 1000
    target_new = n_new / 2.0
    if tleft() > 400:
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, n_new, target_new, [200, 1000, 5000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        elif tleft() > 200:
            h_interp = interpolate_to(best_h, n_new, target_new)
            n = n_new
            target = target_new
            best_h = h_interp

        if n == n_new and tleft() > 200:
            cd_time = min(200, tleft() * 0.3)
            hcd, c5cd = greedy_pair_cd(best_h, n, target, time_limit=cd_time, max_sweeps=200000)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

        if n == n_new and tleft() > 80:
            hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 1500, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = hp

        if n == n_new and tleft() > 150:
            cd_time = min(150, tleft() * 0.3)
            hcd, c5cd = exhaustive_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=50000)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

        if n == n_new and tleft() > 60:
            hp, sc = slsqp_round(best_h, n, target, [50000, 200000], 1000, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = hp

    if tleft() > 80:

        hcd, c5cd = greedy_pair_cd(best_h, n_use, target_use, time_limit=cd_time, max_sweeps=200000)

    n_use = len(best_h)

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
