        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()

                momentum = np.zeros(n)
                no_improve = 0

        weights /= weights.sum()

        for s_idx, w in zip(top_k, weights):

            v2 = (idxs + k >= 0) & (idxs + k < n)

            grad[v2] -= w * h[idxs[v2] + k]

        momentum = beta * momentum + (1 - beta) * grad
        momentum -= np.mean(momentum)  # Keep sum constraint
        
        # Take step
