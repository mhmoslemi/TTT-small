def compute_c5(h):

def project_box_sum(y, target_sum):
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff > 0:
        margins = 1.0 - y
        total = margins.sum()
        if total > 1e-14:
            y = y + (diff / total) * margins
    else:
        total = y.sum()
        if total > 1e-14:
            y = y + (diff / total) * y
    return np.clip(y, 0.0, 1.0)

1. Start with small alpha (smooth landscape, easy to navigate)
2. Gradually increase alpha (sharper landscape, more focused)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(y, target_sum):
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff > 0:
        margins = 1.0 - y
        total = margins.sum()
        if total > 1e-14:
            y = y + (diff / total) * margins
    else:
        total = y.sum()
        if total > 1e-14:
            y = y + (diff / total) * y
    return np.clip(y, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    elif current_sum < target_sum:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_smooth_grad(h, corr, alpha):

    target = int(n // 2)

    positions = np.arange(n)

        if time.time() - start_time > time_limit:

        zeros = np.where(h == 0.0)[0]

        top5_idx = np.argsort(corr)[-5:]

                #           or flip h[i] from 0->1 (reduces by h[i+s])

                #   sum over i in [0,n-1-s] where p = i+s: (1-h[i]) = (1-h[p-s]) if p>=s
                # Score for flipping position p from 0 to 1:
                #   sum over i in [0,n-1-s] where p = i: h[i+s] = h[p+s] if p < n-s
                score_to_zero = np.zeros(n)
                if s < n:
                    score_to_zero[s:] = 1.0 - h[:n - s]
                score_to_one = np.zeros(n)
                if s < n:
                    score_to_one[:n - s] = h[s:]
            else:
                m = -s
                # corr_{-m} = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])

                score_to_zero[m:] = 1.0 - h[:n - m]

            best_i = -1
            best_j = -1
            best_gain = 0
            top_i_candidates = ones[np.argsort(-score_to_zero[ones])[:min(20, len(ones))]]

                        best_i = i
                        best_j = j

                new_corr = np.correlate(h, 1.0 - h, mode='full')

def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    def make_obj(alpha):
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        return obj
    def cons_fun(h):
        return np.sum(h) - target
    def cons_jac(h):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(make_obj(alpha), h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)
