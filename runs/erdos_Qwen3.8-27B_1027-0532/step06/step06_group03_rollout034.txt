```python
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

    if diff > 0:

    grad_half = np.zeros(n_half)
    for i in range(n_half):

    return grad_half
```

**Key new idea: Use the Fourier domain**

In the Fourier domain, the autocorrelation C(k) is the inverse Fourier transform of |H(ω)|². Minimizing the peak of the inverse FT of |H(ω)|² is equivalent to making |H(ω)|² as flat as possible.

So we want to find h (with h(x) = 1-h(2-x)) such that |H(ω)|² is as flat as possible, subject to 0≤h≤1 and ∫h=1.

This is like designing a window function with flat spectrum! The optimal solution would be close to a "Kaiser window" or similar.

Let me try a Fourier-based approach:
1. Start with a symmetric h
2. Compute its Fourier transform H(ω)
3. Compute P(ω) = |H(ω)|²
4. The time-domain peak is related to the deviation of P(ω) from its mean
5. Update h to flatten P(ω): h_new = h * (P_mean / P(ω)) in Fourier domain
6. Clip to [0,1] and re-symmetrize
7. Repeat

This is similar to the Gerchberg-Saxton algorithm or significant-component interpolation.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h):
    n = len(h)
    h_sym = np.empty(n)
    n_half = n // 2
    h_half = np.zeros(n_half)
    for i in range(n_half):
        h_half[i] = 0.5 * (h[i] + 1.0 - h[n-1-i])
    h_sym[:n_half] = h_half
    h_sym[n_half:] = 1.0 - h_half[::-1]
    if n % 2 == 1:
        h_sym[n_half] = 0.5
    return h_sym

def half_to_full(h_half):
    n = 2 * len(h_half)
    h = np.empty(n)
    h[:n//2] = h_half
    h[n//2:] = 1.0 - h_half[::-1]
    return h

def full_to_sym_half(h):
    n = len(h)
    n_half = n // 2
    h_half = np.zeros(n_half)
    for i in range(n_half):
        h_half[i] = 0.5 * (h[i] + 1.0 - h[n-1-i])
    return h_half

def smooth_obj_half(h_half, alpha):
    h = half_to_full(h_half)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad_half(h_half, alpha):
    h = half_to_full(h_half)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    n_half = n // 2
    grad_full = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad_full[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad_full[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad_full *= dx
    grad_half = np.zeros(n_half)
    for i in range(n_half):
        grad_half[i] = grad_full[i] - grad_full[n-1-i]
    return grad_half
```

```python
def coord_descent_half(h_half, n_iters, time_limit, start_time):
    n_half = len(h_half)
    full_n = 2 * n_half
    best_h = h_half.copy()
    best_c5 = compute_c5(half_to_full(best_h))
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n_half):
            # Try to optimize h_half[i]
            old_val = h_half[i]
            # Golden section search on [0, 1]
            a, b = 0.0, 1.0
            gr = (np.sqrt(5) - 1) / 2
            for _ in range(30):
                c = b - gr * (b - a)
                d = a + gr * (b - a)
                h_half[i] = c
                fc = compute_c5(half_to_full(h_half))
                h_half[i] = d
                fd = compute_c5(half_to_full(h_half))
                if fc < fd:
                    b = d
                else:
                    a = c
            h_half[i] = 0.5 * (a + b)
            new_c5 = compute_c5(half_to_full(h_half))
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_half.copy()
                improved = True
            else:
                h_half[i] = old_val
        if not improved:
            break
    return best_h, best_c5
```

This is great! It means coordinate descent is efficient: for each i, we can find the optimal δ by:
1. Computing the vector v[s] = g_i(s) - g_{n-1-i}(s) (the coefficient of δ in Δcorr[s])
2. The new corr is corr_old + δ*v
3. We want to minimize max_s (corr_old[s] + δ*v[s]) * dx over δ such that h[i]+δ ∈ [0,1] and h[n-1-i]-δ ∈ [0,1]
4. This is a 1D convex optimization: minimize f(δ) = max_s (a_s + δ*v_s) subject to δ ∈ [δ_lo, δ_hi]

The function f(δ) = max_s (a_s + δ*v_s) is the maximum of linear functions, hence convex. We can minimize it using:
- Subgradient method: f'(δ) = v_s* where s = argmax
- Bisection on the subgradient: find δ where 0 ∈ subgradient(f)(δ)
- Or just ternary search / golden section (since it's convex 1D)

For ternary search, each evaluation is O(n_corr) = O(2n-1), and we need about 40 evaluations for good precision. So per coordinate i, it's O(40 * 2n) = O(80n).
```python
def coord_descent_sym(h_half, time_limit, start_time, max_sweeps=1000):
    n_half = len(h_half)
    full_n = 2 * n_half
    dx = 2.0 / full_n
    best_h = h_half.copy()
    best_c5 = compute_c5(half_to_full(best_h))
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n_half):
            h = half_to_full(h_half)
            corr = np.correlate(h, 1.0 - h, mode='full')
            n = full_n
            mid = n - 1
            
            # Compute v[s] = coefficient of delta in corr[s]
            # v[s] = (1-h[i-(s-mid)]) * 1_{0<=i-(s-mid)<n} - h[i+(s-mid)] * 1_{0<=i+(s-mid)<n}
            #        - (1-h[n-1-i-(s-mid)]) * 1_{0<=n-1-i-(s-mid)<n} + h[n-1-i+(s-mid)] * 1_{0<=n-1-i+(s-mid)<n}
            
            v = np.zeros(2*n-1)
            j1 = i
            j2 = n-1-i
            for s in range(2*n-1):
                k = s - mid
                # Term from j1 as first factor
                if 0 <= j1 - k < n:
                    v[s] += (1.0 - h[j1 - k])
                # Term from j1 as second factor
                if 0 <= j1 + k < n:
                    v[s] -= h[j1 + k]
                # Term from j2 as first factor (with -delta)
                if 0 <= j2 - k < n:
                    v[s] -= (1.0 - h[j2 - k])
                # Term from j2 as second factor (with -delta)
                if 0 <= j2 + k < n:
                    v[s] += h[j2 + k]
            
            # Now minimize max_s (corr[s] + delta * v[s]) * dx
            # subject to 0 <= h[i] + delta <= 1 and 0 <= h[n-1-i] - delta <= 1
            # i.e., -h[i] <= delta <= 1 - h[i] and h[n-1-i] - 1 <= delta <= h[n-1-i]
            # Since h[n-1-i] = 1 - h[i] (by symmetry):
            # -h[i] <= delta <= 1 - h[i] and -h[i] <= delta <= 1 - h[i]
            # So delta ∈ [-h[i], 1-h[i]]
            
            delta_lo = -h_half[i]
            delta_hi = 1.0 - h_half[i]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h):
    n = len(h)
    n_half = n // 2
    h_half = np.zeros(n_half)
    for i in range(n_half):
        h_half[i] = 0.5 * (h[i] + 1.0 - h[n-1-i])
    h_sym = np.empty(n)
    h_sym[:n_half] = h_half
    h_sym[n_half:] = 1.0 - h_half[::-1]
    if n % 2 == 1:
        h_sym[n_half] = 0.5
    return h_sym

def half_to_full(h_half):
    n = 2 * len(h_half)
    h = np.empty(n)
    h[:n//2] = h_half
    h[n//2:] = 1.0 - h_half[::-1]
    return h

def full_to_sym_half(h):
    n = len(h)
    n_half = n // 2
    h_half = np.zeros(n_half)
    for i in range(n_half):
        h_half[i] = 0.5 * (h[i] + 1.0 - h[n-1-i])
    return h_half

def smooth_obj_half(h_half, alpha):
    h = half_to_full(h_half)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_grad_half(h_half, alpha):
    h = half_to_full(h_half)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    switched = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_switched = np.exp(switched)
    sum_exp = np.sum(exp_switched)
    weights = exp_switched / sum_exp
    mid = n - 1
    n_half = n // 2
    grad_full = np.zeros(n)
    for j in range(n):
        s_lo = max(j - n + 1, -(n - 1))
        s_hi = min(j, n - 1)
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            k_vals = j - s_vals
            grad_full[j] += np.dot(weights[s_vals + mid], 1.0 - h[k_vals])
        s_lo2 = max(-j, -(n - 1))
        s_hi2 = min(n - 1 - j, n - 1)
        if s_hi2 >= s_lo2:
            s_vals2 = np.arange(s_lo2, s_hi2 + 1)
            k_vals2 = j + s_vals2
            grad_full[j] -= np.dot(weights[s_vals2 + mid], h[k_vals2])
    grad_full *= dx
    grad_half = np.zeros(n_half)
    for i in range(n_half):
        grad_half[i] = grad_full[i] - grad_full[n-1-i]
    return grad_half

def slsqp_half(h_half, alphas, maxiter, ftol, time_limit, start_time):
    n_half = len(h_half)
    bounds = [(0.0, 1.0)] * n_half
    h_cur = h_half.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj_half, h_cur, args=(alpha,),
                           jac=smooth_grad_half,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = np.clip(res.x, 0.0, 1.0)
            c_new = compute_c5(half_to_full(h_new))
            c_cur = compute_c5(half_to_full(h_cur))
            if c_new < c_cur:
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def coord_descent_sym_fast(h_half, time_limit, start_time, max_sweeps=2000):
    n_half = len(h_half)
    full_n = 2 * n_half
    dx = 2.0 / full_n
    best_h = h_half.copy()
    best_c5 = compute_c5(half_to_full(best_h))
    k_range = np.arange(-(full_n-1), full_n)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        h = half_to_full(h_half)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mid = full_n - 1
        
        for i in range(n_half):
            if time.time() - start_time > time_limit:
                break
            j1 = i
            j2 = full_n - 1 - i
            
            v = np.zeros(2*full_n-1)
            m1 = (j1 - k_range >= 0) & (j1 - k_range < full_n)
            v[m1] += (1.0 - h[j1 - k_range[m1]])
            m2 = (j1 + k_range >= 0) & (j1 + k_range < full_n)
            v[m2] -= h[j1 + k_range[m2]]
            m3 = (j2 - k_range >= 0) & (j2 - k_range < full_n)
            v[m3] -= (1.0 - h[j2 - k_range[m3]])
            m4 = (j2 + k_range >= 0) & (j2 + k_range < full_n)
            v[m4] += h[j2 + k_range[m4]]
            
            delta_lo = -h_half[i]
            delta_hi = 1.0 - h_half[i]
            if delta_hi - delta_lo < 1e-14:
                continue
            
            # Ternary search
            a, b = delta_lo, delta_hi
            for _ in range(30):
                m1t = a + (b - a) / 3.0
                m2t = b - (b - a) / 3.0
                f1 = np.max(corr + m1t * v)
                f2 = np.max(corr + m2t * v)
                if f1 < f2:
                    b = m2t
                else:
                    a = m1t
            delta_star = 0.5 * (a + b)
            
            if abs(delta_star) > 1e-10:
                h_half[i] += delta_star
                # Update corr incrementally
                corr += delta_star * v
                # Update h
                h[j1] += delta_star
                h[j2] -= delta_star
                new_c5 = float(np.max(corr)) * dx
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best_h = h_half.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
```
