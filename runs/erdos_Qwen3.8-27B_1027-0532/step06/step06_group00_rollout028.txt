from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=40, n_shifts=5, top_k_weight=2.0):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        else:
            consecutive_no_improve += 1

            break

        c_max = np.max(corr)
        # Use exponential weighting for shifts near the max
        weights = np.exp(top_k_weight * (corr - c_max) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)
        
        grad = np.zeros(n)
        for s_idx in range(n_corr):
            w = weights[s_idx]
            if w < 1e-6:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):

            for jj in range(n_l):
                j = int(low_set[jj])

                    continue

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):

                    vals = corr + mid_d * g

                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0

                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def p_power_descent(h_init, n, target, time_limit, p=20, n_iter=5000, step_decay=0.999):
    """Gradient descent on (1/p) * sum corr_k^p with projection."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.01

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        # Gradient of (1/p) * sum corr_k^p
        corr_p = np.power(corr, p - 1)  # d/dh of corr^p = p * corr^(p-1) * dcorr/dh
        # grad[i] = sum_k corr_k^(p-1) * dcorr_k/dh[i]
        # dcorr_k/dh[i] = (1-h[i+k]) for valid i+k, minus h[i-k] for valid i-k
        # (this is the change matrix row)
        # Use weighted sum of change rows
        grad = np.zeros(n)
        shifts = np.arange(-(n-1), n)
        # Only use top correlations to save time
        top_k = 50
        top_idx = np.argsort(corr)[-top_k:]
        for s_idx in top_idx:
            w = corr[s_idx] ** (p - 1)
            k = int(s_idx) - (n - 1)
            i_arr = np.arange(n)
            v1 = (i_arr - k >= 0) & (i_arr - k < n)
            v2 = (i_arr + k >= 0) & (i_arr + k < n)
            grad[v1] += w * (1.0 - h[i_arr[v1] - k])
            grad[v2] -= w * h[i_arr[v2] + k]
        
        grad -= np.mean(grad)
        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
        else:
            step *= 0.9

    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)

        if abs(diff) < 1e-14:

            h = np.clip(h + diff / n, 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(30, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def block_pair_cd(h_init, n, target, time_limit, max_sweeps=15000,
                  block_size=3, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    n_blocks = n // block_size

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        n_cand_b = min(20, n_blocks // 2)
        sorted_b = np.argsort(block_grads)
        low_blocks = sorted_b[:n_cand_b]
        high_blocks = sorted_b[-n_cand_b:]

        best_trial_c5 = c5_val
        best_move = None

        for bh in high_blocks:
            bl = low_blocks[(int(bh) + 7) % len(low_blocks)]
            i_cent = int(bh) * block_size + block_size // 2
            j_cent = int(bl) * block_size + block_size // 2

            i_lo = max(0, int(bh) * block_size)
            i_hi = min(n, (int(bh) + 1) * block_size)
            j_lo = max(0, int(bl) * block_size)
            j_hi = min(n, (int(bl) + 1) * block_size)

            for i in range(i_lo, i_hi):
                for j in range(j_lo, j_hi):
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue

                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(16):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue

                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def p_power_gradient(h_init, n, target, time_limit, p=10, n_iter=8000, step_init=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(20, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx] ** (p - 1)
            if w < 1e-20:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.85

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000],
                                500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Weighted pair CD at initial resolution
    if tleft() > 120:
        h_cd, c5_cd = weighted_pair_cd(best_h, n0, target0,
                                       time_limit=min(150, tleft() * 0.5),
                                       max_sweeps=25000,
                                       n_cand_start=15, n_cand_end=55,
                                       max_no_improve=350,
                                       weight_beta=5.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800 and optimize
    n1 = 800
    target1 = n1 / 2.0
    if tleft() > 350:
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 300:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [50, 200, 1000, 5000],
                                      400, 1e-14)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
                n0 = n1
                target0 = target1
            else:
                n0 = n1
                target0 = target1
                best_h = h_pol1  # still use the polished version for further optimization
                if best_score > compute_c5(best_h):
                    # keep original best
                    pass

        if tleft() > 250:
            h_cd1, c5_cd1 = weighted_pair_cd(best_h, n0, target0,
                                             time_limit=min(280, tleft() * 0.5),
                                             max_sweeps=20000,
                                             n_cand_start=15, n_cand_end=50,
                                             max_no_improve=300,
                                             weight_beta=6.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 100:
            h_pol1b, sc1b = slsqp_round(best_h, n0, target0,
                                        [5000, 20000],
                                        500, 1e-14)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 4: Annealed pair CD to escape local minima
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_an, c5_an = weighted_pair_cd(best_h, n_cur, target_cur,
                                       time_limit=min(130, tleft() * 0.5),
                                       max_sweeps=15000,
                                       n_cand_start=20, n_cand_end=50,
                                       max_no_improve=400,
                                       weight_beta=4.0,
                                       use_anneal=True, T0=0.0005)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 5: p-power gradient descent
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pp, c5_pp = p_power_gradient(best_h, n_cur, target_cur,
                                       time_limit=min(80, tleft() * 0.5),
                                       p=8, n_iter=5000, step_init=0.003)
        if c5_pp < best_score:
            best_score = c5_pp
            best_h = h_pp

    # Phase 6: Block pair CD (structured moves)
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bc, c5_bc = block_pair_cd(best_h, n_cur, target_cur,
                                    time_limit=min(80, tleft() * 0.5),
                                    max_sweeps=8000,
                                    block_size=3, max_no_improve=250)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 7: SLSQP with very high alpha (near-exact max)
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pol2, sc2 = slsqp_round(best_h, n_cur, target_cur,
                                  [10000, 50000, 100000],
                                  600, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 8: Multi-start with different perturbations
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.08]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = slsqp_round(h_pert, n_cur, target_cur,
                                        [200, 2000, 10000],
                                        400, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 50:
                h_cd_ref, c5_cd_ref = weighted_pair_cd(best_h, n_cur, target_cur,
                                                       time_limit=min(35, tleft() * 0.4),
                                                       max_sweeps=8000,
                                                       n_cand_start=15, n_cand_end=40,
                                                       max_no_improve=150,
                                                       weight_beta=5.0)
                if c5_cd_ref < best_score:
                    best_score = c5_cd_ref
                    best_h = h_cd_ref

    # Phase 9: Try finer grid if time allows
    if tleft() > 250:
        n_cur = len(best_h)
        if n_cur < 1200:
            n2 = 1200
            target2 = n2 / 2.0
            h2 = interpolate_to(best_h, n2, target2)

            if tleft() > 200:
                h_pol2h, sc2h = slsqp_round(h2, n2, target2,
                                            [100, 500, 2000],
                                            300, 1e-14)
                if tleft() > 150:
                    h_cd2, c5_cd2 = weighted_pair_cd(h_pol2h, n2, target2,
                                                     time_limit=min(180, tleft() * 0.5),
                                                     max_sweeps=12000,
                                                     n_cand_start=15, n_cand_end=40,
                                                     max_no_improve=250,
                                                     weight_beta=5.0)
                    if c5_cd2 < best_score:
                        best_score = c5_cd2
                        best_h = h_cd2
                    if tleft() > 50:
                        h_pol3, sc3 = slsqp_round(best_h, n2, target2,
                                                  [5000, 20000],
                                                  400, 1e-14)
                        if sc3 < best_score:
                            best_score = sc3
                            best_h = h_pol3

    # Phase 10: Final weighted pair CD polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_cd, c5_final = weighted_pair_cd(best_h, n_cur, target_cur,
                                                time_limit=min(50, tleft() * 0.6),
                                                max_sweeps=8000,
                                                n_cand_start=25, n_cand_end=60,
                                                max_no_improve=200,
                                                weight_beta=8.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [50000, 100000],
                                    400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
