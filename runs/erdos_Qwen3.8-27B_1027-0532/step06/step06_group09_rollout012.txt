import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_step(h_cur, target, epsilon=0.05, top_m=None):
    """
    One step of the LP-based active set method.
    Returns the LP solution (threshold function).
    """
    n = len(h_cur)
    corr = compute_corr(h_cur)
    c_max = np.max(corr)
    
    # Determine active set
    if top_m is not None:
        active = np.argsort(corr)[-top_m:]
    else:
        threshold = c_max * (1.0 - epsilon)
        active = np.where(corr >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Compute cost vector
    mid = n - 1
    c_cost = np.zeros(n)
    for m in active:
        s = m - mid  # shift value
        # c_cost[j] += (1 - h_cur[j+s]) - h_cur[j-s]
        # with valid index checks
        for j in range(n):
            jp = j + s
            jm = j - s
            if 0 <= jp < n:
                c_cost[j] += (1.0 - h_cur[jp])
            if 0 <= jm < n:
                c_cost[j] -= h_cur[jm]
    
    # Solve the LP: minimize sum c_cost[j] * h[j] s.t. sum h = target, 0<=h<=1
    # Solution: sort c_cost, set h=1 for smallest, h=0 for largest, one fractional
    sorted_idx = np.argsort(c_cost)
    h_lp = np.zeros(n)
    remaining = target
    for j in sorted_idx:
        if remaining >= 1.0 - 1e-12:
            h_lp[j] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            h_lp[j] = remaining
            remaining = 0.0
        else:
            break
    
    return h_lp, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target):
    h_bin = (h >= 0.5).astype(float)
    current_sum = h_bin.sum()
    if current_sum > target:
        above = np.where(h_bin == 1)[0]
        n_flip = int(round(current_sum - target))
        for pos in np.argsort(h[above])[:n_flip]:
            h_bin[above[pos]] = 0.0
    elif current_sum < target:

def lp_active_step(h_cur, target, epsilon, use_top_m=None):
    n = len(h_cur)
    corr = compute_corr(h_cur)
    c_max = np.max(corr)
    if use_top_m is not None:
        active = np.argsort(corr)[-use_top_m:]
    else:
        threshold = c_max * (1.0 - epsilon)
        active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    mid = n - 1
    c_cost = np.zeros(n)
    idx = np.arange(n)
    for m in active:
        s = m - mid
        jp_mask = (idx + s >= 0) & (idx + s < n)
        c_cost[jp_mask] += (1.0 - h_cur[idx[jp_mask] + s])
        jm_mask = (idx - s >= 0) & (idx - s < n)
        c_cost[jm_mask] -= h_cur[idx[jm_mask] - s]
    sorted_idx = np.argsort(c_cost)
    h_lp = np.zeros(n)
    remaining = target
    for pos in range(n):
        j = sorted_idx[pos]
        if remaining >= 1.0 - 1e-12:
            h_lp[j] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            h_lp[j] = remaining
            remaining = 0.0
        else:
            break
    return h_lp

            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        if no_improve > patience // 3 and n_cand < n_cand_max:
            n_cand = min(n_cand + 3, n_cand_max)
        elif no_improve < patience // 5 and n_cand > n_cand_start:
            n_cand = max(n_cand - 2, n_cand_start)
        k_star = int(np.argmax(corr))
        s_star = k_star - mid
        grad = np.zeros(n)
        v1 = (idxs + s_star >= 0) & (idxs + s_star < n)
        v2 = (idxs - s_star >= 0) & (idxs - s_star < n)
        grad[v1] += (1.0 - h[idxs[v1] + s_star])
        grad[v2] -= h[idxs[v2] - s_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            ip = (i + shifts >= 0) & (i + shifts < n)
            i_sp = np.zeros(n_corr)
            i_sp[ip] = h[i + shifts[ip]]
            im = (i - shifts >= 0) & (i - shifts < n)
            i_sm = np.zeros(n_corr)
            i_sm[im] = h[i - shifts[im]]
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                jp = (j + shifts >= 0) & (j + shifts < n)
                j_pm = np.zeros(n_corr)
                j_pm[jp] = h[j + shifts[jp]]
                jm = (j - shifts >= 0) & (j - shifts < n)
                j_sm = np.zeros(n_corr)
                j_sm[jm] = h[j - shifts[jm]]
                g = i_sp + i_sm - j_pm - j_sm
                a_d = max(delta_lo, -1e-10)
                b_d = min(delta_hi, 1e10)
                if b_d - a_d < 1e-10:
                    continue
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lbfags_polish(h_start, n, target, alpha, maxiter=1500):
    def smooth_obj(h):
        corr = compute_corr(h)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def game_theoretic_descent(h_init, n, target, time_limit, max_steps=8000, eta_base=1.5, m_top=10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    best_lr = 0.05
    no_improve_count = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 300:
                break
        top_m = np.argsort(corr)[-m_top:]
        lam_top = lam[top_m]
        lam_top = lam_top / (lam_top.sum() + 1e-30)
        grad = np.zeros(n)
        for wi, s_idx in zip(lam_top, top_m):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        lr = best_lr * max(0.1, 1.0 - step / (max_steps * 1.5))
        for attempt in range(8):
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
            lr *= 0.5
        if improved:
            best_lr = min(best_lr * 1.05, 0.5)
        else:
            best_lr *= 0.9
            if best_lr < 0.001:
                break
        exp_arg = np.clip(eta_base * (corr - c_max) / (c_max + 1e-10), -30, 0)
        lam_new = np.exp(exp_arg)
        lam_new = lam_new / (lam_new.sum() + 1e-30)
        lam = 0.7 * lam + 0.3 * lam_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    # Phase 1: LP-based active set descent
    if tleft() > 200:
        lp_time = min(150, tleft() * 0.15)
        h_lp, c5_lp = lp_descent(best_h, n0, target0, time_limit=lp_time, max_iters=2000, seed=seed)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
    # Phase 2: L-BFGS-B polish
    if tleft() > 150:
        for alpha in [200, 1000, 5000]:
            if tleft() < 40:
                break
            h_try = lbfags_polish(best_h, n0, target0, alpha, maxiter=1000)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
    # Phase 3: Game-theoretic descent
    if tleft() > 200:
        gt_time = min(150, tleft() * 0.15)
        h_gt, c5_gt = game_theoretic_descent(best_h, n0, target0, time_limit=gt_time, max_steps=8000, eta_base=2.0, m_top=10)
        if c5_gt < best_score:
            best_score = c5_gt
            best_h = h_gt
    # Phase 4: Continuous pair CD
    if tleft() > 250:
        cd_time = min(300, tleft() * 0.3)
        h_cd, c5_cd = continuous_pair_cd(best_h, n0, target0, time_limit=cd_time, max_sweeps=8000, n_cand_start=30, n_cand_max=70, patience=200, seed=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    # Phase 5: Binary pair CD
    if tleft() > 100:
        bc_time = min(80, tleft() * 0.1)
        h_bin, c5_bin = binary_pair_cd(best_h, n0, target0, time_limit=bc_time, max_sweeps=3000, n_cand=30, patience=100, seed=seed+1)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin
    # Phase 6: LP descent again (now from a better point)
    if tleft() > 150:
        lp_time = min(100, tleft() * 0.12)
        h_lp2, c5_lp2 = lp_descent(best_h, n0, target0, time_limit=lp_time, max_iters=2000, seed=seed+2)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2
    # Phase 7: Second round continuous pair CD with different params
    if tleft() > 200:
        cd_time = min(200, tleft() * 0.25)
        h_cd2, c5_cd2 = continuous_pair_cd(best_h, n0, target0, time_limit=cd_time, max_sweeps=6000, n_cand_start=50, n_cand_max=100, patience=150, seed=seed+3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
    # Phase 8: L-BFGS-B high alpha
    if tleft() > 100:
        for alpha in [10000, 50000]:
            if tleft() < 30:
                break
            h_try = lbfags_polish(best_h, n0, target0, alpha, maxiter=800)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try
    # Phase 9: Multi-restart from perturbations
    rng = np.random.RandomState(seed + 10)
    if tleft() > 150:
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = lbfags_polish(h_pert, n0, target0, 5000, maxiter=600)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_cd3, c5_cd3 = continuous_pair_cd(best_h, n0, target0, time_limit=min(50, tleft() * 0.2), max_sweeps=3000, n_cand_start=30, n_cand_max=60, patience=100, seed=seed+20)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3
    # Phase 10: Try higher resolution
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, target_try, 2000, maxiter=800)
            sc1 = compute_c5(h_hires)
            h_hires2, c5_hires2 = continuous_pair_cd(h_hires, n_try, target_try, time_limit=min(60, tleft() * 0.25), max_sweeps=3000, n_cand_start=30, n_cand_max=60, patience=100, seed=seed+30)
            if c5_hires2 < best_score:
                best_score = c5_hires2
                best_h = h_hires2
            elif sc1 < best_score:
                best_score = sc1
                best_h = h_hires
    # Phase 11: Final polish at current resolution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = lbfags_polish(best_h, n_use, target_use, 100000, maxiter=500)
        sc_final = compute_c5(h_final_try)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_try
    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        final_score = best_score
    return h_final, final_score, n_use
