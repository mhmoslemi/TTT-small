    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

    if abs(diff) < 1e-14:

    h = np.clip(h, 0.0, 1.0)

        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

    cm = np.max(C)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]

        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)

        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5

def subgrad_soft(h_start, n, target, time_limit, max_steps, alpha_w, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 600:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]

        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 3

    return best_h, best_c5

def round_to_binary(h, n, target):
    h_bin = np.zeros(n)
    h_sorted = np.sort(h)[::-1]
    k_ones = int(target)
    if k_ones >= n:
        k_ones = n - 1
    h_bin[np.argsort(h)[::-1][:k_ones]] = 1.0
    residual = target - np.sum(h_bin)
    if abs(residual) > 1e-14:
        frac_idx = np.argmin(np.abs(h - 0.5))
        if residual > 0:
            h_bin[frac_idx] = min(1.0, residual)
        else:
            candidates = np.where(h_bin > 0)[0]
            if len(candidates) > 0:
                pick = candidates[np.argmin(h_bin[candidates])]
                h_bin[pick] = max(0.0, h_bin[pick] + residual)
    return project_feasible(h_bin, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_slsqp(h_init, n, target, n_restarts, time_limit, rng):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        noise = 0.05 * rng.randn(n)
        h_try = project_feasible(h_init + noise, target)
        h_pol, sc = slsqp_opt(h_try, n, target, [500, 2000, 10000, 50000], 800, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pol
    return best_h, best_c5
