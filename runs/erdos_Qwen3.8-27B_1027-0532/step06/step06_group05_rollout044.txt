        m = np.zeros(n)
        v = np.zeros(n)
        t = 0

            # Cosine annealing

            lr = lr_base * 0.5 * (1 + np.cos(np.pi * phase_progress))
            lr = max(lr, lr_base * 0.01)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)

        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
            best_smooth = local_best
        else:
            h = best_h.copy()

- h[a] appears in the (1-h[...]) factor if a = i+k for some i, i.e., i = a-k, valid if 0 ≤ a-k and a-k+k ≤ n-1-k, i.e., 0 ≤ a < n-k

Δcorr[k] = -(1-h[a+k]) + h[a-k] + (1-h[b+k]) - h[b-k]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

        v = np.zeros(n)
        t = 0

        local_best = best_smooth
        max_iter_alpha = 400

        for _ in range(max_iter_alpha):
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-12:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 250:
                break

            phase_progress = t / max_iter_alpha
            lr = lr_base * 0.5 * (1.0 + np.cos(np.pi * phase_progress))
            lr = max(lr, lr_base * 0.01)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
            best_smooth = local_best
        else:
            h = best_h.copy()

    return best_h, best_actual


def flip_effect(a, b, h, top_indices, n):
    ea = np.zeros(len(top_indices))
    eb = np.zeros(len(top_indices))
    for im in range(len(top_indices)):
        k = int(top_indices[im]) - (n - 1)
        da = 0.0
        db = 0.0
        if k >= 0:
            if 0 <= a and a <= n - 1 - k:
                da -= (1.0 - h[a + k])
            if 0 <= a - k and a - k < n:
                da += h[a - k]
            if 0 <= b and b <= n - 1 - k:
                db += (1.0 - h[b + k])
            if 0 <= b - k and b - k < n:
                db -= h[b - k]
        else:
            kk = -k
            if -kk <= a and a <= n - 1:
                da -= (1.0 - h[a + kk])
            if 0 <= a - kk and a - kk < n:
                da += h[a - kk]
            if -kk <= b and b <= n - 1:
                db += (1.0 - h[b + kk])
            if 0 <= b - kk and b - kk < n:
                db -= h[b - kk]
        ea[im] = da
        eb[im] = db
    return ea, eb


def steepest_descent_binary(h0, target_sum, max_iters, time_limit, start_time, top_m=40):
    n = len(h0)
    h = h0.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5_raw = float(corr.max())
    best_h = h.copy()
    dx = 2.0 / n
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_m_actual = min(top_m, 2 * n - 1)
        top_indices = np.argsort(corr)[-top_m_actual:]
        top_vals = corr[top_indices].astype(float)

        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 0.0
        best_a = -1
        best_b = -1

        n_ones = len(ones)
        n_zeros = len(zeros)

        for ia in range(n_ones):
            a = int(ones[ia])
            ea = np.zeros(top_m_actual)
            for im in range(top_m_actual):
                k = int(top_indices[im]) - (n - 1)
                d = 0.0
                if k >= 0:
                    if 0 <= a and a <= n - 1 - k:
                        d -= (1.0 - h[a + k])
                    if 0 <= a - k and a - k < n:
                        d += h[a - k]
                else:
                    kk = -k
                    if -kk <= a and a <= n - 1:
                        d -= (1.0 - h[a + kk])
                    if 0 <= a - kk and a - kk < n:
                        d += h[a - kk]
                ea[im] = d

            for ib in range(n_zeros):
                b = int(zeros[ib])
                eb = np.zeros(top_m_actual)
                for im in range(top_m_actual):
                    k = int(top_indices[im]) - (n - 1)
                    d = 0.0
                    if k >= 0:
                        if 0 <= b and b <= n - 1 - k:
                            d += (1.0 - h[b + k])
                        if 0 <= b - k and b - k < n:
                            d -= h[b - k]
                    else:
                        kk = -k
                        if -kk <= b and b <= n - 1:
                            d += (1.0 - h[b + kk])
                        if 0 <= b - kk and b - kk < n:
                            d -= h[b - kk]
                    eb[im] = d

                new_vals = top_vals + ea + eb
                new_max = float(np.max(new_vals))
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_a = a
                    best_b = b

        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max())
            if sc < best_c5_raw:
                best_c5_raw = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5_raw * dx


def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width=25):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 250

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        top_threshold = max_val * 0.85
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices[:15]:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
                    if best_improve > 3.0:
                        break
            if best_improve > 3.0:
                break
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 12000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.45:
            worst = np.argsort(corr)[-10:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.80:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def subgradient_refine_fast(h0, target_sum, start_time, time_budget, n_iters=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(n_iters):
        if time.time() - start_time > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        threshold = c_max - 1.0
        active_k = np.where(corr >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for k_idx in active_k:
            k = int(k_idx) - (n - 1)
            if k >= 0:
                if n - 1 - k >= 0:
                    grad[:n - k] += (1.0 - h[k:n])
                if k > 0:
                    grad[k:] -= h[:n - k]
            else:
                kk = -k
                grad[kk:] += (1.0 - h[:n - kk])
                grad[:n - kk] -= h[kk:n]

        grad /= float(len(active_k))
        step = 0.08 / (1.0 + 0.005 * it)
        h_new = h - step * grad
        h = project_box_sum(h_new, target_sum)

        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    return best_h, best_score


def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy on binary projection
    h_bin = project_binary(best_h, int(target))
    if tleft() > 120:
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=25000,
                                  time_limit=min(60, tleft() * 0.08),
                                  start_time=start_time, scan_width=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Adam continuous with cosine annealing (main phase)
    if tleft() > 350:
        adam_time = min(500, tleft() - 250)
        h_a, s_a = adam_cosine(
            best_h, target, start_time, adam_time,
            alphas=[20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000],
            lr_base=0.003, beta1=0.85, beta2=0.999, eps=1e-8
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 3: Subgradient refinement
    if tleft() > 200:
        sg_time = min(100, tleft() - 150)
        h_sg, s_sg = subgradient_refine_fast(best_h, target, start_time, sg_time,
                                             n_iters=3000)
        if s_sg < best_score:
            best_score = s_sg
            best_h = h_sg.copy()

    # Phase 4: Multi-restart Adam from perturbed points
    if tleft() > 250:
        n_restarts = 4
        for ri in range(n_restarts):
            if tleft() < 100:
                break
            jitter_scale = 0.004 + 0.004 * ri
            h_pert = best_h + rng.randn(n) * jitter_scale
            h_pert = project_box_sum(h_pert, target)
            rt = min(120, (tleft() - 100) / (n_restarts - ri))
            h_ar, s_ar = adam_cosine(
                h_pert, target, start_time, rt,
                alphas=[100, 500, 2000, 8000, 30000, 80000],
                lr_base=0.002, beta1=0.85, beta2=0.999, eps=1e-8
            )
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    # Phase 5: Steepest descent binary local search (key improvement)
    if tleft() > 150:
        h_bin3 = project_binary(best_h, int(target))
        sd_time = min(200, tleft() - 100)
        h_sd, c5_sd = steepest_descent_binary(h_bin3, int(target),
                                              max_iters=5000,
                                              time_limit=sd_time,
                                              start_time=start_time,
                                              top_m=35)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd.copy()

    # Phase 6: SA for further binary refinement
    if tleft() > 120:
        h_bin4 = project_binary(best_h, int(target))
        sa_time = min(120, tleft() - 80)
        max_it = int(min(100000, sa_time * 400))
        h_sa, c5_sa = sa_binary(h_bin4, int(target), max_it,
                                0.005, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 7: SLSQP polish
    if tleft() > 70:
        slsqp_time = min(150, tleft() - 40)
        alphas_sl = [5000, 10000, 30000, 80000, 200000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     2000, 1e-18, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 8: Final short Adam polish
    if tleft() > 40:
        polish_time = max(15.0, tleft() - 15)
        h_p, s_p = adam_cosine(
            best_h, target, start_time, polish_time,
            alphas=[50000, 100000, 300000],
            lr_base=0.001, beta1=0.9, beta2=0.999, eps=1e-9
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Phase 9: Final steepest descent on the polished solution
    if tleft() > 60:
        h_bin5 = project_binary(best_h, int(target))
        sd_time2 = min(80, tleft() - 20)
        h_sd2, c5_sd2 = steepest_descent_binary(h_bin5, int(target),
                                                max_iters=3000,
                                                time_limit=sd_time2,
                                                start_time=start_time,
                                                top_m=30)
        if c5_sd2 < best_score:
            best_score = c5_sd2
            best_h = h_sd2.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
