def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target):
    n = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def get_delta(h, a, b, k, n):
    delta = 0.0
    if k >= 0:
        if k <= a <= n - 1:
            delta += h[a - k] - 1.0
        if 0 <= a <= n - 1 - k:
            if a + k < n:
                delta += h[a + k]
        if k <= b <= n - 1:
            if b - k >= 0:
                delta += 1.0 - h[b - k]
        if 0 <= b <= n - 1 - k:
            if b + k < n:
                delta -= h[b + k]
    else:
        m = -k
        if 0 <= a <= n - 1 - m:
            if a + m < n:
                delta += h[a + m] - 1.0
        if m <= a <= n - 1:
            delta += h[a - m]
        if 0 <= b <= n - 1 - m:
            if b + m < n:
                delta += 1.0 - h[b + m]
        if m <= b <= n - 1:
            delta -= h[b - m]
    return delta

def fast_binary_ls(h_start, target_sum, time_limit, start_time, K=60, max_moves_per_iter=400):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(corr.max())
    best_c5 = current_max * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200

    while time.time() - start_time < time_limit:
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))

        top_vals, top_idxs = np.sort(corr)[::-1][:K]
        critical_shifts = (top_idxs - (n - 1)).tolist()
        critical_indexed = [(int(corr[idx]), int(idx)) for idx in top_idxs]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 0.0
        best_a = -1
        best_b = -1

        for k_star in critical_shifts[:5]:
            if k_star == 0:
                continue
            p10 = get_10_pairs(h, k_star, n)
            p01 = get_01_pairs(h, k_star, n)
            if len(p10) == 0 or len(p01) == 0:
                continue

            p10.sort(key=lambda x: h[x] - (1.0 if 0 <= x + abs(k_star) < n or 0 <= x - abs(k_star) < n else 0.0))
            p01.sort(key=lambda x: h[x])

            max_p10 = min(len(p10), 25)
            max_p01 = min(len(p01), 25)

            for a in p10[:max_p10]:
                for b in p01[:max_p01]:
                    if a == b:
                        continue
                    new_max = 0.0
                    valid = True
                    for (_, idx) in critical_indexed:
                        k_i = idx - (n - 1)
                        d = get_delta(h, a, b, k_i, n)
                        v = corr[idx] + d
                        if v > new_max:
                            new_max = v
                        if new_max >= max_val:
                            valid = False
                            break

            h[best_b] = 1

                best_h = h.copy()
            stall = 0

            stall += 1

def smooth_objective(h, beta, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.sum(np.exp(shifted))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def smooth_gradient(h, beta, target, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)

    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        mask1 = (idx1 >= 0) & (idx1 < 2 * N - 1)
        mask2 = (idx2 >= 0) & (idx2 < 2 * N - 1)
        grad[j] = np.dot(one_minus_h[mask1], weights[idx1[mask1]]) - np.dot(h[mask2], weights[idx2[mask2]])

    grad *= dx
    if penalty > 0 and abs(float(np.sum(h)) - target) > 1e-10:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def lbfgs_refine(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()

    if time.time() - start_time > time_limit:
        return h_cur

    try:
        res = minimize(smooth_objective, h_cur,
                      args=(beta, target, penalty),
                      jac=smooth_gradient,
                      method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1})
        h_out = project_box_sum(res.x, target)
        if compute_c5(h_out) <= compute_c5(h_cur) + 1e-6:
            return h_out
    except Exception:
        pass
    return h_cur

def subgradient_refine(h_start, target, time_limit, start_time, steps=2000):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_box_sum(h, target)

    for step in range(steps):
        if time.time() - start_time > time_limit:
            break
        if step % 500 == 0:
            h = project_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)

        alpha = 0.1 / (1.0 + 0.01 * step)

        grad = np.zeros(n)
        if k_star >= 0:
            for j in range(n):
                if 0 <= j <= n - 1 - k_star:
                    if j + k_star < n:
                        grad[j] += (1.0 - h[j + k_star])
                if 0 <= j - k_star <= n - 1 - k_star and k_star <= j:
                    grad[j] -= h[j - k_star]
        else:
            m = -k_star
            for j in range(n):
                if 0 <= j <= n - 1 - m and j + m < n:
                    grad[j] += (1.0 - h[j + m])
                if m <= j <= n - 1 and j - m >= 0:
                    grad[j] -= h[j - m]

        grad *= dx
        h -= alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target)

    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

            best_h = h_final_ls.copy()

    # Phase 10: Final continuous polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use / 2)
        h_final_cont = project_box_sum(best_h.copy(), target_use)
        time_alloc = min(20, tleft() - 5)
        h_polished = lbfgs_refine(h_final_cont, 2000, target_use, penalty=3000.0,
                                   maxiter=100, time_limit=time_alloc,
                                   start_time=start_time)
        sc = compute_c5(h_polished)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_polished.copy()

    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    if final_c5 > best_c5:
        best_h = project_box_sum(best_h, target_final)
        final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
