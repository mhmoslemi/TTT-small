def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def perturb_swap(h, n_pairs, rng):
    h_new = h.copy()
    n = len(h)
    for _ in range(n_pairs):
        i, j = rng.choice(n, size=2, replace=False)
        vi, vj = h_new[i], h_new[j]
        h_new[i] = vj
        h_new[j] = vi
    return h_new

def perturb_block(h, rng, block_lens=(3, 5, 8)):
    h_new = h.copy()
    n = len(h)
    bl = int(rng.choice(block_lens))
    if bl >= n:
        return h_new
    start = int(rng.integers(0, n - bl))
    block = h_new[start:start + bl].copy()
    shift = int(rng.integers(-2, 3))
    if abs(shift) >= bl:
        return h_new
    if shift > 0:
        h_new[start:start + bl + shift] = np.concatenate([h_new[start + bl:start + bl + shift], block])
    else:
        h_new[start + shift:start + bl - shift] = np.concatenate([block, h_new[start:start + bl + shift]])
    return h_new

def perturb_value(h, n_vals, rng, scale=0.05):
    h_new = h.copy()
    n = len(h)
    for _ in range(n_vals):
        i = int(rng.integers(0, n))
        h_new[i] += rng.normal(0, scale)
    return h_new

def multistart_search(h_init, n, target, time_limit, rng, n_trials=8, polish_alphas=(500, 2000, 10000)):
    start = time.time()

    time_per_trial = time_limit / (n_trials + 1)

    for trial in range(n_trials):
        if time.time() - start > time_limit - 5:
            break
        h_try = h_init.copy()
        ptype = trial % 4
        if ptype == 0:
            h_try = perturb_swap(h_try, rng.integers(2, 5), rng)
        elif ptype == 1:
            h_try = perturb_block(h_try, rng)
        elif ptype == 2:

        else:
            h_try = perturb_swap(h_try, rng.integers(1, 3), rng)
            h_try = perturb_value(h_try, 2, rng, scale=0.02)
        h_try = project_feasible(h_try, target)

    if tleft() > 120:
        ms_time = min(150, tleft() * 0.4)
        hms, c5ms = multistart_search(best_h, n, target, time_limit=ms_time, rng=rng, n_trials=8,
                                      polish_alphas=(500, 2000, 10000))
        if c5ms < best_score:
            best_score = c5ms
            best_h = hms

    # Phase 5: Pair CD multi-shift
    if tleft() > 80:
        cd_time = min(100, tleft() * 0.4)
        hcd, c5cd = pair_cd_multi(best_h, n, target, time_limit=cd_time, n_cand=35, top_k=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: SLSQP
    if tleft() > 40:
        hp, sc = slsqp_optimize(best_h, n, target, [20000, 100000, 500000], 1500, 1e-15,
                                time_limit=min(35, tleft() * 0.7), t_start=time.time())
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Another multistart round
    if tleft() > 100:
        ms_time = min(120, tleft() * 0.5)
        hms, c5ms = multistart_search(best_h, n, target, time_limit=ms_time, rng=rng, n_trials=6,
                                      polish_alphas=(2000, 10000, 50000))
        if c5ms < best_score:
            best_score = c5ms
            best_h = hms

    # Phase 8: Pair CD
    if tleft() > 60:
        cd_time = min(80, tleft() * 0.5)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Swap local search
    if tleft() > 50:
        sls_time = min(60, tleft() * 0.5)
        hsl, c5sl = swap_local_search(best_h, n, target, time_limit=sls_time)
        if c5sl < best_score:
            best_score = c5sl
            best_h = hsl

    # Phase 10: Final SLSQP
    if tleft() > 30:
        hp, sc = slsqp_optimize(best_h, n, target, [50000, 200000, 1000000], 1000, 1e-15,
                                time_limit=min(25, tleft() * 0.8), t_start=time.time())
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final pair CD
    if tleft() > 30:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(25, tleft() * 0.7), n_cand=35)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 12: Final SLSQP
    if tleft() > 15:
        hp, sc = slsqp_optimize(best_h, n, target, [200000, 1000000], 800, 1e-15,
                                time_limit=min(12, tleft() * 0.8), t_start=time.time())
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
