            res = minimize(_smooth_obj, h, jac=_smooth_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-12, 'factr': 1e12, 'pgtol': 1e-12})
            h_new = project_box_sum(res.x, target)
            if compute_c5(h_new) < compute_c5(h) + 1e-15:
                h = h_new
        except Exception:
            pass
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_dx, _g_alpha, _g_pen
    global start_time_global, budget_s_global

    start_time_global = time.time()
    budget_s_global = budget_s

    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 40.0 - (time.time() - start_time_global)

    # Phase 1: FISTA with moderate K
    tl = remaining()
    if tl > 200:
        fw_time = min(300, tl * 0.35)
        h1, c1 = fista_subgrad(best_h, n, target, fw_time, K=60, lr_init=0.4)
        if c1 < best_score:
            best_score = c1
            best_h = h1

    # Phase 2: FISTA with different K and lr
    tl = remaining()

        fw_time = min(200, tl * 0.25)

        if c2 < best_score:
            best_score = c2
            best_h = h2

    # Phase 3: L-BFGS-B smooth optimization (main polishing)
    tl = remaining()
    if tl > 200:
        lp_time = min(350, tl * 0.4)
        h3 = smooth_only_phase(best_h, n, target, lp_time)
        c3 = compute_c5(h3)
        if c3 < best_score:
            best_score = c3
            best_h = h3

    # Phase 4: Another FISTA round from improved point
    tl = remaining()

        fw_time = min(120, tl * 0.3)
        h4, c4 = fista_subgrad(best_h, n, target, fw_time, K=40, lr_init=0.3)
        if c4 < best_score:
            best_score = c4
            best_h = h4

    # Phase 5: L-BFGS-B with higher alpha (sharper)
    tl = remaining()
    if tl > 150:
        lp_time = min(200, tl * 0.35)
        h5 = smooth_only_phase(best_h, n, target, lp_time)
        c5v = compute_c5(h5)
        if c5v < best_score:
            best_score = c5v
            best_h = h5

    # Phase 6: Multi-start FISTA for diversity
    tl = remaining()
    if tl > 120:
        ms_time = min(150, tl * 0.4)
        h6, c6 = multi_start_fista(best_h, n, target, ms_time, rng, n_starts=3)
        if c6 < best_score:
            best_score = c6
            best_h = h6

    # Phase 7: Final L-BFGS-B polish
    tl = remaining()
    if tl > 100:
        lp_time = min(120, tl * 0.4)
        h7 = smooth_only_phase(best_h, n, target, lp_time)
        c7 = compute_c5(h7)
        if c7 < best_score:
            best_score = c7
            best_h = h7

    # Phase 8: Final FISTA
    tl = remaining()
    if tl > 60:
        fw_time = min(50, tl * 0.5)
        h8, c8 = fista_subgrad(best_h, n, target, fw_time, K=25, lr_init=0.2)
        if c8 < best_score:
            best_score = c8
            best_h = h8

    # Phase 9: Final polish with very high alpha
    tl = remaining()
    if tl > 40:
        _g_alpha = 500000
        _g_pen = 100000
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_smooth_obj, best_h, jac=_smooth_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-20,
                                    'gtol': 1e-12, 'factr': 1e12, 'pgtol': 1e-12})
            h9 = project_box_sum(res.x, target)
            c9 = compute_c5(h9)
            if c9 < best_score:
                best_score = c9
                best_h = h9
        except Exception:
            pass

    # Final feasibility fix
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
