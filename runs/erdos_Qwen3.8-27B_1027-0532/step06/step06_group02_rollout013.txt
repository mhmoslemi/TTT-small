def compute_smooth_gradient(h, alpha, corr_full):
    n = len(h)
    max_c = corr_full.max()
    shifted = alpha * (corr_full - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    grad = 1.0 - 2.0 * conv
    return grad

                break
            corr = full_corr_fast(h)
            grad = compute_smooth_gradient(h, alpha, corr)

            step = lr
            for _bt in range(10):
                h_new = project_box_sum(h + step * v, target)
                sc = compute_c5(h_new)
                if sc < compute_c5(h) + 1e-10 or _bt == 9:

            h = h_new
            if it % 30 == 0 or it == n_iter - 1:

                    best_score = sc
                    best_h = h.copy()
                    backtrack_count = 0
                else:

                    if backtrack_count > 5:

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(-alpha * (c_max - corr), -500.0, 0.0)
    return c_max - float(np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-10:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    remaining = total_time - elapsed_at_start

        return best_h, best_score

    restart_count = 0
    max_restarts = 15
    t_init = 0.03
    iters_per_round = 30000

    for round_i in range(max_restarts):
        if time.time() - start_time > total_time - 5:
            break
        round_time = min(60.0, total_time - 5 - (time.time() - start_time))
        if round_time < 5:
            break
        t_f = t_init * (0.001 ** (1.0 / max(iters_per_round, 1)))
        h_curr = best_h.copy()
        h_sa, c5_sa = sa_improved(h_curr, iters_per_round, t_init, max(t_f, 0.00001),
                                   rng, round_time, start_time, k_arr,
                                   block_frac=0.15, max_block_len=10, double_swap_frac=0.1)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
            restart_count = 0
            t_init *= 0.85
        else:
            restart_count += 1
            if restart_count > 4:
                h_pert = perturb_diverse(best_h, rng, rng.randint(3, 8))
                best_h = h_pert
                restart_count = 0
                t_init = min(t_init, 0.025)
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    if tleft() < 20:
        return best_h, best_score, n_init

    # Phase 1: Quick greedy to improve initial
    allowed = min(30, tleft() * 0.05)
    h_g, c5_g = fast_greedy(best_h, max_iters=20000, rng=rng,
                            time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Intensive SA with restarts (main search)
    if tleft() > 100:
        allowed = min(250, tleft() * 0.3)
        h_sa, c5_sa = sa_with_restarts(best_h, allowed + (time.time() - start_time),
                                        start_time, rng, k_arr, n_init)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Focused SA from best
    if tleft() > 80:
        allowed = min(120, tleft() * 0.2)
        n_sa = int(min(2000000, allowed * 50000))
        h_sa2, c5_sa2 = sa_focused(best_h, n_sa, 0.015, 0.00003, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Gradient-based continuous refinement
    if tleft() > 60:
        target_use = float(n_init) / 2.0
        alphas = [10, 30, 80, 200, 600, 2000, 6000, 15000]
        allowed = min(200, tleft() * 0.25)
        h_ref, sc = gradient_refine(best_h, target_use, start_time, allowed, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: Project to binary and do focused SA again
    if tleft() > 60:
        h_bin = project_binary(best_h, target_init)
        allowed = min(60, tleft() * 0.12)
        n_sa3 = int(min(1500000, allowed * 50000))
        h_sa3, c5_sa3 = sa_focused(h_bin, n_sa3, 0.01, 0.00002, rng,
                                    time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 6: Multi-start with diverse perturbations
    for restart in range(6):
        if tleft() < 50:
            break
        r = rng.random()
        if r < 0.4:
            h_pert = perturb_diverse(best_h, rng, rng.randint(5, 15))
        elif r < 0.7:
            h_pert = perturb_block(best_h, rng, 15)
        else:
            h_pert = perturb_diverse(best_h, rng, 3)
            h_pert = perturb_block(h_pert, rng, 10)

        allowed = min(40, tleft() * 0.08)
        h_gr, c5_gr = fast_greedy(h_pert, max_iters=8000, rng=rng,
                                   time_limit=min(allowed*0.3, 10), start_time=start_time, k_arr=k_arr)
        n_sa_r = int(min(1000000, allowed * 40000))
        h_sr, c5_sr = sa_improved(h_gr, n_sa_r, 0.01, 0.00002, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr,
                                   block_frac=0.15, max_block_len=8, double_swap_frac=0.1)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: More gradient refinement
    if tleft() > 50:
        target_use = float(n_init) / 2.0
        alphas2 = [50, 150, 500, 1500, 5000, 15000, 30000]
        allowed = min(120, tleft() * 0.2)
        h_ref2, sc2 = gradient_refine(best_h, target_use, start_time, allowed, alphas2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 8: SLSQP refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_slsqp = [100, 400, 1500, 5000, 15000, 40000]
        allowed = min(tleft() - 15, 150)
        h_sl, sc_sl = slsqp_refine(best_h, n_use, target_use, alphas_slsqp, 2000, 1e-14,
                                    time_limit=allowed, start_time=start_time)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl.copy()

    # Phase 9: Final SA burst
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        k_arr_f = np.arange(2 * n_use - 1) - (n_use - 1)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(500000, max(30000, (tleft() - 10) * 30000)))
        h_sa_f, c5_f = sa_focused(h_bin_final, n_sa_f, 0.005, 0.00001,
                                   rng, time_limit=max(5.0, tleft() - 8),
                                   start_time=start_time, k_arr=k_arr_f)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 10: Final gradient polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = max(5.0, tleft() - 8)
        h_ref3, sc3 = gradient_refine(best_h, target_use, start_time, allowed,
                                       [5000, 15000, 40000, 80000])
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
