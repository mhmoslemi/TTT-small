        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)

def compute_g(h, i, j, shifts, n):
    """Compute g_k for pair (i,j)"""
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    ip = (i + shifts >= 0) & (i + shifts < n)
    im = (i - shifts >= 0) & (i - shifts < n)
    g[ip] = h[i + shifts[ip]]
    g[im] += h[i - shifts[im]]
    jp = (j + shifts >= 0) & (j + shifts < n)
    jm = (j - shifts >= 0) & (j - shifts < n)
    g[jp] -= h[j + shifts[jp]]
    g[jm] -= h[j - shifts[jm]]
    return g

def bisection_delta(corr, g, dlo, dhi):
    """Find delta minimizing max(corr + delta*g)"""
    for _ in range(16):
        mid = 0.5 * (dlo + dhi)
        v = corr + mid * g
        fm = np.max(v)
        active = v >= fm - 1e-13
        fp = np.max(g[active]) if active.any() else 0.0
        if fp > 1e-11:
            dhi = mid
        else:
            dlo = mid
    return 0.5 * (dlo + dhi)

def pair_cd(h, n, target, tlimit, n_cand=50, patience=200, seed=0):

    dx = 2.0 / n

            best_h = h.copy()
            npat = 0
        else:
            npat += 1
            if npat > patience:
                break
        
        # Gradient at worst shift
        ks = int(np.argmax(corr)) - mid
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - ks >= 0) & (idxs - ks < n)
        v2 = (idxs + ks >= 0) & (idxs + ks < n)
        grad[v1] += 1.0 - h[idxs[v1] - ks]
        grad[v2] -= h[idxs[v2] + ks]
        grad -= grad.mean()
        
        si = np.argsort(grad)
        lo = si[:n_cand]
        hi = si[-n_cand:]

        bm = None

            ip = (i + shifts >= 0) & (i + shifts < n)

                dlo = max(h[i] - 1.0, -h[j])

                jp = (j + shifts >= 0) & (j + shifts < n)
                jm = (j - shifts >= 0) & (j - shifts < n)
                g[jp] -= h[j + shifts[jp]]
                g[jm] -= h[j - shifts[jm]]
                ds = bisection_delta(corr, g, dlo, dhi)
                if abs(ds) < 1e-12:
                    continue
                amax = np.max(corr + ds * g) * dx
                if amax < bmc - 1e-15:

def lbbf(h, n, target, alpha, maxiter=1000):
    """L-BFGS-B with smooth max approximation"""
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        shifted = np.clip(alpha * (corr - cmax), -500, 0)
        return (cmax + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except:
        return h.copy()

    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

            best_c5 = c5v
            best_h = h.copy()
            npat = 0
        else:
            npat += 1
            if npat > patience:
                break

        ks = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - ks >= 0) & (idxs - ks < n)
        v2 = (idxs + ks >= 0) & (idxs + ks < n)
        grad[v1] += 1.0 - h[idxs[v1] - ks]
        grad[v2] -= h[idxs[v2] + ks]
        grad -= grad.mean()

        # Adaptive n_cand: increase when converging slowly
        nc = n_cand
        if npat > patience // 2:
            nc = min(nc + 10, 120)

        si = np.argsort(grad)
        lo = si[:nc]
        hi = si[-nc:]

        bmc = c5v
        bm = None
        for i in hi:
            n_corr = 2 * n - 1
            g_i = np.zeros(n_corr)
            ip = (i + shifts >= 0) & (i + shifts < n)
            im = (i - shifts >= 0) & (i - shifts < n)
            g_i[ip] = h[i + shifts[ip]]
            g_i[im] += h[i - shifts[im]]
            for j in lo:
                if i == j:
                    continue
                dhi_val = min(h[i], 1.0 - h[j])
                dlo_val = max(h[i] - 1.0, -h[j])
                if dhi_val - dlo_val < 1e-12:
                    continue
                g = g_i.copy()
                jp = (j + shifts >= 0) & (j + shifts < n)
                jm = (j - shifts >= 0) & (j - shifts < n)
                g[jp] -= h[j + shifts[jp]]
                g[jm] -= h[j - shifts[jm]]
                ds = bisection_delta(corr, g, dlo_val, dhi_val)
                if abs(ds) < 1e-12:
                    continue
                amax = np.max(corr + ds * g) * dx
                if amax < bmc - 1e-15:
                    bmc = amax
                    bm = (int(i), int(j), ds)

        if bm is None:
            for _ in range(30):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                dhi_val = min(h[i], 1.0 - h[j])
                dlo_val = max(h[i] - 1.0, -h[j])
                if dhi_val - dlo_val < 1e-12:
                    continue
                g = compute_g(h, i, j, shifts, n)
                ds = bisection_delta(corr, g, dlo_val, dhi_val)
                if abs(ds) < 1e-12:
                    continue
                amax = np.max(corr + ds * g) * dx
                if amax < bmc - 1e-15:
                    bmc = amax
                    bm = (i, j, ds)

        if bm is not None:
            i, j, ds = bm
            h[i] -= ds
            h[j] += ds
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            npat += 5
            if npat > patience:
                break

    return best_h, best_c5

def sa_pair_moves(h, n, target, tlimit, t_init=0.05, t_final=0.0005, seed=0):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)
    total_est = max(100, int(tlimit * 200))
    npat = 0

    for step in range(total_est * 10):
        if time.time() - start > tlimit:
            break
        prog = min(1.0, step / total_est)
        #adaptive T: boost if stuck
        T = t_init * (t_final / t_init) ** prog
        if npat > 200:
            T = max(T, t_init * 0.3)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr)) * dx

        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
            npat = 0
        else:
            npat += 1
            if npat > 2000:
                break

        # Biased pair selection: prefer interior values
        weights = 1.0 + 4.0 * h * (1.0 - h)
        wsum = weights.sum()
        weights /= wsum
        i = int(rng.choice(n, p=weights))
        j = int(rng.randint(n))
        if i == j:
            j = (j + 1) % n
            if i == j:
                continue

        dhi_val = min(h[i], 1.0 - h[j])
        dlo_val = max(h[i] - 1.0, -h[j])
        if dhi_val - dlo_val < 1e-12:
            continue

        g = compute_g(h, i, j, shifts, n)
        ds = bisection_delta(corr, g, dlo_val, dhi_val)
        if abs(ds) < 1e-12:
            continue

        new_c5 = np.max(corr + ds * g) * dx
        de = new_c5 - c5v

        if de < 0 or rng.rand() < np.exp(-de / max(T, 1e-12)):
            h[i] -= ds
            h[j] += ds
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def smooth_obj_penalty(h, alpha, penalty, target, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    smooth_max = (cmax + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    penalty_term = penalty * (np.sum(h) - target) ** 2
    return smooth_max + penalty_term

def lbfgs_with_penalty(h, n, target, alpha, penalty, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_penalty, h, args=(alpha, penalty, target, n),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

def structured_perturb(h, n, target, seed):
    rng = np.random.RandomState(seed)
    h2 = h.copy()
    # Add a smooth sinusoidal perturbation
    freq = rng.choice([1, 2, 3, 4, 5])
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.1)
    x = np.linspace(0, 2 * np.pi, n)
    h2 += amp * np.sin(freq * x + phase)
    # Add small random noise
    h2 += rng.randn(n) * 0.02
    return project_feasible(h2, target)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def tl():
        return budget_s - 25.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Pair CD convergence
    if tl() > 120:
        t1 = min(150, tl() * 0.18)
        h_r, c5_r = pair_cd(h, n, target, t1, n_cand=50, patience=250, seed=seed)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 2: Simulated annealing (key innovation)
    if tl() > 150:
        t2 = min(300, tl() * 0.28)
        h_sa, c5_sa = sa_pair_moves(best_h, n, target, t2, t_init=0.06, t_final=0.0003, seed=seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 3: Pair CD after SA
    if tl() > 120:
        t3 = min(200, tl() * 0.22)
        h_r, c5_r = pair_cd(best_h, n, target, t3, n_cand=70, patience=300, seed=seed + 2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: L-BFGS-B with penalty + pair CD
    if tl() > 100:
        for alpha, pen in [(500, 500.0), (2000, 2000.0), (8000, 10000.0)]:
            if tl() < 30:
                break
            h_try = lbfgs_with_penalty(best_h, n, target, alpha, pen, maxiter=500)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try
        if tl() > 60:
            h_r, c5_r = pair_cd(best_h, n, target, min(80, tl() * 0.3), n_cand=80, patience=150, seed=seed + 3)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 5: Higher resolution
    if tl() > 200:
        for n_try in [600, 800]:
            if tl() < 130:
                break
            target_try = n_try / 2.0
            h_interp = project_feasible(interpolate_h(best_h, n_try), target_try)
            h_hires = lbfgs_with_penalty(h_interp, n_try, target_try, 1000, 1000.0, maxiter=300)
            h_hires, c5_hi = pair_cd(h_hires, n_try, target_try, min(100, tl() * 0.45), n_cand=45, patience=100, seed=seed + n_try)
            sc = compute_c5(h_hires)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_hires

    # Phase 6: Second SA round on best
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tl() > 80:
        h_sa2, c5_sa2 = sa_pair_moves(best_h, n_cur, target_cur, min(80, tl() * 0.5), t_init=0.03, t_final=0.0002, seed=seed + 100)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2
        if tl() > 40:
            h_r, c5_r = pair_cd(best_h, n_cur, target_cur, min(40, tl() * 0.5), n_cand=60, patience=80, seed=seed + 200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 7: Structured restart + optimization
    if tl() > 100:
        for s in range(3):
            if tl() < 50:
                break
            h_pert = structured_perturb(best_h, n_cur, target_cur, seed + 500 + s)
            h_try, c5_try = pair_cd(h_pert, n_cur, target_cur, min(30, tl() * 0.4), n_cand=40, patience=50, seed=seed + 600 + s)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 8: Final polish
    if tl() > 30:
        h_f = lbfgs_with_penalty(best_h, n_cur, target_cur, 20000, 50000.0, maxiter=400)
        sc = compute_c5(h_f)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_f
        if tl() > 15:
            h_r, c5_r = pair_cd(best_h, n_cur, target_cur, min(15, tl() * 0.6), n_cand=100, patience=30, seed=seed + 999)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
