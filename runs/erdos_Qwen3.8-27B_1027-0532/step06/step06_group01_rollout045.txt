import numpy as np
from scipy.optimize import minimize, linprog
import time

_SLQ_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET


def _slq_con_jac(x):
    return np.ones(len(x))


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n

    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

            res = minimize(_smooth_obj, h_cur, args=(alpha,),

                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)

            h_cur = h_new

            pass

                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, max_steps, n_shifts=3, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 500:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def frank_wolfe_step(h, n, target, n_active=5, temperature=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    top_idx = np.argsort(corr)[-n_active:]
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    w = np.exp(temperature * (corr[top_idx] - cur_max))
    w = w / w.sum()
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    s_indices = np.argsort(grad)
    s = np.zeros(n)
    s[s_indices[:int(target)]] = 1.0
    return s, grad


def fw_cd_hybrid(h_init, n, target, time_limit, max_iters=50000, n_active=5, n_cand=15):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        s, grad = frank_wolfe_step(h, n, target, n_active, 50.0)
        toward = np.where((s == 1.0) & (h < 1.0 - 1e-10))[0]
        away = np.where((s == 0.0) & (h > 1e-10))[0]
        if len(toward) == 0 or len(away) == 0:
            no_improve += 50
            continue
        toward_top = toward[np.argsort(-grad[toward])[:n_cand]]
        away_top = away[np.argsort(grad[away])[:n_cand]]

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        ks = np.arange(-(n - 1), n)
        n_corr = 2 * n - 1

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in away_top:
            hi_val = float(h[int(i)])
            if hi_val < 1e-12:
                continue
            m_i = int(i)
            idx1i = m_i - ks
            idx2i = m_i + ks
            vali1 = (idx1i >= 0) & (idx1i < n)
            vali2 = (idx2i >= 0) & (idx2i < n)
            pi = np.zeros(n_corr)
            if np.any(vali1):
                pi[vali1] += h[idx1i[vali1]]
            if np.any(vali2):
                pi[vali2] += h[idx2i[vali2]]
            for j in toward_top:
                if int(i) == int(j):
                    continue
                hj_val = float(h[int(j)])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                m_j = int(j)
                idx1j = m_j - ks
                idx2j = m_j + ks
                valj1 = (idx1j >= 0) & (idx1j < n)
                valj2 = (idx2j >= 0) & (idx2j < n)
                pj = np.zeros(n_corr)
                if np.any(valj1):
                    pj[valj1] += h[idx1j[valj1]]
                if np.any(valj2):
                    pj[valj2] += h[idx2j[valj2]]
                g = pi - pj
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 50:

    if tleft() > 200:

                if c5cd < sc:
