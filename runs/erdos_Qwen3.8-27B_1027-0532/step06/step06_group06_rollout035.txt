    h = h.copy().astype(float)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 300:
            break

        # Get top M lags
        top_idx = np.argsort(corr)[-top_m:]
        top_idx = top_idx[::-1]  # descending

        #                = h[i+k] + h[i-k] - h[j+k] - h[j-k] + (something with validity)

        # h[i-k] is 1 if h[i-k]=1, 0 otherwise

        # influence_remove[p, k] = -(1-h[p+k])*[valid] + h[p-k]*[valid]  (contribution when h[p] goes 1->0)
        # influence_add[p, k] = (1-h[p+k])*[valid] - h[p-k]*[valid]  (contribution when h[p] goes 0->1)
        
        # delta_k(i,j) = influence_remove[i, k] + influence_add[j, k]
        
        # Precompute influence matrices for top lags
        # influence_remove for ones: shape (n_ones, top_m)
        # influence_add for zeros: shape (n_zeros, top_m)
        
        inf_rem = np.zeros((len(ones), top_m))
        inf_add = np.zeros((len(zeros), top_m))

            k = k_idx - (n - 1)  # actual lag
            # influence_remove[i, k]: h[i] goes 1->0
            # = -(1-h[i+k])*[0<=i+k<n] + h[i-k]*[0<=i-k<n]
            i_plus = ones + k

            i_minus = ones - k
            valid_minus = (i_minus >= 0) & (i_minus < n)
            
            inf_rem[:, m_idx] = -np.where(valid_plus, 1.0 - h[i_plus[valid_plus]], 0.0) + \

            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)) + 1e-300)) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

def lse_gradient(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    wsum = float(weights.sum())
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        wsum = 1.0
    weights /= wsum
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        k_min = max(-j, -(n - 1))
        k_max = min(n - 1 - j, n - 1)
        if k_max >= k_min:
            ks = np.arange(k_min, k_max + 1)
            ki = ks + (n - 1)
            grad[j] += np.dot(weights[ki], one_minus_h[j + ks])
        k_min2 = max(j - (n - 1), -(n - 1))
        k_max2 = min(j, n - 1)
        if k_max2 >= k_min2:
            ks2 = np.arange(k_min2, k_max2 + 1)
            ki2 = ks2 + (n - 1)
            grad[j] -= np.dot(weights[ki2], h[j - ks2])
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

def continuous_optimize(h_start, beta, target, penalty, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    try:
        res = minimize(lse_objective, h_cur, args=(beta, target, penalty),
                      jac=lse_gradient, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': -1})
        h_proj = project_box_sum(res.x, target)
        sc = compute_c5(h_proj)
        ob = lse_objective(h_proj, beta, target, penalty)
        ob_cur = lse_objective(h_cur, beta, target, penalty)
        if ob < ob_cur or sc < compute_c5(h_cur):
            return h_proj
        return h_cur
    except Exception:
        return h_cur

def sa_binary_guided(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 8000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new = h.copy()
        moved = False
        r = rng.random()
        if r < 0.6:
            top_lags = np.argsort(corr)[-15:]
            k = top_lags[rng.randint(15)] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
            else:
                h_new[ones[rng.randint(len(ones))]] = 0
                h_new[zeros[rng.randint(len(zeros))]] = 1
                moved = True
        elif r < 0.85:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n / 2)
    target_int = int(n / 2)

    best_h = project_box_sum(h_init.copy(), target)
    best_c5 = compute_c5(best_h)
    h_bin = project_binary(h_init, target_int)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Enhanced binary local search
    if tleft() > 60:
        alloc = min(300, tleft() * 0.3)
        h_ls, c5_ls = enhanced_binary_ls(best_h, target_int, time_limit=alloc,
                                         start_time=start_time, top_m=25, max_stall=500)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 2: Continuous optimization from best
    if tleft() > 60:
        h_cont = project_box_sum(best_h.copy(), target)
        for beta in [10, 30, 80, 200]:
            if tleft() < 30:
                break
            alloc2 = min(40, tleft() * 0.15)
            h_opt = continuous_optimize(h_cont, beta, target, 500.0, 200, alloc2, start_time)
            h_cont = h_opt
            sc = compute_c5(h_cont)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont.copy()

    # Phase 3: Binary local search after continuous
    if tleft() > 50:
        h_bin2 = project_binary(best_h, target_int)
        alloc = min(200, tleft() * 0.2)
        h_ls2, c5_ls2 = enhanced_binary_ls(h_bin2, target_int, time_limit=alloc,
                                           start_time=start_time, top_m=30, max_stall=600)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 4: SA guided
    if tleft() > 80:
        h_bin3 = project_binary(best_h, target_int)
        alloc = min(150, tleft() * 0.2)
        max_it = int(min(150000, alloc * 2000))
        h_sa, c5_sa = sa_binary_guided(h_bin3, target_int, max_it, 0.006, 0.00003,
                                       rng, time_limit=alloc, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: More continuous with higher beta
    if tleft() > 60:
        h_cont2 = project_box_sum(best_h.copy(), target)
        for beta in [100, 300, 800, 1500]:
            if tleft() < 25:
                break
            alloc3 = min(35, tleft() * 0.15)
            h_opt = continuous_optimize(h_cont2, beta, target, 1000.0, 150, alloc3, start_time)
            h_cont2 = h_opt
            sc = compute_c5(h_cont2)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont2.copy()

    # Phase 6: Multi-restart continuous
    if tleft() > 80:
        for restart in range(3):
            if tleft() < 40:
                break
            pert = rng.normal(0, 0.04, n)
            h_pert = project_box_sum(best_h + pert, target)
            h_r = h_pert
            for beta_r in [50, 150, 500]:
                if tleft() < 20:
                    break
                alloc_r = min(25, tleft() * 0.2)
                h_opt = continuous_optimize(h_r, beta_r, target, 800.0, 180, alloc_r, start_time)
                h_r = h_opt
                sc = compute_c5(h_r)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_r.copy()

    # Phase 7: Binary LS from continuous result
    if tleft() > 50:
        h_bin4 = project_binary(best_h, target_int)
        alloc = min(100, tleft() * 0.2)
        h_ls4, c5_ls4 = enhanced_binary_ls(h_bin4, target_int, time_limit=alloc,
                                           start_time=start_time, top_m=20, max_stall=400)
        if c5_ls4 < best_c5:
            best_c5 = c5_ls4
            best_h = h_ls4.copy()

    # Phase 8: Try different n
    n_options = [n + 4, n - 4, n + 8, n - 8]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n]
    for n_try in n_options[:3]:
        if tleft() < 60:
            break
        tgt_try = float(n_try / 2)
        tgt_try_int = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, tgt_try)
        alloc_i = min(25, tleft() * 0.12)
        h_opt_i = continuous_optimize(h_interp, 30, tgt_try, 500.0, 150, alloc_i, start_time)
        sc_i = compute_c5(h_opt_i)
        if sc_i < best_c5:
            best_c5 = sc_i
            best_h = h_opt_i.copy()
        if tleft() < 40:
            continue
        h_bin_i = project_binary(h_opt_i, tgt_try_int)
        alloc_b = min(40, tleft() * 0.15)
        h_ls_i, c5_ls_i = enhanced_binary_ls(h_bin_i, tgt_try_int, time_limit=alloc_b,
                                             start_time=start_time, top_m=20, max_stall=300)
        if c5_ls_i < best_c5:
            best_c5 = c5_ls_i
            best_h = h_ls_i.copy()

    # Phase 9: Final SA polish
    if tleft() > 50:
        n_use = len(best_h)
        tgt_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, tgt_use)
        alloc_f = min(80, tleft() - 10)
        max_it_f = int(min(100000, alloc_f * 2500))
        h_sf, c5_sf = sa_binary_guided(h_bin_f, tgt_use, max_it_f, 0.003, 0.00001,
                                       rng, time_limit=alloc_f, start_time=start_time)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Phase 10: Final binary LS
    if tleft() > 30:
        n_use = len(best_h)
        tgt_use = int(n_use / 2)
        h_bin_f2 = project_binary(best_h, tgt_use)
        alloc_f2 = min(50, tleft() - 10)
        h_sf2, c5_sf2 = enhanced_binary_ls(h_bin_f2, tgt_use, time_limit=alloc_f2,
                                           start_time=start_time, top_m=25, max_stall=300)
        if c5_sf2 < best_c5:
            best_c5 = c5_sf2
            best_h = h_sf2.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final / 2)
    if target_final == int(target_final):
        best_h = project_binary(best_h, int(target_final))
    else:
        best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
