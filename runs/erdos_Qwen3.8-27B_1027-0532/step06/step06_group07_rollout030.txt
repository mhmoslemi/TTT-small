```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=10000, n_cand=50, n_bisect=12, top_k=15, n_active_lags=3):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    adapt_count = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            adapt_count = 0
        else:
            no_improve += 1

        if no_improve > 100:
            if n_cand < 80:
                n_cand = min(80, n_cand + 10)
                no_improve = 0
            else:
                break

        # Compute gradient using top active lags
        sorted_corr = np.argsort(corr)
        active_lags = sorted_corr[-n_active_lags:]
        c_min_active = corr[sorted_corr[-(n_active_lags + 1)]] if len(sorted_corr) > n_active_lags else 0.0

        grad = np.zeros(n)
        for s_idx in active_lags:
            k = int(s_idx) - mid
            w = max(0.0, corr[int(s_idx)] - c_min_active) + 0.1
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        nc = min(n_cand, n // 2)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        # Precompute g_i for each high_set member
        g_cache = {}
        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]
            g_cache[i] = g_i

        # Quick scoring with small delta
        test_d = 0.02
        scores = np.full((len(high_set), len(low_set)), c5_val / dx)

        for ai, i in enumerate(high_set):
            hi = h[i]
            delta_hi = hi  # max we can take from i
            if delta_hi < 1e-12:
                continue
            g_i = g_cache[i]
            td = min(test_d, delta_hi)
            for aj, j in enumerate(low_set):
                if i == j:
                    continue
                if h[j] >= 1.0 - 1e-12:
                    continue
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i + 0
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]
                scores[ai, aj] = np.max(corr + td * g)

        # Get top pairs
        flat_scores = scores.ravel()
        top_pairs = np.argsort(flat_scores)[:top_k]

        for pi in top_pairs:
            ai, aj = divmod(int(pi), len(low_set))
            i = high_set[ai]
            j = low_set[aj]
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g_i = g_cache[i]
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g = g_i.copy()
            g[v_jp] += (1.0 - h[j + shifts[v_jp]])
            g[v_jm] -= h[j - shifts[v_jm]]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        elif no_improve > 40:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_out, target_dst)

def fourier_exploration(h, n, target, time_limit, max_steps=2000, n_fqs=8):
    """Try modifying h in Fourier space to escape local optima."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    rng = np.random.RandomState(42)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        # Pick random frequencies
        freqs = rng.choice(1, n // 2 + 1, size=n_fqs, replace=False)
        best_trial = best_c5
        best_basis = None
        best_delta = 0.0

        for freq in freqs:
            for func in [np.cos, np.sin]:
                basis = func(2.0 * np.pi * freq * idxs / n)
                basis = basis - np.mean(basis)
                bnorm = np.linalg.norm(basis)
                if bnorm < 1e-10:
                    continue
                basis = basis / bnorm
                for delta in [0.02, 0.05, -0.02, -0.05]:
                    h_try = h + delta * basis
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial - 1e-12:
                        best_trial = sc
                        best_basis = basis.copy()
                        best_delta = delta

        if best_basis is not None:
            h = h + best_delta * best_basis
            h = project_feasible(h, target)
            best_c5 = best_trial
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_h, best_c5
