            return o

            _, g = smooth_max_and_grad(x, alpha, n)
            return g

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    for _ in range(30):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def smooth_max_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    obj = (c_max + np.log(ws) / alpha) * dx
    mid = n - 1
    grad = np.zeros(n)
    active_s = np.where(w > 1e-8)[0]
    for s in active_s:
        lag = int(s) - mid
        if lag >= 0:
            n_lo = 0
            n_hi = n - 1 - lag
        else:
            n_lo = -lag
            n_hi = n - 1
        if n_lo > n_hi:
            continue
        i1_lo = max(0, n_lo + lag)
        i1_hi = min(n - 1, n_hi + lag)
        if i1_lo <= i1_hi:
            i1 = np.arange(i1_lo, i1_hi + 1)
            grad[i1] += w[s] * (1.0 - h[i1 - lag])
        i2 = np.arange(n_lo, n_hi + 1)
        grad[i2] += w[s] * (-h[i2 + lag])
    grad *= dx
    grad -= np.mean(grad)
    return obj, grad

def constraint_sum_eq(x, target):
    return np.sum(x) - target

def constraint_sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    dead = start_time + time_limit
    for alpha in alphas:
        if time.time() > dead:
            break
        obj_fn = functools.partial(smooth_max_obj_wrap, alpha=alpha, n=n)
        grad_fn = functools.partial(smooth_max_grad_wrap, alpha=alpha, n=n)
        cons = [{'type': 'eq',
                 'fun': functools.partial(constraint_sum_eq, target=target),
                 'jac': functools.partial(constraint_sum_eq_jac, n=n)}]
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def smooth_max_obj_wrap(x, alpha, n):
    o, _ = smooth_max_and_grad(x, alpha, n)
    return o

def smooth_max_grad_wrap(x, alpha, n):

    return g

def fast_subgradient_cd(h0, n, target, time_limit, max_sweeps=50000, n_top=5):

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        else:

def lp_refine_all(h0, n, target, time_limit, n_active_frac=0.15, max_rounds=200, eps_prox=0.15):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        c_max = float(np.max(corr))
        thresh = c_max * (1.0 - n_active_frac)
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        t_target = c5_val * 0.999

        h_var = cp.Variable(n)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        constraints.append(h_var >= h - eps_prox)
        constraints.append(h_var <= h + eps_prox)

        for s_idx in active:
            lag = int(s_idx) - mid
            if lag >= 0:
                n_lo = 0
                n_hi = n - 1 - lag
            else:
                n_lo = -lag
                n_hi = n - 1
            if n_lo > n_hi:
                continue
            grad_k = np.zeros(n)
            i1_lo = max(0, n_lo + lag)
            i1_hi = min(n - 1, n_hi + lag)
            if i1_lo <= i1_hi:
                i1 = np.arange(i1_lo, i1_hi + 1)
                grad_k[i1] += (1.0 - h[i1 - lag])
            i2 = np.arange(n_lo, n_hi + 1)
            grad_k[i2] += (-h[i2 + lag])
            lin_expr = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_expr <= t_target / dx)

        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), constraints)
        h_new = None
        for solver in [cp.ECOS, cp.CVXOPT, cp.SCIP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    h_new = np.clip(h_var.value, 0.0, 1.0)
                    h_new = project_feasible(h_new, target)
                    break
            except Exception:
                continue
        if h_new is None:
            break

        c5_new = compute_c5(h_new)
        if c5_new < c5_val - 1e-12:
            h = h_new
        else:
            best_mix = None
            best_mix_c5 = c5_val
            for mix in [0.5, 0.3, 0.7, 0.2]:
                h_mix = project_feasible(mix * h + (1.0 - mix) * h_new, target)

                    best_mix_c5 = c5_mix
                    best_mix = h_mix
            if best_mix is not None and best_mix_c5 < c5_val - 1e-12:
                h = best_mix
            else:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

                    time_limit=min(40.0, tleft() * 0.25),
                    n_active_frac=0.1, max_rounds=40, eps_prox=0.1)
                if s_hr3 < best_score:
                    best_score = s_hr3
                    best_h = h_hr3

    # Phase 8: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_f, s_f = slsqp_optimize(best_h, n_use, target_use,
                                    [2000, 10000, 50000, 200000],
                                    1500, 1e-15,
                                    min(30.0, tleft() * 0.7), start_time)
        if s_f < best_score:
            best_score = s_f
            best_h = h_f

    # Phase 9: Final CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_f2, s_f2 = fast_subgradient_cd(best_h, n_use, target_use,
                                           time_limit=min(15.0, tleft() * 0.7),
                                           max_sweeps=20000, n_top=5)
        if s_f2 < best_score:
            best_score = s_f2
            best_h = h_f2

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
