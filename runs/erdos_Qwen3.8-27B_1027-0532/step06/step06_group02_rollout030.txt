def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            # Remove the ones with lowest h values
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            # Add the zeros with highest h values
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def adam_optimize(h0, target, alphas, iters_per_alpha, lr, beta1, beta2, eps, start_time, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    N_fft = 1
    while N_fft < 2*n:
        N_fft <<= 1
    
    global_t = 0
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            global_t += 1
            
            corr = full_corr_fast(h)
            grad = compute_smooth_gradient_fft(h, alpha, corr, N_fft)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**global_t)
            v_hat = v / (1 - beta2**global_t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_box_sum(h_new, target)
            h = h_new
            
            if (it + 1) % 200 == 0 or it == iters_per_alpha - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    return best_h, best_score

        # After change h[j]: 0→1, the change in corr_k is +(1-h[j+k]) if j<n-k
