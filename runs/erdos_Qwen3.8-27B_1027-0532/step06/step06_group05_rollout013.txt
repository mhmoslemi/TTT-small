def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    # h(x)(1-h(x+k)) summed over x
    # = sum h(x) - sum h(x)h(x+k)
    # = sum(h) - autocorrelation(h)[k]
    # autocorrelation via FFT
    N = 2 * n
    H = np.fft.fft(h, N)
    acorr = np.fft.ifft(H * np.conj(H)).real
    # acorr[k] = sum_x h[x] h[x+k] (circular, but we zero-padded so it's linear)
    # The linear autocorrelation is in acorr[0:n] and acorr[N-n+1:]
    # Full autocorrelation: acorr[0:n] for k=0..n-1, acorr[N-1:n-1:-1] for k=-1..-(n-1)
    full_acorr = np.concatenate([acorr[N-n+1:N][::-1], acorr[0:n]])
    # C5 = max_k (sum(h) - acorr[k]) * dx
    total = float(np.sum(h))
    c5 = (total - full_acorr).max() * dx
    return c5

    l: length of first interval
    a: start of first interval  
    g: gap
    Returns C5 value.
    """
    n = 606
    l2 = 303 - l
    b = a + l + g
    if b + l2 > n:
        return 1.0  # infeasible
    
    dx = 2.0 / n
    min_count = 303  # at k=0, count = 303
    
    for k in range(1, n):

        c1 = max(0, l - k)
        # Self-correlation of interval 2
        c2 = max(0, l2 - k)

        c12 = max(0, hi - lo + 1) if hi >= lo else 0

Actually, let me reconsider. The problem min max_k g_k(h) where g_k is concave... The minimum of the max of concave functions over a convex set is actually a well-studied problem. It's equivalent to finding the center of the smallest ball (in the "g_k space") that contains the feasible region.

One approach: bisection on t. For a given t, check if there exists h such that g_k(h) ≤ t for all k. Since g_k is concave, g_k(h) ≤ t is a convex constraint (the sublevel set of a concave function is convex). Wait no: the superlevel set of a concave function is convex, not the sublevel set.

g_k(h) ≤ t: this is the sublevel set of a concave function, which is NOT convex in general.
g_k(h) ≥ t: this is the superlevel set of a concave function, which IS convex.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def fix_binary_sum(h, target_sum):
    h = h.copy()
    current = int(h.sum())
    if current == target_sum:
        return h
    if current > target_sum:
        excess = current - target_sum
        ones = np.where(h == 1)[0]
        h[ones[:excess]] = 0
    else:
        deficit = target_sum - current
        zeros = np.where(h == 0)[0]
        h[zeros[:deficit]] = 1
    return h

def make_two_interval(n, target, l, a, g):
    l2 = target - l
    h = np.zeros(n)
    h[a:a + l] = 1.0
    b = a + l + g
    if b + l2 <= n:
        h[b:b + l2] = 1.0
    return h

def two_interval_search(n, target, max_time):

                    return None, 1.0

def compute_deltas(h, top_lags, n):
    """Compute delta arrays for the local search.
    Returns delta_zero (n, K) and delta_one (n, K).
    delta_zero[i, m] = change in C[top_lags[m]] if h[i] goes 1->0
    delta_one[j, m] = change in C[top_lags[m]] if h[j] goes 0->1
    """
    K = len(top_lags)
    k_lags = top_lags - (n - 1)
    delta_zero = np.zeros((n, K))
    delta_one = np.zeros((n, K))
    
    for m in range(K):
        k = k_lags[m]
        # Effect of h[i]: 1->0 at lag k:
        # -(1-h[i+k]) if 0<=i+k<n
        # + h[i-k] if 0<=i-k<n
        
        # Range for i+k in [0, n): i in [max(0,-k), min(n-1, n-1-k)]
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - 1 - k) if k >= 0 else min(n - 1, n - 1 - k)
        # Wait, for k >= 0: i+k < n → i < n-k, so i <= n-k-1
        # For k < 0: i+k >= 0 → i >= -k, so i >= -k = |k|
        # Let me just use:
        # 0 <= i+k < n → -k <= i < n-k
        # Intersect with [0, n-1]:
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - 1 - k)  # n-k-1
        
        # Hmm, for k >= 0: i+k < n → i <= n-1-k. So hi1 = n-1-k.
        # For k < 0: i+k >= 0 → i >= -k. So lo1 = -k.
        # Always: lo1 = max(0, -k), hi1 = min(n-1, n-1-k)
        
        # Wait, n-1-k for k>0: e.g., n=606, k=100: hi1 = 505. i+k = i+100 <= 505+100 = 605 < 606. ✓
        # For k=-100: lo1 = 100, hi1 = min(605, 605+100) = 605. i+k = i-100, for i=100: 0. For i=605: 505. All in [0,606). ✓
        
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            vals = 1.0 - h[idx + k]
            delta_zero[idx, m] -= vals
            delta_one[idx, m] += vals
        
        # Range for i-k in [0, n-1]: k <= i <= k+n-1
        # Intersection with [0, n-1]:
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)  # = n-1 always since k+n-1 >= n-1 for k>=0, and for k<0: k+n-1 < n-1
        
        # For k >= 0: lo2 = k, hi2 = n-1
        # For k < 0: lo2 = 0, hi2 = k+n-1
        
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            vals = h[idx - k]
            delta_zero[idx, m] += vals
            delta_one[idx, m] -= vals
    
    return delta_zero, delta_one


def local_binary_search(h, target, deadline, rng, K=30):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = float(np.max(corr)) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    stall = 0
    max_stall = 3000
    it = 0
    last_recompute = 0
    
    while time.time() < deadline:
        it += 1
        
        # Recompute top K lags periodically
        if it - last_recompute >= 200:
            top_lags = np.argsort(corr)[-K:]
            last_recompute = it
        
        delta_zero, delta_one = compute_deltas(h, top_lags, n)
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        base = corr[top_lags].astype(float)
        dz = delta_zero[ones_idx]  # (n_ones, K)
        do = delta_one[zeros_idx]  # (n_zeros, K)
        
        current_max = float(np.max(base))
        
        best_improve = 1e-10
        best_ii = -1
        best_jj = -1
        
        for ii in range(len(ones_idx)):
            new_vals = base[None, :] + dz[ii][None, :] + do  # (n_zeros, K)
            new_maxes = new_vals.max(axis=1)
            candidate_best = float(new_maxes.min())
            if candidate_best < current_max - best_improve:
                best_improve = current_max - candidate_best
                best_ii = ii
                best_jj = int(np.argmin(new_maxes))
        
        if best_ii >= 0:
            i = ones_idx[best_ii]
            j = zeros_idx[best_jj]
            h[i] = 0.0
            h[j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_c5 = float(np.max(corr)) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            stall = 0
        else:
            # No improving swap found among top K lags
            # Try a random swap (SA-style)
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h[i] = 0.0
            h[j] = 1.0
            corr_new = np.correlate(h, 1.0 - h, mode='full')
            c5_new = float(np.max(corr_new)) * dx
            if c5_new < current_c5 + 0.001 * rng.exponential():  # Accept worse moves sometimes
                h[i] = 0.0  # already set
                h[j] = 1.0  # already set
                corr = corr_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    stall = 0
            else:
                # Revert
                h[i] = 1.0
                h[j] = 0.0
                # corr unchanged
            stall += 1
        
        if stall > max_stall:
            break
    
    return best_h, best_c5


def smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    return c_max + np.log(w_sum) / alpha
