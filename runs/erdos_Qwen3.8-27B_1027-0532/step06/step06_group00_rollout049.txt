import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible_fast(h_raw, target_sum):
    """Faster projection using bisection on lambda."""
    n = len(h_raw)
    # Find lambda such that sum(clip(h_raw - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_proj = np.clip(h_raw - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid  # need to decrease values more → increase lambda
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h_raw - lam, 0.0, 1.0)

def compute_smooth_gradient(h, alpha):
    """Compute gradient of the log-sum-exp smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    corr_shifted = alpha * (corr - np.max(corr))
    weights = np.exp(corr_shifted)
    weights /= np.sum(weights)
    
    # Build weight array indexed by shift k = -(n-1) to (n-1)
    # weights[k_idx] corresponds to shift k = k_idx - (n-1)
    n_corr = 2 * n - 1
    mid = n - 1
    
    # For each shift k, the gradient of corr_k w.r.t. h[i] is:
    # (1 - h[i+k]) if 0 <= i+k < n, else 0
    # minus h[i-k] if 0 <= i-k < n, else 0

    # B[i] = sum_k weights[k_idx] * h[i+k] * 1[0<=i+k<n]
    # This is correlate(h, w_arr, 'full')[mid + i] where w_arr is weights
    
    w_arr = weights  # already in the right order (index 0 = shift -(n-1), index n-1 = shift 0, etc.)
    
    B = np.correlate(h, w_arr, mode='full')[mid : mid + n]
    
    # C[i] = sum_k weights[k_idx] * h[i-k] * 1[0<=i-k<n]
    # = sum_k weights[mid + k] * h[i-k] ... 
    # Let me think: shift k means we look at h[i-k]. In w_arr, shift k is at index mid + k.
    # So C[i] = sum_j w_arr[j] * h[i - (j - mid)] = sum_j w_arr[j] * h[i + mid - j]
    # Let m = i + mid - j, so j = i + mid - m
    # C[i] = sum_m w_arr[i + mid - m] * h[m]
    # This is correlate(w_arr, h, 'full') at position... 
    # np.correlate(a, v, 'full')[p] = sum_i a[i] * v[i + len(v) - 1 - p]
    # np.correlate(w_arr, h, 'full')[p] = sum_i w_arr[i] * h[i + n - 1 - p]
    # I want sum_m w_arr[i + mid - m] * h[m] = sum_m w_arr[mid + i - m] * h[m]
    # With the formula: sum_i w_arr[i] * h[i + n - 1 - p], set i = mid + i_target - m... 
    # This is getting circular. Let me use the reversed array.
    
    w_rev = w_arr[::-1]
    # np.correlate(w_rev, h, 'full')[p] = sum_i w_rev[i] * h[i + n - 1 - p]
    # = sum_i w_arr[n_corr - 1 - i] * h[i + n - 1 - p]
    # Let j = n_corr - 1 - i, so i = n_corr - 1 - j
    # = sum_j w_arr[j] * h[n_corr - 1 - j + n - 1 - p]
    # = sum_j w_arr[j] * h[2n - 2 - j - p]
    # I want C[i] = sum_j w_arr[j] * h[i + n - 1 - j] (from the earlier derivation)
    # So I need 2n - 2 - j - p = i + n - 1 - j → 2n - 2 - p = i + n - 1 → p = n - 1 - i
    
    C = np.correlate(w_rev, h, mode='full')[n - 1 - np.arange(n)]
    # C[i] = correlate(w_rev, h, 'full')[n-1-i]
    
    # A[i] = sum_k weights[k_idx] * 1[0 <= i+k < n]
    # = sum_{k: -i <= k < n-i} weights[mid + k]
    # = sum of w_arr from index mid-i to mid+n-i-1 (clamped to [0, n_corr-1])
    
    # Use cumsum for efficient sliding window
    w_cumsum = np.zeros(n_corr + 1)
    w_cumsum[1:] = np.cumsum(w_arr)
    
    A = np.zeros(n)
    for i in range(n):
        lo_idx = max(0, mid - i)
        hi_idx = min(n_corr - 1, mid + n - 1 - i)
        A[i] = w_cumsum[hi_idx + 1] - w_cumsum[lo_idx]
    
    grad = A - B - C
    grad -= np.mean(grad)  # Center for the sum constraint
    
    return grad, corr

def smooth_projected_gradient(h_init, n, target, alpha_seq, iters_per_alpha, time_limit, 
                                eta_init=0.3, backtrack_factor=0.5, max_backtracks=20):
    start_time = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alpha_seq:
        if time.time() - start_time > time_limit:
            break
        eta = eta_init
        corr = np.correlate(h, 1.0 - h, mode='full')
        smooth_val = np.max(corr) + np.log(np.sum(np.exp(alpha * (corr - np.max(corr))))) / alpha
        
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            
            grad, corr = compute_smooth_gradient(h, alpha)
            
            # Backtracking line search
            smooth_old = np.max(corr) + np.log(np.sum(np.exp(alpha * (corr - np.max(corr))))) / alpha
            
            improved = False
            for bt in range(max_backtracks):
                h_try = project_feasible_fast(h - eta * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                smooth_try = np.max(corr_try) + np.log(np.sum(np.exp(alpha * (corr_try - np.max(corr_try))))) / alpha
                
                if smooth_try < smooth_old - 1e-16:
                    h = h_try
                    smooth_old = smooth_try
                    improved = True
                    break
                eta *= backtrack_factor
            
            if not improved:
                eta *= backtrack_factor
                if eta < 1e-10:
                    break
            
            # Track best
            c5_val = float(np.max(corr_try)) * (2.0 / n)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
    
    return best_h, best_c5
