            if s == 0:
                grad += w * one_minus_h
                grad -= w * h
            else:
                grad[s:] += w * one_minus_h[:n - s]
                grad[:n - s] -= w * h[s:]
        else:
            p = -s
            grad[:n - p] += w * one_minus_h[p:]
            grad[p:] -= w * h[:n - p]
    return value, grad

def adam_optimize(h0, target, start_time, time_limit, alpha_schedule, lr_init, beta1, beta2, eps, clip_norm):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_score = compute_c5(h)
    max_alpha_time = max(30.0, time_limit / max(len(alpha_schedule), 1) * 0.8)

    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        alpha_start = time.time()
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')

        while True:
            if time.time() - start_time > time_limit:
                break
            if time.time() - alpha_start > max_alpha_time:
                break
            if stall > 400:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            gnorm = float(np.linalg.norm(grad))
            if gnorm > clip_norm:
                grad = grad * (clip_norm / gnorm)
            if val < local_best - 1e-11:
                local_best = val
                stall = 0
            else:
                stall += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr_init / (1.0 + 0.0005 * t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target)

        actual = compute_c5(h)
        if actual < best_score:
            best_score = actual
            best_h = h.copy()

    return best_h, best_score

def binary_sa(h0, target, start_time, time_limit, max_iters, t_init, t_final):
    n = len(h0)
    target_int = int(target)
    h = (h0 >= 0.5).astype(float)
    cnt = int(h.sum())
    if cnt > target_int:
        excess = cnt - target_int
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(h0[ones])[:excess]]
        h[order] = 0.0
    elif cnt < target_int:
        deficit = target_int - cnt
        zeros = np.where(h == 0)[0]
        order = zeros[np.argsort(-h0[zeros])[:deficit]]
        h[order] = 1.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    cur_c5 = best_c5
    stall = 0
    rng = np.random.RandomState(12345)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 4000:
            break
        ratio = float(it) / float(max(max_iters - 1, 1))
        temp = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new = h.copy()
        r = rng.random()
        if r < 0.6:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
        elif r < 0.85:
            worst = np.argsort(corr)[-10:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [x for x in ones if 0 <= x + k < n]
            jp = [x for x in zeros if 0 <= x + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0.0
                h_new[j] = 1.0
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0.0
                h_new[j] = 1.0

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - cur_c5
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / max(temp, 1e-30))):
            h = h_new
            corr = corr_new
            cur_c5 = c5_new
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def structured_init(n, freq, amplitude):
    x = np.arange(n) / float(n) * 2.0
    h = 0.5 + amplitude * np.cos(2.0 * np.pi * freq * x / 2.0)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = float(n0) / 2.0

    best_h = project_box_sum(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick Adam on original resolution
    if tleft() > 250:
        h1, s1 = adam_optimize(
            best_h, target0, start_time, min(120, tleft()),
            alpha_schedule=[50, 150, 500, 2000, 8000, 30000, 100000],
            lr_init=0.004, beta1=0.9, beta2=0.999, eps=1e-8,
            clip_norm=5.0
        )
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Upsample to 2x resolution and optimize
    n2 = 2 * n0
    if tleft() > 350:
        h2_init = upsample(best_h, n2)
        h2_init = project_box_sum(h2_init, float(n2) / 2.0)
        h2, s2 = adam_optimize(
            h2_init, float(n2) / 2.0, start_time, min(250, tleft()),
            alpha_schedule=[50, 200, 800, 4000, 20000, 80000, 300000],
            lr_init=0.003, beta1=0.9, beta2=0.999, eps=1e-8,
            clip_norm=5.0
        )
        if s2 < best_score:
            best_score = s2
            best_h = h2
            best_n = n2
            best_target = float(n2) / 2.0

    # Phase 3: Further optimization on current best with high alpha
    if tleft() > 250:
        h3, s3 = adam_optimize(
            best_h, best_target, start_time, min(180, tleft()),
            alpha_schedule=[2000, 10000, 50000, 200000, 800000],
            lr_init=0.002, beta1=0.9, beta2=0.999, eps=1e-8,
            clip_norm=3.0
        )
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Multi-restart with structured initializations
    if tleft() > 200:
        n_work = best_n
        target_work = best_target
        restart_time = min(60, (tleft() - 20) / 3.0)
        for ri in range(3):
            if tleft() < 50:
                break
            if ri == 0:
                h_pert = best_h + rng.randn(n_work) * 0.02
            elif ri == 1:
                h_pert = structured_init(n_work, 3, 0.3)
                h_pert = project_box_sum(h_pert, target_work)
            else:
                h_pert = structured_init(n_work, 5, 0.25)
                h_pert = project_box_sum(h_pert, target_work)
            h_pert = project_box_sum(h_pert, target_work)
            h_r, s_r = adam_optimize(
                h_pert, target_work, start_time, restart_time,
                alpha_schedule=[100, 500, 3000, 15000, 80000],
                lr_init=0.003, beta1=0.9, beta2=0.999, eps=1e-8,
                clip_norm=5.0
            )
            if s_r < best_score:
                best_score = s_r
                best_h = h_r

    # Phase 5: Binary SA on best solution
    if tleft() > 120:
        sa_time = min(150, tleft() - 20)
        max_iter = int(max(5000, sa_time * 800))
        h5, s5 = binary_sa(best_h, best_target, start_time, sa_time, max_iter, 0.005, 0.00001)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Final continuous polish
    if tleft() > 40:
        h6, s6 = adam_optimize(
            best_h, best_target, start_time, max(20, tleft()),
            alpha_schedule=[5000, 30000, 150000, 600000],
            lr_init=0.001, beta1=0.9, beta2=0.999, eps=1e-8,
            clip_norm=2.0
        )
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Final feasibility check
    h_final = project_box_sum(best_h, best_target)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-12:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
