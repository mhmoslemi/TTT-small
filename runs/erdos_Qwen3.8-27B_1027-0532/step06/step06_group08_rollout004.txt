def project_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff < 0:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_reduce = h[idx]
            if can_reduce > remaining:
                h[idx] -= remaining
                remaining = 0
            else:
                remaining -= can_reduce
                h[idx] = 0.0
    else:
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            can_inc = 1.0 - h[idx]
            if can_inc > remaining:
                h[idx] += remaining
                remaining = 0
            else:
                remaining -= can_inc
                h[idx] = 1.0
    return h

def project_binary(h, target_sum):

    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:

        if need <= len(zeros):

            h_r[zeros[order]] = 1

                A[valid] += h[idx[valid]]
                
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                
                # Pick top 2 i's (highest A) and top 2 j's (lowest B... wait)
                # We want to reduce correlation at k_star
                # A[i] high means h[i] has many neighbors at distance k_star that are 1
                # B[j] low means h[j] has few neighbors at distance k_star that are 1
                # We want to remove 1s that contribute to correlation and add 0s that don't
                # So pick i with high A (removing them reduces correlation)
                # and j with low B (adding them adds less correlation)
                top_i = ones[np.argsort(-A[ones])[:min(8, len(ones))]]
                top_j = zeros[np.argsort(B[zeros])[:min(8, len(zeros))]]
                
                # Pick 2 from each
                i1 = top_i[rng.randint(len(top_i))]
                i2_pool = np.delete(top_i, np.where(top_i == i1)[0][0])
                i2 = i2_pool[rng.randint(len(i2_pool))]
                j1 = top_j[rng.randint(len(top_j))]
                j2_pool = np.delete(top_j, np.where(top_j == j1)[0][0])
                j2 = j2_pool[rng.randint(len(j2_pool))]
            else:
                # Random selection
                idx_i = rng.choice(len(ones), 2, replace=False)
                i1, i2 = ones[idx_i[0]], ones[idx_i[1]]
                idx_j = rng.choice(len(zeros), 2, replace=False)
                j1, j2 = zeros[idx_j[0]], zeros[idx_j[1]]
            
            if i1 == i2 or j1 == j2:
                continue
            
            delta = compute_delta_2swap(h, i1, i2, j1, j2)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # 1-swap move
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            
            if k_star != 0 and rng.random() < 0.7:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                
                top_i = ones[np.argsort(-A[ones])[:min(16, len(ones))]]
                top_j = zeros[np.argsort(B[zeros])[:min(16, len(zeros))]]
                
                i = top_i[rng.randint(len(top_i))]
                j = top_j[rng.randint(len(top_j))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            
            delta = compute_delta_1swap(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def greedy_improve(h, max_iters, rng, time_limit, start_time):
    """Greedy: always take the best improving 1-swap or 2-swap move."""
    n = len(h)
    h = h.copy().astype(float)
    full = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        
        max_val = float(full.max())
        top_idx = np.argsort(full)[-5:]
        
        found = False
        for k_idx in top_idx:
            k_star = int(k_idx) - (n - 1)
            if k_star == 0:
                continue
            
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            
            # Try top candidates
            top_i = ones[np.argsort(-A[ones])[:min(20, len(ones))]]
            top_j = zeros[np.argsort(B[zeros])[:min(20, len(zeros))]]
            
            best_delta_max = max_val
            best_i, best_j = -1, -1
            
            for i in top_i:
                for j in top_j:
                    delta = compute_delta_1swap(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < best_delta_max:
                        best_delta_max = new_max
                        best_i, best_j = int(i), int(j)
                        # Also try 2-swap
                    if new_max < max_val:
                        # Found improvement, accept
                        pass
            
            if best_i >= 0 and best_delta_max < max_val:
                delta = compute_delta_1swap(h, best_i, best_j)
                h[best_i] = 0.0
                h[best_j] = 1.0
                full += delta
                stall = 0
                new_c5 = float(full.max()) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                found = True
            else:
                # Try 2-swap
                if len(ones) >= 2 and len(zeros) >= 2:
                    top_i2 = ones[np.argsort(-A[ones])[:min(6, len(ones))]]
                    top_j2 = zeros[np.argsort(B[zeros])[:min(6, len(zeros))]]
                    
                    best2_max = max_val
                    best2_i1, best2_i2, best2_j1, best2_j2 = -1, -1, -1, -1
                    
                    for i1 in top_i2:
                        for i2 in top_i2:
                            if i1 == i2:
                                continue
                            for j1 in top_j2:
                                for j2 in top_j2:
                                    if j1 == j2 or j1 == i1 or j1 == i2 or j2 == i1 or j2 == i2:
                                        continue
                                    delta = compute_delta_2swap(h, int(i1), int(i2), int(j1), int(j2))
                                    new_max = float((full + delta).max())
                                    if new_max < best2_max:
                                        best2_max = new_max
                                        best2_i1, best2_i2 = int(i1), int(i2)
                                        best2_j1, best2_j2 = int(j1), int(j2)
                    
                    if best2_i1 >= 0 and best2_max < max_val:
                        delta = compute_delta_2swap(h, best2_i1, best2_i2, best2_j1, best2_j2)
                        h[best2_i1] = 0.0
                        h[best2_i2] = 0.0
                        h[best2_j1] = 1.0
                        h[best2_j2] = 1.0
                        full += delta
                        stall = 0
