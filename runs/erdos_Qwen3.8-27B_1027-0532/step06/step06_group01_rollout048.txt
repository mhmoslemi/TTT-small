import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_autocorr(h):
    """Compute R(k) = sum_i h[i]*h[i+k] for all lags k"""
    return np.correlate(h, h, mode='full')

def autocorr_grad(h, k):
    """Gradient of R(k) with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    # R(k) = sum_i h[i]*h[i+k]
    # dR/dh[j] = h[j+k] (if valid) + h[j-k] (if valid)
    for j in range(n):
        if 0 <= j + k < n:
            grad[j] += h[j + k]
        if 0 <= j - k < n:
            grad[j] += h[j - k]
    return grad

def slp_iteration(h0, n, solver='GLOP'):
    """One SLP iteration: solve LP with linearized constraints"""
    dx = 2.0 / n
    target = n / 2.0
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Current autocorrelation
    R0 = np.correlate(h0, h0, mode='full')
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    
    # Linearized constraints: R0[k] + grad_k^T (h - h0) >= 1 - t
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        if k == 0:
            # R(0) = sum h[i]^2, grad = 2*h
            grad = 2 * h0
        else:
            grad = np.zeros(n)
            for j in range(n):
                if 0 <= j + k < n:
                    grad[j] += h0[j + k]
                if 0 <= j - k < n:
                    grad[j] += h0[j - k]
        
        # R0[k_idx] + grad^T (h - h0) >= 1 - t
        # R0[k_idx] + grad^T h - grad^T h0 >= 1 - t
        # grad^T h + t >= 1 - R0[k_idx] + grad^T h0
        lhs = grad @ h + t
        rhs = 1.0 - R0[k_idx] + grad @ h0
        constraints.append(lhs >= rhs)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    result = prob.solve(solver=solver)
    
    if h.value is not None:
        h_new = np.clip(h.value, 0, 1)
        # Project to satisfy sum constraint
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Simple projection
            if diff > 0:
                idx = np.where(h_new < 1 - 1e-10)[0]
            else:
                idx = np.where(h_new > 1e-10)[0]
            if len(idx) > 0:
                h_new[idx] += diff / len(idx)
                h_new = np.clip(h_new, 0, 1)
        return h_new, result
    return h0, None

We want to maximize the decrease in max_k C(k), i.e., find δ > 0 such that max_k (C(k) + δ·g(k)) is minimized.

import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grads_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for k_idx in range(n_corr):
        k = abs(k_idx - (n - 1))
        if k == 0:
            grads[k_idx] = 2.0 * h
        else:
            g = np.zeros(n)
            g[:n - k] += h[k:]
            g[k:] += h[:n - k]
            grads[k_idx] = g
    return grads

def slp_single_step(h0, n, solver):
    target = n / 2.0
    R0 = np.correlate(h0, h0, mode='full')
    grads = compute_grads_matrix(h0)
    n_corr = 2 * n - 1
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    for k_idx in range(n_corr):
        constraints.append(grads[k_idx] @ h_var - t_var >= R0[k_idx])
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return h0, None
    if h_var.value is None:
        return h0, None
    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    t_val = float(t_var.value) if t_var.value is not None else None
    return h_new, t_val

def pair_cd(h_init, n, target, time_limit, n_cand=20, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def smoothed_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(lambda x: smoothed_obj(x, alpha, n), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def randomized_restart(h0, n, target, n_restarts=3, perturbation_scale=0.05):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for r in range(n_restarts):
        noise = np.random.randn(n) * perturbation_scale
        h_pert = h0 + noise
        h_pert = project_feasible(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    return best_h, best_c5

    if tleft() > 200:
        slp_time = min(180, tleft() * 0.3)
        hp, sc = slp_loop(best_h, n, time_limit=slp_time, solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Pair CD polish
    if tleft() > 150:
        cd_time = min(120, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=25)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: SLP again
    if tleft() > 150:
        slp_time = min(130, tleft() * 0.3)
        hp, sc = slp_loop(best_h, n, time_limit=slp_time, solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: SLSQP polish
    if tleft() > 80:
        hp, sc = slsqp_polish(best_h, n, target, [1000, 5000, 20000], 1500,
                              time_limit=min(60, tleft() * 0.3))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Randomized restart + SLP
    if tleft() > 200:
        hr, scr = randomized_restart(best_h, n, target, n_restarts=5,
                                     perturbation_scale=0.03)
        if scr < best_score:
            best_score = scr
            best_h = hr
        slp_time = min(100, tleft() * 0.3)
        hp, sc = slp_loop(best_h, n, time_limit=slp_time, solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(60, tleft() * 0.2), n_cand=20)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: Higher resolution n=1200
    if tleft() > 400:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slp_loop(h_interp, n_new, time_limit=min(150, tleft() * 0.3), solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        if tleft() > 150:
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.3), n_cand=20)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
        if tleft() > 80:
            hp, sc = slsqp_polish(best_h, n, target, [2000, 10000], 1000,
                                  time_limit=min(50, tleft() * 0.3))
            if sc < best_score:
                best_score = sc
                best_h = hp

    # Phase 7: More SLP at current resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        slp_time = min(80, tleft() * 0.4)
        hp, sc = slp_loop(best_h, n_use, time_limit=slp_time, solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Pair CD final
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(50, tleft() * 0.5), n_cand=30)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Final SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_polish(best_h, n_use, target_use, [5000, 20000, 100000], 800,
                              time_limit=min(30, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 10: Final SLP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slp_loop(best_h, n_use, time_limit=min(20, tleft() * 0.5), solver='GLOP')
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
