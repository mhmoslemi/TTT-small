def active_set_gradient(h, n, target, n_steps=1000, time_limit=300):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    mid = n - 1
    idxs = np.arange(n)
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr) * dx
        if c_max < best_c5 - 1e-12:
            best_c5 = c_max
            best_h = h.copy()
        
        # Active set: shifts where corr is close to max
        active = np.where(corr > c_max / dx - 0.01 / dx)[0]
        
        if len(active) == 0:
            active = np.where(corr == c_max / dx)[0]
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        
        grad /= len(active)
        grad -= np.mean(grad)  # maintain sum constraint
        
        # Line search
        best_lr = 0.01
        best_penalty = c_max
        for lr in [0.1, 0.05, 0.01, 0.005, 0.001]:
            h_try = project_feasible(h - lr * grad, target)
            c_try = compute_c5(h_try)
            if c_try < best_penalty:
                best_penalty = c_try
                best_lr = lr
        
        h = project_feasible(h - best_lr * grad, target)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def fft_correlate(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = 1.0 - h
    fa = np.fft.rfft(a)
    fb = np.fft.rfft(b)
    corr = np.fft.irfft(fa * np.conj(fb))[:2 * n - 1]
    return corr


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:

        if len(idx2) > 0:

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > stall_threshold:
            if n_perturbs < 20:
                n_perturbs += 1
                rng_pert = np.random.RandomState(sweep * 7 + 13)
                n_pert = rng_pert.randint(3, 15)
                pert_idx = rng_pert.choice(n, size=n_pert, replace=False)
                h[pert_idx] += rng_pert.randn(n_pert) * 0.05
                h = project_feasible(h, target)
                consecutive_no_improve = 0
                continue
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_high = len(high_set)
        n_low = len(low_set)
        step_h = max(1, n_high // 20)
        step_l = max(1, n_low // 20)

        for ii in range(0, n_high, step_h):
            i = high_set[ii]
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for jj in range(0, n_low, step_l):
                j = low_set[jj]
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > stall_threshold // 2:
                break

    return best_h, best_c5


def active_set_gradient(h, n, target, time_limit, n_steps=20000, n_active=5):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    lr = 0.02

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if (step + 1) % 500 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-13:
                best_c5 = c5_now
                best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)

        active = np.argsort(corr)[-n_active:]
        w = np.zeros(len(active))
        for idx, s in enumerate(active):
            w[idx] = max(corr[s] - (c_max - 2.0), 0.0) + 1.0
        w = w / np.sum(w)

        grad = np.zeros(n)
        for idx, s in enumerate(active):
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[idx] * h[idxs[v2] + k]

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        improved = False
        for lr_mult in [2.0, 1.0, 0.5, 0.25, 0.1]:
            lr_try = lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14 or c5_try < compute_c5(h):
                h = h_try
                lr = lr_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()

    return best_h, best_c5


def multi_coord_cd(h, n, target, time_limit, max_sweeps=2000, n_subset=4, n_bisect=12):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    rng = np.random.RandomState(12345)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_subset * 3]
        high_set = sorted_idx[-n_subset * 3:]

        n_tries = 10
        best_trial_c5 = c5_val
        best_move = None

        for _ in range(n_tries):
            n_h = rng.randint(1, n_subset + 1)
            n_l = rng.randint(1, n_subset + 1)
            h_exists = list(set(high_set) - set(low_set))
            if len(h_exists) < n_h:
                continue
            h_idx = rng.choice(h_exists, size=n_h, replace=False)
            l_idx = rng.choice(low_set, size=n_l, replace=False)

            max_delta = min(min(h[h_idx]), min(1.0 - h[l_idx]))
            if max_delta < 1e-12:
                continue

            g = np.zeros(n_corr)
            for i in h_idx:
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_ip] -= (1.0 - h[i + shifts[v_ip]])
                g[v_im] += h[i - shifts[v_im]]
            for j in l_idx:
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

            a_d = 0.0
            b_d = max_delta
            for _b in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-14:
                best_trial_c5 = approx_max
                best_move = (h_idx, l_idx, delta_star)

        if best_move is not None:
            h_idx, l_idx, delta = best_move
            for i in h_idx:
                h[i] -= delta
            for j in l_idx:
                h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [300, 1500, 8000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Long pair CD (main workhorse)
    if tleft() > 300:
        cd_time = min(350, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_improved(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Active-set gradient descent
    if tleft() > 120:
        as_time = min(150, tleft() * 0.15)
        h_as, c5_as = active_set_gradient(best_h, n, target,
                                          time_limit=as_time,
                                          n_steps=50000, n_active=5)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Multi-coordinate CD
    if tleft() > 150:
        mc_time = min(180, tleft() * 0.18)
        h_mc, c5_mc = multi_coord_cd(best_h, n, target,
                                     time_limit=mc_time,
                                     max_sweeps=3000, n_subset=4, n_bisect=13)
        if c5_mc < best_score:
            best_score = c5_mc
            best_h = h_mc

    # Phase 6: Another pair CD round with different params
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.2)
        h_cd, c5_cd = pair_cd_improved(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=6000, n_cand=70, n_bisect=16,
                                       stall_threshold=80)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Active-set gradient again
    if tleft() > 80:
        as_time = min(100, tleft() * 0.12)
        h_as, c5_as = active_set_gradient(best_h, n, target,
                                          time_limit=as_time,
                                          n_steps=30000, n_active=3)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 8: Multi-restart with small perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.02, 0.05, 0.1, 0.2]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        cd_t = min(60, tleft() * 0.15)
        h_try, c5_try = pair_cd_improved(h_pert, n, target,
                                         time_limit=cd_t,
                                         max_sweeps=3000, n_cand=45, n_bisect=13,
                                         stall_threshold=50)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 9: Try higher resolution
    if tleft() > 150:
        for n_try in [700, 800]:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [300, 1000, 5000], 600, 1e-14)
            sc = compute_c5(h_hires)
            if tleft() > 60 and sc < best_score + 0.001:
                cd_t = min(60, tleft() * 0.3)
                h_hires2, sc2 = pair_cd_improved(h_hires, n_try, target_try,
                                                 time_limit=cd_t,
                                                 max_sweeps=2000, n_cand=40, n_bisect=13)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 10: Final active-set gradient
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        as_time = min(40, tleft() * 0.4)
        h_as, c5_as = active_set_gradient(best_h, n_use, target_use,
                                          time_limit=as_time,
                                          n_steps=20000, n_active=4)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000, 150000], 600, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 12: Final pair CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(20, tleft() * 0.5)
        h_cd, c5_cd = pair_cd_improved(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=2000, n_cand=50, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
