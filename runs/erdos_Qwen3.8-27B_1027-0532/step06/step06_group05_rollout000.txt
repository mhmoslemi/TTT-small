    return float(np.sum(h)) - _slsqp_target

def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)

def slsqp_optimize(h_start, n, target, alpha, maxiter, time_limit, start_time):
    global _slsqp_n, _slsqp_target, _slsqp_alpha
    _slsqp_n = n
    _slsqp_target = target
    _slsqp_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    try:
        res = minimize(_slsqp_objective, h_start, jac=_slsqp_gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and not np.any(np.isnan(res.x)):
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return h_start

def adam_optimize(h0, target_sum, start_time, time_budget, alphas, lr, beta1, beta2, eps, topk_list=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    for idx, alpha in enumerate(alphas):
        if time.time() - start_time > time_budget:
            break
        m_arr = np.zeros(n)
        v_arr = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        tk = None
        if topk_list is not None:
            tk = topk_list[min(idx, len(topk_list) - 1)]
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha, topk=tk)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break
            m_arr = beta1 * m_arr + (1.0 - beta1) * grad
            v_arr = beta2 * v_arr + (1.0 - beta2) * (grad * grad)
            m_hat = m_arr / (1.0 - beta1 ** t)
            v_hat = v_arr / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
    return best_h, best_actual

def targeted_binary_search(h, target_sum, time_limit, start_time, max_iters=50000):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 3000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        top_indices = np.argsort(corr)[-10:]
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for t_idx in top_indices:
            k_star = int(t_idx) - (n - 1)
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i_cand = np.array([i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5])
            j_cand = np.array([j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5])
            if len(i_cand) == 0 or len(j_cand) == 0:
                i_cand = np.array([i for i in ones if 0 <= i + k_star < n])
                j_cand = np.array([j for j in zeros if 0 <= j + k_star < n])
                if len(i_cand) == 0 or len(j_cand) == 0:
                    continue
            n_i = min(len(i_cand), 30)
            n_j = min(len(j_cand), 30)
            i_sample = i_cand[:n_i]
            j_sample = j_cand[:n_j]
            for ii in range(len(i_sample)):
                i = i_sample[ii]
                for jj in range(len(j_sample)):
                    j = j_sample[jj]
                    nm = max_val
                    for t2 in top_indices:
                        k = int(t2) - (n - 1)
                        d = 0.0
                        if 0 <= i + k < n:
                            d -= (1.0 - h[i + k])
                        if k < 0 and 0 <= i + k < n:
                            d += h[i + k]
                        if 0 <= j + k < n:
                            d += (1.0 - h[j + k])
                        if k < 0 and 0 <= j + k < n:
                            d -= h[j + k]
                        v = corr[int(t2)] + d
                        if v > nm:
                            nm = v
                            break
                    if nm < max_val - best_improve:
                        best_improve = max_val - nm
                        best_i = i
                        best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                h[best_i] = 1.0
                h[best_j] = 0.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 5000
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new = h.copy()
        r = rng.random()
        if r < 0.4:
            worst = np.argsort(corr)[-5:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = np.array([i for i in ones if 0 <= i + k < n])
            jp = np.array([j for j in zeros if 0 <= j + k < n])
            if len(ip) == 0 or len(jp) == 0:
                continue
            h_new[ip[rng.randint(len(ip))]] = 0.0
            h_new[jp[rng.randint(len(jp))]] = 1.0
        elif r < 0.8:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
            else:
                continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target, _slsqp_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha (main optimization)
    if tleft() > 300:
        alphas_sl = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
        for a in alphas_sl:
            if tleft() < 80:
                break
            h_new = slsqp_optimize(best_h, n, target, a, maxiter=2000,
                                   time_limit=60, start_time=start_time)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Targeted binary search
    if tleft() > 200:
        h_bin = project_binary(best_h, int(target))
        tb_time = min(120, tleft() * 0.25)
        h_tb, c5_tb = targeted_binary_search(h_bin, int(target),
                                             time_limit=tb_time,
                                             start_time=start_time,
                                             max_iters=80000)
        if c5_tb < best_score:
            best_score = c5_tb
            best_h = h_tb.copy()

    # Phase 3: SLSQP from binary solution
    if tleft() > 150:
        h_bin2 = project_binary(best_h, int(target))
        alphas_sl2 = [500, 2000, 5000, 15000, 40000]
        for a in alphas_sl2:
            if tleft() < 50:
                break
            h_new = slsqp_optimize(h_bin2, n, target, a, maxiter=1500,
                                   time_limit=40, start_time=start_time)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Adam with top-k focus
    if tleft() > 200:
        adam_time = min(300, tleft() - 150)
        h_a, s_a = adam_optimize(best_h, target, start_time, adam_time,
                                  alphas=[200, 500, 1000, 3000, 8000, 20000, 50000],
                                  lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8,
                                  topk_list=[50, 30, 20, 10, 5, 3, 2])
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 5: Multiple restarts with diverse perturbations
    if tleft() > 200:
        n_restarts = 6
        for ri in range(n_restarts):
            if tleft() < 80:
                break
            if ri % 3 == 0:
                h_pert = best_h + rng.randn(n) * (0.01 + 0.005 * ri)
            elif ri % 3 == 1:
                h_pert = best_h.copy()
                flip_count = n // 20
                idx = rng.choice(n, flip_count, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]
            else:
                block_size = 10
                h_pert = best_h.copy()
                for _ in range(5):
                    start = rng.randint(0, n - block_size)
                    h_pert[start:start+block_size] = 1.0 - h_pert[start:start+block_size]
            h_pert = project_box_sum(h_pert, target)
            rt = min(80, (tleft() - 60) / max(n_restarts - ri, 1))
            if rt < 10:
                break
            h_r, s_r = adam_optimize(h_pert, target, start_time, rt,
                                      alphas=[200, 1000, 5000, 20000],
                                      lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8,
                                      topk_list=[30, 15, 8, 4])
            if s_r < best_score:
                best_score = s_r
                best_h = h_r.copy()

    # Phase 6: Binary SA polish
    if tleft() > 100:
        h_bin3 = project_binary(best_h, int(target))
        sa_time = min(120, tleft() - 40)
        max_it = int(min(150000, sa_time * 800))
        h_sa, c5_sa = sa_binary(h_bin3, int(target), max_it,
                                0.005, 0.000005, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 7: Final SLSQP polish
    if tleft() > 40:
        h_new = slsqp_optimize(best_h, n, target, 100000, maxiter=3000,
                               time_limit=max(10, tleft() - 20), start_time=start_time)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final targeted binary search
    if tleft() > 30:
        h_bin4 = project_binary(best_h, int(target))
        h_tb2, c5_tb2 = targeted_binary_search(h_bin4, int(target),
                                               time_limit=max(5, tleft() - 15),
                                               start_time=start_time,
                                               max_iters=30000)
        if c5_tb2 < best_score:
            best_score = c5_tb2
            best_h = h_tb2.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score
    if not np.all(np.isfinite(h_final)):
        h_final = project_box_sum(h_init, target)
        final_score = compute_c5(h_final)

    return h_final, final_score, n
