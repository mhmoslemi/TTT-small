def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for any residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(x, alpha):
    corr = get_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    logsum = float(np.log(np.sum(np.exp(shifted))))
    n = len(x)
    dx = 2.0 / n
    return (c_max + logsum / alpha) * dx

def smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = get_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    # Subtract mean to maintain sum constraint (project gradient onto feasible manifold)
    grad -= np.mean(grad)
    return grad

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_con_jac(x):
    return np.ones(len(x))

    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h, args=(alpha,),
                          jac=lambda x, a=alpha: smooth_grad(x, a),
                          method='SLSQP', bounds=bounds, cons=cons,
                          options={'maxiter': maxiter_each, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_score:
                best_score = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_score

where k = s - (n-1).

minimize t
subject to: corr[s] + grad_s · (h' - h) ≤ t for s in active_set
            0 ≤ h' ≤ 1
            sum(h') = n/2

This gives a lower bound on the achievable C₅ in the linearization. If this lower bound is close to the current C₅, we're near a local optimum.

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

        idx = np.where(h < 1.0 - 1e-12)[0]
    else:

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(x, alpha):
