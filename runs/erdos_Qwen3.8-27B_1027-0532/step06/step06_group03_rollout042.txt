                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = d_lo
                b_d = d_hi
                for _ in range(25):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    am = vals >= f_m - 1e-14
                    if am.any():
                        fp = np.max(g[am])
                    else:
                        fp = 0.0
                    if fp > 1e-14:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                approx_max = np.max(corr + d_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = d_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def momentum_steps(h_start, n, target, alpha, time_limit, max_steps, mu, base_lr, top_k):
    start_t = time.time()
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        grad = smooth_grad(h, alpha, top_k)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        v = mu * v + grad / gnorm
        lr = base_lr / (1.0 + 0.05 * step)
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * mult * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14 or c5_try < compute_c5(h) - 1e-14:
                h = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: FISTA with low alpha for basin finding
    if tleft() > 100:
        h_f, c5_f, L_est = fista_optimize(best_h, n, target, alpha=20,
                                          max_iter=20000, top_k=10,
                                          time_limit=min(40, tleft() * 0.3))
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 2: FISTA with moderate alpha
    if tleft() > 100:
        h_f, c5_f, L_est = fista_optimize(best_h, n, target, alpha=100,
                                          max_iter=20000, top_k=8,
                                          time_limit=min(50, tleft() * 0.3), l0=L_est)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 3: FISTA with high alpha
    if tleft() > 80:
        h_f, c5_f, L_est = fista_optimize(best_h, n, target, alpha=500,
                                          max_iter=15000, top_k=6,
                                          time_limit=min(40, tleft() * 0.3), l0=L_est)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_pol = slsqp_polish(best_h, n, target, [200, 1000, 5000],
                             2000, 1e-16, min(60, tleft() * 0.4), top_k=6)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Subgradient steps with exact objective
    if tleft() > 60:
        h_sg, c5_sg = subgradient_steps(best_h, n, target,
                                        max_iter=5000, time_limit=min(30, tleft() * 0.25),
                                        step_scale=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Momentum with high alpha
    if tleft() > 80:
        for a_m in [500, 2000]:
            if tleft() < 40:
                break
            h_pm, c5_pm = momentum_steps(best_h, n, target, alpha=a_m,
                                         time_limit=min(30, tleft() * 0.25),
                                         max_steps=20000, mu=0.9, base_lr=0.02, top_k=5)
            if c5_pm < best_score:
                best_score = c5_pm
                best_h = h_pm

    # Phase 7: SLSQP with very high alpha
    if tleft() > 80:
        h_pol = slsqp_polish(best_h, n, target, [5000, 20000, 100000],
                             2000, 1e-16, min(50, tleft() * 0.35), top_k=4)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Pair coordinate descent
    if tleft() > 60:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=min(40, tleft() * 0.3),
                              max_sweeps=20000, n_cand=60, top_k=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        for trial in range(5):
            if tleft() < 40:
                break
            noise_scale = 0.02 + 0.03 * trial
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_f, c5_f, _ = fista_optimize(h_pert, n, target, alpha=100,
                                          max_iter=10000, top_k=6,
                                          time_limit=min(20, tleft() * 0.2))
            if tleft() > 20:
                h_pol = slsqp_polish(h_f, n, target, [1000, 5000],
                                     1000, 1e-15, min(15, tleft() * 0.2), top_k=5)
                h_f = h_pol
            sc = compute_c5(h_f)
            if sc < best_score:
                best_score = sc
                best_h = h_f

    # Phase 10: Final FISTA + subgradient
    if tleft() > 40:
        h_f, c5_f, _ = fista_optimize(best_h, n, target, alpha=2000,
                                      max_iter=10000, top_k=4,
                                      time_limit=min(15, tleft() * 0.4))
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f
        if tleft() > 15:
            h_sg, c5_sg = subgradient_steps(best_h, n, target,
                                            max_iter=3000, time_limit=min(10, tleft() * 0.4),
                                            step_scale=0.002)
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg

    # Phase 11: Final pair CD
    if tleft() > 25:
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=min(15, tleft() * 0.5),
                              max_sweeps=10000, n_cand=70, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Try higher resolution
    if tleft() > 80:
        for n_try in [600, 800]:
            if tleft() < 50:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_f, c5_f, _ = fista_optimize(h_interp, n_try, target_try, alpha=50,
                                          max_iter=10000, top_k=10,
                                          time_limit=min(25, tleft() * 0.3))
            if tleft() > 20:
                h_pol = slsqp_polish(h_f, n_try, target_try, [200, 1000],
                                     1000, 1e-15, min(15, tleft() * 0.2), top_k=6)
                h_f = h_pol
            sc = compute_c5(h_f)
            if sc < best_score:
                best_score = sc
                best_h = h_f
                n = n_try
                target = target_try

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h.copy()
        final_score = best_score
    return h_final, final_score, len(h_final)
