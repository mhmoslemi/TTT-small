def project_box_sum(y, target_sum):

where c_s(m) = (1-h[m+s])*[0≤m+s<n] - h[m-s]*[0≤m-s<n]

So the new correlation is: corr_new(s) = corr(s) + (x_new - x_old) * c_s(m)

And we want to choose x_new ∈ [0,1] to minimize max_s corr_new(s), subject to the sum constraint.

If we ignore the sum constraint for a moment:
We want to minimize max_s [corr(s) + (x_new - x_old) * c_s(m)]

This is a 1D minimax problem! The optimal x_new is where the top correlations "cross".

Specifically, let's define f(x) = max_s [corr(s) + (x - x_old) * c_s(m)]
We want to minimize f(x) over x ∈ [0,1].

f(x) is the max of linear functions of x, so it's convex in x. We can find the minimum by:
- If all c_s(m) > 0: f is increasing, minimum at x=0
- If all c_s(m) < 0: f is decreasing, minimum at x=1
- Otherwise: the minimum is where the "active" lines cross

More precisely, let's define d_s = c_s(m) for each s. Then:
f(x) = max_s [corr(s) + (x-x_old) * d_s]

Let a_s = corr(s), b_s = d_s. Then:
f(x) = max_s [a_s + b_s * (x - x_old)]

The minimum of this convex piecewise-linear function occurs at a "kink" point, which is where two lines intersect and are both on the upper envelope.

This can be solved efficiently. But for practical purposes, we can just do a binary search or use the fact that the derivative of f is:
f'(x) = b_{s*(x)} where s* is the active shift (the one achieving the max)

At the optimum, f'(x) changes sign, meaning the active shift with positive b_s equals the active shift with negative b_s.

For simplicity, I'll use a simpler approach: project the unconstrained optimum.

The unconstrained optimum (ignoring x ∈ [0,1]) satisfies:
The maximum of {a_s + b_s * t} over s is minimized where t makes the top values equal.

Let t* be the optimal (x - x_old). Then max_s [a_s + b_s * t*] is minimized.

A simple approach: 
- Find the current max: M = max_s a_s, achieved at s*
- The optimal t* should decrease the maximum. If b_{s*} > 0, we want t* < 0 (decrease x). If b_{s*} < 0, we want t* > 0 (increase x).
- More precisely, we want to find t* that minimizes max_s [a_s + b_s * t*]
- This is equivalent to: find t* such that the weighted average of b_s (weighted by the "active" set) is 0.

def project_box_sum(y, target_sum):
    n = len(y)
    y = np.clip(y.astype(float), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-14:
        return y
    if diff < 0:
        order = np.argsort(y)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            reduce = min(y[idx], remaining)
            y[idx] -= reduce
            remaining -= reduce
    else:
        order = np.argsort(y)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            increase = min(1.0 - y[idx], remaining)

            remaining -= increase

            break

        max_val = float(np.max(corr))
        top3 = np.argsort(corr)[-3:]

        best_move = None
        best_new_max = max_val

        for k_idx in top3:
            k = int(k_idx)
            s = s_vals[k]
            if s == 0:
                continue

            A = np.zeros(n)
            ms = np.arange(n) + s
            valid = (ms >= 0) & (ms < n)
            A[valid] += h[ms[valid]]
            ms2 = np.arange(n) - s
            valid2 = (ms2 >= 0) & (ms2 < n)
            A[valid2] += h[ms2[valid2]]

            top_ones = ones[np.argsort(A[ones])[:min(30, len(ones))]]
            top_zeros = zeros[np.argsort(-A[zeros])[:min(30, len(zeros))]]

            for i in top_ones:
                for j in top_zeros:
                    delta_i = np.zeros(2 * n - 1)
                    ms_i = i + s_vals
                    fmask = (ms_i >= 0) & (ms_i < n)
                    delta_i[fmask] -= (1.0 - h[ms_i[fmask]])
                    ms_neg = i - s_vals
                    bmask = (ms_neg >= 0) & (ms_neg < n)
                    delta_i[bmask] += h[ms_neg[bmask]]

                    delta_j = np.zeros(2 * n - 1)
                    ms_j = j + s_vals
                    fmask_j = (ms_j >= 0) & (ms_j < n)
                    delta_j[fmask_j] += (1.0 - h[ms_j[fmask_j]])
                    ms_neg_j = j - s_vals
                    bmask_j = (ms_neg_j >= 0) & (ms_neg_j < n)
                    delta_j[bmask_j] -= h[ms_neg_j[bmask_j]]

                    new_corr = corr - delta_i + delta_j
                    nm = float(np.max(new_corr))
                    if nm < best_new_max:
                        best_new_max = nm
                        best_move = (i, j, delta_i, delta_j)

        if best_move is not None:
            i, j, di, dj = best_move
            h[i] = 0.0
            h[j] = 1.0
            corr += -di + dj
            stall = 0
            if best_new_max * dx < best_c5:
                best_c5 = best_new_max * dx
                best_h = h.copy()
        else:
            stall += 1

    return best_h, best_c5

def binary_two_swap(h, time_limit, start_time, max_iters=100000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    s_vals = np.arange(2 * n - 1) - (n - 1)
    stall = 0

    def swap_delta(m1, m2):
        d1 = np.zeros(2 * n - 1)
        ms = m1 + s_vals
        fm = (ms >= 0) & (ms < n)
        d1[fm] -= (1.0 - h[ms[fm]])
        mn = m1 - s_vals
        bm = (mn >= 0) & (mn < n)
        d1[bm] += h[mn[bm]]

        d2 = np.zeros(2 * n - 1)
        ms2 = m2 + s_vals
        fm2 = (ms2 >= 0) & (ms2 < n)
        d2[fm2] += (1.0 - h[ms2[fm2]])
        mn2 = m2 - s_vals
        bm2 = (mn2 >= 0) & (mn2 < n)
        d2[bm2] -= h[mn2[bm2]]
        return d1, d2

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        max_val = float(np.max(corr))
        k_star = int(np.argmax(corr))
        s = s_vals[k_star]
        if s == 0:
            s = s_vals[int(np.argsort(corr)[-2])]
        if s == 0:
            stall += 1
            continue

        A = np.zeros(n)
        ms = np.arange(n) + s
        valid = (ms >= 0) & (ms < n)
        A[valid] += h[ms[valid]]
        ms2 = np.arange(n) - s
        valid2 = (ms2 >= 0) & (ms2 < n)
        A[valid2] += h[ms2[valid2]]

        cand_ones = ones[np.argsort(A[ones])[:min(15, len(ones))]]
        cand_zeros = zeros[np.argsort(-A[zeros])[:min(15, len(zeros))]]

        best_move = None
        best_nm = max_val

        for i1 in cand_ones:
            for i2 in cand_ones:
                if i1 == i2:
                    continue
                for j1 in cand_zeros:
                    for j2 in cand_zeros:
                        if j1 == j2:
                            continue
                        d1, d2 = swap_delta(i1, i2)
                        new_max = float(np.max(corr - d1 - d2))
                        if new_max < best_nm:
                            best_nm = new_max
                            best_move = (i1, i2, d1, d2)

        if best_move is not None:
            i1, i2, d1, d2 = best_move
            h[i1] = 0.0
            h[i2] = 0.0
            h[best_move[3].shape[0] // (2 * n - 1) * 0 + 0]  # placeholder
            # Actually need to store j1, j2 too
            pass
        stall += 1

    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def diverse_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0

    best_h = project_binary(h_init, int(n_init / 2))
    best_score = compute_c5(best_h)
    best_cont_h = project_box_sum(h_init.copy(), target_init)
    best_cont_score = compute_c5(best_cont_h)

    if best_cont_score < best_score:
        best_score = best_cont_score
        best_h = best_cont_h.copy()

    # Phase 1: Vectorized coordinate descent
    allowed = min(120.0, tleft() * 0.15)
    if allowed > 20:
        h_cd, c5_cd = vectorized_coord_descent(best_cont_h, n_passes=50,
                                                time_limit=allowed, start_time=start_time)
        if c5_cd < best_cont_score:
            best_cont_score = c5_cd
            best_cont_h = h_cd

    # Phase 2: Momentum subgradient
    allowed = min(200.0, tleft() * 0.20)
    if allowed > 30:
        n_iters = int(min(500000, allowed * 5000))
        h_ms, c5_ms = momentum_subgradient(best_cont_h, n_iters,
                                            tau_init=0.5, tau_end=0.05,
                                            lr_init=0.05, lr_end=0.001,
                                            time_limit=allowed, start_time=start_time)
        if c5_ms < best_cont_score:
            best_cont_score = c5_ms
            best_cont_h = h_ms

    # Phase 3: Binary projection + local search
    h_bin = project_binary(best_cont_h, int(n_init / 2))
    h_bin_ls, c5_bin_ls = binary_local_search(h_bin, time_limit=min(100.0, tleft() * 0.12),
                                               start_time=start_time, max_iters=100000)
    if c5_bin_ls < best_score:
        best_score = c5_bin_ls
        best_h = h_bin_ls

    # Phase 4: Coordinate descent on binary (relaxed)
    allowed = min(80.0, tleft() * 0.10)
    if allowed > 15:
        h_cd2, c5_cd2 = vectorized_coord_descent(best_cont_h, n_passes=30,
                                                  time_limit=allowed, start_time=start_time)
        if c5_cd2 < best_cont_score:
            best_cont_score = c5_cd2
            best_cont_h = h_cd2

    # Phase 5: Re-project to binary and search again
    h_bin2 = project_binary(best_cont_h, int(n_init / 2))
    h_bin2_ls, c5_bin2_ls = binary_local_search(h_bin2, time_limit=min(80.0, tleft() * 0.10),
                                                 start_time=start_time, max_iters=80000)
    if c5_bin2_ls < best_score:
        best_score = c5_bin2_ls
        best_h = h_bin2_ls

    # Phase 6: SLSQP polish
    allowed = min(120.0, tleft() * 0.15)
    if allowed > 20:
        h_ref, sc_ref = slsqp_polish(best_cont_h, n_init, target_init,
                                      alphas=[30, 100, 500, 2000],
                                      maxiter=3000, ftol=1e-14,
                                      time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref

    # Phase 7: Final binary from polished continuous
    h_bin3 = project_binary(best_cont_h, int(n_init / 2))
    h_bin3_ls, c5_bin3_ls = binary_local_search(h_bin3, time_limit=min(60.0, tleft() * 0.08),
                                                 start_time=start_time, max_iters=50000)
    if c5_bin3_ls < best_score:
        best_score = c5_bin3_ls
        best_h = h_bin3_ls

    # Phase 8: Iterate local search from perturbed states
    for perturb_round in range(4):
        if tleft() < 60:
            break
        n_pert = rng.randint(3, 8)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        alloc = min(30.0, tleft() * 0.08)
        if alloc > 10:
            h_pls, c5_pls = binary_local_search(h_pert, time_limit=alloc,
                                                 start_time=start_time, max_iters=30000)
            if c5_pls < best_score:
                best_score = c5_pls
                best_h = h_pls

    # Phase 9: Multi-resolution
    for n_try in [n_init + 2, n_init - 2, n_init + 4]:
        if tleft() < 80:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_cont_h, n_try, target_try)

        h_cd3, c5_cd3 = vectorized_coord_descent(h_interp, n_passes=20,
                                                  time_limit=min(30.0, tleft() * 0.06),
                                                  start_time=start_time)
        h_bin_try = project_binary(h_cd3, n_try // 2)
        h_bin_try_ls, c5_bt_ls = binary_local_search(h_bin_try, time_limit=min(30.0, tleft() * 0.05),
                                                      start_time=start_time, max_iters=20000)

        best_at_res = min(c5_cd3, c5_bt_ls)
        if c5_cd3 < c5_bt_ls:
            best_at_res_h = h_cd3
        else:
            best_at_res_h = h_bin_try_ls

        if best_at_res < best_score:
            best_score = best_at_res
            best_h = best_at_res_h
            best_cont_h = h_cd3
            best_cont_score = c5_cd3

    # Phase 10: Final SLSQP + coordinate descent burst
    n_use = len(best_cont_h)
    if tleft() > 40:
        h_ref2, sc_ref2 = slsqp_polish(best_cont_h, n_use, float(n_use) / 2.0,
                                        alphas=[200, 1000, 5000],
                                        maxiter=5000, ftol=1e-15,
                                        time_limit=max(10.0, tleft() - 15),
                                        start_time=start_time)
        if sc_ref2 < best_cont_score:
            best_cont_score = sc_ref2
            best_cont_h = h_ref2

    if tleft() > 15:
        h_cd_final, c5_final = vectorized_coord_descent(best_cont_h, n_passes=10,
                                                         time_limit=max(5.0, tleft() - 5),
                                                         start_time=start_time)
        if c5_final < best_cont_score:
            best_cont_score = c5_final
            best_cont_h = h_cd_final

    # Final selection
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    n_bin = len(best_h)
    h_bin_final = project_binary(best_h, n_bin // 2)
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_bin
    else:
        return h_final, final_score, n_final
