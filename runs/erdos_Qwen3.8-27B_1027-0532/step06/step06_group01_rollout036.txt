import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

_SLQ_TARGET = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

min_d max_i (c_i + d * g_i) for d in [0, d_max]

- d=0 if f is non-decreasing at 0
- d=d_max if f is non-increasing at d_max  
- Otherwise, at an intersection point

The intersection of lines i and j is at d* = (c_j - c_i) / (g_i - g_j) (if g_i ≠ g_j).

We check all intersection points in [0, d_max] and the boundaries, and take the best.

One more optimization for the pair CD: instead of doing a full bisection for each pair, I can use the following trick:

For a pair (i,j) with gradient g = profiles[i] - profiles[j]:
- The new max is max(corr + delta * g)
- At delta = 0, the max is cur_max (achieved at the active lags)
- As delta increases from 0, the max changes based on the lags with the largest g values

If the active lag has g < 0 (decreasing), then increasing delta reduces the active lag's value. The max will either stay at the active lag (if no other lag catches up) or shift to another lag.

The "catch-up point" is where another lag's value equals the active lag's value:
corr[k_other] + delta * g[k_other] = corr[k_active] + delta * g[k_active]
delta = (corr[k_active] - corr[k_other]) / (g[k_other] - g[k_active])

If this delta is positive and less than delta_max, it's a candidate for the optimal delta.

So the algorithm is:
1. Find the active lag k* (argmax corr)
2. For the pair, compute g(k*)
3. If g(k*) < 0: the active lag is decreasing. Find the "catch-up" delta (where some other lag becomes the max). The optimal delta is min(catch_up_delta, delta_max).
4. If g(k*) >= 0: the active lag is increasing. The optimal delta might be 0 (no improvement possible).

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def _pair_cd_core(h_init, n, target, time_limit, n_cand=25, max_sweeps=500000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    idxs = np.arange(n)
    ks = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        top3 = np.argsort(corr)[-3:]
        grad_comb = np.zeros(n)
        for li in top3:
            k = int(li) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad_comb[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad_comb[v2] -= h[idxs[v2] + k]

        source_scores = grad_comb.copy()
        source_scores[h < 1e-8] = -1e30
        sink_scores = -grad_comb
        sink_scores[h > 1.0 - 1e-8] = -1e30

        top_sources = np.argsort(-source_scores)[:n_cand]
        top_sinks = np.argsort(-sink_scores)[:n_cand]

        all_cands = np.unique(np.concatenate([top_sources.astype(int), top_sinks.astype(int)]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            p1 = m - ks
            p2 = m + ks
            v1 = (p1 >= 0) & (p1 < n)
            v2 = (p2 >= 0) & (p2 < n)
            if np.any(v1):
                profiles[ci, v1] += h[p1[v1]]
            if np.any(v2):
                profiles[ci, v2] += h[p2[v2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in top_sources:
            hi_val = float(h[i])
            if hi_val < 1e-10:
                continue
            ci = cand_pos.get(int(i), -1)
            if ci < 0:
                continue
            for j in top_sinks:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-10:
                    continue
                cj = cand_pos.get(int(j), -1)
                if cj < 0:
                    continue

                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    f_mid = float(np.max(corr + mid_d * g))
                    if f_mid <= cur_max + 1e-15:
                        lo = mid_d
                    else:
                        hi_d = mid_d

                delta_star = lo
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def random_perturb(h, target, sigma, rng):
    n = len(h)
    noise = rng.standard_normal(n) * sigma
    noise -= noise.mean()
    h_pert = h + noise
    return project_feasible(h_pert, target)

def _multi_shift_subgrad(h_init, n, target, time_limit, max_steps=50000, n_shifts=3, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 2000:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [200, 1000, 5000, 20000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        cd_time = min(200, tleft() * 0.2)
        hcd, c5cd = _pair_cd_core(best_h, n, target, time_limit=cd_time, n_cand=25)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 50:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 80:
        hsg, c5sg = _multi_shift_subgrad(best_h, n, target, time_limit=min(40, tleft() * 0.2),
                                          max_steps=30000, n_shifts=3, base_lr=0.015)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    n_perturb_rounds = 0
    while tleft() > 100:
        n_perturb_rounds += 1
        frac = 1.0 - n_perturb_rounds * 0.05
        sigma = max(0.01, 0.08 * max(frac, 0.1))

        h_pert = random_perturb(best_h, target, sigma, rng)
        hcd, c5cd = _pair_cd_core(h_pert, n, target, time_limit=min(35, tleft() * 0.25), n_cand=25)
        if c5cd < best_score - 1e-14:
            best_score = c5cd
            best_h = hcd

        if tleft() > 60 and c5cd < best_score + 1e-5:
            hp, sc = slsqp_round(best_h, n, target, [10000, 50000], 1500, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = hp

    if tleft() > 50:
        hcd, c5cd = _pair_cd_core(best_h, n, target, time_limit=min(40, tleft() * 0.6), n_cand=30)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 30:
        hp, sc = slsqp_round(best_h, n, target, [20000, 100000, 500000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 20:
        hcd, c5cd = _pair_cd_core(best_h, n, target, time_limit=min(15, tleft() - 5), n_cand=20)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
