def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

For arrays a and b of length n, the cross-correlation (mode='full') at lag k (where k ranges from -(n-1) to (n-1)) is:
(a ★ b)[k] = Σ_i a[i] * b[i+k] (where the sum is over valid indices)

So corr[k + (n-1)] = Σ_i h[i] * (1-h[i+k]) = Σ_i h[i] - Σ_i h[i]*h[i+k]

The first term Σ_i h[i] = n/2 (by constraint), which is constant. So:
corr[k + (n-1)] = n/2 - autocorr_h[k]

where autocorr_h[k] = Σ_i h[i]*h[i+k] (the autocorrelation of h at lag k).

            break

        else:

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def compute_change_matrix(h, n):

    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
