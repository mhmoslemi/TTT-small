```python
def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase sum: add to positions that can accept more
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        # Need to decrease sum: subtract from positions that can give more
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

    return h

def lse_obj_and_grad(h, alpha, n):
    """Compute log-sum-exp objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = alpha * (corr - c_max)

    ws = w.sum()
    obj = (c_max + np.log(ws) / alpha) * dx
    
    # Gradient
    w = w / ws
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]

        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= grad.mean()  # Project gradient to satisfy sum constraint
    return obj, grad
```

```python
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-14:
            h = h_lp
        else:
            # Try smaller steps
            improved = False
            for mix in [0.5, 0.3, 0.1, 0.05]:
                h_try = project_feasible(mix * h + (1 - mix) * h_lp, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5
```

This is cleaner and should be more effective. The key improvement is using a threshold-based active set (all correlations within 5% of max) rather than a fixed number.
