        return h

    residual = target_sum - np.sum(h)

                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        improved = False
        decay = 1.0 / (1.0 + 0.005 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 2000:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B with moderate alpha (new: replaces SLSQP)
    if tleft() > 100:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[50, 200, 1000, 5000],
                            maxiter=500, time_limit=min(80, tleft() * 0.35),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Pair CD - main workhorse
    if tleft() > 180:
        cd_time = min(220, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=200000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: L-BFGS-B with higher alpha
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 20000, 100000],
                            maxiter=400, time_limit=min(70, tleft() * 0.3),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Pair CD with multi-shift gradient
    if tleft() > 150:
        cd_time = min(180, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=200000, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Subgradient with more shifts
    if tleft() > 60:
        hsg, c5sg = subgradient_phase(best_h, n, target, time_limit=min(50, tleft() * 0.35),
                                       max_steps=200000, n_shifts=7, base_lr=0.08)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 6: L-BFGS-B very high alpha
    if tleft() > 70:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[100000, 500000, 2000000],
                            maxiter=300, time_limit=min(60, tleft() * 0.4),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Pair CD with single shift, more candidates
    if tleft() > 100:
        cd_time = min(120, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=50, max_sweeps=200000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 8: Try higher resolution
    if tleft() > 250:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                            alphas=[50, 500, 5000, 50000],
                            maxiter=300, time_limit=min(100, tleft() * 0.3),
                            penalty=100.0)
        improved = sc < best_score
        if improved:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        elif time.time() - start_time + 100 < budget_s - 30:
            # Even if not improved yet, try pair CD
            pass
        if tleft() > 100:
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.35),
                                 n_cand=30, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n
                target = n / 2.0

    # Phase 9: Random perturbation + re-optimize
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        noise_scale = 0.02
        h_perturbed = best_h + rng.randn(n_use) * noise_scale
        h_perturbed = project_feasible(h_perturbed, target_use)
        # Quick L-BFGS to recover
        hp, sc = lbfgs_phase(h_perturbed, n_use, target_use,
                            alphas=[2000, 20000],
                            maxiter=200, time_limit=min(40, tleft() * 0.25),
                            penalty=500.0)
        if sc < best_score and tleft() > 50:
            hcd, c5cd = pair_cd(hp, n_use, target_use, time_limit=min(50, tleft() * 0.3),
                                 n_cand=25, max_sweeps=50000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
        elif sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 10: Final L-BFGS-B
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[500000, 2000000, 10000000],
                            maxiter=200, time_limit=min(40, tleft() * 0.6),
                            penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final pair CD
    if tleft() > 35:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(25, tleft() * 0.7),
                             n_cand=25, max_sweeps=50000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
