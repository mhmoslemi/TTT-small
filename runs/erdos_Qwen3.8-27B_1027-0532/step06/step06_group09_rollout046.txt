def smooth_max_obj(h, alpha, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def lbfags_polish(h_start, n, target, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        x_feas = np.clip(x, 0.0, 1.0)
        return smooth_max_obj(x_feas, alpha, target)

    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_lag_subgradient(h_init, n, target, time_limit, max_steps=20000,
                          m_top=8, lr=0.02, decay=0.999):

    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        top_indices = np.argsort(corr)[-m_top:]
        c_max = corr[top_indices[-1]]
        exp_args = np.clip(2.0 * (corr[top_indices] - c_max) / (c_max + 1e-10), -30, 0)
        lam = np.exp(exp_args)
        lam = lam / lam.sum()

        S = compute_sensitivity(h, n_corr, mid)
        grad = np.zeros(n)
        for w, ti in zip(lam, top_indices):
            grad += w * S[:, ti]

        grad *= dx
        grad -= np.mean(grad)

        current_lr = lr * (decay ** step)
        h_try = project_feasible(h - current_lr * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try
        else:
            h_try2 = project_feasible(h - 0.5 * current_lr * grad, target)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < c5_val:
                h = h_try2

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast coordinate descent - main workhorse
    if tleft() > 200:
        cd_time = min(350, tleft() * 0.25)
        h_cd, c5_cd = fast_coordinate_descent(best_h, n, target, time_limit=cd_time,
                                              max_iters=20000, n_cand_init=30,
                                              n_cand_max=80, patience=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: L-BFGS-B polish
    if tleft() > 100:
        for alpha in [500, 2000, 8000]:
            if tleft() < 40:
                break
            h_try = lbfags_polish(best_h, n, target, alpha, maxiter=1200)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 3: Second round of coordinate descent with different params
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.25)
        h_cd, c5_cd = fast_coordinate_descent(best_h, n, target, time_limit=cd_time,
                                              max_iters=20000, n_cand_init=40,
                                              n_cand_max=100, patience=250)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Multi-lag subgradient for exploration
    if tleft() > 120:
        sg_time = min(80, tleft() * 0.12)
        h_sg, c5_sg = multi_lag_subgradient(best_h, n, target, time_limit=sg_time,
                                            max_steps=15000, m_top=10,
                                            lr=0.025, decay=0.9995)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: More coordinate descent
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.22)
        h_cd, c5_cd = fast_coordinate_descent(best_h, n, target, time_limit=cd_time,
                                              max_iters=20000, n_cand_init=50,
                                              n_cand_max=100, patience=200)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Random restarts with diverse perturbations
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        for noise_scale in [0.02, 0.05, 0.1, 0.15, 0.25]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfags_polish(h_pert, n, target, 3000, maxiter=600)
            h_ref = fast_coordinate_descent(h_ref, n, target, time_limit=min(30, tleft() * 0.15),
                                            max_iters=5000, n_cand_init=30,
                                            n_cand_max=60, patience=100)[0]
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 7: Try higher resolution
    if tleft() > 200:
        for n_try in [500, 700, 900]:
            if tleft() < 100:
                break
            target_try = float(n_try // 2)
            if n_try % 2 != 0:
                target_try = float(n_try // 2)
            h_interp = interpolate_to_n(best_h, n_try, target_try)
            h_hires = lbfags_polish(h_interp, n_try, float(n_try // 2), 2000, maxiter=800)
            h_hires = fast_coordinate_descent(h_hires, n_try, float(n_try // 2),
                                              time_limit=min(60, tleft() * 0.25),
                                              max_iters=8000, n_cand_init=30,
                                              n_cand_max=70, patience=150)[0]
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = float(n_try // 2)

    # Phase 8: Final coordinate descent polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = fast_coordinate_descent(best_h, n_use, target_use,
                                              time_limit=min(80, tleft() * 0.5),
                                              max_iters=15000, n_cand_init=50,
                                              n_cand_max=100, patience=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: SLSQP final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pol = slsqp_round(best_h, n_use, target_use, [10000, 50000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
