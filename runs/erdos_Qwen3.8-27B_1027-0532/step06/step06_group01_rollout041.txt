    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        # Get top K lags
        top_indices = np.argsort(corr)[-K:]
        top_indices = sorted(top_indices, key=lambda x: -corr[x])
        
        # Compute combined gradient with softmax weights
        alpha_w = 50.0
        w = np.exp(alpha_w * (corr[top_indices] - corr[top_indices[0]]))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_indices):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        # Select candidates
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n-1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)

            if np.any(valid1):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

    return np.ones(len(x))


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def topk_pair_cd(h_init, n, target, time_limit, n_cand, max_sweeps, K=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        top_indices = np.argsort(corr)[-K:]
        top_indices = sorted(top_indices, key=lambda x: -corr[x])

        alpha_w = 60.0
        w = np.exp(alpha_w * (corr[top_indices] - corr[top_indices[0]]))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_indices):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, max_steps, n_shifts=5, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 40.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 1000:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP to smooth/flatten correlation profile
    if tleft() > 70:
        hp, sc = slsqp_round(best_h, n, target, [50, 200, 1000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Top-K paired CD (K=5)
    if tleft() > 250:
        cd_time = min(280, tleft() * 0.35)
        hcd, c5cd = topk_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=20, max_sweeps=50000, K=5)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: SLSQP polish
    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [500, 2000, 10000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Top-K CD with K=10 (address more lags simultaneously)
    if tleft() > 150:
        cd_time = min(180, tleft() * 0.3)
        hcd, c5cd = topk_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=30, max_sweeps=50000, K=10)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Multi-shift subgradient
    if tleft() > 50:
        hsg, c5sg = multi_shift_subgrad(best_h, n, target, time_limit=min(40, tleft() * 0.3),
                                        max_steps=30000, n_shifts=5, base_lr=0.015)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 6: SLSQP with higher alpha
    if tleft() > 50:
        hp, sc = slsqp_round(best_h, n, target, [10000, 50000, 200000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Try higher resolution n=1200
    n_high = 1200
    if tleft() > 350:
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        hp, sc = slsqp_round(h_interp, n_high, target_high, [50, 200, 1000, 5000], 1500, 1e-15)
        if tleft() > 200:
            hcd, c5cd = topk_pair_cd(hp, n_high, target_high, time_limit=min(180, tleft() * 0.4),
                                     n_cand=15, max_sweeps=30000, K=5)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_high
                target = target_high
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_high
            target = target_high

    # Phase 8: SLSQP at current resolution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [50000, 200000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 9: Final top-K CD
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(100, tleft() * 0.4)
        hcd, c5cd = topk_pair_cd(best_h, n_use, target_use, time_limit=cd_time, n_cand=20, max_sweeps=50000, K=8)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [100000, 500000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Tiny final CD
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = topk_pair_cd(best_h, n_use, target_use, time_limit=min(15, tleft() * 0.6),
                                 n_cand=15, max_sweeps=20000, K=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
