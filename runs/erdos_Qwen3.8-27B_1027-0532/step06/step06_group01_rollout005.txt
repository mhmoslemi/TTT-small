from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

Actually, here's another idea: **use the KKT conditions to verify and guide optimization**.

At the optimum, the KKT conditions require:
- For the active set A (shifts achieving the max): the gradients of f_k for k∈A should be "balanced" in a certain way
- Specifically, the subdifferential of max_k f_k at the optimum is conv{∇f_k : k ∈ A}
- The minimum of max_k f_k occurs where 0 ∈ conv{∇f_k : k ∈ A} + N_feasible

This means: there should exist weights w_k ≥ 0, sum w_k = 1, such that
sum_k w_k * ∇f_k ∈ N_feasible (the normal cone of the feasible set at h*)

For the feasible set {h : 0≤h≤1, sum(h)=n/2}, the normal cone at an interior point (0<h<1, sum=n/2) is the set of vectors orthogonal to the tangent space, which is span{1} (the all-ones vector).

So at the optimum (if h is interior), we need:
sum_k w_k * ∇f_k = λ * 1 for some λ

This gives us a way to check how close we are to optimal, and to guide the optimization.

1. Identify active shifts (within tolerance of max)

def fast_cd(h_init, n, target, time_limit, n_cand=20, max_sweeps=100000, ps=15):
    """Fast pair coordinate descent"""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Find worst shift
        k_star = int(np.argmax(corr)) - (n - 1)
        
        # Compute gradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Select candidates
        sorted_idx = np.argsort(-grad)
        high_set = sorted_idx[:n_cand]
        low_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles
        ks = np.arange(-(n - 1), n)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        # For each pair (i in high, j in low), find best delta
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            pi = profiles[ci]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = pi - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(ps):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

    cons = [{'type': 'eq', 'fun': _eql_fun, 'jac': _eql_jac}]
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_start, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_start = h_new
        except Exception:
            pass
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=20, m_shifts=1, max_sweeps=100000, ps=15):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Gradient from top-m shifts
        if m_shifts == 1:
            k_list = [int(np.argmax(corr)) - (n - 1)]
        else:
            top_idx = np.argsort(corr)[-m_shifts:]
            k_list = [int(si) - (n - 1) for si in top_idx]
        
        grad = np.zeros(n)
        for k in k_list:
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k]
        
        sorted_idx = np.argsort(-grad)
        high_set = sorted_idx[:n_cand]
        low_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        
        # Precompute profiles
        ks_arr = np.arange(-(n - 1), n)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            idx1 = m - ks_arr
            idx2 = m + ks_arr
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {int(all_cands[p]): p for p in range(n_c)}
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            pi = profiles[ci]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = pi - profiles[cj]
                
                lo = 0.0
                hi_d = delta_max
                for _ in range(ps):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
