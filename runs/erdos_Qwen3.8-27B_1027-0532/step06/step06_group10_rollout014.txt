def lp_mccormick(h0, n, target, time_limit, n_active=20, max_rounds=20):
    """LP relaxation with McCormick envelopes for active shifts."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    mid = n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        # Identify active shifts
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        
        # Build LP with McCormick envelopes
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target
        ]
        
        for s_idx in active:
            k = int(s_idx) - mid
            # Shift k: we need h[i] * h[i+k] for valid i
            # corr[s_idx] = sum of h[i] * (1-h[i+k]) over valid i
            # = sum h[i] - sum h[i]*h[i+k]
            
            # We need McCormick for z[i] = h[i] * h[i+k]
            # For each valid i, z[i] = h[i] * h[i+k]
            
            # Valid indices: i such that 0≤i<n and 0≤i+k<n
            if k >= 0:
                valid_i = np.arange(0, n - k)
                h_i = h_var[valid_i]
                h_j = h_var[valid_i + k]
            else:
                valid_i = np.arange(-k, n)
                h_i = h_var[valid_i]
                h_j = h_var[valid_i + k]
            
            m = len(valid_i)
            z = cp.Variable(m)
            
            # McCormick: z ≥ h_i + h_j - 1, z ≥ 0, z ≤ h_i, z ≤ h_j
            constraints.append(z >= h_i + h_j - 1)
            constraints.append(z >= 0)
            constraints.append(z <= h_i)
            constraints.append(z <= h_j)
            
            # The overlap for shift k is sum(h_i) - sum(z) ≤ t/dx
            # But wait, we also need to account for the part outside the overlap region
            # Actually, corr[s_idx] = sum over valid i of h[i]*(1-h[i+k])
            # = sum(valid_i) h[i] - sum(valid_i) h[i]*h[i+k]
            
            overlap = cp.sum(h_i) - cp.sum(z)
            constraints.append(overlap <= t_var / dx)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_val - 1e-12:
            h = h_lp
            best_c5 = c5_lp
            best_h = h_lp.copy()
        else:

                    h = h_mix
                    best_c5 = c5_mix
                    best_h = h_mix.copy()
                    break
            else:
                break
    
    return best_h, best_c5

def linearized_minmax(h0, n, target, time_limit, n_active=10, max_rounds=500, step_fracs=None):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    mid = n - 1
    if step_fracs is None:
        step_fracs = [1.0, 0.5, 0.25, 0.1]
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = float(c_max) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Get active shifts
        threshold = c_max - 1e-10  # active shifts
        active_mask = corr >= threshold
        active_indices = np.where(active_mask)[0]
        
        # If too many active, take top n_active
        if len(active_indices) > n_active:
            sorted_active = active_indices[np.argsort(corr[active_indices])[::-1]][:n_active]
        else:
            sorted_active = active_indices
        
        if len(sorted_active) == 0:
            break
        
        # Build linearized LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target
        ]
        
        for s_idx in sorted_active:
            k = int(s_idx) - mid
            # Compute gradient of f_k at current h
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            
            # Linearized: corr[s_idx] + grad_k @ (h_var - h) ≤ t_var / dx
            lin_val = corr[s_idx] + cp.sum(cp.multiply(grad_k, h_var - h))
            constraints.append(lin_val <= t_var / dx)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        
        # Try different step fractions
        improved = False
        for frac in step_fracs:
            h_try = project_feasible((1-frac)*h + frac*h_lp, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-14:
                h = h_try
                improved = True
                break
        
        if not improved:
            # Try the LP solution directly
            c5_lp = compute_c5(h_lp)
            if c5_lp < c5_val - 1e-14:
                h = h_lp
            else:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def linearized_minmax_lp(h0, n, target, time_limit, n_active=15, max_rounds=200):
    """LP-based min-max optimization with linearization."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    mid = n - 1
    
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        
        # Get active shifts
        threshold = c_max - 1e-8
        active_mask = corr >= threshold
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) > n_active:
            sorted_active = active_indices[np.argsort(corr[active_indices])[::-1]][:n_active]
        else:

def subgradient_adaptive(h0, n, target, time_limit, n_top=5):
    """Subgradient method with Barzilai-Borwein step size."""
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    mid = n - 1
    
    h_prev = h.copy()
    grad_prev = None
    step_count = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = float(c_max) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute subgradient (average of gradients of top shifts)
        sorted_idx = np.argsort(corr)[::-1][:n_top]
        w = np.exp(20.0 * (corr[sorted_idx] - c_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(w, sorted_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        
        # Barzilai-Borwein step size
        if step_count > 0 and grad_prev is not None:
            s = h - h_prev
            y = grad - grad_prev
            sy = s @ y
            yy = y @ y
            ss = s @ s
            if yy > 1e-10:
                lr = sy / yy if sy > 0 else ss / sy if sy < 0 else 0.01
            else:
                lr = 0.01
            lr = np.clip(lr, 0.001, 0.1)
        else:
            lr = 0.02
        
        h_new = project_feasible(h - lr * grad, target)
        h_prev = h
        h = h_new
        grad_prev = grad
        step_count += 1
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_resolution(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(28):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def smooth_obj_val(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_smooth_opt(h0, n, target, alphas, time_limit, maxiter_per_alpha=2000, ftol=1e-14):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        h = _slsqp_single_alpha(h, n, target, alpha, bounds, maxiter_per_alpha, ftol)
        h = project_feasible(h, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def _slsqp_single_alpha(h, n, target, alpha, bounds, maxiter, ftol):
    def _obj(x):
        return smooth_obj_val(x, alpha)
    def _grad(x):
        return smooth_obj_grad(x, alpha, n)
    def _con(x):
        return np.sum(x) - target
    def _con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def random_perturb_optimize(h0, n, target, time_limit, seed, n_restarts=5):
    start = time.time()
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    rng = np.random.RandomState(seed)
    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        n_perturb = rng.randint(5, 20)
        idx_perturb = rng.choice(n, n_perturb, replace=False)
        h_perturbed = best_h.copy()
        for idx in idx_perturb:
            h_perturbed[idx] += rng.uniform(-0.08, 0.08)
        h_perturbed = project_feasible(h_perturbed, target)
        t_remain = time.time() - start
        t_alloc = min(20.0, (time_limit - t_remain) * 0.4)
        if t_alloc < 3.0:
            break
        h_opt, s_opt = linmax_lp_optimize(h_perturbed, n, target, t_alloc, n_active=10, max_rounds=80)
        if s_opt < best_c5 - 1e-14:
            best_c5 = s_opt
            best_h = h_opt
        h_cd, s_cd = coord_descent(h_opt, n, target, min(5.0, t_alloc * 0.3), n_cand=15, max_sweeps=3000)
        if s_cd < best_c5 - 1e-14:
            best_c5 = s_cd
            best_h = h_cd
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return max(0.0, deadline - time.time())

    # Phase 1: Linearized LP (primary new method)
    if tleft() > 100:
        h1, s1 = linmax_lp_optimize(best_h, n, target, min(200.0, tleft() * 0.35),
                                     n_active=12, max_rounds=400)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: CD polish
    if tleft() > 80:
        h2, s2 = coord_descent(best_h, n, target, min(80.0, tleft() * 0.25),
                                n_cand=20, max_sweeps=15000)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: LP with more active shifts
    if tleft() > 100:
        h3, s3 = linmax_lp_optimize(best_h, n, target, min(120.0, tleft() * 0.3),
                                     n_active=25, max_rounds=250)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: McCormick LP (stricter relaxation)
    if tleft() > 80:
        h4, s4 = mck_lp_optimize(best_h, n, target, min(80.0, tleft() * 0.25),
                                  n_active=8, max_rounds=40)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SLSQP with smoothed objective
    if tleft() > 100:
        h5, s5 = slsqp_smooth_opt(best_h, n, target,
                                   [100, 500, 2000, 8000, 30000],
                                   time_limit=min(100.0, tleft() * 0.3),
                                   maxiter_per_alpha=2000, ftol=1e-14)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: CD after SLSQP
    if tleft() > 60:
        h6, s6 = coord_descent(best_h, n, target, min(60.0, tleft() * 0.3),
                                n_cand=25, max_sweeps=10000)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Random restarts
    if tleft() > 120:
        h7, s7 = random_perturb_optimize(best_h, n, target,
                                          min(120.0, tleft() * 0.3), seed, n_restarts=6)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: LP again after restarts
    if tleft() > 60:
        h8, s8 = linmax_lp_optimize(best_h, n, target, min(60.0, tleft() * 0.3),
                                     n_active=15, max_rounds=150)
        if s8 < best_score:
            best_score = s8
            best_h = h8
        if tleft() > 30:
            h8b, s8b = coord_descent(best_h, n, target, min(25.0, tleft() * 0.3),
                                      n_cand=25, max_sweeps=8000)
            if s8b < best_score:
                best_score = s8b
                best_h = h8b

    # Phase 9: Multi-resolution attempt
    if tleft() > 200:
        for n_try in [800, 1200, 2000]:
            if tleft() < 150:
                break
            target_try = float(n_try) / 2.0
            h_interp = interp_to_resolution(best_h, n_try, target_try)
            t_alloc = min(70.0, tleft() * 0.3)
            h_hr, s_hr = linmax_lp_optimize(h_interp, n_try, target_try,
                                             t_alloc, n_active=12, max_rounds=80)
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr
            if tleft() < 80:
                break
            h_cd_hr, s_cd_hr = coord_descent(best_h, len(best_h),
                                              float(len(best_h)) / 2.0,
                                              min(30.0, tleft() * 0.2),
                                              n_cand=15, max_sweeps=5000)
            if s_cd_hr < best_score:
                best_score = s_cd_hr
                best_h = h_cd_hr

    # Phase 10: Final LP + CD polish at best resolution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h10, s10 = linmax_lp_optimize(best_h, n_use, target_use,
                                       min(30.0, tleft() * 0.4),
                                       n_active=20, max_rounds=100)
        if s10 < best_score:
            best_score = s10
            best_h = h10
        if tleft() > 20:
            h10b, s10b = coord_descent(best_h, n_use, target_use,
                                        min(15.0, tleft() * 0.5),
                                        n_cand=30, max_sweeps=10000)
            if s10b < best_score:
                best_score = s10b
                best_h = h10b

    # Phase 11: Final SLSQP polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h11, s11 = slsqp_smooth_opt(best_h, n_use, target_use,
                                     [500, 2000, 10000, 50000],
                                     time_limit=min(20.0, tleft() * 0.6),
                                     maxiter_per_alpha=1500, ftol=1e-15)
        if s11 < best_score:
            best_score = s11
            best_h = h11

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
