```python
def block_slsqp_cd(h_init, n, target, block_size, time_limit, alpha=10000, n_blocks=None):
    start_time = time.time()
    dx = 2.0/n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    
    if n_blocks is None:
        n_blocks = n // block_size
    
    for sweep in range(10000):
        if time.time() - start_time > time_limit:
            break
        
        # Shuffle block order
        if sweep % 10 == 0:
            block_order = rng.permutation(n_blocks)
        
        for b_idx in block_order[sweep % 10]:
            if time.time() - start_time > time_limit:
                break
            
            # Select block
            start_idx = (b_idx * block_size) % (n - block_size + 1)
            if start_idx + block_size > n:
                start_idx = n - block_size
            idx = np.arange(start_idx, start_idx + block_size)
            
            # Local SLSQP on block
            h_block = h[idx].copy()
            
            def local_obj(x_block):
                h_trial = h.copy()
                h_trial[idx] = x_block
                # Project just the block to [0,1] and fix sum
                h_trial = np.clip(h_trial, 0, 1)
                # Fix sum by adjusting other elements slightly
                diff = target - np.sum(h_trial)
                if abs(diff) > 1e-12:
                    # Distribute adjustment to non-block elements
                    other_idx = np.setdiff1d(np.arange(n), idx)
                    if len(other_idx) > 0:
                        h_trial[other_idx] += diff / len(other_idx)
                        h_trial = np.clip(h_trial, 0, 1)
                corr = np.correlate(h_trial, 1 - h_trial, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500, 0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
            
            # SLSQP on block
            bounds = [(0,1)] * block_size
            try:
                res = minimize(local_obj, h_block, method='SLSQP', 
                             bounds=bounds, options={'maxiter': 50, 'ftol': 1e-10})
                if res.success or res.fun < local_obj(h_block):
                    h[idx] = np.clip(res.x, 0, 1)
                    h = project_feasible(h, target)
                    c5_now = compute_c5(h)
                    if c5_now < best_c5 - 1e-14:
                        best_c5 = c5_now
                        best_h = h.copy()
            except:
                pass
    
    return best_h, best_c5
```

            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 120:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def lp_cutting_plane(h_init, target, time_limit, max_iters=300):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h_cur = project_feasible(h_init.copy(), target)
    h_best = h_cur.copy()
    c5_best = compute_c5(h_cur)

    cuts = []

    def compute_grad_row(h_ref, m):
        k = m - (n - 1)
        idxs = np.arange(n)
        grad_row = np.zeros(n)
        v_pos = (idxs + k >= 0) & (idxs + k < n)
        v_neg = (idxs - k >= 0) & (idxs - k < n)
        grad_row[v_pos] += (1.0 - h_ref[idxs[v_pos] + k])
        grad_row[v_neg] -= h_ref[idxs[v_neg] - k]
        return grad_row

    def solve_lp():
        n_cuts = len(cuts)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        if n_cuts == 0:
            A_ub = np.zeros((1, n + 1))
            b_ub = np.zeros(1)
        else:
            A_ub = np.zeros((n_cuts, n + 1))
            b_ub = np.zeros(n_cuts)
            for i in range(n_cuts):
                A_ub[i, :n] = cuts[i][0]
                A_ub[i, n] = -1.0
                b_ub[i] = cuts[i][1]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([float(target)])
        bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        return res

    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.02)[0]
    for m in active:
        gr = compute_grad_row(h_cur, int(m))
        rhs = float(np.dot(gr, h_cur) - corr[int(m)])
        cuts.append((gr, rhs))

    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        try:
            res = solve_lp()
        except Exception:
            break
        if not res.success:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        h_lp = project_feasible(res.x[:n], target)
        c5_lp = compute_c5(h_lp)

        if c5_lp < c5_best - 1e-10:
            c5_best = c5_lp
            h_best = h_lp.copy()
            h_cur = h_lp.copy()
            no_improve = 0

            corr_new = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            max_new = np.max(corr_new)
            new_active = np.where(corr_new >= max_new - 0.005)[0]
            for m in new_active:
                mi = int(m)
                gr = compute_grad_row(h_cur, mi)
                rhs = float(np.dot(gr, h_cur) - corr_new[mi])
                cuts.append((gr, rhs))

            if len(cuts) > 300:
                tight_idx = []
                t_lp = float(res.x[-1])
                for ci in range(len(cuts)):
                    lhs = float(np.dot(cuts[ci][0], h_cur)) - t_lp
                    if lhs > cuts[ci][1] - 0.01:
                        tight_idx.append(ci)
                if len(tight_idx) >= 3:
                    cuts = [cuts[i] for i in tight_idx]
                else:
                    cuts = cuts[-150:]
        else:
            no_improve += 1
            corr_lp = np.correlate(h_lp, 1.0 - h_lp, mode='full')
            worst_m = int(np.argmax(corr_lp))
            gr = compute_grad_row(h_lp, worst_m)
            rhs = float(np.dot(gr, h_lp) - corr_lp[worst_m])
            cuts.append((gr, rhs))
            if len(cuts) > 400:
                cuts = cuts[-200:]

    return h_best, c5_best


def fourier_perturb_and_optimize(h_base, n, target, time_limit, seed_val):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h_best = h_base.copy()
    c5_best = compute_c5(h_base)

    for trial in range(5):
        if time.time() - start > time_limit:
            break
        h_trial = h_base.copy()
        n_modes = rng.choice([3, 5, 8, 12, 16])
        scale = rng.choice([0.05, 0.1, 0.2, 0.3])

        freq_signal = np.zeros(n)
        for _ in range(int(n_modes)):
            freq_idx = int(rng.randint(1, n // 2))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.randn() * scale
            freq_signal[np.minimum(freq_idx, n - freq_idx)] += amp * np.cos(phase)
            if freq_idx < n - freq_idx:
                freq_signal[n - freq_idx] += amp * np.cos(phase)

        h_perturbed = h_base + freq_signal
        h_perturbed = project_feasible(h_perturbed, target)

        cd_time = min(30, (time_limit - (time.time() - start)) * 0.5)
        if cd_time < 5:
            break
        h_h, c5_h = fast_pair_cd(h_perturbed, n, target, time_limit=cd_time,
                                  max_sweeps=2000, n_cand=30, n_bisect=12)
        if c5_h < c5_best:
            c5_best = c5_h
            h_best = h_h

    return h_best, c5_best


def micro_cd(h_init, n, target, time_limit, max_steps=20000):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(55555)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid

        n_candidates = 8
        candidates = []
        for _ in range(n_candidates * 3):
            i = int(rng.randint(n))
            if i not in candidates:
                candidates.append(i)
            if len(candidates) >= n_candidates:
                break

        best_move = None
        best_move_c5 = c5_now

        for i in candidates:
            delta_small = 1e-4
            for sign in [1, -1]:
                dh = sign * delta_small * (0.5 + h[i])
                if h[i] + dh < 0 or h[i] + dh > 1:
                    continue
                h_trial = h.copy()
                h_trial[i] += dh
                other = np.setdiff1d(np.arange(n), [i])
                adj = -dh / len(other)
                h_trial[other] += adj
                h_trial = np.clip(h_trial, 0, 1)
                sc = compute_c5(h_trial)
                if sc < best_move_c5 - 1e-14:
                    best_move_c5 = sc
                    best_move = h_trial.copy()

        if best_move is not None:
            h = best_move

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [500, 2000, 10000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main pair CD
    if tleft() > 250:
        cd_time = min(280, tleft() * 0.35)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=8000, n_cand=50, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: LP cutting plane (novel approach)
    if tleft() > 80:
        lp_time = min(60, tleft() * 0.15)
        h_lp, c5_lp = lp_cutting_plane(best_h, target, time_limit=lp_time, max_iters=300)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
            # Polish with CD
            if tleft() > 40:
                h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=min(30, tleft() * 0.3),
                                           max_sweeps=3000, n_cand=40, n_bisect=14)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 4: SLSQP high alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [10000, 50000, 200000], 2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Pair CD round 2
    if tleft() > 100:
        cd_time = min(120, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=6000, n_cand=60, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Micro CD (tiny perturbations)
    if tleft() > 60:
        mc_time = min(50, tleft() * 0.2)
        h_mc, c5_mc = micro_cd(best_h, n, target, time_limit=mc_time, max_steps=30000)
        if c5_mc < best_score:
            best_score = c5_mc
            best_h = h_mc

    # Phase 7: Fourier perturbations for diversity
    if tleft() > 100:
        ft_time = min(100, tleft() * 0.25)
        h_fp, c5_fp = fourier_perturb_and_optimize(best_h, n, target, time_limit=ft_time,
                                                    seed_val=seed + 1)
        if c5_fp < best_score:
            best_score = c5_fp
            best_h = h_fp
            if tleft() > 30:
                h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=min(30, tleft() * 0.3),
                                           max_sweeps=3000, n_cand=45, n_bisect=14)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 8: Higher resolution
    if tleft() > 200:
        for n_try in [800, 1000]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [500, 2000, 10000], 800, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 80 and sc < best_score + 0.002:
                cd_t = min(60, tleft() * 0.3)
                h_hires2, sc2 = fast_pair_cd(h_hires, n_try, target_try,
                                             time_limit=cd_t,
                                             max_sweeps=3000, n_cand=40, n_bisect=14)
                if tleft() > 30 and sc2 < best_score + 0.001:
                    h_pol2 = slsqp_round(h_hires2, n_try, target_try,
                                         [5000, 20000], 1000, 1e-15)
                    sc3 = compute_c5(h_pol2)
                    if sc3 < best_score:
                        best_score = sc3
                        best_h = h_pol2
                        n = n_try
                        target = target_try

    # Phase 9: Final CD
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=4000, n_cand=55, n_bisect=16)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [50000, 200000, 500000], 1500, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
