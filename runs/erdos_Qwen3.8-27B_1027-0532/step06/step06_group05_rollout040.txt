def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1

            h_r[idx[0]] = 0
            final_sum -= 1
        else:

            h_r[idx[0]] = 1
            final_sum += 1
    return h_r

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

                local_best_smooth = val

                stall += 1
            if stall > stall_limit:
                break
            
            m = beta1 * m + (1.0 - beta1) * grad

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr)) * dx


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-12:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad

                local_best_smooth = val
                stall = 0
            else:
                stall += 1
            if stall > stall_limit:
                break

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - eff_lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()

    return best_h, best_actual


def two_interval_c5(n, a, L1, c, L2):
    h = np.zeros(n)
    h[a:a + L1] = 1.0
    h[c:c + L2] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def two_interval_search(n, target, rng, time_limit, start_time):
    half = int(target)
    best_c5 = float('inf')
    best_params = None

    step_L = max(3, half // 60)
    step_pos = max(3, n // 80)

    for L1 in range(half // 5, 4 * half // 5, step_L):
        L2 = half - L1
        if L1 <= 0 or L2 <= 0:
            continue
        for a in range(0, n - L1, step_pos):
            if time.time() - start_time > time_limit:
                return best_params, best_c5
            gap_min = a + L1
            gap_max = n - L2
            if gap_min >= gap_max:
                continue
            for c in range(gap_min, gap_max, step_pos):
                sc = two_interval_c5(n, a, L1, c, L2)
                if sc < best_c5:
                    best_c5 = sc
                    best_params = (a, L1, c, L2)

    if best_params is not None and time.time() - start_time < time_limit * 0.7:
        a0, L10, c0, L20 = best_params
        for da in range(-step_L, step_L + 1, 1):
            for dc in range(-step_pos, step_pos + 1, 1):
                for dL in range(-step_L, step_L + 1, 1):
                    na = a0 + da
                    nL1 = L10 + dL
                    nc = c0 + dc
                    nL2 = L20 - dL
                    if na < 0 or na + nL1 > n or nc < 0 or nc + nL2 > n:
                        continue
                    if nL1 <= 0 or nL2 <= 0:
                        continue
                    if na + nL1 > nc and nc + nL2 > na:
                        continue
                    sc = two_interval_c5(n, na, nL1, nc, nL2)
                    if sc < best_c5:
                        best_c5 = sc
                        best_params = (na, nL1, nc, nL2)
                        a0, L10, c0, L20 = na, nL1, nc, nL2

    if best_params is not None:
        a, L1, c, L2 = best_params
        h = np.zeros(n)
        h[a:a + L1] = 1.0
        h[c:c + L2] = 1.0
        return h, best_c5
    return None, float('inf')


def greedy_binary(h, target_sum, max_iters, time_limit, start_time, scan_width=30):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 400

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] < 0.5]
        j_cand = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] >= 0.5]

        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue

        threshold = max_val * 0.82
        top_indices = np.where(corr >= threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])

        best_improve = 0.0
        best_i = -1
        best_j = -1
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_idx in top_indices:
                    k = int(t_idx) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def sa_binary(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 20000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        h_new = h.copy()
        moved = False
        r = rng.random()

        if r < 0.45:
            worst = np.argsort(corr)[-12:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.8:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        elif r < 0.92:
            blen = int(rng.randint(2, 5))
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i + blen <= n - 1 and j + blen <= n - 1:
                all_valid = True
                for b in range(blen):
                    if not (h[i + b] >= 0.5 and h[j + b] < 0.5):
                        all_valid = False
                        break
                if all_valid:
                    for b in range(blen):
                        h_new[i + b] = 0.0
                        h_new[j + b] = 1.0
                    moved = True
            if not moved:
                h_new[i] = 0.0
                h_new[j] = 1.0
                moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True

        if not moved:
            continue

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def subgradient_step(h, target_sum, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i] += (1.0 - h[i + k])
        for i in range(k, n):
            grad[i] -= h[i - k]
    else:
        p = -k
        for i in range(p, n):
            grad[i] += (1.0 - h[i - p])
        for i in range(n - p):
            grad[i] -= h[i + p]
    h_new = h - lr * grad
    return project_box_sum(h_new, target_sum)


def subgradient_method(h0, target_sum, start_time, time_budget, lr_init, lr_end, n_steps=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    if n_steps is None:
        n_steps = 500

    for t in range(n_steps):
        if time.time() - start_time > time_budget:
            break
        frac = t / max(n_steps - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** frac
        h = subgradient_step(h, target_sum, lr)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = 606
    target = float(n) // 2

    def tleft():
        return max(0.0, budget_s - 35.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Two-interval structured search
    if tleft() > 250:
        ti_time = min(120, tleft() * 0.12)
        h_ti, c5_ti = two_interval_search(n, target, rng, ti_time, start_time)
        if h_ti is not None and c5_ti < best_score:
            best_score = c5_ti
            best_h = h_ti.copy()
            if tleft() > 350:
                h_ti_ref, s_ti_ref = adam_optimize(
                    h_ti, target, start_time, min(180, tleft() * 0.15),
                    alphas=[200, 1000, 5000, 20000, 80000, 300000],
                    lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8, stall_limit=300)
                if s_ti_ref < best_score:
                    best_score = s_ti_ref
                    best_h = h_ti_ref.copy()

    # Phase 2: Binary greedy from current best
    if tleft() > 200:
        h_bin = project_binary(best_h, int(target))
        g_time = min(120, tleft() * 0.1)
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=50000,
                                  time_limit=g_time, start_time=start_time, scan_width=35)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 3: Main Adam optimization with aggressive alpha schedule
    if tleft() > 250:
        adam_time = min(450, tleft() * 0.3)
        h_a, s_a = adam_optimize(
            best_h, target, start_time, adam_time,
            alphas=[80, 200, 600, 2000, 6000, 20000, 60000, 200000, 600000],
            lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8, stall_limit=500,
            lr_decay=0.85)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 4: Subgradient method (different optimization landscape)
    if tleft() > 150:
        sg_time = min(100, tleft() * 0.1)
        h_sg, s_sg = subgradient_method(best_h, target, start_time, sg_time,
                                        lr_init=0.01, lr_end=0.0001, n_steps=300)
        if s_sg < best_score:
            best_score = s_sg
            best_h = h_sg.copy()

    # Phase 5: Multi-restart Adam from diverse starting points
    if tleft() > 250:
        n_restarts = 4
        per_rt = (tleft() - 60) / (n_restarts + 1)

        for ri in range(n_restarts):
            if tleft() < 80:
                break

            if ri == 0:
                h_pert = best_h + rng.randn(n) * 0.004
            elif ri == 1:
                h_pert = np.zeros(n)
                L1 = 140 + int(rng.randint(-30, 31))
                L2 = int(target) - L1
                if L2 <= 0:
                    L1 = 150
                    L2 = int(target) - L1
                a = int(rng.randint(0, 80))
                c = a + L1 + int(rng.randint(40, max(41, n - a - L1 - L2 - 20)))
                if c + L2 > n:
                    c = n - L2 - int(rng.randint(0, 50))
                    if c < 0:
                        c = 0
                h_pert[a:a + L1] = 1.0
                h_pert[c:c + L2] = 1.0
                h_pert += rng.randn(n) * 0.015
            elif ri == 2:
                h_pert = np.zeros(n)
                freq = int(rng.randint(2, 4))
                phase = rng.uniform(0, 2 * np.pi)
                h_pert = 0.5 + 0.25 * np.cos(2 * np.pi * freq * np.arange(n, dtype=float) / n + phase)
                h_pert = np.clip(h_pert, 0, 1)
            else:
                h_pert = project_binary(best_h, int(target)).astype(float)
                positions = rng.choice(n, size=20, replace=False)
                ones_pos = [p for p in positions if h_pert[p] >= 0.5]
                zeros_pos = [p for p in positions if h_pert[p] < 0.5]
                for p in ones_pos:
                    h_pert[p] = 0.0
                for p in zeros_pos:
                    h_pert[p] = 1.0

            h_pert = project_box_sum(h_pert, target)
            rt = max(40.0, min(per_rt, tleft() - 60))
            h_ar, s_ar = adam_optimize(
                h_pert, target, start_time, rt,
                alphas=[200, 1000, 5000, 20000, 80000, 300000],
                lr=0.0015, beta1=0.9, beta2=0.999, eps=1e-8, stall_limit=350,
                lr_decay=0.8)
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    # Phase 6: Binary SA
    if tleft() > 120:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(180, tleft() * 0.2)
        max_it = int(min(150000, sa_time * 400))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                0.006, 0.000003, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 7: SLSQP polish
    if tleft() > 80:
        slsqp_time = min(120, tleft() - 30)
        alphas_sl = [5000, 20000, 80000, 300000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     3000, 1e-14, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 8: Final high-alpha Adam polish
    if tleft() > 25:
        polish_time = max(8.0, tleft() - 8)
        h_p, s_p = adam_optimize(
            best_h, target, start_time, polish_time,
            alphas=[200000, 600000, 2000000],
            lr=0.0005, beta1=0.9, beta2=0.999, eps=1e-8, stall_limit=150)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
