import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        res = minimize(
            lambda x: smooth_obj_val(x, alpha), h,
            jac=lambda x: smooth_obj_grad(x, alpha, n),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x):

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _sgrad(x):
    alpha = _g_alpha
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

    _g_n = n
    _g_target = target
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:

            break
        _g_alpha = alpha
        try:

                          bounds=bounds,

                best_c5 = c5_new
                best_h = h_new.copy()

    return best_h, best_c5

Hmm, the "fast_cd" above is actually slow because it calls compute_c5 inside the inner loop. Let me fix this.

A better approach for CD: use the gradient to estimate the improvement without computing the full C5.

For a small perturbation δ moving mass from i to j:
The change in corr[s] for each s is:
Δcorr[s] ≈ δ * (grad_s[i] - grad_s[j])

where grad_s is the gradient of corr[s] with respect to h.

So the new max is approximately:

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_alpha = 100.0
_g_n = 400
_g_target = 200.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x):
    alpha = _g_alpha
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _sgrad(x):
    alpha = _g_alpha
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w[s] * x[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def _con_eq(x):
    return np.sum(x) - _g_target

def _con_eq_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h0, n, target, alphas, maxiter, ftol, deadline):
    global _g_alpha, _g_n, _g_target
    _g_n = n
    _g_target = target
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        _g_alpha = alpha
        try:
            res = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def compute_all_grads(h):
    n = len(h)
    mid = n - 1
    grads = np.zeros((2 * n - 1, n))
    grads[mid] = 1.0 - 2.0 * h
    for k in range(1, n):
        g1 = np.zeros(n)
        g1[:n - k] = 1.0 - h[k:]
        g1[k:] = -h[:n - k]
        grads[mid + k] = g1
        g2 = np.zeros(n)
        g2[k:] = 1.0 - h[:n - k]
        g2[:n - k] = -h[k:]
        grads[mid - k] = g2
    return grads

def lp_all(h0, n, target, deadline, max_rounds=100):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        grads = compute_all_grads(h)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        rhs = t_var - corr + grads @ h
        constraints.append(grads @ h_var <= rhs)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        improved = False
        if c5_lp < c5_val - 1e-10:
            h = h_lp
            improved = True
        else:
            for mix in [0.7, 0.5, 0.3, 0.1]:
                h_mix = project_feasible((1.0 - mix) * h + mix * h_lp, target)
                c5_mix = compute_c5(h_mix)
                if c5_mix < c5_val - 1e-10:
                    h = h_mix
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def greedy_cd(h0, n, target, deadline, n_cand=20, max_sweeps=30000, corr_interval=50):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % corr_interval == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 300:
                break
            k_star = int(np.argmax(corr)) - mid
        else:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(2 * n - 1)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def perturb_reoptimize(h0, n, target, deadline, rng, n_attempts=8):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    attempts_per_epsilon = max(1, n_attempts // 4)
    epsilons = [0.005, 0.01, 0.02, 0.05]
    for eps in epsilons:
        for _ in range(attempts_per_epsilon):
            if time.time() > deadline:
                return best_h, best_c5
            noise = rng.standard_normal(n) * eps
            h_pert = project_feasible(h0 + noise, target)
            c5_pert = compute_c5(h_pert)
            if c5_pert >= best_c5 + 0.005:
                continue
            sub_deadline = min(time.time() + 30.0, deadline - 5.0)
            h_s, s_s = slsqp_optimize(h_pert, n, target, [100, 500, 2000], 2000, 1e-13, sub_deadline)
            sub_deadline2 = min(time.time() + 20.0, deadline - 5.0)
            h_c, c_c = greedy_cd(h_s, n, target, sub_deadline2, n_cand=15, max_sweeps=5000, corr_interval=50)
            if c_c < best_c5 - 1e-12:
                best_c5 = c_c
                best_h = h_c.copy()
    return best_h, best_c5

def nesterov_smooth(h0, n, target, deadline, max_steps=20000):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    stop_count = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        alpha = 50.0 * (1.0 + step * 0.03)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stop_count = 0
        else:
            stop_count += 1
        if stop_count > 2000:
            break
        _g_n = n
        _g_alpha = alpha
        grad = _sgrad(h)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_try = project_feasible(h + beta * v - lr_trial * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                v = 0.9 * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
