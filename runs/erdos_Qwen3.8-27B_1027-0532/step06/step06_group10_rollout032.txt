    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

            break

            grad[v2] -= (corr[s] - t) * h[idxs[v2] + k]

        grad -= np.mean(grad)  # Remove mean to handle sum constraint

def block_cd(h0, n, target, time_limit, block_size=4, max_sweeps=5000):

    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    rng = np.random.RandomState(42)
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Select a block of coordinates to update
        sorted_idx = np.argsort(grad)
        # Take low and high groups
        low_block = sorted_idx[:block_size]
        high_block = sorted_idx[-block_size:]
        
        # Try to move mass from high to low
        d_max = min(np.min(1.0 - h[low_block]), np.min(h[high_block]))
        if d_max < 1e-12:
            continue
        
        # Bisection on the step size
        # Compute the "gain vector" for moving delta from each high to each low
        best_delta = 0.0
        best_improve = 0.0
        best_assign = None
        
        for hi in high_block:
            for lo in low_block:
                # Compute effect on corr
                g = np.zeros(n_corr)
                v_ip = (hi + shifts >= 0) & (hi + shifts < n)
                g[v_ip] += h[hi + shifts[v_ip]]
                v_im = (hi - shifts >= 0) & (hi - shifts < n)
                g[v_im] += h[hi - shifts[v_im]]
                v_lp = (lo + shifts >= 0) & (lo + shifts < n)
                g[v_lp] -= h[lo + shifts[v_lp]]
                v_lm = (lo - shifts >= 0) & (lo - shifts < n)
                g[v_lm] -= h[lo - shifts[v_lm]]
                
                # Bisection
                a, b = 0.0, d_max
                for _ in range(20):
                    mid_d = 0.5 * (a + b)
                    new_max = np.max(corr + mid_d * g) * dx
                    if new_max < c5_val:
                        a = mid_d
                    else:
                        b = mid_d
                delta = 0.5 * (a + b)
                if delta > best_delta:
                    best_delta = delta
                    best_assign = (hi, lo)
        
        if best_assign is not None and best_delta > 1e-12:
            hi, lo = best_assign
            h[hi] -= best_delta
            h[lo] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def annealed_subgrad(h0, n, target, time_limit, max_steps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    v = np.zeros(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Temperature annealing
        temp = max(0.01, 1.0 * (1.0 - step / max_steps))
        
        # Softmax weighting over top shifts
        top_k = min(20, len(corr))
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        w = np.exp(corr[sorted_idx] / (temp * dx))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, s_idx in zip(w, sorted_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        
        # Momentum update
        t_nest = (1.0 + np.sqrt(1.0 + 4.0 * (step+1)**2)) / 2.0
        beta = (t_nest - 1.0 - 1.0) / t_nest  # approximate momentum
        beta = max(0.0, min(0.9, (step / (1.0 + step))))
        lr = 0.1 / np.sqrt(step + 1)
        
        v = beta * v + (1 - beta) * grad
        h_new = project_feasible(h - lr * v, target)
        
        # Accept if improves or with small probability (simulated annealing)
        c5_try = compute_c5(h_new)
        if c5_try < c5_now or (np.random.random() < np.exp(-(c5_try - c5_now) / temp)):
            h = h_new
        else:
            if step > 1000 and c5_try > c5_now + 0.01:
                break
    
    return best_h, best_c5

            h[best_j] += best_delta

**Idea: Iterative refinement with "active set" tracking**

1. At each iteration, identify the set of "active" shifts (those where the overlap is close to the max)
2. Solve a QP: minimize $\max_{k \in \text{active}} g_k(h)$ approximately, using a smooth approximation
3. After the QP step, re-evaluate the full max and update the active set
4. If a non-active shift becomes the max, add it to the active set

This is essentially what the previous code does with SLSQP + log-sum-exp, but the "top n" trick (only considering top 50 shifts) is a key improvement.

**Another idea: Use the "cutting plane" method**
- Maintain a set of "cuts" (linear approximations of the max)
- At each step, solve a linear program with the current cuts
- Add a new cut based on the optimal solution
- This converges to the optimum of the max

But the $g_k(h)$ are quadratic, not linear, so cutting planes don't directly apply. However, we can linearize at the current point:
$g_k(h) \approx g_k(h_0) + \nabla g_k(h_0)^T (h - h_0)$

This gives a linear approximation, and we can use a cutting plane method:
1. Start with a few cuts
2. Solve LP: min t subject to $t \geq g_k^{\text{lin}}(h)$ for all cuts, plus feasibility constraints
3. Evaluate the true max at the LP solution
4. Add a new cut for the shift that achieves the max
5. Repeat

This could be more efficient than log-sum-exp because it directly targets the max.

Let me implement this:

def cutting_plane(h0, n, target, time_limit, max_rounds=100):

    cuts = []  # List of (grad_vector, dc) where g_k(h) ≈ dc + grad^T h
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        # Evaluate current corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid

        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

        k = k_star
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)

        # Actually: g_k(h_new) ≈ g_k(h) + grad_k(h)^T (h_new - h)
        # So: g_k(h_new) ≈ (g_k(h) - grad_k^T h) + grad_k^T h_new

        # In the LP: t >= g_k(h) + grad_k^T (h_new - h)
        # Which is: t - grad_k^T h_new >= g_k(h) - grad_k^T h
        
        # Formulate LP: min t
        # s.t. t - grad_k^T h_var >= g_k(h) - grad_k^T h  for each cut

        import cvxpy as cp
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        for g_cut, val_cut, h_cut in cuts:
            # t >= val_cut + g_cut^T (h_var - h_cut)
            constraints.append(t_var >= val_cut + g_cut @ (h_var - h_cut))

        grad_k_scaled = grad_k  # already scaled by dx
        val_current = float(corr[int(k_star + mid)]) * dx
        constraints.append(t_var >= val_current + grad_k_scaled @ (h_var - h))
        cuts.append((grad_k_scaled.copy(), val_current, h.copy()))
        
        # Keep only the most recent 30 cuts (oldest are likely not active)
        if len(cuts) > 30:
            cuts = cuts[-30:]
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_lp = project_feasible(h_var.value, target)
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_now - 1e-12:
            h = h_lp
        else:

            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_now:
                h = h_mix
            elif c5_lp < best_c5:
                h = h_lp
            else:
                break
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

        sorted_idx = np.argsort(corr)[::-1][:top_k]
        w = np.exp(corr[sorted_idx] / (temp * dx * 50.0))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, sorted_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        beta = min(0.85, step / (10.0 + step))
        lr = 0.05 / np.sqrt(1.0 + step * 0.01)
        v = beta * v + (1.0 - beta) * grad
        h_new = project_feasible(h - lr * v, target)
        c5_try = compute_c5(h_new)
        if c5_try <= c5_now or rng.random() < np.exp(-(c5_try - c5_now) / (temp * dx)):
            h = h_new
        else:
            if step > 2000 and c5_try > c5_now + 0.005:
                v *= 0.5
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 18.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP with top-n smooth max, adaptive n_top
    if tleft() > 120:
        n_top_start = 80
        h_p, s_p = slsqp_opt(best_h, n, target,
                             [100, 500, 2000, 8000],
                             n_top_start, 3000, 1e-14,
                             min(100.0, tleft() * 0.45), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Tighter top-n SLSQP (fewer active constraints = sharper)
    if tleft() > 100:
        h_p, s_p = slsqp_opt(best_h, n, target,
                             [500, 2000, 10000, 50000],
                             20, 3000, 1e-15,
                             min(80.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 3: Greedy coordinate descent
    if tleft() > 120:
        h_cd, c5_cd = greedy_cd(best_h, n, target,
                                time_limit=min(100.0, tleft() * 0.35),
                                max_sweeps=15000, n_cand=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Cutting plane LP refinement (novel approach)
    if tleft() > 80:
        h_cp, c5_cp = cutting_plane_refine(best_h, n, target,
                                           time_limit=min(70.0, tleft() * 0.3),
                                           max_rounds=50)
        if c5_cp < best_score:
            best_score = c5_cp
            best_h = h_cp

    # Phase 5: Multi-start perturbation
    if tleft() > 150:
        h_ms, c5_ms = multi_start(best_h, n, target,
                                   time_limit=min(130.0, tleft() * 0.4),
                                   n_starts=4, perturb_scale=0.02, n_top=30)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 6: Subgradient annealing (escape local minima)
    if tleft() > 80:
        h_as, c5_as = subgrad_anneal(best_h, n, target,
                                      time_limit=min(60.0, tleft() * 0.3),
                                      max_steps=15000)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 7: SLSQP + CD polish
    if tleft() > 100:
        h_p, s_p = slsqp_opt(best_h, n, target,
                             [1000, 5000, 20000, 100000],
                             15, 2500, 1e-15,
                             min(60.0, tleft() * 0.4), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p
        if tleft() > 50:
            h_cd, c5_cd = greedy_cd(best_h, n, target,
                                    time_limit=min(40.0, tleft() * 0.35),
                                    max_sweeps=10000, n_cand=22)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Higher resolution
    if tleft() > 150:
        for n_try in [800, 1200]:
            if tleft() < 100:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hr, s_hr = slsqp_opt(h_interp, n_try, target_try,
                                    [100, 500, 2000, 8000],
                                    40, 2000, 1e-14,
                                    min(70.0, tleft() * 0.35), start_time)
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr
            n_use = len(best_h)
            t_use = float(n_use) / 2.0
            if tleft() > 50:
                h_cd2, c5_cd2 = greedy_cd(best_h, n_use, t_use,
                                          time_limit=min(40.0, tleft() * 0.3),
                                          max_sweeps=8000, n_cand=15)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2

    # Phase 9: Final cutting plane
    if tleft() > 50:
        n_use = len(best_h)
        t_use = float(n_use) / 2.0
        h_cp, c5_cp = cutting_plane_refine(best_h, n_use, t_use,
                                           time_limit=min(40.0, tleft() * 0.5),
                                           max_rounds=40)
        if c5_cp < best_score:
            best_score = c5_cp
            best_h = h_cp

    # Phase 10: Final SLSQP + CD
    if tleft() > 40:
        n_use = len(best_h)
        t_use = float(n_use) / 2.0
        h_p, s_p = slsqp_opt(best_h, n_use, t_use,
                             [5000, 20000, 100000],
                             10, 2000, 1e-15,
                             min(25.0, tleft() * 0.5), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p
        if tleft() > 15:
            h_cd, c5_cd = greedy_cd(best_h, n_use, t_use,
                                    time_limit=min(12.0, tleft() * 0.6),
                                    max_sweeps=5000, n_cand=20)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
