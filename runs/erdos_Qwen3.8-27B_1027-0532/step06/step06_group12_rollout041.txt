        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_value(h, alpha):
    ov = compute_overlaps(h)
    m = np.max(ov)
    s = alpha * (ov - m)
    s = np.clip(s, -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_gradient(h, alpha):
    ov = compute_overlaps(h)
    m = np.max(ov)
    s = alpha * (ov - m)
    s = np.clip(s, -500.0, 0.0)
    e = np.exp(s)
    total = np.sum(e)
    if total < 1e-300:
        return np.zeros(len(h))
    w = e / total
    n = len(h)
    dx = 2.0 / n
    nh = n - 1
    g = np.zeros(n)
    for idx in range(len(w)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - nh
        if k >= 0:
            g[:n-k] += wi * dx * (1.0 - h[k:])
            g[k:] += wi * dx * (-h[:n-k])
        else:
            m2 = -k
            g[m2:] += wi * dx * (1.0 - h[:n-m2])
            g[:n-m2] += wi * dx * (-h[m2:])
    return g

def _smooth_obj(h):
    return smooth_value(h, _g_alpha)

def _smooth_grad(h):
    return smooth_gradient(h, _g_alpha)

def _eq_fun(h):
    return np.sum(h) - _g_target

def _eq_jac(h):
    return np.ones(_g_n)

def run_slsqp(h_start, n, target, alpha, maxiter, ftol):
    global _g_alpha, _g_target, _g_n
    _g_alpha = alpha
    _g_target = target
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_smooth_obj, h_start, jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_constraint(res.x, target)
    except Exception:
        return project_constraint(h_start.copy(), target)

def proj_subgrad(h_init, n, target, time_limit, max_iter=500000):
    """Projected subgradient with Nesterov acceleration for the max function"""
    start = time.time()
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_t = 0
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        ov = compute_overlaps(y)
        max_ov = np.max(ov)
        # Active set: lags within 1e-4 of max
        active_mask = ov > (max_ov - 1e-4)
        n_actives = max(1, int(np.sum(active_mask)))
        
        # Compute average subgradient
        g = np.zeros(n)
        nh = n - 1
        for idx in range(len(ov)):
            if not active_mask[idx]:
                continue
            k = idx - nh
            if k >= 0:
                g[:n-k] += (2.0/n) * (1.0 - y[k:])
                g[k:] += (2.0/n) * (-y[:n-k])
            else:
                m2 = -k
                g[m2:] += (2.0/n) * (1.0 - y[:n-m2])
                g[:n-m2] += (2.0/n) * (-y[m2:])
        g /= n_actives
        
        # Step size
        step = 0.5 / np.sqrt(t + 1)
        
        # Nesterov
        t_plus = t + 1
        beta = (1.0 - np.sqrt(1.0 + 4.0 * t_plus * t_plus)) / 2.0
        # Actually, standard Nesterov: beta_t = (1 - sqrt(1+4t^2))/(2t)
        # Let me use the simpler form: gamma = 2/(t+2)
        gamma = 2.0 / (t + 2.0)
        
        h_new = project_constraint(y - step * g, target)
        y_new = h_new + (h_new - h) * gamma / (1.0 + gamma)
        # Actually, the Nesterov update is:
        # y_{t+1} = (t/(t+2)) * x_{t+1} + (2/(t+2)) * y_t
        # Or equivalently: y_{t+1} = x_{t+1} + (t/(t+2))(x_{t+1} - x_t)
        # Let me use the simpler momentum form
        
        y_new = (1.0 + gamma) * h_new - gamma * h
        y_new = project_constraint(y_new, target)
        
        h = h_new
        y = y_new
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            best_t = t
        
        if t > 10000 and t - best_t > 200000:
            break
    
    return best_h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    def _tl():
        return budget_s - 20.0 - (time.time() - start_time)
    
    # Phase 1: SLSQP warm-start chain
    alpha_chain = [5, 20, 100, 500, 2000, 10000, 50000, 200000]
    for alpha in alpha_chain:
        tl = _tl()
        if tl < 40:
            break
        iters = 2000 if alpha < 1000 else 3000
        h_new = run_slsqp(best_h, n, target, alpha, maxiter=iters, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Projected subgradient with Nesterov
    tl = _tl()
    if tl > 80:
        h_new = proj_subgrad(best_h, n, target, time_limit=min(60, tl * 0.3))
        c5_new = compute_c5(h_new)
