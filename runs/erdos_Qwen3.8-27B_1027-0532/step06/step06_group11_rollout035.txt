```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g
```

```python
def lbgfbs_optimize(h_start, n, target, alpha_start, alpha_end, n_alphas, 
                    mu_start, mu_end, time_limit, maxiter=500):
    import time
    dx = 2.0 / n
    mid = n - 1
    start = time.time()
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    
    # Precompute: for each shift k, which indices are valid
    # We'll compute gradient by iterating over significant weights
    
    n_alphas = int(n_alphas)
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    mus = np.geomspace(mu_start, mu_end, n_alphas)
    
    for alpha, mu in zip(alphas, mus):
        if time.time() - start > time_limit:
            break
        
        def obj_and_grad(z):
            h = z
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            max_c = np.max(corr)
            shifted = alpha * (corr - max_c)
            shifted = np.clip(shifted, -500, 0)
            exp_shifted = np.exp(shifted)
            sum_exp = np.sum(exp_shifted)
            f_lse = max_c + np.log(sum_exp) / alpha
            
            # Penalty for constraint
            sum_h = np.sum(h)
            f_pen = mu * (sum_h - target) ** 2
            f = f_lse + f_pen
            
            # Gradient
            weights = exp_shifted / sum_exp
            grad = np.zeros(n)
            
            # Only iterate over significant weights
            sig_idx = np.where(weights > 1e-6)[0]
            for s_idx in sig_idx:
                k = int(s_idx) - mid
                w = weights[s_idx]
                # dC_k/dh
                idxs = np.arange(n)
                g = np.zeros(n)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += (1.0 - h[idxs[v1] - k])
                g[v2] -= h[idxs[v2] + k]
                grad += w * g
            grad *= dx
            
            # Penalty gradient
            grad += 2.0 * mu * (sum_h - target)
            
            return f, grad
        
        try:
            res = minimize(obj_and_grad, best_h, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 
                                   'gtol': 1e-8, 'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
```

```python
            h_try = project_feasible(h - lr_try * m_hat / (np.sqrt(v_hat) + eps), target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 10
            lr_decay = 0.95
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_perturbation(h_init, n, target, rng, n_blocks, block_sizes):
    h_try = h_init.copy()
    for _ in range(n_blocks):
        bs = int(rng.choice(block_sizes))
        bs = min(bs, n)
        start = int(rng.randint(0, n - bs))
        h_try[start:start+bs] = 1.0 - h_try[start:start+bs]
    h_try = project_feasible(h_try, target)
    return h_try

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B polish at initial resolution
    if tleft() > 200:
        h_l, sc = lbgfbs_optimize(best_h, n, target,
                                  alpha_list=[50, 200, 1000, 5000, 20000],
                                  mu_list=[100, 500, 2000, 10000, 50000],
                                  time_limit=min(80, tleft() * 0.3),
                                  maxiter=300)
        if sc < best_score:
            best_score = sc
            best_h = h_l

    # Phase 2: Top-k subgradient with adaptive LR
    if tleft() > 150:
        h_ad, sc = topk_subgrad_adaptive(best_h, n, target,
                                         time_limit=min(60, tleft() * 0.2),
                                         max_steps=20000, base_lr=0.02,
                                         n_top_start=3, n_top_end=12)
        if sc < best_score:
            best_score = sc
            best_h = h_ad

    # Phase 3: Multi-resolution bridge upward
    if tleft() > 300:
        n2 = 600
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        h_l2, sc2 = lbgfbs_optimize(h_interp, n2, target2,
                                    alpha_list=[100, 500, 2000, 10000, 50000],
                                    mu_list=[200, 1000, 5000, 20000, 100000],
                                    time_limit=min(100, tleft() * 0.25),
                                    maxiter=250)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_l2

    if tleft() > 250:
        n3 = 800
        target3 = n3 / 2.0
        h_interp3 = interpolate_to(best_h, len(best_h), n3, target3)
        h_l3, sc3 = lbgfbs_optimize(h_interp3, n3, target3,
                                    alpha_list=[50, 200, 1000, 5000, 20000],
                                    mu_list=[100, 500, 2000, 10000, 50000],
                                    time_limit=min(100, tleft() * 0.25),
                                    maxiter=200)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_l3

    # Phase 4: Interpolate back to n=400 for final refinement
    n_final = 400
    target_final = n_final / 2.0
    h_back = interpolate_to(best_h, n_final, target_final)
    if compute_c5(h_back) < best_score:
        best_score = compute_c5(h_back)
        best_h = h_back
    else:
        # Keep the original n=400 solution if it's better
        pass

    # Use the best at n=400 going forward
    n_use = 400
    target_use = n_use / 2.0
    h_work = interpolate_to(best_h, n_use, target_use) if len(best_h) != n_use else best_h.copy()
    work_score = compute_c5(h_work)
    if work_score < best_score:
        best_score = work_score
        best_h = h_work

    # Phase 5: L-BFGS-B at n=400 with high alpha
    if tleft() > 100:
        h_l5, sc5 = lbgfbs_optimize(h_work, n_use, target_use,
                                    alpha_list=[200, 1000, 5000, 20000, 100000, 500000],
                                    mu_list=[500, 2000, 10000, 50000, 200000, 500000],
                                    time_limit=min(80, tleft() * 0.3),
                                    maxiter=400)
        if sc5 < best_score:
            best_score = sc5
            best_h = h_l5

    # Phase 6: Pair CD
    if tleft() > 80:
        h_cd, sc_cd = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                              time_limit=min(60, tleft() * 0.3),
                              max_sweeps=3000, n_cand=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 7: SLSQP final polish
    if tleft() > 60:
        h_s, sc_s = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                   alphas=[5000, 20000, 100000, 500000, 2000000],
                                   maxiter_per_alpha=1000, ftol=1e-16)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 8: Random restarts with structured perturbation
    for noise_scale in [0.02, 0.05, 0.08, 0.12]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise_scale
        h_pert = project_feasible(h_pert, len(best_h) / 2.0)
        h_rp, sc_rp = lbgfbs_optimize(h_pert, len(best_h), len(best_h) / 2.0,
                                      alpha_list=[100, 1000, 10000, 100000],
                                      mu_list=[200, 2000, 20000, 200000],
                                      time_limit=min(40, tleft() * 0.2),
                                      maxiter=200)
        if sc_rp < best_score:
            best_score = sc_rp
            best_h = h_rp

    # Phase 9: Block perturbation restart
    if tleft() > 80:
        h_bp = block_perturbation(best_h, len(best_h), len(best_h) / 2.0,
                                  rng, n_blocks=2, block_sizes=[10, 20, 40, 60])
        h_rb, sc_rb = lbgfbs_optimize(h_bp, len(best_h), len(best_h) / 2.0,
                                      alpha_list=[100, 1000, 10000, 100000],
                                      mu_list=[200, 2000, 20000, 200000],
                                      time_limit=min(40, tleft() * 0.2),
                                      maxiter=200)
        if sc_rb < best_score:
            best_score = sc_rb
            best_h = h_rb

    # Phase 10: Final pair CD + SLSQP
    if tleft() > 50:
        h_cd2, sc_cd2 = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                time_limit=min(30, tleft() * 0.4),
                                max_sweeps=2000, n_cand=25)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 30:
        n_u = len(best_h)
        target_u = n_u / 2.0
        h_final_pol, sc_fp = slsqp_optimize(best_h, n_u, target_u,
                                            alphas=[10000, 50000, 200000, 1000000],
                                            maxiter_per_alpha=800, ftol=1e-16)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_final_pol

    # Final validation
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
