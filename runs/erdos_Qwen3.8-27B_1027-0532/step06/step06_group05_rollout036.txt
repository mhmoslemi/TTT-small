def adam_continuous(h0, target_sum, start_time, time_budget, alphas, lr, beta1, beta2, eps):

    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_actual = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')

        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-10:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)

        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()

    return best_h, best_actual

def binary_local_search_vectorized(h, target_sum, max_iters, time_limit, start_time, top_m=50, cand_width=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(np.max(corr))
        top_m_eff = min(top_m, 2*n-1)
        top_indices = np.argsort(corr)[-top_m_eff:]
        top_k = top_indices.astype(int) - (n - 1)  # actual shift values
        
        # For each candidate, compute the "profile" over top shifts
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Limit candidates
        i_cand = ones[:cand_width] if len(ones) <= cand_width else ones[rng.randint(len(ones), size=cand_width)]
        j_cand = zeros[:cand_width] if len(zeros) <= cand_width else zeros[rng.randint(len(zeros), size=cand_width)]
        
        # For each i in i_cand, compute the best j
        best_improve_total = 0
        best_i_idx = -1
        best_j_idx = -1
        
        for idx_i in range(len(i_cand)):
            i = i_cand[idx_i]
            # val_i[k] = h[i+k] + h[i-k] for k in top_k (with bounds checking)
            i_k = i + top_k
            i_mk = i - top_k
            valid_plus = (i_k >= 0) & (i_k < n)
            valid_minus = (i_mk >= 0) & (i_mk < n)
            val_i = np.zeros(top_m_eff)
            val_i[valid_plus] = h[i_k[valid_plus]]
            val_i[valid_minus] += h[i_mk[valid_minus]]
            
            # For all j candidates: val_j[k] = h[j+k] + h[j-k]
            j_plus = j_cand[:, None] + top_k[None, :]
            j_minus = j_cand[:, None] - top_k[None, :]
            valid_p = (j_plus >= 0) & (j_plus < n)
            valid_m = (j_minus >= 0) & (j_minus < n)
            val_j = np.zeros((len(j_cand), top_m_eff))
            val_j = np.where(valid_p, h[np.clip(j_plus, 0, n-1)], 0.0) + \
                  np.where(valid_m, h[np.clip(j_minus, 0, n-1)], 0.0)
            
            # new_c[j, k] = corr[top_indices][k] + val_i[k] - val_j[j, k]

    n = len(h)
    dx = 2.0 / n

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):

        if w < 1e-12:
            continue
        grad[:n - s] += w * one_minus_h[s:]

        if w < 1e-12:

        grad[:n - p] -= w * h[p:]

    best_actual = compute_c5(h)

        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')

        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1

            if val < local_best - 1e-10:
                local_best = val

            else:

            v = beta2 * v + (1.0 - beta2) * (grad * grad)

_slsqp_target = 0.0

def _slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_cons_fun(h):
    return float(np.sum(h)) - _slsqp_target

def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14 and float(np.min(h)) >= -1e-14 and float(np.max(h)) <= 1.0 + 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = (h >= 0.5).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    h_r = h_r.copy()
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        grad[:n - s] += w * one_minus_h[s:]
        if s > 0:
            grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-14:
            continue
        grad[p:] += w * one_minus_h[:n - p]
        grad[:n - p] -= w * h[p:]
    return value, grad


def adam_opt(h0, target_sum, start_time, time_budget, alphas, lr, beta1, beta2, eps):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_actual = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        stall = 0
        local_best = float('inf')
        while True:
            if time.time() - start_time > time_budget:
                break
            t += 1
            val, grad = smooth_obj_and_grad(h, alpha)
            if val < local_best - 1e-11:
                local_best = val
                stall = 0
            else:
                stall += 1
            if stall > 400:
                break
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_box_sum(h_new, target_sum)
        actual = compute_c5(h)
        if actual < best_actual:
            best_actual = actual
            best_h = h.copy()
    return best_h, best_actual


def binary_local_search(h, max_iters, time_limit, start_time, top_m, cand_width, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(np.max(corr))
        max_idx = int(np.argmax(corr))
        k_worst = max_idx - (n - 1)
        top_m_eff = min(top_m, 2 * n - 1)
        top_indices = np.argsort(corr)[-top_m_eff:]
        top_k = top_indices.astype(np.int64) - (n - 1)
        top_corr = corr[top_indices].astype(float)
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_score = np.zeros(len(ones))
        for idx in range(len(ones)):
            i = ones[idx]
            s = 0.0
            if 0 <= i + k_worst < n:
                s += h[i + k_worst]
            if 0 <= i - k_worst < n:
                s += h[i - k_worst]
            i_score[idx] = s
        j_score = np.zeros(len(zeros))
        for idx in range(len(zeros)):
            j = zeros[idx]
            s = 0.0
            if 0 <= j + k_worst < n:
                s += h[j + k_worst]
            if 0 <= j - k_worst < n:
                s += h[j - k_worst]
            j_score[idx] = s
        i_order = np.argsort(i_score)[:cand_width]
        j_order = np.argsort(-j_score)[:cand_width]
        if rng.random() < 0.15:
            i_order = rng.choice(len(ones), size=min(cand_width, len(ones)), replace=False)
            j_order = rng.choice(len(zeros), size=min(cand_width, len(zeros)), replace=False)
        i_cand = ones[i_order]
        j_cand = zeros[j_order]
        n_j = len(j_cand)
        j_plus = j_cand[:, None].astype(np.int64) + top_k[None, :]
        j_minus = j_cand[:, None].astype(np.int64) - top_k[None, :]
        valid_p = (j_plus >= 0) & (j_plus < n)
        valid_m = (j_minus >= 0) & (j_minus < n)
        j_plus_c = np.clip(j_plus, 0, n - 1)
        j_minus_c = np.clip(j_minus, 0, n - 1)
        val_j = np.where(valid_p, h[j_plus_c], 0.0) + np.where(valid_m, h[j_minus_c], 0.0)
        best_improve = 0.0
        best_i_pos = -1
        best_j_pos = -1
        for idx_i in range(len(i_cand)):
            i = i_cand[idx_i]
            i_k = i + top_k
            i_mk = i - top_k
            vp = (i_k >= 0) & (i_k < n)
            vm = (i_mk >= 0) & (i_mk < n)
            val_i = np.zeros(top_m_eff)
            i_k_c = np.clip(i_k, 0, n - 1)
            i_mk_c = np.clip(i_mk, 0, n - 1)
            val_i[vp] = h[i_k_c[vp]]
            val_i[vm] += h[i_mk_c[vm]]
            new_c = top_corr[None, :] + val_i[None, :] - val_j
            new_max = new_c.max(axis=1)
            bjl = int(np.argmin(new_max))
            imp = max_val - new_max[bjl]
            if imp > best_improve:
                best_improve = imp
                best_i_pos = idx_i
                best_j_pos = bjl
                if imp >= dx * 15:
                    break
        if best_i_pos >= 0 and best_improve > 0:
            h[i_cand[best_i_pos]] = 0.0
            h[j_cand[best_j_pos]] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(corr)) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if stall > 60 and rng.random() < 0.25:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = float(np.max(corr)) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
            else:
                stall += 1
    return best_h, best_c5


def _slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _slsqp_cons_fun(h):
    return float(np.sum(h)) - _slsqp_target


def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)


def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary local search from initial construction
    if tleft() > 100:
        h_bin = project_binary(best_h, int(target))
        t1 = min(250, tleft() * 0.18)
        h_bs, c5_bs = binary_local_search(h_bin, 100000, t1, start_time,
                                          top_m=40, cand_width=80, rng=rng)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs.copy()

    # Phase 2: Continuous Adam optimization
    if tleft() > 200:
        adam_time = min(450, tleft() * 0.30)
        h_a, s_a = adam_opt(
            best_h, target, start_time, adam_time,
            alphas=[40, 80, 160, 320, 640, 1200, 2500, 5000, 12000, 30000, 80000],
            lr=0.003, beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 3: Binary local search from continuous result
    if tleft() > 150:
        h_bin2 = project_binary(best_h, int(target))
        t3 = min(280, tleft() * 0.22)
        h_bs2, c5_bs2 = binary_local_search(h_bin2, 100000, t3, start_time,
                                            top_m=50, cand_width=100, rng=rng)
        if c5_bs2 < best_score:
            best_score = c5_bs2
            best_h = h_bs2.copy()

    # Phase 4: Multi-restart Adam from perturbed points
    if tleft() > 150:
        for restart_i in range(6):
            if tleft() < 60:
                break
            jitter_scale = 0.002 + 0.004 * restart_i
            h_pert = best_h + rng.randn(n) * jitter_scale
            h_pert = project_box_sum(h_pert, target)
            rt = min(120, (tleft() - 60) / (6 - restart_i))
            h_ar, s_ar = adam_opt(
                h_pert, target, start_time, rt,
                alphas=[100, 500, 2000, 8000, 30000, 100000],
                lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8
            )
            if s_ar < best_score:
                best_score = s_ar
                h_h_ar = h_ar.copy()
                h_bin_r = project_binary(h_h_ar, int(target))
                t_br = min(80, tleft() * 0.1)
                if t_br > 10:
                    h_br, c5_br = binary_local_search(h_bin_r, 50000, t_br,
                                                      start_time, top_m=50,
                                                      cand_width=100, rng=rng)
                    if c5_br < best_score:
                        best_score = c5_br
                        best_h = h_br.copy()
                else:
                    best_h = h_h_ar.copy()

    # Phase 5: SLSQP polish
    if tleft() > 80:
        slsqp_time = min(180, tleft() * 0.3)
        h_ref, sc_ref = slsqp_polish(best_h, n, target,
                                     [3000, 10000, 30000, 100000],
                                     3000, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 6: Final binary polish
    if tleft() > 40:
        h_bin3 = project_binary(best_h, int(target))
        t6 = max(10.0, tleft() * 0.5)
        h_bs3, c5_bs3 = binary_local_search(h_bin3, 100000, t6, start_time,
                                            top_m=60, cand_width=120, rng=rng)
        if c5_bs3 < best_score:
            best_score = c5_bs3
            best_h = h_bs3.copy()

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
