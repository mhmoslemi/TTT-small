        # d corr[k] / d h[i] = (1 - h[i+k]) if i+k valid, else 0
        #                        + h[i-k] * (-1) if i-k valid... wait
        # corr[k] = sum_i h[i] * (1 - h[i+k])
        # d corr[k] / d h[i] = (1 - h[i+k]) [from the term where i is the first index]
        #                      + h[i-k] * (-1) [from the term where i-k is the first index, i.e., h[i-k]*(1-h[i])]
        # Wait, corr[k] = sum_j h[j] * (1 - h[j+k])
        # The terms involving h[i] are:
        #   j = i: h[i] * (1 - h[i+k])  → derivative wrt h[i] is (1 - h[i+k])
        #   j = i-k: h[i-k] * (1 - h[i])  → derivative wrt h[i] is -h[i-k]
        # So d corr[k] / d h[i] = (1 - h[i+k]) - h[i-k] (with boundary handling)
        if valid_p.any():
            hotness[valid_p] += (1.0 - h[idx_p[valid_p]])
        if valid_m.any():
            hotness[valid_m] -= h[idx_m[valid_m]]
    return hotness
