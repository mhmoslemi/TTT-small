def project_box_sum(h, target):
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:

            h_r[zeros[order]] = 1

            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / beta
    d = float(np.sum(h)) - target
    val += penalty * d * d
    return val

    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(corr.max())
    shifted = beta * (corr - c_max)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()
    t_arr = np.arange(N)
    one_minus_h = 1.0 - h
    grad = np.zeros(N)
    for j in range(N):
        idx1 = j - t_arr + N - 1
        idx2 = t_arr - j + N - 1
        grad[j] = np.dot(one_minus_h, weights[idx1]) - np.dot(h, weights[idx2])
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (float(np.sum(h)) - target)
    return grad

    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    tabu = {}
    stall = 0
    max_stall = 800

    for it in range(2000000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_k_indices = np.argsort(corr)[-top_k:][::-1]
        max_idx = int(top_k_indices[0])
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_cand = []
        for i in ones:
            if tabu.get(i, -1) < it:
                if 0 <= i + k_star < n and h[i + k_star] == 0:
                    i_cand.append(i)
                if 0 <= i - k_star < n and h[i - k_star] == 0:
                    pass

        j_cand = []
        for j in zeros:
            if tabu.get(j, -1) < it:
                if 0 <= j + k_star < n and h[j + k_star] == 1:
                    j_cand.append(j)

        if len(i_cand) == 0 or len(j_cand) == 0:
            tabu.clear()
            i_cand = [i for i in ones if (0 <= i + k_star < n and h[i + k_star] == 0)]
            j_cand = [j for j in zeros if (0 <= j + k_star < n and h[j + k_star] == 1)]
            if len(i_cand) == 0 or len(j_cand) == 0:
                stall += 1
                continue

        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)

        best_score = max_val + 1.0
        best_i = -1
        best_j = -1

        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                nm = max_val
                for t_i in top_k_indices:
                    k = int(t_i) - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[int(t_i)] + d
                    if v > nm:
                        nm = v
                        break
                if nm < best_score:
                    best_score = nm
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            tabu[best_i] = it + tabu_len
            tabu[best_j] = it + tabu_len
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def kick_and_tabu(h_start, target_sum, rng, time_limit, start_time, kick_strength=6, tabu_len=50, top_k=3, scan_width=35):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < time_limit:
        h_kicked = best_h.copy()
        ones = np.where(h_kicked == 1)[0]
        zeros = np.where(h_kicked == 0)[0]
        n_swap = min(kick_strength, len(ones) // 2, len(zeros) // 2)
        if n_swap > 0:
            swap_ones = rng.choice(len(ones), n_swap, replace=False)
            swap_zeros = rng.choice(len(zeros), n_swap, replace=False)
            h_kicked[ones[swap_ones]] = 0
            h_kicked[zeros[swap_zeros]] = 1

        kick_time = min(30, (time_limit - (time.time() - start_time)) * 0.5)
        if kick_time < 3:
            break
        h_pol, c5_pol = tabu_search(h_kicked, target_sum, time_limit=kick_time,
                                     start_time=start_time, tabu_len=tabu_len,
                                     top_k=top_k, scan_width=scan_width)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5

def coarse_to_fine(h_init, time_limit, start_time, rng):
    n_full = len(h_init)
    levels = [60, 120, 240, n_full]
    h_cur = None

    for n_level in levels:
        if time.time() - start_time > time_limit:
            break
        target_l = float(n_level / 2)
        target_l_int = int(n_level / 2)

        if h_cur is None or len(h_cur) != n_level:
            if h_cur is None:
                x_new = np.linspace(0.0, 1.0, n_level)
                x_old = np.linspace(0.0, 1.0, n_full)
                h_cur = np.interp(x_new, x_old, h_init)
            else:
                x_new = np.linspace(0.0, 1.0, n_level)
                x_old = np.linspace(0.0, 1.0, len(h_cur))
                h_cur = np.interp(x_new, x_old, h_cur)
        h_cur = project_box_sum(h_cur, target_l)

        betas = [20, 100, 400] if n_level <= 240 else [100, 400, 1000, 2000]
        for beta in betas:
            if time.time() - start_time > time_limit:
                break
            alloc = min(25, (time_limit - (time.time() - start_time)) * 0.12)
            if alloc < 5:
                break
            h_opt = lbfgs_opt(h_cur, beta, target_l, penalty=500.0,
                               maxiter=150, time_limit=alloc, start_time=start_time)
            h_cur = h_opt

        if n_level == n_full:
            return h_cur

        h_bin = project_binary(h_cur, target_l_int)
        alloc2 = min(15, (time_limit - (time.time() - start_time)) * 0.12)
        if alloc2 > 3:
            h_bin_opt, _ = tabu_search(h_bin, target_l_int, time_limit=alloc2,
                                         start_time=start_time, tabu_len=30,
                                         top_k=2, scan_width=25)
            h_cur = h_bin_opt

    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_full = len(h_init)
    target_full = float(n_full / 2)
    target_full_int = int(n_full / 2)

    best_h = project_box_sum(h_init.copy(), target_full)
    best_c5 = compute_c5(best_h)

    h_bin_init = project_binary(h_init, target_full_int)
    c5_bin = compute_c5(h_bin_init)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin_init.copy()

    # Phase 1: Coarse-to-fine to potentially find different structure
    t_c2f = min(120, (budget_s - 150) * 0.12)
    if budget_s - 150 > 60:
        h_c2f = coarse_to_fine(h_init, time_limit=t_c2f, start_time=start_time, rng=rng)
        sc_c2f = compute_c5(h_c2f)
        if sc_c2f < best_c5:
            best_c5 = sc_c2f
            best_h = h_c2f.copy()
        h_bin_c2f = project_binary(h_c2f, target_full_int)
        c5_c2f_bin = compute_c5(h_bin_c2f)
        if c5_c2f_bin < best_c5:
            best_c5 = c5_c2f_bin
            best_h = h_bin_c2f.copy()

    # Phase 2: Main tabu search from best binary
    h_work = project_binary(best_h, target_full_int)
    t_main = min(450, (budget_s - 100) * 0.35)
    h_tab, c5_tab = tabu_search(h_work, target_full_int, time_limit=t_main,
                                 start_time=start_time, tabu_len=50, top_k=3, scan_width=40)
    if c5_tab < best_c5:
        best_c5 = c5_tab
        best_h = h_tab.copy()

    # Phase 3: Kick and tabu cycles (diversification)
    t_kick = min(350, (budget_s - 80) * 0.28)
    h_kick = project_binary(best_h, target_full_int)
    h_kp, c5_kp = kick_and_tabu(h_kick, target_full_int, rng, time_limit=t_kick,
                                 start_time=start_time, kick_strength=6,
                                 tabu_len=40, top_k=3, scan_width=35)
    if c5_kp < best_c5:
        best_c5 = c5_kp
        best_h = h_kp.copy()

    # Phase 4: Another kick with different parameters
    if time.time() - start_time < budget_s - 60:
        h_kick2 = project_binary(best_h, target_full_int)
        t_kick2 = min(200, (budget_s - 60) * 0.25)
        h_kp2, c5_kp2 = kick_and_tabu(h_kick2, target_full_int, rng, time_limit=t_kick2,
                                        start_time=start_time, kick_strength=10,
                                        tabu_len=60, top_k=2, scan_width=30)
        if c5_kp2 < best_c5:
            best_c5 = c5_kp2
            best_h = h_kp2.copy()

    # Phase 5: Continuous refinement
    if time.time() - start_time < budget_s - 40:
        h_cont = project_box_sum(best_h.copy(), target_full)
        for beta in [200, 500, 1500, 3000]:
            if time.time() - start_time > budget_s - 20:
                break
            alloc = min(25, (budget_s - 20 - (time.time() - start_time)) * 0.3)
            if alloc < 5:
                break
            h_opt = lbfgs_opt(h_cont, beta, target_full, penalty=1000.0,
                               maxiter=200, time_limit=alloc, start_time=start_time)
            h_cont = h_opt
            sc = compute_c5(h_cont)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_cont.copy()

    # Phase 6: Final binary polish on the best
    if time.time() - start_time < budget_s - 20:
        h_final_bin = project_binary(best_h, target_full_int)
        t_final = min(60, budget_s - 20 - (time.time() - start_time))
        if t_final > 5:
            h_fin, c5_fin = tabu_search(h_final_bin, target_full_int, time_limit=t_final,
                                         start_time=start_time, tabu_len=30, top_k=3, scan_width=35)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final / 2)
    best_h = project_box_sum(best_h, target_final)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
