import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    target = int(target_sum)
    h_r = (h >= 0.5).astype(float)
    current = int(h_r.sum())
    if current > target:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:current - target]
        h_r[ones[order]] = 0
    elif current < target:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:target - current]
        h_r[zeros[order]] = 1
    # Fix if still off (shouldn't happen)
    while int(h_r.sum()) != target:
        if int(h_r.sum()) > target:
            h_r[np.where(h_r == 1)[0][0]] = 0
        else:
            h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    # Gradient computation
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w > 1e-12:
            grad[:n - s] += w * one_minus_h[s:]
            if s > 0:
                grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w > 1e-12:
            grad[p:] += w * one_minus_h[:n - p]
            grad[:n - p] -= w * h[p:]
    return value, grad

def lbfgsb_smooth(h0, target, alpha, maxiter, time_limit, start_time):
    n = len(h0)
    
    def obj(h):
        val, _ = smooth_max_value_and_grad(h, alpha)
        return val
    
    def obj_and_grad(h):
        val, grad = smooth_max_value_and_grad(h, alpha)
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    # Note: L-BFGS-B doesn't handle equality constraints well
    # We'll use a penalty approach or just rely on projection
    
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for attempt in range(3):
        if time.time() - start_time > time_limit:

        res = minimize(obj_and_grad, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                                'maxfun': maxiter * 3})
        h_new = project_box_sum(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_val:
            best_val = sc
            best_h = h_new.copy()
        h = best_h.copy()
    
    return best_h, best_val

def active_set_optimize(h0, target, top_m, n_iters, time_limit, start_time):
    """Optimize using active set (top M correlations) gradient."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.001
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(corr.max())
        
        # Find top M shifts
        top_idx = np.argsort(corr)[-top_m:]
        top_weights = np.zeros(2 * n - 1)
        top_weights[top_idx] = 1.0 / top_m
        
        # Compute gradient of average of top M correlations
        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        for s in range(n):
            w = top_weights[n - 1 + s]
            if w > 0:
                grad[:n - s] += w * one_minus_h[s:]
                if s > 0:
                    grad[s:] -= w * h[:n - s]
        for p in range(1, n):
            w = top_weights[n - 1 - p]
            if w > 0:
                grad[p:] += w * one_minus_h[:n - p]
                grad[:n - p] -= w * h[p:]
        
        # Gradient step
        h_new = h - lr * grad
        h = project_box_sum(h_new, target)
        
        # Check improvement
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # Increase lr slightly
            lr = min(lr * 1.01, 0.01)
        else:
            lr = max(lr * 0.9, 0.0001)
    
    return best_h, best_c5

    n = len(h0)
    h = h0.copy()
    corr = compute_corr(h).astype(float)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()

            stall += 1

For the binary local search, I want to make it more efficient. The key insight is:

For a binary h, corr[k] is an integer. The max is some integer M. To reduce M, we need to find a swap that reduces the max.

A swap (i: 1→0, j: 0→1) changes corr[k] by:
Δ(k) = h[i+k] + h[i-k] - h[j+k] - h[j-k] (with out-of-range terms = 0)

Since h is binary, Δ(k) ∈ {-2, -1, 0, 1, 2}.

For the max to decrease, we need Δ(k*) < 0 for all k* where corr[k*] = M (the max).

So the problem becomes: find i (with h[i]=1) and j (with h[j]=0) such that for ALL max shifts k*, Δ(k*) ≤ -1.

If there are M_max shifts at the maximum, this is a system of constraints. For each max shift k*, we need:
h[i+k*] + h[i-k*] - h[j+k*] - h[j-k*] ≤ -1

This is a combinatorial constraint satisfaction problem. We can solve it by:
1. For each candidate i, determine which j's satisfy the constraints
2. Pick any valid (i,j) pair

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -3.0
    hi = 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(h_proj)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    return result

def project_binary(h, target_sum):
    n = len(h)
    target = int(target_sum)
    h_r = (h >= 0.5).astype(float)
    current = int(h_r.sum())
    if current > target:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:current - target]
        h_r[ones[order]] = 0
    elif current < target:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:target - current]
        h_r[zeros[order]] = 1
    while int(h_r.sum()) != target:
        if int(h_r.sum()) > target:
            h_r[np.where(h_r == 1)[0][0]] = 0
        else:
            h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w_sum = float(np.sum(exp_vals))
    weights = exp_vals / w_sum
    value = c_max + np.log(w_sum) / alpha
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(n):
        w = weights[n - 1 + s]
        if w > 1e-14:
            grad[:n - s] += w * one_minus_h[s:]
            if s > 0:
                grad[s:] -= w * h[:n - s]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w > 1e-14:
            grad[p:] += w * one_minus_h[:n - p]
            grad[:n - p] -= w * h[p:]
    return value, grad

def lbfgsb_with_sum(h0, target, alpha, maxiter, time_limit, start_time):
    """L-BFGS-B with penalty for sum constraint."""
    n = len(h0)
    penalty = 1000.0
    
    def obj_grad(h):
        val, grad = smooth_max_and_grad(h, alpha)
        s = float(np.sum(h)) - target
        val += penalty * s * s
        grad += 2.0 * penalty * s * np.ones(n)
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'maxfun': maxiter * 5})
    h_new = project_box_sum(res.x, target)
    sc = compute_c5(h_new)
    if sc < best_c5:
        best_c5 = sc
        best_h = h_new.copy()
    
    return best_h, best_c5

def slsqp_optimize(h0, target, alpha, maxiter, time_limit, start_time):
    n = len(h0)
    
    def obj(h):
        val, _ = smooth_max_and_grad(h, alpha)
        return val
    
    def obj_grad(h):
        return smooth_max_and_grad(h, alpha)
    
    def cons_fun(h):
        return float(np.sum(h)) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    try:
        res = minimize(obj_grad, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        sc = compute_c5(h_new)
        if sc < best_c5:

    except Exception:
        pass

def subgradient_max(h0, target, n_iters, lr_init, time_limit, start_time):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    patience = 0
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if patience > 2000:
            break
        
        corr = compute_corr(h).astype(float)
        c_max = float(corr.max())
        
        # Find all shifts within 1e-6 of max (tied or nearly tied)
        active = np.where(corr >= c_max - 1e-6)[0]
        
        # Average gradient of active shifts
        grad = np.zeros(n)
        one_minus_h = 1.0 - h
        for a_idx in active:
            k = int(a_idx) - (n - 1)
            # grad of corr[n-1+k] w.r.t. h
            for s in range(n):
                # corr[n-1+s]: h[i] * (1-h[i+s]) for valid i
                # d/dh[j] = (1-h[j+s]) if j is the "first" index
                #           -h[j-s] if j is the "second" index
                pass
            # More efficient: 
            # corr[n-1+k] = sum_{i=max(0,-k)}^{min(n-1,n-1-k)} h[i]*(1-h[i+k])
            # d/dh[j] = (1-h[j+k]) [if j+k is valid] - h[j-k] [if j-k is valid]
            for j in range(n):
                g = 0.0
                if 0 <= j + k < n:
                    g += (1.0 - h[j + k])
                if 0 <= j - k < n:
                    g -= h[j - k]
                grad[j] += g
        
        grad /= len(active)
        # Center gradient for sum constraint
        grad -= np.mean(grad)
        
        h_new = h - lr * grad
        h = project_box_sum(h_new, target)
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
            lr = min(lr * 1.01, 0.01)
        else:
            patience += 1
            lr = max(lr * 0.98, 0.0005)
    
    return best_h, best_c5

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -3.0

        else:
