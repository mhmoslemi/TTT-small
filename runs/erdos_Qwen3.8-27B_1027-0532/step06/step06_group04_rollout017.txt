def coord_descent_v2(h_init, n_active, max_passes, time_limit, rng, topk_update=1):
    """Coordinate descent on the max of linear functions."""
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argsort(corr)[-n_active:]
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        order = rng.permutation(n)
        for i in order:
            slopes = np.empty(n_active)
            intercepts = np.empty(n_active)
            for idx in range(n_active):
                k_idx = active_k[idx]
                k = k_idx - half_n
                A = 0.0
                if 0 <= i + k < n:
                    A += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    A -= h[i - k]
                A *= dx
                slopes[idx] = A
                intercepts[idx] = corr[k_idx] - A * h[i]
            # Solve: min max_j (slopes[j]*x + intercepts[j]) s.t. 0<=x<=1
            # This is a 1D convex problem
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = 0.5*(lo+hi)
                vals = slopes*mid + intercepts
                mx = np.max(vals)
                if mx < 0:
                    hi = mid
                elif np.all(slopes > 1e-14):
                    hi = mid
                else:
                    active = np.abs(vals - mx) < 1e-10 * max(1.0, abs(mx))
                    if not np.any(active):
                        active = np.zeros(n_active, dtype=bool)
                        active[np.argmax(vals)] = True
                    avg_slope = np.mean(slopes[active])
                    if avg_slope > 1e-14:
                        hi = mid
                    elif avg_slope < -1e-14:
                        lo = mid
                    else:
                        break
            h[i] = 0.5*(lo+hi)
        
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        if p % topk_update == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            active_k = np.argsort(corr)[-n_active:]
    
    return best_h, best_c5
