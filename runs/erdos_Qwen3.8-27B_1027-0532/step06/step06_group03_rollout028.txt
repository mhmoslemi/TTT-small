        # corr_k = sum_m h[m] * (1-h[m+k-(n-1)])

    shifts = np.arange(n_corr) - (n - 1)  # k' = k - (n-1) ranges from -(n-1) to n-1
    
    for i in range(n):
        # First term: (1-h[i + k']) where k' = k-(n-1)
        # Valid when 0 <= i+k' < n
        k_prime = shifts  # (n-1) to (n-1) ... wait
        # shifts = np.arange(n_corr) - (n-1) = [-(n-1), ..., 0, ..., n-1]
        # i + k' valid when 0 <= i + k' < n, i.e., -i <= k' < n-i
        valid1 = (idxs[i] + shifts >= 0) & (idxs[i] + shifts < n)
        M[i, valid1] += one_minus_h[idxs[i] + shifts[valid1]]
        
        # Second term: -h[i - k'] where i-k' valid when 0 <= i-k' < n, i.e., i-n+1 <= k' <= i
        valid2 = (idxs[i] - shifts >= 0) & (idxs[i] - shifts < n)
        M[i, valid2] -= h[idxs[i] - shifts[valid2]]
    
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        
        # Update corr and M
        corr = np.correlate(h, one_minus_h, mode='full')
        c5_val = np.max(corr) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 200:
            break
        
        # Update M
        one_minus_h = 1.0 - h
        M = np.zeros((n, n_corr))
        for i in range(n):
            valid1 = (i + shifts >= 0) & (i + shifts < n)
            M[i, valid1] += one_minus_h[i + shifts[valid1]]
            valid2 = (i - shifts >= 0) & (i - shifts < n)
            M[i, valid2] -= h[i - shifts[valid2]]
        
        # Select candidate indices based on gradient
        # Gradient (for the max of corr) is approximately M[:, top_k]
        top_k_ = 3
        top_idx = np.argsort(corr)[-top_k_]
        grad = np.zeros(n)
        for s in top_idx:
            grad += M[:, s]
        grad /= top_k_
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Compute delta bounds for all pairs
        # For pair (i, j): delta in [-(1-h[j]), h[i]] (decrease h[i] by delta, increase h[j] by delta)
        # Actually: h[i] - delta >= 0 => delta <= h[i]
        #           h[j] + delta <= 1 => delta <= 1-h[j]  (wait, delta is the amount transferred)
        # Let me redefine: new_h[i] = h[i] - delta, new_h[j] = h[j] + delta
        # Constraints: 0 <= h[i]-delta, h[i]-delta <= 1, 0 <= h[j]+delta, h[j]+delta <= 1
        # => delta <= h[i], delta >= h[i]-1, delta <= 1-h[j], delta >= -h[j]
        # => max(h[i]-1, -h[j]) <= delta <= min(h[i], 1-h[j])
        
        hi_vals = h[high_set]  # (n_cand,)
        lo_vals = h[low_set]  # (n_cand,)
        
        # delta_max[i,j] = min(hi_vals[i], 1-lo_vals[j])
        # delta_min[i,j] = max(hi_vals[i]-1, -lo_vals[j])
        
        delta_max = np.minimum(hi_vals[:, None], 1.0 - lo_vals[None, :])  # (n_cand, n_cand)
        delta_min = np.maximum(hi_vals[:, None] - 1.0, -lo_vals[None, :])  # (n_cand, n_cand)
        
        # Filter valid pairs
        valid_pairs = delta_max > delta_min + 1e-12
        if not valid_pairs.any():
            break
        
        # Compute Δcorr for all pairs: (n_cand, n_cand, n_corr)
        # Δcorr[i,j,:] = M[high_set[i], :] - M[low_set[j], :]
        # This might be too much memory for large n_cand
        # For n_cand=30, n_corr=799: 30*30*799*8 = 5.7 MB, fine
        
        M_hi = M[high_set]  # (n_cand, n_corr)
        M_lo = M[low_set]  # (n_cand, n_corr)
        
        # For each valid pair, find optimal delta
        best_c5_sweep = c5_val
        best_pair = None
        best_delta = 0
        
        # Process in batches to save memory
        batch_size = 100
        num_pairs = len(high_set) * len(low_set)
        
        for batch_start in range(0, num_pairs, batch_size):
            batch_end = min(batch_start + batch_size, num_pairs)
            for pi in range(batch_start, batch_end):
                i_idx = pi // len(low_set)
                j_idx = pi % len(low_set)
                if not valid_pairs[i_idx, j_idx]:
                    continue
                
                d_max = delta_max[i_idx, j_idx]
                d_min = delta_min[i_idx, j_idx]
                
                dc = M_hi[i_idx] - M_lo[j_idx]  # (n_corr,)
                
                # Bisection to find optimal delta
                a_db = d_min
                b_db = d_max
                for _ in range(15):
                    mid_db = 0.5 * (a_db + b_db)
                    vals = corr + mid_db * dc
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(dc[active])
                    if f_prime > 1e-14:
                        b_db = mid_db
                    else:
                        a_db = mid_db
                
                delta_star = 0.5 * (a_db + b_db)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * dc) * dx
                if approx_max < best_c5_sweep:
                    best_c5_sweep = approx_max
                    best_pair = (high_set[i_idx], low_set[j_idx])
                    best_delta = delta_star
        
        if best_pair is not None:
            i, j = best_pair
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            break
    
    return best_h, best_c5

Hmm, this is still O(n_cand² * 15 * n_corr) per sweep due to the bisection. The vectorization helps a little (numpy operations are faster than Python loops) but the fundamental complexity is the same.

Let me think of a faster approach. What if I skip the bisection and just use a fixed delta based on the slope at 0?

**Simplified pair CD:**
- For each valid pair (i,j), compute slope = max(dc[active]) at delta=0
- If slope > 0, the function is increasing at 0, so we should decrease delta (i.e., make it negative or 0)
- If slope < 0, the function is decreasing at 0, so we should increase delta
- Use a simple step: delta = -slope * step_size (clamped to valid range)

This is O(n_cand² * n_corr) per sweep (just the slope computation), which is 15x faster than the bisection version.

But it might not find the optimal delta. Let me use a compromise:
- First pass: compute slope for all pairs, pick the top 5 most promising
- Second pass: do bisection only for those 5 pairs

This is O(5 * 15 * n_corr) = O(75 * n_corr) ≈ O(60000) per sweep for the bisection part, plus O(n_cand² * n_corr) for the slope computation.

Total per sweep: O(n_cand² * n_corr + 75 * n_corr) ≈ O(900 * 799 + 60000) ≈ 780K operations. Much better!

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    mid = n - 1
    one_minus_h = 1.0 - h
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - mid
    idxs = np.arange(n)
    i_plus = idxs[:, None] + shifts[None, :]
    i_minus = idxs[:, None] - shifts[None, :]
    valid_p = (i_plus >= 0) & (i_plus < n)
    valid_m = (i_minus >= 0) & (i_minus < n)
    i_plus_s = np.clip(i_plus, 0, n - 1)
    i_minus_s = np.clip(i_minus, 0, n - 1)
    M = np.where(valid_p, one_minus_h[i_plus_s], 0.0) - np.where(valid_m, h[i_minus_s], 0.0)
    grad = M.T @ weights
    grad *= dx
    return grad


def _con_fun(h):
    return np.sum(h) - _global_target


def _con_jac(h):
    return np.ones(_global_n)


def slsqp_analytic(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_influence(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    one_minus_h = 1.0 - h
    shifts = np.arange(n_corr) - mid
    idxs = np.arange(n)
    i_plus = idxs[:, None] + shifts[None, :]
    i_minus = idxs[:, None] - shifts[None, :]
    valid_p = (i_plus >= 0) & (i_plus < n)
    valid_m = (i_minus >= 0) & (i_minus < n)
    i_plus_s = np.clip(i_plus, 0, n - 1)
    i_minus_s = np.clip(i_minus, 0, n - 1)
    M = np.where(valid_p, one_minus_h[i_plus_s], 0.0) - np.where(valid_m, h[i_minus_s], 0.0)
    return M
