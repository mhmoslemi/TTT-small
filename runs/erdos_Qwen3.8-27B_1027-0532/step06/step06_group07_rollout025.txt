```python
import numpy as np

# Global variables for SLSQP
_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def active_set_descent(h_init, n, target, time_limit, n_active=5, lr_init=0.1):
    """Focus on active constraints and do projected gradient descent."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    lr = lr_init

    for step in range(5000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        # Find top n_active shifts
        top_shifts = np.argsort(corr)[::-1][:n_active]
        if corr[top_shifts[0]] - corr[top_shifts[-1]] < 1e-10:
            # All active shifts are very close, reduce n_active
            pass

        # Compute gradient as weighted sum of active constraint gradients
        grad = np.zeros(n)
        total_weight = 0.0
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            w = corr[int(s_idx)]  # weight by violation magnitude
            total_weight += w
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad /= total_weight if total_weight > 0 else 1.0
        grad *= 2.0 * dx
        grad -= np.mean(grad)

        # Projected gradient step with backtracking
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c_try = compute_c5(h_try)
            if c_try < c5_now - 1e-15:
                h = h_try
                lr = lr_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 50:
                break

    return best_h, best_c5


def active_set_descent(h_init, n, target, time_limit, n_active=5, lr_init=0.08):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    lr = lr_init

    for step in range(8000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        top_shifts = np.argsort(corr)[::-1][:n_active]
        gap = corr[top_shifts[0]] - corr[top_shifts[-1]]
        if gap < 1e-8:
            effective_n_active = 3
        elif gap < 1e-5:
            effective_n_active = n_active
        else:
            effective_n_active = 2
        top_shifts = top_shifts[:effective_n_active]

        grad = np.zeros(n)
        total_weight = 0.0
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            w = corr[int(s_idx)]
            total_weight += w
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        if total_weight > 0:
            grad /= total_weight
        grad *= 2.0 * dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c_try = compute_c5(h_try)
            if c_try < c5_now - 1e-15:
                h = h_try
                lr = lr_try
                improved = True
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def random_binary_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    h = project_feasible(h, n / 2.0)
    return h


def structured_start(n, pattern_type, seed_val=0):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    if pattern_type == 'blocks3':
        third = n // 3
        h[:third] = 1.0
        idx_mid = rng.choice(third, size=third // 2, replace=False)
        h[third + idx_mid] = 1.0
        h[third * 2:third * 2 + third // 2] = 1.0
    elif pattern_type == 'blocks4':
        quarter = n // 4
        h[:quarter] = 1.0
        h[quarter * 2:quarter * 3] = 1.0
    elif pattern_type == 'sine_squared':
        x = np.linspace(0, 2 * np.pi, n)
        h = 0.5 + 0.5 * np.sin(x) ** 2
        h = project_feasible(h, n / 2.0)
    elif pattern_type == 'double_sine':
        x = np.linspace(0, 4 * np.pi, n)
        h = 0.5 + 0.4 * np.sin(x) * 0.3 + 0.3 * np.sin(3 * x) * 0.2
        h = project_feasible(h, n / 2.0)
    else:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
        h = project_feasible(h, n / 2.0)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP on initial
    if tleft() > 60:
        h_pol = slsqp_round(best_h, n, target, [1000, 5000, 15000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Piecewise global search for new starting points
    pw_candidates = []
    if tleft() > 200:
        for m_pieces in [12, 18]:
            if tleft() < 100:
                break
            pw_time = min(80, tleft() * 0.2)
            h_pw, c5_pw = piecewise_global_search(n, m_pieces, pw_time, seed_val=seed + m_pieces)
            pw_candidates.append((h_pw, c5_pw))

    for h_pw, c5_pw in pw_candidates:
        if c5_pw < best_score:
            best_score = c5_pw
            best_h = h_pw

    # Phase 3: Enhanced pair CD (main workhorse)
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=10000,
                                       n_cand_init=35, n_cand_max=65,
                                       n_bisect=14, n_active_shifts=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Active set descent for equalization
    if tleft() > 80:
        as_time = min(80, tleft() * 0.12)
        h_as, c5_as = active_set_descent(best_h, n, target,
                                         time_limit=as_time, n_active=6, lr_init=0.06)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 5: SLSQP polish
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 60000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Higher resolution n=800
    n_hires = 800
    target_hires = n_hires / 2.0
    if tleft() > 250:
        h_hires = interpolate_to(best_h, n_hires, target_hires)
        sc_hires = compute_c5(h_hires)

        if sc_hires < best_score + 0.002 and tleft() > 100:
            h_hires = slsqp_round(h_hires, n_hires, target_hires,
                                   [500, 2000, 8000], 500, 1e-15)
            sc_hires = compute_c5(h_hires)

        if tleft() > 150:
            cd_time = min(200, tleft() * 0.3)
            h_hires2, sc_hires2 = enhanced_pair_cd(h_hires, n_hires, target_hires,
                                                   time_limit=cd_time,
                                                   max_sweeps=8000,
                                                   n_cand_init=30, n_cand_max=55,
                                                   n_bisect=13, n_active_shifts=3)
            if sc_hires2 < best_score:
                best_score = sc_hires2
                best_h = h_hires2
            elif sc_hires2 < sc_hires:
                h_hires = h_hires2
                sc_hires = sc_hires2

        if tleft() > 60:
            h_hires = slsqp_round(h_hires, n_hires, target_hires,
                                   [5000, 20000, 80000], 800, 1e-15)
            sc_hires = compute_c5(h_hires)
            if sc_hires < best_score:
                best_score = sc_hires
                best_h = h_hires

    # Phase 7: Another CD round at best resolution
    if tleft() > 120:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(150, tleft() * 0.25)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=6000,
                                       n_cand_init=45, n_cand_max=70,
                                       n_bisect=15, n_active_shifts=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Active set + SLSQP
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        as_time = min(50, tleft() * 0.12)
        h_as, c5_as = active_set_descent(best_h, n_use, target_use,
                                         time_limit=as_time, n_active=8, lr_init=0.04)
        if c5_as < best_score:
            best_score = c5_as
            best_h = h_as

    # Phase 9: Multi-start with structured and piecewise inits
    rng = np.random.RandomState(seed)
    starts_tried = 0
    for init_type in ['blocks3', 'blocks4', 'sine_squared', 'random_binary']:
        if tleft() < 80:
            break
        starts_tried += 1
        if init_type == 'random_binary':
            h_str = random_binary_start(n, seed + starts_tried * 1000)
        else:
            h_str = structured_start(n, init_type, seed + starts_tried)
        
        h_str = project_feasible(h_str, n / 2.0)
        sc_str = compute_c5(h_str)
        
        if sc_str < best_score + 0.01:
            cd_t = min(60, tleft() * 0.15)
            h_try, c5_try = enhanced_pair_cd(h_str, n, n / 2.0,
                                             time_limit=cd_t,
                                             max_sweeps=2000,
                                             n_cand_init=30, n_cand_max=50,
                                             n_bisect=12, n_active_shifts=3)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try

    # Phase 10: Try n=1000 if time allows
    if tleft() > 250:
        n_try = 1000
        target_try = n_try / 2.0
        h_try = interpolate_to(best_h, n_try, target_try)
        if tleft() > 150:
            h_try = slsqp_round(h_try, n_try, target_try,
                                 [500, 3000, 10000], 500, 1e-15)
        sc_try = compute_c5(h_try)
        if tleft() > 120 and sc_try < best_score + 0.001:
            h_try2, sc_try2 = enhanced_pair_cd(h_try, n_try, target_try,
                                               time_limit=min(100, tleft() * 0.3),
                                               max_sweeps=4000,
                                               n_cand_init=30, n_cand_max=50,
                                               n_bisect=12, n_active_shifts=3)
            if sc_try2 < best_score:
                best_score = sc_try2
                best_h = h_try2

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000, 150000], 1000, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
