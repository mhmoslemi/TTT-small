def project_box_sum(h, target):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:

        else:

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def swap_delta(h, i, j, N):
    """Compute exact delta in correlation for swap h[i]:1->0, h[j]:0->1"""
    offset = N - 1
    delta = np.zeros(2 * N - 1)
    
    # i: 1->0, h-side contribution removed
    m_lo = max(0, offset - i)
    m_hi = min(2 * N - 2, offset - i + N - 1)
    if m_lo <= m_hi:
        idx = i + np.arange(m_lo, m_hi + 1) - offset
        delta[m_lo:m_hi + 1] -= (1.0 - h[idx])
    
    # i: 1->0, (1-h)-side contribution added (was 0, now 1)
    m_lo2 = max(0, i)
    m_hi2 = min(2 * N - 2, i + N - 1)
    if m_lo2 <= m_hi2:
        idx2 = i - np.arange(m_lo2, m_hi2 + 1) + offset
        delta[m_lo2:m_hi2 + 1] += h[idx2]
    
    # j: 0->1, h-side contribution added (was 0, now 1)
    m_lo3 = max(0, offset - j)
    m_hi3 = min(2 * N - 2, offset - j + N - 1)
    if m_lo3 <= m_hi3:
        idx3 = j + np.arange(m_lo3, m_hi3 + 1) - offset
        delta[m_lo3:m_hi3 + 1] += (1.0 - h[idx3])
    
    # j: 0->1, (1-h)-side contribution removed (was 1, now 0)
    m_lo4 = max(0, j)
    m_hi4 = min(2 * N - 2, j + N - 1)
    if m_lo4 <= m_hi4:
        idx4 = j - np.arange(m_lo4, m_hi4 + 1) + offset
        delta[m_lo4:m_hi4 + 1] -= h[idx4]
    
    return delta

    iteration = 0
    while time.time() - start_time < time_limit:

        worst_idx = int(np.argmax(corr))
        worst_k = worst_idx - offset
        
        # Generate candidates: focus on the worst shift

        # j should have h[j]=0 and h[j+worst_k]=1 (or j-worst_k in range with h[j-worst_k]=1)

            j_cands.extend([x for x in random_j if x not in j_cands])

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            iteration += 1
        else:
            stall += 1

```python
def batch_swap_deltas(h, i_list, j_list, N):
    """Compute deltas for multiple candidate swaps (i_k, j_k) at once"""
    offset = N - 1
    comp = 1.0 - h
    n_cands = len(i_list)
    # Delta shape: (n_cands, 2N-1)
    # For each candidate, delta[m] = -comp[i+m-offset] + comp[j+m-offset] + h[i-m+offset] - h[j-m+offset]
    
    # This is hard to vectorize nicely because of the bounds.
    # Let me just do it in a loop for each candidate.
    deltas = np.zeros((n_cands, 2*N-1))
    for k in range(n_cands):
        i = i_list[k]
        j = j_list[k]
        # i: 1->0, h-side
        m_lo = max(0, offset - i)
        m_hi = min(2*N-2, offset - i + N - 1)
        if m_lo <= m_hi:
            idx = i + np.arange(m_lo, m_hi+1) - offset
            deltas[k, m_lo:m_hi+1] -= comp[idx]
        # i: 1->0, (1-h)-side
        m_lo2 = max(0, i)
        m_hi2 = min(2*N-2, i + N - 1)
        if m_lo2 <= m_hi2:
            idx2 = i - np.arange(m_lo2, m_hi2+1) + offset
            deltas[k, m_lo2:m_hi2+1] += h[idx2]
        # j: 0->1, h-side
        m_lo3 = max(0, offset - j)
        m_hi3 = min(2*N-2, offset - j + N - 1)
        if m_lo3 <= m_hi3:
            idx3 = j + np.arange(m_lo3, m_hi3+1) - offset
            deltas[k, m_lo3:m_hi3+1] += comp[idx3]
        # j: 0->1, (1-h)-side
        m_lo4 = max(0, j)
        m_hi4 = min(2*N-2, j + N - 1)
        if m_lo4 <= m_hi4:
            idx4 = j - np.arange(m_lo4, m_hi4+1) + offset
            deltas[k, m_lo4:m_hi4+1] -= h[idx4]
    return deltas
```
