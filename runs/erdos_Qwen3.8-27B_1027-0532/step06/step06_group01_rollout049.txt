    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit and time.time() - start_t > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj_fn(x, alpha, n), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def discrete_ratchet(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = np.rint(h_init)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    while time.time() - start < time_limit:
        if time.time() - start > time_limit - 2.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        ov = corr * dx
        cur_max = float(np.max(ov))
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        k_star = int(np.argmax(ov)) - (n - 1)
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star])

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cands = ones[np.argsort(-grad[ones])[:20]]
        j_cands = zeros[np.argsort(grad[zeros])[:20]]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        for i in i_cands:
            for j in j_cands:
                if int(i) == int(j):
                    continue
                ks = np.arange(-(n - 1), n)
                g = np.zeros(2 * n - 1)
                idx_ik = int(i) - ks
                valid_ik = (idx_ik >= 0) & (idx_ik < n)
                idx_iak = int(i) + ks
                valid_iak = (idx_iak >= 0) & (idx_iak < n)
                idx_jk = int(j) - ks
                valid_jk = (idx_jk >= 0) & (idx_jk < n)
                idx_jak = int(j) + ks
                valid_jak = (idx_jak >= 0) & (idx_jak < n)
                if np.any(valid_ik):
                    g[valid_ik] += h[idx_ik[valid_ik]]
                if np.any(valid_iak):
                    g[valid_iak] += h[idx_iak[valid_iak]]
                if np.any(valid_jk):
                    g[valid_jk] -= h[idx_jk[valid_jk]]
                if np.any(valid_jak):
                    g[valid_jak] -= h[idx_jak[valid_jak]]
                new_ov = ov + dx * g
                new_max = float(np.max(new_ov))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            h = project_feasible(h, target)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP cutting plane (new approach)
    if tleft() > 200:
        lp_time = min(300, tleft() * 0.35)
        h_lp, sc_lp = lp_cutting_plane(best_h, n, target, time_limit=lp_time, M=100)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 2: Pair CD with weighted multi-shift gradient
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, sc_cd = pair_cd_weighted(best_h, n, target, time_limit=cd_time, n_cand=30, n_active=5)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [2000, 10000, 50000], 2000, 1e-15, time_limit=min(60, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Second LP round
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.3)
        h_lp, sc_lp = lp_cutting_plane(best_h, n, target, time_limit=lp_time, M=150)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 5: Pair CD again
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.35)
        h_cd, sc_cd = pair_cd_weighted(best_h, n, target, time_limit=cd_time, n_cand=40, n_active=7)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: Discrete ratchet
    if tleft() > 60:
        h_dr, sc_dr = discrete_ratchet(best_h, n, target, time_limit=min(40, tleft() * 0.4))
        if sc_dr < best_score:
            best_score = sc_dr
            best_h = h_dr

    # Phase 7: SLSQP with high alpha
    if tleft() > 50:
        hp, sc = slsqp_round(best_h, n, target, [50000, 200000], 1500, 1e-15, time_limit=min(40, tleft() * 0.6))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Try higher resolution n=1200
    if tleft() > 300:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        # LP at higher resolution
        hp, sc = lp_cutting_plane(h_interp, n_new, target_new, time_limit=min(120, tleft() * 0.3), M=80)
        if tleft() > 100:
            h_cd, sc_cd = pair_cd_weighted(hp, n_new, target_new, time_limit=min(100, tleft() * 0.3), n_cand=25, n_active=5)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 9: Final LP
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, sc_lp = lp_cutting_plane(best_h, n_use, target_use, time_limit=min(60, tleft() * 0.6), M=100)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp
            n = n_use
            target = target_use

    # Phase 10: Final pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, sc_cd = pair_cd_weighted(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.7), n_cand=20, n_active=5)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            n = n_use
            target = target_use

    # Phase 11: Final SLSQP
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [100000, 500000], 500, 1e-15, time_limit=min(20, tleft() * 0.8))
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
