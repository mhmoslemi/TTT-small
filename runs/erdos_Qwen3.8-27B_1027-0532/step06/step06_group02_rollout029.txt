    k_offset = n - 1

    dx = 2.0 / n

    best_c5 = current_max * dx
    best_h = h.copy()

    src = precompute_src(h)

        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
            
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        top_k = 8
        top_lags = np.argsort(corr)[::-1][:top_k]

        best_j = -1
        best_new_max = current_max + 1e-10

            j_cands = zeros[np.argsort(-src[zeros, k_idx])[:min(15, len(zeros))]]

                # For each j, new_corr = base - src[j]
                # new_max = max(base - src[j])
                new_maxs = np.max(base[None, :] - src[j_cands, :], axis=1)
                j_local = np.argmin(new_maxs)
                j = j_cands[j_local]
                nm = new_maxs[j_local]
                if nm < best_new_max:
                    best_new_max = nm
                    best_i = i
                    best_j = j
        
        if best_i < 0:

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

With M=10 top i's and n_zeros=300: each i requires computing a (300, 1199) matrix and taking max. That's 300*1199 ≈ 360K operations per i, 3.6M total. At ~1ns per op (numpy), that's ~3.6ms per iteration. So ~280 iterations per second.

Let me also think about a different approach: **minimax optimization**.

The problem can be viewed as: minimize max_k (corr_k) subject to the constraint.

A Lagrangian/soft approach: minimize sum_k (corr_k - t)^+ for some threshold t, or minimize the smooth approximation (1/alpha) * log(sum exp(alpha * corr_k)).

The gradient of this with respect to h[i] is:
d/dh[i] of (1/alpha) * log(sum_k exp(alpha * corr_k))
= sum_k exp(alpha*(corr_k - max_k)) * d(corr_k)/dh[i] / sum_k exp(alpha*(corr_k - max_k))
= weighted average of d(corr_k)/dh[i] over k, weighted by exp(alpha*(corr_k-max_k))

And d(corr_k)/dh[i] = (1-h[i+k]) + (1-h[i-k]) for binary h (with boundary handling).
Wait, for general h: d(corr_k)/dh[i] = (1-h[i+k]) + h[i-k]... 

Actually, corr_k = sum_j h[j]*(1-h[j+k]). 
d(corr_k)/dh[i] = (1-h[i+k]) + sum over j where j+k=i of (1-h[j+k])*(-1)*dh[i]/dh[i]... 

Hmm, this is getting complicated. Let me just use the smooth approach with the gradient:

corr_k = sum_j h[j]*(1-h[j+k])
d(corr_k)/dh[i] = (1-h[i+k]) [from the term j=i] - h[i-k] [from the term j=i-k, since d(1-h[i])/dh[i] = -1, so h[i-k]*(-1)]

Wait: corr_k = sum_j h[j]*(1-h[j+k])
- Term j=i: contribution is h[i]*(1-h[i+k]), derivative wrt h[i] is (1-h[i+k])
- Term j where j+k=i, i.e., j=i-k: contribution is h[i-k]*(1-h[i]), derivative wrt h[i] is -h[i-k]

So d(corr_k)/dh[i] = (1-h[i+k]) - h[i-k] (with boundary: h[x]=0 if x outside [0,n))

This is exactly src[i, k] that we already computed! (Well, src[i,k] = -(1-h[i+k]) + h[i-k] = -((1-h[i+k]) - h[i-k]) = -d(corr_k)/dh[i])

So the gradient of the smooth objective is:
grad[i] = sum_k w_k * d(corr_k)/dh[i] = -sum_k w_k * src[i,k]

where w_k = exp(alpha*(corr_k - max_k)) / sum_j exp(alpha*(corr_j - max_j))

This is just a matrix-vector product: w @ src, negated. And w is a (2n-1,) vector, src is (n, 2n-1), so the gradient is (n,).

This is very fast! The smooth gradient computation is O(n*2n) = O(n^2) per step. With n=600, that's 720K operations, ~1ms.

So we can do gradient-based optimization at ~1000 steps per second.

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fast(h)
    return float(np.max(corr)) * dx

def precompute_src(h):
    n = len(h)
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)
    src = np.zeros((n, n_corr))
    cols = np.arange(n_corr)
    for p in range(n):
        pk = p + k_vals
        valid_pk = (pk >= 0) & (pk < n)
        src[p, valid_pk] -= (1.0 - h[pk[valid_pk]])
        pmk = p - k_vals
        valid_pmk = (pmk >= 0) & (pmk < n)
        src[p, valid_pmk] += h[pmk[valid_pmk]]
    return src

def update_src_after_flip(src, i, j, k_vals, n):
    """Update src after flipping h[i]: 1->0 and h[j]: 0->1"""
    n_corr = 2 * n - 1
    cols = np.arange(n_corr)
    
    # For i (1->0): src decreases by 1 where p+k=i or p-k=i
    rows_ik = i - k_vals
    v1 = (rows_ik >= 0) & (rows_ik < n)
    if v1.any():
        src[rows_ik[v1], cols[v1]] -= 1.0
    
    rows_ipk = i + k_vals
    v2 = (rows_ipk >= 0) & (rows_ipk < n)
    if v2.any():
        src[rows_ipk[v2], cols[v2]] -= 1.0
    
    # For j (0->1): src increases by 1 where p+k=j or p-k=j
    rows_jk = j - k_vals
    v3 = (rows_jk >= 0) & (rows_jk < n)
    if v3.any():
        src[rows_jk[v3], cols[v3]] += 1.0
    
    rows_jpk = j + k_vals
    v4 = (rows_jpk >= 0) & (rows_jpk < n)
    if v4.any():
        src[rows_jpk[v4], cols[v4]] += 1.0

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_fast(h, max_iters, rng, time_limit, start_time):
    """Fast greedy using precomputed src matrix."""
    n = len(h)
    h = h.copy().astype(float)
    n_corr = 2*n - 1
    k_vals = np.arange(n_corr) - (n-1)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    current_max = float(corr.max())
    best_c5 = current_max * dx
    best_h = h.copy()
    src = precompute_src(h)
    
    stall = 0
    max_stall = 300
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Find best move
        # Strategy: for top bottleneck lags, find best (i,j)
        n_top_lags = 6
        top_lag_indices = np.argsort(corr)[::-1][:n_top_lags]
        
        best_new_max = current_max + 0.5
        best_i = -1
        best_j = -1
        
        # For each bottleneck lag, find candidates
        for k_idx in top_lag_indices:
            # At lag k_idx, delta = src[j, k_idx] - src[i, k_idx]
            # Want to maximize this: minimize src[i, k_idx] for i in ones, maximize src[j, k_idx] for j in zeros
            n_cand = min(20, len(ones), len(zeros))
            i_cands = ones[np.argsort(src[ones, k_idx])[:n_cand]]
            j_cands = zeros[np.argsort(-src[zeros, k_idx])[:n_cand]]
            
            for i in i_cands:
                base = corr + src[i]
                # For each j in j_cands: new_max = max(base - src[j])
                new_maxs = np.max(base[None, :] - src[j_cands, :], axis=1)
                j_best = j_cands[np.argmin(new_maxs)]
                nm = float(new_maxs.min())
                if nm < best_new_max:
                    best_new_max = nm
                    best_i = i
                    best_j = j_best
        
        if best_i < 0 or best_new_max >= current_max:
            # Try random moves
            found = False
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                nm = float(np.max(corr + src[i] - src[j]))
                if nm < current_max:
                    best_i = i
                    best_j = j
                    best_new_max = nm
                    found = True
                    break
            if not found:
                stall += 1
                continue
        
        # Apply move
        delta = src[best_i] - src[best_j]
        corr += delta
        h[best_i] = 0.0
        h[best_j] = 1.0
        current_max = best_new_max
        stall = 0
        update_src_after_flip(src, best_i, best_j, k_vals, n)
        
        if current_max * dx < best_c5:
            best_c5 = current_max * dx
            best_h = h.copy()
    
    return best_h, best_c5

def sa_fast(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    """Fast SA using precomputed src matrix."""
    n = len(h)
    h = h.copy().astype(float)
    n_corr = 2*n - 1
    k_vals = np.arange(n_corr) - (n-1)
    corr = full_corr_fast(h)
    dx = 2.0 / n
    current_max = float(corr.max())
    current_c5 = current_max * dx
    best_c5 = current_c5
    best_h = h.copy()
    src = precompute_src(h)
    
    stall = 0
    max_stall = 50000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        if it % 5 == 0:
            # Focused move: target worst lag
            worst_k = int(np.argmax(corr))
            k_star = worst_k - (n - 1)
            if k_star == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                # Find i with smallest src[i, worst_k] among ones
                # and j with largest src[j, worst_k] among zeros
                i_scores = src[ones, worst_k]
                j_scores = src[zeros, worst_k]
                n_pick = min(10, len(ones))
                i = ones[np.argsort(i_scores)[:n_pick][rng.randint(n_pick)]]
                j = zeros[np.argsort(-j_scores)[:n_pick][rng.randint(n_pick)]]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        delta = src[i] - src[j]
        new_max = float(np.max(corr + delta))
        delta_c5 = (new_max - current_max) * dx
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            corr += delta
            h[i] = 0.0
            h[j] = 1.0
            current_max = new_max
            update_src_after_flip(src, i, j, k_vals, n)
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def gradient_refine(h0, target, start_time, time_limit):
    """Smooth gradient-based refinement."""
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    n_corr = 2*n - 1
    
    alphas = [5, 20, 80, 300, 1000, 4000, 12000, 30000]
    for alpha in alphas:
        remaining = time_limit - (time.time() - start_time)
        if remaining < 3:
            break
        phase_time = min(remaining * 0.5, 60.0)
        lr = 1.0 / (alpha * 0.1 + 1.0)
        lr = min(lr, 0.5)
        momentum = 0.8
        v = np.zeros(n)
        
        n_iter = int(phase_time * 800)
        n_iter = max(n_iter, 30)
        
        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            max_c = float(corr.max())
            exp_c = np.exp(alpha * (corr - max_c))
            w = exp_c / exp_c.sum()
            # gradient[i] = sum_k w[k] * d(corr_k)/dh[i] = -sum_k w[k] * src[i,k]
            # d(corr_k)/dh[i] = (1-h[i+k]) - h[i-k] = -src[i,k]
            # So gradient = w @ src (negative of what we want for minimization)
            # We want to decrease corr, so we go in direction -gradient = src
            # Wait: we want to MINIMIZE the objective, so we go in direction of negative gradient
            # gradient of objective wrt h[i] = sum_k w_k * (1-h[i+k] - h[i-k])
            # = sum_k w_k * (-src[i,k]) = -(w @ src)[i]
            # So update: h -= lr * gradient = h += lr * (w @ src)
            
            grad = -(w @ src)  # This is the gradient of the objective
            # Wait, let me re-derive.
            # objective = (1/alpha) * log(sum exp(alpha * corr_k))
            # d(objective)/dh[i] = sum_k (alpha * exp(alpha*corr_k) / sum(exp(alpha*corr_j))) * d(corr_k)/dh[i]
            # = sum_k w_k * d(corr_k)/dh[i]
            # where w_k = exp(alpha*corr_k) / sum(exp(alpha*corr_j))
            # d(corr_k)/dh[i] = (1-h[i+k]) - h[i-k] = -src[i,k]
            # So d(objective)/dh[i] = -sum_k w_k * src[i,k] = -(w @ src)[i]
            # For minimization: h_new = h - lr * gradient = h + lr * (w @ src)
            
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            
            if it % 20 == 0:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                
                # Recompute src if h changed significantly
                src = precompute_src(h)
    
    return best_h, best_score

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fast(h)
    return float(np.max(corr)) * dx

def precompute_src(h):
    n = len(h)
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)
    src = np.zeros((n, n_corr))
    for p in range(n):
        pk = p + k_vals
        valid_pk = (pk >= 0) & (pk < n)
        src[p, valid_pk] -= (1.0 - h[pk[valid_pk]])
        pmk = p - k_vals
        valid_pmk = (pmk >= 0) & (pmk < n)
        src[p, valid_pmk] += h[pmk[valid_pmk]]
    return src

def update_src_after_flip(src, i, j, k_vals, n):
    n_corr = 2 * n - 1
    cols = np.arange(n_corr)
    rows_ik = i - k_vals
    v1 = (rows_ik >= 0) & (rows_ik < n)
    if v1.any():
        src[rows_ik[v1], cols[v1]] -= 1.0
    rows_ipk = i + k_vals
    v2 = (rows_ipk >= 0) & (rows_ipk < n)
    if v2.any():
        src[rows_ipk[v2], cols[v2]] -= 1.0
    rows_jk = j - k_vals
    v3 = (rows_jk >= 0) & (rows_jk < n)
    if v3.any():
        src[rows_jk[v3], cols[v3]] += 1.0
    rows_jpk = j + k_vals

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())

        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1

    # Phase 6: SA again
    if tleft() > 50:
        allowed = min(150.0, tleft() * 0.15)
        dl = deadline_after(allowed)
        n_sa3 = int(allowed * 350000)
        h_sa3, c5_sa3 = sa_fast(best_h, n_sa3, 0.01, 0.00002, rng, dl)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3

    # Phase 7: Multi-start perturbations
    for restart in range(12):
        if tleft() < 35:
            break
        r = restart % 3
        if r == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(4, 10))
        elif r == 1:
            h_pert = perturb_block(best_h, rng, 15)
        else:
            h_pert = perturb_diverse(best_h, rng, rng.randint(2, 6))
            h_pert = perturb_block(h_pert, rng, 8)

        allowed = min(25.0, tleft() * 0.03)
        dl = deadline_after(allowed)
        n_sr = int(allowed * 250000)
        h_sr, c5_sr = sa_fast(h_pert, n_sr, 0.008, 0.00002, rng, dl)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr

    # Phase 8: Final gradient polish
    if tleft() > 40:
        allowed = min(80.0, tleft() * 0.25)
        dl = deadline_after(allowed)
        h_ref2, sc2 = gradient_refine(best_h, float(target), dl)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Phase 9: Final binary + greedy
    if tleft() > 25:
        h_bin2 = project_binary(best_h, target)
        allowed = min(30.0, tleft() * 0.3)
        dl = deadline_after(allowed)
        h_g3, c5_g3 = greedy_fast(h_bin2, max_iters=5000000, rng=rng, deadline=dl)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3

    # Final feasibility
    h_final = project_box_sum(best_h, float(target))
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = project_binary(best_h, target)
        final_score = compute_c5(h_final)

    return h_final, final_score, n
