    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_binary(h, target_sum):

    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(round(h_r.sum()))
    if current_sum == target_sum:

    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0
    # Fix any remaining discrepancy
    final_sum = int(round(h_r.sum()))
    if final_sum > target_sum:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:final_sum - target_sum]] = 0.0
    elif final_sum < target_sum:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:target_sum - final_sum]] = 1.0
    return h_r

def compute_gradient(h, corr_full, alpha, n_cache=None):
    """Compute gradient of (1/alpha)*log(sum exp(alpha*c_k)) w.r.t. h."""
    n = len(h)
    max_c = np.max(corr_full)
    exp_vals = np.exp(np.clip(alpha * (corr_full - max_c), -500, 0))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    
    for idx in range(len(shifts)):
        if w[idx] < 1e-14:
            continue
        k = shifts[idx]
        # Role 1: h[j] appears as h[i+k] where i=j-k, valid if 0<=j-k<n and 0<=i<n
        #   → max(0,k) <= j < min(n, n+k)
        #   Contribution: (1-h[j-k])
        r1_lo = max(0, k)
        r1_hi = min(n, n + k)
        if r1_hi > r1_lo:
            grad[r1_lo:r1_hi] += w[idx] * (1.0 - h[r1_lo - k:r1_hi - k])
        
        # Role 2: h[j] appears as h[i] where i=j, valid if 0<=j<n and 0<=j+k<n
        #   → max(0,-k) <= j < min(n, n-k)
        #   Contribution: -h[j+k]
        r2_lo = max(0, -k)
        r2_hi = min(n, n - k)
        if r2_hi > r2_lo:
            grad[r2_lo:r2_hi] -= w[idx] * h[r2_lo + k:r2_hi + k]
    
    return grad / alpha

        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta_binary(h, i, j, k_arr, n)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta_binary(h, i, j, k_arr, n)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def continuous_optimize(h0, target, start_time, time_limit, alpha_list):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    dx = 2.0 / n
    corr = full_corr_fft(h)
    best_h = h.copy()
    best_score = float(corr.max()) * dx
    v = np.zeros(n)
    total_iters_done = 0

    for alpha in alpha_list:
        if time.time() - start_time > time_limit:
            break
        phase_remaining = time_limit - (time.time() - start_time)
        if phase_remaining < 2:
            break
        lr = 3.0 / (alpha + 1.0)
        momentum = min(0.95, 0.7 + 0.01 * np.log10(alpha + 1))
        n_iter = max(20, int(phase_remaining * 300))

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fft(h)
            grad = compute_gradient(h, corr, alpha)
            v = momentum * v - lr * grad
            h_candidate = h + v
            h_candidate = project_box_sum(h_candidate, target)
            h = h_candidate
            total_iters_done += 1
            if it % 80 == 0:
                sc = float(full_corr_fft(h).max()) * dx
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()

    sc = float(full_corr_fft(h).max()) * dx
    if sc < best_score:
        best_score = sc
        best_h = h.copy()
    return best_h, best_score

def slsqp_obj(h, alpha, corr_cache):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_cons_n = 0
_cons_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    for _ in range(n_pert):
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a + blen] = h[a:a + blen][::-1]
    return h_pert

def focused_binary_sa(h0, target, rng, time_limit, start_time, k_arr, t_init=0.015, t_final=0.00003):
    n = len(h0)
    h = h0.copy().astype(float)
    full = full_corr_fft(h)
    dx = 2.0 / n
    current_c5 = float(full.max()) * dx
    best_c5 = current_c5
    best_h = h.copy()
    stall = 0
    max_stall = 40000
    positions = np.arange(n)

    for it in range(20000000):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = min(it / 300000.0, 1.0)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 5 == 0:
            top_k = 3
            top_indices = np.argsort(full)[::-1][:top_k]
            best_move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                if len(i_cands) == 0:
                    continue
                i = i_cands[rng.randint(len(i_cands))]
                delta = compute_delta_binary(h, i, zeros[rng.randint(len(zeros))], k_arr, n)
                new_max = float((full + delta).max())
                c5_new = new_max * dx
                delta_c5 = c5_new - current_c5
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    j = zeros[rng.randint(len(zeros))]
                    delta = compute_delta_binary(h, i, j, k_arr, n)
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    current_c5 = float(full.max()) * dx
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                    best_move_found = True
                    break
            if not best_move_found:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_binary(h, i, j, k_arr, n)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) // 2.0
    target_int = n // 2
    k_arr = np.arange(2 * n - 1) - (n - 1)

    best_h = project_binary(h_init, target_int).astype(float)
    best_score = compute_c5(best_h)
    best_cont_h = project_box_sum(h_init, target)
    best_cont_score = compute_c5(best_cont_h)

    # Phase 1: Binary greedy + SA (100s)
    allowed = min(40, tleft() * 0.08)
    h_g, c5_g = greedy_binary(best_h, target_int, rng, allowed, start_time, k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    allowed = min(60, tleft() * 0.1)
    h_sa, c5_sa = binary_sa(best_h, target_int, rng, allowed, start_time, k_arr,
                            t_init=0.02, t_final=0.0001, block_frac=0.25, max_block_len=12)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 2: Main continuous optimization (600s)
    if tleft() > 100:
        h_start = project_box_sum(best_h, target)
        alphas_main = [3, 5, 8, 12, 20, 35, 60, 100, 200, 400, 800, 1500, 3000, 6000, 12000, 25000]
        allowed = min(500, tleft() * 0.45)
        h_cont, c5_cont = continuous_optimize(h_start, target, start_time, allowed, alphas_main)
        if c5_cont < best_cont_score:
            best_cont_score = c5_cont
            best_cont_h = h_cont.copy()
        if c5_cont < best_score:
            best_score = c5_cont
            best_h = h_cont.copy()

    # Phase 3: Binary SA from best continuous (80s)
    if tleft() > 50:
        h_bin = project_binary(best_cont_h, target_int)
        allowed = min(70, tleft() * 0.12)
        h_bs, c5_bs = focused_binary_sa(h_bin, target_int, rng, allowed, start_time, k_arr,
                                        t_init=0.012, t_final=0.00002)
        if c5_bs < best_score:
            best_score = c5_bs
            best_h = h_bs.copy()

    # Phase 4: ILS - iterative local search with continuous + binary (350s)
    n_ils = 6
    for restart in range(n_ils):
        if tleft() < 80:
            break
        # Create perturbation
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(5, 10))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 15)
        else:
            h_pert = perturb_diverse(best_h, rng, 3)
            h_pert = perturb_block(h_pert, rng, 8)

        # Quick binary SA
        allowed_bin = min(25, tleft() * 0.06)
        h_bs2, c5_bs2 = binary_sa(h_pert, target_int, rng, allowed_bin, start_time, k_arr,
                                  t_init=0.01, t_final=0.00003, block_frac=0.2, max_block_len=8)
        if c5_bs2 < best_score:
            best_score = c5_bs2
            best_h = h_bs2.copy()

        # Continuous optimize from perturbed
        if tleft() > 40:
            h_cont_start = project_box_sum(h_bs2, target)
            alphas_ils = [10, 30, 80, 200, 500, 1500, 5000]
            allowed_cont = min(60, tleft() * 0.15)
            h_ic, c5_ic = continuous_optimize(h_cont_start, target, start_time, allowed_cont, alphas_ils)
            if c5_ic < best_score:
                best_score = c5_ic
                best_h = h_ic.copy()
            if c5_ic < best_cont_score:
                best_cont_score = c5_ic
                best_cont_h = h_ic.copy()

    # Phase 5: SLSQP refinement (120s)
    if tleft() > 60:
        h_start_sl = project_box_sum(best_h, target)
        alphas_sl = [50, 200, 800, 3000, 10000, 30000]
        allowed = min(100, tleft() * 0.2)
        h_sl, c5_sl = slsqp_refine(h_start_sl, n, target, alphas_sl, 2000, allowed, start_time)
        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl.copy()
        if c5_sl < best_cont_score:
            best_cont_score = c5_sl
            best_cont_h = h_sl.copy()

    # Phase 6: Final continuous polish with very high alpha (80s)
    if tleft() > 30:
        h_start_fin = project_box_sum(best_cont_h, target)
        alphas_fin = [5000, 15000, 50000, 150000]
        allowed = min(70, tleft() * 0.25)
        h_fin, c5_fin = continuous_optimize(h_start_fin, target, start_time, allowed, alphas_fin)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Final binary burst (40s)
    if tleft() > 25:
        h_bin_final = project_binary(best_h, target_int)
        allowed = min(35, tleft() * 0.3)
        h_fb, c5_fb = focused_binary_sa(h_bin_final, target_int, rng, allowed, start_time, k_arr,
                                        t_init=0.008, t_final=0.00001)
        if c5_fb < best_score:
            best_score = c5_fb
            best_h = h_fb.copy()

    # Final SLSQP polish
    if tleft() > 15:
        h_start_p = project_box_sum(best_h, target)
        alphas_p = [20000, 80000, 200000]
        allowed = max(5.0, tleft() - 5)
        h_p, c5_p = slsqp_refine(h_start_p, n, target, alphas_p, 1500, allowed, start_time)
        if c5_p < best_score:
            best_score = c5_p
            best_h = h_p.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) // 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)
    best_h = h_final
    best_score = final_score

    return best_h, best_score, n_final
