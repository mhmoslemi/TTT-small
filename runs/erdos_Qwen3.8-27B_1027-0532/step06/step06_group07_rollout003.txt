import numpy as np

import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):

    _global_target = target

            g[v1] += (1.0 - h[idxs[v1] - k])
            g[v2] -= h[idxs[v2] + k]
            grad += weights[np.where(near_active == s_idx)[0][0]] * g
        
        grad -= np.mean(grad)
        
        # Step size
        gamma = gamma0 / (1.0 + decay * step)
        
        # Momentum
        v = momentum * v - gamma * grad
        h_try = project_feasible(h + v, target)
        
        # Check improvement
        if compute_c5(h_try) < c5_val + 1e-10:
            h = h_try
        else:
            v *= 0.5
    
    return best_h, best_c5

def block_perturbation(h, n, target, rng, n_blocks=5, block_size=20):
    """Perturb h by adding structured noise in blocks."""
    h_try = h.copy()
    for _ in range(n_blocks):
        start_idx = int(rng.randint(0, max(1, n - block_size)))
        end_idx = start_idx + int(block_size * rng.uniform(0.5, 1.5))
        end_idx = min(end_idx, n)
        h_try[start_idx:end_idx] += rng.randn(end_idx - start_idx) * 0.05
    return project_feasible(h_try, target)
