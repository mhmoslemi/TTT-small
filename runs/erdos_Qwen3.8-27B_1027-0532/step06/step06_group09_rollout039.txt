            h[best_j] += best_delta

    return best_h, best_c5

def fast_pair_cd(h, n, target, time_limit, max_sweeps=50000, n_cand=30, patience=200):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        
        # Compute gradient at active shift
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        n_c = min(n_cand, n // 4)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-15)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    cmax = np.max(corr)
    sh = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(sh))) / alpha)

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    sh = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    w = np.exp(sh)
    w = w / w.sum()
    sig = np.zeros(n)
    mid = n - 1
    for s_idx in range(2 * n - 1):
        wi = w[s_idx]
        if wi < 1e-13:
            continue
        k = int(s_idx) - mid
        for j in range(n):
            jp = j - k
            jm = j + k
            contrib = 0.0
            if 0 <= jp < n:
                contrib += (1.0 - h[jp])
            if 0 <= jm < n:
                contrib -= h[jm]
            sig[j] += wi * contrib
    return sig * dx

                       jac=lambda x: lse_gradient(x, alpha),

                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})

        K = min(40, n_c * n_c)
        flat_scores = pair_scores.ravel()
        top_K = np.argpartition(flat_scores, K)[:K]
        top_K = top_K[flat_scores[top_K] < 1e9]

        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_c5 = c5_val

        for flat_idx in top_K:
            i_pos = int(flat_idx) // n_c
            j_pos = int(flat_idx) % n_c
            if i_pos >= n_c or j_pos >= n_c:
                continue
            dmax = delta_maxes[i_pos, j_pos]
            if dmax < 1e-8:
                continue
            g = low_sigs[j_pos] - high_sigs[i_pos]
            f_check = np.max(corr + 1e-14 * g)
            if f_check >= c_max - 1e-12:
                continue
            lo = 1e-14
            hi = dmax
            for _ in range(20):
                mid_d = 0.5 * (lo + hi)
                vals = corr + mid_d * g
                slope = g[int(np.argmax(vals))]
                if slope > 1e-14:
                    hi = mid_d
                elif slope < -1e-14:
                    lo = mid_d
                else:
                    break
            delta_star = 0.5 * (lo + hi)
            if delta_star < 1e-12:
                continue
            new_c5 = np.max(corr + delta_star * g) * dx
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_delta = delta_star
                best_i = i_pos
                best_j = j_pos

        if best_i >= 0:
            i_actual = int(high_set[best_i])
            j_actual = int(low_set[best_j])
            h[i_actual] -= best_delta
            h[j_actual] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def momentum_escape(h_init, n, target, time_limit, max_steps=10000,
                    mu=0.85, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        top_k = int(np.argmax(corr))
        k_star = top_k - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        lr = base_lr * max(0.05, 1.0 - step / max_steps)
        h_new = project_feasible(h - lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        h = h_new
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_restart(h, n, target, rng):
    choice = int(rng.randint(5))
    if choice == 0:
        h_new = h[::-1].copy()
    elif choice == 1:
        shift = int(rng.randint(1, n // 4))
        h_new = np.roll(h, shift).copy()
    elif choice == 2:
        kernel_size = int(rng.choice([5, 7, 11]))
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3.0))**2)
        kernel /= kernel.sum()
        h_new = np.convolve(h, kernel, mode='same')
    elif choice == 3:
        h_new = h + rng.randn(n) * 0.05
    else:
        block_size = int(rng.choice([10, 20, 30]))
        h_new = h.copy()
        n_blocks = n // block_size
        for bi in range(n_blocks):
            if rng.rand() < 0.3:
                start_b = bi * block_size
                end_b = start_b + block_size
                h_new[start_b:end_b] = 1.0 - h[start_b:end_b]
    return project_feasible(h_new, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Adaptive alpha L-BFGS-B
    if tleft() > 100:
        alphas = [5, 20, 80, 300, 1000, 3000, 10000, 30000, 80000]
        for alpha in alphas:
            if tleft() < 30:
                break
            h_try = lbb_optimize(best_h, n, target, alpha, maxiter=1200)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 2: Fast pair CD
    if tleft() > 200:
        cd_time = min(350, tleft() * 0.35)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=50000, n_cand=30, patience=120, K_pairs=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B polish
    if tleft() > 100:
        for alpha in [20000, 60000, 150000]:
            if tleft() < 30:
                break
            h_try = lbb_optimize(best_h, n, target, alpha, maxiter=1500)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 4: Multi-shift CD
    if tleft() > 150:
        ms_time = min(200, tleft() * 0.3)
        h_ms, c5_ms = multi_shift_cd(best_h, n, target, time_limit=ms_time,
                                      max_sweeps=30000, n_cand=25, patience=100)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 5: Second pair CD round
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                    max_sweeps=40000, n_cand=40, patience=100, K_pairs=80)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Momentum escape + structured restarts
    rng = np.random.RandomState(seed)
    if tleft() > 120:
        h_mom, c5_mom = momentum_escape(best_h, n, target,
                                         time_limit=min(40, tleft() * 0.2),
                                         max_steps=8000, mu=0.85, base_lr=0.03)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

        for _ in range(4):
            if tleft() < 80:
                break
            h_re = structured_restart(best_h, n, target, rng)
            h_pol = lbb_optimize(h_re, n, target, 5000, maxiter=800)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
                h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                            time_limit=min(30, tleft() * 0.15),
                                            max_sweeps=10000, n_cand=30,
                                            patience=80, K_pairs=40)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 7: Higher resolution
    if tleft() > 200:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbb_optimize(h_interp, n_try, target_try, 3000, maxiter=1000)
            h_hires, _ = fast_pair_cd(h_hires, n_try, target_try,
                                       time_limit=min(80, tleft() * 0.3),
                                       max_sweeps=20000, n_cand=30,
                                       patience=80, K_pairs=40)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 8: Final intensive pair CD
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        cd_time = min(80, tleft() * 0.6)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use, time_limit=cd_time,
                                    max_sweeps=30000, n_cand=20, patience=60, K_pairs=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Final L-BFGS-B
    if tleft() > 20:
        h_try = lbb_optimize(best_h, n_use, target_use, 200000, maxiter=800)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
