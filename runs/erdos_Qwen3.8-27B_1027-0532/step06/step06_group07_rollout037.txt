```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
```

For step 4, with the vectorized approach:
- We have n_cand pairs. For each pair (i,j), the change in corr is δ * (g_j - g_i).
- We want min_δ max_k (corr_k + δ * (g_j - g_i)_k)

Let me define d_{ij} = g_j - g_i (a vector of length n_corr).
Then f_{ij}(δ) = max_k (corr_k + δ * d_{ij,k})

This is a convex piecewise-linear function of δ. Its minimum is at:
- δ = 0 if the subgradient at 0 contains 0 (i.e., if all active k have d_{ij,k} >= 0 or all have d_{ij,k} <= 0... actually no)
- More precisely, at δ*, the set of active k's {k : corr_k + δ*_k = f(δ*)} must have both positive and negative d values, OR f is flat.

The bisection approach works: we bisect on [0, δ_max] to find where f stops decreasing.

For vectorization, I can compute f(δ) for all pairs simultaneously:
f_all(δ) = max_k (corr + δ * D) where D is the (n_pairs × n_corr) matrix of d_{ij}.

This is a matrix max operation, which numpy handles efficiently.

So the algorithm becomes:
1. Precompute D (n_cand^2 × n_corr matrix) - this might be too large for n_cand=30: 900 × 1199 ≈ 1M floats = 8MB. That's OK.
2. Bisect: for all pairs simultaneously.
3. Pick the best pair.

For a given pair (i, j), the function f(δ) = max_k (corr_k + δ * d_k) is a max of linear functions. The minimum of this convex PWL function occurs at a breakpoint, which is where two lines cross. The breakpoints are at δ = (corr_k - corr_l) / (d_l - d_k) for pairs of active k, l.

Finding the exact breakpoint is complex. Let me just use a fast heuristic.

**Alternative: Use the "steepest descent" direction**

Instead of pair moves, use a gradient direction:
- d = -∇f(h) projected to the feasible set
- Take a step in direction d

The gradient of f(h) = max_k g_k(h) is in the normal cone of the active set:
∇f(h) = (1/|A|) ∑_{k∈A} ∇g_k(h) where A = {k : g_k = max}

This is a single direction, and we can do a line search along it. The advantage is it's O(n) per step (since we only need the gradient at the active k's).

```python
def gradient_cd(h, n, target, time_limit, max_sweeps=5000):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break
        
        # Find active set (k values at the max)
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-10)[0]
        
        # Compute gradient: average gradient over active k's
        grad = np.zeros(n)
        for s in active:
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        # Subtract mean to satisfy the sum constraint
        grad -= np.mean(grad)
        
        # The direction to decrease f is -grad (projected to sum=0)
        # Line search: find best step in direction -grad
        # We want to minimize f(h - α * grad) for α > 0
        # With box constraints: h - α*grad must be in [0,1]^n
        
        # Max step from box constraints:
        positive_grad = grad > 1e-12
        negative_grad = grad < -1e-12
        alpha_max = 1e10
        if np.any(positive_grad):
            alpha_max = min(alpha_max, np.min(h[positive_grad] / grad[positive_grad]))
        if np.any(negative_grad):
            alpha_max = min(alpha_max, np.min((1.0 - h[negative_grad]) / (-grad[negative_grad])))
        
        # Try several steps
        best_alpha = 0.0
        best_f = c5_val
        for frac in [1.0, 0.75, 0.5, 0.25, 0.1, 0.05]:
            alpha = frac * alpha_max
            if alpha < 1e-14:
                break
            h_try = h - alpha * grad
            h_try = project_feasible(h_try, target)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            f_try = np.max(corr_try) * dx
            if f_try < best_f:
                best_f = f_try
                best_alpha = alpha
        
        if best_alpha > 0:
            h = project_feasible(h - best_alpha * grad, target)
        else:
            if no_improve > 20:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
```

```python
            grad[v2] += w * (-h[idxs[v2] + k])

        h_prev = h.copy()
        h_new = h + v
        h_new = project_feasible(h_new, target)

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        f_new = np.max(corr_new) * dx
        if f_new > c5_val + 1e-10:

                h_try = project_feasible(h_try, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')

                    h_new = h_try
                    break
            else:
                h_new = h  # No improvement possible
        
        h = h_new
    
    return best_h, best_c5
```

**Alternative: use the "active set" method more directly.**

At the current iterate, identify the set A of active k's (those achieving the max). If |A| = m, then at the optimum (assuming smoothness), we have:
- g_k(h*) = C* for all k ∈ A
- The gradient of g_k for k ∈ A are "balanced" (their convex hull contains 0 in the projected space).

The KKT conditions give us a system of equations:
- g_k(h) = C for k ∈ A (m equations)
- ∑h = n/2 (1 equation)
- h ∈ [0,1]^n (complementarity)
- ∑_{k∈A} λ_k ∇g_k(h) + μ * 1 + ν = 0 (gradient condition, where ν handles the box constraints)

This is a system of n+m+1 equations in n+m+1 unknowns (h, C, λ). We can solve it with Newton's method!

**Newton's method on the KKT system**:
1. Guess the active set A (start with the top few k's).
2. Solve the KKT system for (h, C, λ) with the current A.
3. Check if the solution is feasible (h ∈ [0,1], and no other k exceeds C).
4. If not feasible, update A and repeat.

This is the "active set method" from mathematical programming. It can be very effective when the active set is small.

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
