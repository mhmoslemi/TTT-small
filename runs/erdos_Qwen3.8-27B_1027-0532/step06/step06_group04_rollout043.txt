def solve_mccormick_lp(h0, n):
    """Solve the McCormick LP relaxation."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Variables: h (n), z (n^2 flattened), t (1)
    # z is indexed by (k, i) where k ranges from -(n-1) to (n-1)
    # and i ranges from max(0,-k) to min(n, n-k)-1 (so that i+k is in [0,n))
    
    n_k = 2 * n - 1  # number of k values
    # For each k_idx (0 to n_k-1), k = k_idx - (n-1)
    # valid i: max(0, -k) to min(n-1, n-1-k)
    
    total_z = n * n  # Σ (n-|k|) over all k = n²
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    z_var = cp.Variable(total_z)
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        z_var >= 0
    ]
    
    z_idx = 0
    k_offsets = []
    k_lengths = []
    for k_idx in range(n_k):
        k = k_idx - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)  # exclusive
        length = i_end - i_start
        k_offsets.append(z_idx)
        k_lengths.append(length)
        
        z_slice = z_var[z_idx:z_idx + length]
        
        # R_k = sum_i z_{i,k} * dx >= t
        constraints.append(cp.sum(z_slice) * dx >= t_var)
        
        # McCormick: z <= h_i, z <= h_{i+k}, z >= h_i + h_{i+k} - 1
        h_i = h_var[i_start:i_end]
        h_ik = h_var[i_start + k:i_end + k]
        constraints.append(z_slice <= h_i)
        constraints.append(z_slice <= h_ik)
        constraints.append(z_slice >= h_i + h_ik - 1)
        
        z_idx += length
    
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    prob.solve(solver=cp.SCS, eps=1e-9, max_iters=200000)
    
    return t_var.value, h_var.value

For the Erdős problem, the optimal function (in the continuous case) is known to be a "balanced" function. The key insight is that the optimal h should have its autocorrelation as flat as possible.

A flat autocorrelation means that the Fourier transform of h has a flat magnitude spectrum (like a "thumbtack" spectrum in radar terminology). This is related to the concept of a "Barker sequence" but in continuous form.

So the ideal h would be one whose Fourier coefficients have equal magnitude: |c_k| = constant for all k (within the bandwidth).

For a function on [0,2] with ∫h = 1, the DC component is c_0 = 1/2 (since ∫h dx = 2*c_0 = 1, so c_0 = 1/2).

If all other coefficients have magnitude c, then by Parseval:
∫|h|² dx = 2(c_0² + Σ_{k≠0} c_k²) = 2(1/4 + M*c²)

where M is the number of nonzero coefficients.

For h ∈ [0,1], we have ∫h² ≤ ∫h = 1, so 2(1/4 + M*c²) ≤ 1, giving M*c² ≤ 1/4, so c ≤ 1/√(4M).

The autocorrelation R(k) = 2 Σ |c_j|² cos(πjk) = 2(1/4 + c² Σ_{j≠0} cos(πjk))

For the "flat spectrum" case: R(k) = 2(1/4 + c² Σ_{j≠0} cos(πjk))

The sum Σ_{j=1}^{M} cos(πjk) for k ≠ 0 (mod 2M) is roughly -1/2 (by the Dirichlet kernel), and for k = 0 it's M.

So R(0) ≈ 2(1/4 + M*c²) and R(k≠0) ≈ 2(1/4 - c²/2) = 1/2 - c².

For the flat autocorrelation, we want R(k) ≈ constant for all k. But R(0) = ∫h² dx which is always ≥ (∫h)²/n = 1/n (by Cauchy-Schwarz, in the discrete case). For the continuous case, R(0) = ∫h² dx.

Actually, in the continuous case with period 2:

R(k) = ∫₀² h(x)h(x+k) dx (for integer k, by periodicity)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, target, alpha, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = corr * dx
    B_max = np.max(B)
    shifted = np.clip(alpha * (B - B_max), -500.0, 0.0)
    lse = B_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = mu * (np.sum(h) - target) ** 2
    return lse + penalty

def lse_gradient(h, target, alpha, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = corr * dx
    B_max = np.max(B)
    shifted = np.clip(alpha * (B - B_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Only use top-k for efficiency
    n_active = 80
    top_idx = np.argsort(B)[-n_active:]
    # Recompute weights for just these
    top_B = B[top_idx]
    top_shifted = np.clip(alpha * (top_B - B_max), -500.0, 0.0)
    top_exp = np.exp(top_shifted)
    top_weights = top_exp / np.sum(top_exp)
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for j, k_idx in enumerate(top_idx):
        k = k_idx - (n - 1)
        w = top_weights[j]
        valid_plus = (idxs + k >= 0) & (idxs + k < n)
        valid_minus = (idxs - k >= 0) & (idxs - k < n)
        grad[valid_plus] += w * (1.0 - h[idxs[valid_plus] + k])
        grad[valid_minus] -= w * h[idxs[valid_minus] - k]
    grad *= dx
    
    grad_pen = 2.0 * mu * (np.sum(h) - target) * np.ones(n)
    return grad + grad_pen

def coord_descent_full(h, target, max_sweeps, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    
    start = time.time()
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        for i in range(n):
            # Compute a_k and b_k for all k
            # B_k(h) = sum_j h[j]*(1-h[j+k])*dx (with boundary handling)
            # As function of h_i: B_k = a_k * h_i + b_k
            # a_k = dx * [(1-h_{i+k}) if 0<=i+k<n else 0] - dx * [h_{i-k} if 0<=i-k<n else 0]
            # b_k = B_k(h) - a_k * h_i
            
            k_vals = np.arange(-half_n, half_n + 1)
            valid_plus = (idxs[i] + k_vals >= 0) & (idxs[i] + k_vals < n)
            valid_minus = (idxs[i] - k_vals >= 0) & (idxs[i] - k_vals < n)
            
            a_k = np.zeros(len(k_vals))
            a_k[valid_plus] += (1.0 - h[i + k_vals[valid_plus]]) * dx
            a_k[valid_minus] -= h[i - k_vals[valid_minus]] * dx
            
            # Compute current B_k
            corr = np.correlate(h, 1.0 - h, mode='full')
            B_k = corr * dx  # B_k for k = -(n-1), ..., n-1
            # corr has length 2n-1, indexed 0 to 2n-2
            # k = index - (n-1)
            
            b_k = B_k - a_k * h[i]
            
            # Find min of max_k (a_k * t + b_k) over t in [0,1]
            # This is a 1D convex optimization
            # Use bisection on the subgradient
            
            # The function f(t) = max_k (a_k*t + b_k) is convex
            # f'(t) = max_{k: a_k*t+b_k = f(t)} a_k (subgradient)
            # We want f'(t) = 0, i.e., the max is achieved by lines with both pos and neg slopes
            
            # Efficient approach: find the upper envelope
            # For large n, use the fact that f(t) = max(a_k*t + b_k)
            # Evaluate at t=0, t=0.5, t=1 to get bounds, then refine
            
            # Simple approach: ternary search (since f is convex)
            # But binary search on subgradient is better
            
            t_lo = 0.0
            t_hi = 1.0
            for _ in range(40):
                t_mid = 0.5 * (t_lo + t_hi)
                vals = a_k * t_mid + b_k
                max_val = np.max(vals)
                # Active constraints (within tolerance)
                active = np.abs(vals - max_val) < 1e-10 * max(1.0, abs(max_val))
                if np.all(a_k[active] > 1e-12):
                    t_lo = t_mid  # slope positive, move right
                elif np.all(a_k[active] < -1e-12):
                    t_hi = t_mid  # slope negative, move left
                else:
                    break  # 0 is in the subgradient
            
            new_val = 0.5 * (t_lo + t_hi)
            h[i] = new_val
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_full(h_init, target, max_sweeps, time_limit):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    k_vals = np.arange(-half_n, half_n + 1)
    start = time.time()
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        for i in range(n):
            valid_plus = (i + k_vals >= 0) & (i + k_vals < n)
            valid_minus = (i - k_vals >= 0) & (i - k_vals < n)
            a_k = np.zeros(len(k_vals))
            a_k[valid_plus] += (1.0 - h[i + k_vals[valid_plus]])
            a_k[valid_minus] -= h[i - k_vals[valid_minus]]
            a_k *= dx
            corr = np.correlate(h, 1.0 - h, mode='full')
            B_k = corr * dx
            b_k = B_k - a_k * h[i]
            t_lo = 0.0
            t_hi = 1.0
            for _ in range(50):
                t_mid = 0.5 * (t_lo + t_hi)
                vals = a_k * t_mid + b_k
                max_val = np.max(vals)
                tol = 1e-12 * max(1.0, abs(max_val))
                active = np.abs(vals - max_val) <= tol
                if not np.any(active):
                    active = np.zeros(len(k_vals), dtype=bool)
                    active[np.argmax(vals)] = True
                if np.all(a_k[active] > 1e-14):
                    t_lo = t_mid
                elif np.all(a_k[active] < -1e-14):
                    t_hi = t_mid
                else:
                    break
            h[i] = 0.5 * (t_lo + t_hi)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def lse_vec(h, target, alpha, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = corr * dx
    B_max = np.max(B)
    shifted = np.clip(alpha * (B - B_max), -500.0, 0.0)
    lse = B_max + np.log(np.sum(np.exp(shifted))) / alpha
    pen = mu * (np.sum(h) - target) ** 2
    return lse + pen

def lse_grad(h, target, alpha, mu, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = corr * dx
    B_max = np.max(B)
    n_active = 100
    top_idx = np.argsort(B)[-n_active:]
    top_B = B[top_idx]
    top_shifted = np.clip(alpha * (top_B - B_max), -500.0, 0.0)
    top_exp = np.exp(top_shifted)
    top_w = top_exp / np.sum(top_exp)
    half_n = n - 1
    k_vals = np.arange(-half_n, half_n + 1)
    grad = np.zeros(n)
    for j, k_idx in enumerate(top_idx):
        k = k_idx - half_n
        w = top_w[j]
        idxs = np.arange(n)
        vp = (idxs + k >= 0) & (idxs + k < n)
        vm = (idxs - k >= 0) & (idxs - k < n)
        grad[vp] += w * (1.0 - h[idxs[vp] + k])
        grad[vm] -= w * h[idxs[vm] - k]
    grad *= dx
    grad_pen = 2.0 * mu * (np.sum(h) - target) * np.ones(n)
    return grad + grad_pen

def lbfgs_optimize(h0, target, time_limit, alphas=None):
    n = len(h0)
    if alphas is None:
        alphas = [50.0, 100.0, 300.0, 1000.0, 3000.0]
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    start = time.time()
    mu = 100.0
    for alpha in alphas:
        remaining = time_limit - (time.time() - start)
        if remaining < 10:
            break
        maxiter = min(300, int(remaining * 8))
        if maxiter < 20:
            break
        try:
            res = minimize(
                lambda x: lse_vec(x, target, alpha, mu, n),
                lambda x: lse_grad(x, target, alpha, mu, n),
                h_cur, method='L-BFGS-B', jac=True,
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
        mu = max(mu * 0.5, 1.0)
    return best_h, best_score

def slsqp_optimize(h0, target, time_limit, alphas=None, maxiter=2000):
    n = len(h0)
    if alphas is None:
        alphas = [200.0, 1000.0, 5000.0]
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        remaining = time_limit - (time.time() - start)
        if remaining < 15:
            break
        mi = min(maxiter, int(remaining * 3))
        if mi < 50:
            break
        try:
            res = minimize(
                lambda x: lse_vec(x, target, alpha, 0.0, n),
                h_cur, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': mi, 'ftol': 1e-16}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def subgradient_refine(h0, target, max_steps, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    start = time.time()
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        vp = (idxs - k >= 0) & (idxs - k < n)
        vm = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[vp] += (1.0 - h_cur[idxs[vp] - k])
        grad[vm] -= h_cur[idxs[vm] + k]
        grad *= dx
        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h0, target, time_limit, n_active=80, scs_eps=1e-10):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    damping = 0.85
    half_n = n - 1
    reg = 1e-4
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_set = set(np.argsort(corr * dx)[-n_active:].tolist())
    it = 0
    while time.time() - start < time_limit:
        it += 1
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k_idx in sorted(active_set):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_set and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 3e-5:
                    active_set.add(k_idx)
                    break
        active_set = top_k | active_set
        if len(active_set) > 5 * n_active:
            active_set = set(np.argsort(corr)[-5 * n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.4, damping * 0.95)
        else:
            damping = min(0.99, damping + 0.02)
        if it % 40 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.7)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    def tleft():

    # Phase 1: Coordinate descent with full constraints (fast and effective)

        h_cd, sc = coord_descent_full(best_h, target, max_sweeps=2000, time_limit=cd_time)
        if sc < best_score:

            best_h = h_cd

    # Phase 2: L-BFGS-B with log-sum-exp smoothing

        h_lb, sc = lbfgs_optimize(best_h, target, time_limit=lb_time,
                                   alphas=[50.0, 100.0, 300.0, 1000.0, 3000.0])
        if sc < best_score:

            best_h = h_lb

    # Phase 3: More coordinate descent

        cd_time = min(120, tleft() * 0.25)
        h_cd, sc = coord_descent_full(best_h, target, max_sweeps=3000, time_limit=cd_time)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 4: Alternating LP
    if tleft() > 200:
        lp_time = min(180, tleft() * 0.28)
        h_lp, sc = alternating_lp(best_h, target, time_limit=lp_time, n_active=100, scs_eps=1e-10)

            best_h = h_lp

    # Phase 5: SLSQP with high alpha for precision
    if tleft() > 150:
        slq_time = min(120, tleft() * 0.25)
        h_sl, sc = slsqp_optimize(best_h, target, time_limit=slq_time,
                                   alphas=[500.0, 2000.0, 10000.0], maxiter=1500)

            best_h = h_sl

    # Phase 6: Coordinate descent again
    if tleft() > 100:
        cd_time = min(80, tleft() * 0.25)
        h_cd, sc = coord_descent_full(best_h, target, max_sweeps=5000, time_limit=cd_time)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 7: Try n=200 (coarser grid might capture structure better)
    if tleft() > 250:
        n2 = 200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if tleft() > 150:
            h_cd2, sc_cd = coord_descent_full(h2, target2, max_sweeps=2000, time_limit=min(80, tleft() * 0.3))
            if sc_cd < sc2:

        if tleft() > 100:

                sc2 = sc_lb
                h2 = h_lb2
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n = n2
            target = target2

    # Phase 8: Perturbation restarts
    for noise in [0.01, 0.03, 0.08, 0.15]:
        if tleft() < 100:

        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.25)
        h_cd, sc = coord_descent_full(h_pert, target_use, max_sweeps=1000, time_limit=cd_time)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 9: Final coordinate descent (long)
    if tleft() > 80:
        cd_time = min(60, tleft() * 0.4)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, sc = coord_descent_full(best_h, target_use, max_sweeps=5000, time_limit=cd_time)

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, sc = subgradient_refine(best_h, target_use, max_steps=3000, time_limit=min(25, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 11: Final coordinate descent
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, sc = coord_descent_full(best_h, target_use, max_sweeps=2000, time_limit=min(15, tleft() * 0.6))
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
