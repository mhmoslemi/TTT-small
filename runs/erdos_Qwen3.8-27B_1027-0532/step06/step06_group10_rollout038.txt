import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    per = diff / len(idx)
    h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            return smooth_obj_val(x, alpha)
        def _grad(x):
            return smooth_obj_grad(x, alpha, n)
        def _con(x):
            return np.sum(x) - target
        def _con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)

            try:

                break

        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    per = diff / len(idx)
    h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -60.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / ws
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad

def slsqp_round(h0, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def _obj(x):
            return smooth_obj_val(x, alpha)
        def _grad(x):
            return smooth_obj_grad(x, alpha, n)
        def _con(x):
            return np.sum(x) - target
        def _con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def build_effect_vectors(h, n, shifts):
    n_corr = 2 * n - 1
    v = np.zeros((n, n_corr))
    for p in range(n):
        im = (p - shifts >= 0) & (p - shifts < n)
        ip = (p + shifts >= 0) & (p + shifts < n)
        vp = np.zeros(n_corr)
        if np.any(im):
            vp[im] += h[p - shifts[im]]
        if np.any(ip):
            vp[ip] += h[p + shifts[ip]]
        v[p] = vp
    return v

def vectorized_cd(h0, n, target, time_limit, n_cand=20, max_sweeps=100000,
                  perturb_every=800, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    v = build_effect_vectors(h, n, shifts)

    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 4000:
                break
            if no_improve == perturb_every:
                n_pert = max(5, n // 50)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h[pert_idx] += rng.uniform(-0.03, 0.03, size=n_pert)
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                v = build_effect_vectors(h, n, shifts)
                no_improve = 0
                continue

        # Multi-shift gradient (top 3 shifts)
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        corr_masked = corr.copy()
        corr_masked[int(np.argmax(corr))] = -np.inf
        k_star2 = int(np.argmax(corr_masked)) - mid
        v1b = (idxs - k_star2 >= 0) & (idxs - k_star2 < n)
        v2b = (idxs + k_star2 >= 0) & (idxs + k_star2 < n)
        grad[v1b] += 0.4 * (1.0 - h[idxs[v1b] - k_star2])
        grad[v2b] -= 0.4 * h[idxs[v2b] + k_star2]

        corr_masked2 = corr.copy()
        corr_masked2[[int(np.argmax(corr)), k_star2 + mid]] = -np.inf
        k_star3 = int(np.argmax(corr_masked2)) - mid
        v1c = (idxs - k_star3 >= 0) & (idxs - k_star3 < n)
        v2c = (idxs + k_star3 >= 0) & (idxs + k_star3 < n)
        grad[v1c] += 0.2 * (1.0 - h[idxs[v1c] - k_star3])
        grad[v2c] -= 0.2 * h[idxs[v2c] + k_star3]

        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-12:
                    continue
                g = v[i] - v[j]
                # Quick rejection: if min possible max >= current best, skip
                # f(0) = max(corr) = c5_val/dx
                # If g is all >= 0, best is at delta=0
                if np.min(g) >= -1e-15:
                    continue
                a_d = 0.0
                b_d = d_max
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid * dx >= best_approx + 1e-15:
                        b_d = mid_d
                        continue
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx - 1e-15:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            for p in (best_i, best_j):
                im = (p - shifts >= 0) & (p - shifts < n)
                ip = (p + shifts >= 0) & (p + shifts < n)
                vp = np.zeros(n_corr)
                if np.any(im):
                    vp[im] += h[p - shifts[im]]
                if np.any(ip):
                    vp[ip] += h[p + shifts[ip]]
                v[p] = vp

    return best_h, best_c5

def full_lp_refine(h0, n, target, time_limit, max_rounds=20):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        # Use top 60 constraints (top 60 by correlation value)
        sorted_idx = np.argsort(corr)[::-1][:min(60, n_corr)]
        for s_idx in sorted_idx:
            k = int(s_idx) - mid
            g_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g_k[v1] += (1.0 - h[idxs[v1] - k])
            g_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + g_k @ (h_var - h)
            constraints.append(lin_val <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_iters=10000)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break

        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-12:
            h = h_lp
        else:
            break

    return best_h, best_c5

def multiplicative_weight_opt(h0, n, target, time_limit, max_steps=30000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        alpha_w = 20.0 + step * 0.3
        c_max = np.max(corr)
        w_shifts = np.exp(alpha_w * (corr - c_max))
        w_sum = w_shifts.sum()
        if w_sum > 1e-300:
            w_shifts = w_shifts / w_sum

        grad = np.zeros(n)
        active = np.where(w_shifts > 1e-10)[0]
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w_shifts[s] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w_shifts[s] * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        lr = 0.05 / (1.0 + step * 0.005)
        improved = False
        for lr_t in (lr, lr * 0.5, lr * 0.25, lr * 0.1):
            h_try = project_feasible(h - lr_t * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            h = project_feasible(h - lr * 0.05 * grad, target)

    return best_h, best_c5

def nesterov_step(h0, n, target, time_limit, max_steps=20000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nt = 1.0
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        alpha = 100.0 + step * 2.0
        grad = smooth_obj_grad(h, alpha, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nt * t_nt)) / 2.0
        beta = (t_nt - 1.0) / t_new
        t_nt = t_new
        lr = 2.5 / (1.0 + t_nt)

        improved = False
        for lr_t in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05):
            y = project_feasible(h + beta * v, target)
            h_try = project_feasible(y - lr_t * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                dist = h_try - y
                v = beta * v + dist
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    return h_out, final_score, n_final
