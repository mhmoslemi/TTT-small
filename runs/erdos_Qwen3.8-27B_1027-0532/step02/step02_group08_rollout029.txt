    g = 1.0 - h

    k = min(top_k, len(corrs))
    top_idx = np.argsort(corrs)[-k:]
    subset = corrs[top_idx]
    m_sub = np.max(subset)
    shifted = alpha * (subset - m_sub)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = m_sub + np.log(sum_exp) / alpha
    full_w = np.zeros_like(corrs)

    w_rev = full_w[::-1]

    conv2 = np.convolve(h, full_w, mode='full')
    idx = np.arange(n) + n - 1

                    best_score = s
                    best_h = h_new.copy()
        except Exception:
            pass

    best_h = project_to_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
