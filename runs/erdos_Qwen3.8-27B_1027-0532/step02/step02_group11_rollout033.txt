```python
def swap_delta(corr, h, i, j, n):
    """Compute the change in correlation array when swapping h[i] and h[j].
    i has h[i]=1, j has h[j]=0.
    Returns delta array of length 2n-1."""
    # h[i]: 1→0, h[j]: 0→1
    # For lag m (where m = k - (n-1), k = 0..2n-2):
    # corr at index k = sum_{j': 0<=j'<n, 0<=j'+m<n} h[j']*(1-h[j'+m])
    
    # Changes from h[i]: 1→0
    # As "first" (j'=i): term was 1*(1-h[i+m]), now 0. Delta = -(1-h[i+m]), valid if 0<=i+m<n
    # As "second" (j'=i-m, so j'+m=i): term was h[i-m]*(1-1)=0, now h[i-m]*(1-0)=h[i-m]. Delta = +h[i-m], valid if 0<=i-m<n
    
    # Changes from h[j]: 0→1
    # As "first" (j'=j): term was 0*(1-h[j+m])=0, now 1*(1-h[j+m]). Delta = +(1-h[j+m]), valid if 0<=j+m<n
    # As "second" (j'=j-m, so j'+m=j): term was h[j-m]*(1-0)=h[j-m], now h[j-m]*(1-1)=0. Delta = -h[j-m], valid if 0<=j-m<n
    
    # IMPORTANT: use OLD values of h
    delta = np.zeros(2*n-1)
    
    # h[i] 1→0:
    m1 = i + (n-1)  # This gives the index in the full corr array
    # "first" role: for lag m, i+m must be in [0,n). The lag index is (n-1)+m
    # i+m in [0,n) means m in [-i, n-1-i]
    # Index in corr array: (n-1)+m for m in [-i, n-1-i]
    # The value to subtract is (1-h[i+m]) for each such m
    for m in range(-i, n-i):
        k = (n-1) + m
        delta[k] -= (1 - h[i+m])
    
    # "second" role: for lag m, i-m must be in [0,n). m in [i-(n-1), i]
    for m in range(i-(n-1), i+1):
        k = (n-1) + m
        delta[k] += h[i-m]
    
    # h[j] 0→1:
    for m in range(-j, n-j):
        k = (n-1) + m
        delta[k] += (1 - h[j+m])
    
    for m in range(j-(n-1), j+1):
        k = (n-1) + m
        delta[k] -= h[j-m]
    
    return delta
```

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        np.random.shuffle(one_idx)
        for idx in one_idx[:num_ones - target]:
            h_bin[idx] = 0
    elif num_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        np.random.shuffle(zero_idx)
        for idx in zero_idx[:target - num_ones]:
            h_bin[idx] = 1
    return h_bin

def sa_solve(h, n, T_start, T_end, n_iters, rng, time_limit_fn):
    """Simulated annealing on binary solutions."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    current_max = np.max(corr)
    current_c5 = current_max * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Precompute positions
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for it in range(n_iters):
        if time_limit_fn:
            if time_limit_fn():
                break
        
        T = T_start * (T_end / T_start) ** (it / max(n_iters, 1))
        
        # Pick random i from ones, j from zeros
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Perform swap
        h[i] = 0
        h[j] = 1
        
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = np.max(new_corr)
        new_c5 = new_max * dx
        
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            current_max = new_max
            current_c5 = new_c5
            corr = new_corr
            
            # Update position lists
            ones = np.where(h == 1)[0]  # Expensive! O(n)
            zeros = np.where(h == 0)[0]
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

For m<0: let m=-p, p>0: corr[-p] = #{j in [p, n-1]: h[j]=1, h[j-p]=0} = #{j in [0, n-1-p]: h[j+p]=1, h[j]=0} = #{j in [0, n-1-p]: h[j]=0, h[j+p]=1}

= sum_{p=1}^{n-1} #{(j, j+p): h[j]≠h[j+p]}

For each pair (a,b) with a<b, let p=b-a. The pair (a,b) is counted in the p-th term if h[a]≠h[b].

So sum_{m≠0} corr[m] = #{(a,b): a<b, h[a]≠h[b]} = number of disagreeing pairs.

def hill_climb(h, n, rng, max_sweeps=500, time_limit_fn=None):
    """First-improvement hill climbing on binary solutions."""
    h = h.copy()
    dx = 2.0 / n
    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    best_c5 = current_c5

        if time_limit_fn is not None and time_limit_fn():
            break
        improved = False

        zeros_arr = np.where(h == 0)[0].copy()
        rng.shuffle(ones_arr)
        rng.shuffle(zeros_arr)
        
        for i in ones_arr:

                break
            for j in zeros_arr:
