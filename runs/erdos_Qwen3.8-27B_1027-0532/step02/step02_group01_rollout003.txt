    dx = 2.0 / n

    x = (corr - cmax) / eps
    x -= np.max(x)

    omh = 1.0 - h

        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c5

def make_perturbation(h_best, rng, n, target_sum, strat):
    if strat == 0:
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n), target_sum)
    elif strat == 1:
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)
    elif strat == 2:
        hp = h_best.copy()
        nf = max(2, n // 10)
        idx = rng.choice(n, size=nf, replace=False)
        hp[idx] = 1.0 - hp[idx]
        return project_onto_constraint(hp, target_sum)
    elif strat == 3:
        hp = h_best.copy()
        bl = int(rng.integers(3, max(4, n // 5)))
        a = int(rng.integers(0, max(1, n - bl)))
        b = int(rng.integers(0, max(1, n - bl)))
        if abs(a - b) >= bl:
            i1, i2 = min(a, b), max(a, b)
            tmp = hp[i1:i1+bl].copy()
            hp[i1:i1+bl] = hp[i2:i2+bl]
            hp[i2:i2+bl] = tmp
        return project_onto_constraint(hp, target_sum)
    elif strat == 4:
        alpha = rng.uniform(0.3, 0.7)
        hr = project_onto_constraint(0.5 + 0.3 * rng.standard_normal(n), target_sum)
        return project_onto_constraint(alpha * h_best + (1.0 - alpha) * hr, target_sum)
    else:
        hb = np.where(h_best + 0.08 * rng.standard_normal(n) > 0.5, 1.0, 0.0)
        no = int(hb.sum())
        if no > n // 2:
            oi = np.where(hb == 1)[0]
            ex = rng.choice(oi, size=no - n // 2, replace=False)
            hb[ex] = 0.0
        elif no < n // 2:
            zo = np.where(hb == 0)[0]
            hb[zo[:n // 2 - no]] = 1.0
        return project_onto_constraint(hb + 0.005 * rng.standard_normal(n), target_sum)
