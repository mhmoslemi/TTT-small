```python
def binary_local_search(h, n, num_iters, deadline):
    """Local search on binary h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    current_max = c.max()
    best_h = h.copy()
    best_max = current_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.9999
    
    for it in range(num_iters):
        if time.time() > deadline:
            break
        
        # Propose random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Compute new correlation
        h[i], h[j] = h[j], h[i]
        c_new = np.correlate(h, 1.0 - h, mode='full') * dx
        new_max = c_new.max()
        
        # Accept/reject
        if new_max <= current_max or np.random.rand() < np.exp((current_max - new_max) / T):
            current_max = new_max
            c = c_new
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
        
        T = max(T * cooling, T_min)
    
    return best_h, best_max
```

```python
def gen_mseq(m, seed=0):
    """Generate a binary m-sequence of length 2^m - 1."""
    # Use a primitive polynomial
    # For m=8: x^8 + x^4 + x^3 + x^2 + 1 (0x1D)
    # For m=9: x^9 + x^5 + x^3 + x^2 + 1 (0x27B)
    # For m=10: x^10 + x^3 + 1 (0x409)
    # For m=7: x^7 + x^3 + 1 (0x61)
    
    polys = {
        7: 0x61,   # x^7 + x^3 + 1
        8: 0x1D,   # x^8 + x^4 + x^3 + x^2 + 1
        9: 0x27B,  # x^9 + x^5 + x^3 + x^2 + 1
        10: 0x409, # x^10 + x^3 + 1
    }
    
    poly = polys[m]
    L = 2**m - 1
    # LFSR
    state = 1
    seq = np.zeros(L, dtype=np.int8)
    for i in range(L):
        seq[i] = state & 1
        # Feedback
        lsb = state & 1
        state >>= 1
        if lsb:
            state ^= (poly >> 1)  # XOR with taps (excluding the MSB)
    return seq

def circ_corr(h):
    """Circular autocorrelation via FFT."""
    n = len(h)
    H = np.fft.fft(h.astype(float))
    return np.real(np.fft.ifft(H * np.conj(H)))

def compute_c_circ(h):

    A = circ_corr(h)

def project_to_sum(h, target):
```
