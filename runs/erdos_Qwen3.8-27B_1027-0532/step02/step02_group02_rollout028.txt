def _project_affine_box(x, s):
    # Project x onto {y : sum(y) = s, 0 <= y_i <= 1}
    lo, hi = -x.max(), 1 - x.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        sm = y.sum()
        if sm > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fix tiny numerical error
    diff = s - y.sum()
    if abs(diff) < 1e-10:
        return y
    # Adjust the element with most room
    if diff > 0:
        idx = np.argmax(1 - y)
        y[idx] = min(1, y[idx] + diff)
    else:
        idx = np.argmax(y)
        y[idx] = max(0, y[idx] + diff)
    return y

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_affine_box(x, s):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    sm = x.sum()
    if abs(sm - s) < 1e-13:
        return x
    lo = -x.max() - 0.1
    hi = 1.0 - x.min() + 0.1
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = s - y.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - y)
            y[idx] = min(1.0, y[idx] + diff)
        else:
            idx = np.argmax(y)
            y[idx] = max(0.0, y[idx] + diff)
    return y

def _smooth_max_val(corr, alpha):
    m = np.max(corr)
    if alpha <= 1:
        return m
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _weight_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_vals = _corr_full(h)
    m = np.max(corr_vals)
    shifted = np.clip(alpha * (corr_vals - m), -700, 700)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            i_max = n - lag
            for i in range(i_max):
                grad[i] += wi * (1.0 - h[i + lag])
            for i in range(lag, n):
                grad[i] -= wi * h[i - lag]
        else:
            L = -lag
            i_max = n - L
            for i in range(L, n):
                grad[i] += wi * (1.0 - h[i - L])
            for i in range(i_max):
                grad[i] -= wi * h[i + L]
    return grad * dx

def _softmax_pg(h_init, n, target_sum, n_iter, alpha, deadline):
    h = h_init.copy()
    lr = 0.5
    best_c5 = _c5_of(h)
    best_h = h.copy()
    for t in range(n_iter):
        if time.time() > deadline - 10:
            break
        if t > 0 and t % 200 == 0:
            lr *= 0.97
        grad = _weight_grad(h, alpha)
        h_new = h - lr * grad
        h_new = _project_affine_box(h_new, target_sum)
        obj_new = _smooth_max_val(_corr_full(h_new), alpha)
        obj_old = _smooth_max_val(_corr_full(h), alpha)
        if obj_new > obj_old + 1e-12:
            lr *= 0.7
            h_new = h - lr * grad
            h_new = _project_affine_box(h_new, target_sum)
        h = h_new
        if t % 500 == 0 or t == n_iter - 1:
            c5 = _c5_of(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_smooth(h_init, n, target_sum, alpha, maxiter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        dx = 2.0 / n
        m = np.max(corr) * dx
        corr_dx = corr * dx
        shifted = np.clip(alpha * (corr_dx - m), -700, 700)
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _sum_constraint, 'args': (target_sum,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.success or res.fun < _smooth_max_val(_corr_full(h_init), alpha):
        return res.x
    return h_init.copy()

def _sum_constraint(x, target_sum):
    return x.sum() - target_sum

def _slsqp_smooth_fast(h_init, n, target_sum, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -700, 700)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _sum_constraint, 'args': (target_sum,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.success or _c5_of(res.x) < _c5_of(h_init):
        return res.x
    return h_init.copy()

def _upsample(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def _coordinate_descent(h, n, target_sum, n_sweeps, deadline):
    h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline - 10:
            break
        improved = False
        order = np.random.RandomState(sweep + 7).permutation(n)
        for i in order:
            if time.time() > deadline - 10:
                break
            best_val = h[i]
            best_obj = _c5_of(h)
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                cand = h.copy()
                cand[i] = frac
                cand = _project_affine_box(cand, target_sum)
                obj = _c5_of(cand)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = frac
            if best_val != h[i]:
                h[i] = best_val
                h = _project_affine_box(h, target_sum)
                improved = True
    return h

def _make_alternative_start(n, seed, stage):
    rng = np.random.RandomState(seed + stage * 104729)
    x = np.linspace(0, 2, n, endpoint=False)
    if stage == 0:

    elif stage == 1:

    elif stage == 4:

def _optimize_pipeline(h0, n, target_sum, deadline, budget_frac):
    t_deadline = deadline - budget_frac
    h = _project_affine_box(h0, target_sum)
    best_h = h.copy()
    best_c5 = _c5_of(h)

    h, c5 = _softmax_pg(h, n, target_sum, n_iter=15000, alpha=60,

    if c5 < best_c5:
        best_c5 = c5

    alphas = [80, 200, 500, 1200, 3000]

    for a in alphas:

        h_new = _slsqp_smooth_fast(h_cur, n, target_sum, a, maxiter=200)
        h_new = _project_affine_box(h_new, target_sum)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new.copy()

    return best_h, best_c5
