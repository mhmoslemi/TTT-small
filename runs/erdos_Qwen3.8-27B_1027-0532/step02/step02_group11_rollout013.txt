Valid if 0 ≤ i-d < n.

```python
def swap_delta_vector(h, i, j, n):
    """Compute delta in full correlation when swapping h[i] (1) and h[j] (0) in binary h."""
    # d = n-1-k, so for array index k, the shift is d = n-1-k
    # d ranges from n-1 (k=0) to -(n-1) (k=2n-2)
    d = np.arange(n-1, -(n), -1)  # d[k] = n-1-k
    
    # Term 1: -(1-h[i+d]) * [0 <= i+d < n]
    ip_d = i + d
    valid1 = (ip_d >= 0) & (ip_d < n)
    t1 = np.zeros(2*n-1)
    t1[valid1] = -(1 - h[ip_d[valid1]])
    
    # Term 2: (1-h[j+d]) * [0 <= j+d < n, j+d != i]
    jp_d = j + d
    valid2 = (jp_d >= 0) & (jp_d < n) & (jp_d != i)
    t2 = np.zeros(2*n-1)
    t2[valid2] = (1 - h[jp_d[valid2]])
    # Special case: j+d = i
    valid2b = (jp_d == i) & (jp_d >= 0) & (jp_d < n)
    t2[valid2b] = 1.0
    
    # Term 3: h[i-d] * [0 <= i-d < n]
    im_d = i - d
    valid3 = (im_d >= 0) & (im_d < n)
    t3 = np.zeros(2*n-1)
    t3[valid3] = h[im_d[valid3]]
    
    # Term 4: -h[j-d] * [0 <= j-d < n]
    jm_d = j - d
    valid4 = (jm_d >= 0) & (jm_d < n)
    t4 = np.zeros(2*n-1)
    t4[valid4] = -h[jm_d[valid4]]
    
    return t1 + t2 + t3 + t4
```

This is O(n) with small constants. For n=170, this is very fast.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def swap_delta_binary(h, i, j, n):
    d = (n - 1) - np.arange(2 * n - 1)
    ip = i + d
    jp = j + d
    im = i - d
    jm = j - d
    v1 = (ip >= 0) & (ip < n)
    v2 = (jp >= 0) & (jp < n)
    v3 = (im >= 0) & (im < n)
    v4 = (jm >= 0) & (jm < n)
    delta = np.zeros(2 * n - 1)
    if np.any(v1):
        delta[v1] += -(1.0 - h[ip[v1]])
    if np.any(v2):
        idx = jp[v2]
        vals = 1.0 - h[idx]
        vals[idx == i] = 1.0
        delta[v2] += vals
    if np.any(v3):
        delta[v3] += h[im[v3]]
    if np.any(v4):
        delta[v4] += -h[jm[v4]]
    return delta

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def make_binary(h, n, seed=0):
    """Project to binary with exactly n//2 ones."""
    if n % 2 == 1:
        n_half = n // 2
    else:
        n_half = n // 2
    if np.sum(h) == n_half:
        # Already has right number of ones if binary
        if np.all((h == 0) | (h == 1)):
            return h
    # Threshold approach
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:n_half]] = 1.0
    return binary

def local_search(h, n, time_fn, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full").astype(float)
    best_max = np.max(corr)
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    stuck = 0
    kick_size = 3
    iter_count = 0
    
    while not time_fn():
        iter_count += 1
        current_max = np.max(corr)
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_h = h.copy()
        
        peak_idx = np.argmax(corr)
        d_peak = peak_idx - (n - 1)
        
        # Generate candidates
        candidates = []
        
        # Targeted: positions contributing to the peak
        sample_ones = ones if len(ones) <= 50 else [ones[rng.randint(len(ones))] for _ in range(50)]
        for i in sample_ones:
            ip = i + d_peak
            if 0 <= ip < n and h[ip] == 0:
                for _ in range(2):
                    j = zeros[rng.randint(len(zeros))]
                    if j != i:
                        candidates.append((i, j))
        
        # Random
        n_rand = 100
        for _ in range(n_rand):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidates.append((i, j))
        
        # Evaluate
        best_new_max = current_max
        best_pair = None
        best_delta = None
        
        for i, j in candidates:
            if h[i] != 1.0 or h[j] != 0.0:
                continue
            delta = swap_delta_binary(h, i, j, n)
            new_max = np.max(corr + delta)
            if new_max < best_new_max - 1e-10:
                best_new_max = new_max
                best_pair = (i, j)
                best_delta = delta
        
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            corr += best_delta
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            stuck = 0
        else:
            stuck += 1
            if stuck >= 100:
                for _ in range(kick_size):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                corr = np.correlate(h, 1 - h, mode="full").astype(float)
                stuck = 0
                kick_size = min(kick_size + 2, 30)
        
        # Periodic recomputation to avoid drift
        if iter_count % 500 == 0:
            corr = np.correlate(h, 1 - h, mode="full").astype(float)
    
    return best_h, best_max * (2.0 / n)

def slsqp_refine(h, n, time_fn, maxiter=300):
    """Try SLSQP with smooth max to refine from binary."""
    eps = 500.0
    dx = 2.0 / n
    
    def smooth_obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        m = np.max(corr)
        shifted = eps * (corr - m)
        return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx
    
    smooth_h = h.copy()
    result = minimize(
        smooth_obj, smooth_h,
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    if np.all(np.isfinite(result.x)) and not time_fn():
        h_new = project_feasible(result.x, n)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new, c5_new
    return h, evaluate_c5(h)

    start_time = time.time()
    
    def elapsed():
        return time.time() - start_time
    
    def time_check(frac=0.95):
        return elapsed() > budget_s * frac
    
    # Start from initial

    # Phase 3: More local search from refined solution (25% budget)
    h_from_refined = make_binary(best_h, n, seed + 1)
    def tc3():
        return elapsed() > budget_s * 0.70
    h_ls2, c5_ls2 = local_search(h_from_refined, n, tc3, seed=seed + 100)
    if c5_ls2 < best_c5:
        best_c5 = c5_ls2
        best_h = h_ls2
    
    # Phase 4: Try different initializations (15% budget)
    initializations = []
    # Thue-Morse
    tm = thue_morse(n)
    # Adjust to have exactly n//2 ones
    if np.sum(tm) != n // 2:
        # Flip some bits
        ones_idx = np.where(tm == 1)[0]
        zeros_idx = np.where(tm == 0)[0]
        n_flip = int(np.abs(np.sum(tm) - n // 2))
        if len(ones_idx) >= n_flip:
            tm[ones_idx[:n_flip]] = 0
        elif len(zeros_idx) >= n_flip:
            tm[zeros_idx[:n_flip]] = 1
    initializations.append(tm)
    
    # Random
    for rs in range(3):
        h_rand = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        initializations.append(h_rand)
    
    for init_i, h_init in enumerate(initializations):
        def tc_init():
            return time_check(0.88)
        h_ls3, c5_ls3 = local_search(h_init, n, tc_init, seed=seed + 200 + init_i)
        if c5_ls3 < best_c5:
            best_c5 = c5_ls3
            best_h = h_ls3
    
    # Phase 5: Final SLSQP refinement (5% budget)
    def tc5():
        return elapsed() > budget_s * 0.97
    best_h, best_c5 = slsqp_refine(best_h, n, tc5, maxiter=200)

    best_c5 = evaluate_c5(best_h)
