        return h
    if diff > 0:

        for i in order:

            h[i] += add

            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

def _coord_descent(h, n, n_sweeps, deadline, check_every=10):
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()

    for sweep in range(n_sweeps):
        if time.time() > deadline - 3:
            break
        for i in range(n):
            if time.time() > deadline - 3:
                break
            old_x = h[i]
            # Compute corr at x=0 and x=1
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            
            # Linear part for m != n-1
            # Quadratic part for m == n-1: c0[n-1] + x - x^2

def _slsqp_refine(h_init, n, alpha, maxiter, ftol, deadline):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Smooth max: (1/alpha) * log(sum(exp(alpha * corr)))
        m = np.max(corr)
        if m > 500 / alpha:  # overflow protection
            return 500.0
        return (np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha + m) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m > 500 / alpha:
            return np.zeros(npts)
        weights = np.exp(alpha * (corr - m))
        weights /= np.sum(weights)
        # Gradient of corr[m] w.r.t. x[i]
        # corr[m] = sum_j x[j+k]*(1-x[j]) where k = m-(n-1)
        # d(corr[m])/dx[i] = 
        #   (1-x[i-k]) if 0<=i-k<n (from x[i] = x[(i-k)+k] term)
        #   - x[i+k] if 0<=i+k<n (from (1-x[i]) term)
        #   For k=0: (1-x[i]) - x[i] = 1-2x[i] (but wait, both terms give the same j=i)
        #   Actually for k=0: corr[m] = sum_j x[j]*(1-x[j]), d/dx[i] = 1-2x[i]
        #   For k≠0: d/dx[i] = (1-x[i-k]) if i-k valid - x[i+k] if i+k valid
        #             But we need to be careful: if k>0, i-k is valid when i>=k
        #             and i+k is valid when i+k<n.
        
        g = np.zeros(npts)
        for m in range(2 * npts - 1):
            k = m - (npts - 1)
            w = weights[m]
            if abs(w) < 1e-10:
                continue
            if k == 0:
                g += w * (1.0 - 2.0 * x)
            elif k > 0:
                # corr = sum_{j=0}^{n-1-k} x[j+k]*(1-x[j])
                # d/dx[i] for i in [k, n-1]: (1-x[i-k]) [from x[i] term]
                # d/dx[i] for i in [0, n-k-1]: -x[i+k] [from (1-x[i]) term]
                g[k:] += w * (1.0 - x[:npts-k])
                g[:npts-k] -= w * x[k:]
            else:
                L = -k
                # corr = sum_{j=L}^{n-1} x[j-L]*(1-x[j])  wait...
                # k < 0, let L = -k > 0
                # corr[m] = sum_{j: 0<=j<n, 0<=j+k<n} x[j+k]*(1-x[j])
                # j+k >= 0 => j >= -k = L
                # j+k < n => j < n-L
                # So sum over j = L to n-L-1: x[j-L]*(1-x[j])
                # d/dx[i]:
                #   x[i] appears as x[j-L] when j-L = i, j = i+L, valid if L<=i+L<n => 0<=i<n-L
                #     contribution: (1-x[i+L])
                #   x[i] appears as (1-x[j]) when j = i, valid if L<=i<n-L
                #     contribution: -x[i-L]
                # So for i in [0, n-L-1]: g[i] += w*(1-x[i+L])
                # For i in [L, n-1]: g[i] -= w*x[i-L]
                g[:npts-L] += w * (1.0 - x[L:])
                g[L:] -= w * x[:npts-L]
        return g * dx
    
    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj(h_init):
            return res.x
        else:
            return h_init.copy()
    except Exception:
        return h_init.copy()

def _make_starting_points(n, seed, target_sum):

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _slsqp_obj_and_grad(x, n, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shift = np.max(shifted)
    if max_shift > 500:
        return 500.0, np.zeros(n)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    val = (m + log_sum / alpha) * dx
    weights = exp_shifted / np.sum(exp_shifted)
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        w = weights[mi]
        if w < 1e-12:
            continue
        k = mi - (n - 1)
        if k == 0:
            g += w * (1.0 - 2.0 * x)
        elif k > 0:
            g[k:] += w * (1.0 - x[:n - k])
            g[:n - k] -= w * x[k:]
        else:
            L = -k
            g[:n - L] += w * (1.0 - x[L:])
            g[L:] -= w * x[:n - L]
    g *= dx
    return val, g

def _slsqp_refine(h_init, n, alpha, maxiter, ftol, deadline):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h_init)

    def obj(x):
        v, _ = _slsqp_obj_and_grad(x, npts, alpha, dx)
        return v

    def grad(x):
        _, g = _slsqp_obj_and_grad(x, npts, alpha, dx)
        return g

    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_out = res.x if res.fun < obj(h_init) else h_init.copy()
        return np.clip(h_out, 0.0, 1.0)
    except Exception:
        return h_init.copy()

            break
        if it % 200 == 0:
            t_cur *= 0.95
        i = rng.randint(n)
        old = h[i]
        h[i] = rng.rand()
        c5_new = _c5_of(h)
        c5_old = best_c5
        if c5_new < c5_old or rng.rand() < np.exp(-(c5_new - c5_old) / max(t_cur, 1e-6)):
            h = _fix_sum(h, target_sum)
            c5_new = _c5_of(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        else:
            h[i] = old
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_values = [100, 86, 120]

    for n in n_values:
        if time.time() > deadline - 80:
            break
        target_sum = n / 2.0

        starts = _make_starts(n, seed, target_sum)

        if n == 100 and 'initial_h_values' in globals() and initial_h_values is not None:
            iv = np.array(initial_h_values, dtype=float)
            if len(iv) == n:
                starts.insert(0, _fix_sum(iv, target_sum))

        for si, h0 in enumerate(starts):
            if time.time() > deadline - 40:
                break

            t_rem = deadline - time.time()
            n_sweeps = min(300, max(50, int(t_rem * 3)))
            h_cd, c5_cd = _coord_descent(h0, n, n_sweeps, deadline)

            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

            if time.time() > deadline - 20:
                continue

            h_cur = h_cd.copy()
            for alpha in [200, 1000, 5000]:
                if time.time() > deadline - 12:
                    break
                maxiter = min(300, max(30, int((deadline - time.time()) * 3)))
                h_r = _slsqp_refine(h_cur, n, alpha, maxiter, 1e-15, deadline)
                h_r = _fix_sum(h_r, target_sum)
                c5_r = _c5_of(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n
                h_cur = h_r

            if time.time() > deadline - 15:
                continue

            h_cd2, c5_cd2 = _coord_descent(h_cur, n, 100, deadline)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n

    if best_h is not None and time.time() < deadline - 30:
        n = best_n
        target_sum = n / 2.0
        rng = np.random.RandomState(seed + 7777)
        for attempt in range(15):
            if time.time() > deadline - 12:
                break
            scale = 0.03 + 0.07 * rng.rand()
            h_p = _fix_sum(best_h + scale * rng.randn(n), target_sum)
            h_a, c5_a = _anneal(h_p, n, target_sum, deadline, seed + attempt * 13)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
                best_n = n
            if time.time() > deadline - 18:
                break
            h_cd3, c5_3 = _coord_descent(h_a, n, 60, deadline)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h_cd3.copy()
                best_n = n
            if time.time() > deadline - 15:
                break
            h_r = _slsqp_refine(h_cd3, n, 2000, 100, 1e-15, deadline)
            h_r = _fix_sum(h_r, target_sum)
            c5_r = _c5_of(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _fix_sum(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
