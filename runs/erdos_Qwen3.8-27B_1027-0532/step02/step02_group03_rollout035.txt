import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / n)

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _de_optimize(h_init, n, target_sum, pop_size, n_gens, F_range, CR, seed, deadline):
    rng = np.random.RandomState(seed)
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = _fix_sum(h_init, target_sum)
    for i in range(1, pop_size):
        # Mix of random and perturbed init
        if rng.random() < 0.3:
            cand = h_init + rng.randn(n) * 0.3
        else:
            cand = rng.rand(n)
        pop[i] = _fix_sum(np.clip(cand, 0, 1), target_sum)
    
    fitness = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_val = fitness[best_idx]
    best = pop[best_idx].copy()
    
    stagnant = 0
    for gen in range(n_gens):
        if time.time() > deadline:
            break
        F = F_range[0] - (F_range[0] - F_range[1]) * (gen / max(1, n_gens))
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            # DE/best/1 or DE/rand/1
            if rng.random() < 0.5:
                j = (best_idx + rng.randint(1, pop_size)) % pop_size
                k = (j + rng.randint(1, pop_size)) % pop_size
                v = pop[best_idx] + F * (pop[j] - pop[k])
            else:
                indices = [i] + list(rng.choice([x for x in range(pop_size) if x != i], 2, replace=False))
                v = pop[indices[0]] + F * (pop[indices[1]] - pop[indices[2]])
            
            # Binary crossover
            trial = pop[i].copy()
            cr_point = rng.randint(n)
            for j in range(n):
                if rng.random() < CR or j == cr_point:
                    trial[j] = v[j]
            trial = _fix_sum(np.clip(trial, 0, 1), target_sum)
            
            # Selection
            f_trial = _c5_of(trial)
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_val - 1e-10:
                    best_val = f_trial
                    best = trial.copy()
                    best_idx = i
                    stagnant = 0
                elif f_trial < best_val:
                    stagnant = 0

                else:

                fitness[i] = _c5_of(pop[i])
            stagnant = 0
    
    return best, best_val

def _coord_descent_line(h0, n, target_sum, max_rounds=20, n_trials=5):

                        best_v = v
                h[j] = best_v

        if not improved:

    return h

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / n)

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _nesterov(h0, n, target_sum, n_iters, alpha0, use_smooth_sg=False, sg_m=5, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        if use_smooth_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            weights = np.exp(50.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m = np.argsort(weights)[::-1][:sg_m]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _de_optimize(h_init, n, target_sum, pop_size, n_gens, seed, deadline):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = _fix_sum(h_init.copy(), target_sum)
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.3:
            cand = h_init + rng.randn(n) * 0.3
        elif r < 0.6:
            cand = rng.rand(n)
        else:
            cand = h_init + rng.randn(n) * 0.15
        pop[i] = _fix_sum(np.clip(cand, 0.0, 1.0), target_sum)

    fitness = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_val = fitness[best_idx]
    best = pop[best_idx].copy()
    stagnant = 0

    F_start = 0.9
    F_end = 0.4
    CR = 0.7

    for gen in range(n_gens):
        if time.time() > deadline:
            break
        F = F_start - (F_start - F_end) * (gen / max(1, n_gens))

        for i in range(pop_size):
            if time.time() > deadline:
                break
            if rng.random() < 0.4:
                a = best_idx
                b = (a + rng.randint(1, pop_size)) % pop_size
                c = (b + rng.randint(1, pop_size)) % pop_size
                v = pop[a] + F * (pop[b] - pop[c])
            else:
                others = [x for x in range(pop_size) if x != i]
                a = i
                b, c = rng.choice(others, 2, replace=False)
                v = pop[a] + F * (pop[b] - pop[c])

            trial = pop[i].copy()
            cr_point = rng.randint(n)
            mask = rng.random(n) < CR
            mask[cr_point] = True
            trial[mask] = v[mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)

            f_trial = _c5_of(trial)
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_val - 1e-10:
                    best_val = f_trial
                    best = trial.copy()
                    best_idx = i
                    stagnant = 0

        stagnant += 1
        if stagnant > 30:
            n_div = max(3, pop_size // 4)
            for i in range(n_div):
                idx = int(rng.randint(pop_size))
                if idx == best_idx:
                    continue
                r = rng.random()
                if r < 0.5:
                    pop[idx] = _fix_sum(rng.rand(n), target_sum)
                else:
                    pop[idx] = _fix_sum(np.clip(pop[best_idx] + rng.randn(n) * 0.25, 0.0, 1.0), target_sum)
                fitness[idx] = _c5_of(pop[idx])
            stagnant = 0

    return best, best_val

def _coord_descent(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            return h
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            if old <= 0.05 or old >= 0.95:
                candidates = [0.0, 1.0, 0.1, 0.9]
            else:
                candidates = [0.0, 1.0, max(0.0, old - 0.2), max(0.0, old - 0.1),
                              min(1.0, old + 0.1), min(1.0, old + 0.2), 0.25, 0.75]
            for v in candidates:
                if abs(v - old) < 1e-8:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _iter_refine(h0, n, target_sum, deadline, seed):
    rng = np.random.RandomState(seed)
    best = h0.copy()
    best_val = _c5_of(h0)
    for r in range(8):
        if time.time() > deadline:
            break
        sigma = 0.15 * (1.0 - r / 10.0)
        h_pert = _fix_sum(np.clip(h0 + rng.randn(n) * sigma, 0.0, 1.0), target_sum)
        h_r, c5_r = _nesterov(h_pert, n, target_sum, n_iters=150, alpha0=0.3, deadline=deadline)
        if c5_r < best_val - 1e-10:
            best = h_r.copy()
            best_val = c5_r
            h0 = h_r.copy()
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    for nn in [80, 90, 100, 110, 120, 130, 140, 160, 180]:
        if nn % 2 == 0 and 40 <= nn <= 200:
            if nn not in n_candidates:
                n_candidates.append(nn)

    n_candidates.sort(key=lambda x: abs(x - n_init))

    for n in n_candidates:

            break
        target_sum = n / 2.0

        if n == n_init:
            h_interp = np.clip(h_init, 0.0, 1.0).copy()
        else:
            x_old = np.linspace(0, n_init - 1, n_init)

            h_interp = np.interp(x_new, x_old, h_init)
        h_interp = _fix_sum(h_interp, target_sum)

        de_pop = 35
