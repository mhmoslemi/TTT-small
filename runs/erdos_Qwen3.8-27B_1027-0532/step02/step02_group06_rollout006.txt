Let me also think about the gradient computation. For the smoothed objective:
f(h) = max_k corr_k * dx (approximately, after smoothing)

The gradient of the log-sum-exp smoothing is:
∂/∂h_i [log(sum_k exp(alpha * corr_k))/alpha] = sum_k p_k * ∂corr_k/∂h_i

where p_k = exp(alpha * corr_k) / sum_j exp(alpha * corr_j)

This is what the previous code does. The key is to compute this efficiently.

For a given lag l (where l = k - (n-1), so l ranges from -(n-1) to n-1):
corr_l = sum_i h[i] * (1-h[i+l]) (with appropriate bounds)

∂corr_l/∂h_i:
- For the h[i] term: (1 - h[i+l]) (if i+l is in range)
- For the h[i+l] term: -h[i] (if i-l is in range, i.e., h[i] = h[(i-l)+l])

So ∂corr_l/∂h_i = (1-h[i+l]) * 1_{i+l < n} - h[i-l] * 1_{i-l >= 0}

This is what compute_gradient_for_lag does.

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def smoothed_obj_and_grad(h, alpha, target_sum=None):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h) * dx
    m = np.max(corr)
    # Shift for numerical stability
    shifted = corr - m
    log_sum = np.log(np.sum(np.exp(alpha * shifted))) / alpha
    obj = m + log_sum
    # Gradient
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    threshold = 1e-8
    active = np.where(weights > threshold)[0]
    for idx in active:
        lag = int(idx - (n - 1))
        grad += weights[idx] * compute_gradient_for_lag(h, lag)
    return obj, grad

def subgradient_run(h0, alpha_schedule, n_top, step_init, seed, t0, time_limit):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0
    it = 0
    
    for s_alpha in alpha_schedule:
        if time.time() - t0 > t0 + time_limit:
            break
        max_iters = int(time_limit * 15)  # rough
        for _ in range(max_iters):
            if time.time() - t0 > t0 + time_limit:
                break
            it += 1
            corr = compute_correlations(h)
            top_idx = np.argsort(-corr)[:n_top]
            top_vals = corr[top_idx]
            shifted = top_vals - top_vals[0]
            weights = np.exp(s_alpha * shifted)
            weights /= weights.sum()
            grad = np.zeros(n)
            for w, idx in zip(weights, top_idx):
                lag = int(idx - (n - 1))
                grad += w * compute_gradient_for_lag(h, lag)
            grad_norm = np.sqrt(np.sum(grad * grad))
            if grad_norm < 1e-14:
                break
            grad /= grad_norm  # normalize for stability
            improved = False
            for trial in range(30):
                h_new = h - step * grad
                h_new = project_to_feasible(h_new, target_sum)

                    h = h_new

                    h = project_to_feasible(h, target_sum)
                    step = step_init * (0.5 + rng.rand())
                    stall = 0
    return best_h, best_score

def lbfgsb_refine(h, alpha, maxiter=5000):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        obj_val, _ = smoothed_obj_and_grad(hv, alpha)
        return obj_val
    
    def grad(hv):
        _, g = smoothed_obj_and_grad(hv, alpha)
        return g
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        result = project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
        return result
    return None

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def constraint_fun(hv):
        return np.sum(hv) - target_sum
    
    def constraint_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

def coordinate_descent(h, target_sum, max_rounds=1000, t0=None, time_limit=None):
    """Try to improve by adjusting individual coordinates."""
    n = len(h)

        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + dv, 0.0, 1.0)
        # Need to maintain sum constraint - adjust another coordinate
        delta = h_new[i] - h[i]
        if abs(delta) > 1e-15:

            if np.abs(np.sum(h_new) - target_sum) > 1e-10:
                # Project back
                h_new = project_to_feasible(h_new, target_sum)
        score = evaluate_score(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
            h = h_new
    return best_h, best_score

        grad[:] = 1.0 - 2.0 * h

        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

        m = np.max(corr)

from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def lbfgsb_obj(hv, alpha, mu, n, target_sum):
    corr = compute_correlations(hv)
    m = np.max(corr)
    shifted = corr - m
    smoothed = m + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    penalty = mu * (np.sum(hv) - target_sum) ** 2
    return smoothed + penalty


def lbfgsb_grad(hv, alpha, mu, n, target_sum):
    corr = compute_correlations(hv)
    m = np.max(corr)
    shifted = corr - m
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    g = np.zeros(n)
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for idx in active:
        lag = int(idx - (n - 1))
        g += weights[idx] * compute_gradient_for_lag(hv, lag)
    g += 2.0 * mu * (np.sum(hv) - target_sum)
    return g


def lbfgsb_refine(h, alpha, mu, maxiter):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    res = minimize(lbfgsb_obj, h, jac=lbfgsb_grad, method='L-BFGS-B',
                   bounds=bounds, args=(alpha, mu, n, target_sum),
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'maxls': 100})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def slsqp_obj(hv, alpha):
    corr = compute_correlations(hv)
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def slsqp_con_fun(hv, target_sum):
    return np.sum(hv) - target_sum


def slsqp_con_jac(hv, target_sum):
    return np.ones(len(hv))


def slsqp_refine(h, alpha, maxiter):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac, 'args': (target_sum,)}]
    res = minimize(slsqp_obj, h, method='SLSQP', bounds=bounds, args=(alpha,),
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
