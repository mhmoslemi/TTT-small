            hi_V = mid_V
        else:
            lo_V = mid_V

    lo_t = 0.0

        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

def coordinate_descent(h, n_sweeps=2000, tol=1e-12, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 50 == 49:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def block_perturbation(h, rng):

    n = len(h)

    method = rng.randint(4)

    elif method == 1:

        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.05, 0.2)
        h[start:start+length] += amp * np.sin(2 * np.pi * freq * x / length + rng.uniform(0, 2*np.pi))
    elif method == 2:
        # Block swap
        if n >= 10:
            l1 = rng.randint(2, max(3, n // 5))
            l2 = l1
            s1 = rng.randint(0, n - l1)
            s2 = rng.randint(0, n - l2)
            if s1 == s2 or abs(s1 - s2) < max(l1, l2):

            tmp = h[s1:s1+l1].copy()
            h[s1:s1+l1] = h[s2:s2+l2]
            h[s2:s2+l2] = tmp
    else:
        # Gaussian perturbation on random interval

        start = rng.randint(0, n - length)
        amp = rng.uniform(0.05, 0.15)
        h[start:start+length] += rng.randn(length) * amp
    
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, n / 2.0)
    return h


def subgradient_step(h, eta, momentum=0.9):
    """Take a subgradient step with momentum."""
    n = len(h)
    corr = compute_correlations(h)
    max_val = np.max(corr)
    # Active set: shifts where correlation is close to max
    active = corr > max_val - 0.01 * max_val
    
    # Compute average slope over active set
    # For each i, the slope of corr[k] w.r.t. h[i] is:
    # (1 - h[i-k]) for valid k where i-k is the "first" index
    # minus h[i+k] for valid k where i is the "first" index
    # This is complex, so let's use a simpler approach:
    # The subgradient of max_k corr[k] is the slope vector for k* that achieves max
    k_star = np.argmax(corr)
    k = k_star - (n - 1)  # shift value
    
    # Slope for coordinate i when shift is k:
    # corr[n-1+k] = sum_j h[j+k] * (1-h[j]) for valid j
    # d/dcot h[i]: 
    #   if i+k is valid: (1-h[i-k]) ... wait this is getting confusing
    
    # Let me use the same formulation as optimize_coordinate
    m = 2 * n - 1
    p = np.arange(m) - k_star  # this gives the shift for each correlation element
    
    # Actually, let me just compute the subgradient numerically
    # For the purpose of this, use the slope from the coordinate optimization
    # The subgradient direction is -slopes for the active shift
    # But we want the average over active shifts
    
    # For simplicity, use the slope from the max shift
    # slope[i] = (1-h[i-k]) * [valid] - h[i+k] * [valid]
    slopes = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = i - k
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i + k
        if 0 <= j2 < n:
            val -= h[j2]
        slopes[i] = val
    
    # Center the slope to maintain sum constraint
    slopes -= np.mean(slopes)
    
    # Step
    h_new = h - eta * slopes
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n / 2.0)
    return h_new, slopes

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha


def smooth_obj(h, alpha):
    return smooth_max(compute_correlations(h), alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))

        else:

    lo_t = 0.0
