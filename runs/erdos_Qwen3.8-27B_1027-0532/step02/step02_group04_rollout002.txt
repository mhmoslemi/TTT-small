```python
    # R(k) = sum_i h[i] * (1 - h[i + lag_k])
    # For coordinate j, the coefficient in R(k) is:
    #   (1 - h[j + lag_k]) if j + lag_k in [0, n-1]
    #   minus h[j - lag_k] if j - lag_k in [0, n-1]
    #   (for lag_k = 0: coefficient is (1 - 2h[j]) + derivative of -h[j]^2 which is -2h[j])
    #   Actually for lag=0: term is h[j]*(1-h[j]), derivative is 1-2h[j]

    corr = compute_correlations(h)
    max_corr = np.max(corr)
    
    for j in range(n):
        # Try to find optimal h[j]
        # For each k, R(k) = corr[k] + δh * coef_k where coef_k is the coefficient of h[j] in R(k)
        # We want to minimize max_k (corr[k] + δh * coef_k) subject to h[j] + δh ∈ [0,1]
        # Plus the constraint sum(h) = n/2 (handled separately)
        # And the lag-0 term is quadratic: h[j]*(1-h[j]) so R(0) = corr[0] - old_val + (h[j]+δh)*(1-h[j]-δh)
        # = corr[0] - h[j]*(1-h[j]) + (h[j]+δh)*(1-h[j]-δh)
        # = corr[0] + δh*(1-h[j]) - h[j]*δh - δh^2
        # = corr[0] + δh*(1-2h[j]) - δh^2
        
        # For simplicity, let's just do a 1D search
        best_val = max_corr
        best_hj = h[j]
        
        # Try several values
        for trial in np.linspace(0, 1, 20):
            δh = trial - h[j]
            if abs(δh) < 1e-10:
                continue
            # Update correlations
            new_corr = corr.copy()
            for k in range(2*n - 1):
                lag = k - (n - 1)
                coef = 0.0
                if lag == 0:
                    # h[j]*(1-h[j]) changes
                    old_term = h[j] * (1 - h[j])
                    new_term = trial * (1 - trial)
                    new_corr[k] += new_term - old_term
                    continue
                if 0 <= j + lag < n:
                    coef += (1 - h[j + lag])
                if 0 <= j - lag < n:
                    coef += -h[j - lag]
                new_corr[k] += δh * coef
            
            max_trial = np.max(new_corr)
            if max_trial < best_val:
                best_val = max_trial
                best_hj = trial
        
        if best_hj != h[j]:
            # Apply the change
            h[j] = best_hj
            corr = compute_correlations(h)  # Recompute (expensive but safe)
            max_corr = np.max(corr)
```

```python
def subgradient_descent(h, n, n_iters, step_init=0.01):
    """Subgradient method with projection."""
    best = h.copy()
    best_score = np.max(compute_correlations(best))
    step = step_init
    
    for it in range(n_iters):
        corr = compute_correlations(best)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        active_lag = max_idx - (n - 1)
        
        # Compute gradient of R(active_lag)
        grad = np.zeros(n)
        if active_lag == 0:
            grad = 1.0 - 2.0 * best
        elif active_lag > 0:
            grad[active_lag:] += (1.0 - best[:n-active_lag])
            grad[:n-active_lag] += -best[active_lag:]
        else:
            l = -active_lag
            grad[:n-l] += (1.0 - best[l:])
            grad[l:] += -best[:n-l]
        
        # Projected subgradient step: move in -grad direction, preserving sum
        # Zero-mean gradient
        grad_centered = grad - np.mean(grad)
        new_h = best - step * grad_centered
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_feasible(new_h, n / 2.0)
        
        new_score = np.max(compute_correlations(new_h))
        if new_score < best_score:
            best = new_h
            best_score = new_score
            # Slowly decrease step
            step *= 0.999
        else:
            step *= 0.9
        
        if step < 1e-6:
            break
    
    return best, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = index - (N - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[lag:] += (1.0 - h[:N - lag])
                grad[:N - lag] += -h[lag:]
        else:
            l = -lag
            if l < N:
                grad[:N - l] += (1.0 - h[l:])
                grad[l:] += -h[:N - l]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def lbfgs_optimize(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def subgradient_step(h, n, step):
    corr = compute_correlations(h)
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    active_lag = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if active_lag == 0:
        grad = 1.0 - 2.0 * h
    elif active_lag > 0:
        if active_lag < n:
            grad[active_lag:] += (1.0 - h[:n - active_lag])
            grad[:n - active_lag] += -h[active_lag:]
    else:
        l = -active_lag
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = index - (N - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[lag:] += (1.0 - h[:N - lag])
                grad[:N - lag] += -h[lag:]
        else:
            l = -lag
            if l < N:
                grad[:N - l] += (1.0 - h[l:])
                grad[l:] += -h[:N - l]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.
