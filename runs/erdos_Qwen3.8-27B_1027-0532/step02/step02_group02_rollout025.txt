for k = 0, 1, ..., 2n-2, with the understanding that we only sum over valid i.

So corr_full[n-1] = Σ_i h[i] * (1-h[i]) (lag 0)
corr_full[n-1+lag] = Σ_{i=0}^{n-1-lag} h[i] * (1-h[i+lag]) (positive lag)
corr_full[n-1-lag] = Σ_{i=lag}^{n-1} h[i] * (1-h[i-lag]) (negative lag)

The gradient d(corr_full[k])/dh[j]:
For k = n-1+lag (positive lag, lag > 0):
  corr = Σ_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
  d/dh[j]:

    - If j in [lag, n-1]: contribution from h[j-lag]*(1-h[j]) → -h[j-lag]
    - Both can apply if j in [lag, n-1-lag]

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _softmax_corr_grad(h, alpha, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if len(corr) > top_m:
        idx = np.argsort(corr)[-top_m:]
    else:
        idx = np.arange(len(corr))
    w = np.exp(alpha * (corr[idx] - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[-1] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for j, k in enumerate(idx):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad[:] += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wj * (-h[lag:])
            grad[lag:] += wj * (1.0 - h[:n - lag])
        else:
            L = -lag
            grad[:n - L] += wj * (1.0 - h[L:])
            grad[L:] += wj * (-h[:n - L])
    return grad * dx

def _softmax_descent(h, n, n_iter, a0, a1, growth, lr0, decay, deadline, momentum, check_every):
    target_sum = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    no_improve = 0
    patience = check_every * 25
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        alpha = min(a1, a0 * (1.0 + t * growth))
        lr = lr0 / (1.0 + t * decay)
        g = _softmax_corr_grad(h, alpha, top_m=60)
        v = momentum * v - lr * g
        h = _fix_sum(h + v, target_sum)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h)
            if c5 < best_c5 - 5e-7:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_solve(h, n, alpha, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    npts = len(h)

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        if m < -500:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (c - m))))) / alpha * dx

    def jcb(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        w = np.exp(alpha * (c - m))
        s = w.sum()
        if s < 1e-300:
            w = np.zeros_like(w)
            w[np.argmax(c)] = 1.0
        else:
            w /= s
        g = np.zeros(npts)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-12:
                continue
            lag = k - (npts - 1)
            if lag == 0:
                g[:] += wk * (1.0 - 2.0 * x)
            elif lag > 0:
                g[:npts - lag] += wk * (-x[lag:])
                g[lag:] += wk * (1.0 - x[:npts - lag])
            else:
                L = -lag
                g[:npts - L] += wk * (1.0 - x[L:])
                g[L:] += wk * (-x[:npts - L])
        return g * dx

    bounds = [(0.0, 1.0)] * npts
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(npts)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jcb,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def _lbfgsb_solve(h, n, alpha, maxiter, ftol, pen):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        if m < -500:
            sm = 0.0
        else:
            sm = (m + np.log(np.sum(np.exp(alpha * (c - m))))) / alpha
        return sm * dx + pen * (x.sum() - target_sum) ** 2

    def jcb(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        w = np.exp(alpha * (c - m))
        s = w.sum()
        if s < 1e-300:
            w = np.zeros_like(w)
            w[np.argmax(c)] = 1.0
        else:
            w /= s
        g = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-12:
                continue
            lag = k - (n - 1)
            if lag == 0:
                g[:] += wk * (1.0 - 2.0 * x)
            elif lag > 0:
                g[:n - lag] += wk * (-x[lag:])
                g[lag:] += wk * (1.0 - x[:n - lag])
            else:
                L = -lag
                g[:n - L] += wk * (1.0 - x[L:])
                g[L:] += wk * (-x[:n - L])
        g *= dx
        g += 2.0 * pen * (x.sum() - target_sum)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jcb,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def _perturb_polish(h, n, target_sum, rng, deadline):
    best = h.copy()
    best_c5 = _c5_of(h)
    for attempt in range(12):
        if time.time() > deadline - 10:
            break
        scale = 0.005 + 0.03 * rng.rand()
        h_p = _fix_sum(h + scale * rng.randn(n), target_sum)
        for alpha in [80, 400, 1500]:
            if time.time() > deadline - 10:
                break
            h_r = _slsqp_solve(h_p, n, alpha, maxiter=200, ftol=1e-14)
            h_r = _fix_sum(h_r, target_sum)
            c5_r = _c5_of(h_r)
            if c5_r < best_c5 - 1e-14:
                best_c5 = c5_r
                best = h_r.copy()
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    best_h = None
    best_c5 = np.inf
    best_n = 0

    n_work = 100
    target_sum = n_work / 2.0

    h_starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv = np.array(initial_h_values, dtype=float)
        if len(iv) == n_work:
            h_starts.append(_fix_sum(iv, target_sum))
        else:
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 2, len(iv), endpoint=False)
            x_new = np.linspace(0, 2, n_work, endpoint=False)
            h_interp = interp1d(x_old, iv, kind='cubic', fill_value='extrapolate')(x_new)
            h_starts.append(_fix_sum(h_interp, target_sum))

    x = np.linspace(0, 2, n_work, endpoint=False)
    rng = np.random.RandomState(seed)
    h_starts.append(_fix_sum(0.5 + 0.18 * np.sin(2*np.pi*x) + 0.08*np.sin(4*np.pi*x), target_sum))
    h_starts.append(_fix_sum(0.5 + 0.25*np.sin(3*np.pi*x) + 0.05*np.sin(5*np.pi*x), target_sum))
    h_starts.append(_fix_sum(rng.rand(n_work), target_sum))

    for h0 in h_starts:
        if time.time() > deadline - 60:
            break
        h1, c5_1 = _softmax_descent(h0, n_work, n_iter=20000,
                                     a0=10.0, a1=300.0, growth=0.0001,
                                     lr0=0.6, decay=0.0001,
                                     deadline=deadline, momentum=0.92, check_every=300)
        if c5_1 < best_c5:
            best_c5, best_h, best_n = c5_1, h1.copy(), n_work

        if time.time() > deadline - 40:
            continue

        h2, c5_2 = _softmax_descent(h1, n_work, n_iter=15000,
                                     a0=100.0, a1=3000.0, growth=0.0003,
                                     lr0=0.25, decay=0.0004,
                                     deadline=deadline, momentum=0.88, check_every=300)
        if c5_2 < best_c5:

            h_s = _slsqp_solve(h_cur, n_work, alpha, maxiter=400, ftol=1e-14)
            h_s = _fix_sum(h_s, target_sum)
            c5_s = _c5_of(h_s)
            if c5_s < best_c5:

    if best_h is None:
        best_h = _fix_sum(np.full(n_work, 0.5), target_sum)

        best_n = n_work

    if time.time() < deadline - 20:
        rng2 = np.random.RandomState(seed + 12345)
        best_h, best_c5 = _perturb_polish(best_h, best_n, best_n / 2.0, rng2, deadline)

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
