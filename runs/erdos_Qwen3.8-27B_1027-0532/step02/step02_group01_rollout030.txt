    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx

    x -= np.max(x)

        h_opt = project_onto_constraint(result.x, target_sum)

        h[h_bin] = 0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        h_bin = rng.choice(zeros, n // 2 - n_ones, replace=False)
        h[h_bin] = 1
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    for i in range(n_steps):
        t = t_start * (t_end / t_start) ** (i / n_steps)
        oi = ones_list[rng.integers(len(ones_list))]
        zi = zeros_list[rng.integers(len(zeros_list))]
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            oi_idx = np.where(ones_list == oi)[0]
            if len(oi_idx) > 0:
                ones_list[oi_idx[0]] = zi
            zi_idx = np.where(zeros_list == zi)[0]
            if len(zi_idx) > 0:
                zeros_list[zi_idx[0]] = oi
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c5

def pairwise_cd(h_init, n_sweeps, rng, full_sweeps=0):
    n = len(h_init)
    h = h_init.astype(float).copy()
    h = project_onto_constraint(h, n / 2.0)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        improved = False
        if full_sweeps > 0 and sweep < full_sweeps:
            pairs = [(j, (j + 1) % n) for j in range(n)]
        else:
            pairs = [(j, rng.integers(n)) for j in range(n)]
        for (j, l) in pairs:
            if j == l:
                l = (j + 1) % n
            s = h[j] + h[l]
            t_min = max(0.0, s - 1.0)
            t_max = min(1.0, s)
            if t_max - t_min < 1e-10:
                continue
            h_save_j = h[j]
            h_save_l = h[l]
            h[j] = 0.0
            h[l] = s
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            h[l] = s - 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = h_save_j
            h[l] = h_save_l
            A = corr1 - corr0
            B = corr0
            a = t_min
            b = t_max
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = np.max(A * c_pt + B)
            fd = np.max(A * d_pt + B)
            for _ in range(60):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = np.max(A * c_pt + B)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = np.max(A * d_pt + B)
            t_opt = (a + b) / 2.0
            new_c5 = compute_c5(np.where(np.arange(n) == j, t_opt, np.where(np.arange(n) == l, s - t_opt, h)))
            if new_c5 < best_c5 - 1e-15:
                h[j] = t_opt
                h[l] = s - t_opt
                best_c5 = new_c5
                improved = True
        if not improved:
            break
    return h, best_c5

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

        h0 = np.array(initial_h_values, dtype=float)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    remaining = budget_s - (time.time() - start_time)

        n_ones = int(np.sum(h_bin))
        if n_ones > n0 // 2:
            ones = np.where(h_bin == 1)[0]
            h_bin[rng.choice(ones, n_ones - n0 // 2, replace=False)] = 0
        elif n_ones < n0 // 2:
            zeros = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zeros, n0 // 2 - n_ones, replace=False)] = 1
        h_sa, c5_sa = fast_binary_sa(h_bin, 300000, 0.5, 0.005, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_cd, c5_cd = pairwise_cd(best_h, 50, rng, full_sweeps=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    remaining = budget_s - (time.time() - start_time)
    n_restarts = 8
    for restart in range(n_restarts):
        if remaining < 80:
            break
        strategy = restart % 4
        if strategy == 0:
            h_start = best_h + 0.05 * rng.standard_normal(n0)
            h_start = project_onto_constraint(h_start, target_sum)
        elif strategy == 1:
            h_start = (rng.random(n0) < 0.5).astype(float)
            h_start = project_onto_constraint(h_start, target_sum)
        elif strategy == 2:
            alpha = rng.uniform(0.4, 0.8)
            h_rand = project_onto_constraint(
                np.clip(0.5 + 0.4 * rng.standard_normal(n0), 0, 1), target_sum)
            h_start = alpha * best_h + (1.0 - alpha) * h_rand
            h_start = project_onto_constraint(h_start, target_sum)
        else:
            h_start = best_h.copy()
            nflip = max(3, n0 // 5)
            idx = rng.choice(n0, nflip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            h_start = project_onto_constraint(h_start, target_sum)

        h_bin = np.round(h_start).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones > n0 // 2:
            ones = np.where(h_bin == 1)[0]
            h_bin[rng.choice(ones, n_ones - n0 // 2, replace=False)] = 0
        elif n_ones < n0 // 2:
            zeros = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zeros, n0 // 2 - n_ones, replace=False)] = 1

        n_sa = 80000
        h_sa, c5_sa = fast_binary_sa(h_bin, n_sa, 0.4, 0.01, rng)
        h_cd, c5_cd = pairwise_cd(h_sa, 40, rng, full_sweeps=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        remaining = budget_s - (time.time() - start_time)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n1 = 2 * n0
        h_up = np.repeat(best_h, 2)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.02 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_cd_up, c5_cd_up = pairwise_cd(h_up, 30, rng, full_sweeps=3)
        if c5_cd_up < best_c5:
            best_c5 = c5_cd_up
            best_h = h_cd_up.copy()
            n0 = n1
            target_sum = n0 / 2.0

    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_cur = len(best_h)
        tx = n_cur / 2.0
        for eps in [0.002, 0.001, 0.0005, 0.0002]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, eps, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            h_cd2, c5_cd2 = pairwise_cd(best_h, 20, rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    remaining = budget_s - (time.time() - start_time)

    n_cur = len(best_h)
    final_h = project_onto_constraint(best_h, n_cur / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_cur
