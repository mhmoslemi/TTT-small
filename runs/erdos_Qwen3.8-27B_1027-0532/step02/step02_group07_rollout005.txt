def subgradient(h):
    """Compute a subgradient of C(h) = max_k f_k(h) at h."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-12)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    grad = np.zeros(n)
    for p in active:
        shift = p - (n - 1)
        # Term 1: (1-h[j-shift]) for valid j
        # j-shift in [0,n-1] => j in [shift, shift+n-1], intersect [0,n-1]
        j1_lo = max(0, shift)
        j1_hi = min(n, shift + n)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += (1.0 - h[j1_lo - shift : j1_hi - shift])
        # Term 2: -h[j+shift] for valid j
        # j+shift in [0,n-1] => j in [-shift, -shift+n-1], intersect [0,n-1]
        j2_lo = max(0, -shift)
        j2_hi = min(n, n - shift)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= h[j2_lo + shift : j2_hi + shift]
    
    grad /= (len(active) * 1.0)
    return dx * grad

def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j, slopes

    for _ in range(50):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:

    t0 = time.time()

                h[j] = t_opt

        if max_change < tol:
            improved += 1
            if improved > 3:

        else:
            improved = 0

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = h.copy()
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    mx = np.max(shifted)
    if mx > 50:
        shifted = shifted - mx
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_obj_fn(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return dx * smooth_max(corr, alpha)


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        smooth_obj_fn, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0}
    )
    return res.x if res.x is not None else None


def optimize_coordinate(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)
    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]
    a = corr - h_j * slopes
    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14
    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))
    if lo_V >= hi_V - 1e-14:
        return h_j, slopes
    for _ in range(50):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False
        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V_star = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))
    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5
    return float(np.clip(t_star, 0.0, 1.0)), slopes


def coordinate_descent(h, n_sweeps=200, tol=1e-12, time_budget=60.0, rng_seed=None):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()
    rng = np.random.RandomState(rng_seed if rng_seed is not None else 0)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        order = rng.permutation(n)
        for j in order:
            t_opt, slopes = optimize_coordinate(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-14:
                h[j] = t_opt
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 10 == 9:
            corr = compute_correlations(h)
        if max_change < tol:
            break
    h = project_sum(h, n / 2.0)
    return h


def subgradient_method(h, time_budget=100.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)
    step = 0.05
    for it in range(100000):
        if time.time() - t0 > time_budget:
            break
        corr = compute_correlations(h)
        max_val = np.max(corr)
        active = np.where(np.abs(corr - max_val) < 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for p in active:
            shift = p - (n - 1)
            j1_lo = max(0, shift)
            j1_hi = min(n, shift + n)
            if j1_lo < j1_hi:
                grad[j1_lo:j1_hi] += (1.0 - h[j1_lo - shift:j1_hi - shift])
            j2_lo = max(0, -shift)
            j2_hi = min(n, n - shift)
            if j2_lo < j2_hi:
                grad[j2_lo:j2_hi] -= h[j2_lo + shift:j2_hi + shift]
        grad /= len(active)
        h_new = h - step * grad * dx
        h_new = project_sum(h_new, n / 2.0)
        s = evaluate_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()
        h = h_new
        step *= 0.9999
    return best_h, best_score


def weighted_qp_optimize(h, time_budget=60.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    best_h = h.copy()
    best_score = evaluate_score(h)
    weights = np.ones(2 * n - 1)
    for it in range(500):
        if time.time() - t0 > time_budget:
            break
        corr = compute_correlations(h)
        max_val = np.max(corr)
        weights = np.exp(50.0 * (corr - max_val))
        weights /= np.sum(weights)
        def obj(hv):
            c = compute_correlations(hv)
            return float(np.sum(weights * c)) * dx
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 50, 'ftol': 1e-14})
        if res.x is not None:
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new
    return best_h, best_score


def interpolate_to(h, n_new):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_sum(h_new, n_new / 2.0)
    return h_new


def binary_flip_search(h, time_budget=20.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_sum(h, n / 2.0)
    corr = compute_correlations(h)
    current_max = float(np.max(corr))
    t0 = time.time()
    rng = np.random.RandomState(777)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for it in range(200000):
        if time.time() - t0 > time_budget:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        corr_new = compute_correlations(h)
        new_max = float(np.max(corr_new))
        if new_max < current_max - 1e-15:
            current_max = new_max
            corr = corr_new
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h

    if time_left() > 60:
        h_wq, s_wq = weighted_qp_optimize(best_h, time_budget=min(45, time_left() - 30))
        if s_wq < best_score:
            best_score = s_wq
            best_h = h_wq
            h_cd = coordinate_descent(best_h, n_sweeps=200, time_budget=min(20, time_left() - 30))
            s = evaluate_score(h_cd)
            if s < best_score:
                best_score = s
                best_h = h_cd

    # Phase 4: SLSQP with multiple alphas
    if time_left() > 80:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        for alpha in [100, 500, 2000, 10000, 50000]:
            if time_left() < 20:
                break
            maxiter = max(200, 300000 // n_cur)
            res_x = slsqp_optimize(best_h, alpha, maxiter, bounds)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_cur / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new

    # Phase 5: Multi-resolution search
    res_list = sorted(set([2 * n_base, 3 * n_base, int(n_base * 1.5)]), reverse=True)
    res_list = [r for r in res_list if 100 <= r <= 4000]
    for n_try in res_list:
        if time_left() < 100:
            break
        h_interp = interpolate_to(best_h, n_try)
        tb_cd = min(40, time_left() - 60)
        h_cd = coordinate_descent(h_interp, n_sweeps=300, time_budget=tb_cd)

            best_h = h_cd

                res_x = slsqp_optimize(h_cd, alpha, maxiter, bounds_try)

                best_h = h_cd2

        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 7: Binary flip search

        h_bin = binary_flip_search(best_h, time_budget=min(20, time_left() - 15))
        s = evaluate_score(h_bin)
        if s < best_score:
            best_score = s
            best_h = h_bin

    # Phase 8: Translation and reflection perturbations
    if time_left() > 40:
        n_cur = len(best_h)
        for shift_amt in [1, 2, 3, int(n_cur * 0.1), int(n_cur * 0.25)]:
            if time_left() < 15:
                break
            h_shift = np.roll(best_h, shift_amt)
            h_shift = project_sum(h_shift, n_cur / 2.0)
            h_cd = coordinate_descent(h_shift, n_sweeps=200, time_budget=min(8, time_left() - 10))

                best_h = h_cd
        h_rev = best_h[::-1].copy()
