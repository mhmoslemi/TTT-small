def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = float(np.max(x))
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-13:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_smooth_max(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)

        w = p[j]
        if w < 1e-15:

            n_terms = N - k
            grad[k:] += w * (1.0 - h[:n_terms])
            grad[:n_terms] -= w * h[k:]
        else:
            kp = -k
            n_terms = N - kp
            grad[:n_terms] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n_terms]
    return grad

    h[half] = 0.5

    grad_h = grad_smooth_max(h, alpha)
    half = len(g)

                    improved = True

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = float(np.max(x))
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-13:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_smooth_max(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N - 1)
        if k >= 0:
            n_terms = N - k
            grad[k:] += w * (1.0 - h[:n_terms])
            grad[:n_terms] -= w * h[k:]
        else:
            kp = -k
            n_terms = N - kp
            grad[:n_terms] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n_terms]
    return grad

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def ap_obj(g, alpha):
    h = antipal_h(g)
    return smooth_max_val(compute_correlations(h), alpha)

def ap_grad(g, alpha):
    h = antipal_h(g)
    grad_h = grad_smooth_max(h, alpha)
    half = len(g)
    return grad_h[:half] - grad_h[2 * half:half:-1]

def _coord_eval(g, i, val, alpha):
    old = g[i]
    g[i] = val
    h = antipal_h(g)
    s = smooth_max_val(compute_correlations(h), alpha)
    g[i] = old
    return s

def _coord_golden(g, i, alpha, n_iter=50):
    a = 0.0
    b = 1.0
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - inv_phi * (b - a)
    d = a + inv_phi * (b - a)
    fc = _coord_eval(g, i, c, alpha)
    fd = _coord_eval(g, i, d, alpha)
    for _ in range(n_iter):
        if abs(b - a) < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - inv_phi * (b - a)
            fc = _coord_eval(g, i, c, alpha)
        else:
            a = c
            c = d
            fc = fd
            d = a + inv_phi * (b - a)
            fd = _coord_eval(g, i, d, alpha)
    best_val = (a + b) / 2.0
    # Also check endpoints
    f_best = _coord_eval(g, i, best_val, alpha)
    for v in [0.0, 1.0]:
        f_v = _coord_eval(g, i, v, alpha)
        if f_v < f_best:
            f_best = f_v
            best_val = v
    return best_val

def coord_desc_ap(g, alpha, n_passes=5):
    half = len(g)
    g = g.copy()
    for _pass in range(n_passes):
        improved = False

            if abs(best_val - g[i]) > 1e-10:
                g[i] = best_val
                improved = True
        if not improved:
            break
    return g
