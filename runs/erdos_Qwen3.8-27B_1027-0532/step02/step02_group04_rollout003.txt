import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    if np.max(h) - np.min(h) < 1e-15:
        return np.full_like(h, target_sum / len(h))
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smax_objective(h, alpha, n_pts, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sm = m + np.log(np.sum(exp_shifted)) / alpha
    s = np.sum(h) - n_pts * 0.5
    return sm + mu * s * s

def smax_gradient(h, alpha, n_pts, mu):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    active = np.where(weights > 1e-9)[0]
    grad = np.zeros(N)
    for idx in active:
        w = weights[idx]
        lag = N - 1 - idx
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += w * (1.0 - h[:N - l])
            grad[:N - l] -= w * h[l:]
    
    s = np.sum(h) - n_pts * 0.5
    grad += 2.0 * mu * s
    return grad

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(smax_objective, h0, args=(alpha, n_pts, mu),
                   jac=smax_gradient, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    return res.x

def targeted_local_search(h, n, rng, n_iters, top_k=8, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr)
    
    for it in range(n_iters):
        top_indices = np.argsort(best_corr)[-top_k:]
        lag_idx = top_indices[rng.randint(top_k)]
        lag = n - 1 - lag_idx
        l = abs(lag)
        
        if 0 < l < n:
            i = rng.randint(0, n - l)
            step = step_sizes[rng.randint(len(step_sizes))]
            h_try = best.copy()
            if lag > 0:
                h_try[i] -= step
                h_try[i + l] += step
            else:
                h_try[i + l] -= step
                h_try[i] += step
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = step_sizes[rng.randint(len(step_sizes))]
            h_try = best.copy()
            h_try[i] += step
            h_try[j] -= step
        
        if np.any(h_try < -1e-10) or np.any(h_try > 1.0 + 1e-10):
            continue
        
        corr_try = compute_correlations(h_try)
        max_try = np.max(corr_try)
        if max_try < best_max - 1e-13:
            best = h_try
            best_corr = corr_try
            best_max = max_try
    
    return best, best_max

# Top-level helper functions

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smax_objective(h, alpha, n_pts, mu):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sm = m + np.log(np.sum(exp_shifted)) / alpha
    s = np.sum(h) - n_pts * 0.5
    return sm + mu * s * s

def smax_gradient(h, alpha, n_pts, mu):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    active = np.where(weights > 1e-9)[0]
    grad = np.zeros(N)
    for idx in active:
        w = weights[idx]
        lag = N - 1 - idx
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += w * (1.0 - h[:N - l])
            grad[:N - l] -= w * h[l:]
    
    s = np.sum(h) - n_pts * 0.5
    grad += 2.0 * mu * s
    return grad

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(smax_objective, h0, args=(alpha, n_pts, mu),
                   jac=smax_gradient, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    return res.x
