- For shift s: the term h[i]*(1-h[i+s]) disappears (if valid), and the term h[i-s]*(1-h[i]) changes from 0 to h[i-s] (if valid)
- So ΔC_i(s) = -(1-h[i+s]) * valid(i, s) + h[i-s] * valid(i-s, s)
  where valid(m, s) = 1 if 0<=m<n and 0<=m+s<n

The contribution of changing h[j] from 0 to 1:
- For shift s: the term h[j]*(1-h[j+s]) appears (if valid), and the term h[j-s]*(1-h[j]) changes from h[j-s] to 0 (if valid)
- So ΔC_j(s) = (1-h[j+s]) * valid(j, s) - h[j-s] * valid(j-s, s)

So for swap (i, j):
ΔC(s) = ΔC_i(s) + ΔC_j(s)
new_C(s) = C(s) + ΔC_i(s) + ΔC_j(s)
new_C5 = max_s new_C(s)

This is O(n) per swap (computing the max over 2n-1 shifts), and O(n^2) to precompute the contribution vectors (n/2 ones * n shifts + n/2 zeros * n shifts).

Total for one full pass: O(n^3) = O(5.8M) for n=180. Very fast!

    # s ranges from -(n-1) to n-1
    s_range = np.arange(-(n-1), n)  # length 2n-1
    # valid[m, s_idx] = 1 if 0 <= m < n and 0 <= m + s < n
    # m + s >= 0 => s >= -m => s_idx >= n-1-m
    # m + s <= n-1 => s <= n-1-m => s_idx <= 2n-2-m
    # So valid[m, s_idx] = 1 if (n-1-m) <= s_idx <= (2n-2-m)
    
    # For contribution vectors:
    # dC_i(s) = -(1-h[i+s]) * valid(i, s) + h[i-s] * valid(i-s, s)
    # where valid(m, s) means 0<=m<n and 0<=m+s<n
    
    # Let's precompute for each position p the vectors:
    # out_vec[p][s_idx] = contribution to corr[s] when h[p] goes from 1 to 0
    # in_vec[p][s_idx] = contribution to corr[s] when h[p] goes from 0 to 1
    
    # For position p going 1->0:
    # - Remove term p as first index: -(1-h[p+s]) for valid s (where p and p+s both in range)
    # - Add term p as second index: h[p-s] for valid s (where p-s and p both in range, h[p-s]=1)
    #   Wait, the term is h[p-s]*(1-h[p]) which was 0 (since h[p]=1), becomes h[p-s]*1
    #   So we add h[p-s] for s where p-s is valid first index and p is valid second index
    
    # For position p going 0->1:
    # - Add term p as first index: (1-h[p+s]) for valid s
    # - Remove term p as second index: -h[p-s] for valid s (was h[p-s]*(1-0)=h[p-s], becomes h[p-s]*(1-1)=0)
    
    def compute_contrib_out(p):
        """Contribution to corr when h[p] changes from 1 to 0."""
        vec = np.zeros(2*n-1)
        # p as first index: s ranges such that 0<=p<n (always) and 0<=p+s<n
        # p+s >= 0 => s >= -p, p+s < n => s < n-p
        # s_idx = s + n - 1, so s_idx ranges from n-1-p to n-1+p-1 = 2n-p-2... 
        # wait: s >= -p => s_idx >= n-1-p
        # s < n-p => s_idx < n-1+n-p = 2n-1-p => s_idx <= 2n-2-p
        s_min = max(0, n-1-p)
        s_max = min(2*n-2, 2*n-2-p)
        for s_idx in range(s_min, s_max+1):
            s = s_idx - (n-1)
            if 0 <= p+s < n:
                # valid, contribution is -(1-h[p+s])
                vec[s_idx] = -(1.0 - h[p+s])
        # p as second index: s such that 0<=p-s<n and 0<=p<n
        # p-s >= 0 => s <= p, p-s < n => s > p-n
        # s_idx = s + n - 1, so s_idx ranges from n-1+(p-n+1) to n-1+p = p to n-1+p
        # s > p-n => s >= p-n+1 => s_idx >= n-1+p-n+1 = p
        # s <= p => s_idx <= n-1+p
        s_min2 = max(0, p)
        s_max2 = min(2*n-2, n-1+p)
        for s_idx in range(s_min2, s_max2+1):
            s = s_idx - (n-1)
            m = p - s  # first index
            if 0 <= m < n:
                # contribution is h[m] (since h[p] was 1, (1-h[p])=0, now h[p]=0, (1-h[p])=1)
                # wait: the term is h[m]*(1-h[p]). Before: h[m]*(1-1)=0. After: h[m]*(1-0)=h[m]
                vec[s_idx] += h[m]
        return vec
    
    def compute_contrib_in(p):
        """Contribution to corr when h[p] changes from 0 to 1."""
        vec = np.zeros(2*n-1)
        # p as first index: add (1-h[p+s]) for valid s
        s_min = max(0, n-1-p)
        s_max = min(2*n-2, 2*n-2-p)
        for s_idx in range(s_min, s_max+1):
            s = s_idx - (n-1)
            if 0 <= p+s < n:
                vec[s_idx] = (1.0 - h[p+s])
        # p as second index: remove h[m] for valid s
        s_min2 = max(0, p)
        s_max2 = min(2*n-2, n-1+p)
        for s_idx in range(s_min2, s_max2+1):
            s = s_idx - (n-1)
            m = p - s
            if 0 <= m < n:
                vec[s_idx] -= h[m]
        return vec

Wait, let me double-check the gradient. 

corr[s] = sum_i h[i] * (1-h[i+s]) where the sum is over valid i.

d(corr[s])/dh[k]:
- If k is in the valid range as first index (i=k): contribution is (1-h[k+s])
- If k-s is in the valid range as first index (i=k-s, so i+s=k): contribution is -h[k-s]

So: d(corr[s])/dh[k] = (1-h[k+s]) * I(k valid, k+s valid) - h[k-s] * I(k-s valid, k valid)

Where "valid" means in [0, n-1].

For shift = s - (n-1):
- First term: k+s in [0,n-1] means k+shift in [0,n-1]. Combined with k in [0,n-1]: k in [0, n-1] ∩ [-shift, n-1-shift]
- Second term: k-s in [0,n-1] means k-shift in [0,n-1]. Combined with k in [0,n-1]: k in [shift, n-1] ∩ [0, n-1]

For shift >= 0:
- First term valid for k in [0, n-1-shift]: d = (1-h[k+shift])
- Second term valid for k in [shift, n-1]: d = -h[k-shift]

For shift < 0, let r = -shift > 0:
- First term valid for k in [r, n-1]: d = (1-h[k-r])  [since k+s = k-r must be in [0,n-1], so k >= r]
- Second term valid for k in [0, n-1-r]: d = -h[k+r]  [since k-s = k+r must be in [0,n-1], so k <= n-1-r]

from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def to_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x -= np.max(x)
    return m + eps * np.log(np.sum(np.exp(x)))

    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)

    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s] * dx
        if ws < 1e-15:
            continue
        shift = s - (n - 1)
        if shift >= 0:
            mc = n - shift
            grad[:mc] += ws * omh[shift:]
            grad[shift:] -= ws * h[:mc]
        else:
            r = -shift
            mc = n - r
            grad[r:] += ws * omh[:mc]
            grad[:mc] -= ws * h[r:]
    return grad

def binary_local_search(h, start_time, time_budget):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for iteration in range(100):
        if time.time() - start_time > start_time + time_budget:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        best_delta = 0.0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                delta = new_c5 - current_c5
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            current_c5 += best_delta
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def lbfgs_optimize(h_init, eps, maxiter, ftol):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smooth_obj, h0, args=(eps,),
            jac=smooth_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 30}
        )
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def continuation_optimize(h_init, eps_list, time_budget, start_time):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for eps in eps_list:
        elapsed = time.time() - start_time
        if elapsed > start_time + time_budget:
            break
        h_opt, c5_opt = lbfgs_optimize(h, eps, 2000, 1e-15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
    
    return best_h, best_c5
