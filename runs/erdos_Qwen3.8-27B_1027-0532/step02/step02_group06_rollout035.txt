import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha


def grad_smooth_max(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    threshold = 1e-10
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < threshold:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            ma = -k
            grad[:n - ma] += wi * (1.0 - h[ma:])
            grad[ma:] -= wi * h[:n - ma]
    return grad


def subgradient_run(h0, alpha, max_iters, step_init, t_deadline):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()

    for it in range(max_iters):

def slsqp_refine(h, alpha, maxiter=300):
    n = len(h)
    target_sum = n / 2.0

    def obj(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def obj_grad(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        w /= w.sum()
        grad = np.zeros(n)
        threshold = 1e-10
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < threshold:
                continue
            k = idx - (n - 1)
            if k == 0:
                grad += wi * (1.0 - 2.0 * hv)
            elif k > 0:
                grad[k:] += wi * (1.0 - hv[:n - k])
                grad[:n - k] -= wi * hv[k:]
            else:
                ma = -k
                grad[:n - ma] += wi * (1.0 - hv[ma:])
                grad[ma:] -= wi * hv[:n - ma]
        return grad

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    except Exception:
        pass
    return None


def coord_descent_sweep(h, n_sweeps, seed=42, t_deadline=None):
    n = len(h)
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()

    for sweep in range(n_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break
        order = rng.permutation(n)
        for i in order:
            curr = h[i]
            B = np.zeros(2 * n - 1)
            for k in range(1, n):
                b_k = 0.0
                if i >= k:
                    b_k += (1.0 - h[i - k])
                if i <= n - 1 - k:
                    b_k += (-h[i + k])
                B[n - 1 + k] = b_k
                b_neg = 0.0
                if i <= n - 1 - k:
                    b_neg += (1.0 - h[i + k])
                if i >= k:
                    b_neg += (-h[i - k])
                B[n - 1 - k] = b_neg

            corr = compute_correlations(h)
            A = corr - B * curr
            A_0 = np.sum(h * (1.0 - h)) - curr * (1.0 - curr)

            def f_1d(t):
                vals = A + B * t
                vals[n - 1] = A_0 + t - t * t
                return np.max(vals)

            res = minimize_scalar(f_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            h[i] = np.clip(res.x, 0.0, 1.0)

        h = project_to_feasible(h, target_sum)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    return best_h, best_score


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new / 2.0)
