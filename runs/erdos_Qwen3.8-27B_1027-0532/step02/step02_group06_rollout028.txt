def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):

    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):

    for _ in range(60):

        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    shifted = corr - np.max(corr)
    log_sum = np.log(np.sum(np.exp(beta * shifted)))
    return np.max(corr) + np.log(np.sum(np.exp(beta * shifted))) / beta

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta, weights

def compute_lag_gradient(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Use log-sum-exp trick
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = m + log_sum / beta
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    return obj, weights

        v_hat = v_buf / (1.0 - b2 ** it)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, target_sum)

        score = evaluate_score(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new

    return best_h, best_score


def get_active_lags(h, tol):
    n = len(h)
    corr = compute_correlations(h)
    mx = np.max(corr)
    center = n - 1
    active = []
    for k in range(len(corr)):
        if corr[k] >= mx - tol:
            active.append(k - center)
    return active


def slsqp_refine(h, active_lags, maxiter):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    center = n - 1
    active_idx = [lag + center for lag in active_lags]

    def obj_func(hv):
        corr = compute_correlations(hv)
        vals = np.array([corr[i] for i in active_idx])
        m = np.max(vals)
        return m * dx

    def obj_grad(hv):
        corr = compute_correlations(hv)
        vals = np.array([corr[i] for i in active_idx])
        m = np.max(vals)
        mask = vals >= m - 1e-8
        if np.sum(mask) == 0:
            mask = np.ones(len(vals), dtype=bool)
        w = mask / np.sum(mask)
        grad = np.zeros(n)
        for wi, idx in zip(w, active_idx):
            lag = idx - center
            if lag == 0:
                grad[:] += wi * (1.0 - 2.0 * hv)
            elif lag > 0:
                grad[lag:] += wi * (1.0 - hv[:-lag])
                grad[:n - lag] -= wi * hv[lag:]
            else:
                la = -lag
                grad[:n - la] += wi * (1.0 - hv[la:])
                grad[la:] -= wi * hv[:n - la]
        return grad * dx

    def cons_fun(hv):
        return np.sum(hv) - target_sum

    def cons_jac(hv):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                   jac=obj_grad,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None


def try_resolution(h_best, n_best, n_new, t_start, duration, seed):
    x_old = np.linspace(0.0, 2.0, n_best, endpoint=False) + 1.0 / n_best
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h_best, period=2.0)
    h_interp = project_to_feasible(h_interp, n_new / 2.0)
    h_out, score = adam_round(h_interp, n_new / 2.0, t_start, duration,
                              10.0, 60.0, 0.03, 0.75, 0.95, seed)
    return h_out, score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_score = evaluate_score(h)
    best_n = n

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Moderate exploration
    if time_left() > 200:
        dur = min(200, time_left() * 0.25)
        h_new, score = adam_round(best_h, target_sum, time.time(), dur,
                                  5.0, 40.0, 0.05, 0.75, 0.95, seed)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Phase 2: Higher beta exploitation
    if time_left() > 150:
        dur = min(200, time_left() * 0.25)
        h_start = best_h + rng.randn(n) * 0.005 / np.sqrt(n)
        h_start = project_to_feasible(h_start, target_sum)
        h_new, score = adam_round(h_start, target_sum, time.time(), dur,
                                  30.0, 120.0, 0.03, 0.8, 0.94, seed + 1)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Phase 3: Very high beta fine-tuning
    if time_left() > 100:
        dur = min(180, time_left() * 0.25)
        h_new, score = adam_round(best_h, target_sum, time.time(), dur,
                                  80.0, 300.0, 0.015, 0.85, 0.93, seed + 2)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Phase 4: SLSQP active-set polish
    if time_left() > 60:
        active = get_active_lags(best_h, 0.005)
        h_ref = slsqp_refine(best_h, active, maxiter=500)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Phase 5: Try one higher resolution
    if time_left() > 150:
        n_try = min(n + 200, 1000)
        if n_try > best_n:
            dur = min(100, time_left() * 0.15)
            h_new, score = try_resolution(best_h, best_n, n_try,
                                          time.time(), dur, seed + 100)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                best_n = n_try

    # Phase 6: Second SLSQP polish after higher resolution
    if time_left() > 50:
        active = get_active_lags(best_h, 0.003)
        h_ref = slsqp_refine(best_h, active, maxiter=800)
        if h_ref is not None:
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Phase 7: Final Adam with very high beta
    if time_left() > 30:
        dur = min(60, time_left() - 15)
        h_new, score = adam_round(best_h, len(best_h) / 2.0, time.time(), dur,
                                  150.0, 500.0, 0.008, 0.9, 0.92, seed + 999)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
