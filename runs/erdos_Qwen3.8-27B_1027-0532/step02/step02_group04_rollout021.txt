def weighted_autocorr_gradient(h, weights):
    """Compute sum_k weights[k] * grad R(k) for k=0,...,n-1.
    weights should be a vector of length n (weights[0] can be 0)."""

        w = weights[k]
        grad[:-k] += w * h[k:]
        grad[k:] += w * h[:-k]
    return grad

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = smooth_max_gradient_for_correlations(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def autocorrelation(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    return np.fft.irfft(H * np.conj(H), N)[:n]


def smooth_max_value(vals, alpha):
    m = np.max(vals)
    shifted = alpha * (vals - m)
    return m + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / alpha


def smooth_max_gradient_for_correlations(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def weighted_autocorr_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    if weights[0] > 1e-10:
        grad += 2.0 * weights[0] * h
    for k in range(1, n):
        if weights[k] < 1e-10:
            continue
        w = weights[k]
        grad[:-k] += w * h[k:]
        grad[k:] += w * h[:-k]
    return grad


def subgradient_optimize(h0, n_iters, alpha, momentum_coef, lr_init, lr_min, lr_decay):
    n = len(h0)
    best = h0.copy()
    best_score = evaluate_score(best)
    momentum = np.zeros(n)
    lr = lr_init
    stagnant = 0

    for it in range(n_iters):
        R = autocorrelation(best)
        weights_full = np.zeros(n)
        r_vals = R[1:]
        shifted = -alpha * (r_vals - np.min(r_vals))
        max_shifted = np.max(shifted)
        weights_full[1:] = np.exp(shifted - max_shifted)
        wsum = weights_full.sum()
        if wsum > 0:
            weights_full /= wsum

        grad = weighted_autocorr_gradient(best, weights_full)
        momentum = momentum_coef * momentum + (1.0 - momentum_coef) * grad
        h_new = best + lr * momentum
        h_new = project_feasible(h_new, n / 2.0)
        score_new = evaluate_score(h_new)

        if score_new < best_score:
            best = h_new
            best_score = score_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 80:
                lr = max(lr_min, lr * lr_decay)
                stagnant = 0

        if lr <= lr_min * (1 + 1e-6):
            break

    return best, best_score


def subgradient_topk(h0, n_iters, K, lr_init, lr_min, lr_decay):
    n = len(h0)
    best = h0.copy()
    best_score = evaluate_score(best)
    momentum = np.zeros(n)
    lr = lr_init
    stagnant = 0

    for it in range(n_iters):
        R = autocorrelation(best)
        r_vals = R[1:]
        worst_indices = np.argsort(r_vals)[:K]
        weights_full = np.zeros(n)
        for idx in worst_indices:
            weights_full[idx + 1] = 1.0 / K

        grad = weighted_autocorr_gradient(best, weights_full)
        momentum = 0.85 * momentum + 0.15 * grad
        h_new = best + lr * momentum
        h_new = project_feasible(h_new, n / 2.0)
        score_new = evaluate_score(h_new)

        if score_new < best_score:
            best = h_new
            best_score = score_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 60:
                lr = max(lr_min, lr * lr_decay)
                stagnant = 0

        if lr <= lr_min * (1 + 1e-6):
            break

    return best, best_score


def ljbfgs_call(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient_for_correlations(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def local_search_pair(h, n_pts, n_iters, rng, step_sizes):
    best = h.copy()
    best_score = evaluate_score(best)
    for _ in range(n_iters):
        i = rng.randint(n_pts)
        j = rng.randint(n_pts)
        if i == j:
            continue
        for step in step_sizes:
            h_try = best.copy()
            h_try[i] = best[i] + step
            h_try[j] = best[j] - step
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            s_try = evaluate_score(h_try)
            if s_try < best_score - 1e-14:
                best = h_try
                best_score = s_try
                break
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    best_h = project_feasible(init, n / 2.0)
    best_score = evaluate_score(best_h)

    MU = 50000.0

    # Phase 1: Subgradient optimization with diverse alpha and top-K
    for alpha in [3, 8, 15, 30, 60, 120]:
        if time.time() - t0 > budget_s - 500:
            break
        n_iters = min(4000, int(150 * (100 / (alpha + 1)) + 500))
        h_sub, s_sub = subgradient_optimize(best_h, n_iters, alpha,
                                            momentum_coef=0.85, lr_init=0.015,
                                            lr_min=0.0005, lr_decay=0.93)
        h_sub = project_feasible(h_sub, n / 2.0)
        s_sub = evaluate_score(h_sub)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()

    # Top-K subgradient variants
    for K in [2, 3, 5, 8]:
        if time.time() - t0 > budget_s - 450:
            break
        h_sub, s_sub = subgradient_topk(best_h, 2000, K,
                                        lr_init=0.012, lr_min=0.0005, lr_decay=0.93)
        h_sub = project_feasible(h_sub, n / 2.0)
        s_sub = evaluate_score(h_sub)
        if s_sub < best_score:
            best_score = s_sub
            best_h = h_sub.copy()

    # Phase 2: L-BFGS-B progressive polish
    for alpha in [50, 150, 400, 1000, 2500, 6000]:
        if time.time() - t0 > budget_s - 350:
            break
        maxiter = max(150, 100000 // n)
        x = ljbfgs_call(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: Multi-start with diverse strategies
    n_restarts = 250
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 200:
            break

        strategy = i % 7
        if strategy == 0:
            scale = 0.015 + 0.06 * (i / float(n_restarts))
            pert = best_h + rng.randn(n) * scale
        elif strategy == 1:
            pert = best_h.copy()
            blen = rng.randint(4, min(35, n // 5))
            bstart = rng.randint(0, n - blen)
            pert[bstart:bstart + blen] = 1.0 - pert[bstart:bstart + blen]
        elif strategy == 2:
            pert = best_h.copy()
            n_freq = rng.randint(1, 6)
            x_idx = np.arange(n, dtype=float)
            for k in range(n_freq):
                f = rng.randint(1, 20)
                amp = rng.uniform(0.008, 0.06)
                phase = rng.rand() * 2.0 * np.pi
                pert += amp * np.sin(2.0 * np.pi * f * x_idx / n + phase)
        elif strategy == 3:
            n_coarse = max(25, n // 4)
            x_fine = np.linspace(0, 2, n, endpoint=False)
            x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
            coarse = np.interp(x_coarse, x_fine, best_h)
            coarse = project_feasible(coarse, n_coarse / 2.0)
            pert = np.interp(x_fine, x_coarse, coarse)
        elif strategy == 4:
            pert = best_h.copy()
            n_blocks = rng.randint(2, 7)
            block_size = n // n_blocks
            for b in range(n_blocks):
                s0 = b * block_size
                s1 = min(s0 + block_size, n)
                if rng.rand() < 0.4:
                    pert[s0:s1] = 1.0 - pert[s0:s1]
                else:
                    pert[s0:s1] += rng.randn(s1 - s0) * rng.uniform(0.03, 0.1)
        elif strategy == 5:
            pert = best_h + rng.randn(n) * 0.04
            pert, _ = subgradient_topk(pert, 300, K=3,
                                       lr_init=0.01, lr_min=0.001, lr_decay=0.9)
        else:
            pert = best_h.copy()
            n_swap = rng.randint(5, n // 4)
            idx = rng.choice(n, n_swap, replace=False)
            pert[idx] = 1.0 - pert[idx]

        pert = project_feasible(pert, n / 2.0)

        # Subgradient warm-up
        alpha_sub = [5, 15, 40, 100][rng.randint(4)]
        h_sub, s_sub = subgradient_optimize(pert, 400, alpha_sub,
                                            momentum_coef=0.8, lr_init=0.01,
                                            lr_min=0.0005, lr_decay=0.94)
        h_sub = project_feasible(h_sub, n / 2.0)
        s_sub = evaluate_score(h_sub)
        candidate = h_sub if s_sub < evaluate_score(pert) else pert

        if time.time() - t0 > budget_s - 150:
            if s_sub < best_score:
                best_score = s_sub
                best_h = h_sub.copy()
            continue

        # L-BFGS-B refinement
        alpha_lbfgs = [80, 250, 600, 1500][rng.randint(4)]
        maxiter_r = max(60, 60000 // n)
        x = ljbfgs_call(candidate, alpha_lbfgs, n, MU, maxiter_r)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 4: Resolution increase
    for n_try in [2 * n, 3 * n]:
        if n_try > 2200:
            continue
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n_try / 2.0)

        # Subgradient at new resolution
        h_sub, s_sub = subgradient_optimize(h_init, 2000, alpha=15,
                                            momentum_coef=0.85, lr_init=0.012,
                                            lr_min=0.0008, lr_decay=0.95)
        h_sub = project_feasible(h_sub, n_try / 2.0)
        s_sub = evaluate_score(h_sub)
        current = h_sub if s_sub < evaluate_score(h_init) else h_init

        for alpha in [150, 600, 2000, 8000]:
            if time.time() - t0 > budget_s - 60:
                break
            maxiter_n = max(50, 300000 // n_try)
            x = ljbfgs_call(current, alpha, n_try, MU, maxiter_n)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    current = h_new.copy()

    # Phase 5: Final high-alpha polish
    n_final = len(best_h)
    for alpha in [8000, 20000, 50000]:
        if time.time() - t0 > budget_s - 25:
            break
        maxiter_f = max(200, 250000 // n_final)
        x = ljbfgs_call(best_h, alpha, n_final, MU, maxiter_f)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_final / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 6: Local search
    if time.time() - t0 < budget_s - 10:
        n_local = len(best_h)
        remaining = budget_s - 12 - (time.time() - t0)
        n_ls = int(remaining * 120)
        n_ls = max(0, min(n_ls, 30000))
        if n_ls > 0:
            steps = [0.004, 0.002, 0.001, 0.0005, 0.0002]
            best_h, _ = local_search_pair(best_h, n_local, n_ls, rng, steps)
            best_h = project_feasible(best_h, n_local / 2.0)
            best_score = evaluate_score(best_h)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
