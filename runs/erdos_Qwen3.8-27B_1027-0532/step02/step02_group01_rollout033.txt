The subgradient of max_k corr[k] with respect to h[i] is:

∂corr[k] / ∂h[i] = (1-h[i+s]) - h[i-s] (with appropriate boundary conditions)

This is O(n) per evaluation (for a specific k).

So the subgradient method would be:
1. Compute corr array
2. Find k* = argmax
3. Compute gradient of corr[k*] (O(n))
4. Update: h_new = clip(h - lr * grad, 0, 1), then project onto sum constraint
5. Repeat

This is much faster than the log-sum-exp approach! And for the max problem, subgradient methods can work well, especially with averaging.

def subgradient_step(h, k_star, n):
    """Compute the subgradient of corr[k_star] with respect to h."""
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        # corr[k] = sum_{i=0}^{n-s-1} h[i] * (1-h[i+s])
        # d/dh[i] = (1-h[i+s]) for i in [0, m-1]
        # d/dh[i] = -h[i-s] for i in [s, n-1]
        grad[:m] += (1.0 - h[s:s+m])  # wait, h[i+s] for i in [0, m-1] is h[s:s+m]
        grad[s:] -= h[:m]  # h[i-s] for i in [s, n-1] is h[0:n-s] = h[:m]
    else:
        r = -s
        m = n - r
        # corr[k] = sum_{i=r}^{n-1} h[i] * (1-h[i-r])
        # Let j = i-r, so j in [0, m-1], i = j+r
        # corr[k] = sum_{j=0}^{m-1} h[j+r] * (1-h[j])
        # d/dh[j] = -h[j+r] for j in [0, m-1]  (from the (1-h[j]) term)
        # d/dh[j+r] = (1-h[j]) for j in [0, m-1]  (from the h[j+r] term)
        grad[r:r+m] += (1.0 - h[:m])  # (1-h[j]) for j in [0, m-1]
        grad[:m] -= h[r:r+m]  # -h[j+r] for j in [0, m-1]
    return grad

```python
def subgradient_descent(h_init, n_steps, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    h_sum = np.zeros(n)  # for averaging
    lr_0 = 0.5
    
    for t in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h)
        k_star = np.argmax(corr)
        grad = subgradient_step(h, k_star, n)
        lr = lr_0 / (1 + 0.01 * t)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        h_sum += h_new
        h_avg = h_sum / (t + 1)
        c5_avg = compute_c5(h_avg)
        c5_curr = compute_c5(h_new)
        h = h_new
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg.copy()
    
    return best_h, best_c5
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    h_bin = np.zeros(n)

def binary_sa(h_init, n_steps, t_start, t_end, rng, time_limit=None, start_time=None):
    n = len(h_init)
    h = h_init.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    no_improve = 0
    
    for step in range(n_steps):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        oi = int(ones[rng.integers(len(ones))])
        zi = int(zeros[rng.integers(len(zeros))])
        
        h[oi] = 0.0
        h[zi] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[oi] = 1.0
            h[zi] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def subgradient_descent(h_init, n_steps, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    h_avg = np.zeros(n)
    lr_decay = 0.999
    
    lr = 0.1
    for step in range(n_steps):
        if (time.time() - start_time) > time_limit:
            break
        
        corr = compute_corr_array(h)
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        
        # Compute subgradient of corr[k_star]
        grad = np.zeros(n)
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:s+m])
            grad[s:] -= h[:m]
        else:
            r = -s
            m = n - r
            grad[r:r+m] += (1.0 - h[:m])
            grad[:m] -= h[r:r+m]
        
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        h_avg += (h_new - h_avg) / (step + 1)
        h = h_new
        
        c5_curr = compute_c5(h)
        if c5_curr < best_c5:

            best_h = h.copy()
        
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg.copy()
        
        lr *= lr_decay
    
    return best_h, best_c5
