            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            cur_score = compute_c5(h)
            if score >= cur_score:
                lr = max(lr * 0.5, 1e-8)
                h_new = h - lr * momentum
                h_new = euclidean_project(h_new, target_sum)
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
                    stall = 0
        h = h_new
        lr = max(lr0 * (decay ** it), 1e-8)
        if stall > 800 and it > 500:
            break
    return best_h, best_score


def run_slsqp_phase(h_start, n, target, alphas, maxiter, ftol):
    global _c5_n, _c5_target, _c5_alpha
    _c5_n = n
    _c5_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        _c5_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = euclidean_project(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return euclidean_project(h_src.copy(), target_dst)
    n_fft = max(1024, 1 << (max(n_src, n_dst) - 1).bit_length())
    pad_src = n_fft - n_src
    pad_dst = n_fft - n_dst
    F_src = np.fft.rfft(h_src, n_fft)
    F_dst = np.zeros(n_fft // 2 + 1, dtype=complex)
    if n_dst > n_src:
        for j in range(n_src):
            F_dst[j] = F_src[j]
    else:
        for j in range(n_dst):
            F_dst[j] = F_src[j]
    h_dst = np.fft.irfft(F_dst, n_fft)[:n_dst]
    return euclidean_project(h_dst, target_dst)


def generate_perturbation(h, rng, pert_type, n):
    if pert_type == 0:
        return h + rng.randn(n) * 0.04
    elif pert_type == 1:
        h_p = h.copy()
        n_flips = max(1, n // 25)
        idx_1 = np.where(h_p > 0.3)[0]
        idx_0 = np.where(h_p <= 0.7)[0]
        if len(idx_1) >= n_flips and len(idx_0) >= n_flips:
            f1 = rng.choice(idx_1, size=n_flips, replace=False)
            f0 = rng.choice(idx_0, size=n_flips, replace=False)
            h_p[f1] = 0.0
            h_p[f0] = 1.0
        return h_p
    elif pert_type == 2:
        h_p = h.copy()
        n_blocks = rng.randint(2, 5)
        for _ in range(n_blocks):
            bs = rng.randint(3, 15)
            st = rng.randint(0, max(1, n - bs))
            h_p[st:st + bs] += rng.choice([-1.0, 1.0]) * rng.uniform(0.1, 0.25)
        return h_p
    elif pert_type == 3:
        h_p = h.copy()
        h_p[rng.randint(0, n)] += rng.uniform(-0.3, 0.3)
        return h_p
    else:
        h_p = h.copy()
        perm = rng.permutation(n)
        h_p[perm[:n // 8]] += rng.uniform(-0.15, 0.15, size=n // 8)
        return h_p


def run(seed=42, budget_s=1300, **kwargs):
    global _c5_n, _c5_target, _c5_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = euclidean_project(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: PGD refinement on initial solution
    if tleft() > 300:
        h1, s1 = run_pgd(best_h, target_init, 8000, K=8, lr0=0.004, decay=0.9998, beta=0.75)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: SLSQP polish
    if tleft() > 200:
        h2, s2 = run_slsqp_phase(best_h, n_init, target_init,
                                  alphas=[30, 100, 300, 800, 1500], maxiter=800, ftol=1e-14)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Multi-start with diverse perturbations
    n_restarts = 15
    for r in range(n_restarts):
        if tleft() < 150:
            break
        pert_type = r % 5
        h_pert = generate_perturbation(best_h, rng, pert_type, n_init)
        h_pert = euclidean_project(h_pert, target_init)
        h_new, s_new = run_pgd(h_pert, target_init, 4000, K=8, lr0=0.003, decay=0.9997, beta=0.8)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new
        if tleft() > 80:
            h_new2, s_new2 = run_slsqp_phase(best_h, n_init, target_init,
                                              alphas=[200, 600, 1200, 2500], maxiter=500, ftol=1e-14)
            if s_new2 < best_score:
                best_score = s_new2
                best_h = h_new2

    # Phase 4: Higher resolution attempts
    for n_try in [550, 700, 900]:
        if tleft() < 250:
            break
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_new, s_new = run_pgd(h_interp, target_try, 4000, K=10, lr0=0.003, decay=0.9998, beta=0.75)
        if tleft() > 100:
            h_new, s_new = run_slsqp_phase(h_new, n_try, target_try,
                                            alphas=[100, 400, 1000, 2000], maxiter=600, ftol=1e-14)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    # Phase 5: Final fine-tuning at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 100:
        h_fin, s_fin = run_slsqp_phase(best_h, n_use, target_use,
                                        alphas=[500, 1000, 2000, 4000, 8000], maxiter=800, ftol=1e-15)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    # Phase 6: Final PGD micro-tuning
    if tleft() > 40:
        h_fin2, s_fin2 = run_pgd(best_h, target_use, 3000, K=5, lr0=0.001, decay=0.9999, beta=0.9)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2

    h_final = euclidean_project(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
