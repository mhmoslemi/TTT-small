import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)

    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-10:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad


def lbfgs_optimize(h0, alpha, n_pts, mu, maxiter):
    def _obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s

    def _grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x


def binarize(h):
    n = len(h)
    target = n / 2.0
    sorted_idx = np.argsort(h)
    result = np.zeros(n)
    count = 0.0
    for i in range(n):
        idx = sorted_idx[n - 1 - i]
        remaining = target - count
        if remaining >= 1.0:
            result[idx] = 1.0
            count += 1.0
        elif remaining > 0:
            result[idx] = remaining
            break
    return result

    if sa_iters > 100:
        best_bin, _ = simulated_annealing(h_bin, sa_iters, rng, deadline_sa)
        bin_score = evaluate_score(best_bin)
        if bin_score < best_score:
            best_score = bin_score
            best_h = best_bin.copy()

    # Phase 2: L-BFGS-B with increasing alpha at original resolution
    deadline_p2 = t0 + budget_s * 0.25
    alphas_p2 = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
    for alpha in alphas_p2:
        if time.time() > deadline_p2:
            break
        maxiter = max(100, int(100000 / n))
        x = lbfgs_optimize(best_h, alpha, n, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, target)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 3: Diverse restarts
    deadline_p3 = t0 + budget_s * 0.55
    n_restarts = 300
    for i in range(n_restarts):
        if time.time() > deadline_p3:
            break

        strategy = i % 7
        if strategy == 0:
            scale = 0.02 + 0.1 * (i / float(n_restarts))
            pert = best_h + rng.randn(n) * scale
        elif strategy == 1:
            pert = best_h.copy()
            blen = rng.randint(4, min(30, n // 5))
            bstart = rng.randint(0, n - blen)
            pert[bstart:bstart + blen] = 1.0 - pert[bstart:bstart + blen]
        elif strategy == 2:
            pert = best_h.copy()
            x_idx = np.arange(n, dtype=float)
            n_freq = rng.randint(1, 6)
            for k in range(n_freq):
                f = rng.randint(1, 20)
                amp = rng.uniform(0.02, 0.15)
                phase = rng.rand() * 2.0 * np.pi
                pert += amp * np.sin(2.0 * np.pi * f * x_idx / n + phase)
        elif strategy == 3:
            pert = binarize(best_h + rng.randn(n) * 0.15)
        elif strategy == 4:
            n_coarse = max(40, n // 5)
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
            coarse = np.interp(x_coarse, x_old, best_h)
            coarse = project_feasible(coarse, n_coarse / 2.0)
            coarse += rng.randn(n_coarse) * 0.15
            coarse = project_feasible(coarse, n_coarse / 2.0)
            pert = np.interp(x_old, x_coarse, coarse)
        elif strategy == 5:
            pert = best_h.copy()
            n_blocks = rng.randint(1, 5)
            for _ in range(n_blocks):
                blen = rng.randint(4, min(20, n // 8))
                bstart = rng.randint(0, n - blen)
                pert[bstart:bstart + blen] = best_h[bstart:bstart + blen][::-1]
        else:
            pert = best_h.copy()
            n_swap = rng.randint(2, 8)
            ones = np.where(pert > 0.4)[0]
            zeros = np.where(pert < 0.6)[0]
            for _ in range(n_swap):
                if len(ones) > 0 and len(zeros) > 0:
                    oi = ones[rng.randint(len(ones))]
                    zi = zeros[rng.randint(len(zeros))]
                    if oi != zi:
                        pert[oi] -= 0.5
                        pert[zi] += 0.5
                        ones = np.where(pert > 0.4)[0]
                        zeros = np.where(pert < 0.6)[0]

        pert = project_feasible(pert, target)
        alpha_pick = [30, 80, 200, 500, 1500, 5000][rng.randint(6)]
        maxiter_r = max(60, int(80000 / n))
        x = lbfgs_optimize(pert, alpha_pick, n, MU, maxiter_r)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, target)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 4: Higher resolution
    deadline_p4 = t0 + budget_s * 0.82
    for n_try in [2 * n]:
        if n_try > 1200 or time.time() > deadline_p4:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_try / 2.0)

        for alpha in [50, 200, 800, 3000, 10000, 30000]:
            if time.time() > deadline_p4:
                break
            maxiter_hi = max(30, int(600000 / n_try))
            x = lbfgs_optimize(h_hi, alpha, n_try, MU, maxiter_hi)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_hi = h_new.copy()

        # Lag-aware local search at high resolution
        if time.time() < deadline_p4:
            n_ls = min(5000, int((deadline_p4 - time.time()) * 80))
            if n_ls > 200:
                best_h, _ = lag_aware_local_search(best_h, n_ls, rng, deadline_p4)
                best_h = project_feasible(best_h, len(best_h) / 2.0)
                s = evaluate_score(best_h)
                if s < best_score:
                    best_score = s

    # Phase 5: Final high-alpha polish
    deadline_p5 = t0 + budget_s - 60
    n_final = len(best_h)
    target_final = n_final / 2.0
    for alpha in [30000, 80000, 200000]:
        if time.time() > deadline_p5:
            break
        maxiter_f = max(100, int(400000 / n_final))
        x = lbfgs_optimize(best_h, alpha, n_final, MU, maxiter_f)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, target_final)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 6: Final local search (lag-aware + mixed)
    deadline_final = t0 + budget_s - 10
    remaining = deadline_final - time.time()
    if remaining > 5:
        n_ls1 = min(8000, int(remaining * 0.6 * 100))
        if n_ls1 > 100:
            best_h, _ = lag_aware_local_search(best_h, n_ls1, rng, deadline_final)
            best_h = project_feasible(best_h, len(best_h) / 2.0)
            best_score = evaluate_score(best_h)

        remaining2 = deadline_final - time.time()
        if remaining2 > 3:
            n_ls2 = min(8000, int(remaining2 * 0.5 * 80))
            if n_ls2 > 100:
                best_h, _ = local_search_mixed(best_h, n_ls2, rng, deadline_final)
                best_h = project_feasible(best_h, len(best_h) / 2.0)
                best_score = evaluate_score(best_h)

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, n_final)
