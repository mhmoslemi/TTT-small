```python
    import cvxpy as cp
    dx = 2.0 / n
    h = cp.Variable(n)

    for k in range(n):  # all lags

        # corr_k = sum_{i} h[i] * (1-h[i+k]) (with appropriate ranges)
        # Linearize: h[i]*h[i+k] ≈ h_prev[i]*h[i+k] + h[i]*h_prev[i+k] - h_prev[i]*h_prev[i+k]

            # Linearize h[i]^2 ≈ h_prev[i]*h[i] + h[i]*h_prev[i] - h_prev[i]^2
```

- h[i]*h[j] ≤ min(h[i], h[j]) ≤ (h[i] + h[j])/2
- h[i]*h[j] ≥ max(0, h[i] + h[j] - 1)

These are the valid inequalities for the bilinear term.

So: corr_k = sum h[i]*(1-h[i+k]) = sum h[i] - sum h[i]*h[i+k]

Using the upper bound on h[i]*h[i+k]:
corr_k ≤ sum h[i] - sum max(0, h[i] + h[i+k] - 1)

This gives a valid upper bound on corr_k (i.e., if the upper bound ≤ C, then corr_k ≤ C).

Similarly, using the lower bound:
corr_k ≥ sum h[i] - sum min(h[i], h[i+k])

This gives a valid lower bound.

So we can formulate an LP:
minimize C
subject to: sum_{i in overlap_k} h[i] - sum_{i in overlap_k} max(0, h[i] + h[i+k] - 1) ≤ C (upper bound on corr_k)
           0 ≤ h_i ≤ 1
           sum h_i = n/2

t_i ≥ 0, t_i ≥ h[i] + h[i+k] - 1
Then corr_k ≤ sum h[i] - sum t_i

The LP becomes:
minimize C
subject to: sum_{i} h[i] - sum_{i} t_i ≤ C (for each k)
           t_i ≥ h[i] + h[i+k] - 1
           t_i ≥ 0
           0 ≤ h_i ≤ 1
           sum h_i = n/2

This is a valid LP! And it gives an upper bound on the true C₅ (because it uses a relaxation).

But we want a LOWER bound (a construction that achieves C₅ ≤ some value). The LP gives us an h that we can then evaluate to get the true C₅.

So the algorithm is:
1. Solve the LP to get h
2. Evaluate the true C₅ for this h
3. Use this h as a starting point for further local optimization

```python
def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj
```

```python
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

    """Project h onto anti-palindromic structure."""
    n = len(h)
    half = (n - 1) // 2
    g = 0.5 * (h[:half] + (1.0 - h[n-1:half-1:-1]))
    return antipal_h(np.clip(g, 0, 1))

    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

    """Gradient of smooth max of correlations w.r.t. h."""
    N = len(h)

    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)

        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad

    grad_g = grad_h[:half].copy()
    for i in range(half):
        j = 2 * half - i
        grad_g[i] += -grad_h[j]
    return grad_g

    """FISTA for anti-palindromic optimization."""
    N = len(antipal_h(g0))
    g = g0.copy()
    z = g0.copy()
    t_k = 1.0
    best_g = g0.copy()
    best_score = evaluate_score(antipal_h(g0))
    
    # Lipschitz estimate
    L = alpha * N / 2  # rough estimate
    
    for it in range(max_iter):

        lr = 1.0 / L
        g_new = np.clip(z - lr * gz, 0.0, 1.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z_new = g_new + ((t_k - 1.0) / t_new) * (g_new - g)
        z_new = np.clip(z_new, 0.0, 1.0)
        
        score = evaluate_score(antipal_h(z_new))
        if score < best_score:
            best_score = score
            best_g = z_new.copy()
        
        g = g_new
        z = z_new
        t_k = t_new
    
    return best_g, best_score

def slsqp_antipal(g0, alpha, maxiter=300):
    """SLSQP for anti-palindromic optimization."""
    half = len(g0)

        return np.clip(res.x, 0.0, 1.0)
    return g0
```

For a more accurate estimate, I can use the backtracking line search in FISTA:
- Start with a large step size
- If the function doesn't decrease, halve the step size
- This ensures convergence

Let me implement backtracking FISTA:

```python
def fista_antipal_bt(g0, alpha, max_iter=5000, ftol=1e-14):
    """FISTA with backtracking for anti-palindromic optimization."""
    g = g0.copy()
    z = g0.copy()
    t_k = 1.0
    best_g = g0.copy()
    best_score = evaluate_score(antipal_h(g0))
    
    lr = 1.0 / (alpha * len(g))  # initial step size
    
    for it in range(max_iter):
        # Compute gradient at z
        gz = grad_corr_g(z, alpha)
        
        # Backtracking line search
        g_candidate = np.clip(z - lr * gz, 0.0, 1.0)
        f_candidate = smooth_max_val(compute_correlations(antipal_h(g_candidate)), alpha)
        f_z = smooth_max_val(compute_correlations(antipal_h(z)), alpha)
        # Descent condition (simplified for projected case)
        while f_candidate > f_z + 0.5 * lr * np.sum(gz**2):
            lr *= 0.5
            g_candidate = np.clip(z - lr * gz, 0.0, 1.0)
            f_candidate = smooth_max_val(compute_correlations(antipal_h(g_candidate)), alpha)
            if lr < 1e-10:
                break
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z_new = g_candidate + ((t_k - 1.0) / t_new) * (g_candidate - g)
        z_new = np.clip(z_new, 0.0, 1.0)
        
        score = evaluate_score(antipal_h(z_new))
        if score < best_score:
            best_score = score
            best_g = z_new.copy()
        
        g = g_candidate
        z = z_new
        t_k = t_new
        
        # Convergence check
        if it > 100 and abs(score - best_score) < ftol:
            break
    
    return best_g, best_score
```

- Gradient computation: O(n²) as well
- Each FISTA iteration: a few ms

Let me also think about what the "Lipschitz constant" should be. The gradient of the smooth max involves:
∂/∂h_i [softmax_α(corr)] = sum_k p_k * ∂corr_k/∂h_i

And ∂corr_k/∂h_i is either 0, 1, -h_j, or -(1-h_j) for various j. The magnitude is at most 1.

So ||∇ softmax_α(corr(h))|| ≤ sum_k p_k * ||∂corr_k/∂h|| ≤ max_i ||∂corr_k/∂h_i|| * (number of non-zero terms)

For a fixed k, ∂corr_k/∂h_i is non-zero for at most 2 values of i (the pair (i, i+k) or (i-k, i)). So the gradient of corr_k w.r.t. h has at most 2(n-k) non-zero entries, each with magnitude ≤ 1.

The gradient of the smooth max is a convex combination of these, so its infinity norm is at most 1 (approximately). But the Lipschitz constant of the gradient (i.e., the bound on the Hessian) is different.

For the smooth max, the Hessian involves second derivatives of the correlations (which are 0 since correlations are bilinear) plus terms from the softmax (which involve p_k). The Hessian of the softmax part has norm roughly alpha * max(p_k).

So the total Lipschitz constant is roughly α. For α=10000, the step size should be about 1/10000 = 0.0001.

But this is for the full h-space. In the g-space (anti-palindromic), it's similar.

Let me use lr = 1/(alpha * 2) as a safe initial step size, and let backtracking handle the rest.

Actually, for the bilinear terms, the Hessian is not bounded by α alone. Let me just use the backtracking approach and not worry about the exact Lipschitz constant.

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half+1:] = 1.0 - g[::-1]
    return h

def make_antipal(h):
    n = len(h)
    half = (n - 1) // 2
    g = 0.5 * (h[:half] + (1.0 - h[n-1:half-1:-1]))
    return antipal_h(np.clip(g, 0.0, 1.0))

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def grad_smoothmax_h(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    m = float(np.max(corr))
    ex = np.exp(alpha * (corr - m))
    p = ex / np.sum(ex)
    grad = np.zeros(N)
    for j in range(2*N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (N-1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N-k])
            grad[:N-k] -= w * h[k:]
        else:
            kp = -k
            grad[:N-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N-kp]
    return grad

def grad_smoothmax_g(g, alpha):
    h = antipal_h(g)
    gh = grad_smoothmax_h(h, alpha)
    half = len(g)
    gg = gh[:half].copy()
    for i in range(half):
        gg[i] += -gh[2*half - i]
    return gg

def fista_ap(g0, alpha, max_iter, use_bt=True):
    g = g0.copy()
    z = g0.copy()
    t_k = 1.0
    best_g = g0.copy()
    best_s = evaluate_score(antipal_h(g0))
    lr = 1.0 / (alpha * 2.0)
    
    for it in range(max_iter):
        gz = grad_smoothmax_g(z, alpha)
        g_new = np.clip(z - lr * gz, 0.0, 1.0)
        
        if use_bt:
            # Simple backtracking
            f_new = smooth_max_val(compute_correlations(antipal_h(g_new)), alpha)
            f_z = smooth_max_val(compute_correlations(antipal_h(z)), alpha)
            gsc = np.sum(gz * (g_new - z))
            while f_new > f_z + gsc + 0.5 * lr * np.sum(gz**2) and lr > 1e-12:
                lr *= 0.5
                g_new = np.clip(z - lr * gz, 0.0, 1.0)
                f_new = smooth_max_val(compute_correlations(antipal_h(g_new)), alpha)
                gsc = np.sum(gz * (g_new - z))
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t_k*t_k)) / 2.0
        z_new = g_new + ((t_k - 1.0)/t_new) * (g_new - g)
        z_new = np.clip(z_new, 0.0, 1.0)

        if s < best_s:
            best_s = s
            best_g = z_new.copy()
        
        g = g_new
        z = z_new
        t_k = t_new
    
    return best_g, best_s
