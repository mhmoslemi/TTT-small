```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_hypersimplex(x, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def get_correlations(h):
    return np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))

def topk_smooth_max(corr, eps, k):
    # Take top k values and compute smooth max over them
    idx = np.argsort(corr)[-k:]
    vals = corr[idx]
    m = np.max(vals)
    return m + (1.0/eps) * np.log(np.sum(np.exp(eps * (vals - m))))

def topk_smooth_max_grad(corr, h, eps, k, n):
    dx = 2.0 / n
    idx = np.argsort(corr)[-k:]
    vals = corr[idx]
    m = np.max(vals)
    weights = np.exp(eps * (vals - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for ii in range(k):
        wi = weights[ii]
        if wi < 1e-14:
            continue
        l = idx[ii] - (n - 1)
        for j in range(n):
            term = 0.0
            jml = j - l
            if 0 <= jml < n:
                term += (1.0 - h[jml])
            jpl = j + l
            if 0 <= jpl < n:
                term += (-h[jpl])
            grad[j] += wi * term
    grad *= dx
    return grad

def frank_wolfe(h, eps, k_top, max_iters, time_limit_fn):
    n = len(h)
    h_cur = h.copy()
    for it in range(max_iters):
        if time_limit_fn():
            break
        corr = get_correlations(h_cur)
        g = topk_smooth_max_grad(corr, h_cur, eps, k_top, n)
        # Frank-Wolfe step: find v in hypersimplex minimizing g·v
        # v is binary with exactly n/2 ones at positions where g is smallest
        n_ones = n // 2
        v_indices = np.argsort(g)[:n_ones]
        v = np.zeros(n)
        v[v_indices] = 1.0
        # Line search or fixed step
        t = 2.0 / (it + 2)
        h_new = (1 - t) * h_cur + t * v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to maintain sum constraint (approximately)
        h_new = project_to_hypersimplex(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h_cur) or it == 0:
            h_cur = h_new
    return h_cur, evaluate_c5(h_cur)
```

```python
term2 = np.where(mask2, h[i + lags], 0.0)
# term3: +(1-h[j-l]) for valid j-l
mask3 = (j - lags >= 0) & (j - lags < n)
term3 = np.where(mask3, (1.0 - h[j - lags]), 0.0)
# term4: -h[j+l] for valid j+l
mask4 = (j + lags >= 0) & (j + lags < n)
term4 = np.where(mask4, -h[j + lags], 0.0)
delta = term1 + term2 + term3 + term4
```

This is O(n) with numpy vectorization, which is fast.

Let me vectorize the gradient computation:

For a given lag l and weight w:
grad[j] += w * [(1-h[j-l]) * 1_{0<=j-l<n} + (-h[j+l]) * 1_{0<=j+l<n}]

This can be written as:
grad += w * (shifted_1_minus_h + shifted_neg_h)

where shifted_1_minus_h[j] = (1-h[j-l]) if 0<=j-l<n else 0
and shifted_neg_h[j] = -h[j+l] if 0<=j+l<n else 0

I can compute these using np.roll or slicing:

For lag l > 0:
- (1-h[j-l]) is valid for j >= l: result[l:] = 1-h[:n-l]
- -h[j+l] is valid for j < n-l: result[:n-l] = -h[l:]

For lag l < 0:
- Let l' = -l > 0
- (1-h[j-l]) = (1-h[j+l']) is valid for j < n-l': result[:n-l'] = 1-h[l':]
- -h[j+l] = -h[j-l'] is valid for j >= l': result[l':] = -h[:n-l']

For lag l = 0:
- (1-h[j]) for all j
- -h[j] for all j
- Combined: 1-2h

```python
def compute_fw_gradient(h, eps, k_top, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = 2 * n - 1
    # Get top k indices
    if k_top >= n_corr:
        idx = np.arange(n_corr)
        vals = corr
    else:
        idx = np.argsort(corr)[-k_top:]
        vals = corr[idx]
    m = np.max(vals)
    exp_vals = np.exp(eps * (vals - m))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for ii in range(min(k_top, n_corr)):
        wi = weights[ii]
        if wi < 1e-14:
            continue
        l = idx[ii] - (n - 1)
        # Contribution to grad from lag l with weight wi
        temp = np.zeros(n)
        if l == 0:
            temp = 1.0 - 2.0 * h
        elif l > 0:
            # (1-h[j-l]) for j >= l
            temp[l:] += (1.0 - h[:n-l])
            # -h[j+l] for j < n-l
            temp[:n-l] += (-h[l:])
        else:
            ll = -l
            # (1-h[j+l]) for j < n-ll
            temp[:n-ll] += (1.0 - h[ll:])
            # -h[j-l] for j >= ll
            temp[ll:] += (-h[:n-ll])
        grad += wi * temp
    grad *= dx
    return grad
```

This is O(k * n) per gradient call, which is much better.

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def binary_round(h, n):
    """Round h to a binary vector with exactly n/2 ones."""
    target = n // 2
    # Sort by value descending, pick top n/2
    idx = np.argsort(h)[::-1]
    v = np.zeros(n)
    v[idx[:target]] = 1.0
    return v

def fw_gradient(h, corr, eps, k_top, n):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    if k_top >= n_corr:
        idx = np.arange(n_corr)
        vals = corr
    else:
        idx = np.argsort(corr)[-k_top:]
        vals = corr[idx]
    m = np.max(vals)
    exp_vals = np.exp(eps * (vals - m))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for ii in range(min(k_top, len(idx))):
        wi = weights[ii]
        if wi < 1e-14:
            continue
        l = int(idx[ii]) - (n - 1)
        temp = np.zeros(n)
        if l == 0:
            temp = 1.0 - 2.0 * h
        elif l > 0:
            temp[l:] += (1.0 - h[:n-l])
            temp[:n-l] += (-h[l:])
        else:
            ll = -l
            temp[:n-ll] += (1.0 - h[ll:])
            temp[ll:] += (-h[:n-ll])
        grad += wi * temp
    grad *= dx
    return grad

def frank_wolfe_iter(h, eps, k_top, max_iters, time_fn):
    n = len(h)
    h_cur = h.copy().astype(float)
    best_c5 = evaluate_c5(h_cur)
    best_h = h_cur.copy()
    n_ones = n // 2
    
    for it in range(max_iters):
        if time_fn():
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        g = fw_gradient(h_cur, corr, eps, k_top, n)
        # FW step: v = binary vector with n/2 ones at smallest g positions
        v_idx = np.argsort(g)[:n_ones]
        v = np.zeros(n)
        v[v_idx] = 1.0
        t = 2.0 / (it + 2)
        h_new = (1.0 - t) * h_cur + t * v
        # h_new is automatically feasible (convex combo of feasible points)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_c5

def swap_local_search(h, n, max_swaps=5000, time_fn=None):
    """Binary local search using swaps (maintains sum constraint exactly)."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Adjust to exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            # Remove some ones
            ones_idx = np.where(h == 1.0)[0]
            to_remove = ones_idx[np.random.choice(len(ones_idx), n_ones - target, replace=False)]
            h[to_remove] = 0.0
        else:
            zeros_idx = np.where(h == 0.0)[0]
            to_add = zeros_idx[np.random.choice(len(zeros_idx), target - n_ones, replace=False)]
            h[to_add] = 1.0
    
    corr_base = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr_base)
    n_corr = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    swaps_done = 0
    for _ in range(1000):
        if time_fn is not None and time_fn():
            break
        if swaps_done >= max_swaps:
            break
        improved = False
        n_attempts = min(len(ones_list) * len(zeros_list) // 10, 200)
        n_attempts = max(n_attempts, 20)
        for _ in range(n_attempts):
            if time_fn is not None and time_fn():
                break
            i = ones_list[np.random.randint(len(ones_list))]
            j = zeros_list[np.random.randint(len(zeros_list))]
            if i == j:
                continue
            # Compute delta correlation
            # delta(l) = -(1-h[i-l]) + h[i+l] + (1-h[j-l]) - h[j+l] (with validity)
            # Vectorized:
            i_minus = i - lags
            j_minus = j - lags
            i_plus = i + lags
            j_plus = j + lags
            
            d = np.zeros(n_corr)
            m1 = (i_minus >= 0) & (i_minus < n)
            d[m1] -= (1.0 - h[i_minus[m1]])
            m2 = (i_plus >= 0) & (i_plus < n)
            d[m2] += h[i_plus[m2]]
            m3 = (j_minus >= 0) & (j_minus < n)
            d[m3] += (1.0 - h[j_minus[m3]])
            m4 = (j_plus >= 0) & (j_plus < n)
            d[m4] -= h[j_plus[m4]]
            
            new_corr = corr_base + d
            max_new = np.max(new_corr)
            if max_new < c5 - 1e-10:
                c5 = max_new
                corr_base = new_corr
                h[i] = 0.0
                h[j] = 1.0
                ones_list.remove(i)
                zeros_list.remove(j)
                ones_list.append(j)
                zeros_list.append(i)
                improved = True
                swaps_done += 1
        if not improved:
            break
    
    return h, c5 * dx

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def elapsed():
        return time.time() - start_time
    
    def time_frac(frac):
        return lambda: elapsed() > budget_s * frac
    
    # Get initial values

        if c5_new < best_c5:

        # Also try from a slightly perturbed version

            j = zeros[np.random.randint(len(zeros))]
            if i != j:

                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
```

```python
def annealed_swap_search(h, n, max_swaps=10000, T_start=0.01, T_end=0.001, time_fn=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones_idx = np.where(h == 1.0)[0]
            to_remove = ones_idx[np.random.choice(len(ones_idx), n_ones - target, replace=False)]
            h[to_remove] = 0.0
        else:
            zeros_idx = np.where(h == 0.0)[0]
            to_add = zeros_idx[np.random.choice(len(zeros_idx), target - n_ones, replace=False)]
            h[to_add] = 1.0
    
    corr_base = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr_base)
    n_corr = 2 * n - 1
    lags = np.arange(-(n-1), n)
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    best_h = h.copy()
    best_c5 = c5
    best_corr = corr_base.copy()
    
    total_steps = max_swaps
    for step in range(total_steps):
        if time_fn is not None and time_fn():
            break
        T = T_start * (T_end / T_start) ** (step / total_steps)
        
        i = ones_list[np.random.randint(len(ones_list))]
        j = zeros_list[np.random.randint(len(zeros_list))]
        if i == j:
            continue
        
        i_minus = i - lags
        j_minus = j - lags
        i_plus = i + lags
        j_plus = j + lags
        
        d = np.zeros(n_corr)
        m1 = (i_minus >= 0) & (i_minus < n)
        d[m1] -= (1.0 - h[i_minus[m1]])
        m2 = (i_plus >= 0) & (i_plus < n)
        d[m2] += h[i_plus[m2]]
        m3 = (j_minus >= 0) & (j_minus < n)
        d[m3] += (1.0 - h[j_minus[m3]])
        m4 = (j_plus >= 0) & (j_plus < n)
        d[m4] -= h[j_plus[m4]]
        
        new_corr = corr_base + d
        max_new = np.max(new_corr)
        delta = max_new - c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # Accept
            c5 = max_new
            corr_base = new_corr
            h[i] = 0.0
            h[j] = 1.0
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5 * dx
```

```python
def best_improvement_sweep(h, n):
    """Try all possible swaps, perform the best improving one."""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr)
    n_corr = 2 * n - 1

    zeros = np.where(h == 0.0)[0]
    
    best_delta = 0

    for i in ones:
        i_minus = i - lags
        i_plus = i + lags
        m1 = (i_minus >= 0) & (i_minus < n)
        m2 = (i_plus >= 0) & (i_plus < n)
        di = np.zeros(n_corr)

        di[m2] += h[i_plus[m2]]
```

```python
import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def binary_round(h, n, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    v = np.zeros(n)
    v[idx[:target]] = 1.0
    return v

def fw_gradient_vec(h, corr, eps, k_top, n):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    if k_top >= n_corr:
        idx = np.arange(n_corr)
        vals = corr
    else:
        idx = np.argsort(corr)[-k_top:]
        vals = corr[idx]
    m = np.max(vals)
    exp_vals = np.exp(eps * (vals - m))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wi = weights[ii]
        if wi < 1e-14:
            continue
        l = int(idx[ii]) - (n - 1)
        if l == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif l > 0:
            temp = np.zeros(n)
            temp[l:] += (1.0 - h[:n - l])
            temp[:n - l] += (-h[l:])
            grad += wi * temp
        else:
            ll = -l
            temp = np.zeros(n)
            temp[:n - ll] += (1.0 - h[ll:])
            temp[ll:] += (-h[:n - ll])
            grad += wi * temp
    grad *= dx
    return grad

def frank_wolfe_opt(h, n, eps, k_top, max_iters, time_fn):
    h_cur = h.copy().astype(float)
    best_c5 = evaluate_c5(h_cur)
    best_h = h_cur.copy()
    n_ones = n // 2
    for it in range(max_iters):
        if time_fn():
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        g = fw_gradient_vec(h_cur, corr, eps, k_top, n)
        v_idx = np.argsort(g)[:n_ones]
        v = np.zeros(n)
        v[v_idx] = 1.0
        t = 2.0 / (it + 2.0)
        h_new = (1.0 - t) * h_cur + t * v
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c5

def annealed_swap_search(h, n, n_steps=20000, T_start=0.005, T_end=0.0001, time_fn=None):
    dx = 2.0 / n
    h = h.astype(float).copy()
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:

        to_remove = ones_idx[np.random.choice(len(ones_idx), n_ones - target, replace=False)]
        h[to_remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0.0)[0]
        to_add = zeros_idx[np.random.choice(len(zeros_idx), target - n_ones, replace=False)]
        h[to_add] = 1.0
    corr_base = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = np.max(corr_base)
    n_corr = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    best_h = h.copy()
    best_c5 = c5_curr
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    for step in range(n_steps):
        if time_fn is not None and time_fn():
            break
        T = T_start * (T_end / T_start) ** (float(step) / float(n_steps))
        i = ones_list[np.random.randint(len(ones_list))]
        j = zeros_list[np.random.randint(len(zeros_list))]
        if i == j:
            continue
        i_minus = i - lags
        j_minus = j - lags
        i_plus = i + lags
        j_plus = j + lags
        d = np.zeros(n_corr)
        m1 = (i_minus >= 0) & (i_minus < n)
        d[m1] -= (1.0 - h[i_minus[m1]])
        m2 = (i_plus >= 0) & (i_plus < n)
        d[m2] += h[i_plus[m2]]
        m3 = (j_minus >= 0) & (j_minus < n)
        d[m3] += (1.0 - h[j_minus[m3]])
        m4 = (j_plus >= 0) & (j_plus < n)
        d[m4] -= h[j_plus[m4]]
        new_corr = corr_base + d
        max_new = np.max(new_corr)
        delta = max_new - c5_curr
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            c5_curr = max_new
            corr_base = new_corr
            h[i] = 0.0
            h[j] = 1.0
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
    return best_h, best_c5 * dx

    h_bin, c5_bin = best_improvement_search(h_bin, n, max_sweeps=500, time_fn=time_frac(0.45))
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    h_anneal, c5_anneal = annealed_swap_search(best_h, n, n_steps=15000, T_start=0.008, T_end=0.0002, time_fn=time_frac(0.55))
    if c5_anneal < best_c5:
        best_c5 = c5_anneal
        best_h = h_anneal

    h_bi2, c5_bi2 = best_improvement_search(best_h, n, max_sweeps=500, time_fn=time_frac(0.60))
    if c5_bi2 < best_c5:
        best_c5 = c5_bi2
        best_h = h_bi2

    h_ms, c5_ms = multistart_swap_search(best_h, n, n_starts=20, n_steps_per=8000, time_fn=time_frac(0.80))
    if c5_ms < best_c5:
        best_c5 = c5_ms
        best_h = h_ms

    h_bi3, c5_bi3 = best_improvement_search(best_h, n, max_sweeps=1000, time_fn=time_frac(0.90))
    if c5_bi3 < best_c5:
        best_c5 = c5_bi3
        best_h = h_bi3

    if elapsed() < budget_s * 0.95:
        h_anneal2, c5_anneal2 = annealed_swap_search(best_h, n, n_steps=15000, T_start=0.003, T_end=0.00005, time_fn=time_frac(0.95))
        if c5_anneal2 < best_c5:
            best_c5 = c5_anneal2
            best_h = h_anneal2
        h_bi4, c5_bi4 = best_improvement_search(best_h, n, max_sweeps=500, time_fn=time_frac(0.97))
        if c5_bi4 < best_c5:
            best_c5 = c5_bi4
            best_h = h_bi4

    best_h = (best_h > 0.5).astype(float)
    s = int(np.sum(best_h))
    target = n // 2
    if s != target:
        if s > target:
            oi = np.where(best_h == 1.0)[0]
            best_h[oi[np.random.choice(len(oi), s - target, replace=False)]] = 0.0
        else:
            zi = np.where(best_h == 0.0)[0]
            best_h[zi[np.random.choice(len(zi), target - s, replace=False)]] = 1.0

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
```
