```python
def compute_gradient_fft(h, weights, dx):
    n = len(h)
    W = weights  # length 2n-1
    # S1 = full_conv(W[::-1], 1-h)
    # S2 = full_conv(W, h)
    # Use FFT
    fft_size = 2 * (2 * n - 1)  # or next power of 2
    # Actually, for convolution of length (2n-1) and n, result length is 3n-2
    # FFT size should be >= 3n-2
    fft_size = 1
    while fft_size < 3 * n - 1:
        fft_size *= 2
    
    W_rev_fft = np.fft.fft(W[::-1], fft_size)
    one_minus_h_fft = np.fft.fft(1.0 - h, fft_size)
    S1_full = np.fft.ifft(W_rev_fft * one_minus_h_fft).real[:3*n-2]
    
    W_fft = np.fft.fft(W, fft_size)
    h_fft = np.fft.fft(h, fft_size)
    S2_full = np.fft.ifft(W_fft * h_fft).real[:3*n-2]
    
    S1 = S1_full[n-1:2*n-1]  # n elements
    S2 = S2_full[0:n]  # n elements
    
    grad = (S1 - S2) * dx
    return grad
```

```python
import numpy as np
from scipy.signal import fftconvolve
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], 'full')
    return corr * dx

def compute_smoo_max_and_weights(overlaps, alpha):
    m = np.max(overlaps)
    shifted = alpha * (overlaps - m)
    exp_shifted = np.exp(np.clip(shifted, -500, 0))
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient_fft(h, weights, dx):
    n = len(h)
    W = weights
    # S1[i] = full_conv(W[::-1], 1-h)[i + n - 1] for i=0,...,n-1
    # S2[i] = full_conv(W, h)[i] for i=0,...,n-1
    S1_full = fftconvolve(W[::-1], 1.0 - h, 'full')
    S2_full = fftconvolve(W, h, 'full')
    S1 = S1_full[n-1:2*n-1]
    S2 = S2_full[0:n]
    return (S1 - S2) * dx

def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-10:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], 'full')
    return np.max(corr) * dx

def adam_optimize(h, alpha, target, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_x = x.copy()
    best_score = evaluate_score(x)
    
    while time.time() < deadline:
        t += 1
        overlaps = compute_overlaps(x)
        sm_val, weights = compute_smoo_max_and_weights(overlaps, alpha)
        grad = compute_gradient_fft(x, weights, dx)
        
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_onto_constraint(x_new, target)
        
        x = x_new
        score = evaluate_score(x)
        if score < best_score:
            best_score = score
            best_x = x.copy()
    
    return best_x, best_score

def pgd_optimize(h, alpha, target, deadline, lr=None):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    if lr is None:
        lr = 1.0 / (alpha * 4 + 1)
    best_x = x.copy()
    best_sm = np.inf
    stall = 0
    
    while time.time() < deadline:
        overlaps = compute_overlaps(x)
        sm_val, weights = compute_smoo_max_and_weights(overlaps, alpha)
        if sm_val < best_sm - 1e-12:
            best_sm = sm_val
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        
        grad = compute_gradient_fft(x, weights, dx)
        
        # Line search
        step = lr
        improved = False
        for _ in range(20):
            x_trial = project_onto_constraint(x - step * grad, target)
            ov_t = compute_overlaps(x_trial)
            sm_t, _ = compute_smoo_max_and_weights(ov_t, alpha)
            if sm_t < sm_val - 1e-14:
                x = x_trial
                improved = True
                break
            step *= 0.5
            if step < 1e-15:
                break
        if not improved:
            lr *= 0.5
            if lr < 1e-15:
                break
    
    return best_x

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Start from initial values
    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)

    n = 4000
    x_old = np.linspace(0, 2, n0, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_start = np.interp(x_new, x_old, init)
    h_start = project_onto_constraint(h_start, n / 2.0)
    
    best_h = h_start.copy()
    best_score = evaluate_score(best_h)
    
    # Phase 1: Adam optimization with increasing alpha
    alphas = [50, 100, 200, 500, 1000]
    for alpha in alphas:
        remaining = budget_s - 120 - (time.time() - t0)
        if remaining < 5:
            break
        deadline = time.time() + min(remaining * 0.25, 100)
        h, sc = adam_optimize(best_h, alpha, n/2.0, deadline, lr=0.005)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    # Phase 2: PGD with high alpha
    for alpha in [2000, 5000, 10000, 30000]:
        remaining = budget_s - 60 - (time.time() - t0)
        if remaining < 5:
            break
        deadline = time.time() + min(remaining * 0.2, 80)
        h = pgd_optimize(best_h, alpha, n/2.0, deadline)
        h = project_onto_constraint(h, n/2.0)
        sc = evaluate_score(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    # Phase 3: Exploration - perturb and re-optimize
    for i in range(50):
        remaining = budget_s - 30 - (time.time() - t0)
        if remaining < 10:
            break
        deadline = time.time() + min(remaining * 0.5, 15)
        
        # Different perturbation strategies
        if i % 5 == 0:
            # Binary + noise
            sorted_idx = np.argsort(-best_h)
            binary = np.zeros(n)
            binary[sorted_idx[:n//2]] = 1.0
            pert = binary + rng.randn(n) * 0.3
        elif i % 5 == 1:
            # Moderate noise
            pert = best_h + rng.randn(n) * 0.05
        elif i % 5 == 2:

            pert = best_h.copy()
            idx = rng.randint(n)
            w = 50
            lo = max(0, idx - w)
            hi = min(n, idx + w)
            pert[lo:hi] += rng.randn(hi - lo) * 0.1
        elif i % 5 == 3:
            # Shift-based perturbation
            shift = rng.randint(1, n // 4)
            pert = np.roll(best_h, shift)
            pert = project_onto_constraint(pert, n/2.0)
        else:
            # Smooth perturbation
            freq = rng.randint(1, 20)
            phase = rng.uniform(0, 2*np.pi)
            pert = best_h + 0.05 * np.sin(2*np.pi*freq*x_new/n*2 + phase)

        alpha = [200, 500, 1000, 5000][rng.randint(4)]
        h = pgd_optimize(pert, alpha, n/2.0, deadline)
        h = project_onto_constraint(h, n/2.0)
        sc = evaluate_score(h)
        if sc < best_score:
```

```python
import numpy as np
from scipy.signal import fftconvolve
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = fftconvolve(h, (1.0 - h)[::-1], 'full')
    return corr * dx


def compute_overlaps_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(np.clip(shifted, -400.0, 0.0))
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient_fft(h, weights, dx):
    n = len(h)
    W = weights
    S1_full = fftconvolve(W[::-1], 1.0 - h, 'full')
    S2_full = fftconvolve(W, h, 'full')
    S1 = S1_full[n - 1:2 * n - 1]
    S2 = S2_full[0:n]
    return (S1 - S2) * dx


def compute_gradient_loop(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 - offset:hi2 - offset]
    grad *= dx
    return grad


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        if mask.sum() == 0:
            mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()
        h[mask] += adj
        np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        for _ in range(1000):
            diff = target_sum - h.sum()
            if abs(diff) < 1e-14:
                break
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
    return h


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def adam_optimize(h, alpha, target, deadline, base_lr=0.005):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_x = x.copy()
    best_sm = np.inf
    stall = 0
    lr = base_lr

    while time.time() < deadline:
        t += 1
        lr_t = lr / (1.0 + 0.01 * t * 0.001)
        overlaps = compute_overlaps(x)
        sm_val, weights = smooth_max_and_weights(overlaps, alpha)
        if sm_val < best_sm - 1e-11:
            best_sm = sm_val
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                break

        grad = compute_gradient_fft(x, weights, dx)
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad * grad
        m_hat = m_v / (1.0 - 0.9 ** t)
        v_hat = v_v / (1.0 - 0.999 ** t)
        x_new = x - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        x = project_onto_constraint(x_new, target)

    return best_x


def pgd_refine(h, alpha, target, deadline):

def upsample_h(h_old, n_new):
    n_old = len(h_old)
```
