```python
import time
import numpy as np

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_simplex_box(vec, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(vec - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)

def _full_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_smooth_max(h, corr, alpha):
    n_pts = len(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: use only the argmax
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
        w_sum = 1.0
    w /= w_sum
    
    grad = np.zeros(n_pts)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n_pts - 1)
        wk = w[k]
        if lag > 0:
            grad[:n_pts - lag] -= wk * h[lag:]
            grad[lag:] += wk * (1.0 - h[:n_pts - lag])
        elif lag < 0:
            L = -lag
            grad[:n_pts - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n_pts - L]
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return grad

def _grad_topk(h, corr, top_k=3):
    n_pts = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n_pts)
    for k in top_idx:
        lag = k - (n_pts - 1)
        if lag > 0:
            grad[:n_pts - lag] -= h[lag:]
            grad[lag:] += (1.0 - h[:n_pts - lag])
        elif lag < 0:
            L = -lag
            grad[:n_pts - L] += (1.0 - h[L:])
            grad[L:] -= h[:n_pts - L]
        else:
            grad[:] += (1.0 - 2.0 * h)
    grad /= len(top_idx)
    return grad

def _optimize_smooth_max(h_init, n, deadline, verbose=False):
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    
    # Phase parameters: (alpha, n_iter, lr, top_k_or_full)
    phases = [
        (15, 8000, 0.12, 'full'),
        (40, 8000, 0.08, 'full'),
        (100, 6000, 0.05, 'full'),
        (300, 6000, 0.03, 3),
        (800, 4000, 0.015, 2),
        (2000, 3000, 0.008, 1),
    ]
    
    for phase_idx, (alpha, n_iter, lr, mode) in enumerate(phases):
        stall = 0
        for it in range(n_iter):
            if time.time() > deadline - 15:
                return best_h, best_c5
            
            corr = _full_correlation(h)
            
            if mode == 'full':
                grad = _grad_smooth_max(h, corr, alpha)
            else:
                grad = _grad_topk(h, corr, top_k=mode)
            
            h_new = _project_simplex_box(h - lr * grad, target_sum)
            
            c5_new = _c5_of(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
            
            h = h_new
            
            # Adaptive: if stalled for too long, reduce lr
            if stall > 500:
                lr *= 0.7
                stall = 0
                if lr < 0.001:
                    break
    
    return best_h, best_c5

def _interpolate_to(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def _balance_top_lags(h, n, n_steps=200, deadline=None):
    """Try to equalize the top few lags."""
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    
    for step in range(n_steps):
        if deadline and time.time() > deadline - 10:
            break
        
        corr = _full_correlation(h_cur)
        # Find top 3 lags
        top_3 = np.argsort(corr)[-3:]
        c_vals = corr[top_3]
        
        if len(c_vals) < 2 or c_vals[-1] - c_vals[0] < 1e-8:
            break
        
        # Direction: decrease the max lag, increase the min lag
        # grad of (corr[max] - corr[min])
        k_max = top_3[-1]
        k_min = top_3[0]
        lag_max = k_max - (n - 1)
        lag_min = k_min - (n - 1)
        
        # Gradient of corr at k_max
        g_max = np.zeros(n)
        if lag_max > 0:
            g_max[:n - lag_max] = -h_cur[lag_max:]
            g_max[lag_max:] = (1.0 - h_cur[:n - lag_max])
        elif lag_max < 0:
            L = -lag_max
            g_max[:n - L] = (1.0 - h_cur[L:])
            g_max[L:] = -h_cur[:n - L]
        else:
            g_max[:] = 1.0 - 2.0 * h_cur
        
        # Gradient of corr at k_min
        g_min = np.zeros(n)
        if lag_min > 0:
            g_min[:n - lag_min] = -h_cur[lag_min:]
            g_min[lag_min:] = (1.0 - h_cur[:n - lag_min])
        elif lag_min < 0:
            L = -lag_min
            g_min[:n - L] = (1.0 - h_cur[L:])
            g_min[L:] = -h_cur[:n - L]
        else:
            g_min[:] = 1.0 - 2.0 * h_cur
        
        # Direction: decrease corr_max, increase corr_min
        # Step: h - lr * (g_max - g_min) / 2
        direction = (g_max - g_min)
        lr = 0.01
        h_new = _project_simplex_box(h_cur - lr * direction, target_sum)
        c5_new = _c5_of(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_c5

    best_c5 = _c5_of(best_h)
    
    return (best_h, best_c5, best_n)
```

```python
import time
import numpy as np

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_simplex_box(vec, target_sum):
    """Euclidean projection onto {0<=x<=1, sum(x)=target_sum}."""
    s = np.clip(vec, 0.0, 1.0).sum()
    if abs(s - target_sum) < 1e-15:
        return np.clip(vec, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(vec - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - 0.5 * (lo + hi), 0.0, 1.0)

def _lag_gradient(h, lag):
    """Exact gradient of correlation at given lag."""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n - lag] -= h[lag:]
        grad[lag:] += (1.0 - h[:n - lag])
    else:
        L = -lag
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    return grad

def _grad_full(h, corr, alpha):
    """Full gradient of smooth max using softmax weights."""
    n = len(h)
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-8:
            continue
        lag = k - (n - 1)
        grad += w[k] * _lag_gradient(h, lag)
    return grad

def _grad_topk(h, corr, top_k):
    """Gradient using only top-k lags (equal weight)."""
    n = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        grad += _lag_gradient(h, lag)
    grad /= top_k
    return grad

def _optimize_multi_phase(h_init, n, deadline):
    """Multi-phase smooth max optimization."""
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    
    phases = [
        (10, 6000, 0.12, 'full'),
        (30, 6000, 0.08, 'full'),
        (80, 5000, 0.05, 'full'),
        (200, 5000, 0.03, 4),
        (500, 4000, 0.02, 3),
        (1500, 3000, 0.01, 2),
        (5000, 2000, 0.005, 1),
    ]
    
    for phase_idx, (alpha, n_iter, lr, mode) in enumerate(phases):
        stall_count = 0
        max_stall = 1500
        for it in range(n_iter):
            if time.time() > deadline - 20:
                return best_h, best_c5
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            if mode == 'full':
                grad = _grad_full(h, corr, alpha)
            else:
                grad = _grad_topk(h, corr, mode)
            
            h_new = _project_simplex_box(h - lr * grad, target_sum)
            c5_new = _c5_of(h_new)
            
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            h = h_new
            
            if stall_count >= max_stall:
                lr *= 0.6
                stall_count = 0
                if lr < 0.0005:
                    break
    
    return best_h, best_c5

def _interpolate_h(h_src, n_src, n_dst):
    """Linearly interpolate h from n_src points to n_dst points on [0,2)."""
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def _balance_lags(h, n, n_steps, deadline):
    """Try to equalize the top lags to reduce the max."""
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _c5_of(h_cur)
    
    for step in range(n_steps):
        if time.time() > deadline - 10:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        # Get top 4 lags
```

```python
import time
import numpy as np

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_simplex_box(vec, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(vec - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - 0.5 * (lo + hi), 0.0, 1.0)

def _lag_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] -= h[lag:]
        g[lag:] += (1.0 - h[:n - lag])
    else:
        L = -lag
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    return g

def _topk_grad(h, corr, top_k):
    n = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        if lag > 0:
            grad[:n - lag] -= h[lag:]
            grad[lag:] += (1.0 - h[:n - lag])
        elif lag < 0:
            L = -lag
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            grad[:] += 1.0 - 2.0 * h
    grad /= float(top_k)
    return grad

def _smooth_grad(h, corr, alpha):
    n = len(h)
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    ws = w.sum()
    if ws < 1e-300:
        idx = np.argmax(corr)
        return _lag_grad(h, idx - (n - 1))
    w /= ws
    grad = np.zeros(n)
    thresh = w.max() * 1e-5
    active = np.where(w > thresh)[0]
    for k in active:
        lag = k - (n - 1)
        wk = w[k]
        if lag > 0:
            grad[:n - lag] -= wk * h[lag:]
            grad[lag:] += wk * (1.0 - h[:n - lag])
        elif lag < 0:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return grad

def _optimize_phases(h_init, n, deadline):

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)

    phases = [
        (5, 7000, 0.10, 'topk'),
        (4, 6000, 0.06, 'topk'),
        (3, 6000, 0.04, 'topk'),

        (1, 5000, 0.01, 'smooth'),
        (1, 4000, 0.005, 'smooth'),
    ]

    for phase_idx, (top_k, n_iter, lr_init, mode) in enumerate(phases):
        lr = lr_init
        stall = 0
        alpha = 50 + phase_idx * 100
        for it in range(n_iter):

            corr = np.correlate(h, 1.0 - h, mode='full')

            else:

def _interpolate_h(h_src, n_src, n_dst):

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
```
