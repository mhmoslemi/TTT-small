                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    if time_left() > 20:
        n_foc_final = min(5000, int(time_left() * 15))
        h_final, s_final = adaptive_focused(best_h, n_foc_final, 0.3, 0.9995)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final.copy()

    best_h = project_box_sum(best_h, n / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, n)
