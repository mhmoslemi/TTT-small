def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_objective(h, alpha):
    corr = compute_correlations(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def lse_gradient(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    exp_a = np.exp(alpha * (corr - m))
    Z = np.sum(exp_a)

    partA = np.correlate(w, 1.0 - h, mode='full')[:n]
    partB = np.correlate(w[::-1], h, mode='full')[n-1:2*n-1][::-1]
