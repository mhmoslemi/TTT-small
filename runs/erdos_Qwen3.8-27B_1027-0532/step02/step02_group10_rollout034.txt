        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)

    for _ in range(50):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break

        nf = free.sum()
        if nf == 0:
            break

        np.clip(h, 0.0, 1.0, out=h)
    # Final correction
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 0:
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient_smooth(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            grad[:n-offset] += w * (1.0 - h[offset:n])
            grad[offset:n] -= w * h[:n-offset]
        else:
            grad[-offset:] += w * (1.0 - h[:n+offset])
            grad[:n+offset] -= w * h[-offset:]
    grad *= dx
    return grad

def pgd_optimize(h0, alpha, target, max_iters, deadline, use_exact_max=False):

    dx = 2.0 / n
    h = h0.copy()
    step = 0.5
    prev_score = None
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break

            ov = compute_overlaps(h)
            score = np.max(ov)
            # Subgradient: use active set
            active = np.where(ov >= score - 1e-12)[0]
            wts = np.zeros(len(ov))
            wts[active] = 1.0 / len(active)
        else:
            ov = compute_overlaps(h)
            score, wts = smooth_max_and_weights(ov, alpha)
        
        grad = compute_gradient_smooth(h, wts, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Check improvement
        if prev_score is not None:
            if score < prev_score - 1e-15:
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break
        prev_score = score
        
        # Backtracking line search
        improved = False
        s = step
        for bt in range(50):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            if use_exact_max:
                new_score = np.max(ov_t)
            else:
                new_score, _ = smooth_max_and_weights(ov_t, alpha)
            if new_score < score - 1e-16:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-15:
                break
        
        if not improved:
            step = s
            if step < 1e-14:
                break
        else:
            step = min(s * 1.6, 2.0)
    
    return h

m + (1/α) log(∑ exp(α(x_i - m)))

When only one value is the max (say x_1 = m and all others are < m - δ), the smooth max ≈ m + (1/α) log(1 + ∑exp(-αδ)) ≈ m + (1/α) exp(-αδ) for large α.

import numpy as np

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-16) & (h < 1.0 - 1e-16)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -400.0, 400.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    value = m + np.log(total) / alpha
    weights = exps / total
    return value, weights


def make_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(len(weights)):
        w = weights[m_idx]
        if w < 1e-14:
            continue
        offset = m_idx - (n - 1)
        if offset >= 0:
            grad[:n - offset] += w * (1.0 - h[offset:n])
            grad[offset:n] -= w * h[:n - offset]
        else:
            grad[-offset:] += w * (1.0 - h[:n + offset])
            grad[:n + offset] -= w * h[-offset:]
    grad *= dx
    return grad

    n = len(h0)
    h = h0.copy()
    step = 0.3
    prev_sc = None
    stall = 0
    while time.time() < deadline:
        ov = compute_overlaps(h)
        if alpha > 0:
            sc, wts = smooth_max_and_weights(ov, alpha)
        else:
            sc = np.max(ov)
            active = np.where(ov >= sc - 1e-12)[0]
            wts = np.zeros(len(ov))
            wts[active] = 1.0 / len(active)
        if prev_sc is not None:
            if sc < prev_sc - 1e-15:
                stall = 0
            else:
                stall += 1
            if stall > max_stall:
                break
        prev_sc = sc
        grad = make_gradient(h, wts)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        improved = False
        s = step
        for _ in range(50):
            h_try = project_onto_constraint(h - s * grad, target)
            ov_try = compute_overlaps(h_try)
            if alpha > 0:
                sc_try, _ = smooth_max_and_weights(ov_try, alpha)
            else:
                sc_try = np.max(ov_try)
            if sc_try < sc - 1e-16:
                h = h_try
                improved = True
                break
            s *= 0.5
            if s < 1e-16:
                break
        if not improved:
            step = s * 0.5
            if step < 1e-16:

        else:

    return h
