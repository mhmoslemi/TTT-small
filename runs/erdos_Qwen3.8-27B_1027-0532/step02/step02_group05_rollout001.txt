        return smooth_max_val(compute_correlations(h), alpha)

    def jac(g):
        return grad_reduced(g, alpha)

    res = minimize(fun, g0, method='SLSQP', bounds=bounds,
                   jac=jac, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    return res.x if res.x is not None else None


def project_feasible(h, target_sum):
    N = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_g = None
    best_n = n

    def elapsed():
        return time.time() - t0

    def try_update(h_cand):
        nonlocal best_h, best_score, best_g, best_n
        s = evaluate_score(h_cand)
        if s < best_score:
            best_score = s
            best_h = h_cand.copy()
            best_n = len(h_cand)
            if best_n % 2 == 1:
                m_c = (best_n - 1) // 2
                best_g = (h_cand[:m_c] + 1.0 - h_cand[m_c + 1:][::-1]) / 2.0
                best_g = np.clip(best_g, 0.0, 1.0)

    if n % 2 == 1:
        m = (n - 1) // 2
        g = (init[:m] + 1.0 - init[m + 1:][::-1]) / 2.0
        g = np.clip(g, 0.0, 1.0)
    else:
        m = n // 2
        g = None

    # Phase 1: FISTA backtracking on reduced anti-palindrome space
    if g is not None:
        for alpha in [100, 300, 1000, 3000, 10000]:
            if elapsed() > budget_s - 400:
                break
            max_iter = max(3000, int(500000 // m))
            g_new, _ = fista_backtrack(g, alpha, max_iter)
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            try_update(h_new)
            if best_g is not None:
                g = best_g.copy()

    # Phase 2: SLSQP on reduced space
    if g is not None:
        for alpha in [300, 1000, 3000]:
            if elapsed() > budget_s - 300:
                break
            maxiter = max(300, int(200000 // m))
            g_res = slsqp_reduced(g, alpha, maxiter)
            if g_res is not None and np.all(np.isfinite(g_res)):
                h_new = antipal_h(np.clip(g_res, 0.0, 1.0))
                try_update(h_new)
                if best_g is not None:
                    g = best_g.copy()

    # Phase 3: Coordinate descent polish
    if best_g is not None and elapsed() < budget_s - 250:
        m_cd = len(best_g)
        for alpha in [1000, 5000]:
            if elapsed() > budget_s - 200:
                break
            g_cd, _ = coordinate_descent_polish(best_g, alpha, n_cycles=4)
            h_cd = antipal_h(g_cd)
            try_update(h_cd)
            if best_g is not None:
                best_g = best_g.copy()

    # Phase 4: Full-space FISTA (in case anti-palindrome is suboptimal)
    if elapsed() < budget_s - 250:
        target_sum = len(best_h) / 2.0
        x = project_feasible(best_h, target_sum)
        z = x.copy()
        t_k = 1.0
        L = 10.0
        for it in range(3000):
            if elapsed() > budget_s - 200:
                break
            alpha = 1000
            h_z = z
            corr_z = compute_correlations(h_z)
            m_z = float(np.max(corr_z))
            fz = m_z
            gz = grad_obj(h_z, alpha)
            z_new = z - gz / L
            z_new = np.clip(z_new, 0.0, 1.0)
            diff_s = target_sum - np.sum(z_new)
            mask = (z_new > 1e-8) & (z_new < 1.0 - 1e-8)
            cnt = mask.sum()
            if cnt > 0 and abs(diff_s) > 1e-10:
                z_new[mask] += diff_s / cnt
                np.clip(z_new, 0.0, 1.0, out=z_new)
            h_zen = z_new
            corr_zen = compute_correlations(h_zen)
            fz_new = float(np.max(corr_zen))
            lin = fz + gz.dot(h_zen - h_z) + 0.5 * L * np.sum((h_zen - h_z) ** 2)
            if fz_new > lin + 1e-8:
                L *= 2.0
                z_new = z - gz / L
                z_new = np.clip(z_new, 0.0, 1.0)
                diff_s = target_sum - np.sum(z_new)
                mask = (z_new > 1e-8) & (z_new < 1.0 - 1e-8)
                cnt = mask.sum()
                if cnt > 0:
                    z_new[mask] += diff_s / cnt
                    np.clip(z_new, 0.0, 1.0, out=z_new)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            x_new = z_new + ((t_k - 1.0) / t_new) * (z_new - x)
            x_new = np.clip(x_new, 0.0, 1.0)
            diff_s = target_sum - np.sum(x_new)
            mask = (x_new > 1e-8) & (x_new < 1.0 - 1e-8)
            cnt = mask.sum()
            if cnt > 0:
                x_new[mask] += diff_s / cnt
                np.clip(x_new, 0.0, 1.0, out=x_new)
            try_update(x_new)
            x = x_new
            z = z_new
            t_k = t_new
            L = max(5.0, L / 1.3)

    # Phase 5: Random restarts in reduced space
    if best_g is not None and elapsed() < budget_s - 150:
        m_rs = len(best_g)
        for r in range(30):
            if elapsed() > budget_s - 120:
                break
            scale = 0.01 + 0.10 * (r / 30.0)
            g_pert = best_g + rng.randn(m_rs) * scale
            g_pert = np.clip(g_pert, 0.0, 1.0)
            alpha = 500 + 2000 * (r % 4)
            g_new, _ = fista_backtrack(g_pert, alpha, 2000)
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            try_update(h_new)
            if best_g is not None:
                best_g = best_g.copy()

    # Phase 6: Binary snapping
    if best_g is not None and elapsed() < budget_s - 100:
        g_bin = np.round(best_g)
        h_bin = antipal_h(g_bin)
        try_update(h_bin)
        g_half = np.round(best_g * 2.0) / 2.0
        h_half = antipal_h(g_half)
        try_update(h_half)

    # Phase 7: Multi-resolution
    for n_try in [213, 425, 637, 849, 1061]:
        if n_try <= 2 or n_try > 1500:
            continue
        if elapsed() > budget_s - 80:
            break
        if n_try == len(best_h):
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new_arr, x_old, best_h)
        if n_try % 2 == 1:
            m_try = (n_try - 1) // 2
            g_interp = (h_interp[:m_try] + 1.0 - h_interp[m_try + 1:][::-1]) / 2.0
            g_interp = np.clip(g_interp, 0.0, 1.0)
            for alpha in [500, 2000, 5000]:
                if elapsed() > budget_s - 60:
                    break
                g_new, _ = fista_backtrack(g_interp, alpha, 2000)
                h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
                try_update(h_new)
                if best_g is not None and len(best_g) == m_try:
                    g_interp = best_g.copy()

    # Phase 8: Final high-alpha polish
    if best_g is not None and elapsed() < budget_s - 10:
        for alpha in [5000, 10000, 20000, 50000]:
            if elapsed() > budget_s - 5:
                break
            g_new, _ = fista_backtrack(best_g, alpha, 3000)
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            try_update(h_new)
            if best_g is not None:
                best_g = best_g.copy()

    # Coordinate descent final polish
    if best_g is not None and elapsed() < budget_s - 5:
        g_cd, _ = coordinate_descent_polish(best_g, 5000, n_cycles=3)
        h_cd = antipal_h(g_cd)
        try_update(h_cd)

    return (best_h, best_score, len(best_h))
