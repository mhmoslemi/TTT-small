def _slsqp_step(h_init, eps, maxiter):

        cmax = np.max(corr)
        x = (corr - cmax) / eps
        x -= np.max(x)
        return cmax + eps * np.log(np.sum(np.exp(x)))
    
    def obj_grad(h):

from scipy.optimize import minimize
import time

    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_to_binary(h, target_sum):
    n = len(h)
    h = h.copy()
    # Round and fix
    rounded = (h >= 0.5).astype(float)
    n_ones = int(rounded.sum())
    if n_ones == target_sum:
        return rounded
    # Adjustment needed
    # Use the fractional parts to decide
    if n_ones > target_sum:
        # Turn off some 1s
        ones_idx = np.where(rounded == 1.0)[0]
        # Prioritize turning off those closest to 0.5
        sort_order = np.argsort(h[ones_idx])  # lowest h values first (closest to 0.5)
        to_flip = n_ones - target_sum
        for i in range(to_flip):
            rounded[ones_idx[sort_order[i]]] = 0.0
    else:
        zeros_idx = np.where(rounded == 0.0)[0]
        sort_order = np.argsort(h[zeros_idx])[::-1]  # highest h values first
        to_flip = target_sum - n_ones
        for i in range(to_flip):
            rounded[zeros_idx[sort_order[i]]] = 1.0
    return rounded

def eval_swap(h, i1, i2, n):
    """Evaluate C5 after swapping h[i1] and h[i2] (one is 1, other is 0)."""
    # i1 should be 1, i2 should be 0
    h[i1] = 0.0
    h[i2] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr)
    h[i1] = 1.0
    h[i2] = 0.0
    return c5

def eval_2swap(h, i1, i2, i3, i4, n):
    """Evaluate C5 after 2-swap: i1,i3 are 1s to become 0s, i2,i4 are 0s to become 1s."""
    orig1, orig2, orig3, orig4 = h[i1], h[i2], h[i3], h[i4]
    h[i1] = 0.0
    h[i2] = 1.0
    h[i3] = 0.0
    h[i4] = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr)
    h[i1] = orig1
    h[i2] = orig2
    h[i3] = orig3
    h[i4] = orig4
    return c5

def guided_local_search(h, max_iter, time_limit, start_time, rng):
    n = len(h)
    target_sum = n // 2
    dx = 2.0 / n
    h = project_to_binary(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = int(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Find active lags (within 1 of max for more aggressive search)
        threshold = current_max - 1
        active_lags = np.where(corr >= threshold)[0]
        
        best_new_max = current_max
        best_move = None
        move_type = 0  # 0=single, 1=double

        for lag_idx in active_lags:

            if s > 0:

                mask = (h[valid] == 1.0) & (h[valid + s] == 0.0)
                candidates = valid[mask]
            elif s < 0:
                r = -s
                valid = np.arange(0, n - r)
                mask = (h[valid + r] == 1.0) & (h[valid] == 0.0)
                candidates = valid[mask]
            else:
                continue
            
            if len(candidates) == 0:
                continue
            
            # Try up to 8 random candidates from this lag
            n_try = min(8, len(candidates))
            if n_try < len(candidates):

            else:
                try_idx = candidates

                if s > 0:
                    c5_new = eval_swap(h, ci, ci + s, n)
                    if c5_new < best_new_max:
                        best_new_max = c5_new
                        best_move = (ci, ci + s)
                        move_type = 0
                else:
                    r2 = -s
                    c5_new = eval_swap(h, ci + r2, ci, n)
                    if c5_new < best_new_max:
                        best_new_max = c5_new
                        best_move = (ci + r2, ci)
                        move_type = 0

def mirror_descent_solve(h_init, n_iters, lr, eta, start_time, time_limit, rng):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    # Initialize lag weights
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Compute gradient of weighted sum of correlations
        # grad[i] = sum_k w[k] * d(corr_k)/d(h[i])
        # For corr at shift s (s = k - (n-1)):
        #   d(corr_s)/d(h[i]) = (1-h[i+s]) if i in [0, n-1-s] (first factor)
        #                      - h[i-s] if i in [s, n-1] (second factor)
        # We can compute this using convolution
        omh = 1.0 - h
        # First factor contribution: for each s, add (1-h[i+s]) to positions [0, n-1-s]
        # This is: for position i, sum over s where 0<=i<=n-1-s of w_{n-1+s} * (1-h[i+s])
        #         = (w * omh)_i (circular or linear convolution?)
        # Actually, let's use the direct formula:
        # grad[i] = sum_s w_{n-1+s} * [(1-h[i+s]) * 1_{0<=i<=n-1-s} + (-h[i-s]) * 1_{0<=i-s<=n-1-s}]
        
        # Let's compute using numpy:
        # For s > 0: positions i in [0, n-1-s] get (1-h[i+s]), and positions i in [s, n-1] get -h[i-s]
        # For s < 0 (let r = -s): positions i in [r, n-1] get (1-h[i-r]), and positions i in [0, n-1-r] get -h[i+r]
        # For s = 0: none (corr_0 = sum h[i]*(1-h[i]) which is fixed given sum h = n/2)
        
        # Simpler: use the formula from before
        grad = np.zeros(n)
        for k in range(n_lags):
            wk = w[k]
            if wk < 1e-15:

            s = k - (n - 1)
            if s >= 0:

                if m > 0:

                    grad[s:] -= wk * h[:m]   # -h[i-s] for i in [s, n-1]

                    grad[:m] -= wk * h[r:]    # -h[i+r] for i in [0, m-1]

            best_c5 = c5

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def ew_solve(h_init, max_iter, lr_init, eta_init, start_time, time_limit):
    """Exponential weights / mirror descent for min-max correlation."""
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    n_lags = 2 * n - 1
    h = h_init.copy()
    w = np.ones(n_lags) / n_lags
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = lr_init
    eta = eta_init
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Compute gradient of weighted sum
        omh = 1.0 - h
        grad = np.zeros(n)
        for k in range(n_lags):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                if m > 0:
                    grad[:m] += wk * omh[s:]
                    grad[s:] -= wk * h[:m]
            else:
                r = -s
                m = n - r
                if m > 0:
                    grad[r:] += wk * omh[:m]
                    grad[:m] -= wk * h[r:]
        
        # Update h with projection
        h = project_onto_constraint(h - lr * grad, target_sum)
        
        # Update weights (exponential update)
        cmax = np.max(corr)
        if cmax > 1e-12:
            log_w = eta * corr / cmax
            log_w -= np.max(log_w)
            w = np.exp(log_w)
            w_sum = np.sum(w)
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.ones(n_lags) / n_lags
        
        # Track best
        c5 = dx * np.max(corr)  # use the pre-update corr for speed
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Decay learning rate
        lr = lr_init / (1.0 + 0.01 * it)
        eta = eta_init * (1.0 + 0.005 * it)
    
    return best_h, best_c5

def smooth_obj(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))
