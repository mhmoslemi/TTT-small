        if s > target_sum:

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_max_corr(h):
    """Return the max integer correlation value (for binary h)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def hill_climb_binary(h, max_rounds=200, deadline=None):
    import bisect
    n = len(h)
    h = h.astype(float).copy()
    current_c5 = compute_c5(h)
    ones = sorted(np.where(h == 1.0)[0].tolist())
    zeros = sorted(np.where(h == 0.0)[0].tolist())
    
    for rd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        best_c5 = current_c5
        best_oi = -1
        best_zi = -1
        for oi in ones:
            if deadline and time.time() > deadline:
                break
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c5 = compute_c5(h)
                h[oi] = 1.0
                h[zi] = 0.0

                    best_c5 = c5
                    best_oi = oi
                    best_zi = zi

            h[best_zi] = 1.0
            current_c5 = best_c5
            ones.remove(best_oi)
            zeros.remove(best_zi)
            bisect.insort(ones, best_zi)
            bisect.insort(zeros, best_oi)
        else:
            break
    return h, current_c5
