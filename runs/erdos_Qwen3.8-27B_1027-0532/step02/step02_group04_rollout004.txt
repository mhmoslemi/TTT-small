                grad[:N - lag] += w
                grad[:N - lag] -= w * h[lag:]
                grad[lag:] -= w * h[:N - lag]
        else:
            l = -lag
            if l < N:
                grad[l:] += w
                grad[l:] -= w * h[:N - l]
                grad[:N - l] -= w * h[l:]
    return grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def project_binary(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    sorted_idx = np.argsort(-h)
    result = np.zeros(n)
    result[sorted_idx[:int(target_sum)]] = 1.0
    if target_sum != int(target_sum):
        frac = target_sum - int(target_sum)
        result[sorted_idx[int(target_sum)]] = frac
    return result

def binary_local_search(h_bin, n_iters, rng, t_start=1.0, t_end=0.001):
    h = h_bin.copy()
    n = len(h)
    corr = compute_correlations(h)
    best_max = np.max(corr)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        if p == q:
            continue
        h[p], h[q] = h[q], h[p]
        corr_new = compute_correlations(h)
        max_new = np.max(corr_new)
        delta = max_new - best_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            best_max = max_new
            corr = corr_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[p], h[q] = h[q], h[p]
    return h, best_max

def gradient_guided_binary_search(h_bin, n_iters, rng, alpha_smooth=200.0):
    h = h_bin.copy()
    n = len(h)
    best_max = np.max(compute_correlations(h))
    for it in range(n_iters):
        grad = smooth_max_gradient(h, alpha_smooth)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p_candidates = ones[rng.randint(0, len(ones), size=min(20, len(ones)))]
        q_candidates = zeros[rng.randint(0, len(zeros), size=min(20, len(zeros)))]
        best_delta = 0
        best_p, best_q = -1, -1
        for p in p_candidates:
            for q in q_candidates:
                if p == q:
                    continue
                if grad[p] > 0 and grad[q] < 0:
                    h[p], h[q] = 0.0, 1.0
                    new_max = np.max(compute_correlations(h))
                    delta = new_max - best_max
                    h[p], h[q] = 1.0, 0.0
                    if delta < best_delta:
                        best_delta = delta
                        best_p, best_q = p, q
        if best_p >= 0:
            h[best_p], h[best_q] = 0.0, 1.0
            best_max = best_max + best_delta
    return h, best_max

def objective_with_penalty(h, alpha, mu, beta):
    corr = compute_correlations(h)
    sm = smooth_max_value(corr, alpha)
    n = len(h)
    s = np.sum(h) - n / 2.0
    binary_pen = np.sum(h * (1.0 - h))
    return sm + mu * s * s + beta * binary_pen

def gradient_with_penalty(h, alpha, mu, beta):
    g = smooth_max_gradient(h, alpha)
    n = len(h)
    s = np.sum(h) - n / 2.0
    g += 2.0 * mu * s
    g += beta * (1.0 - 2.0 * h)
    return g

def ljbfgs_call(h0, alpha, n_pts, mu, maxiter, beta=0.0):
    def obj(h):
        return objective_with_penalty(h, alpha, mu, beta)
    def grad(h):
        return gradient_with_penalty(h, alpha, mu, beta)
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def ljbfgs_call_standard(h0, alpha, n_pts, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n_pts / 2.0
        return sm + mu * s * s
    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n_pts / 2.0
        g += 2.0 * mu * s
        return g
    bounds = [(0.0, 1.0)] * n_pts
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    best_h = project_feasible(init, n / 2.0)
    best_score = evaluate_score(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    MU = 50000.0

    # Phase 1: Adaptive alpha smooth max optimization
    alpha = 50.0
    prev_best = best_score
    for phase1_iter in range(30):
        if time_left() < budget_s * 0.35:
            break
        maxiter_p1 = max(100, int(60000.0 / n))
        x = ljbfgs_call_standard(best_h, alpha, n, MU, maxiter_p1)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
            improvement = prev_best - best_score
            if improvement < 1e-7:
                alpha *= 3.0
            else:
                alpha *= 1.5
            prev_best = best_score
        if alpha > 20000:
            break

    # Phase 2: Binary local search with simulated annealing
    if time_left() > budget_s * 0.30:
        h_bin = project_binary(best_h, n / 2.0)
        n_sa_iters = int(min(50000, time_left() * 100))
        h_bin, bin_max = binary_local_search(h_bin, n_sa_iters, rng, t_start=0.5, t_end=0.005)
        h_bin_proj = project_feasible(h_bin, n / 2.0)
        bin_score = evaluate_score(h_bin_proj)
        if bin_score < best_score:
            best_score = bin_score
            best_h = h_bin_proj.copy()

    # Phase 3: Gradient-guided binary search
    if time_left() > budget_s * 0.25:
        h_bin = project_binary(best_h, n / 2.0)
        n_guided_iters = int(min(5000, time_left() * 30))
        h_guided, guided_max = gradient_guided_binary_search(h_bin, n_guided_iters, rng, alpha_smooth=300.0)
        h_guided_proj = project_feasible(h_guided, n / 2.0)
        guided_score = evaluate_score(h_guided_proj)
        if guided_score < best_score:
            best_score = guided_score
            best_h = h_guided_proj.copy()

    # Phase 4: Penalty-based optimization (encourage binary)
    if time_left() > budget_s * 0.25:
        for beta_val in [0.5, 1.0, 2.0, 5.0, 10.0]:
            if time_left() < budget_s * 0.15:
                break
            starts = [best_h.copy()]
            h_bin_start = project_binary(best_h, n / 2.0)
            h_bin_noise = h_bin_start + rng.randn(n) * 0.05
            starts.append(project_feasible(h_bin_noise, n / 2.0))
            for start_h in starts:
                if time_left() < budget_s * 0.12:
                    break
                maxiter_p4 = max(100, int(40000.0 / n))
                for alpha_p4 in [100, 500, 2000]:
                    if time_left() < budget_s * 0.10:
                        break
                    x = ljbfgs_call(start_h, alpha_p4, n, MU, maxiter_p4, beta=beta_val)
                    if x is not None and np.all(np.isfinite(x)):
                        h_new = project_feasible(x, n / 2.0)
                        s = evaluate_score(h_new)
                        if s < best_score:
                            best_score = s
                            best_h = h_new.copy()
                            start_h = h_new.copy()

    # Phase 5: Multi-resolution refinement
    for n_try in [2 * n, 4 * n]:
        if n_try > 3000 or time_left() < budget_s * 0.15:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_feasible(h_init, n_try / 2.0)
        for alpha_m in [200, 1000, 5000, 20000]:
            if time_left() < budget_s * 0.08:
                break
            maxiter_m = max(50, int(200000.0 / n_try))
            x = ljbfgs_call_standard(h_init, alpha_m, n_try, MU, maxiter_m)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_try / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_init = h_new.copy()

    # Phase 6: Restart-based refinement at current resolution
    n_current = len(best_h)
    n_restarts = 80
    for i in range(n_restarts):
        if time_left() < budget_s * 0.08:
            break
        if i < n_restarts // 2:
            scale = 0.02 + 0.05 * (i / float(n_restarts))
            pert = best_h + rng.randn(n_current) * scale
        else:
            pert = best_h.copy()
            blen = rng.randint(4, min(30, n_current // 6))
            bstart = rng.randint(0, n_current - blen)
            pert[bstart:bstart + blen] = 1.0 - pert[bstart:bstart + blen]
        pert = project_feasible(pert, n_current / 2.0)
        alpha_pick = [100, 500, 2000, 5000, 10000][rng.randint(5)]
        maxiter_r = max(80, int(30000.0 / n_current))
        x = ljbfgs_call_standard(pert, alpha_pick, n_current, MU, maxiter_r)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_current / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 7: Very high alpha polish
    n_final = len(best_h)
    for alpha_f in [20000, 50000, 100000, 200000]:
        if time_left() < 30:
            break
        maxiter_f = max(200, int(100000.0 / n_final))
        x = ljbfgs_call_standard(best_h, alpha_f, n_final, MU, maxiter_f)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_final / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 8: Final local search with pair transfers
    if time_left() > 15:
        n_ls = len(best_h)
        steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        best_corr_ls = compute_correlations(best_h)
        best_max_ls = np.max(best_corr_ls)
        n_ls_iters = int(min(20000, (time_left() - 10) * 60))
        for _ in range(n_ls_iters):
            i = rng.randint(n_ls)
            j = rng.randint(n_ls)
            if i == j:
                continue
            for step in steps:
                h_try = best_h.copy()
                h_try[i] = best_h[i] + step
                h_try[j] = best_h[j] - step
                if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                    continue
                corr_try = compute_correlations(h_try)
                max_try = np.max(corr_try)
                if max_try < best_max_ls - 1e-12:
                    best_h = h_try
                    best_max_ls = max_try
                    break

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
