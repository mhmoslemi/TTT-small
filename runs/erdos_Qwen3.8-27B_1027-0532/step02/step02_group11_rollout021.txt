import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    if np.sum(x) == target:
        return np.clip(x, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    # Fix any small numerical issues
    diff = target - np.sum(result)
    if abs(diff) < 0.5:
        # Adjust one element
        idx = np.argmax(result < 1.0) if diff > 0 else np.argmax(result > 0.0)
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = h.copy()
    n_ones = n // 2
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def sa_swaps(h_bin, n, n_iters, T_start, T_end, time_limit_fn):
    """Simulated annealing with paired swaps for binary vectors."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    curr_c5 = best_c5
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    for it in range(n_iters):
        if time_limit_fn():
            break
        T = T_start * (T_end / T_start) ** (float(it) / max(n_iters, 1))
        
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        
        h[i] = 0
        h[j] = 1
        
        new_c5 = evaluate_c5(h)
        delta = new_c5 - curr_c5
        
        if delta < 0 or (T > 1e-15 and np.random.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def evaluate_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1][:n_ones]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def fix_binary_count(h, n):
    target = n // 2
    cnt = int(h.sum())
    if cnt < target:
        z = np.where(h == 0)[0]
        np.random.shuffle(z)
        h[z[:target - cnt]] = 1.0
    elif cnt > target:
        o = np.where(h == 1)[0]
        np.random.shuffle(o)
        h[o[-(cnt - target):]] = 0.0
    return h

def upsample_binary(h, nf, nt):

    out = np.zeros(nt)
    for i in range(nf):
        a = int(i * ratio)
        b = int((i + 1) * ratio)
        out[a:b] = h[i]
    return fix_binary_count(out, nt)
