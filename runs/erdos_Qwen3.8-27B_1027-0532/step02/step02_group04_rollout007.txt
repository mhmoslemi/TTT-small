                mx2 = np.max(v2)
                f2 = mx2 + np.log(np.sum(np.exp(alpha * (v2 - mx2)))) / alpha + mu * (sum_others + m2 - target) ** 2
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1

            t_opt = (lo_t + hi_t) / 2.0
            corrs = base + t_opt * dcorr_i
            sum_h = sum_others + t_opt
            h_work[i] = t_opt

    return h_work


def local_search_blocks(h, n_iters, rng, temperature=0.0):
    n = len(h)
    corrs = compute_correlations(h)
    best_max = np.max(corrs)
    h_cur = h.copy()

    for _ in range(n_iters):
        strategy = rng.randint(4)
        h_try = h_cur.copy()
        accepted = False

        if strategy == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.choice([-0.02, -0.01, -0.005, 0.005, 0.01, 0.02])
            h_try[i] += delta
            h_try[j] -= delta
            if h_try[i] < -1e-10 or h_try[i] > 1 + 1e-10 or h_try[j] < -1e-10 or h_try[j] > 1 + 1e-10:
                continue
            h_try = np.clip(h_try, 0.0, 1.0)
        elif strategy == 1:
            blen = rng.randint(3, min(25, n // 10))
            bstart = rng.randint(0, n - blen)
            h_try[bstart:bstart + blen] = 1.0 - h_try[bstart:bstart + blen]
        elif strategy == 2:
            blen = rng.randint(2, min(15, n // 10))
            bstart = rng.randint(0, n - blen)
            shift = rng.randint(-blen, blen + 1)
            if shift == 0:
                continue
            new_start = bstart + shift
            if new_start < 0 or new_start + blen > n:
                continue
            block = h_try[bstart:bstart + blen].copy()
            h_try[bstart:bstart + blen] = h_try[new_start:new_start + blen]
            h_try[new_start:new_start + blen] = block
        else:
            n_swap = rng.randint(2, 8)
            pos1 = rng.choice(n, size=n_swap, replace=False)
            pos2 = rng.choice(n, size=n_swap, replace=False)
            mask = np.setdiff1d(np.arange(n), pos1)
            tmp = h_try[pos1].copy()
            h_try[pos1] = h_try[pos2]
            h_try[pos2] = tmp

        corr_try = compute_correlations(h_try)
        score_try = np.max(corr_try)
        delta_score = score_try - best_max
        if delta_score < -1e-13:
            h_cur = h_try
            corrs = corr_try
            best_max = score_try
        elif temperature > 0 and rng.rand() < np.exp(-delta_score / temperature):
            h_cur = h_try
            corrs = corr_try
            best_max = score_try

    return h_cur, best_max


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)
    best_h = project_feasible(init, n0 / 2.0)
    best_score = evaluate_score(best_h)

    MU = 50000.0

    phase1_end = t0 + budget_s * 0.12
    phase2_end = t0 + budget_s * 0.20
    phase3_end = t0 + budget_s * 0.65
    phase4_end = t0 + budget_s * 0.85

    # Phase 1: L-BFGS-B at original resolution
    alphas_p1 = [80, 200, 500, 1200, 3000]
    for alpha in alphas_p1:
        if time.time() > phase1_end:
            break
        maxiter = max(80, 50000 // n0)
        x = ljbfgs_call(best_h, alpha, n0, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n0 / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 2: Coordinate descent at original resolution
    if time.time() < phase2_end:
        n_sweeps = max(3, int((phase2_end - time.time()) / (30.0 * n0 * 0.001)))
        n_sweeps = min(n_sweeps, 15)
        best_h = coordinate_descent(best_h, 2000, MU, n_sweeps, rng)
        best_h = project_feasible(best_h, n0 / 2.0)
        best_score = evaluate_score(best_h)

    # Phase 3: Higher resolution optimization
    n_high = 2 * n0
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_high, endpoint=False)
    h_high = np.interp(x_new, x_old, best_h)
    h_high = project_feasible(h_high, n_high / 2.0)

    alphas_p3 = [100, 300, 800, 2000]
    for alpha in alphas_p3:
        if time.time() > phase3_end - 120:
            break
        maxiter = max(40, 200000 // n_high)
        x = ljbfgs_call(h_high, alpha, n_high, MU, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n_high / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                h_high = h_new.copy()

    # Coordinate descent at high resolution
    if time.time() < phase3_end:
        time_left = phase3_end - time.time()
        n_sweeps_high = max(2, int(time_left / (n_high * 0.004)))
        n_sweeps_high = min(n_sweeps_high, 8)
        h_high = coordinate_descent(h_high, 3000, MU, n_sweeps_high, rng)
        h_high = project_feasible(h_high, n_high / 2.0)
        s = evaluate_score(h_high)
        if s < best_score:
            best_score = s
            best_h = h_high.copy()

    # Phase 4: Local search
    if time.time() < phase4_end:
        n_ls = int((phase4_end - time.time()) / 0.008)
        n_ls = min(n_ls, 8000)
        temp = 0.0001
        best_h, ls_score = local_search_blocks(best_h, n_ls, rng, temperature=temp)
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        best_score = evaluate_score(best_h)

    # Phase 5: Final polish
    if time.time() < t0 + budget_s * 0.95:
        n_final = len(best_h)
        for alpha in [5000, 15000]:
            if time.time() > t0 + budget_s * 0.97:
                break
            maxiter_f = max(30, 100000 // n_final)
            x = ljbfgs_call(best_h, alpha, n_final, MU, maxiter_f)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_final / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

        if time.time() < t0 + budget_s * 0.98:
            best_h = coordinate_descent(best_h, 8000, MU, 3, rng)
            best_h = project_feasible(best_h, len(best_h) / 2.0)
            s = evaluate_score(best_h)
            if s < best_score:
                best_score = s

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
