    # Compute subgradient s_i = (1-h[i+k*] - h[i-k*]) * dx
    s = np.zeros(n)
    for i in range(n):

        h_minus = h[i-k_star] if 0 <= i-k_star < n else 0.0
        s[i] = (1.0 - h_plus - h_minus) * dx

    v = np.zeros(n)
    v[np.argpartition(s, n_select)[:n_select]] = 1.0
    
    return v, s

        ov = compute_overlaps(h)
        sm = smooth_max_val(ov, alpha)
        if sm < best_sm - 1e-13:
            best_sm = sm
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        w = smooth_max_weights(ov, alpha)
        s = grad_smooth_max(h, n, dx, w, offsets)
        # Line search
        improved = False
        for bt in range(30):
            h_new = project_onto_constraint(h - step * s, target)
            sm_new = smooth_max_val(compute_overlaps(h_new), alpha)
            if sm_new < sm - 1e-15:
                h = h_new
                step = min(step * 1.3, 1.0)
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if not improved:
            break
    return h

def cd_polish(h, n, target, max_passes, deadline):
    """Coordinate descent with projection for polishing."""
    dx = 2.0 / n
    h = h.copy()
    best_score = evaluate_score(h)
    rng = np.random.RandomState(42)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            # Try setting h[i] to 0 and 1
            best_v = h[i]
            best_s = best_score
            for v in [0.0, 1.0]:
                if abs(v - h[i]) < 1e-10:
                    continue
                h_trial = h.copy()
                h_trial[i] = v
                h_trial = project_onto_constraint(h_trial, target)
                s = evaluate_score(h_trial)
                if s < best_s - 1e-12:
                    best_s = s
                    best_v = v
            if best_v != h[i]:
                h[i] = best_v
                h = project_onto_constraint(h, target)
                best_score = best_s
                improved = True
        if not improved:
            break
    return h, best_score
