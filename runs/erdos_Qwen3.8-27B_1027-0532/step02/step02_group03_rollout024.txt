        starts.append(_fix_sum(0.5 + 0.4 * np.cos(np.pi * x_grid), target))
        starts.append(_fix_sum(0.5 + 0.3 * np.sin(2.0 * np.pi * x_grid), target))

        for h0 in starts:
            if _remaining() < 80:
                break
            c5 = _c5_of(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_n = n

            if _remaining() < 80:
                continue

            h_opt = _lbbg_cont(h0, n, mu=50.0, alphas=alphas_cont, maxit=300)
            h_opt = _fix_sum(h_opt, target)
            c5 = _c5_of(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

            if _remaining() < 80:
                continue

            h_cd = _coord_polish(h_opt, n, max_rounds=15)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

            if _remaining() < 40:
                continue

            h_fw = _frank_wolfe(h_cd, n, max_it=400, alpha_fw=200.0)
            h_fw = _fix_sum(h_fw, target)
            c5_fw = _c5_of(h_fw)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw.copy()
                best_n = n

            if _remaining() < 40:
                continue

            h_ns, c5_ns = _nesterov_subgrad(h_fw, n, n_iters=200, step_init=0.12)
            if c5_ns < best_c5:
                best_c5 = c5_ns
                best_h = h_ns.copy()
                best_n = n

    if best_h is not None:
        n_best = best_n
        tt = n_best / 2.0

        if _remaining() > 30:
            h_fin = _lbbg_cont(best_h, n_best, mu=100.0, alphas=[5000, 20000], maxit=500)
            h_fin = _fix_sum(h_fin, tt)
            c5_f = _c5_of(h_fin)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_fin

        if _remaining() > 30:
            h_cd2 = _coord_polish(best_h, n_best, max_rounds=20)
            c5_c = _c5_of(h_cd2)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_cd2

        if _remaining() > 20:
            h_ns2, c5_n = _nesterov_subgrad(best_h, n_best, n_iters=300, step_init=0.08)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_ns2

        if _remaining() > 20:
            h_fin2 = _lbbg_cont(best_h, n_best, mu=200.0, alphas=[20000, 50000], maxit=500)
            h_fin2 = _fix_sum(h_fin2, tt)
            c5_f2 = _c5_of(h_fin2)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_fin2

        if _remaining() > 10:
            h_cd3 = _coord_polish(best_h, n_best, max_rounds=25)
            c5_c3 = _c5_of(h_cd3)
            if c5_c3 < best_c5:
                best_c5 = c5_c3
                best_h = h_cd3

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
