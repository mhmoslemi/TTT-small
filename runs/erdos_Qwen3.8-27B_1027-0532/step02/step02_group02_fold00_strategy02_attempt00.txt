def _fix_sum(h, target_sum=100.0):  # target_sum = n/2 for n=100
    h = h.copy().clip(0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        # Add probability mass
        added = 0.0
        for i in order:
            if diff <= 1e-12:
                break
            max_val = 1.0 - h[i]
            add_amount = min(diff, max_val)
            added += add_amount
            h[i] += add_amount
            diff -= add_amount
        assert added > 0
    else:
        # Remove probability mass
        order = np.argsort(h)
