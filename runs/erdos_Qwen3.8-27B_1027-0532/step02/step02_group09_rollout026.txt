        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_cand_raw = np.inf
        best_cand_ij = None

        for _ in range(best_k):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            raw_test = float(np.max(compute_overlaps(h)))
            h[i] = 1.0
            h[j] = 0.0
            if raw_test < best_cand_raw:
                best_cand_raw = raw_test
                best_cand_ij = (i, j)

        if best_cand_ij is None:
            break

        i, j = best_cand_ij
        h[i] = 0.0
        h[j] = 1.0
        delta = best_cand_raw - curr_raw

        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            curr_raw = best_cand_raw
            if curr_raw < best_raw:
                best_raw = curr_raw
                best_h = h.copy()
                it_no_improve = 0
            else:
                it_no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            it_no_improve += 1

    return best_h, best_raw


def guided_local_search(h0, n_iters, rng):
    n = len(h0)
    h = make_nearly_binary(h0, rng).copy()
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    curr_raw = best_raw

    for it in range(n_iters):
        overlaps = compute_overlaps(h)
        max_val = float(np.max(overlaps))
        threshold = max_val - 0.001 * (2.0 / n)
        active = np.where(overlaps > threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(overlaps)])

        sensitivity = np.zeros(n)
        for idx in active:
            L = idx - (n - 1)
            if L >= 0:
                mask_first = np.zeros(n)
                mask_first[:n - L] = 1.0
                shift_up = np.zeros(n)
                if n - L > 0:
                    shift_up[:n - L] = h[L:]
                shift_dn = np.zeros(n)
                if L < n:
                    shift_dn[L:] = h[:n - L]
                sensitivity += mask_first - shift_up - shift_dn
            else:
                M = -L
                mask_first = np.zeros(n)
                mask_first[M:] = 1.0
                shift_up = np.zeros(n)
                if n - M > 0:
                    shift_up[M:] = h[:n - M]
                shift_dn = np.zeros(n)
                if M < n:
                    shift_dn[:n - M] = h[M:]
                sensitivity += mask_first - shift_up - shift_dn

        dec_idx = np.where((h >= 0.5) & (sensitivity > 0))[0]
        inc_idx = np.where((h < 0.5) & (sensitivity < 0))[0]

        moved = False
        if len(dec_idx) > 0 and len(inc_idx) > 0:
            i = dec_idx[rng.randint(len(dec_idx))]
            j = inc_idx[rng.randint(len(inc_idx))]
            h[i] = 0.0
            h[j] = 1.0
            new_raw = float(np.max(compute_overlaps(h)))
            if new_raw < curr_raw:
                curr_raw = new_raw
                if curr_raw < best_raw:
                    best_raw = curr_raw
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
            moved = True

        if not moved:
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_raw = float(np.max(compute_overlaps(h)))
                if new_raw < curr_raw:
                    curr_raw = new_raw
                    if curr_raw < best_raw:
                        best_raw = curr_raw
                        best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0

    return best_h, best_raw


def try_different_n(base_h, base_score, base_n, rng, time_budget):
    best_result = (base_h, base_score, base_n)
    t_start = time.time()
    for n_try in [base_n - 25, base_n + 25, base_n + 50]:
        if n_try < 100 or n_try > 800:
            continue
        if time.time() - t_start > time_budget:
            break
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, base_h)
        h_init = project_box_sum(h_init, n_try / 2.0)
        bounds = [(0.0, 1.0)] * n_try
        for alpha in [500, 2000]:
            if time.time() - t_start > time_budget:
                break
            penalty = alpha * 30.0
            res_x = lbfgsb_optimize(h_init, alpha, penalty, 400, bounds)
            if res_x is not None:
                h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_result[1]:
                    best_result = (h_new.copy(), s, n_try)
                    h_init = h_new.copy()
        h_anneal, raw_anneal = anneal_best_of_k(h_init, 20000, rng, 0.005, 1e-8, 3)
        s_anneal = raw_anneal * (2.0 / n_try)
        if s_anneal < best_result[1]:
            best_result = (h_anneal.copy(), s_anneal, n_try)
    return best_result


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_score = evaluate_score(h)

    def time_left():
        return budget_s - 25.0 - (time.time() - t0)

    # Phase 1: SLSQP optimization
    for alpha in [50, 200, 500, 1000]:
        if time_left() < 120:
            break
        bounds = [(0.0, 1.0)] * n
        maxiter = max(100, 250000 // n)
        res_x = slsqp_optimize(h, alpha, maxiter, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 2: L-BFGS-B polishing
    for alpha in [2000, 5000, 10000]:
        if time_left() < 90:
            break
        penalty = alpha * 30.0
        bounds = [(0.0, 1.0)] * n
        res_x = lbfgsb_optimize(h, alpha, penalty, 600, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()
            h = h_new

    # Phase 3: Intensive simulated annealing (best-of-k)
    if time_left() > 150:
        n_iters = int(time_left() * 18)
        h_a, raw_a = anneal_best_of_k(best_h, n_iters, rng, 0.008, 1e-8, 3)
        s_a = raw_a * (2.0 / n)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 4: Guided local search
    if time_left() > 100:
        n_iters = int(time_left() * 12)
        h_g, raw_g = guided_local_search(best_h, n_iters, rng)
        s_g = raw_g * (2.0 / n)
        if s_g < best_score:
            best_score = s_g
            best_h = h_g.copy()

    # Phase 5: More annealing from perturbed states
    for restart in range(8):
        if time_left() < 60:
            break
        noise_scale = 0.06 + 0.02 * restart
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_box_sum(h_pert, n / 2.0)
        n_iters = max(5000, min(50000, int(time_left() * 10)))
        h_a, raw_a = anneal_best_of_k(h_pert, n_iters, rng, 0.01, 1e-8, 3)
        s_a = raw_a * (2.0 / n)
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 6: Guided search from best
    if time_left() > 60:
        n_iters = max(3000, int(time_left() * 10))
        h_g, raw_g = guided_local_search(best_h, n_iters, rng)
        s_g = raw_g * (2.0 / n)
        if s_g < best_score:
            best_score = s_g
            best_h = h_g.copy()

    # Phase 7: Try different n
    if time_left() > 120:
        result = try_different_n(best_h, best_score, n, rng, time_left() - 40)
        if result[1] < best_score:
            best_h = result[0].copy()
            best_score = result[1]
            n = result[2]

    # Phase 8: Final continuous polishing
    n_final = len(best_h)
    for alpha in [5000, 10000, 20000]:
        if time_left() < 25:
            break
        penalty = alpha * 50.0
        bounds = [(0.0, 1.0)] * n_final
        res_x = lbfgsb_optimize(best_h, alpha, penalty, 800, bounds)
        if res_x is not None:
            h_new = project_box_sum(np.clip(res_x, 0.0, 1.0), n_final / 2.0)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 9: Final annealing pass
    if time_left() > 40:
        n_iters = max(5000, int(time_left() * 15))
        h_a, raw_a = anneal_best_of_k(best_h, n_iters, rng, 0.004, 1e-9, 3)
        s_a = raw_a * (2.0 / len(best_h))
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Final projection
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
