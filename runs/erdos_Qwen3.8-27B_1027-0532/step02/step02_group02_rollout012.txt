def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project_box_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Fix any remaining numerical error
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)
        for i in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - result[i])
                result[i] += add
                diff -= add
            else:
                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub
    return result

def _smooth_max_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    m = np.max(c)
    shifted = alpha * (c - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = m + (max_shifted + np.log(sum_exp)) / alpha
    p = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    threshold = 1e-8
    active = np.where(p > threshold)[0]
    for k in active:
        lag = k - (n - 1)
        if lag == 0:
            grad += p[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
            grad += p[k] * g

            L = -lag
            g = np.zeros(n)
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
            grad += p[k] * g

def _interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def _slsqp_refine(h_init, target_sum, alpha, maxiter, ftol):
    n = len(h_init)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        if m < -50:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

    top_idx = np.argsort(c)[-top_k:]

    for k in top_idx:
        lag = k - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
            grad += g
        else:
            L = -lag
            g = np.zeros(n)
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
            grad += g
    return grad / len(top_idx) * dx

def _adam_run(h0, target_sum, alpha_schedule, lr, n_iter, deadline,
              beta1=0.85, beta2=0.999, eps=1e-8, check_every=50):
    n = len(h0)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    no_improve = 0
    patience = check_every * 25
    for t in range(1, n_iter + 1):
        if time.time() > deadline - 5:
            break
        frac = (t - 1) / max(1, n_iter - 1)
        idx = min(int(frac * len(alpha_schedule)), len(alpha_schedule) - 1)
        alpha = alpha_schedule[idx]
        g = _smooth_max_grad(h0, alpha)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * g * g
        m_unbias = m_hat / (1.0 - beta1 ** t)
        v_unbias = v_hat / (1.0 - beta2 ** t)
        h0 = h0 - lr * m_unbias / (np.sqrt(v_unbias) + eps)
        h0 = _project_box_sum(h0, target_sum)
        if t % check_every == 0:
            c5 = _c5_of(h0)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h0.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h0.copy()
    return best_h, best_c5

def _topk_subgrad(h0, target_sum, n_iter, lr0, lr_decay, top_k,
                  momentum, deadline, check_every=50):
    n = len(h0)
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    no_improve = 0
    patience = check_every * 15
    for t in range(n_iter):
        if time.time() > deadline - 5:
            break
        lr = lr0 / (1.0 + t * lr_decay)
        g = _topk_grad(h0, top_k)
        v = momentum * v - lr * g
        h0 = h0 + v
        h0 = _project_box_sum(h0, target_sum)
        if (t + 1) % check_every == 0:
            c5 = _c5_of(h0)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h0.copy()
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve > patience:
                break
    c5 = _c5_of(h0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h0.copy()
    return best_h, best_c5

def _interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def _slsqp_refine(h_init, target_sum, alpha, maxiter, ftol):
    n = len(h_init)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        if m < -50:
            return 0.0
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def _perturb_restart(h_base, target_sum, scale, rng, n, alpha, lr, n_iter, deadline):
    h_p = _project_box_sum(h_base + scale * rng.randn(n), target_sum)
    h_r, c5_r = _adam_run(h_p, target_sum, [alpha], lr, n_iter, deadline)
    return h_r, c5_r

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    best_h = None
    best_c5 = np.inf
    best_n = 0

    h_init_available = ('initial_h_values' in globals() and
                        initial_h_values is not None)

    n_values = [100, 120, 150]

    for n in n_values:
        t_rem = deadline - time.time()
        if t_rem < 180:
            break
        target_sum = n / 2.0

        h0_list = []
        if h_init_available:
            h_src = np.array(initial_h_values, dtype=float)
            if n == len(h_src):
                h0_list.append(h_src.copy())
            else:
                h0_list.append(_project_box_sum(
                    _interpolate_h(h_src, len(h_src), n), target_sum))

        x = np.linspace(0, 2, n, endpoint=False)
        h0_list.append(_project_box_sum(
            0.5 + 0.15 * np.sin(2 * np.pi * x) + 0.08 * np.sin(4 * np.pi * x),
            target_sum))

        rng = np.random.RandomState(seed + n * 13)
        h0_list.append(_project_box_sum(0.5 + 0.1 * rng.randn(n), target_sum))

        for h0 in h0_list:
            t_rem = deadline - time.time()
            if t_rem < 120:
                break

            h1, c5_1 = _adam_run(h0, target_sum, [40, 100, 200, 500, 1000],
                                 lr=0.06, n_iter=6000, deadline=deadline)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
                best_n = n

            t_rem = deadline - time.time()
            if t_rem > 60:
                h2, c5_2 = _adam_run(h1, target_sum, [300, 600, 1200, 2500],
                                     lr=0.03, n_iter=5000, deadline=deadline)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = n
                h1 = h2

            t_rem = deadline - time.time()
            if t_rem > 40:
                h3, c5_3 = _topk_subgrad(h1, target_sum, n_iter=12000,
                                         lr0=0.25, lr_decay=0.0008,
                                         top_k=1, momentum=0.92,
                                         deadline=deadline)
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = h3.copy()
                    best_n = n

            t_rem = deadline - time.time()
            if t_rem > 35 and n <= 120:
                h4 = _slsqp_refine(h1, target_sum, alpha=400,
                                   maxiter=200, ftol=1e-14)
                h4 = _project_box_sum(h4, target_sum)
                c5_4 = _c5_of(h4)
                if c5_4 < best_c5:
                    best_c5 = c5_4
                    best_h = h4.copy()
                    best_n = n

            t_rem = deadline - time.time()
            if t_rem > 50 and best_n == n:
                rng2 = np.random.RandomState(seed + 7777)
                for scale in [0.03, 0.06]:
                    if time.time() > deadline - 25:
                        break
                    hp, cp = _perturb_restart(h1, target_sum, scale, rng2,
                                              n, alpha=500, lr=0.03,
                                              n_iter=2000, deadline=deadline)
                    if cp < best_c5:
                        best_c5 = cp
                        best_h = hp.copy()
                        best_n = n

    if best_h is not None and time.time() < deadline - 20:
        n_best = best_n
        tf = best_h.copy()
        tf, c5f = _adam_run(tf, n_best / 2.0, [400, 800, 1500, 3000],
                            lr=0.015, n_iter=3000, deadline=deadline)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = tf.copy()

        if time.time() < deadline - 12:
            tf2, c5f2 = _topk_subgrad(tf, n_best / 2.0, n_iter=8000,
                                      lr0=0.12, lr_decay=0.002,
                                      top_k=1, momentum=0.9,
                                      deadline=deadline)
            if c5f2 < best_c5:
                best_c5 = c5f2
                best_h = tf2.copy()

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_h = _project_box_sum(best_h, n / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n

    best_h = _project_box_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
