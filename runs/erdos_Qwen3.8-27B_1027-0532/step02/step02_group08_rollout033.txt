    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    max_shift = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shift)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = (m + (np.log(sum_exp) + max_shift) / alpha) * dx
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, weights, mode='full')
    idx = np.arange(n) + n - 1
    grad = (conv1[idx] - conv2[idx]) * dx
    return obj, grad


def sm_fun(h, alpha):
    return smooth_max_obj_grad(h, alpha)[0]


def sm_jac(h, alpha):
    return smooth_max_obj_grad(h, alpha)[1]


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h0, alpha, maxiter, bounds):
    res = minimize(
        sm_fun, h0, method='SLSQP', bounds=bounds, args=(alpha,),
        jac=sm_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x


def lbfgs_optimize(h0, alpha, mu, maxiter, bounds):
    def penalty_fun(x):
        obj, grad = smooth_max_obj_grad(x, alpha)
        viol = np.sum(x) - len(x) / 2.0
        obj += 0.5 * mu * viol * viol
        grad = grad + mu * viol
        return obj, grad
    res = minimize(
        penalty_fun, h0, method='L-BFGS-B', bounds=bounds,
        jac=True, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16, 'iprint': 0, 'maxls': 40}
    )
    return res.x


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
    else:
        init = np.full(425, 0.5)

    n0 = len(init)
    best_h = project_to_box_sum(init, n0 / 2.0)
    best_score = evaluate_score(best_h)

    # Phase 1: Long CD at initial resolution
    cd1_time = min(350, time_left() * 0.35)
    if cd1_time > 5:
        h_cd, score_cd, nsw = cd_optimize(best_h, n0 / 2.0, cd1_time)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd.copy()

    # Phase 2: Try slightly higher resolutions
    for n_try in [500, 600, 750, 900, 1100]:
        if time_left() < 80:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0.0, 2.0, n_cur, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_to_box_sum(h_init, n_try / 2.0)
        cd2_time = min(60, time_left() * 0.08)
        if cd2_time < 5:
            break
        h_try, score_try, _ = cd_optimize(h_init, n_try / 2.0, cd2_time)
        if score_try < best_score:
            best_score = score_try
            best_h = h_try.copy()

    # Phase 3: SLSQP polish at best resolution
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    for alpha in [3000, 8000, 20000, 50000]:
        if time_left() < 25:
            break
        res_x = slsqp_optimize(best_h, alpha, 150, bounds_best)
        if res_x is not None:
            h_new = project_to_box_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 4: CD again after SLSQP (SLSQP may introduce small smooth-max bias)
    if time_left() > 40:
        cd4_time = min(150, time_left() * 0.5)
        h_cd4, score_cd4, _ = cd_optimize(best_h, n_best / 2.0, cd4_time)
        if score_cd4 < best_score:
            best_score = score_cd4
            best_h = h_cd4.copy()

    # Phase 5: L-BFGS-B with very high alpha for final polish
    if time_left() > 20:
        for alpha in [100000, 200000]:
            if time_left() < 10:
                break
            res_x = lbfgs_optimize(best_h, alpha, 50000.0, 200, bounds_best)
            if res_x is not None:
                h_new = project_to_box_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()

    # Phase 6: Final short CD
    if time_left() > 10:
        h_final, score_final, _ = cd_optimize(best_h, n_best / 2.0, min(60, time_left() - 5))
        if score_final < best_score:
            best_score = score_final
            best_h = h_final.copy()

    # Final projection and evaluation
    best_h = project_to_box_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
