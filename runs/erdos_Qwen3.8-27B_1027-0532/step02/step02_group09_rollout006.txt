    best_h = h_bin.copy()

            if 0 <= a-d < n:
                val -= best_h[a-d]

            val = 0.0
            if 0 <= b+d < n:
                val += (1 - best_h[b+d])
            if 0 <= b-d < n:
                val -= best_h[b-d]
            delta_b[i] = val

            break
        else:

            best_max = new_max

    return lse_grad_wrt_h(h, alpha)

def sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def optimize_lbfgsb(h0, alpha, penalty, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h0, args=(alpha, penalty), jac=lse_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return res.x
    except:
        return None

def optimize_slsqp(h0, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_obj, h0, args=(alpha,), jac=slsqp_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}],
                       options={'maxiter': maxiter, 'iprint': 0})
        return res.x
    except:
        return None

def binary_greedy_search(h_bin, max_rounds=500):
    """Greedy binary local search: find the best swap that reduces the max overlap."""
    n = len(h_bin)
    h = h_bin.copy()
    ov = compute_overlaps(h)
    best_max = float(ov.max())
    
    for r in range(max_rounds):
        # Find current max lag
        max_idx = int(np.argmax(ov))
        d = max_idx - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For each a in ones, compute delta_a at lag d
        # delta_a = -(1-h[a+d])*[0<=a+d<n] - h[a-d]*[0<=a-d<n]
        delta_a = np.zeros(len(ones))
        for i, a in enumerate(ones):
            v = 0.0
            if 0 <= a + d < n:
                v -= (1.0 - h[a + d])
            if 0 <= a - d < n:
                v -= h[a - d]
            delta_a[i] = v
        
        # For each b in zeros, compute delta_b at lag d
        # delta_b = (1-h[b+d])*[0<=b+d<n] - h[b-d]*[0<=b-d<n]
        delta_b = np.zeros(len(zeros))
        for i, b in enumerate(zeros):
            v = 0.0
            if 0 <= b + d < n:
                v += (1.0 - h[b + d])
            if 0 <= b - d < n:
                v -= h[b - d]
            delta_b[i] = v
        
        # Best swap minimizes delta_a[i] + delta_b[j]
        best_i = int(np.argmin(delta_a))
        best_j = int(np.argmin(delta_b))
        
        a = ones[best_i]
        b = zeros[best_j]
        
        # Apply swap and check
        h[a] = 0.0
        h[b] = 1.0
        new_ov = compute_overlaps(h)
        new_max = float(new_ov.max())
        
        if new_max >= best_max - 1e-12:
            # Revert
            h[a] = 1.0
            h[b] = 0.0
            break
        else:
            ov = new_ov
            best_max = new_max
    
    return h, best_max

def binary_sa_search(h_bin, n_iters, rng, t_init=0.1, t_final=0.001):
    """Simulated annealing on binary solutions."""
    n = len(h_bin)
    h = h_bin.copy()
    ov = compute_overlaps(h)
    best_h = h.copy()
    best_score = float(ov.max())
    current_score = best_score
    n_lags = 2 * n - 1
    
    for it in range(n_iters):
        t = t_init * (t_final / t_init) ** (it / max(n_iters - 1, 1))
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a] = 0.0
        h[b] = 1.0
        new_ov = compute_overlaps(h)
        new_score = float(new_ov.max())
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_score
