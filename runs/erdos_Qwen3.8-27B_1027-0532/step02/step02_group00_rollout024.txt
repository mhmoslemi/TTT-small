```python
def compute_grad_approx(h, alpha, top_K=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    # Get top K lags
    top_idx = np.argsort(corr)[-top_K:]
    top_corr = corr[top_idx]
    exps = np.exp(alpha * (top_corr - c_max))
    weights = exps / exps.sum()
    
    lags = np.arange(-(n-1), n)  # full lag array
    top_lags = lags[top_idx]
    
    grad = np.zeros(n)
    for k in range(top_K):
        l = top_lags[k]
        w = weights[k]
        # ∂corr[k]/∂h[j] = (1-h[j+l])*I(0<=j+l<n) - h[j-l]*I(0<=j-l<n)
        # Vectorized:
        j_indices = np.arange(n)
        jl = j_indices + l
        j_nl = j_indices - l
        valid1 = (jl >= 0) & (jl < n)
        valid2 = (j_nl >= 0) & (j_nl < n)
        grad += w * ((1.0 - h[jl[valid1]]) * valid1 - h[j_nl[valid2]] * valid2)
    
    return dx * grad
```
