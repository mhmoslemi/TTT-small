       return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

       m = np.max(overlaps)
       exp_vals = np.exp(alpha * (overlaps - m))

       weights = exp_vals / wsum
       threshold = 1e-10

       grad = np.zeros(n)

           w = weights[idx]
           L = idx - (n - 1)
           if L >= 0:
               grad[:n - L] += w * (1.0 - h[L:])
               grad[L:] -= w * h[:n - L]
           else:
               M = -L
               grad[M:] += w * (1.0 - h[:n - M])
               grad[:n - M] -= w * h[M:]
       return grad
