```python
    # At v, the subgradient is the convex hull of slopes[k] for k in argmax
    # Minimum is where 0 is in the subgradient
    
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        # Compute f and subgradient at mid
        vals = corr + slopes * (mid - v0)
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        min_slope = np.min(slopes[active])
        max_slope = np.max(slopes[active])
        
        if min_slope > 0:
            # f is increasing at mid, go left
            hi = mid
        elif max_slope < 0:
            # f is decreasing at mid, go right
            lo = mid
        else:
            # 0 is in subgradient, we found the minimum
            break
    
    return (lo + hi) / 2.0
```

```python
    """Compute d corr[k] / d h[j] for all k."""
    slopes = np.zeros(2 * n - 1)
    offset = n - 1
    # j+l valid: l from -j to n-1-j

    hi1 = n - 1 - j

        slopes[lags + offset] += (1.0 - h[j + lags])

    hi2 = j

        slopes[lags + offset] += (-h[j - lags])
```

The question is: what's the minimum achievable max_corr for a given n?

For a "perfect" solution where all correlations are equal, we'd have:
max_corr = (sum of all correlation values) / (number of lags)

The sum of all correlation values:
sum_k corr[k] = sum_k sum_i h[i](1-h[i+k])
= sum_i h[i] * sum_k (1-h[i+k])
= sum_i h[i] * (number of valid k for position i - sum_k h[i+k])

For position i, the number of valid k values where i+k is in [0,n-1] is n (for k from -i to n-1-i).
And sum_k h[i+k] (over valid k) = sum_{j=0}^{n-1} h[j] = n/2.

So sum_k corr[k] = sum_i h[i] * (n - n/2) = sum_i h[i] * n/2 = (n/2) * n/2 = n²/4.

Number of lags = 2n-1.

So the average correlation = n²/(4(2n-1)) ≈ n/8 for large n.

For n=170: average ≈ 21.25
For n=204: average ≈ 25.5
For n=256: average ≈ 32

If we could make all correlations equal to the average, C₅ = (n/8) * (2/n) = 1/4 = 0.25. But that's the theoretical minimum and not achievable in practice.

The best we can do is make the correlations as equal as possible. The "spread" of the correlations determines how much above 0.25 we end up.

corr[k] = sum_i h[i] * (1 - h[i+l]) for i where both i and i+l are in [0, n-1]

If we vary h[j] by δ:

  → derivative w.r.t. h[j] is -h[j-l]

So total slope at lag l: (1-h[j+l]) * [j+l valid] - h[j-l] * [j-l valid]

If there are both positive and negative (or zero), we've found the minimum.

```python
import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_slopes(h, j, n):
    slopes = np.zeros(2 * n - 1)
    offset = n - 1
    lo1 = -j
    hi1 = n - 1 - j
    if lo1 <= hi1:
        lags = np.arange(lo1, hi1 + 1)
        slopes[lags + offset] += (1.0 - h[j + lags])
    lo2 = j - (n - 1)
    hi2 = j
    if lo2 <= hi2:
        lags = np.arange(lo2, hi2 + 1)
        slopes[lags + offset] += (-h[j - lags])
    return slopes

def exact_coord_update(corr, v0, slopes):
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = corr + slopes * (mid - v0)
        max_val = np.max(vals)
        tol = max(1e-12, abs(max_val) * 1e-10)
        active = vals >= max_val - tol
        min_s = np.min(slopes[active])
        max_s = np.max(slopes[active])

            hi = mid
        elif max_s < -1e-10:
```
