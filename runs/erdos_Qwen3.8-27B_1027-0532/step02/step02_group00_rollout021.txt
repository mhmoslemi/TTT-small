        return h

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

corr at k=-1 = h[1]*v[0] + h[2]*v[1] = 0.5*(0.5-ε) + 0.5*0.5 = 0.25-0.5ε+0.25 = 0.5-0.5ε
Derivative w.r.t. h[0] = -0.5 ✓

If I increase h[2] by ε: h=[0.5, 0.5, 0.5+ε], v=[0.5, 0.5, 0.5-ε]
corr at k=-1 = h[1]*v[0] + h[2]*v[1] = 0.5*0.5 + (0.5+ε)*0.5 = 0.25+0.25+0.5ε = 0.5+0.5ε
Derivative w.r.t. h[2] = 0.5 ✓

So the correct formula for k<0 (kp=-k):
d(corr)/d(h[j]) = (1-h[j-kp])*I(0≤j-kp<n) - h[j+kp]*I(0≤j+kp<n)

For the first term (1-h[j-kp]):
- j-kp ≥ 0 → j ≥ kp
- j-kp < n → j < n+
