        _obj_ap, g0, method='SLSQP', bounds=bounds,
        jac=_jac_ap, args=(alpha,),
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    if res.x is not None:
        return res.x
    return None


def slsqp_full(h0, alpha, maxiter, target):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        _obj_full, h0, method='SLSQP', bounds=bounds,
        jac=_jac_full, args=(alpha,),
        constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac, 'args': (target,)}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    if res.x is not None:
        return res.x
    return None


def nesterov_pg(h0, alpha, target, max_iter, lr_init, decay):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    t_k = 1.0
    lr = lr_init
    best_x = h0.copy()
    best_score = evaluate_score(h0)
    for it in range(max_iter):
        g = grad_smooth_max(z, alpha)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        z_new = project_feasible(z - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = z_new + ((t_k - 1.0) / t_new) * (z_new - x)
        x_new = project_feasible(x_new, target)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
        z = z_new
        t_k = t_new
        lr = max(lr_init * (decay ** it), 1e-12)
    return best_x, best_score


def nesterov_ap(g0, alpha, max_iter, lr_init, decay):
    best_x = g0.copy()
    best_score = evaluate_score(antipal_h(g0))
    z = g0.copy()
    t_k = 1.0
    lr = lr_init
    for it in range(max_iter):
        g = _jac_ap(z, alpha)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        z_clipped = np.clip(z - lr * g, 0.0, 1.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x = z_clipped + ((t_k - 1.0) / t_new) * (z_clipped - best_x)
        x = np.clip(x, 0.0, 1.0)
        score = evaluate_score(antipal_h(x))
        if score < best_score:
            best_score = score
            best_x = x.copy()
        z = z_clipped
        t_k = t_new
        lr = max(lr_init * (decay ** it), 1e-12)
    return best_x, best_score


def active_lag_grad(h, top_k_count, gap_epsilon):
    N = len(h)
    corr = compute_correlations(h)
    C = float(np.max(corr))
    active = corr > (C - gap_epsilon)
    if top_k_count is not None:
        sorted_idx = np.argsort(corr)[::-1]
        for idx in sorted_idx:
            if top_k_count <= 0:
                break
            if active[idx]:
                active[idx] = True
            else:
                active[idx] = True
            top_k_count -= 1
        active = active & (np.arange(len(corr)) < len(corr))
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        if not active[j]:
            continue
        w = 1.0 / float(np.sum(active))
        k = j - (N - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            kp = -k
            grad[:N - kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:N - kp]
    return grad, C


def active_lag_subgrad_step(h, target, gap_epsilon, lr, max_steps):
    N = len(h)
    x = h.copy()
    best_x = h.copy()
    best_score = evaluate_score(h)
    for _ in range(max_steps):
        grad, C = active_lag_grad(x, None, gap_epsilon)
        x_new = project_feasible(x - lr * grad, target)
        score = evaluate_score(x_new)
        if score < best_score:
            best_score = score
            best_x = x_new.copy()
        x = x_new
        lr *= 0.95
        if lr < 1e-10:
            break
    return best_x, best_score


def _make_structured_init(n, seed_val, pattern_type):
    rng_local = np.random.RandomState(seed_val)
    if pattern_type == 0:
        g = rng_local.rand((n - 1) // 2)
    elif pattern_type == 1:
        half = (n - 1) // 2
        g = np.where(np.arange(half) < half // 2, 1.0, 0.0) + rng_local.rand(half) * 0.1
    elif pattern_type == 2:
        half = (n - 1) // 2
        g = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(half) / half)
        g += rng_local.rand(half) * 0.05
    elif pattern_type == 3:
        half = (n - 1) // 2
        g = np.clip(0.5 + 0.3 * np.cos(4.0 * np.pi * np.arange(half) / half), 0, 1)
        g += rng_local.rand(half) * 0.1
    else:
        half = (n - 1) // 2
        blocks = rng_local.randint(2, 6)
        block_size = half // blocks
        g = np.zeros(half)
        for b in range(blocks):
            val = rng_local.rand()
            g[b * block_size:(b + 1) * block_size] = val
        g += rng_local.rand(half) * 0.05
    return np.clip(g, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n = len(init)
    target = n / 2.0
    init = project_feasible(init, target)

    best_h = init.copy()
    best_score = evaluate_score(init)
    half = n // 2
    g_best = best_h[:half].copy()

    # Phase 1: Anti-palindromic SLSQP, gradual alpha increase
    alpha_schedule = [50, 150, 400, 1000, 3000, 8000, 20000, 50000]
    for alpha in alpha_schedule:
        if time.time() - t0 > budget_s - 350:
            break
        maxiter = max(200, min(1200, 400000 // n))
        g_new = slsqp_ap(g_best, alpha, maxiter)
        if g_new is not None and np.all(np.isfinite(g_new)):
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                g_best = h_new[:half].copy()

    # Phase 2: Nesterov accelerated gradient on anti-palindromic params
    if time.time() - t0 < budget_s - 250:
        for alpha in [2000, 8000, 30000]:
            if time.time() - t0 > budget_s - 200:
                break
            lr_init = 2.0 / (alpha * half)
            g_new, s = nesterov_ap(g_best, alpha, 15000, lr_init, 0.9999)
            if s < best_score:
                best_score = s
                best_h = antipal_h(g_new)
                g_best = g_best.copy()
                g_best = g_new.copy()

    # Phase 3: Active-lag subgradient method
    if time.time() - t0 < budget_s - 200:
        for gap in [0.001, 0.005, 0.01, 0.05]:
            if time.time() - t0 > budget_s - 150:
                break
            lr_init = 5.0 / (n * gap)
            h_new, s = active_lag_subgrad_step(best_h, target, gap, lr_init, 5000)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                g_best = best_h[:half].copy()

    # Phase 4: Diverse random restarts
    n_restarts = 80
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 180:
            break

        rt = r % 7
        if rt == 0:
            scale = 0.02 + 0.10 * (r / float(n_restarts))
            pert = best_h + rng.randn(n) * scale
        elif rt == 1:
            pert = best_h.copy()
            mask = rng.random(n) < 0.04
            pert[mask] = 1.0 - pert[mask]
        elif rt == 2:
            pert = best_h.copy()
            start = rng.randint(0, max(1, n - n // 4))
            end = min(n, start + n // 6)
            pert[start:end] = 1.0 - pert[start:end]
        elif rt == 3:
            w = 0.6 + 0.3 * rng.rand()
            pert = w * best_h + (1.0 - w) * rng.rand(n)
        elif rt == 4:
            shift = rng.randint(1, max(2, n // 6))
            pert = np.roll(best_h, shift)
        elif rt == 5:
            g_init = _make_structured_init(n, seed + r, rng.randint(5))
            pert = antipal_h(g_init)
        else:
            g_init = _make_structured_init(n, seed * 7 + r, rng.randint(5))
            pert = antipal_h(g_init)

        pert = project_feasible(np.clip(pert, 0.0, 1.0), target)
        g_pert = pert[:half].copy()
        g_pert = np.clip(g_pert, 0.0, 1.0)

        alpha = 300 + 2500 * rng.randint(6)
        maxiter = max(150, min(600, 200000 // n))
        g_new = slsqp_ap(g_pert, alpha, maxiter)
        if g_new is not None and np.all(np.isfinite(g_new)):
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                g_best = h_new[:half].copy()

    # Phase 5: Multi-resolution exploration
    res_factors = [2, 3, 5, 0.5]
    for factor in res_factors:
        n_try = int(n * factor)
        if n_try < 11 or n_try > 4000:
            continue
        if time.time() - t0 > budget_s - 130:
            break

        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new_arr, x_old, best_h, period=2.0)
        h_try = project_feasible(h_try, n_try / 2.0)
        half_try = n_try // 2
        g_try = h_try[:half_try].copy()

        for alpha in [300, 1500, 6000]:
            if time.time() - t0 > budget_s - 100:
                break
            maxiter = max(100, min(500, 250000 // n_try))
            g_new = slsqp_ap(g_try, alpha, maxiter)
            if g_new is not None and np.all(np.isfinite(g_new)):
                h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    n = n_try
                    target = n / 2.0
                    half = n // 2
                    g_best = best_h[:half].copy()

    # Phase 6: Final high-alpha polish with multiple seeds of perturbation
    n_final = len(best_h)
    target_final = n_final / 2.0
    half_final = n_final // 2
    g_final = best_h[:half_final].copy()

    for alpha in [30000, 80000, 200000]:
        if time.time() - t0 > budget_s - 8:
            break
        maxiter = max(500, min(2000, 800000 // n_final))
        g_new = slsqp_ap(g_final, alpha, maxiter)
        if g_new is not None and np.all(np.isfinite(g_new)):
            h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                g_final = h_new[:half_final].copy()

    # Phase 7: Brief Nesterov polish at the current best
    if time.time() - t0 < budget_s - 5:
        for alpha in [50000, 150000]:
            if time.time() - t0 > budget_s - 3:
                break
            lr_init = 1.0 / (alpha * half_final)
            g_new, s = nesterov_ap(g_final, alpha, 5000, lr_init, 0.99999)
            if s < best_score:
                best_score = s
                best_h = antipal_h(g_new)
                g_final = g_new.copy()

    # Final cleanup
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
