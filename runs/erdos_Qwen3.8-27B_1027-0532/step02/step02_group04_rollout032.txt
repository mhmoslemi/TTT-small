def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def top_k_gradient(h, corr, alpha, k=20):
    """Compute gradient using only top k correlations"""
    N = len(h)
    # Get top k indices
    top_idx = np.argsort(corr)[-k:]
    sorted_c = np.sort(corr)[-k:]
    
    # Compute weights (softmax over top k)
    shifted = alpha * (sorted_c - sorted_c[-1])
    weights_k = np.exp(shifted - np.max(shifted))
    weights_k /= np.sum(weights_k)
    
    grad = np.zeros(N)
    for idx, wi in zip(top_idx, weights_k):
        if wi < 1e-8:

        lag = N - 1 - idx

        if abs_lag == 0:

            # grad[i] -= h[i-abs_lag] for i >= abs_lag

            grad[abs_lag:] -= wi * h[:N - abs_lag]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def top_k_gradient(h, corr, alpha, k=30):
    N = len(h)
    top_idx = np.argsort(corr)[-k:]
    sorted_c = corr[top_idx]
    shifted = alpha * (sorted_c - sorted_c[-1])
    weights_k = np.exp(shifted - np.max(shifted))
    weights_k /= np.sum(weights_k)
    grad = np.zeros(N)
    for i in range(len(top_idx)):
        wi = weights_k[i]
        if wi < 1e-8:
            continue
        lag = N - 1 - top_idx[i]
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += wi
            grad[:N - lag] -= wi * h[lag:N]
            grad[lag:] -= wi * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += wi
            grad[l:] -= wi * h[:N - l]
            grad[:N - l] -= wi * h[l:]
    return grad


def full_smooth_max_grad(h, corr, alpha):
    N = len(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(N)
    for i in range(len(weights)):
        wi = weights[i]
        if wi < 1e-10:
            continue
        lag = N - 1 - i
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < lag < N:
            grad[:N - lag] += wi
            grad[:N - lag] -= wi * h[lag:N]
            grad[lag:] -= wi * h[:N - lag]
        elif -N < lag < 0:
            l = -lag
            grad[l:] += wi
            grad[l:] -= wi * h[:N - l]
            grad[:N - l] -= wi * h[l:]
    return grad


def lbgfgs_smooth(h0, alpha, target_sum, maxiter, k=30):
    N = len(h0)

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max_value(corr, alpha)

    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if k < len(corr):
            return top_k_gradient(h, corr, alpha, k)
        return full_smooth_max_grad(h, corr, alpha)

    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x


def momentum_gd(h0, alpha, n_iters, lr0, momentum, k, target_sum, time_limit=None):

def active_set_gd(h0, n_active, n_iters, lr, target_sum, time_limit=None):
    t_start = time.time()
    h = h0.copy()
    N = len(h)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = np.max(corr)
    cur_lr = lr
    for it in range(n_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        g = np.zeros(N)
        for idx in top_idx:
            lag = N - 1 - idx
            if lag == 0:
                g += (1.0 - 2.0 * h)
            elif 0 < lag < N:
                g[:N - lag] += 1.0
                g[:N - lag] -= h[lag:N]
                g[lag:] -= h[:N - lag]
            elif -N < lag < 0:
                l = -lag
                g[l:] += 1.0
                g[l:] -= h[:N - l]
                g[:N - l] -= h[l:]
        g /= n_active
        h = h - cur_lr * g
        h = project_feasible(h, target_sum)
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        max_new = np.max(corr_new)
        if max_new < best_max:
            best_h = h.copy()
            best_max = max_new
            if it % 50 == 0 and it > 0:
                cur_lr *= 0.9
        else:
            cur_lr *= 0.98
    return best_h, best_max * (2.0 / N)


def binary_swap_search(h0, n_swaps, rng, target_sum, time_limit=None):
    t_start = time.time()
    N = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    best_h = h.copy()
    for _ in range(n_swaps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        diff = h[j] - h[i]
        if abs(diff) < 1e-6:
            continue
        h[i] += diff
        h[j] -= diff
        if h[i] < -1e-6 or h[i] > 1 + 1e-6 or h[j] < -1e-6 or h[j] > 1 + 1e-6:
            h[i] -= diff
            h[j] += diff
            continue
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        max_new = np.max(corr_new)
        if max_new < best_max - 1e-10:
            best_max = max_new
            best_h = h.copy()
        else:
            h[i] -= diff
            h[j] += diff
    return best_h, best_max * (2.0 / N)


def pair_transfer_search(h0, n_iters, rng, steps, target_sum, time_limit=None):
    t_start = time.time()
    N = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    best_h = h.copy()
    for _ in range(n_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        i = rng.randint(N)
        j = rng.randint(N)
        if i == j:
            continue
        for step in steps:
            h_try = h.copy()
            h_try[i] = h[i] + step
            h_try[j] = h[j] - step
            if h_try[i] < -1e-6 or h_try[i] > 1 + 1e-6 or h_try[j] < -1e-6 or h_try[j] > 1 + 1e-6:
                continue
            h_try[i] = np.clip(h_try[i], 0.0, 1.0)
            h_try[j] = np.clip(h_try[j], 0.0, 1.0)
            if abs(np.sum(h_try) - target_sum) > 1e-4:
                continue
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            max_try = np.max(corr_try)
            if max_try < best_max - 1e-10:
                best_max = max_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
    return best_h, best_max * (2.0 / N)


def make_binary(h, N, rng):
    binary = (h > 0.5).astype(float)
    current_sum = np.sum(binary)
    target = N / 2.0
    if N % 2 == 0:
        excess = current_sum - target
        if excess > 0:
            one_idx = np.where(binary == 1)[0]
            rng.shuffle(one_idx)
            binary[one_idx[:int(excess)]] = 0
        elif excess < 0:
            zero_idx = np.where(binary == 0)[0]
            rng.shuffle(zero_idx)
            binary[zero_idx[:int(-excess)]] = 1
    else:
        excess = current_sum - target
        if excess >= 0.5:
            one_idx = np.where(binary == 1)[0]
            rng.shuffle(one_idx)
            binary[one_idx[:int(excess)]] = 0
        elif excess <= -0.5:
            zero_idx = np.where(binary == 0)[0]
            rng.shuffle(zero_idx)
            binary[zero_idx[:int(-excess)]] = 1
        binary = project_feasible(binary, target)
    return binary

    n_final = len(best_h)
    if time_left() > 60:
        for alpha in [5000, 20000, 50000]:
            if time_left() < 40:
                break
            maxiter_f = max(50, 100000 // n_final)
            x = lbgfgs_smooth(best_h, alpha, n_final / 2.0, maxiter_f, k=10)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n_final / 2.0)
                s_new, _ = compute_c5(h_new)
                if s_new < best_score:
                    best_h = h_new
                    best_score = s_new

    # Phase 9: Very fine pair transfer
    if time_left() > 40:
        steps_fine = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        tl = max(10.0, time_left() - 20)
        n_it = max(500, int(tl * 200))
        h_new, s_new = pair_transfer_search(best_h, n_it, rng, steps_fine, n_final / 2.0, time_limit=tl)
        h_new = project_feasible(h_new, n_final / 2.0)
        s_new, _ = compute_c5(h_new)
        if s_new < best_score:
            best_h = h_new
            best_score = s_new

    # Final
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_score, _ = compute_c5(best_h)
    return (best_h, best_score, n_out)
