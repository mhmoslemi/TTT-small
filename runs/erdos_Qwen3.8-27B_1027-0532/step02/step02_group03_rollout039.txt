    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
                if diff < 1e-13:
                    break
        else:
            order = np.argsort(x)
            for i in order:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
                if diff > -1e-13:
                    break
    return x

def _smooth_corr_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_corr_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return np.zeros(n)
    exp_t = np.exp(alpha * (corr - m))
    sum_e = exp_t.sum()
    if sum_e < 1e-300:
        return np.zeros(n)
    w = exp_t / sum_e
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            ma = lag
            if ma < n:
                grad[:n - ma] += wk * (1.0 - h[ma:])
                grad[ma:] += -wk * h[:n - ma]
        else:
            ma = -lag
            if ma < n:
                grad[ma:] += wk * (1.0 - h[:n - ma])
                grad[:n - ma] += -wk * h[ma:]
    grad *= (2.0 / n)
    return grad

def _lbfgs_run(h0, n, target_sum, alpha, mu, maxiter=400):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v = _smooth_corr_val(x, alpha)
        return v * (2.0 / n) + mu * (x.sum() - target_sum) ** 2

    def grad(x):
        g = _smooth_corr_grad(x, alpha, n)
        g += 2.0 * mu * (x.sum() - target_sum)
        return g

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50})
        return _project_box_sum(res.x, target_sum)
    except Exception:
        return _project_box_sum(h0, target_sum)

def _binary_cd(h, n, target_sum, max_rounds=50):
    h = h.copy()
    for r in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            c_old = _c5_of(h)
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            if c0 < c_old - 1e-14:
                h[j] = 0.0
                improved = True
            elif c1 < c_old - 1e-14:
                h[j] = 1.0
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return _project_box_sum(h, target_sum)

def _frac_cd(h, n, target_sum, max_rounds=20):
    h = h.copy()
    vals = (0.0, 0.25, 0.5, 0.75, 1.0)
    for r in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-12:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return _project_box_sum(h, target_sum)

def _de_optimize(h_init, n, target_sum, pop_size=25, n_gen=150, time_limit=80):
    t0 = time.time()
    F = 0.8
    CR = 0.8
    rng = np.random.RandomState(int(t0 * 1000) % 100000)

    pop = [h_init.copy()]
    for i in range(1, pop_size):
        if i < pop_size // 4:
            pert = h_init + 0.4 * rng.randn(n)
            pop.append(_project_box_sum(pert, target_sum))
        elif i < pop_size // 2:
            h_rand = _project_box_sum((rng.rand(n) < 0.5).astype(float), target_sum)
            pop.append(h_rand)
        else:
            pop.append(_project_box_sum(rng.rand(n), target_sum))

    vals = [_c5_of(p) for p in pop]
    best_h = pop[int(np.argmin(vals))].copy()
    best_val = min(vals)
    no_improve = 0

    for gen in range(n_gen):
        if time.time() - t0 > time_limit:
            break
        if no_improve > 40:
            for i in range(pop_size // 3):
                pop[i] = _project_box_sum(rng.rand(n), target_sum)
                vals[i] = _c5_of(pop[i])
            no_improve = 0

        F_gen = max(0.3, F * (1.0 - gen / (n_gen * 2.0)))
        new_pop = []
        new_vals = []
        n_elite = max(2, pop_size // 5)
        elite_idx = np.argsort(vals)[:n_elite]
        for i in elite_idx:
            new_pop.append(pop[i].copy())
            new_vals.append(vals[i])

        for i in range(pop_size - len(new_pop)):
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            best_i = int(np.argmin(vals))
            mutant = pop[best_i] + F_gen * (pop[r1] - pop[r2])
            mutant = _project_box_sum(mutant, target_sum)
            trial = pop[r1].copy()
            for j in range(n):
                if rng.rand() < CR:
                    trial[j] = mutant[j]
            trial = _project_box_sum(trial, target_sum)
            tv = _c5_of(trial)
            if tv < vals[r1]:
                new_pop.append(trial)
                new_vals.append(tv)
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                new_pop.append(pop[r1].copy())
                new_vals.append(vals[r1])
                no_improve += 1

        pop = new_pop[:pop_size]
        vals = new_vals[:pop_size]

    return best_h, best_val

def _nesterov_smooth(h0, n, target_sum, n_iters=200, alpha0=0.3):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        g = _smooth_corr_grad(y, 30.0, n)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _refine_pipeline(h, n, target_sum, deadline):
    h = _project_box_sum(h, target_sum)
    c5 = _c5_of(h)

    if time.time() < deadline - 40:
        h2, cv2 = _nesterov_smooth(h, n, target_sum, n_iters=150, alpha0=0.35)
        if cv2 < c5:
            h, c5 = h2, cv2

    for alpha in [10, 50, 200]:
        if time.time() > deadline - 30:
            break
        mu = max(20.0, 10.0 * target_sum / n)
        h = _lbfgs_run(h, n, target_sum, alpha, mu, maxiter=300)
        cv = _c5_of(h)
        if cv < c5:
            c5 = cv

    if time.time() < deadline - 25:
        h = _binary_cd(h, n, target_sum, max_rounds=40)
        c5 = _c5_of(h)

    if time.time() < deadline - 20:
        h = _frac_cd(h, n, target_sum, max_rounds=15)
        c5 = _c5_of(h)

    if time.time() < deadline - 15:
        h = _lbfgs_run(h, n, target_sum, 1000, 100.0, maxiter=400)
        cv = _c5_of(h)
        if cv < c5:
            h, c5 = h, cv

    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 100
        h_parent = np.full(n_parent, 0.5)

    best_c5 = float('inf')
    best_h = None
    best_n = n_parent

    n_list = sorted(set([n_parent - 2, n_parent - 1, n_parent, n_parent + 1, n_parent + 2]))
    n_list = [nn for nn in n_list if 30 <= nn <= 250]

    for n in n_list:
        if time.time() > deadline - 200:
            break
        target_sum = n / 2.0
        n_deadline = min(deadline - 30, t_start + budget_s * 0.18 * (n_list.index(n) + 1.0))

        if n == n_parent:
            h_start = _project_box_sum(h_parent.copy(), target_sum)
        else:
            h_int = np.interp(np.linspace(0, n_parent - 1, n), np.arange(n_parent), h_parent)
            h_start = _project_box_sum(np.clip(h_int, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        de_time = min(60.0, (n_deadline - time.time()) * 0.35)
        if de_time > 15:
            h_de, c5_de = _de_optimize(h_start, n, target_sum,
                                       pop_size=25, n_gen=120, time_limit=de_time)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n
            h_curr = h_de
        else:
            h_curr = h_start.copy()

        h_curr, c5_curr = _refine_pipeline(h_curr, n, target_sum, n_deadline)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

    if best_h is not None:
        n_best = best_n
        target = n_best / 2.0
        h = best_h.copy()

        if time.time() <= deadline - 15:
            h = _lbfgs_run(h, n_best, target, 5000, 200.0, maxiter=500)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if time.time() <= deadline - 12:
            h = _binary_cd(best_h, n_best, target, max_rounds=60)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if time.time() <= deadline - 10:
            h = _frac_cd(best_h, n_best, target, max_rounds=25)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if time.time() <= deadline - 8:
            h = _lbfgs_run(best_h, n_best, target, 10000, 500.0, maxiter=500)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _c5_of(best_h)

    if best_h is None:
        best_h = _project_box_sum(h_parent, n_parent / 2.0)
        best_c5 = _c5_of(best_h)
        best_n = n_parent

    return (best_h, best_c5, best_n)
