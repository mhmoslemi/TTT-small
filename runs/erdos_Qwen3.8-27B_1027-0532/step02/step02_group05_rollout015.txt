            continue
        if time.time() - t0 > budget_s - 250:
            break
        h_up = upsample(best_h, n_new)
        t_up = n_new / 2.0
        h_up = project_feasible(h_up, t_up)

        for alpha in [100, 500]:
            if time.time() - t0 > budget_s - 200:
                break
            maxiter = max(30, min(200, 150000 // n_new))
            res_x = slsqp_opt(h_up, alpha, maxiter)
            if res_x is not None:
                h_up = project_feasible(np.clip(res_x, 0.0, 1.0), t_up)

        if time.time() - t0 < budget_s - 200:
            h_up = multisweep_cd(h_up, 30, anti_pal=False, rng=rng)
        h_up = project_feasible(h_up, t_up)
        s = evaluate_score(h_up)
        if s < best_score:
            best_score = s
            best_h = h_up.copy()
            n = n_new

    # Phase 6: Final coordinate descent + SLSQP polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    if time.time() - t0 < budget_s - 50:
        best_h = multisweep_cd(best_h, 100, anti_pal=False, rng=rng)
        best_h = project_feasible(best_h, target_final)
        s = evaluate_score(best_h)
        if s < best_score:
            best_score = s

    for alpha in [5000, 15000, 50000]:
        if time.time() - t0 > budget_s - 20:
            break
        maxiter = max(100, min(300, 400000 // n_final))
        res_x = slsqp_opt(best_h, alpha, maxiter)
        if res_x is not None:
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_final)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Final CD polish
    if time.time() - t0 < budget_s - 10:
        best_h = multisweep_cd(best_h, 50, anti_pal=False, rng=None)
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        s = evaluate_score(best_h)
        if s < best_score:
            best_score = s

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
