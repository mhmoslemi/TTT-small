        tl = min(rem() - 20, 200)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.05, 500000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 6: Restart loop
    eps_cycle = [0.002, 0.001, 0.0005, 0.0002]
    n_restart = 0
    while rem() > 100 and n_restart < 25:
        strategy = n_restart % 5
        h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)

        eps_use = eps_cycle[min(n_restart // 6, len(eps_cycle) - 1)]
        h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, 200)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if rem() > 50:
            tl_pg = min(rem() - 30, 40)
            h_pg, c5_pg = proj_grad_topk(best_h, 2, 0.005, 0.04, 100000, tl_pg, start_time)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        n_restart += 1

    # Phase 7: Very fine SLSQP
    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final projected gradient polish
    if rem() > 30:
        tl = min(rem() - 15, 100)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.02, 500000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 9: Resolution increase
    if rem() > 180:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.008 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.005, 300)
        if rem() > 80:
            h_opt_up, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.001, 300)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
        if rem() > 40:
            tl_up = min(rem() - 20, 60)
            h_pg_up, c5_pg_up = proj_grad_topk(h_opt_up, 2, 0.005, 0.04, 200000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n0 = n1

    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
