```python
def compute_delta(i, j, h, n):
    """Compute change in correlation when h[i]:1->0, h[j]:0->1.
    Returns array of length 2n-1."""
    delta = np.zeros(2*n-1)
    # Term 1: -(1-h[i+l]) for l in [-i, n-i-1]
    # In corr index: k = l + (n-1), so k in [n-1-i, n-1+n-i-1] = [n-1-i, 2n-i-2]
    l_start = -i
    l_end = n - i - 1
    for l in range(max(l_start, -(n-1)), min(l_end, n-1) + 1):
        k = l + (n - 1)
        idx = i + l
        if 0 <= idx < n:
            delta[k] -= (1.0 - h[idx])
    
    # Term 2: +(1-h[j+l]) for l in [-j, n-j-1]
    l_start = -j
    l_end = n - j - 1
    for l in range(max(l_start, -(n-1)), min(l_end, n-1) + 1):
        k = l + (n - 1)
        idx = j + l
        if 0 <= idx < n:
            delta[k] += (1.0 - h[idx])
    
    # Term 3: +h[i-l] for l in [i-n+1, i]
    l_start = i - n + 1
    l_end = i
    for l in range(max(l_start, -(n-1)), min(l_end, n-1) + 1):
        k = l + (n - 1)
        idx = i - l
        if 0 <= idx < n:
            delta[k] += h[idx]
    
    # Term 4: -h[j-l] for l in [j-n+1, j]
    l_start = j - n + 1
    l_end = j
    for l in range(max(l_start, -(n-1)), min(l_end, n-1) + 1):
        k = l + (n - 1)
        idx = j - l
        if 0 <= idx < n:
            delta[k] -= h[idx]
    
    return delta
```

```python
def swap_delta(i, j, h, n):
    """Compute delta[l] for all l when swapping h[i]:1->0, h[j]:0->1.
    h is the old state. Returns array of length 2n-1 (indexed by k = l + n - 1)."""
    delta = np.zeros(2*n - 1)
    
    def h_new_at(p):
        if p == i:
            return 0.0
        if p == j:
            return 1.0
        return h[p]
    
    for l in range(-(n-1), n):
        d = 0.0
        # Term 1: a = i, second = i+l
        if 0 <= i + l < n:
            d += -(1.0 - h[i + l])
        # Term 2: a = i-l
        a = i - l
        if 0 <= a < n:
            d += h_new_at(a)
        # Term 3: a = j, second = j+l
        if 0 <= j + l < n:
            d += (1.0 - h_new_at(j + l))
        # Term 4: a = j-l
        a = j - l
        if 0 <= a < n:
            d += -h[a]
        delta[l + (n-1)] = d
    
    return delta
```
