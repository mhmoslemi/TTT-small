import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad


def sqp_solve(h, corr, M, mu, rho_box):
    n = len(h)
    top_idx = np.argsort(-corr)[:M]
    top_lags = top_idx - (n - 1)
    top_vals = corr[top_idx]

    delta = cp.Variable(n)
    t = cp.Variable()

    obj = cp.Minimize(t + mu * cp.sum_squares(delta))

    lower = -np.minimum(h, rho_box)
    upper = np.minimum(1.0 - h, rho_box)

    cons = [cp.sum(delta) == 0,
            delta >= lower,
            delta <= upper]

    for val, lag in zip(top_vals, top_lags):
        g = compute_grad_lag(h, int(lag))
        cons.append(val + g @ delta <= t)

    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None

    if delta.value is None or not np.all(np.isfinite(delta.value)):
        return None, None
    return delta.value, t.value


def subgradient_phase(h, t0, time_limit, alpha_sm, n_top, step_init, seed):
    n = len(h)
    target_sum = n / 2.0
    h = project_to_feasible(h.copy(), target_sum)
    rng = np.random.RandomState(seed)

    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    stall = 0

    while time.time() - t0 < t0 + time_limit:
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        shifted = top_vals - top_vals[0]
        weights = np.exp(alpha_sm * shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_grad_lag(h, lag)

        improved = False
        for trial in range(40):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score - 1e-12:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
                improved = True
                break
            step *= 0.5

        if not improved:
            stall += 1

                h = project_to_feasible(h, target_sum)
                step = step_init * (0.5 + rng.rand())
                stall = 0

def fourier_perturb(h, n_modes, amplitude, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for m in range(1, n_modes + 1):
        coef1 = rng.randn() * amplitude
        coef2 = rng.randn() * amplitude
        pert += coef1 * np.cos(2.0 * np.pi * m * x / n)
        pert += coef2 * np.sin(2.0 * np.pi * m * x / n)
    pert -= np.mean(pert)
    return h + pert

def slsqp_refine(h, alpha, maxiter):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def con_fun(hv):
        return np.sum(hv) - target_sum

    def con_jac(hv):
        return np.ones(n)

    res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
