        weights = smooth_max_weights(corr_y, alpha)

        grad = np.zeros(n)
        active_mask = weights > 1.0 / (3.0 * n)
        active_idx = np.where(active_mask)[0]
        for idx in active_idx:
            lag = int(idx - (n - 1))
            grad += weights[idx] * compute_gradient_for_lag(y, lag, n)

        grad_norm = np.sqrt(np.sum(grad * grad))
        if grad_norm < 1e-14:
            break

        x_new = y - step * grad
        x_new = project_to_feasible(x_new, target_sum)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc)) / 2.0
        momentum = (t_acc - 1.0) / t_new
        y = x_new + momentum * (x_new - h)
        y = np.clip(y, 0.0, 1.0)

        h = x_new
        t_acc = t_new

        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

        if it % 200 == 0 and it > 0:
            step *= 0.97

    return best_h, best_score


def solve_active_lp(h_current, active_lags, n):
    c = np.zeros(n + 1)
    c[n] = 1.0

    full_corr = compute_correlations(h_current)
    A_ub_rows = []
    b_ub_rows = []

    for k in active_lags:
        idx = k + n - 1
        idx = max(0, min(2 * n - 2, idx))
        grad = compute_gradient_for_lag(h_current, k, n)
        corr_val = full_corr[idx]
        const = corr_val - np.dot(grad, h_current)

        row = np.zeros(n + 1)
        row[:n] = grad
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(-const)

    if not A_ub_rows:
        return h_current.copy()

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n]
    return h_current.copy()


def active_set_optimize(h0, max_rounds=80, max_lags=12, t0=None, time_limit=60):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)

    if t0 is None:
        t0 = time.time()

    best_score = evaluate_score(h)
    best_h = h.copy()
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() - t0 > t0 + time_limit:
            break

        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:max_lags]
        active_lags = [int(idx - (n - 1)) for idx in top_idx]

        h_new = solve_active_lp(h, active_lags, n)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, target_sum)

        score_new = evaluate_score(h_new)

        if score_new < best_score - 1e-12:
            h = h_new
            best_score = score_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            improved = False
            for theta in [0.5, 0.3, 0.7, 0.2]:
                h_try = (1.0 - theta) * h + theta * h_new
                h_try = project_to_feasible(h_try, target_sum)
                s = evaluate_score(h_try)
                if s < best_score - 1e-12:
                    h = h_try
                    best_score = s
                    best_h = h_try.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 8:
                    break

    return best_h, best_score


def coordinate_descent(h0, n_sweeps=3, t0=None, time_limit=30):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)

    if t0 is None:
        t0 = time.time()

    best_score = evaluate_score(h)
    best_h = h.copy()

    for sweep in range(n_sweeps):
        if time.time() - t0 > t0 + time_limit:
            break

        for i in range(n):
            if time.time() - t0 > t0 + time_limit:
                break
            old_val = h[i]
            best_val = old_val
            best_local = np.inf

            n_cand = 9
            lo_b = max(0.0, old_val - 0.08)
            hi_b = min(1.0, old_val + 0.08)
            candidates = np.linspace(lo_b, hi_b, n_cand)

            for tv in candidates:
                h[i] = tv
                corr = compute_correlations(h)
                sc = np.max(corr)
                if sc < best_local:
                    best_local = sc
                    best_val = tv

            h[i] = best_val

        h = project_to_feasible(h, target_sum)
        score = evaluate_score(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()

    return best_h, best_score


def upsample_and_refine(h_coarse, n_new, t0, time_limit):
    n_old = len(h_coarse)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_coarse)
    h_new = project_to_feasible(h_new, n_new / 2.0)

    h_out, score = fista_smoothed(h_new, 30.0, 3000, t0, time_limit * 0.4)
    h_out, score = active_set_optimize(h_out, max_rounds=30, max_lags=10,
                                        t0=t0, time_limit=time_limit * 0.5)
    return h_out, score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n_init = len(init)
    init = project_to_feasible(init, n_init / 2.0)

    best_h = init.copy()
    best_score = evaluate_score(init)
    best_n = n_init

    rng = np.random.RandomState(seed)

    # Phase 1: FISTA from various starting points
    if remaining() > budget_s * 0.3:
        starts = [best_h.copy()]
        for p in range(3):
            perturbed = best_h + rng.randn(n_init) * 0.015 / np.sqrt(n_init)
            perturbed = project_to_feasible(perturbed, n_init / 2.0)
            starts.append(perturbed)

        for si, sh in enumerate(starts):
            if remaining() < budget_s * 0.2:
                break
            alpha = 20.0 + 15.0 * si
            t_allow = min(remaining() * 0.25, 120)
            if t_allow < 10:
                break
            h_f, s_f = fista_smoothed(sh, alpha, 8000, t0, t_allow)
            if s_f < best_score:
                best_score = s_f
                best_h = h_f.copy()

    # Phase 2: Active set LP refinement
    if remaining() > budget_s * 0.15:
        t_allow = min(remaining() * 0.35, 300)
        if t_allow > 20:
            h_a, s_a = active_set_optimize(best_h, max_rounds=120, max_lags=15,
                                            t0=t0, time_limit=t_allow)
            if s_a < best_score:
                best_score = s_a
                best_h = h_a.copy()

    # Phase 3: Coordinate descent
    if remaining() > budget_s * 0.1:
        t_allow = min(remaining() * 0.3, 120)
        if t_allow > 10:
            for cs in range(3):
                if remaining() < budget_s * 0.05:
                    break
                h_c, s_c = coordinate_descent(best_h, n_sweeps=4,
                                               t0=t0, time_limit=t_allow / 3)
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()

    # Phase 4: More active set rounds
    if remaining() > budget_s * 0.12:
        t_allow = min(remaining() * 0.4, 250)
        if t_allow > 15:
            h_a, s_a = active_set_optimize(best_h, max_rounds=100, max_lags=20,
                                            t0=t0, time_limit=t_allow)
            if s_a < best_score:
                best_score = s_a
                best_h = h_a.copy()

    # Phase 5: Try upsampling
    if remaining() > budget_s * 0.2:
        n_candidates = [int(best_n * 2), int(best_n * 2.5), int(best_n * 4)]
        n_candidates = sorted(set(x for x in n_candidates if 500 < x <= 3000))
        for n_try in n_candidates:
            if remaining() < budget_s * 0.08:
                break
            t_allow = min(remaining() * 0.15, 100)
            if t_allow < 15:
                break
            h_up, s_up = upsample_and_refine(best_h, n_try, t0, t_allow)
            if s_up < best_score:
                best_score = s_up
                best_h = h_up.copy()
                best_n = n_try

    # Phase 6: Final refinement loop
    if remaining() > 30:
        for _ in range(5):
            if remaining() < 15:
                break
            h_a, s_a = active_set_optimize(best_h, max_rounds=50, max_lags=15,
                                            t0=t0, time_limit=min(remaining() * 0.3, 60))
            if s_a < best_score:
                best_score = s_a
                best_h = h_a.copy()
            else:
                h_c, s_c = coordinate_descent(best_h, n_sweeps=3,
                                               t0=t0, time_limit=min(remaining() * 0.2, 40))
                if s_c < best_score:
                    best_score = s_c
                    best_h = h_c.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
