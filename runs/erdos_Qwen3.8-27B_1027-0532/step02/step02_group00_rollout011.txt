import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_grad_smooth(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_s = corr * dx
    c_max = np.max(corr_s)
    shifted = alpha * (corr_s - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    logsum = np.log(np.sum(exp_shifted))
    f = c_max + logsum / alpha
    
    p = exp_shifted / np.sum(exp_shifted)
    
    T1 = np.convolve(p, 1.0 - h, 'full')[n-1:2*n-1]
    T2 = np.correlate(p, h, 'full')[0:n][::-1].copy()
    T2[0] += p[-1] * h[-1]
    
    grad = dx * (T1 - T2)
    if lam > 0:
        grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    
    f_total = f + lam * (np.sum(h) - n / 2.0) ** 2
    return f_total, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def optimize_smooth(h_start, n, target, alpha, lam, maxiter=10000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        compute_c5_grad_smooth, h_start,
        args=(alpha, lam),
        method='L-BFGS-B',
        jac=True,
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12, 'maxls': 50}
    )
    return project_feasible(res.x, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
