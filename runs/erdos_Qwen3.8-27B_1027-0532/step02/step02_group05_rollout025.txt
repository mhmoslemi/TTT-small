        if k >= 0:
            grad_h[k:] += (1.0 - h[:2 * half + 1 - k])
            grad_h[:2 * half + 1 - k] -= h[k:]
        else:
            kp = -k
            grad_h[:2 * half + 1 - kp] += (1.0 - h[kp:])
            grad_h[kp:] -= h[:2 * half + 1 - kp]
        grad_g = grad_h[:half].copy()
        for i in range(half):
            grad_g[i] += -grad_h[2 * half - i]
        g_new = np.clip(g - lr * grad_g, 0.0, 1.0)
        score = obj_anti_exact(g_new)
        if score < best_score:
            best_score = score
            best_g = g_new.copy()
            g = g_new
            lr *= 1.01
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_g, best_score


def coordinate_descent_anti(g0, max_sweeps=20):
    half = len(g0)
    g = g0.copy()
    best_score = obj_anti_exact(g)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(half):
            old_val = g[i]
            current_score = best_score
            candidates = [old_val + d for d in [-0.1, -0.05, -0.02, -0.01, 0.0, 0.01, 0.02, 0.05, 0.1]]
            candidates.extend([max(0.0, min(1.0, old_val * 0.5)), max(0.0, min(1.0, old_val * 1.5))])
            best_cand = old_val
            best_cand_score = obj_anti_exact(g)
            for c in candidates:
                c = max(0.0, min(1.0, c))
                if c == old_val:
                    continue
                g[i] = c
                s = obj_anti_exact(g)
                if s < best_cand_score:
                    best_cand_score = s
                    best_cand = c
            g[i] = best_cand
            if best_cand != old_val:
                improved = True
        score = obj_anti_exact(g)
        if score < best_score:
            best_score = score
        if not improved:
            break
    return g, best_score


def de_low_res(n, rng, maxiter, popsize):
    half = (n - 1) // 2
    bounds = [(0.0, 1.0)] * half

    def obj_de(g):
        h = build_anti_h(g)
        return evaluate_score(h)

    try:
        res = differential_evolution(
            obj_de, bounds, maxiter=maxiter, popsize=popsize,
            seed=rng.randint(2**31), tol=1e-8, mutation=(0.3, 1.0),
            recombination=0.8, polish=False, init='soothe'
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1300, **kwargs):

    init = np.array(initial_h_values, dtype=float)

    best_score = evaluate_score(h_best)
    n_current = len(h_best)

    for alpha in [50, 100, 200, 500, 1000, 2000, 5000]:
        if remaining() < 200:

        g_new = slsqp_anti(pert, alpha, maxiter)
        if g_new is not None:

    if n_final % 2 == 1:
