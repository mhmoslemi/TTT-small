```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_value(corr, alpha):
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    return np.max(corr) + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha

def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w * (1.0 - h[lag:])
                grad[lag:] += w * (-h[:N - lag])
        else:
            l = -lag
            if l < N:
                grad[l:] += w * (1.0 - h[:N - l])
                grad[:N - l] += w * (-h[l:])
    return grad

def lbfgs_optimize(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s
    
    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

def binary_swap_search(h_bin, n, rng, n_iters, T_initial=0.01, T_final=0.001):
    """Simulated annealing on binary vector with fixed number of ones."""
    h = h_bin.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    corr = compute_correlations(h)
    current_max = np.max(corr)
    best = h.copy()
    best_corr = corr.copy()
    best_max = current_max
    
    n_total = n_iters
    for it in range(n_total):
        # Annealing schedule
        frac = it / max(n_total - 1, 1)
        T = T_initial * (T_final / T_initial) ** frac
        
        # Pick a random 1-position and 0-position to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for all lags
        delta = np.zeros(n)  # delta for lags -n+1 to n-1, stored at index lag+n-1
        # Actually, corr has 2n-1 entries. Let me just compute the new corr directly.
        
        # Update correlation
        corr_new = corr.copy()
        for L in range(-(n-1), n):
            d = 0.0
            # i: 1->0
            ipL = i + L
            if 0 <= ipL < n:
                d -= (1.0 - h[ipL])
            imL = i - L
            if 0 <= imL < n and L != 0:
                d += h[imL]
            # j: 0->1
            jpL = j + L
            if 0 <= jpL < n:
                d += (1.0 - h[jpL])
            jmL = j - L
            if 0 <= jmL < n and L != 0:
                d -= h[jmL]
            corr_new[L + n - 1] += d
        
        new_max = np.max(corr_new)
        
        # Accept or reject
        if new_max <= current_max or (T > 0.0001 and rng.rand() < np.exp(-(new_max - current_max) / T)):
            # Accept
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            corr = corr_new
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best = h.copy()
                best_corr = corr.copy()
        # else: reject, keep current state
    
    return best, best_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_value(corr, alpha):
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    return np.max(corr) + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha


def smooth_max_gradient(h, alpha):
    N = len(h)
    corr = compute_correlations(h)
    shifted = alpha * corr
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(N)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = N - 1 - index
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            if lag < N:
                grad[:N - lag] += w * (1.0 - h[lag:])
                grad[lag:] += w * (-h[:N - lag])
        else:
            l = -lag
            if l < N:
                grad[l:] += w * (1.0 - h[:N - l])
                grad[:N - l] += w * (-h[l:])
    return grad


def lbfgs_optimize(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

    n_total = n_iters

def local_search_continuous(h, n, rng, n_iters, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best = h.copy()
    best_corr = compute_correlations(best)
    best_max = np.max(best_corr) * (2.0 / n)
    corr = best_corr.copy()
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        improved = False
        for step in steps:
            h_try = best.copy()
            h_try[i] = best[i] + step
            h_try[j] = best[j] - step
            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            corr_try = compute_correlations(h_try)
            max_try = np.max(corr_try)

                best = h_try
                corr = corr_try
                if max_try * (2.0 / n) < best_max:
                    best_max = max_try * (2.0 / n)
                improved = True
                break
    return best, best_max

            best_h, _ = local_search_continuous(best_h, n_cur, rng, n_ls)
            best_h = project_feasible(best_h, n_cur / 2.0)
            s = evaluate_score(best_h)
            if s < best_score:
                best_score = s

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
```
