import numpy as np
from scipy.optimize import minimize
import time

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_score(h):
    n = len(h)
    return float(np.max(full_corr(h))) * 2.0 / n

def project(h, target):
    N = len(h)
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    p = np.clip(h - lam, 0.0, 1.0)
    d = target - float(np.sum(p))
    if abs(d) > 1e-14:
        mask = (p > 1e-12) & (p < 1.0 - 1e-12)
        c = int(np.sum(mask))
        if c > 0:
            p[mask] += d / c
            np.clip(p, 0.0, 1.0, out=p)
    return p

def compute_grad(h, weights):
    N = len(h)
    g = np.zeros(N)
    for j in range(2 * N - 1):
        wj = weights[j]
        if wj < 1e-15:
            continue
        k = N - 1 - j
        if k >= 0:
            g[:N - k] += wj * (1.0 - h[k:])
            g[k:] -= wj * h[:N - k]
        else:
            kp = -k
            g[kp:] += wj * (1.0 - h[:N - kp])
            g[:N - kp] -= wj * h[kp:]
    return g

def softmax_weights(corr, alpha):
    m = float(corr.max())
    e = np.exp(alpha * (corr - m))
    s = float(e.sum())
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return e / s

def smooth_max_val(corr, alpha):
    m = float(corr.max())
    return m + float(np.log(np.sum(np.exp(alpha * (corr - m))))) / alpha

def nesterov_opt(h0, target, alpha, max_iters, t_limit, lr_init=None):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    tk = 1.0
    if lr_init is None:
        lr = 0.5 / (alpha + 10.0)
    else:
        lr = lr_init
    best_x = h0.copy()
    best_s = eval_score(h0)
    best_smooth = smooth_max_val(full_corr(h0), alpha)
    for it in range(max_iters):
        if time.time() > t_limit:
            break
        c_z = full_corr(z)
        w = softmax_weights(c_z, alpha)
        gz = compute_grad(z, w)
        norm_g = float(np.linalg.norm(gz))
        if norm_g < 1e-15:
            break
        x_new = project(z - lr * gz, target)
        f_new = smooth_max_val(full_corr(x_new), alpha)
        if f_new > best_smooth + 1.0 / (alpha + 1.0) * norm_g:
            lr *= 0.5
            x_new = project(z - lr * gz, target)
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        z_new = x_new + ((tk - 1.0) / tk_new) * (x_new - x)
        z_new = project(z_new, target)
        x = x_new
        z = z_new
        tk = tk_new
        if f_new < best_smooth:
            best_smooth = f_new
        if it % 25 == 0 or it == max_iters - 1:
            s_actual = eval_score(x)
            if s_actual < best_s:
                best_s = s_actual
                best_x = x.copy()
        lr *= 0.9995
    s_actual = eval_score(x)
    if s_actual < best_s:
        best_s = s_actual
        best_x = x.copy()
    return best_x, best_s

def slsqp_opt(h0, target, alpha, maxiter):
    N = len(h0)
    def fun(x):
        c = full_corr(x)
        m = float(c.max())
        return m + float(np.log(np.sum(np.exp(alpha * (c - m))))) / alpha
    def jac(x):
        c = full_corr(x)
        w = softmax_weights(c, alpha)
        return compute_grad(x, w)
    def cons_fun(x):
        return float(np.sum(x)) - target
    def cons_jac(x):
        return np.ones(N)
    bounds = [(0.0, 1.0)] * N
    res = minimize(fun, h0, method='SLSQP', bounds=bounds,
                   jac=jac,
                   constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project(np.clip(res.x, 0.0, 1.0), target)
    return None

def antipal_h(g):
    half = len(g)
    N = 2 * half + 1
    h = np.empty(N)
    h[:half] = g
    h[half] = 0.5
    h[half + 1:] = 1.0 - g[::-1]
    return h

def ap_smooth_grad(g, alpha):
    h = antipal_h(g)
    c = full_corr(h)
    w = softmax_weights(c, alpha)
    grad_h = compute_grad(h, w)
    half = len(g)
    n_full = len(h)
    grad_g = grad_h[:half].copy()
    for i in range(half):
        j = n_full - 1 - i
        grad_g[i] += -grad_h[j]
    return grad_g

def ap_nesterov_opt(g0, alpha, max_iters, t_limit):
    half = len(g0)
    N_full = 2 * half + 1
    g = g0.copy()
    zg = g0.copy()
    tk = 1.0
    lr = 0.3 / (alpha + 10.0)
    best_g = g0.copy()
    best_s = eval_score(antipal_h(g0))
    for it in range(max_iters):
        if time.time() > t_limit:
            break
        gz = ap_smooth_grad(zg, alpha)
        ng = float(np.linalg.norm(gz))
        if ng < 1e-15:
            break
        g_new_raw = zg - lr * gz
        g_new = np.clip(g_new_raw, 0.0, 1.0)
        h_new = antipal_h(g_new)
        h_new = project(h_new, N_full / 2.0)
        g_new = h_new[:half]
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        zg_new = g_new + ((tk - 1.0) / tk_new) * (g_new - g)
        zg_new = np.clip(zg_new, 0.0, 1.0)
        g = g_new
        zg = zg_new
        tk = tk_new
        if it % 20 == 0 or it == max_iters - 1:
            s_actual = eval_score(antipal_h(g))
            if s_actual < best_s:
                best_s = s_actual
                best_g = g.copy()
        lr *= 0.9997
    s_actual = eval_score(antipal_h(g))
    if s_actual < best_s:
        best_s = s_actual
        best_g = g.copy()
    return best_g, best_s

def topk_nesterov(h0, target, topk, temperature, max_iters, t_limit):
    N = len(h0)
    x = h0.copy()
    z = h0.copy()
    tk = 1.0
    lr = 0.8 / (N * 0.01)

    best_s = eval_score(h0)
    for it in range(max_iters):

        c_z = full_corr(z)
        idx = np.argsort(c_z)[-topk:]
        w = np.zeros(len(c_z))
        vals = c_z[idx]
        m = float(vals.max())
        e = np.exp(temperature * (vals - m))
        w[idx] = e / float(e.sum())
        gz = compute_grad(z, w)
        x_new = project(z - lr * gz, target)
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        z_new = x_new + ((tk - 1.0) / tk_new) * (x_new - x)
        z_new = project(z_new, target)
        x = x_new
        z = z_new
        tk = tk_new
        if it % 30 == 0:
            s = eval_score(x)
            if s < best_s:
                best_s = s
                best_x = x.copy()
        lr *= 0.9998
    s = eval_score(x)
    if s < best_s:
        best_s = s
        best_x = x.copy()
    return best_x, best_s
