import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def nesterov_subgradient(h0, t0, time_limit, alpha, n_top, seed):
    n = len(h0)
    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    v = np.zeros(n)  # momentum
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    
    step = 1.0 / np.sqrt(n)
    momentum = 0.9
    stall = 0
    max_stall = 200
    
    it = 0
    while time.time() - t0 < t0 + time_limit:
        it += 1
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]
        
        # Smoothed max gradient
        m = np.max(top_vals)
        shifted = top_vals - m
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            lag = int(idx - (n - 1))
            grad += w * compute_gradient_for_lag(h, lag)

        v = momentum * v - step * grad
        h_new = h + v
        
        # Project
        h_new = project_to_feasible(h_new, target_sum)
        
        # Evaluate
        score = evaluate_score(h_new)

            best_score = score
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

        # Periodic step size adjustment

If |A| = 1 (only one active lag), the optimum is at t=0 (if slope > 0) or t=1 (if slope < 0).
If |A| = 2, the optimum is at the intersection of the two lines (if it's in [0,1]).
If |A| > 2, it's more complex but still solvable.

In practice, at the optimum, typically 2-3 lags are active (by Carathéodory's theorem in the 1D case, at most 2 lines determine the minimum of their maximum).

So a practical approach:
1. Compute correlations
2. Find the top few lags
3. For pairs of top lags, compute their intersection point
4. Try those intersection points as candidates for h[j]
5. Pick the best

Let me now design the full algorithm:

**Phase 1: Subgradient with momentum (50% of time)**
- Standard subgradient with Nesterov momentum
- Adaptive step size

**Phase 2: Coordinate descent on true max (30% of time)**
- Cycle through coordinates
- For each coordinate, find the best value using the piecewise linear structure
- Multiple passes

**Phase 3: Local refinement (20% of time)**

    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def coordinate_descent_step(h, j, corr):
    """Find optimal value for h[j] given all other coordinates fixed."""
    n = len(h)
    # For each lag, compute slope and intercept of c_lag as function of h[j]
    # c_{n-1+lag} = intercept + slope * h[j]
    slopes = np.zeros(2*n-1)
    # slope for lag ℓ at position j: (1-h[j-ℓ]) - h[j+ℓ] with boundary handling
    
    for lag_idx in range(-(n-1), n):
        # h[j] appears as h[n+lag] when n = j-lag, coeff (1-h[j-lag])
        # h[j] appears as (1-h[n]) when n = j, coeff -h[j+lag]
        a = 0.0
        b = 0.0
        if 0 <= j - lag_idx < n:
            a = 1.0 - h[j - lag_idx]
        if 0 <= j + lag_idx < n:
            b = h[j + lag_idx]
        slopes[lag_idx + n - 1] = a - b
    
    # intercept = corr - h[j] * slope
    intercepts = corr - h[j] * slopes
    
    # Now find t in [0,1] minimizing max_k (intercepts[k] + t * slopes[k])
    # This is a 1D convex piecewise linear problem
    
    # Evaluate at endpoints
    f0 = np.max(intercepts)
    f1 = np.max(intercepts + slopes)
    
    candidates = [0.0, 1.0]
    
    # Find intersection points of pairs of "top" lines
    # For efficiency, only consider the top few lines
    current_max = np.max(intercepts + h[j] * slopes)
    # Actually, let's just find the top lines at t=0 and t=1 and current t
    top_at_0 = np.argsort(-intercepts)[:5]
    top_at_1 = np.argsort(-(intercepts + slopes))[:5]
    top_at_cur = np.argsort(-corr)[:5]
    
    all_top = list(set(list(top_at_0) + list(top_at_1) + list(top_at_cur)))
    
    for i in range(len(all_top)):
        for j2 in range(i+1, len(all_top)):
            si = slopes[all_top[i]]
            sj = slopes[all_top[j2]]
            ci = intercepts[all_top[i]]
            cj = intercepts[all_top[j2]]
            if abs(si - sj) > 1e-12:
                t_int = (cj - ci) / (si - sj)
                if 0.0 <= t_int <= 1.0:
                    candidates.append(t_int)
    
    # Also try current value
    candidates.append(h[j])
    
    best_t = h[j]
    best_val = current_max
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        val = np.max(intercepts + t * slopes)
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    
    return best_t

def coordinate_descent_pass(h, max_rounds=5, t0=None, time_limit=None):

        for j in range(n):
            if t0 is not None and (j % 50 == 0) and (time.time() - t0 > t0 + time_limit):
                break
            
            new_t = coordinate_descent_step(h, j, corr)
            if abs(new_t - h[j]) > 1e-10:
                h[j] = new_t
                improved = True
                # Update correlations (approximate - recompute periodically)
        
        if not improved:
            break
    
    return h

def subgradient_run(h0, t0, time_limit, alpha=20, n_top=10, step_init=0.5, seed=42):

    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = step_init
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    
    it = 0
    while time.time() - t0 < t0 + time_limit:
        it += 1
        corr = compute_correlations(h)
        top_idx = np.argsort(-corr)[:n_top]
        top_vals = corr[top_idx]

        weights = np.exp(alpha * shifted)
        weights /= weights.sum()

                grad[:n-la] += w * (1.0 - h[la:])
                grad[la:] -= w * h[:n-la]

            stall = 0

            step *= 0.95

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    
    def objective(v):
        corr = compute_correlations(v)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

        if np.sum(x) > target_sum:

        else:
            hi = mid

        if np.max(active_slopes) < -1e-15:

    lags = np.arange(-(n-1), n)
    rng = np.random.RandomState(seed)
    corr = compute_correlations(h)

            idx_b = j + lags
            a = np.zeros(2*n-1)
            mask_a = (idx_a >= 0) & (idx_a < n)
            a[mask_a] = 1.0 - h[idx_a[mask_a]]

    target_sum = n / 2.0
    h = project_to_feasible(h0.copy(), target_sum)
    rng = np.random.RandomState(seed)
    best_score = evaluate_score(h)
    best_h = h.copy()

    v = np.zeros(n)
    momentum = 0.8
    stall = 0
    it = 0
    lags = np.arange(-(n-1), n)
    
    while time.time() - t0 < t0 + time_limit:
        it += 1

        top_idx = np.argsort(-corr)[:n_top]

def slsqp_refine(h, alpha, maxiter=200):

    target_sum = n / 2.0
    def objective(v):
        corr = compute_correlations(v)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    def constraint_fun(v):
        return np.sum(v) - target_sum

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def find_optimal_t(slopes, intercepts):
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = intercepts + mid * slopes
        max_val = np.max(vals)
        active_mask = vals >= max_val - 1e-13
        active_slopes = slopes[active_mask]
        if np.max(active_slopes) < -1e-14:
            hi = mid
        elif np.min(active_slopes) > 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def coord_desc_pass(h, max_rounds=10, t0=None, time_limit=None, seed=42):
    n = len(h)
    target_sum = n / 2.0
    lags = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    corr = compute_correlations(h)

            break
        improved = False
        order = rng.permutation(n)

            if t0 is not None and (j % 200 == 0) and (time.time() - t0 > t0 + time_limit):
                break
            idx_a = j - lags
            idx_b = j + lags
            a = np.zeros(2 * n - 1)

            a[mask_a] = 1.0 - h[idx_a[mask_a]]

                improved = True

            break

        corr = compute_correlations(h)

        corr = compute_correlations(h)

        top_vals = corr[top_idx]
        m = np.max(top_vals)
        shifted = top_vals - m
        weights = np.exp(alpha * shifted)
        weights /= weights.sum()

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0

        corr = compute_correlations(v)
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def constraint_fn(v):
        return np.sum(v) - target_sum

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': constraint_fn}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None
