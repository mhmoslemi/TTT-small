from scipy.optimize import minimize
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return float(np.max(corr) * dx)

def make_anti_pal_from_g(g):
    M = len(g)
    n = 2 * M + 1
    h = np.empty(n)
    h[:M] = g
    h[M] = 0.5
    h[M+1:] = 1.0 - g[::-1]
    return h

def smooth_max(x, alpha):
    m = float(np.max(x))
    shifted = alpha * (x - m)
    # Clip to avoid overflow (shouldn't happen since x-m <= 0)
    shifted = np.clip(shifted, -700, 0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / alpha

def ap_fun(g, alpha):
    h = make_anti_pal_from_g(g)
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h) * dx
    return smooth_max(corr, alpha)
