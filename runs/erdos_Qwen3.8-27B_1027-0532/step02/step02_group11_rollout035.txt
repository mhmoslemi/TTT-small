    w = exp_shifted / np.sum(exp_shifted)

    B_ext[1:] = w[:n-1][::-1]  # B_ext[m] = w[n-1-m] for m=1..n-1

    T4 = np.convolve(B_ext, h, mode='full')[0:n]

def lbfgs_optimize(h_start, n, eps, maxiter, time_limit_fn):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, eps)
    
    def grad(x):
        return smooth_max_grad(x, eps)
    
    res = minimize(obj, h_start.copy(), jac=grad, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxls': 50, 'maxfun': 10*maxiter})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def binary_swap_search(h_start, n, max_iters, time_limit_fn):

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_max_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    obj_val = (m + (1.0 / eps) * np.log(sum_exp)) * dx
    A = w[n - 1:]
    B_ext = np.zeros(n)
    B_ext[1:] = w[:n - 1][::-1]
    T1 = np.convolve(A, 1.0 - h, mode='full')[0:n]
    T2 = np.correlate(h, A, mode='full')[n - 1:2 * n - 1]
    T3 = np.correlate(1.0 - h, B_ext, mode='full')[n - 1:2 * n - 1]
    T4 = np.convolve(B_ext, h, mode='full')[0:n]
    grad = dx * (T1 - T2 + T3 - T4)
    return obj_val, grad

def lbfgs_optimize(h_start, n, eps, maxiter, maxfun, time_limit_fn):
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        val, _ = smooth_max_obj_grad(x, eps)
        return val

    def grad_fn(x):
        _, g = smooth_max_obj_grad(x, eps)
        return g

    res = minimize(obj_fn, h_start.copy(), jac=grad_fn, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxls': 100, 'maxfun': maxfun})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start.copy(), evaluate_c5(h_start)

def binary_swap_search(h_start, n, time_limit_fn, max_iters=500000):
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stagnant = 0
    for it in range(max_iters):
        if time_limit_fn():
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_try = h.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        c5 = evaluate_c5(h_try)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            h = h_try
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30000:
                break
    return h, best_c5

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def downsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = float(i) * n_from / n_to
        idx = int(pos)
        if idx >= n_from:
            idx = n_from - 1
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def time_at(frac):
        return lambda: elapsed() > budget_s * frac

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = project_feasible(h0.copy(), n0)
    best_c5 = evaluate_c5(best_h)

    # Phase 1: L-BFGS-B with eps annealing on initial n
    eps_schedule = [200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0, 50000.0, 100000.0]
    for eps in eps_schedule:
        if elapsed() > budget_s * 0.30:
            break
        tl = time_at(0.30)
        h_new, c5_new = lbfgs_optimize(best_h, n0, eps, 2000, 100000, tl)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = project_feasible(h_new, n0)

    # Phase 2: Binary local search
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n0)
    c5_bin = evaluate_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    tl = time_at(0.45)
    h_bin, c5_bin = binary_swap_search(h_bin, n0, tl)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    # Refine with L-BFGS-B after binary search
    if elapsed() < budget_s * 0.52:
        tl = time_at(0.52)
        h_new, c5_new = lbfgs_optimize(best_h, n0, 50000.0, 3000, 100000, tl)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = project_feasible(h_new, n0)

    # Phase 3: Try different resolutions
    for n_try in [85, 255, 340]:
        if elapsed() > budget_s * 0.75:
            break
        if n_try == n0:
            continue
        if n_try > n0:
            h_init = project_feasible(upsample(best_h, n0, n_try), n_try)
        else:
            h_init = project_feasible(downsample(best_h, n0, n_try), n_try)
        tl = time_at(0.75)
        for eps in [2000.0, 10000.0, 50000.0]:
            if elapsed() > budget_s * 0.75:
                break
            h_new, c5_new = lbfgs_optimize(h_init, n_try, eps, 1500, 80000, tl)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = project_feasible(h_new, n_try)
                h_init = best_h
        if elapsed() < budget_s * 0.82 and len(best_h) == n_try:
            h_bi = (best_h > 0.5).astype(float)
            h_bi = project_feasible(h_bi, n_try)
            tl = time_at(0.82)
            h_bi, c5_bi = binary_swap_search(h_bi, n_try, tl, max_iters=200000)
            if c5_bi < best_c5:
                best_c5 = c5_bi
                best_h = h_bi

    # Phase 4: Final refinement on best solution
    n_final = len(best_h)
    if elapsed() < budget_s * 0.90:
        tl = time_at(0.90)
        for eps in [20000.0, 50000.0, 100000.0, 200000.0]:
            if elapsed() > budget_s * 0.90:
                break
            h_new, c5_new = lbfgs_optimize(best_h, n_final, eps, 3000, 100000, tl)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = project_feasible(h_new, n_final)

    # Phase 5: Final binary search pass
    if elapsed() < budget_s * 0.97:
        h_bi = (best_h > 0.5).astype(float)
        h_bi = project_feasible(h_bi, n_final)
        tl = time_at(0.97)
        h_bi, c5_bi = binary_swap_search(h_bi, n_final, tl, max_iters=300000)
        if c5_bi < best_c5:
            best_c5 = c5_bi
            best_h = h_bi
        if elapsed() < budget_s * 0.99:
            tl = time_at(0.99)
            h_new, c5_new = lbfgs_optimize(best_h, n_final, 50000.0, 2000, 60000, tl)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = project_feasible(h_new, n_final)

    final_h = project_feasible(best_h, n_final)
    final_c5 = evaluate_c5(final_h)
    return final_h, final_c5, n_final
