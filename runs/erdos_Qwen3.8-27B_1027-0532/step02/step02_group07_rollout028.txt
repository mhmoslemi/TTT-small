    val = m + np.log(np.sum(exp_x)) / alpha
    return val, w


def compute_smooth_gradient(h, alpha):
    n = len(h)
    corr = compute_correlations(h)
    val, w = smooth_max_with_weights(corr, alpha)
    
    grad = np.zeros(n)
    # grad[j] = sum over p of w[p] * slopes_j[p]
    # slopes_j[p] = (1-h[idx1]) * valid1 + (-h[idx2]) * valid2
    # where idx1 = j + p - (n-1), idx2 = j - p + (n-1)
    
    for j in range(n):
        # First part: sum over p where idx1 is valid
        # idx1 = j + p - n + 1, valid when 0 <= j+p-n+1 < n
        # => n-1-j <= p <= 2n-2-j
        p_lo1 = max(0, n - 1 - j)
        p_hi1 = min(2 * n - 2, 2 * n - 2 - j)
        if p_lo1 <= p_hi1:
            idx1 = j + np.arange(p_lo1, p_hi1 + 1) - (n - 1)
            grad[j] += np.sum(w[p_lo1:p_hi1 + 1] * (1.0 - h[idx1]))
        
        # Second part: sum over p where idx2 is valid
        # idx2 = j - p + n - 1, valid when 0 <= j-p+n-1 < n
        # => j-(n-1) <= p <= j+n-1
        p_lo2 = max(0, j - n + 1)
        p_hi2 = min(2 * n - 2, j + n - 1)
        if p_lo2 <= p_hi2:
            idx2 = j - np.arange(p_lo2, p_hi2 + 1) + (n - 1)
            grad[j] -= np.sum(w[p_lo2:p_hi2 + 1] * h[idx2])
    
    return grad, val


def frank_wolfe_step(n, target_sum, grad):
    """Solve min grad·h s.t. 0<=h<=1, sum(h)=target_sum"""
    sorted_idx = np.argsort(grad)
    hw = np.zeros(n)
    k = int(target_sum)
    hw[sorted_idx[:k]] = 1.0
    frac = target_sum - k
    if frac > 1e-12:
        hw[sorted_idx[k]] = frac
    return hw


def subgradient_fw(h, alpha, step_epochs, time_budget):
    """Frank-Wolfe with subgradient of smooth max"""
    n = len(h)
    target = n / 2.0
    t0 = time.time()
    h = h.copy()
    
    for epoch in range(step_epochs):
        if time.time() - t0 > time_budget:
            break
        
        grad, val = compute_smooth_gradient(h, alpha)
        hw = frank_wolfe_step(n, target, grad)
        
        # Step size: gamma = 2/(epoch+2) (standard FW step)
        gamma = 2.0 / (epoch + 2)
        h_new = (1.0 - gamma) * h + gamma * hw
        h = project_sum(h_new, target)
    
    return h


def coordinate_descent(h, n_sweeps=200, tol=1e-10, time_budget=60.0):
    n = len(h)
    h = h.copy()
    corr = compute_correlations(h)
    t0 = time.time()

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        max_change = 0.0
        for j in range(n):
            t_opt = optimize_coordinate_fast(h, j, corr)
            change = t_opt - h[j]
            if abs(change) > 1e-13:
                h[j] = t_opt
                # Update corr
                p = np.arange(2 * n - 1)
                idx1 = j + p - (n - 1)
                valid1 = (idx1 >= 0) & (idx1 < n)
                idx2 = j - p + (n - 1)
                valid2 = (idx2 >= 0) & (idx2 < n)
                slopes = np.zeros(2 * n - 1)
                if np.any(valid1):
                    slopes[valid1] += (1.0 - h[idx1[valid1]])
                if np.any(valid2):
                    slopes[valid2] -= h[idx2[valid2]]
                corr = corr + change * slopes
                max_change = max(max_change, abs(change))
        if sweep % 20 == 19:
            corr = compute_correlations(h)
        if max_change < tol:
            break

    h = project_sum(h, n / 2.0)
    return h


def optimize_coordinate_fast(h, j, corr):
    n = len(h)
    h_j = h[j]
    m = 2 * n - 1
    p = np.arange(m)

    idx1 = j + p - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j - p + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)

    slopes = np.zeros(m)
    if np.any(valid1):
        slopes[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        slopes[valid2] -= h[idx2[valid2]]

    a = corr - h_j * slopes

    zero_mask = (np.abs(slopes) < 1e-14)
    pos_mask = slopes > 1e-14
    neg_mask = slopes < -1e-14

    lo_V = float(np.max(np.minimum(a, a + slopes)))
    hi_V = float(np.max(corr))

    if lo_V >= hi_V - 1e-14:
        return h_j

    for _ in range(35):
        mid_V = (lo_V + hi_V) * 0.5
        feasible = True
        if np.any(zero_mask):
            if float(np.max(a[zero_mask])) > mid_V + 1e-14:
                feasible = False
        if feasible:
            lo_t = 0.0
            if np.any(neg_mask):
                lo_t = float(np.max((mid_V - a[neg_mask]) / slopes[neg_mask]))
            hi_t = 1.0
            if np.any(pos_mask):
                hi_t = float(np.min((mid_V - a[pos_mask]) / slopes[pos_mask]))
            if lo_t > hi_t + 1e-14:
                feasible = False

        if feasible:
            hi_V = mid_V
        else:
            lo_V = mid_V

    V_star = (lo_V + hi_V) * 0.5

    lo_t = 0.0
    hi_t = 1.0
    if np.any(neg_mask):
        lo_t = float(np.max((V_star - a[neg_mask]) / slopes[neg_mask]))
    if np.any(pos_mask):
        hi_t = float(np.min((V_star - a[pos_mask]) / slopes[pos_mask]))

    lo_t = max(lo_t, 0.0)
    hi_t = min(hi_t, 1.0)
    if lo_t > hi_t:
        t_star = h_j
    else:
        t_star = (lo_t + hi_t) * 0.5

    return float(np.clip(t_star, 0.0, 1.0))

            idx2 = j - np.arange(p_lo2, p_hi2 + 1) + (n - 1)
            grad[j] -= np.dot(w[p_lo2:p_hi2 + 1], h[idx2])

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    h = h.copy()
    n = len(h)
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s

        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)

    m = np.max(x)

    m = np.max(x)

        h_cd = coordinate_descent(h_pert, n_sweeps=300, time_budget=tb)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd
            n_cur = len(best_h)

    # Phase 7: Rounding strategies
    if time_left() > 50:
        n_cur = len(best_h)
        target = n_cur / 2.0
        # Strategy A: threshold at 0.5
        h_round = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_round))
        if n_ones > int(np.floor(target)) + 1:
            order = np.argsort(best_h)
            for idx in order[:n_ones - int(np.floor(target))]:
                h_round[idx] = 0.0
        elif n_ones < int(np.floor(target)):
            order = np.argsort(-best_h)
            for idx in order[:int(np.floor(target)) - n_ones]:
                h_round[idx] = 1.0
        h_round = project_sum(h_round, target)
        h_cd = coordinate_descent(h_round, n_sweeps=500, time_budget=min(25, time_left() - 25))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    if time_left() > 50:
        # Strategy B: histogram-based rounding
        n_cur = len(best_h)
        target = n_cur / 2.0
        h_round2 = np.zeros(n_cur)
        threshold = np.sort(best_h)[int(n_cur * 0.5)]
        h_round2[best_h > threshold] = 1.0
        n_ones2 = int(np.sum(h_round2))
        if n_ones2 != int(np.floor(target)):
            diff = int(np.floor(target)) - n_ones2
            if diff > 0:
                zeros_idx = np.where(h_round2 == 0)[0]
                for idx in zeros_idx[:diff]:
                    h_round2[idx] = 1.0
            elif diff < 0:
                ones_idx = np.where(h_round2 == 1)[0]
                for idx in ones_idx[:(-diff)]:
                    h_round2[idx] = 0.0
        h_round2 = project_sum(h_round2, target)
        h_cd = coordinate_descent(h_round2, n_sweeps=500, time_budget=min(25, time_left() - 25))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Phase 8: Multi-resolution
    base_n = len(best_h)
    res_list = []
    for mult in [1.5, 2, 3]:
        r = int(base_n * mult)
        if 80 <= r <= 4000:
            res_list.append(r)
    # Sort by distance from base_n
    res_sorted = sorted(res_list, key=lambda x: abs(x - base_n))

    for n_try in res_sorted:
        if time_left() < 70:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)

        tb_cd = min(35, time_left() - 50)
        if tb_cd < 10:
            break
        h_cd = coordinate_descent(h_interp, n_sweeps=500, time_budget=tb_cd)
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

        if time_left() > 30:
            bounds_try = [(0.0, 1.0)] * n_try
            for alpha2 in [500, 5000]:
                if time_left() < 20:
                    break
                res_x = slsqp_optimize(h_cd, alpha2, max(50, 200000 // n_try), bounds_try)
                if res_x is not None:
                    h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                    if np.all(np.isfinite(h_new)):
                        s2 = evaluate_score(h_new)
                        if s2 < best_score:
                            best_score = s2
                            best_h = h_new
                            h_cd = h_new

        if time_left() > 20:
            h_cd2 = coordinate_descent(h_cd, n_sweeps=300, time_budget=min(15, time_left() - 10))
            s = evaluate_score(h_cd2)
            if s < best_score:
                best_score = s
                best_h = h_cd2

    # Phase 9: Second round of FW from best
    if time_left() > 60:
        for alpha_fw2 in [200, 2000]:
            if time_left() < 40:
                break
            h_fw2 = frank_wolfe(best_h, alpha_fw2, n_iters=100, time_budget=min(20, time_left() - 25))
            h_cd2 = coordinate_descent(h_fw2, n_sweeps=500, time_budget=min(15, time_left() - 15))
            s = evaluate_score(h_cd2)
            if s < best_score:
                best_score = s
                best_h = h_cd2

    # Phase 10: Final CD (long run)
    if time_left() > 25:
        h_cd = coordinate_descent(best_h, n_sweeps=5000, time_budget=max(10, time_left() - 5))
        s = evaluate_score(h_cd)
        if s < best_score:
            best_score = s
            best_h = h_cd

    # Final cleanup
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
