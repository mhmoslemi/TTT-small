def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(n, dtype=bool)
        h[free] += diff / free.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            i_hi = n - offset
            if i_hi > 0:
                grad[:i_hi] += w * (1.0 - h[offset:offset + i_hi])
                grad[offset:] -= w * h[:i_hi]
        else:
            i_hi = n + offset
            if i_hi > 0:
                grad[-offset:] += w * (1.0 - h[:i_hi])
                grad[:i_hi] -= w * h[-offset:]
    grad *= dx
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-13:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 250:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        s = step
        for bt in range(35):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-14:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            break
        step = min(s * 1.4, 1.0)
    return h


def fista_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    step = min(0.3, 1.0 / (alpha * dx * 5.0 + 1.0))
    prev_sm = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ov_y = compute_overlaps(y)
        sm_y, w_y = smooth_max_and_weights(ov_y, alpha)
        grad_y = compute_gradient(y, w_y, dx)
        x_new = project_onto_constraint(y - step * grad_y, target)
        ov_x = compute_overlaps(x_new)
        sm_x, _ = smooth_max_and_weights(ov_x, alpha)
        if prev_sm is not None and sm_x > prev_sm + 1e-9:
            t_k = 1.0
        prev_sm = sm_x
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_onto_constraint(y, target)
        x = x_new
        t_k = t_new
    return x


def guided_binary_sa(h_init, n_iters, deadline, rng):
    n = len(h_init)
    target = n / 2.0
    h = (h_init > 0.5).astype(float)
    diff = target - h.sum()
    if diff > 0:
        zeros = np.where(h == 0)[0]
        idx = rng.permutation(len(zeros))
        for k in range(int(np.ceil(diff))):
            h[zeros[idx[k]]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        idx = rng.permutation(len(ones))
        for k in range(int(np.ceil(-diff))):
            h[ones[idx[k]]] = 0.0

    best_h = h.copy()
    best_score = evaluate_score(h)
    current_score = best_score
    best_iter = 0

    T = 0.004
    cooling = 0.9998
    tabu = {}
    tabu_len = 60

    for it in range(n_iters):
        if time.time() > deadline:
            break

        if it % 200 == 0:
            overlaps = compute_overlaps(h)
            worst_k = np.argmax(overlaps)
        else:
            worst_k = np.argmax(overlaps) if it < 200 else worst_k

        if it % 100 == 0:
            overlaps = compute_overlaps(h)
            worst_k = np.argmax(overlaps)

        lag = worst_k - (n - 1)
        move_made = False

        if rng.random() < 0.65 and lag != 0:
            if lag > 0:
                bad = np.where((h[:n - lag] == 1) & (h[lag:] == 0))[0]
                if len(bad) > 0:
                    i = bad[rng.randint(len(bad))]
                    j = i + lag
                    mk = (i, j)
                    if tabu.get(mk, -10) > it - tabu_len:
                        move_made = False
                    else:
                        h[i] = 0.0
                        h[j] = 1.0
                        ns = evaluate_score(h)
                        d = ns - current_score
                        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                            current_score = ns
                            tabu[mk] = it
                            move_made = True
                            if ns < best_score:
                                best_score = ns
                                best_h = h.copy()
                                best_iter = it
                        else:
                            h[i] = 1.0
                            h[j] = 0.0
            else:
                al = -lag
                bad = np.where((h[al:] == 1) & (h[:-al] == 0))[0] + al
                if len(bad) > 0:
                    i = bad[rng.randint(len(bad))]
                    j = i - al
                    mk = (i, j)
                    if tabu.get(mk, -10) > it - tabu_len:
                        move_made = False
                    else:
                        h[i] = 0.0
                        h[j] = 1.0
                        ns = evaluate_score(h)
                        d = ns - current_score
                        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                            current_score = ns
                            tabu[mk] = it
                            move_made = True
                            if ns < best_score:
                                best_score = ns
                                best_h = h.copy()
                                best_iter = it
                        else:
                            h[i] = 1.0
                            h[j] = 0.0

        if not move_made:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            mk = (i, j)
            if tabu.get(mk, -10) > it - tabu_len:
                T *= cooling
                continue
            h[i] = 0.0
            h[j] = 1.0
            ns = evaluate_score(h)
            d = ns - current_score
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                current_score = ns
                tabu[mk] = it
                if ns < best_score:
                    best_score = ns
                    best_h = h.copy()
                    best_iter = it
            else:
                h[i] = 1.0
                h[j] = 0.0

        T *= cooling

        if it - best_iter > 4000 and it > 500:
            h = best_h.copy()
            current_score = best_score
            T = 0.004

    return best_h, best_score


def fractional_refine(h_binary, n_max_frac, deadline, rng):
    n = len(h_binary)
    target = n / 2.0
    h = h_binary.copy()
    best_score = evaluate_score(h)
    best_h = h.copy()

    for _ in range(n_max_frac):
        if time.time() > deadline:
            break
        improvements = []
        n_candidates = min(50, n)
        candidates = rng.choice(n, n_candidates, replace=False)
        for i in candidates:
            for val in [0.3, 0.5, 0.7]:
                if abs(h[i] - val) < 0.01:
                    continue
                h_try = h.copy()
                h_try[i] = val
                diff = target - h_try.sum()
                if abs(diff) > 1e-10:
                    others = np.arange(n) != i
                    if diff > 0:
                        free = np.where(others & (h_try < 1.0 - 0.01))[0]
                    else:
                        free = np.where(others & (h_try > 0.01))[0]
                    if len(free) == 0:
                        continue
                    per = diff / len(free)
                    h_try[free] += per
                    if np.any(h_try[free] < 0) or np.any(h_try[free] > 1):
                        continue
                    np.clip(h_try, 0.0, 1.0, out=h_try)
                    h_try = project_onto_constraint(h_try, target)
                s = evaluate_score(h_try)
                if s < best_score - 1e-6:
                    improvements.append((s, i, val))

        if improvements:
            improvements.sort()
            best_s, best_i, best_val = improvements[0]
            h[best_i] = best_val
            h = project_onto_constraint(h, target)
            best_score = evaluate_score(h)
            best_h = h.copy()

    return best_h, best_score


def diverse_initialization(rng, n, method):
    if method == 0:
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
        if n % 2 == 1:
            h[idx[n // 2]] = 0.5
    elif method == 1:
        period = max(3, n // 8)
        h = np.zeros(n)
        for i in range(0, n, period):
            start = min(i, n)
            end = min(i + period // 2, n)
            h[start:end] = 1.0
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-6:
            if diff > 0:
                zeros = np.where(h == 0)[0]
                h[rng.choice(zeros, int(np.ceil(diff)), replace=False)] = 1.0
            else:
                ones = np.where(h == 1)[0]
                h[rng.choice(ones, int(np.ceil(-diff)), replace=False)] = 0.0
    elif method == 2:
        h = np.zeros(n)
        block = max(2, n // 16)
        for b in range(0, n, block * 2):
            h[b:min(b + block, n)] = 1.0
        diff = n / 2.0 - h.sum()
        if abs(diff) > 1e-6:
            if diff > 0:
                zeros = np.where(h == 0)[0]
                h[rng.choice(zeros, min(int(np.ceil(diff)), len(zeros)), replace=False)] = 1.0
            else:
                ones = np.where(h == 1)[0]
                h[rng.choice(ones, min(int(np.ceil(-diff)), len(ones)), replace=False)] = 0.0
    elif method == 3:
        freq = 2.0 * np.pi * rng.randint(3, 12) / n
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.4 * np.sin(freq * x + rng.uniform(0, 2 * np.pi))
        h = np.clip(h, 0, 1)
        h = project_onto_constraint(h, n / 2.0)
    elif method == 4:
        h = rng.rand(n)
        h = project_onto_constraint(h, n / 2.0)
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h
