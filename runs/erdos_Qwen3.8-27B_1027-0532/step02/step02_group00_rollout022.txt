import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_active = np.sum(mask)
        if n_active == 0:
            break
        step = diff / n_active
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def corr_gradient_fft(h, weights):
    """Compute gradient of sum_k weights[k] * corr[k] w.r.t. h using FFT."""
    n = len(h)
    # corr = np.correlate(h, 1-h, 'full')
    # d/dh[j] sum_k weights[k] * corr[k] = t1[j] - t2[j]
    # where t1[j] = sum_i (1-h[i]) * W(j-i) and t2[j] = sum_i h[i] * W(i-j)
    # W(m) = weights[m + (n-1)]

    t2 = fftconvolve(weights[::-1], h, mode='full')[n-1:2*n-1]
    return t1 - t2

def logsumexp_objective_and_grad(h, alpha):
    """Compute log-sum-exp(alpha * corr) and its gradient."""
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    exp_shifted = np.exp(np.clip(shifted, -500, 0))
    log_sum = np.log(np.sum(exp_shifted))
    objective = (max_c + log_sum / alpha)
    # softmax weights
    weights = exp_shifted / np.sum(exp_shifted)
    grad = corr_gradient_fft(h, weights)
    return objective, grad

def pnorm_objective_and_grad(h, p):
    """Compute L_p norm of corr and its gradient."""
    corr = compute_corr(h)
    abs_corr = np.abs(corr)
    max_abs = np.max(abs_corr)
    if max_abs < 1e-15:
        return 0.0, np.zeros_like(h)
    # Use normalized version for numerical stability
    log_p_norm = np.log(max_abs) + np.log(np.sum((abs_corr / max_abs) ** p)) / p
    objective = np.exp(log_p_norm)
    # Gradient
    # d/dh [(sum |c_k|^p)^(1/p)] = (sum |c_k|^p)^(1/p-1) * sum_k |c_k|^(p-1) * sign(c_k) * dc_k/dh
    # weights[k] = |c_k|^(p-1) * sign(c_k) / (sum |c_m|^p)^(1-1/p)
    powered = (abs_corr / max_abs) ** (p - 1)
    weights = powered * np.sign(corr)
    norm_factor = np.sum(powered / max_abs)  # approximate
    # Actually: (sum |c|^p)^(1/p - 1) = (max_abs^p * sum (|c|/max_abs)^p)^(1/p-1)
    # = max_abs * (sum (|c|/max_abs)^p)^(1/p-1)
    s = np.sum((abs_corr / max_abs) ** p)
    norm = max_abs * s ** (1.0/p - 1.0)
    weights = norm * (abs_corr / max_abs) ** (p-1) * np.sign(corr)
    grad = corr_gradient_fft(h, weights)
    return objective, grad

def nesterov_pgd(h0, target, alpha, max_iters, lr_init, time_limit):
    """Nesterov accelerated projected gradient descent."""
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_h = h0.copy()
    best_obj = compute_c5(h0)
    lr = lr_init
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t_new - 1) / t_new
        y = h + beta * (h - h_prev)
        y = project_feasible(y, target)
        
        # Compute gradient at y
        obj, grad = logsumexp_objective_and_grad(y, alpha)
        
        # Gradient step
        y_new = y - lr * grad
        y_new = project_feasible(y_new, target)
        
        # Backtracking line search
        obj_new = compute_c5(y_new)
        # Use the actual C5 for line search (not the smoothed objective)
        # Actually, for the smoothed objective, we should check the smoothed value
        obj_check = logsumexp_objective_and_grad(y_new, alpha)[0]
        
        step_count = 0
        while obj_check > obj + 1e-4 * lr * np.sum(grad * (y_new - y)) and step_count < 20:
            lr *= 0.5
            y_new = y - lr * grad
            y_new = project_feasible(y_new, target)
            obj_check = logsumexp_objective_and_grad(y_new, alpha)[0]
            step_count += 1
        
        h_prev = h
        h = y_new
        t = t_new
        
        if compute_c5(h) < best_obj:
            best_obj = compute_c5(h)
            best_h = h.copy()
    
    return best_h, best_obj

from scipy.signal import fftconvolve

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_act = np.sum(mask)
        if n_act == 0:
            break
        h[mask] = np.clip(h[mask] + diff / n_act, 0.0, 1.0)
    return h

def corr_grad_fft(h, weights):
    n = len(h)
    t1 = fftconvolve(1.0 - h, weights, mode='full')[n-1:2*n-1]
    t2 = fftconvolve(h, weights[::-1], mode='full')[n-1:2*n-1]
    return t1 - t2

def lse_obj_grad(h, alpha):
    corr = compute_corr(h)
    mc = np.max(corr)
    s = alpha * (corr - mc)
    es = np.exp(np.clip(s, -700, 0))
    es_sum = np.sum(es)
    obj = mc + np.log(es_sum) / alpha
    w = es / es_sum
    grad = corr_grad_fft(h, w)
    return obj, grad

def nesterov_pgd(h0, target, alpha, max_it=5000, lr0=0.01, tlimit=60.0):
    h = h0.copy()
    hp = h0.copy()
    t = 1.0
    lr = lr0
    bh = h0.copy()
    bs = compute_c5(h0)
    t0 = time.time()
    for it in range(max_it):
        if time.time() - t0 > tlimit:
            break
        tn = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        beta = (tn - 1.0) / tn
        y = h + beta * (h - hp)
        y = project_feasible(y, target)
        obj, grad = lse_obj_grad(y, alpha)
        # Backtracking line search

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_act = np.sum(mask)
        if n_act == 0:
            break
        h[mask] = np.clip(h[mask] + diff / n_act, 0.0, 1.0)
    return h


def corr_grad_fft(h, weights):
    n = len(h)
    t1 = fftconvolve(1.0 - h, weights, mode='full')[n - 1:2 * n - 1]
    t2 = fftconvolve(h, weights[::-1], mode='full')[n - 1:2 * n - 1]
    return t1 - t2


def lse_obj_grad(h, alpha):
    corr = compute_corr(h)
    mc = np.max(corr)
    s = alpha * (corr - mc)
    es = np.exp(np.clip(s, -700.0, 0.0))
    es_sum = np.sum(es)
    obj = mc + np.log(es_sum) / alpha
    w = es / es_sum
    grad = corr_grad_fft(h, w)
    return obj, grad
