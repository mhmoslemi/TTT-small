    log_Z = ms + np.log(np.sum(np.exp(shifted - ms)))
    return (m + log_Z / alpha) * dx

def _softmax_grad(x, alpha):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    ms = np.max(shifted)
    exp_s = np.exp(shifted - ms)
    Z = np.sum(exp_s)
    w = exp_s / Z
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag > 0:
            grad[:n - lag] -= wk * x[lag:]
            grad[lag:] += wk * (1.0 - x[:n - lag])
        elif lag < 0:
            L = -lag
            grad[:n - L] += wk * (1.0 - x[L:])
            grad[L:] -= wk * x[:n - L]
        else:
            grad += wk * (1.0 - 2.0 * x)
    grad *= dx
    return grad
