                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr(h, k_idx):
    """Gradient of corr[k_idx] * dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    k_prime = abs(k_idx - (n - 1))
    grad = np.zeros(n)
    # For both positive and negative lag with magnitude k':
    # grad[t] = (1-h[t+k'])*I[t < n-k'] - h[|t-k'| adjust]*I[...]
    # Unified formula:
    # The corr is sum over valid pairs (i, i+k') of h[i]*(1-h[i+k'])
    # or equivalently h[i+k']*(1-h[i])
    # Gradient w.r.t. h[t]:
    #   If t appears as the "first" element: contributes (1-h[t+k']) for t in [0, n-1-k']
    #     or (1-h[t-k']) for t in [k', n-1] (if we index it the other way)
    #   If t appears as the "second" element: contributes -h[t-k'] for t in [k', n-1]
    #     or -h[t+k'] for t in [0, n-1-k']
    
    # For lag k': corr = sum_{i=0}^{n-1-k'} h[i]*h_complement[i+k']
    # where h_complement = 1-h
    # = sum_{i=0}^{n-1-k'} h[i]*(1-h[i+k'])
    
    # d/dh[t]:
    # First term: h[t] appears when i=t, t in [0, n-1-k']
    #   contribution: (1-h[t+k'])
    # Second term: h[t] appears as h[i+k'] when i+k'=t, i=t-k', t-k' in [0, n-1-k']
    #   i.e., t in [k', n-1]
    #   contribution: -h[t-k']
    
    m = n - k_prime
    if m > 0:
        grad[:m] += (1.0 - h[k_prime:k_prime+m])
        grad[k_prime:] -= h[:m]
    return grad * dx

def _smoothed_max(corr_vals, alpha):
    m = np.max(corr_vals)
    if m * alpha < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr_vals - m)))) / alpha

def _grad_smoothed_max(h, alpha, n):
    """Gradient of smoothed max of corr * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    max_corr = np.max(corr)
    weights = np.exp(alpha * (corr - max_corr))
    total = weights.sum()
    if total < 1e-300:
        k_star = int(np.argmax(corr))
        return _grad_corr(h, k_star)
    weights /= total
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] > 1e-8:
            grad += weights[idx] * _grad_corr(h, idx)
    return grad

def _nesterov_adaptive(h0, target_sum, n_iters, alpha0, alpha_decay, use_smooth=False, smooth_alpha=50, top_m=3):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t_nest = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):

                g = _grad_smoothed_max(y, smooth_alpha, n)
            else:

                top_indices = np.argsort(corr)[::-1][:top_m]
                max_corr = corr[top_indices[0]]
                weights = np.exp(30.0 * (corr[top_indices] - max_corr))
                weights /= weights.sum()
                g = np.zeros(n)
                for wi, ki in zip(weights, top_indices):
                    g += wi * _grad_corr(y, int(ki))
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr(y, k_star)
        
        alpha = alpha0 / (it + 1) ** alpha_decay
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y_new = x_new + ((t_nest - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t_nest = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
