import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    s = np.sum(h)
    target_exact = target
    if abs(s - target_exact) > 1e-10:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += (target_exact - s) / np.sum(free)
            np.clip(h, 0.0, 1.0, out=h)
    return h


def topk_smooth_max_obj_grad(h, alpha, k):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m_total = len(corrs)
    k_eff = min(k, m_total)
    top_idx = np.argpartition(-corrs, k_eff - 1)[:k_eff]
    top_vals = corrs[top_idx]
    m_top = np.max(top_vals)
    shifted = alpha * (top_vals - m_top)
    shifted = np.clip(shifted, -60, 60)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        weights_top = np.ones(k_eff) / k_eff
    else:
        weights_top = exp_shifted / sum_exp
    obj = m_top + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for j in range(k_eff):
        w = alpha * weights_top[j]
        if w < 1e-14:
            continue
        lag = top_idx[j] - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += w
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + lag:hi2 + lag]
        lo3 = max(0, lag)
        hi3 = min(n, n + lag)
        if lo3 < hi3:
            grad[lo3:hi3] -= w * h[lo3 - lag:hi3 - lag]
    return obj, grad


def penalty_obj_grad(h, alpha, mu, k):
    obj, grad = topk_smooth_max_obj_grad(h, alpha, k)
    viol = np.sum(h) - len(h) / 2.0
    obj += 0.5 * mu * viol * viol
    grad = grad + mu * viol
    return obj, grad


def slsqp_obj(h, alpha, k):
    return topk_smooth_max_obj_grad(h, alpha, k)[0]


def slsqp_jac(h, alpha, k):
    return topk_smooth_max_obj_grad(h, alpha, k)[1]


def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0


def constraint_eq_jac(h):
    return np.ones(len(h))


def lbfgsb_optimize(h0, alpha, mu, k, maxiter, bounds):
    res = minimize(
        penalty_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu, k),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'iprint': 0,
                 'maxls': 40, 'maxfun': maxiter * 30}
    )
    return res.x


def slsqp_optimize(h0, alpha, k, maxiter, bounds):
    res = minimize(
        slsqp_obj, h0, method='SLSQP', bounds=bounds,
        args=(alpha, k), jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x


def block_coordinate_descent(h, alpha, k, n_rounds, block_size, mu):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for _ in range(n_rounds):
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            bsize = end - start
            if bsize < 2:
                continue
            fixed_sum = np.sum(h) - np.sum(h[start:end])
            block_target = target - fixed_sum
            bounds_blk = [(0.0, 1.0)] * bsize

            def blk_obj(x, start=start, end=end):
                h_temp = h.copy()
                h_temp[start:end] = x
                obj, _ = topk_smooth_max_obj_grad(h_temp, alpha, k)
                viol = np.sum(x) - block_target
                obj += 0.5 * mu * viol * viol
                return obj

            def blk_jac(x, start=start, end=end):
                h_temp = h.copy()
                h_temp[start:end] = x
                _, full_grad = topk_smooth_max_obj_grad(h_temp, alpha, k)
                viol = np.sum(x) - block_target
                grad_blk = full_grad[start:end] + mu * viol
                return grad_blk

            if bsize <= 20:
                res = minimize(
                    blk_obj, h[start:end], method='L-BFGS-B',
                    bounds=bounds_blk, jac=blk_jac,
                    options={'maxiter': 100, 'ftol': 1e-18, 'iprint': 0}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h[start:end] = np.clip(res.x, 0.0, 1.0)
    return project_sum(h, target)


def subgradient_step(h, step_size, n_substeps, alpha, k):

    target = n / 2.0
    for _ in range(n_substeps):
        _, grad = topk_smooth_max_obj_grad(h, alpha, k)
        h = h - step_size * grad
        h = project_sum(h, target)
    return h

    target = n / 2.0

    elif mode == 3:

    elif mode == 4:
