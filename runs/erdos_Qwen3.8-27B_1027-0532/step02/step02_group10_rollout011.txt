def solve_1d_minmax(a, b, lo=0.0, hi=1.0):
    """Solve min_x max_k (a[k]*x + b[k]) s.t. lo <= x <= hi
    
    The optimal x is where two of the lines achieve the maximum simultaneously,
    or at a boundary.
    """
    n = len(a)
    # Evaluate at boundaries
    f_lo = np.max(b)
    f_hi = np.max(a + b)
    
    # Find intersection points
    best_val = min(f_lo, f_hi)
    best_x = lo if f_lo < f_hi else hi
    
    # Check all pairs
    for i in range(n):
        for j in range(i+1, n):
            if a[i] == a[j]:
                continue
            x_cross = (b[j] - b[i]) / (a[i] - a[j])
            if lo <= x_cross <= hi:
                # Check if this is actually the max
                vals = a * x_cross + b
                v = np.max(vals)
                if v < best_val - 1e-15:
                    best_val = v
                    best_x = x_cross
    return best_x, best_val

def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        weights = np.zeros_like(x)
        weights[np.argmax(x)] = 1.0
        return m, weights
    value = m + np.log(total) / alpha
    weights = exp_shifted / total
    return value, weights


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-13:
            continue
        offset = m - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w * (1.0 - h[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= w * h[i_lo2 - offset:i_hi2 - offset]
    grad *= dx
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    step = min(0.5, 1.0 / (alpha * dx * 3.0 + 1.0))
    prev_sm = None
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-14:
                stall_count = 0
            else:
                stall_count += 1
            if stall_count > 300:
                break
        prev_sm = sm
        grad = compute_gradient(h, weights, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        improved = False
        s = step
        for bt in range(40):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-14:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step = s
            break
        step = min(s * 1.5, 1.0)
    return h


def active_set_pgd(h, target, top_k_frac, step_size, max_iters, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    prev_score = evaluate_score(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        current_max = np.max(overlaps)
        n_top = max(5, int(top_k_frac * len(overlaps)))
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        vals = overlaps[top_idx]
        spread = current_max - vals[-1]
        if spread < 1e-12:
            active_weights = np.ones(n_top) / n_top
        else:
            active_weights = np.exp(-(current_max - vals) / (spread * 0.3 + 1e-12))
            active_weights /= active_weights.sum()
        full_weights = np.zeros(len(overlaps))
        full_weights[top_idx] = active_weights
        grad = compute_gradient(h, full_weights, dx)
        improved = False
        s = step_size
        for bt in range(30):
            h_trial = project_onto_constraint(h - s * grad, target)
            score_t = evaluate_score(h_trial)
            if score_t < prev_score - 1e-14:
                h = h_trial
                prev_score = score_t
                improved = True
                stall = 0
                break
            s *= 0.5
            if s < 1e-12:
                break
        if not improved:
            step_size = s
            stall += 1
            if stall > 80:
                break
        else:
            step_size = min(step_size * 1.3, 0.3)
    return h

    dx = 2.0 / n
    rng = np.random.RandomState(seed)

            zeros = np.where(h == 0.0)[0]

def simulated_annealing_binary(initial_h, target, deadline, seed):

def structured_perturbation(h, overlaps, rng, scale):
    n = len(h)
    pert = h.copy()
    top_idx = np.argsort(overlaps)[::-1][:10]
    for k_idx in top_idx:
        lag = k_idx - (n - 1)
        mask = (h > 0.01) & (h < 0.99)
        if mask.sum() == 0:
            break
        affected = np.zeros(n)
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo < i_hi:
            affected[i_lo:i_hi] = h[i_lo + lag:i_hi + lag]
        neg_affected = np.zeros(n)
        i_lo2 = max(0, lag)
        i_hi2 = min(n, n + lag)
        if i_lo2 < i_hi2:
            neg_affected[i_lo2:i_hi2] = h[i_lo2 - lag:i_hi2 - lag]
        strength = rng.uniform(0.3, 1.0)
        pert[i_lo:i_hi] -= strength * scale * affected[i_lo:i_hi] * mask[i_lo:i_hi]
        pert[i_lo2:i_hi2] += strength * scale * neg_affected[i_lo2:i_hi2] * mask[i_lo2:i_hi2]
    noise = rng.randn(n) * scale * 0.3
    pert += noise
    return np.clip(pert, 0.0, 1.0)


def upsample_h(h, new_n):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
