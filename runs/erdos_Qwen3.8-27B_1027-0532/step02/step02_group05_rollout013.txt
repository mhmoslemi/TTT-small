def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def smooth_max_val(x, alpha):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def smooth_max_grad(x, alpha):
    m = np.max(x)
    ex = np.exp(alpha * (x - m))
    return ex / np.sum(ex)

    for alpha in [500, 2000, 8000]:
        if elapsed() > budget_s - 350:
            break
        lr_init = 1.0 / max(1.0, alpha * n * 0.1)
        h_new, s_new = fista_backtrack(best_h, alpha, target, 2000, lr_init)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

    # Phase 4: Multi-resolution refinement
    for n_try in [n * 3, n * 5]:
        if n_try > 4000 or n_try < 3:
            continue
        if elapsed() > budget_s - 200:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new_arr = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init_hi = np.interp(x_new_arr, x_old, best_h)
        target_hi = n_try / 2.0
        h_init_hi = project_feasible(h_init_hi, target_hi)

        if n_try % 2 == 1:
            half_hi = n_try // 2
            g0_hi = h_init_hi[:half_hi].copy()
            g0_hi = np.clip(g0_hi, 0.0, 1.0)
            for alpha in [200, 800, 3000]:
                if elapsed() > budget_s - 150:
                    break
                maxiter = max(50, min(250, 150000 // n_try))
                g_new = slsqp_ap(g0_hi, alpha, maxiter)
                if g_new is not None and np.all(np.isfinite(g_new)):
                    h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
                    s = evaluate_score(h_new)
                    if s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        g0_hi = h_new[:half_hi].copy()

        for alpha in [200, 800, 3000]:
            if elapsed() > budget_s - 150:
                break
            maxiter = max(50, min(250, 150000 // n_try))
            res_x = slsqp_full(h_init_hi if n_try % 2 == 0 else best_h, alpha, maxiter)
            if res_x is not None and np.all(np.isfinite(res_x)):
                n_res = len(res_x)
                h_new = project_feasible(np.clip(res_x, 0.0, 1.0), n_res / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_init_hi = res_x.copy()

    # Phase 5: Diverse random restarts
    n_base = len(best_h)
    target_base = n_base / 2.0
    n_restarts = 25
    for r in range(n_restarts):
        if elapsed() > budget_s - 100:
            break
        frac = r / float(n_restarts)
        if r % 5 == 0:
            pert = best_h + rng.randn(n_base) * (0.02 + 0.08 * frac)
        elif r % 5 == 1:
            pert = best_h + rng.randn(n_base) * (0.05 + 0.15 * frac)
        elif r % 5 == 2:
            mask = rng.rand(n_base) < 0.3
            pert = best_h.copy()
            pert[mask] = rng.rand(int(np.sum(mask)))
        elif r % 5 == 3:
            if n_base % 2 == 1:
                half_r = n_base // 2
                g_r = np.clip(best_h[:half_r] + rng.randn(half_r) * (0.03 + 0.1 * frac), 0, 1)
                pert = antipal_h(g_r)
            else:
                pert = best_h + rng.randn(n_base) * 0.06
        else:
            pert = (best_h + rng.randn(n_base) * 0.04 > 0.5).astype(float)

        pert = project_feasible(pert, target_base)
        alpha_r = [300, 800, 2000, 5000, 12000][r % 5]
        maxiter = max(60, min(250, 120000 // n_base))
        res_x = slsqp_full(pert, alpha_r, maxiter)
        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_base)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 6: Binary local search
    if elapsed() < budget_s - 60:
        h_bin, s_bin = binary_swap_search(best_h, rng, 50)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
        # Also try from a slightly different binary
        h_bin2, s_bin2 = binary_swap_search(best_h + rng.randn(len(best_h)) * 0.05, rng, 50)
        if s_bin2 < best_score:
            best_score = s_bin2
            best_h = h_bin2.copy()

    # Phase 7: Final aggressive polishing
    n_final = len(best_h)
    target_final = n_final / 2.0
    for alpha in [10000, 30000, 80000, 200000]:
        if elapsed() > budget_s - 10:
            break
        maxiter = max(200, min(600, 500000 // n_final))
        res_x = slsqp_full(best_h, alpha, maxiter)
        if res_x is not None and np.all(np.isfinite(res_x)):
            h_new = project_feasible(np.clip(res_x, 0.0, 1.0), target_final)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Final antipalindromic polish if odd length
    if n_final % 2 == 1 and elapsed() < budget_s - 5:
        half_f = n_final // 2
        g_f = best_h[:half_f].copy()
        g_f = np.clip(g_f, 0.0, 1.0)
        for alpha in [30000, 100000]:
            if elapsed() > budget_s - 3:
                break
            maxiter = max(200, min(400, 300000 // n_final))
            g_new = slsqp_ap(g_f, alpha, maxiter)
            if g_new is not None and np.all(np.isfinite(g_new)):
                h_new = antipal_h(np.clip(g_new, 0.0, 1.0))
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
