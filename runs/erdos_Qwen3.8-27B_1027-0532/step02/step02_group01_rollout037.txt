import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_AB(h):
    """Compute A_m and B_m arrays for binary h."""
    n = len(h)
    A = np.zeros(n)  # A[m] for m=1..n-1
    B = np.zeros(n)  # B[m] for m=1..n-1
    for m in range(1, n):
        # A[m] = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
        # B[m] = sum_{j=0}^{n-1-m} (1-h[j])*h[j+m]
        overlap = n - m
        A[m] = np.sum(h[:overlap] * (1 - h[m:]))
        B[m] = np.sum((1 - h[:overlap]) * h[m:])
    return A, B

def binary_local_search(h, n_steps, t_start, t_end, rng):
    """Fast binary local search with 2-flip moves."""
    n = len(h)
    h = np.round(h).astype(np.float64).copy()
    # Ensure exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) != n // 2:
        # Adjust
        h = np.zeros(n)
        h[ones[:n//2]] = 1.0
    
    current = compute_c5(h)
    best_h = h.copy()
    best_c5 = current
    A_curr, B_curr = compute_AB(h)
    current_max = max(np.max(A_curr[1:]), np.max(B_curr[1:]))
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Pick a random 1 and a random 0
        ones_pos = np.where(h == 1.0)[0]
        zeros_pos = np.where(h == 0.0)[0]
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        
        # Compute delta for A and B
        # Flip i: 1->0, Flip j: 0->1
        delta_A = np.zeros(n)
        delta_B = np.zeros(n)
        
        for m in range(1, n):
            dA = 0.0
            dB = 0.0
            # i as first factor in A: h[i]*(1-h[i+m])
            if i <= n - 1 - m:
                dA -= (1 - h[i + m])
            # i as second factor in A: h[i-m]*(1-h[i])
            if i >= m:
                dA += h[i - m]
            # j as first factor in A: h[j]*(1-h[j+m])
            if j <= n - 1 - m:
                dA += (1 - h[j + m])
            # j as second factor in A: h[j-m]*(1-h[j])
            if j >= m:
                dA -= h[j - m]
            
            # i in B: (1-h[i])*h[i+m]
            if i <= n - 1 - m:
                dB += h[i + m]
            # i in B: (1-h[i-m])*h[i]
            if i >= m:
                dB -= (1 - h[i - m])
            # j in B: (1-h[j])*h[j+m]
            if j <= n - 1 - m:
                dB -= h[j + m]
            # j in B: (1-h[j-m])*h[j]
            if j >= m:
                dB += (1 - h[j - m])
            
            delta_A[m] = dA
            delta_B[m] = dB
        
        # New max
        new_A = A_curr + delta_A
        new_B = B_curr + delta_B
        new_max = max(np.max(new_A[1:]), np.max(new_B[1:]))
        delta = new_max - current_max
        dx = 2.0 / n
        delta_c5 = delta * dx
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(t, 1e-10)):

            h[i] = 0.0
            h[j] = 1.0
            A_curr = new_A
            B_curr = new_B
            current_max = new_max
            current = new_max * dx
            if current < best_c5:
                best_c5 = current
                best_h = h.copy()
        # else: reject, no change
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_AB_binary(h):
    """Compute A_m and B_m for binary h. Returns arrays indexed by m=0..n-1."""
    n = len(h)
    A = np.zeros(n)
    B = np.zeros(n)
    for m in range(1, n):
        ov = n - m
        A[m] = np.dot(h[:ov], 1.0 - h[m:])
        B[m] = np.dot(1.0 - h[:ov], h[m:])
    return A, B

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, time_limit):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    start = time.time()
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        c5 = compute_c5(h_opt)
    except Exception:
        h_opt = h0
        c5 = compute_c5(h0)
    return h_opt, c5

        to_remove = rng.choice(one_idx, size=excess, replace=False)
        h[to_remove] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h == 0.0)[0]
        deficit = n // 2 - n_ones
        to_add = rng.choice(zero_idx, size=deficit, replace=False)
        h[to_add] = 1.0

    A_curr, B_curr = compute_AB_binary(h)
    current_max = max(np.max(A_curr[1:]), np.max(B_curr[1:]))
    best_h = h.copy()
    best_max = current_max
    dx = 2.0 / n
    m_arr = np.arange(1, n)
    
    n_improve = 0
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Pick random 1 and 0
        ones_pos = np.where(h == 1.0)[0]
        zeros_pos = np.where(h == 0.0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        i = int(ones_pos[rng.integers(len(ones_pos))])
        j = int(zeros_pos[rng.integers(len(zeros_pos))])
        
        # Compute delta A and delta B vectorized
        delta_A = np.zeros(n)
        delta_B = np.zeros(n)
        
        # i as first factor in A: h[i]*(1-h[i+m]) for m where i+m < n
        valid = m_arr < (n - i)
        delta_A[valid] += -(1.0 - h[i + m_arr[valid]])
        
        # i as second factor in A: h[i-m]*(1-h[i]) for m where i-m >= 0
        valid = m_arr <= i
        delta_A[valid] += h[i - m_arr[valid]]
        
        # j as first factor in A
        valid = m_arr < (n - j)
        delta_A[valid] += (1.0 - h[j + m_arr[valid]])
        
        # j as second factor in A
        valid = m_arr <= j
        delta_A[valid] += -h[j - m_arr[valid]]
        
        # B: (1-h[k])*h[k+m]
        # i as (1-h[i])
        valid = m_arr < (n - i)
        delta_B[valid] += h[i + m_arr[valid]]
        # i as h[i] (in h[k+m] where k+m=i, k=i-m)
        valid = m_arr <= i
        delta_B[valid] += -(1.0 - h[i - m_arr[valid]])
        
        # j as (1-h[j])
        valid = m_arr < (n - j)
        delta_B[valid] += -h[j + m_arr[valid]]
        # j as h[j]
        valid = m_arr <= j
        delta_B[valid] += (1.0 - h[j - m_arr[valid]])
        
        new_A_max = np.max(A_curr[1:] + delta_A[1:])
        new_B_max = np.max(B_curr[1:] + delta_B[1:])
        new_max = max(new_A_max, new_B_max)
        
        delta_max = new_max - current_max
        if delta_max < 0 or (t > 1e-10 and rng.random() < np.exp(-delta_max / t)):
            h[i] = 0.0
            h[j] = 1.0
            A_curr = A_curr + delta_A
            B_curr = B_curr + delta_B
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                n_improve += 1
    
    return best_h, best_max * dx

For any binary sequence with n/2 ones, consider the sum of A_m over all m:
sum_{m=1}^{n-1} A_m = sum_{m=1}^{n-1} sum_{k=0}^{n-1-m} h[k]*(1-h[k+m])
= sum_{k=0}^{n-1} h[k] * sum_{m=1}^{n-1-k} (1-h[k+m])
= sum_{k=0}^{n-1} h[k] * (n-1-k - sum_{m=1}^{n-1-k} h[k+m])
= sum_{k=0}^{n-1} h[k] * (n-1-k) - sum_{k=0}^{n-1} sum_{m=1}^{n-1-k} h[k]*h[k+m]
= sum_{k=0}^{n-1} h[k]*(n-1-k) - sum_{0≤i<j≤n-1} h[i]*h[j]

The first term depends on where the 1s are placed. The second term is the number of pairs of 1s.

First term = sum_{t=1}^{90} (n-1-p_t)
Second term = C(90, 2) = 4005 (number of pairs of 1s)
