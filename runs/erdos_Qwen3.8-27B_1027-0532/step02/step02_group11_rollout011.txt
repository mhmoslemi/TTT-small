def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def binary_from_continuous(h, n):
    sorted_idx = np.argsort(h)[::-1]

def smooth_max_obj(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        if l >= 0:
            idx = np.arange(n - l)
            grad[idx] += w * (1.0 - h[idx + l])
            grad[idx + l] += w * (-h[idx])
        else:
            ll = -l
            idx = np.arange(n - ll)
            grad[idx + ll] += w * (1.0 - h[idx])
            grad[idx] += w * (-h[idx + ll])

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def binary_from_continuous(h, n):
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[sorted_idx[:n//2]] = 1.0
    return binary

def corr_max_int(h, n):
    return int(np.max(np.correlate(h, 1 - h, mode="full")))

def smooth_max_obj(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    return (m + (1.0 / eps) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = eps * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        if l >= 0:
            idx = np.arange(n - l)
            grad[idx] += w * (1.0 - h[idx + l])
            grad[idx + l] += w * (-h[idx])
        else:
            ll = -l
            idx = np.arange(n - ll)
            grad[idx + ll] += w * (1.0 - h[idx])
            grad[idx] += w * (-h[idx + ll])
    grad *= dx
    return grad

def slsqp_refine(h_start, n, eps, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, eps, n)
    
    def grad(x):
        return smooth_max_grad(x, eps, n)
    
    def constr(x):
        return np.sum(x) - n / 2.0
    
    res = minimize(obj, h_start.copy(), jac=grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constr}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x, n)
        return h_new, evaluate_c5(h_new)
    return h_start, evaluate_c5(h_start)

def steepest_ascent_binary(h_start, n, time_limit, seed, n_candidates=50, max_iters=500000):
    """Steepest descent: at each step, try n_candidates random swaps and take the best."""
    rng = np.random.RandomState(seed)
    h = h_start.copy()
    best_h = h.copy()
    best_corr = corr_max_int(h, n)
    curr_corr = best_corr
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    stall_iters = 0
    max_stall = 10000
    
    for it in range(max_iters):
        if time_limit():
            break
        if stall_iters > max_stall:
            break
        
        best_delta = 0
        best_move = None
        
        for _ in range(n_candidates):
            po = ones[rng.randint(len(ones))]
            pz = zeros[rng.randint(len(zeros))]
            h[po] = 0.0
            h[pz] = 1.0
            new_corr = corr_max_int(h, n)
            h[po] = 1.0
            h[pz] = 0.0
            delta = new_corr - curr_corr
            if delta < best_delta:
                best_delta = delta
                best_move = (po, pz)
        
        if best_move is not None and best_delta < 0:
            po, pz = best_move
            h[po] = 0.0
            h[pz] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            curr_corr += best_delta
            if curr_corr < best_corr:
                best_corr = curr_corr
                best_h = h.copy()
            stall_iters = 0
        else:
            stall_iters += 1
    
    return best_h, best_corr * (2.0 / n)

def upsample(h, n_from, n_to):
    out = np.zeros(n_to)
    for i in range(n_to):
        pos = i * n_from / float(n_to)
        idx = min(int(pos), n_from - 1)
        frac = pos - idx
        if idx + 1 < n_from:
            out[i] = (1.0 - frac) * h[idx] + frac * h[idx + 1]
        else:
            out[i] = h[idx]
    return out
