def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        grad[la:] -= h[:n - la]
    return grad

def slsqp_with_grad(h0, alpha, maxiter=500):
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_correlations(x) * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    def obj_grad(x):
        corr = compute_correlations(x) * dx
        m = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m))
        weights = exp_vals / np.sum(exp_vals)

        for idx in range(len(corr)):
            if weights[idx] < 1e-10:
                continue
            lag = idx - (n - 1)
            grad += weights[idx] * compute_gradient_for_lag(x, lag)
        return grad
    
    def con_f(x):
        return np.sum(x) - target_sum
    
    def con_j(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', 
                   bounds=bounds,

                   options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})

            best_h = h_new.copy()

    # Phase 3: More SLSQP refinement
    if remaining_frac() > 0.30:
        phase3_alphas = [500, 2000, 8000, 20000]
        for alpha in phase3_alphas:
            if remaining_frac() < 0.25:
                break
            tk = min(30, len(compute_correlations(best_h)))
            h_ref = slsqp_optimize(best_h, alpha, maxiter=600, top_k=tk)
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

    # Phase 4: Try different starting points (structured + random)
    if remaining_frac() > 0.30:
        n_starts = 6
        for s in range(n_starts):
            if remaining_frac() < 0.15:
                break
            if s < 3:
                h_start = make_structured_init(n_init, seed + s + 100)
            else:
                pert = rng.randn(n_init) * (0.15 + 0.1 * s) / np.sqrt(n_init)
                h_start = project_to_feasible(best_h + pert, n_init / 2.0)

            h_new = slsqp_optimize(h_start, 300, maxiter=200, top_k=25)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h_new2 = slsqp_optimize(h_new, 2000, maxiter=300, top_k=25)
                score2 = evaluate_score(h_new2)
                if score2 < best_score:
                    best_score = score2
                    best_h = h_new2.copy()

    # Phase 5: Resolution scaling
    if remaining_frac() > 0.15:
        n_candidates = [int(best_n * 2), int(best_n * 2.5), int(best_n * 3)]
        n_candidates = sorted(set([x for x in n_candidates if 500 < x <= 2500 and x != best_n]))
        for n_try in n_candidates:
            if remaining_frac() < 0.08:
                break
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new_tmp = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new_tmp, x_old, best_h)
            h_interp = project_to_feasible(h_interp, n_try / 2.0)

            tk = min(35, 2 * n_try - 1)
            h_new = slsqp_optimize(h_interp, 200, maxiter=150, top_k=tk)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                for alpha in [800, 3000, 10000]:
                    if remaining_frac() < 0.05:
                        break
                    h_new2 = slsqp_optimize(h_new, alpha, maxiter=150, top_k=tk)
                    score2 = evaluate_score(h_new2)
                    if score2 < best_score:
                        best_score = score2
                        best_h = h_new2.copy()

    # Phase 6: Final intensive polish
    if remaining_frac() > 0.05:
        n_final = len(best_h)
        tk_final = min(40, 2 * n_final - 1)
        final_alphas = [3000, 10000, 30000, 100000]
        for alpha in final_alphas:
            if remaining_frac() < 0.02:
                break
            h_ref = slsqp_optimize(best_h, alpha, maxiter=800, top_k=tk_final)
            score = evaluate_score(h_ref)
            if score < best_score:
                best_score = score
                best_h = h_ref.copy()

        # Last gradient descent push
        if remaining_frac() > 0.03:
            h_new, score = gradient_descent_phase(best_h, 5000, 15, 30000, 0.3,
                                                   t0, remaining_frac() * budget_s * 0.03, rng)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()

    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
