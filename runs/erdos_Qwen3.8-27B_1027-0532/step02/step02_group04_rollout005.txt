    shifted = shifted - np.max(shifted)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def compute_lag_grad(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h)
    elif lag > 0:
        if lag < n:
            grad[:n - lag] += 1.0
            grad[:n - lag] -= h[lag:]
            grad[lag:] -= h[:n - lag]
    else:
        l = -lag
        if l < n:
            grad[l:] += 1.0
            grad[l:] -= h[:n - l]
            grad[:n - l] -= h[l:]
    return grad


def smooth_max_gradient(h, alpha, n):
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = shifted - np.max(shifted)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(n)
    for index in range(len(weights)):
        w = weights[index]
        if w < 1e-12:
            continue
        lag = n - 1 - index
        grad += w * compute_lag_grad(h, lag, n)
    return grad


def bottleneck_gradient(h, n, top_k=3):
    corr = compute_correlations(h)
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    w = 1.0 / len(idx)
    for index in idx:
        lag = n - 1 - index
        grad += w * compute_lag_grad(h, lag, n)
    return grad


def ljbfgs_smooth(h0, alpha, n, mu, maxiter):
    def obj(h):
        corr = compute_correlations(h)
        sm = smooth_max_value(corr, alpha)
        s = np.sum(h) - n / 2.0
        return sm + mu * s * s

    def grad(h):
        g = smooth_max_gradient(h, alpha, n)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x


def ljbfgs_bottleneck(h0, n, mu, maxiter, top_k):
    def obj(h):
        corr = compute_correlations(h)
        m = np.max(corr)
        s = np.sum(h) - n / 2.0
        return m + mu * s * s

    def grad(h):
        corr = compute_correlations(h)
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        w = 1.0 / len(idx)
        for index in idx:
            lag = n - 1 - index
            g += w * compute_lag_grad(h, lag, n)
        s = np.sum(h) - n / 2.0
        g += 2.0 * mu * s
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return res.x


def bottleneck_descent(h, n, n_iters, dt0, dt_decay, momentum_beta=0.8):
    best = h.copy()
    best_score = evaluate_score(best)
    dt = dt0
    prev_grad = np.zeros(n)
    top_k = max(2, n // 100)

    for it in range(n_iters):
        corr = compute_correlations(best)
        max_val = np.max(corr)
        current_score = max_val * 2.0 / n

        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        w = 1.0 / len(idx)
        for index in idx:
            lag = n - 1 - index
            g += w * compute_lag_grad(best, lag, n)

        g_mom = g + momentum_beta * prev_grad
        prev_grad = g

        h_try = best - dt * g_mom
        h_try = project_feasible(h_try, n / 2.0)
        score_try = evaluate_score(h_try)

        if score_try < current_score - 1e-14:
            best = h_try
            best_score = score_try
            dt = min(dt * 1.05, 0.05)
        else:
            dt *= dt_decay
            dt = max(dt, 1e-7)

    return best, best_score


def local_search_structured(h, n, n_iters, rng, top_k=15):
    best = h.copy()
    best_score = evaluate_score(best)
    corr = compute_correlations(best)
    max_lags = np.argsort(corr)[-top_k:]
    steps = [0.03, 0.015, 0.008, 0.004, 0.002]

    for it in range(n_iters):
        lag_idx = max_lags[rng.randint(top_k)]
        lag = n - 1 - lag_idx

        if lag > 0 and lag < n:
            contrib = best[lag:] * (1.0 - best[:n - lag])
            i = np.argmax(contrib)
            j = i + lag
            step = steps[rng.randint(len(steps))]
            h_try = best.copy()
            if best[j] > best[i]:
                mv = min(step, best[j] - best[i])
                h_try[i] += mv
                h_try[j] -= mv
            else:
                mv = min(step, best[i] - best[j])
                h_try[j] += mv
                h_try[i] -= mv
        elif lag == 0:
            i = np.argmax(best * (1.0 - best))
            h_try = best.copy()
            if best[i] > 0.5:
                h_try[i] -= 0.02
            else:
                h_try[i] += 0.02
        else:
            l = -lag
            if l < n:
                contrib = best[:n - l] * (1.0 - best[l:])
                i = np.argmax(contrib)
                j = i + l
                step = steps[rng.randint(len(steps))]
                h_try = best.copy()
                if best[j] > best[i]:
                    mv = min(step, best[j] - best[i])
                    h_try[i] += mv
                    h_try[j] -= mv
                else:
                    mv = min(step, best[i] - best[j])
                    h_try[j] += mv
                    h_try[i] -= mv
            else:
                h_try = best.copy()
                i = rng.randint(n)
                h_try[i] += rng.uniform(-0.02, 0.02)

        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n / 2.0)
        score_try = evaluate_score(h_try)

        if score_try < best_score - 1e-13:
            best = h_try
            best_score = score_try
            if it % 30 == 0:
                corr = compute_correlations(best)
                max_lags = np.argsort(corr)[-top_k:]

    return best, best_score


def random_pair_search(h, n, n_iters, rng):
    best = h.copy()
    best_score = evaluate_score(best)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    for it in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = steps[rng.randint(len(steps))]
        h_try = best.copy()
        h_try[i] += step
        h_try[j] -= step
        if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
            continue
        h_try = project_feasible(h_try, n / 2.0)
        score_try = evaluate_score(h_try)
        if score_try < best_score - 1e-13:
            best = h_try
            best_score = score_try
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    init = np.array(initial_h_values, dtype=float)
    n_parent = len(init)

    def elapsed():
        return time.time() - t0

    n = 850
    x_old = np.linspace(0, 2, n_parent, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h = np.interp(x_new, x_old, init)
    h = project_feasible(h, n / 2.0)

    best_h = h.copy()
    best_score = evaluate_score(h)

    # Phase 1: Smooth max L-BFGS-B sweep
    alphas_p1 = [20, 60, 150, 400, 1000, 2500, 5000, 10000]
    for alpha in alphas_p1:
        if elapsed() > budget_s - 350:
            break
        maxiter = max(100, min(4000, 300000 // n))
        x = ljbfgs_smooth(h, alpha, n, 50000.0, maxiter)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Bottleneck L-BFGS-B (direct max, no smoothing)
    if elapsed() < budget_s - 300:
        for top_k in [1, 2, 3, 5]:
            if elapsed() > budget_s - 250:
                break
            maxiter = max(200, 200000 // n)
            x = ljbfgs_bottleneck(h, n, 50000.0, maxiter, top_k)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 3: Bottleneck gradient descent with momentum
    if elapsed() < budget_s - 250:
        time_avail = min(120, budget_s - elapsed() - 250)
        n_iters = int(time_avail * 60)
        h2, s2 = bottleneck_descent(h, n, n_iters, 0.015, 0.995, 0.85)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()
            h = h2.copy()

    # Phase 4: More smooth max at very high alpha after bottleneck
    if elapsed() < budget_s - 200:
        for alpha in [8000, 15000, 30000]:
            if elapsed() > budget_s - 150:
                break
            maxiter = max(100, 150000 // n)
            x = ljbfgs_smooth(h, alpha, n, 50000.0, maxiter)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 5: Higher resolution
    n2 = 1275
    if elapsed() < budget_s - 180:
        x_old2 = np.linspace(0, 2, n, endpoint=False)
        x_new2 = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new2, x_old2, best_h)
        h_up = project_feasible(h_up, n2 / 2.0)
        n = n2

        for alpha in [100, 500, 2000, 8000]:
            if elapsed() > budget_s - 100:
                break
            maxiter = max(50, 120000 // n)
            x = ljbfgs_smooth(h_up, alpha, n, 50000.0, maxiter)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_up = h_new.copy()

        if elapsed() < budget_s - 80:
            time_avail = min(60, budget_s - elapsed() - 80)
            n_iters = int(time_avail * 40)
            h2, s2 = bottleneck_descent(h_up, n, n_iters, 0.01, 0.996, 0.8)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()
                h = h2.copy()

    # Phase 6: Multiple restarts with diverse strategies
    n_restarts = 150
    n = len(best_h)
    for i in range(n_restarts):
        if elapsed() > budget_s - 100:
            break
        strategy = i % 6
        if strategy == 0:
            scale = 0.02 + 0.04 * (i / float(n_restarts))
            pert = best_h + rng.randn(n) * scale
        elif strategy == 1:
            pert = best_h.copy()
            blen = rng.randint(5, min(40, n // 8))
            bstart = rng.randint(0, n - blen)
            pert[bstart:bstart + blen] = 1.0 - pert[bstart:bstart + blen]
        elif strategy == 2:
            pert = best_h.copy()
            n_freq = rng.randint(1, 5)
            x_idx = np.arange(n, dtype=float)
            for k in range(n_freq):
                f = rng.randint(1, 6)
                amp = rng.uniform(0.03, 0.1)
                phase = rng.rand() * 2.0 * np.pi
                pert += amp * np.sin(2.0 * np.pi * f * x_idx / n + phase)
        elif strategy == 3:
            pert = best_h.copy()
            n_blocks = rng.randint(2, 5)
            block_size = n // n_blocks
            for b in range(n_blocks):
                s0 = b * block_size
                s1 = min(s0 + block_size, n)
                if rng.rand() < 0.4:
                    pert[s0:s1] = 1.0 - pert[s0:s1]
                else:
                    pert[s0:s1] += rng.randn(s1 - s0) * 0.08
        elif strategy == 4:
            pert = best_h + rng.randn(n) * 0.12
        else:
            pert = best_h.copy()
            n_swaps = rng.randint(3, 10)
            for _ in range(n_swaps):
                i1 = rng.randint(n)
                i2 = rng.randint(n)
                pert[i1], pert[i2] = pert[i2], pert[i1]

        pert = project_feasible(pert, n / 2.0)
        alpha_pick = [100, 500, 2000, 5000][rng.randint(4)]
        maxiter_r = max(80, 80000 // n)
        x = ljbfgs_smooth(pert, alpha_pick, n, 50000.0, maxiter_r)
        if x is not None and np.all(np.isfinite(x)):
            h_new = project_feasible(x, n / 2.0)
            s = evaluate_score(h_new)
            if s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 7: Structured local search
    if elapsed() < budget_s - 50:
        n = len(best_h)
        time_avail = max(10, min(80, budget_s - elapsed() - 50))
        n_iters = int(time_avail * 40)
        h2, s2 = local_search_structured(best_h, n, n_iters, rng, top_k=len(n)//80+2)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    # Phase 8: Random pair search
    if elapsed() < budget_s - 30:
        n = len(best_h)
        time_avail = max(5, min(40, budget_s - elapsed() - 30))
        n_iters = int(time_avail * 100)
        h2, s2 = random_pair_search(best_h, n, n_iters, rng)
        if s2 < best_score:
            best_score = s2
            best_h = h2.copy()

    # Phase 9: Final high-alpha polish
    if elapsed() < budget_s - 15:
        n = len(best_h)
        for alpha in [20000, 50000]:
            if elapsed() > budget_s - 8:
                break
            maxiter_f = max(100, 80000 // n)
            x = ljbfgs_smooth(best_h, alpha, n, 50000.0, maxiter_f)
            if x is not None and np.all(np.isfinite(x)):
                h_new = project_feasible(x, n / 2.0)
                s = evaluate_score(h_new)
                if s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Final bottleneck descent squeeze
    if elapsed() < budget_s - 8:
        n = len(best_h)
        time_avail = max(2, budget_s - elapsed() - 8)
        n_iters = int(time_avail * 80)
        if n_iters > 10:
            h2, s2 = bottleneck_descent(best_h, n, n_iters, 0.008, 0.998, 0.7)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)

    return (best_h, best_score, len(best_h))
