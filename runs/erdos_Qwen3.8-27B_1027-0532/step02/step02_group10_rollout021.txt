    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)

        else:

**Approach: Iterative water-filling**

The idea is:
1. Compute the current overlaps.
2. Identify the "bottleneck" lag k* (the one with maximum overlap).
3. Adjust h to specifically reduce overlap at k*, while being careful not to increase other overlaps too much.
4. Repeat.

This is essentially what the pair-swap approach does, but let me implement it more simply:

For each iteration:
1. Compute overlaps, find k*.
2. Compute the gradient of overlap(k*) w.r.t. h:
   grad_k*[j] = d/dh[j] [sum_i h[i+k*](1-h[i])]
             = (1 if j+k* is in range, 1-h[j+k*] if j is the "first" index, etc.)
   
   More precisely:
   overlap(k*) = sum_i h[i+k*](1-h[i])
   d/dh[j] = -h[j+k*] (from the (1-h[j]) term) + (1-h[j-k*]) (from the h[j] = h[i+k*] term where i=j-k*)
   = 1 - h[j-k*] - h[j+k*] (with out-of-range terms = 0)

3. Descent direction: -grad_k* (to reduce overlap at k*).
4. But we need to handle the constraint sum(h) = n/2. The gradient of the constraint is (1,1,...,1).
5. Projected gradient step: h_new = proj_{constraint}(h - step * (grad_k* - mean(grad_k*)))
   (subtracting the mean makes it orthogonal to the constraint gradient).

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(h, target_sum):
    n = len(h)
    h = h.copy()
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            mask2 = (h < 1.0)
            if mask2.sum() == 0:
                break
            h[mask2] += diff / mask2.sum()
        else:
            h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_and_weights(x, alpha):
    m = np.max(x)
    shifted = alpha * (x - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    value = m + np.log(total) / alpha
    weights = exp_s / total
    return value, weights


def compute_grad_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            i_lo = 0
            i_hi = n - offset
        else:
            i_lo = -offset
            i_hi = n
        if i_lo < i_hi:
            grad[i_lo:i_hi] += -w * dx * h[i_lo + offset:i_hi + offset]
            grad[i_lo + offset:i_hi + offset] += w * dx * (1.0 - h[i_lo:i_hi])
    return grad


def pgd_optimize(h, alpha, target, max_iters, deadline):

    dx = 2.0 / n
    h = h.copy()

    prev_sm = None
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h)
        sm, weights = smooth_max_and_weights(overlaps, alpha)
        if prev_sm is not None:
            if sm < prev_sm - 1e-15:
                stall = 0
            else:
                stall += 1
            if stall > 500:
                break
        prev_sm = sm
        grad = compute_grad_weighted(h, weights, dx)
        improved = False
        s = step
        for bt in range(28):
            h_trial = project_onto_constraint(h - s * grad, target)
            ov_t = compute_overlaps(h_trial)
            sm_t, _ = smooth_max_and_weights(ov_t, alpha)
            if sm_t < sm - 1e-15:
                h = h_trial
                improved = True
                break
            s *= 0.5
            if s < 1e-14:
                break
        if not improved:
            break
        step = min(s * 1.4, 0.5)

def lag_gradient(h, k):
    """Compute gradient of overlap at specific lag k w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        i_hi = n - k
        grad[:i_hi] += -h[k:i_hi + k]
        grad[k:n] += (1.0 - h[:n - k])
    else:
        kk = -k
        i_hi = n - kk
        grad[kk:n] += -h[:i_hi]
        grad[:i_hi] += (1.0 - h[kk:n])
    return grad

def pair_transfer_search(h, target, deadline, max_rounds=20000):
    """Local search: transfer small amounts between pairs to reduce max overlap."""
    n = len(h)
    h = h.copy()

    stall = 0

            break
        if stall > 200:
            break

        current_max = np.max(overlaps)
        k_star_idx = np.argmax(overlaps)
        k_star = k_star_idx - (n - 1)

        # Compute D_k for the star lag: D_k[p] = h[p+k] + h[p-k] (with bounds)
        D_p = np.zeros(n)
        for p in range(n):
            val = 0.0
            if 0 <= p + k_star < n:
                val += h[p + k_star]
            if 0 <= p - k_star < n:
                val += h[p - k_star]
            D_p[p] = val

        # We want to transfer from p (high D_p) to q (low D_p)
        # to reduce overlap at k_star
        # Transfer delta from p to q: change at k_star = delta * (D_p[p] - D_p[q])
        # We need D_p[p] > D_p[q] for this to help

        # Find top and bottom D_p values
        p_candidates = np.where(h > 0.01)[0]
        q_candidates = np.where(h < 0.99)[0]
        if len(p_candidates) == 0 or len(q_candidates) == 0:
            stall += 1
            continue

        # Sort by D_p
        p_order = p_candidates[np.argsort(-D_p[p_candidates])]
        q_order = q_candidates[np.argsort(D_p[q_candidates])]

        found = False
        for p in p_order[:50]:
            if found:
                break
            for q in q_order[:50]:
                if q == p:
                    continue
                D_diff = D_p[p] - D_p[q]
                if D_diff <= 0:
                    break
                # Try a transfer

                # Bisect on delta to find the largest that doesn't increase max

                hi_d = max_delta * 0.8
                best_d = 0.0
                for _ in range(15):
                    mid_d = (lo_d + hi_d) / 2.0
                    if mid_d < 1e-12:
                        break
                    h[p] -= mid_d
                    h[q] += mid_d
                    ov_check = compute_overlaps(h)
                    h[p] += mid_d
                    h[q] -= mid_d
                    if np.max(ov_check) < current_max - 1e-14:

                    else:
                        hi_d = mid_d

                    h[p] -= best_d
                    h[q] += best_d
                    found = True
                    stall = 0
                    break

        if not found:
            stall += 1

    return h

        h = project_onto_constraint(h, target)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Phase 8: Multi-resolution
    if remaining() > 120:
        budget_mr = min(remaining() * 0.4, 300)
        best_h_mr, best_score_mr = multi_resolution_optimize(best_h, target, budget_mr, rng)
        if best_score_mr < best_score:
            best_score = best_score_mr
            best_h = best_h_mr.copy()

    # Phase 9: Final refinement at whatever resolution we're at
    n_final = len(best_h)
    target_final = n_final / 2.0
    for alpha in [5000, 20000, 50000]:
        if remaining() < 20:
            break
        dl = time.time() + min(10, remaining() * 0.3)
        h = pgd_optimize(best_h, alpha, target_final, 20000, dl)
        h = project_onto_constraint(h, target_final)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    if remaining() > 15:
        dl = time.time() + min(10, remaining() * 0.4)
        h = single_lag_descent(best_h, target_final, dl)
        h = project_onto_constraint(h, target_final)
        s = evaluate_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    best_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
