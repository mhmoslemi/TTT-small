import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx


def project_sum(h, target):
    n = len(h)
    h = h.copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_obj_grad(h, alpha, mu, dx):
    n = len(h)
    g = 1.0 - h
    corrs = np.correlate(h, g, mode='full')
    m = np.max(corrs)
    shifted = alpha * (corrs - m)
    shifted = shifted - np.max(shifted)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = (m + np.log(sum_exp) / alpha) * dx
    w_rev = weights[::-1]
    conv1 = np.convolve(g, w_rev, mode='full')
    conv2 = np.convolve(h, w_rev, mode='full')
    idx = np.arange(n) + n - 1
    grad = (conv1[idx] - conv2[idx]) * dx
    viol = np.sum(h) - n / 2.0
    obj += 0.5 * mu * viol * viol * dx
    grad += mu * viol * dx
    return obj, grad


def lbfgsb_opt(h0, alpha, mu, dx, maxiter, bounds):
    res = minimize(
        smooth_max_obj_grad, h0, method='L-BFGS-B',
        bounds=bounds, args=(alpha, mu, dx),
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                 'iprint': 0, 'maxls': 100, 'maxfun': maxiter * 50}
    )
    return res.x


def slsqp_obj(h, alpha, mu, dx):
    return smooth_max_obj_grad(h, alpha, mu, dx)[0]


def slsqp_jac(h, alpha, mu, dx):
    return smooth_max_obj_grad(h, alpha, mu, dx)[1]


def eq_constraint(h):
    return np.sum(h) - len(h) / 2.0


def eq_constraint_jac(h):
    return np.ones(len(h))


def slsqp_opt(h0, alpha, mu, dx, maxiter, bounds):
    res = minimize(
        slsqp_obj, h0, method='SLSQP', bounds=bounds,
        args=(alpha, mu, dx), jac=slsqp_jac,
        constraints=[{'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
    )
    return res.x


def coordinate_descent(h, n_sweeps, dx):
    n = len(h)
    corrs = compute_correlations(h).astype(float)
    r_range = np.arange(2 * n - 1) - (n - 1)
    for _ in range(n_sweeps):
        for i in range(n):
            valid1 = (i + r_range >= 0) & (i + r_range < n)
            valid2 = (i - r_range >= 0) & (i - r_range < n)
            a_coeffs = np.zeros(2 * n - 1)
            if np.any(valid1):
                idx1 = i + r_range[valid1]
                a_coeffs[valid1] += (1.0 - h[idx1])
            if np.any(valid2):
                idx2 = i - r_range[valid2]
                a_coeffs[valid2] -= h[idx2]
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-12:
                continue
            best_delta = 0.0
            best_val = np.max(corrs)
            n_eval = 30
            delta_grid = np.linspace(lo, hi, n_eval)
            for d in delta_grid:
                v = np.max(corrs + d * a_coeffs)
                if v < best_val:
                    best_val = v
                    best_delta = d
            for _ref in range(3):
                span = (hi - lo) / (n_eval * 3)
                d_refined = np.linspace(best_delta - span, best_delta + span, 20)
                d_refined = np.clip(d_refined, lo, hi)
                for d in d_refined:
                    v = np.max(corrs + d * a_coeffs)
                    if v < best_val:
                        best_val = v
                        best_delta = d
            if abs(best_delta) > 1e-12:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                corrs += best_delta * a_coeffs
    h = project_sum(h, n / 2.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    init = np.array(initial_h_values, dtype=float)
    n0 = len(init)
    best_h = project_sum(init.copy(), n0 / 2.0)
    best_score = evaluate_score(best_h)

    # Phase 1: Optimize at original resolution
    bounds0 = [(0.0, 1.0)] * n0
    dx0 = 2.0 / n0
    h_curr = best_h.copy()
    for alpha in [50, 100, 200, 500, 1000, 2000, 5000]:
        if time_left() < 250:
            break
        mu = 10000.0
        maxiter = max(200, min(800, 400000 // n0))
        res_x = lbfgsb_opt(h_curr, alpha, mu, dx0, maxiter, bounds0)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n0 / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_curr = h_new.copy()

    # Phase 2: Coordinate descent polish at original resolution
    if time_left() > 60:
        n_cd = max(5, min(20, 300000 // n0))
        h_new = coordinate_descent(h_curr.copy(), n_cd, dx0)
        s = evaluate_score(h_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_new.copy()
            h_curr = h_new.copy()

    # Phase 3: SLSQP at high alpha
    for alpha in [5000, 10000, 20000]:
        if time_left() < 150:
            break
        res_x = slsqp_opt(h_curr, alpha, 50000.0, dx0, 150, bounds0)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n0 / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    h_curr = h_new.copy()

    # Phase 4: Higher resolutions
    for n_try in [500, 700, 900, 1200, 1500, 2000]:
        if time_left() < 200:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_sum(h_init, n_try / 2.0)
        bounds_try = [(0.0, 1.0)] * n_try
        dx_try = 2.0 / n_try
        h_cur = h_init.copy()
        for alpha in [80, 200, 500, 1500, 4000, 10000]:
            if time_left() < 50:
                break
            mu_try = 50000.0
            maxiter = max(80, min(400, 250000 // n_try))
            res_x = lbfgsb_opt(h_cur, alpha, mu_try, dx_try, maxiter, bounds_try)
            if res_x is not None:
                h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_try / 2.0)
                if np.all(np.isfinite(h_new)):
                    s = evaluate_score(h_new)
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_h = h_new.copy()
                        h_cur = h_new.copy()
        # Coordinate descent at this resolution
        if time_left() > 80:
            n_cd = max(3, min(15, 200000 // n_try))
            h_new = coordinate_descent(h_cur.copy(), n_cd, dx_try)
            s = evaluate_score(h_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_h = h_new.copy()

    # Phase 5: Random restarts at best resolution
    n_best = len(best_h)
    bounds_best = [(0.0, 1.0)] * n_best
    dx_best = 2.0 / n_best
    mu_best = 50000.0

    for i in range(300):
        if time_left() < 60:
            break
        strategy = i % 10
        if strategy == 0:
            pert = best_h + rng.randn(n_best) * 0.04
        elif strategy == 1:
            pert = best_h + rng.randn(n_best) * 0.12
        elif strategy == 2:
            pert = best_h.copy()
            n_fl = max(3, n_best // 25)
            idx = rng.choice(n_best, size=n_fl, replace=False)
            pert[idx] = 1.0 - pert[idx]
        elif strategy == 3:
            shift = rng.randint(1, max(2, n_best // 15))
            pert = np.roll(best_h, shift)
        elif strategy == 4:
            t = np.linspace(0, 4 * np.pi, n_best)
            scale = 0.1 + 0.1 * rng.rand()
            pert = best_h + scale * np.sin(t + rng.uniform(0, 2 * np.pi)) * np.cos(0.5 * t)
        elif strategy == 5:
            sorted_idx = np.argsort(best_h)
            pert = np.zeros(n_best)
            pert[sorted_idx[n_best // 5:4 * n_best // 5]] = 1.0
            pert += rng.randn(n_best) * 0.2
        elif strategy == 6:
            noise = rng.randn(n_best)
            kernel = np.ones(7) / 7
            noise = np.convolve(noise, kernel, mode='same')
            pert = best_h + 0.08 * noise / (np.std(noise) + 1e-10)
        elif strategy == 7:
            pert = best_h.copy()
            mask = rng.rand(n_best) < 0.15
            pert[mask] = rng.rand(int(np.sum(mask)))
        elif strategy == 8:
            pert = best_h + rng.randn(n_best) * 0.06
            pert = coordinate_descent(pert, 3, dx_best)
        else:
            pert = best_h * (0.9 + 0.2 * rng.rand()) + 0.1 * rng.rand(n_best)

        pert = project_sum(np.clip(pert, 0.0, 1.0), n_best / 2.0)
        alpha = 150 + 1500 * rng.randint(6)
        maxiter_r = max(80, min(350, 180000 // n_best))
        res_x = lbfgsb_opt(pert, alpha, mu_best, dx_best, maxiter_r, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()
                    if time_left() > 120 and n_best < 2000:
                        n_cd = max(3, min(10, 150000 // n_best))
                        h_poly = coordinate_descent(best_h.copy(), n_cd, dx_best)
                        s2 = evaluate_score(h_poly)
                        if np.isfinite(s2) and s2 < best_score:
                            best_score = s2
                            best_h = h_poly.copy()

    # Phase 6: Final coordinate descent
    if time_left() > 30:
        n_cd = max(5, min(25, 400000 // n_best))
        h_new = coordinate_descent(best_h.copy(), n_cd, dx_best)
        s = evaluate_score(h_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_new.copy()

    # Phase 7: Final SLSQP + very high alpha L-BFGS-B
    for alpha in [10000, 30000, 80000]:
        if time_left() < 20:
            break
        res_x = slsqp_opt(best_h, alpha, 200000.0, dx_best, 80, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    for alpha in [50000, 150000, 300000]:
        if time_left() < 10:
            break
        res_x = lbfgsb_opt(best_h, alpha, 500000.0, dx_best, 150, bounds_best)
        if res_x is not None:
            h_new = project_sum(np.clip(res_x, 0.0, 1.0), n_best / 2.0)
            if np.all(np.isfinite(h_new)):
                s = evaluate_score(h_new)
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_h = h_new.copy()

    # Phase 8: Final coordinate descent squeeze
    if time_left() > 15:
        h_new = coordinate_descent(best_h.copy(), 10, dx_best)
        s = evaluate_score(h_new)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_h = h_new.copy()

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_score = evaluate_score(best_h)
    return (best_h, best_score, len(best_h))
