    """Compute delta to overlap array when h[i] goes 0->1 and h[j] goes 1->0."""
    n = len(h)
    delta = np.zeros(2*n-1)
    # i: 0 -> 1
    # As first index: shift L where i+L in [0,n-1] -> L in [-i, n-1-i]
    #   correlate index m = n-1+L in [n-1-i, 2n-2-i]
    #   contribution: (1-h[i+L]) = (1-h[m-(n-1)+i]) = (1-h[m-n+1+i])

    hi1 = 2 * n - 2 - i
    for m in range(lo1, hi1 + 1):
        idx = m - n + 1 + i
        delta[m] += (1.0 - h[idx])
    # As second index: shift L where i-L in [0,n-1] -> L in [i-n+1, i]
    #   correlate index m = n-1+L in [i, n-1+i]
    #   contribution: -h[i-L] = -h[i-(m-n+1)] = -h[n-1+i-m]

def swap_delta(h, i, j):
    """Compute change in overlap array when h[i]: 0->1, h[j]: 1->0"""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # i as first index in h[i]*(1-h[i+L]): L in [-i, n-1-i], m = n-1+L in [n-1-i, 2n-2-i]
    lo = n - 1 - i
    hi = 2 * n - 2 - i
    idx = np.arange(lo, hi + 1) - n + 1 + i
    delta[lo:hi+1] += 1.0 - h[idx]
    # i as second index: h[i-L]*(1-h[i]): L in [i-n+1, i], m in [i, n-1+i]
    lo = i
    hi = n - 1 + i
    idx = n - 1 + i - np.arange(lo, hi + 1)
    delta[lo:hi+1] -= h[idx]
    # j as first index (removing): L in [-j, n-1-j], m in [n-1-j, 2n-2-j]
    lo = n - 1 - j
    hi = 2 * n - 2 - j
    idx = np.arange(lo, hi + 1) - n + 1 + j
    delta[lo:hi+1] -= 1.0 - h[idx]
    # j as second index (removing): L in [j-n+1, j], m in [j, n-1+j]
    lo = j
    hi = n - 1 + j
    idx = n - 1 + j - np.arange(lo, hi + 1)
    delta[lo:hi+1] += h[idx]
    return delta

def binary_sa(h_bin, overlaps, n_ones, n_iters, t_init, t_decay, rng, check_interval=100):
    """Simulated annealing on binary h with incremental overlap tracking."""
    n = len(h_bin)
    h_c = h_bin.copy().astype(float)
    ov_c = overlaps.copy().astype(float)
    best_ov = ov_c.copy()
    best_score = float(np.max(ov_c))
    
    ones = np.where(h_c == 1.0)[0]
    zeros = np.where(h_c == 0.0)[0]
    
    t = t_init
    for it in range(n_iters):
        if it % check_interval == check_interval - 1:
            t *= t_decay
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        d = swap_delta(h_c, i, j)
        ov_new = ov_c + d
        max_new = float(np.max(ov_new))
        
        if max_new < best_score:
            best_score = max_new
            best_ov = ov_new.copy()
        
        # Acceptance
        delta = max_new - float(np.max(ov_c))
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            # Apply swap
            h_c[i] = 1.0
            h_c[j] = 0.0
            ov_c = ov_new.copy()
            # Update ones and zeros
            ones = ones[ones != j]
            zeros = zeros[zeros != i]
            ones = np.append(ones, i)
            zeros = np.append(zeros, j)
        # else: reject, no change
    
    return h_c, ov_c, best_ov, best_score

Let me also think about the block moves. A block move of length B:
- Select a contiguous block of B zeros
- Select a contiguous block of B ones
- Swap them

This is more powerful than single swaps but less common. I'll try it periodically in the SA.

For the delta computation of a block swap, I can just sum the individual delta computations for each position in the block. But that's O(B*n) which is still fast for small B.

Actually, I realize that for the block swap, I need to be careful: the delta for one position depends on the current state of h, and if I'm swapping multiple positions simultaneously, the deltas interact (the "first index" contribution of one position depends on the value at the "second index" position, which might also be changing).

To handle this correctly, I should compute the delta using the OLD values of h (before the swap), and the formula still works because:
- For position i changing 0→1: the contribution is (1-h[i+L]) - h[i-L] using OLD h values
- For position j changing 1→0: the contribution is -(1-h[j+L]) + h[j-L] using OLD h values

Since all these use OLD h, the total delta for a block swap is just the sum of individual deltas for each position. This is correct!

So for a block swap of positions i_1,...,i_B (zeros) and j_1,...,j_B (ones):
total_delta = sum of swap_delta for each pair (i_k, j_k)

Wait no, that's not right either. swap_delta(h, i, j) computes the change when i goes 0→1 AND j goes 1→0 simultaneously. For a block swap, I want ALL i_k to go 0→1 and ALL j_k to go 1→0 simultaneously.

The total delta is:
For each i_k (0→1): contribution as first and second index
For each j_k (1→0): contribution as first and second index

And since all use OLD h values, the total is:
sum_k [delta_first(i_k) + delta_second(i_k) - delta_first(j_k) - delta_second(j_k)]

Where delta_first(i) is the change when position i goes 0→1 in its role as first index, etc.

This is just the sum of individual single-position deltas. So I can compute it as:
total = sum over all changing positions of (signed single-position delta)

Let me define:
- pos_delta(h, p, sign): change in overlaps when h[p] changes by `sign` (either +1 for 0→1 or -1 for 1→0)

```python
def pos_delta(h, p, sign):
    """Change in overlap array when h[p] changes by `sign`."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # p as first index: m in [n-1-p, 2n-2-p], delta[m] += sign * (1-h[p+L])
    lo = n - 1 - p
    hi = 2 * n - 2 - p
    idx = np.arange(lo, hi + 1) - n + 1 + p
    delta[lo:hi+1] += sign * (1.0 - h[idx])
    # p as second index: m in [p, n-1+p], delta[m] += sign * (-h[p-L]) = -sign * h[p-L]
    lo = p
    hi = n - 1 + p
    idx = n - 1 + p - np.arange(lo, hi + 1)
    delta[lo:hi+1] -= sign * h[idx]
    return delta
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h))) * dx


def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def pos_delta(h, p, sign):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    lo = n - 1 - p
    hi = 2 * n - 2 - p
    idx = np.arange(lo, hi + 1) - n + 1 + p
    delta[lo:hi+1] += sign * (1.0 - h[idx])
    lo = p
    hi = n - 1 + p
    idx = n - 1 + p - np.arange(lo, hi + 1)
    delta[lo:hi+1] -= sign * h[idx]
    return delta


def swap_delta(h, i, j):
    return pos_delta(h, i, 1.0) + pos_delta(h, j, -1.0)


def block_swap_delta(h, zero_block, one_block):
    n = len(h)
    total = np.zeros(2 * n - 1)
    for p in zero_block:
        total += pos_delta(h, p, 1.0)
    for p in one_block:
        total += pos_delta(h, p, -1.0)
    return total


def lse_grad(h, alpha):
    n = len(h)
    ov = compute_overlaps(h)
    m = np.max(ov)
    exp_v = np.exp(alpha * (ov - m))
    w = exp_v / np.sum(exp_v)
    sig = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for idx in sig:
        L = idx - (n - 1)
        if L >= 0:
            grad[:n - L] += w[idx] * (1.0 - h[L:])
            grad[L:] -= w[idx] * h[:n - L]
        else:
            M = -L
            grad[M:] += w[idx] * (1.0 - h[:n - M])
            grad[:n - M] -= w[idx] * h[M:]
    return grad


def continuous_polish(h0, n_iters, alpha, lr, target, decay=0.999):

    h = h0.copy().astype(float)
    best_h = h.copy()
    best_raw = float(np.max(compute_overlaps(h)))
    cur = lr
    for it in range(n_iters):
        grad = lse_grad(h, alpha)
        h_new = project_box_sum(h - cur * grad, target)
        raw = float(np.max(compute_overlaps(h_new)))
        if raw < best_raw:
            best_raw = raw
            best_h = h_new.copy()
        h = h_new
        cur *= decay
    return best_h, best_raw


def binary_sa(h_bin, n_iters, t_init, t_decay, rng, t_interval=50, use_blocks=False):

def optimize_fractional(h_bin, target):
    n = len(h_bin)
    ov_base = compute_overlaps(h_bin)
    s = float(np.sum(h_bin))
    if s > target:
        excess = s - target
        candidates = np.where(h_bin > 0.01)[0]
        best_max = float(np.inf)
        best_p = candidates[0]
        for p in candidates:
            d = pos_delta(h_bin, p, -excess)
            mx = float(np.max(ov_base + d))
            if mx < best_max:
                best_max = mx
                best_p = p
        h_out = h_bin.copy().astype(float)
        h_out[best_p] -= excess
        return h_out
    else:
        deficit = target - s
        candidates = np.where(h_bin < 0.99)[0]
        best_max = float(np.inf)
        best_p = candidates[0]
        for p in candidates:
            d = pos_delta(h_bin, p, deficit)
            mx = float(np.max(ov_base + d))
            if mx < best_max:
                best_max = mx
                best_p = p
        h_out = h_bin.copy().astype(float)
        h_out[best_p] += deficit
        return h_out


def lse_optimize(h0, alpha, maxiter, bounds, target):
    def obj(h):
        ov = compute_overlaps(h)
        m = np.max(ov)
        return float(m + np.log(np.sum(np.exp(alpha * (ov - m)))) / alpha)
    def grad(h):
        return lse_grad(h, alpha)
    def cons_fun(h):
        return np.sum(h) - target
    def cons_jac(h):
        return np.ones(len(h))
    try:
        res = minimize(
            obj, h0, jac=grad, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return res.x
    except Exception:
        return None
