       y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k)

import time

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlation(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_D(h):
    """Compute the delta matrix D[pos, lag_idx] for flipping h[pos] from 0 to 1."""
    n = len(h)
    D = np.zeros((n, 2 * n - 1))
    off = n - 1
    for L in range(n):
        idx = L + off
        D[L:, idx] = 1.0 - h[:n - L]
        D[:n - L, idx] += -h[L:]
    for m in range(1, n):
        idx = off - m
        D[m:, idx] = 1.0 - h[:n - m]
        D[:n - m, idx] += -h[m:]
    return D

def binary_local_search(h, max_passes=100, rng=None):
    """Efficient binary local search using precomputed delta matrix."""
    n = len(h)
    h = h.copy()
    c = compute_correlation(h)
    current_max = float(np.max(c))
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for _ in range(max_passes):
        D = compute_D(h)
        zeros_idx = np.where(h < 0.5)[0]
        ones_idx = np.where(h > 0.5)[0]
        
        if len(zeros_idx) == 0 or len(ones_idx) = 0:
            break
        
        # Shuffle for diversification
        if rng is not None:
            zeros_idx = zeros_idx[rng.permutation(len(zeros_idx))]
        
        improved = False
        for i in zeros_idx:
            if h[i] >= 0.5:
                continue
            A = c + D[i, :]
            M = A[None, :] - D[ones_idx, :]
            new_maxes = M.max(axis=1)
            best_j_pos = new_maxes.argmin()
            if new_maxes[best_j_pos] < current_max - 1e-12:
                j = ones_idx[best_j_pos]
                # Perform swap
                h[i] = 1.0
                h[j] = 0.0
                c = c + D[i, :] - D[j, :]
                current_max = float(np.max(c))
                improved = True
                # Remove i from zeros and j from ones
                zeros_idx = np.delete(zeros_idx, np.argmin(np.abs(zeros_idx - i)))
                ones_idx = np.delete(ones_idx, np.argmin(np.abs(ones_idx - j)))
                break
        
        if not improved:
            break
    
    return h, current_max

    n = len(h)
    h = h.copy()
    zeros_idx = np.where(h < 0.5)[0]
    ones_idx = np.where(h > 0.5)[0]
    for _ in range(min(n_swaps, min(len(zeros_idx), len(ones_idx)))):
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        zi = rng.randint(len(zeros_idx))
        oi = rng.randint(len(ones_idx))
        i, j = zeros_idx[zi], ones_idx[oi]
        h[i], h[j] = 1.0, 0.0
        zeros_idx = np.delete(zeros_idx, zi)
        ones_idx = np.delete(ones_idx, oi)
    return h

**Connection to sequence design:**
The problem is equivalent to finding a binary sequence of length n (with n/2 ones, up to the fractional part) that minimizes the maximum "asymmetric autocorrelation" over all shifts.

Wait, that's the periodic autocorrelation (balanced version). Our overlap is different - it's one-sided.

Actually, our overlap at positive lag k is:
o(k) = #{i: 0≤i<n-k, x_i=0, x_{i+k}=1}

And at negative lag -k:
o(-k) = #{i: k≤i<n, x_i=1, x_{i-k}=0}

Note: o(k) + o(-k) = #{i: 0≤i<n-k, x_i≠x_{i+k}} + #{i: k≤i<n-1, x_i≠x_{i+k}} ... no, that's not right either.

o(-k) = Σ_{i=k}^{n-1} x_i * (1-x_{i-k})

Let j = i-k in the second sum: o(-k) = Σ_{j=0}^{n-1-k} x_{j+k} * (1-x_j)

So o(k) = Σ_{j=0}^{n-1-k} (1-x_j) * x_{j+k} = Σ_{j=0}^{n-1-k} x_{j+k} - x_j * x_{j+k}
And o(-k) = Σ_{j=0}^{n-1-k} x_{j+k} * (1-x_j) = Σ_{j=0}^{n-1-k} x_{j+k} - x_j * x_{j+k}

Wait, they're the same! o(k) = o(-k)?

Let me double-check. o(k) for k>0:
In the correlation c = correlate(h, 1-h, 'full'), the element at index (n-1+k) is:
c[n-1+k] = Σ_{n_valid} h[n_valid + k] * (1-h[n_valid])

where n_valid ranges over values where both indices are valid: 0 ≤ n_valid ≤ n-1-k.
So c[n-1+k] = Σ_{n=0}^{n-1-k} h[n+k] * (1-h[n])

The element at index (n-1-k) is:
c[n-1-k] = Σ_{n_valid} h[n_valid - k] * (1-h[n_valid])

where n_valid ranges over values where both indices are valid: k ≤ n_valid ≤ n-1.
So c[n-1-k] = Σ_{n=k}^{n-1} h[n-k] * (1-h[n])

Let m = n-k: c[n-1-k] = Σ_{m=0}^{n-1-k} h[m] * (1-h[m+k])

So c[n-1+k] = Σ h[n+k]*(1-h[n]) and c[n-1-k] = Σ h[n]*(1-h[n+k])

These are different in general! c[n-1+k] counts "0→1" at shift k, while c[n-1-k] counts "1→0" at shift k.

Their sum: c[n-1+k] + c[n-1-k] = Σ_{n=0}^{n-1-k} [h[n+k]*(1-h[n]) + h[n]*(1-h[n+k])]
= Σ_{n=0}^{n-1-k} [h[n+k] + h[n] - 2*h[n]*h[n+k]]

For binary h, h[n+k]*h[n] = 1 iff both are 1. So:
= Σ [h[n+k] + h[n] - 2*I(both 1)]
= (sum of h[n+k] for n=0..n-1-k) + (sum of h[n] for n=0..n-1-k) - 2*(# of adjacent-1s at shift k)

The max of the overlap is max(c[n-1+k], c[n-1-k]) for k=0..n-1, and also c[n-1] = Σ h[n]*(1-h[n]) = Σ(1-h[n]²) for k=0.

For binary h, c[n-1] = n - (n/2) = n/2 (since h[n]² = h[n] for binary). Wait: c[n-1] = Σ h[n]*(1-h[n]) = Σ h[n] - Σ h[n]² = n/2 - n/2 = 0.

So at lag 0, the overlap is 0 for binary h. Good.

OK so the key metric is: max over k ∈ {1,...,n-1} of max(c[n-1+k], c[n-1-k]).

Now I realize that c[n-1+k] + c[n-1-k] is deterministic given h (it depends on the positions of the 1s), while the difference c[n-1+k] - c[n-1-k] depends on the "direction" of the asymmetry.

c[n-1+k] - c[n-1-k] = Σ [h[n+k]*(1-h[n]) - h[n]*(1-h[n+k])]
= Σ [h[n+k] - h[n+k]*h[n] - h[n] + h[n]*h[n+k]]
= Σ [h[n+k] - h[n]]
= (sum of h[n+k] for n=0..n-1-k) - (sum of h[n] for n=0..n-1-k)

So the difference is just the difference in the sum of h over two windows of the same length. If h is "balanced" in the sense that any window of length L has approximately L/2 ones, then this difference is small.

This gives us a useful decomposition:
max(o(k), o(-k)) = (o(k) + o(-k))/2 + |o(k) - o(-k)|/2
= [sum of h over [k, n-1] + sum of h over [0, n-1-k] - 2*autocorr(k)]/2 + |sum_diff(k)|/2

To minimize the max, we want both the autocorrelation and the window sum difference to be small.

This suggests that good sequences are those that:
1. Have low periodic autocorrelation (all shifts have ≈ (n-k)/4 ones in common)
2. Have balanced window sums (any contiguous block of length L has ≈ L/2 ones)

Condition 2 is related to "low discrepancy" sequences. Condition 1 is related to "low autocorrelation" sequences (like Barker codes, but for longer sequences).

This gives me an idea: I could use known low-discrepancy / low-autocorrelation sequences as starting points!

For example, the Thue-Morse sequence has good autocorrelation properties. Or the van der Corput sequence for low discrepancy.

import numpy as np
from scipy.optimize import minimize
import time

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlation(h))) * dx

def project_box_sum(h, target):
    n = len(h)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_D(h):
    n = len(h)
    D = np.zeros((n, 2 * n - 1))
    off = n - 1
    for L in range(n):
        idx = L + off
        D[L:, idx] = 1.0 - h[:n - L]
        D[:n - L, idx] += -h[L:]
    for m in range(1, n):
        idx = off - m
        D[m:, idx] = 1.0 - h[:n - m]
        D[:n - m, idx] += -h[m:]
    return D

def make_quasi_binary(h, n_target_ones):
    """Convert h to quasi-binary: n_target_ones ones, rest zeros, adjust one value for the fractional part."""
    n = len(h)
    h = h.copy()
    # Sort by value descending
    sorted_idx = np.argsort(-h)
    # Top n_target_ones get set to 1
    h[sorted_idx[:n_target_ones]] = 1.0
    # Next one gets the fractional remainder
    frac = float(np.sum(h)) - n_target_ones  # This should be in [0,1)
    # Actually, let me compute it properly
    # We want sum = n/2. If we set n_target_ones to 1, the sum is n_target_ones + frac_remainder
    # So frac_remainder = n/2 - n_target_ones
    target_sum = n / 2.0
    remainder = target_sum - n_target_ones
    if 0 <= remainder <= 1:
        h[sorted_idx[n_target_ones]] = remainder
    else:
        # Fallback: use project_box_sum
        h = project_box_sum(h, target_sum)
    h[sorted_idx[n_target_ones + 1:]] = 0.0
    return h

def binary_local_search(h, max_swaps=3000, time_limit=30.0):
    n = len(h)
    h = h.copy()
    c = compute_correlation(h)
    current_max = float(np.max(c))
    t_start = time.time()
    
    swaps = 0
    while swaps < max_swaps and (time.time() - t_start) < time_limit:
        D = compute_D(h)
        zeros_idx = np.where(h < 0.3)[0]
        ones_idx = np.where(h > 0.7)[0]
        
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        
        best_improvement = 0.0
        best_i, best_j = -1, -1
        
        # Precompute: for each zero position i, A_i = c + D[i,:]
        # For each one position j, the new max is max(A_i - D[j,:])
        # Try all zeros, find best (i,j) pair
        
        for i in zeros_idx:
            A = c + D[i, :]
            M = A[None, :] - D[ones_idx, :]
            new_maxes = M.max(axis=1)
            min_pos = new_maxes.argmin()
            improvement = current_max - new_maxes[min_pos]
            if improvement > best_improvement:
                best_improvement = improvement
                best_i = i
                best_j = ones_idx[min_pos]
        
        if best_improvement <= 1e-12:
            break
        
        h[best_i] = 1.0
        h[best_j] = 0.0
        c = c + D[best_i, :] - D[best_j, :]
        current_max = float(np.max(c))
        swaps += 1
    
    return h, current_max

def perturb_binary(h, n_swaps, rng):
    n = len(h)
    h = h.copy()
    zeros_idx = np.where(h < 0.3)[0].tolist()
    ones_idx = np.where(h > 0.7)[0].tolist()
    actual_swaps = min(n_swaps, min(len(zeros_idx), len(ones_idx)))
    for _ in range(actual_swaps):
        if not zeros_idx or not ones_idx:
            break
        zi = rng.randint(len(zeros_idx))
        oi = rng.randint(len(ones_idx))
        i, j = zeros_idx[zi], ones_idx[oi]
        h[i], h[j] = 1.0, 0.0
        zeros_idx.pop(zi)
        ones_idx.pop(oi)
    return h

    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def lse_obj_and_grad(h, alpha, penalty):
    n = len(h)
    obj = lse_val(compute_correlation(h), alpha)
    grad = lse_grad(h, alpha)
    s = float(np.sum(h)) - n / 2.0
    obj += penalty * s * s
    grad = grad + 2.0 * penalty * s
    return obj, grad

def optimize_continuous(h0, alpha, penalty, maxiter, n):

    try:
        res = minimize(
            lse_obj_and_grad, h0, args=(alpha, penalty),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16}
        )
        return res.x
    except Exception:
        return None
