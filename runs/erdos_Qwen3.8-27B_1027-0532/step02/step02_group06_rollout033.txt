import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, alpha):
    """Compute smooth approximation of max(corr) and its softmax weights"""
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    weight_sum = np.sum(exp_vals)
    smooth_val = m - np.log(weight_sum) / alpha
    weights = exp_vals / weight_sum
    return smooth_val, weights

def compute_gradient_smooth(h, weights, all_lags=False):
    """Compute gradient of smooth max w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    corr = compute_correlations(h)
    
    if all_lags:
        lags_to_use = np.arange(len(corr))
    else:
        # Use top lags by weight
        n_top = min(20, len(corr))
        top_idx = np.argsort(-weights)[:n_top]
        # Only include lags with significant weight
        threshold = weights[top_idx[-1]] * 0.1
        lags_to_use = top_idx[weights[top_idx] > threshold]
    
    for idx in lags_to_use:
        w = weights[idx]
        if w < 1e-10:
            continue
        lag = int(idx - (n - 1))
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂c_{lag}/∂h[j] = (1-h[j-lag]) * 1_{j>=lag} - h[j+lag] * 1_{j<=n-1-lag}
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n-la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n-la]
    
    return grad

                                momentum=0.9, n_top=30):

    h = h0.copy()
    v = np.zeros(n)  # momentum velocity
    
    best_score = evaluate_score(h)
    best_h = h.copy()
    step = 1.0 / (alpha * 4)  # initial step size
    stall = 0
    
    for it in range(max_iters):

            break
        
        corr = compute_correlations(h)
        smooth_val, weights = smooth_max(corr, alpha)
        grad = compute_gradient_smooth(h, weights, n_top=n_top)

        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_to_feasible(h_new, target_sum)
        
        # Backtracking line search
        score = evaluate_score(h_new)
        if score < best_score or score < smooth_val:
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                stall = 0
            h = h_new
        else:
            # Reduce step
            step *= 0.5
            if step < 1e-10:
                step = 1.0 / (alpha * 4)
                v *= 0.5
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            score = evaluate_score(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    # Restart with larger step
                    step = 1.0 / (alpha * 2)
                    v = -step * grad
                    h_new = h + v
                    h_new = project_to_feasible(h_new, target_sum)
                    if evaluate_score(h_new) < best_score:
                        best_score = evaluate_score(h_new)
                        best_h = h_new.copy()
                        h = h_new
                    stall = 0
                    step = 1.0 / (alpha * 4)
        
        # Adaptive step size adjustment
        if it % 100 == 0 and it > 0:
            # Slightly increase step if we're making progress
            if stall == 0:
                step = min(step * 1.05, 1.0 / alpha)
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx


def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_smooth_gradient(h, alpha, n_top=50):
    n = len(h)
    corr = compute_correlations(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    top_idx = np.argsort(-weights)[:n_top]
    for idx in top_idx:
        w = weights[idx]
        if w < 1e-8:
            continue
        lag = int(idx - (n - 1))
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            la = -lag
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]
    return grad

def slsqp_refine(h, alpha, maxiter=200):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_fun(hv):
        corr = compute_correlations(hv)
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 0)
        return m - np.log(np.sum(np.exp(shifted))) / alpha

    def constraint_fun(hv):
        return np.sum(hv) - target_sum

    def constraint_jac(hv):
        return np.ones(n)

    res = minimize(obj_fun, h, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_to_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
    return None

    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new / 2.0)
